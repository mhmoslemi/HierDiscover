    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,

"""
TriMul outgoing kernel (AlphaFold‑3) – high‑performance forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32→fp16) – custom Triton kernel.
2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   half‑precision GEMM:  [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H] (R = B·N·N).
3️⃣ Compute the output‑gate linear projection (half‑precision GEMM).
4️⃣ Apply optional mask and in‑place sigmoid on the gates, then element‑wise
   gating of the left/right projections.
5️⃣ Batched multiplicative update (the N³ TriMul) – custom Triton kernel that
   performs a batched GEMM for every (batch, head) pair without any transpose
   or extra memory copies.
6️⃣ Fuse the second LayerNorm with the output‑gate multiplication (kernel).
7️⃣ Final linear projection back to `dim` (half‑precision), cast to fp32.

All heavy per‑row normalisations are executed in custom Triton kernels,
while the huge matrix multiplications are performed either by cuBLAS (for
the linear layers) or by a specialised Triton GEMM (the TriMul step).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)  γ
    bias_ptr,            # const   * [dim]               (float32)  β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Kernel 2: LayerNorm + element‑wise gate multiplication (fp16 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,            # input & output, [total_rows, dim] (fp16)
    weight_ptr, bias_ptr,    # LayerNorm parameters γ,β (float32)
    gate_ptr,                # gate tensor, same shape as x (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance (fp32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalize, apply γ/β, then multiply by gate -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # gate

        out = norm * w[None, :] + b[None, :]   # fp32
        out = out * g                          # apply gate (fp32)

        out_fp16 = out.to(tl.float16)
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Kernel 3: Batched TriMul (outgoing) – computes
#            out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # batch‑head = B·H

    # ───── offsets for the i‑ and j‑axes ─────
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ───── decode batch and head from pid_bh ─────
    head = pid_bh % H
    batch = pid_bh // H

    # base pointers for this (batch, head)
    base_left  = left_ptr  + batch * stride_left_batch  + head * stride_left_h
    base_right = right_ptr + batch * stride_right_batch + head * stride_right_h
    base_out   = out_ptr   + batch * stride_out_batch   + head * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ───── loop over the reduction dimension k ─────
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        a_ptrs = base_left  + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        b_ptrs = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # load tiles (fp16 -> fp32 inside tl.dot)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                      # fp32 accumulation

    # write the result (fp16)
    out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [R, dim] float32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates   (4·H outputs)
    # ------------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)

    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)   # [4·H, dim]

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj = F.linear(x_norm, w_concat)      # fp16

    left_proj_raw = proj[:, 0:H]                  # [R, H]
    right_proj_raw = proj[:, H:2 * H]             # [R, H]
    left_gate_raw = proj[:, 2 * H:3 * H]          # [R, H]
    right_gate_raw = proj[:, 3 * H:4 * H]         # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate projection
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Mask (if present) + sigmoid gates
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate   # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM) – custom kernel
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] (no copy, just view)
    left_4d = left.view(B, N, N, H)
    right_4d = right.view(B, N, N, H)
    out_4d = torch.empty_like(left_4d)      # fp16 buffer for the result

    # strides for the kernel (int64)
    s_left_batch = left_4d.stride(0)
    s_left_i     = left_4d.stride(1)
    s_left_k     = left_4d.stride(2)
    s_left_h     = left_4d.stride(3)

    s_right_batch = right_4d.stride(0)
    s_right_i     = right_4d.stride(1)
    s_right_k     = right_4d.stride(2)
    s_right_h     = right_4d.stride(3)

    s_out_batch = out_4d.stride(0)
    s_out_i     = out_4d.stride(1)
    s_out_j     = out_4d.stride(2)
    s_out_h     = out_4d.stride(3)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * H,                            # one program per (batch, head)
    )
    tri_mul_outgoing_kernel[grid](
        left_4d,
        right_4d,
        out_4d,
        B, N, H,
        s_left_batch, s_left_i, s_left_k, s_left_h,
        s_right_batch, s_right_i, s_right_k, s_right_h,
        s_out_batch, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out‑gate multiplication (fused kernel)
    # ------------------------------------------------------------------
    out_flat = out_4d.view(total_rows, H)     # [R, H] fp16
    out_norm = torch.empty_like(out_flat)     # fp16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.view(total_rows, H),    # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm, w_to_out)              # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
