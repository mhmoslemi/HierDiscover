"""
Optimized outgoing Triangle‑Multiplicative‑Update (TriMul) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B,N,N,dim) → fp16.
2️⃣ Fuse the four linear projections/gates (left_proj, right_proj,
   left_gate, right_gate) into one half‑precision GEMM:
        X_norm @ W_proj_gate.T  →  [B·N·N, 4·H]
3️⃣ Compute the output‑gate with a second half‑precision GEMM.
4️⃣ Apply optional mask, sigmoid‑gating and elementwise multiplication
   to obtain gated tensors *left* and *right*.
5️⃣ N³ multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realised as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ using
   torch.matmul (cuBLAS/Tensor‑Core).
6️⃣ Second LayerNorm over the hidden dimension (fp16).
7️⃣ Apply the output‑gate, final linear projection back to ``dim`` and
   cast the result to fp32.

All LayerNorms are implemented with a custom Triton kernel; the
large matrix multiplications leverage cuBLAS Tensor‑Core kernels.
Mixed‑precision (fp16) is used for all heavy arithmetic to maximise
throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32 or float16)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]          (float32)  γ
    bias_ptr,       # const   * [dim]          (float32)  β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as fp32 if the tensor is fp16 we cast after load
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise and apply γ,β → fp16 output
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Apply row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim].

    Args
    ----
    x      : float32 or float16 tensor, shape [rows, dim]
    weight : γ,  float32 tensor of shape [dim]
    bias   : β,  float32 tensor of shape [dim]
    eps    : numerical stability constant
    block_m: rows processed per program (tunable)
    block_d: columns processed per program (tunable)

    Returns
    -------
    y : float16 tensor of shape [rows, dim] (normalized)
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# TriMul forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        eps=eps,
    )                              # → fp16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate GEMM (produces 4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection / gate weight matrices (out_features stacked)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    # Single half‑precision GEMM: X_norm @ W_proj_gate.T  → [R, 4·H]
    proj_gate = F.linear(x_norm, w_proj_gate)   # fp16

    # Split the fused output
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device),  # [H, dim]
    )                                            # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gating and element‑wise multiplication
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # Mask shape: [B, N, N] → [R, 1] broadcast over H
        mask_h = mask_tensor.to(dtype=torch.float16, device=device).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for the three gates
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)        # [B, N, N, H]
    right = right.view(B, N, N, H)       # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H) # [B, N, N, H]

    # Move hidden dimension to batch axis → (B, H, N, N), then make contiguous
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ   →   [B, H, N, N]
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))

    # Back to original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                    # fp16, shape [R, H]

    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Output‑gate & final projection back to ``dim``
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # element‑wise gate, fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device),   # [dim, H]
    )                                                                        # fp16

    # Cast to float32 for the API contract
    final = final_half.view(B, N, N, dim).to(dtype=torch.float32)

    return final
