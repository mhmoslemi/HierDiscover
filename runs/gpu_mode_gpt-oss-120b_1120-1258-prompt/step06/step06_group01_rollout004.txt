"""
optimised outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
    Implemented as a small Triton kernel (reduction in fp32,
    write‑back in fp16).

2️⃣  Fuse **five** linear layers (left/right projections, left/right
    gates, output gate) into a single GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N, H = hidden_dim)

3️⃣  Split the fused result, apply optional mask, apply sigmoid to the
    three gate tensors and produce the gated tensors `left` and `right`.

4️⃣  Reshape to [B, H, N, N] and perform the N³ multiplicative update
    with a *batched* half‑precision GEMM:
        out = left_mat @ right_matᵀ   →   [B·H, N, N]

5️⃣  Second row‑wise LayerNorm over the hidden dimension (fp16).

6️⃣  Multiply by the output‑gate, project back to `dim` with a final
    linear layer and cast the result to float‑32.

The only explicit Triton kernels are the two row‑wise LayerNorms;
all dense matrix multiplications use cuBLAS via `torch.nn.functional.linear`
and `torch.bmm`, which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# --------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    w_ptr,          # const   * [dim]           (float32) γ
    b_ptr,          # const   * [dim]           (float32) β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The reduction (mean / variance) is performed in fp32, the
    normalized result is written in fp16.
    """
    pid = tl.program_id(0)               # one program per row block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized result γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float)
        - weights_dict : dict of torch.Tensors containing all parameters
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows for the first LN
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # flatten to [R, dim] (fp32)
    x_flat = input_tensor.view(R, dim).contiguous()
    # allocate fp16 output buffer
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 32       # columns per block
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + four gates + output gate
    # ------------------------------------------------------------------
    # --------------------------------------------------------------
    # Cast and concatenate all linear weights (no bias)
    # --------------------------------------------------------------
    # shape after cat: [5*H, dim]  (fp16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (fp16)
    proj_all = F.linear(x_norm_half, w_fused)   # fp16

    # split the fused result
    left_proj_raw   = proj_all[:, 0:H]                 # [R, H]
    right_proj_raw  = proj_all[:, H:2 * H]             # [R, H]
    left_gate_raw   = proj_all[:, 2 * H:3 * H]         # [R, H]
    right_gate_raw  = proj_all[:, 3 * H:4 * H]         # [R, H]
    out_gate_raw    = proj_all[:, 4 * H:5 * H]         # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (half precision) – saves a temporary allocation
    left_gate_raw.sigmoid_()
    right_gate_raw.sigmoid_()
    out_gate_raw.sigmoid_()

    # Element‑wise gating
    left  = left_proj_raw * left_gate_raw   # [R, H] fp16
    right = right_proj_raw * right_gate_raw # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate_raw.view(B, N, N, H)   # [B, N, N, H]

    # move hidden dimension to the batch axis for a batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)   # [B·H, N, N]

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate   # [B, N, N, H] fp16

    final_half = F.linear(
        out_norm.view(R, H),               # fp16
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
