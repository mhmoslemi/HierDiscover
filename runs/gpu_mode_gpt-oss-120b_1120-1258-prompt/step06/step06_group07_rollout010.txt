"""
Highly‑optimized forward pass for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – Triton kernel.
2️⃣ Two fused linear‑+‑gate passes (half‑precision):
   * left  = (norm(x)·W_left_proj) ⊙ σ((norm(x)·W_left_gate))
   * right = (norm(x)·W_right_proj) ⊙ σ((norm(x)·W_right_gate))
   The two linear maps share the normalised tensor (concatenated weight).
3️⃣ Optional binary mask applied to *left* and *right* (broadcast on the hidden dim).
4️⃣ **Triangular multiplicative update**
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single ``torch.einsum`` on the original layout,
   therefore avoiding the costly ``permute+contiguous`` copies that a
   batched‑GEMM implementation would need.
5️⃣ Fused LayerNorm + output‑gate (FP16) – Triton kernel.
6️⃣ Final linear projection (hidden → dim) – half‑precision GEMM.
7️⃣ Cast back to ``float32`` for the returned tensor.

Only the two LayerNorm kernels are written in Triton; the heavy matrix
multiplications reuse cuBLAS/Tensor‑Core kernels.  This reduces launch
overhead and, most importantly, eliminates the large memory copy that
the naïve batched‑GEMM formulation performs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16)
    weight_ptr,          # *const float*   [dim]               (γ)
    bias_ptr,            # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_row = row < total_rows

    # ---------------------------------------------------------------
    # first pass: mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        col = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_col = col < dim
        ptr = x_ptr + row[:, None] * dim + col[None, :]
        vals = tl.load(ptr,
                       mask=mask_row[:, None] & mask_col[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # second pass: normalise, apply γ, β and store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        col = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_col = col < dim
        ptr = x_ptr + row[:, None] * dim + col[None, :]
        vals = tl.load(ptr,
                       mask=mask_row[:, None] & mask_col[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + col, mask=mask_col, other=0.0)   # γ
        b = tl.load(bias_ptr   + col, mask=mask_col, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + row[:, None] * dim + col[None, :],
                 out.to(tl.float16),
                 mask=mask_row[:, None] & mask_col[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_row = row < total_rows

    # ---------------------------------------------------------------
    # first pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        col = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_col = col < hidden
        ptr = x_ptr + row[:, None] * hidden + col[None, :]
        vals = tl.load(ptr,
                       mask=mask_row[:, None] & mask_col[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # second pass: normalise, apply γ, β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        col = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_col = col < hidden
        ptr = x_ptr + row[:, None] * hidden + col[None, :]
        vals = tl.load(ptr,
                       mask=mask_row[:, None] & mask_col[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + col, mask=mask_col, other=0.0)   # γ
        b = tl.load(bias_ptr   + col, mask=mask_col, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        gate = tl.load(gate_ptr + row[:, None] * hidden + col[None, :],
                       mask=mask_row[:, None] & mask_col[None, :],
                       other=0.0).to(tl.float32)                # FP32
        out = out * gate

        tl.store(y_ptr + row[:, None] * hidden + col[None, :],
                 out.to(tl.float16),
                 mask=mask_row[:, None] & mask_col[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool)
        - weights_dict : dictionary of model weights (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                 # C
    hidden = cfg["hidden_dim"]          # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # flattened pair‑wise rows

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()      # [R, C]  FP32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                # [R, C]  FP16

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear‑+‑gate passes (left / right)
    # ------------------------------------------------------------------
    # Cast weight matrices to half once
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ------------------------------------------------------------------
    # Left side (project + gate)
    # ------------------------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)               # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                       # (R, H)
    left = left_proj_raw * left_gate                               # (R, H)

    # ------------------------------------------------------------------
    # Right side (project + gate)
    # ------------------------------------------------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)            # (R, 2*H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                    # (R, H)
    right = right_proj_raw * right_gate                          # (R, H)

    # ------------------------------------------------------------------
    # Output gate (layer‑norm → linear → sigmoid)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)              # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                        # (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional binary mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) → (R,1) and broadcast
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update – single einsum
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)   # (B, N, N, H)    FP16
    right = right.view(B, N, N, hidden)  # (B, N, N, H)    FP16

    # einsum contracts over the shared `k` axis
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + output‑gate (half‑precision)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H)   FP16
    out_gate_flat = out_gate                                 # already (R, H) FP16
    out_norm_gate = torch.empty_like(out_flat)               # (R, H)   FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_flat,
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()                # (C, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)           # (R, C)  FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # (B, N, N, C)

    return final
