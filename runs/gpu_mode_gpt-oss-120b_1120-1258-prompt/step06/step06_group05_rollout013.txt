def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # layernorm kernels
    # allocate x_norm fp16
    x_flat = x.reshape(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # fused projection and gates
    # concat weights
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
    ], dim=0).to(torch.float16)
    proj_gate = torch.nn.functional.linear(x_norm, w_concat)  # [R,4H]

    H = hidden_dim
    left_proj_raw = proj_gate[:, :H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out gate
    out_gate_raw = torch.nn.functional.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),
    )  # [R, H]

    # mask
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # sigmoid in-place
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate  # [R, H]
    right = right_proj_raw * right_gate

    # reshape to [B,N,N,H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Permute for BMM
    left = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * H, N, N)   # [B*H,N,N]
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # second layernorm
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)  # fp16

    # apply out_gate and final linear
    out_norm = out_norm * out_gate
    final = torch.nn.functional.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),
    )  # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (B·N·N × dim) → fp16 (Triton kernel).
2️⃣ Fuse the four linear projections/gates (left/right + left/right‑gate)
   into a single half‑precision GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H].
3️⃣ Compute a separate half‑precision GEMM for the output‑gate:
   [R, dim] @ [H, dim]ᵀ → [R, H].
4️⃣ Optional masking, sigmoid the gates and apply them.
5️⃣ Tri‑multiplicative update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ.
6️⃣ Second LayerNorm over the hidden dimension → fp16 (same Triton kernel).
7️⃣ Multiply by the output‑gate, linearly project back to `dim`,
   and cast the result to float32.

All heavy reductions (LayerNorm) are written in Triton; the large matrix
multiplications use cuBLAS via `torch.nn.functional.linear` / `torch.bmm`,
which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim] (float32)
    y_ptr,          # mutable *[rows, dim] (float16)
    weight_ptr,     # const   *[dim]     (float32)  γ
    bias_ptr,       # const   *[dim]     (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    Output is fp16; all accumulations are in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1️⃣ compute mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # linear offset: row*dim + col
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2️⃣ write normalized values (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict with all linear / layer‑norm parameters
                         (stored as float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)          # if True we ignore the mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R in the description
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – Triton
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, dim]
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)                         # [R, dim]

    BLOCK_M = 1024          # rows per program – fits a whole warp block
    BLOCK_D = 64            # column tile size
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates (single GEMM)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices: [4·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # fp16

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_concat)                 # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),      # [H, dim]
    )                                                       # fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and apply gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1]
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid (half precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate       # [R, H]  fp16
    right = right_proj_raw * right_gate     # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension to the batch axis for torch.bmm
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch & head -> (B·H, N, N)
    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension – Triton
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()         # [R, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B, N, N, H] fp16

    # Final projection back to `dim`
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)   # float32

    return final
