"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm (high‑level):
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) → fp16
   – custom Triton kernel, fp32 accumulation, vector‑load width = BLOCK_D.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate)
   into a single half‑precision GEMM: Xnorm @ Wprojᵀ → [R, 4·H].
3️⃣ Compute the output‑gate linear projection separately.
4️⃣ Apply optional mask, sigmoid gates and element‑wise gating.
5️⃣ Reshape to [B·H, N, N] and run a batched half‑precision GEMM:
   out = left @ rightᵀ   (B·H independent matmuls).
6️⃣ Second LayerNorm over the hidden dimension → fp16 (same Triton kernel).
7️⃣ Multiply by the output‑gate, final half‑precision projection, and
   cast the result back to float32.

Only the two LayerNorm steps are written in Triton; all dense matrix
multiplications are performed with PyTorch / cuBLAS, which is optimal on
NVIDIA H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]    (float32)
    y_ptr,          # mutable * [total_rows, dim]    (float16)
    w_ptr,          # const   * [dim]               (float32)  γ
    b_ptr,          # const   * [dim]               (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # vector‑load width (must be power‑of‑2)
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
       μ, σ² are computed per row (fp32), y is stored as fp16.
    """
    pid = tl.program_id(0)                     # one program per row block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows                # guard for last partial block

    # ---------- 1️⃣  Compute mean & variance per row (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the column dimension in vectorized chunks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile [BLOCK_M, BLOCK_D] (float32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)                # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)         # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var + eps)

    # ---------- 2️⃣  Write normalized result (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  shape [B, N, N]     (bool/float)
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain \"dim\", \"hidden_dim\" and optional \"nomask\"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape                     # B, N, N, dim
    R = B * N * N                                        # total rows for the first LN
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()     # [R, dim], fp32
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # launch Triton kernel
    BLOCK_M = 512            # rows per program (≤ 1024 threads)
    BLOCK_D = 128            # vector width (covers dim up to 768 in ≤ 6 steps)
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers
    # ------------------------------------------------------------------
    # Concatenate proj/gate weights → [4·H, dim] (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]   (half‑precision GEMM)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # fp16

    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]           # [R, H]

    # output‑gate linear (separate weight) → [R, H]
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any), sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate   = left_gate_raw.sigmoid_()    # in‑place, fp16
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H]
    right = right_proj_raw * right_gate        # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Reshape & batched N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H]

    # Move head dimension into the batch axis for a single batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B·H, N, N]

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection & cast back to float32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise gating (fp16)

    # Final linear: [R, H] @ [dim, H]ᵀ → [R, dim]
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
