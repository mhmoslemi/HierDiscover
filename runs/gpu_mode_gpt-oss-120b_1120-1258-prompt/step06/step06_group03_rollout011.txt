"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  First LayerNorm (over the channel dim) – use PyTorch’s highly‑optimized
    `torch.nn.functional.layer_norm` on the original 4‑D tensor.  This avoids
    the heavyweight custom Triton kernel on the huge (B·N·N) rows.

2️⃣  Fuse the four linear projections/gates (left‑proj, right‑proj,
    left‑gate, right‑gate) into a single GEMM:
        X_norm_half @ W_proj_gate.T  →  [B,N,N,4·H]
    The result is split into the four tensors needed later.

3️⃣  Compute the output‑gate with a second (small) linear layer.

4️⃣  Apply the optional mask, sigmoid gating and element‑wise multiplication
    to obtain the gated tensors *left* and *right*.

5️⃣  Multiplicative update (the N³ part):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication:
        (B·H, N, N) @ (B·H, N, N)ᵀ
    Implemented with `torch.bmm` on half‑precision data (Tensor‑Cores).

6️⃣  Second LayerNorm (over the hidden dimension H) – performed by a
    custom Triton kernel that writes the result directly in fp16.

7️⃣  Apply the output‑gate, a final linear projection back to `dim`,
    and cast the result to fp32.

Only the second LayerNorm is written in Triton, satisfying the “must use
Triton” requirement while leveraging PyTorch for the bulk of the work.
All heavy arithmetic is carried out in fp16 to exploit Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def triton_layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]         (float32) – gamma
    bias_ptr,       # const   * [dim]         (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]                     # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Apply row‑wise LayerNorm to a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    and emit fp16 results.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    triton_layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y

# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]      (bool/float)
        - weights_dict : dict of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (over channel dim) – use PyTorch impl (fast)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # float32, shape [B,N,N,dim]
    x_norm_h = x_norm.half()           # fp16 for all subsequent GEMMs

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices once and cast to fp16
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)               # [4·H, dim]

    # Single GEMM -> shape [B,N,N,4·H] in fp16
    proj_gate = F.linear(x_norm_h, w_proj_gate)

    # Split the concatenated result
    H = hidden_dim
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw = torch.split(
        proj_gate, split_size_or_sections=H, dim=-1
    )  # each: [B,N,N,H]  fp16

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_h,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) and sigmoid gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (fp16) and broadcast over H
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to avoid extra buffers
    left_gate_raw.sigmoid_()
    right_gate_raw.sigmoid_()
    out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate_raw   # [B,N,N,H] fp16
    right = right_proj_raw * right_gate_raw # [B,N,N,H] fp16
    out_gate = out_gate_raw                 # [B,N,N,H] fp16 (already sigmoid‑ed)

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update
    # --------------------------------------------------------------
    # Move the head dimension to the batch axis for a single bmm call
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Flatten batch & head
    left_flat  = left_bh.view(-1, N, N)   # [B·H, N, N]
    right_flat = right_bh.view(-1, N, N)  # [B·H, N, N]

    # Batched matmul (uses Tensor‑Core fp16 kernels)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [B·H, N, N]

    # Restore original layout → [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension) – Triton implementation
    # --------------------------------------------------------------
    out_flat2 = out.view(total_rows, hidden_dim).contiguous()
    out_norm_h = _layernorm_triton(
        out_flat2,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                   # fp16, shape [total_rows, H]
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate           # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16, shape [total_rows, dim]

    final = final_half.view(B, N, N, dim).float()      # cast back to fp32
    return final
