"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – Triton kernel.
2️⃣ Fused linear + gate for the left and right branches (FP16 GEMM + sigmoid).
3️⃣ Output‑gate (linear + sigmoid) – FP16.
4️⃣ Optional binary mask (broadcasted on the hidden dimension).
5️⃣ Triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a single batched half‑precision GEMM (B*H, N, N) which
   lets cuBLAS/Tensor‑Cores handle the heavy O(N³) work.
6️⃣ Fused LayerNorm + output‑gate (FP16) – Triton kernel.
7️⃣ Final linear projection back to the original channel dimension (FP16 GEMM).
8️⃣ Cast the result to float32 (the model expects FP32 outputs).

Only the two LayerNorms are custom Triton kernels; all dense linear layers
and the O(N³) multiplication use highly‑optimised FP16 cuBLAS kernels.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [rows, dim]   (FP16)
    weight_ptr,          # *const float*   [dim]       (γ)
    bias_ptr,            # *const float*   [dim]       (β)
    rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------------------------------------------------------------
    # 1️⃣ Compute mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+ε)

    # ---------------------------------------------------------------
    # 2️⃣ Normalize, apply γ & β, store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                    # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM) and cast to FP32
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()          # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)      # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
