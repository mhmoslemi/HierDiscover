"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Fast LayerNorm on the input tensor using the built‑in torch API
    (float32 reduction → fp16 output).

2️⃣  Fuse the four linear projections (left/right + left/right‑gate) into a
    single half‑precision GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N).

3️⃣  Compute the output‑gate (also a GEMM) and apply the optional mask.

4️⃣  Apply sigmoid gates (half‑precision) and obtain the gated tensors
    `left` and `right`.

5️⃣  Batched half‑precision GEMM implements the N³ multiplicative update:
        out = left @ rightᵀ   →   [B·H, N, N].

6️⃣  **Fused kernel** – performs the second LayerNorm over the hidden
    dimension, multiplies by the pre‑computed output‑gate and projects
    back to `dim` in a single Triton kernel:
        out_norm = LN(out)                     (row‑wise over H)
        out      = (out_norm * out_gate) @ W   (W = to_out.weight)

   This kernel eliminates the intermediate write/read of the normalized
   tensor, cutting the memory traffic of the second LN by ~500 MiB.

7️⃣  Reshape to `[B, N, N, dim]` and return the result in `float32`.

The implementation keeps a Triton kernel (step 6) as required and
leverages cuBLAS (via `torch.nn.functional.linear` / `torch.bmm`) for
the heavy matrix multiplications, which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:
#   Row‑wise LayerNorm + output‑gate multiplication + final Linear.
#   Input  : out_mat   [total_rows, H]   fp16
#            out_gate  [total_rows, H]   fp16
#   Params : ln_weight [H]               fp32   (γ)
#            ln_bias   [H]               fp32   (β)
#            proj_w    [dim, H]          fp16   (to_out.weight)
#   Output : out       [total_rows, dim] fp32
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_proj_kernel(
    out_ptr,          # *float32   [total_rows, dim]
    in_ptr,           # *float16   [total_rows, H]
    gate_ptr,         # *float16   [total_rows, H]
    ln_weight_ptr,    # *float32   [H]
    ln_bias_ptr,      # *float32   [H]
    proj_weight_ptr,  # *float16   [dim, H]
    total_rows, H, dim,
    eps,              # float32 (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim tile
    BLOCK_D: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance of the hidden dimension (row‑wise)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # load a tile of input (half → fp32)
        ptr = in_ptr + offs_m[:, None] * H + offs_h[None, :]
        x = tl.load(ptr,
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)          # sum over hidden
        sum_sq  += tl.sum(x * x, axis=1)      # sum of squares

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Produce the final output (LN → gate → linear) in one pass
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # accumulator for the output tile
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        # iterate over hidden dimension again
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
            mask_h = offs_h < H

            # ---- load input, gate and LayerNorm parameters -----------------
            ptr_in   = in_ptr   + offs_m[:, None] * H + offs_h[None, :]
            x_h = tl.load(ptr_in,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

            ptr_gate = gate_ptr + offs_m[:, None] * H + offs_h[None, :]
            g_h = tl.load(ptr_gate,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

            gamma_h = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
            beta_h  = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)  # fp32

            # ---- LayerNorm + gating ---------------------------------------
            # (x - μ) / σ   →  norm
            norm = (x_h - mean[:, None]) * rstd[:, None]                # [M, H]
            # γ·norm·gate + β·gate
            term = norm * gamma_h[None, :] * g_h + beta_h[None, :] * g_h   # [M, H]

            # ---- Load projection weight (transposed slice) ----------------
            # proj_weight shape [dim, H]; we need [H, BLOCK_D] for dot
            w_ptr = proj_weight_ptr + offs_h[:, None] + offs_d[None, :] * H
            w = tl.load(w_ptr,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                        # [H, BLOCK_D]

            # ---- Accumulate ------------------------------------------------
            acc += tl.dot(term, w)    # term[M,H] @ w[H,BLOCK_D] → [M,BLOCK_D]

        # ---- Store the finished tile ------------------------------------
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor       [B, N, N, dim]  (float32)
        mask_tensor  : torch.Tensor       [B, N, N]      (bool/float, may be ignored)
        weights      : dict of torch.Tensor (all Linear/LayerNorm parameters)
        config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------

    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                    # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch implementation – fast & fused)
    # --------------------------------------------------------------
    # flatten to [R, dim] (float32, contiguous)
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # cast to fp16 for the following GEMMs
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices → [4·H, dim] (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (fp16)
    proj_gate = F.linear(x_norm_h, w_proj_gate)   # shape [R, 4·H]

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm_h,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim] → stored as (out_features, in_features)
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Compute sigmoid gates in float32 for better accuracy, then cast back
    left_gate  = left_gate_raw.float().sigmoid().to(torch.float16)
    right_gate = right_gate_raw.float().sigmoid().to(torch.float16)
    out_gate   = out_gate_raw.float().sigmoid().to(torch.float16)

    # Element‑wise gating
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM (the N³ update)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]

    # Bring head dimension into the batch axis for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)   # [B·H, N, N]

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate + final projection (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()            # [R, H] fp16
    out_gate_flat = out_gate.view(R, H).contiguous()  # [R, H] fp16

    # Allocate the final output tensor (float32)
    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    # Kernel launch parameters
    BLOCK_M = 128            # rows processed per program
    BLOCK_H = 32             # tile over hidden dimension
    BLOCK_D = 32             # tile over output dimension (dim)

    grid = (triton.cdiv(R, BLOCK_M),)

    fused_ln_gate_proj_kernel[grid](
        final,                     # output fp32
        out_flat,                  # LN input fp16
        out_gate_flat,             # output‑gate fp16
        weights["to_out_norm.weight"],   # LN γ  (fp32)
        weights["to_out_norm.bias"],     # LN β  (fp32)
        weights["to_out.weight"].to(torch.float16),   # final linear weight [dim, H] fp16
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to the original 4‑D layout and return
    # --------------------------------------------------------------
    final = final.view(B, N, N, dim)   # [B, N, N, dim] float32
    return final
```
