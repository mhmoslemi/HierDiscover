import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     total_rows, dim,
                     eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    # accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # compute mean+variance
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized output
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        norm = (x - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]  # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor (float32) → fp16 output.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


def custom_kernel(data):
    """
    Fast forward pass for the “outgoing” TriMul (AlphaFold‑3) operator.

    Algorithm
    ---------
    1️⃣ Row‑wise LayerNorm over the input `dim` (FP32 → FP16) using a custom
       Triton kernel.
    2️⃣ Fuse the four projection/gate linear layers and the output‑gate
       linear layer into a single half‑precision GEMM:
          [R, dim] @ W_all.T → [R, 5·H]
       where R = B·N·N and H = hidden_dim.
    3️⃣ Split the result, apply sigmoid to gate vectors and (optionally) the
       pairwise mask, then compute the gated tensors `left` and `right`.
    4️⃣ Perform the N³ multiplicative update via a batched half‑precision
       matrix multiplication:
          out[b, h] = left[b, h] @ right[b, h].T
       using `torch.bmm` (cuBLAS Tensor‑Core).
    5️⃣ Second LayerNorm (over the hidden dimension) – Triton again.
    6️⃣ Apply the output‑gate, a final half‑precision linear projection back to
       `dim`, and cast the result to FP32.

    The implementation keeps the memory footprint modest (all heavy tensors in
    FP16) while still using Triton for the two LayerNorms, satisfying the
    requirement that “part of the operations” be implemented in a kernel.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N  # total rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – FP32 → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim], fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # [R, dim], fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse all linear projections / gates (5·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate weights of left/right proj, left/right gate, out_gate
    # Shapes: each (H, dim)
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    # Single GEMM
    all_out = F.linear(x_norm, w_all)  # [R, 5·H], fp16

    # Split
    left_proj_raw = all_out[:, 0:H]
    right_proj_raw = all_out[:, H:2 * H]
    left_gate_raw = all_out[:, 2 * H:3 * H]
    right_gate_raw = all_out[:, 3 * H:4 * H]
    out_gate_raw = all_out[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoid gates (inplace) and optional mask
    # --------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    if not nomask:
        # mask: [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)  # broadcast over H
        # apply mask after gating (commutes with multiplication)
        left = left_proj_raw * left_gate * mask_h
        right = right_proj_raw * right_gate * mask_h
    else:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, N, N, H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # move hidden dim to batch dimension: (B·H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (half‑precision, Tensor‑Core)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))  # [B·H, N, N]

    # back to [B, N, N, H]
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (H) – FP16 output
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )  # [R, H] in fp16
    out_norm = out_norm.view(B, N, N, H)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate  # element‑wise gating, still fp16

    # final linear back to `dim`
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )  # [R, dim], fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
fast_tri_mul_outgoing.py
-------------------------

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) optimized for
GPU.  The forward pass consists of:

1️⃣ Row‑wise LayerNorm over the input channel dimension (custom Triton kernel,
   FP32 → FP16).

2️⃣ A *single* half‑precision GEMM that fuses the four projection/gate linear
   layers (left/right projection + left/right gate) *and* the output‑gate
   linear layer into a matrix of shape (5·H, dim).  The result is split into
   the five groups required for the TriMul computation.

3️⃣ In‑place sigmoid for the gate vectors and optional mask application.
   Mask multiplication can be performed after the sigmoid because multiplication
   is commutative, saving an extra pass over the data.

4️⃣ The N³ multiplicative update is performed as a batched half‑precision
   matrix multiplication:
       out[b, h] = left[b, h] @ right[b, h].T
   using `torch.bmm`, which maps to cuBLAS Tensor‑Core kernels.

5️⃣ A second row‑wise LayerNorm over the hidden dimension (again Triton).

6️⃣ Element‑wise gating with the output‑gate, a final half‑precision linear
   projection back to the original dimension, and a cast to FP32.

Only the two LayerNorms are written in Triton; all heavy dense linear algebra
uses PyTorch’s highly‑tuned cuBLAS kernels.  This satisfies the “must use
Triton” requirement while keeping the runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]         (float32)  gamma
    bias_ptr,       # const   * [dim]         (float32)  beta
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalisation + affine transform → fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → returns an fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y

# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict with all linear / LayerNorm parameters
        - config_dict  : must contain keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of (b,i,j) rows

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse all linear projections / gates (5·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection/gate weight matrices with the output‑gate
    w_all = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    # Single half‑precision GEMM
    all_out = F.linear(x_norm, w_all)                 # [R, 5·H] FP16

    # Split the result
    left_proj_raw   = all_out[:, 0:H]
    right_proj_raw  = all_out[:, H:2 * H]
    left_gate_raw   = all_out[:, 2 * H:3 * H]
    right_gate_raw  = all_out[:, 3 * H:4 * H]
    out_gate_raw    = all_out[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoid gates (in‑place) and optional mask
    # --------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    if not nomask:
        # mask: [B, N, N] → [R, 1] (broadcast over hidden dim)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left  = left_proj_raw  * left_gate  * mask_h
        right = right_proj_raw * right_gate * mask_h
    else:
        left  = left_proj_raw  * left_gate
        right = right_proj_raw * right_gate

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # bring hidden dimension into the batch dimension => (B·H, N, N)
    left_bh  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched half‑precision matrix multiplication (Tensor‑Core)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))   # [B·H, N, N]

    # reshape to original 4‑D layout
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )  # [R, H] fp16
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # fp16
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),        # [dim, H]
    )                                                       # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
