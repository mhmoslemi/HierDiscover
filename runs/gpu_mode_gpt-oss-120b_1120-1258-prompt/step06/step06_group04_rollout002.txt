"""
custom_kernel
==============

Highly‑optimized forward pass for the “outgoing” TriMul operator
(AlphaFold‑3).  The implementation performs:

1️⃣ Row‑wise LayerNorm (FP32→FP16) with a custom Triton kernel.
2️⃣ Fused linear projections + sigmoid gates for the left / right
   branches (torch.nn.functional.linear on FP16 tensors).
3️⃣ Optional mask application (broadcast over the hidden dimension).
4️⃣ Batched matrix‑multiply (tri‑multiplicative update) using a
   custom Triton GEMM kernel – this replaces the many calls to
   `torch.bmm` and removes the extra transposes that cuBLAS must do.
5️⃣ Second LayerNorm (FP16) via the same Triton kernel.
6️⃣ Output‑gate multiplication and final projection back to the
   original `dim` (FP16 → FP32).

All heavy arithmetic is carried out in FP16 (Tensor‑Cores) while
the final result is returned as `float32`.  The only PyTorch
operations are the linear layers (highly‑optimized GEMMs) and
element‑wise functions; the two LayerNorms and the tri‑multiplicative
update are pure Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the Tri‑multiplicative update
#   C = A @ B   where A,B are (batch, M, K) and (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,          # half*  [batch, M, K]
    B_ptr,          # half*  [batch, K, N]
    C_ptr,          # half*  [batch, M, N]
    batch,          # int
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_ab,      # batch stride for A
    stride_am,      # row stride for A (M)
    stride_ak,      # col stride for A (K)
    stride_bb,      # batch stride for B
    stride_bk,      # row stride for B (K)
    stride_bn,      # col stride for B (N)
    stride_cb,      # batch stride for C
    stride_cm,      # row stride for C (M)
    stride_cn,      # col stride for C (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in float32 for better accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Compute block matrix product
        acc += tl.dot(a, b)

    # Write back result (cast to fp16)
    c_ptrs = (
        C_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be all‑ones)
        weights_dict : {
            "norm.weight":          Tensor[dim],
            "norm.bias":            Tensor[dim],
            "left_proj.weight":     Tensor[hidden_dim, dim],
            "right_proj.weight":    Tensor[hidden_dim, dim],
            "left_gate.weight":     Tensor[hidden_dim, dim],
            "right_gate.weight":    Tensor[hidden_dim, dim],
            "out_gate.weight":      Tensor[hidden_dim, dim],
            "to_out_norm.weight":   Tensor[hidden_dim],
            "to_out_norm.bias":     Tensor[hidden_dim],
            "to_out.weight":        Tensor[dim, hidden_dim],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) via Triton
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.reshape(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Block sizes – tuned for dim ∈ {128,384}
    BLOCK_M = 128
    BLOCK_D = 128
    grid_norm = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # shape back to [B, N, N, dim] (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gates (fused via torch.nn.functional.linear)
    # --------------------------------------------------------------
    rows = total_rows
    # concatenate proj & gate weights for each side → (2·hidden, dim)
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    # Apply linear maps (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat = F.linear(x_norm.view(rows, dim), left_w)   # FP16
    right_concat = F.linear(x_norm.view(rows, dim), right_w)

    # split into projection and pre‑gate streams
    left_proj = left_concat[:, :hidden_dim]          # (rows, H)
    left_gate_pre = left_concat[:, hidden_dim:]      # (rows, H)
    right_proj = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # output gate (shared across positions)
    out_gate = torch.sigmoid(
        F.linear(
            x_norm.view(rows, dim),
            weights["out_gate.weight"].to(dtype=torch.float16, device=device, non_blocking=True)
        )
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device, non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)   FP16
    right_h = right_proj * right_gate       # (rows, H)   FP16

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM via Triton)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Merge H into batch dimension → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Transpose right for GEMM: (B·H, K, N)
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()

    # Allocate output
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)

    # Strides (row‑major contiguous)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat_t.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Triton GEMM launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_gemm = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid_gemm](
        left_mat,
        right_mat_t,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (FP16) via Triton
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_norm2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)       # FP16
    gated = out_norm * out_gate                         # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(dtype=torch.float16, device=device, non_blocking=True)
    )   # (rows_h, dim) FP16

    # Cast back to float32 for the model output
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
