"""
Optimized forward pass for the **outgoing** AlphaFold‑3 TriMul operator.

Algorithm (high‑level)
----------------------
1️⃣ **Row‑wise LayerNorm** on the flattened input `[B·N·N, C]`.
   A tiny Triton kernel computes μ, σ² and writes the normalized values
   in FP16 (`x_norm_fp16`).  This replaces the PyTorch `LayerNorm`.

2️⃣ **Fused projection & gating**.
   The four linear layers (left‑proj, right‑proj, left‑gate, right‑gate)
   are merged into a single half‑precision GEMM:
        `[R, C] @ [4·H, C]ᵀ → [R, 4·H]`
   where `R = B·N·N`.  The result is split into the four tensors.
   The output‑gate is computed by a second small GEMM.

3️⃣ **Mask & sigmoid**.
   When a mask is supplied it is applied to the left / right projections.
   The three gate tensors are turned into sigmoids *in‑place* and
   multiplied into the corresponding projections.

4️⃣ **Triangular multiplicative update**.
   After reshaping to `[B, H, N, N]` the update is a batched GEMM
        `(B·H, N, N) @ (B·H, N, N)ᵀ`
   performed with `torch.bmm` (cuBLAS Tensor‑Cores on H100).

5️⃣ **Fused LayerNorm + output‑gate**.
   The second LayerNorm (`to_out_norm`) and the element‑wise multiplication
   with the output‑gate are merged into a single Triton kernel
   (`ln_gate_fuse_kernel`).  It emits a tensor `z` of shape
   `[B·N·N, H]` that already contains the normalized *and* gated values.

6️⃣ **Final linear projection**.
   A single half‑precision GEMM maps `z` from the hidden dimension `H`
   back to the model dimension `C`:
        `[R, H] @ [C, H]ᵀ → [R, C]`
   The result is reshaped to `[B, N, N, C]` and cast to `float32`.

Only the two lightweight Triton kernels are required; all large matrix
multiplications use cuBLAS for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)   γ
    bias_ptr,         # const   * [dim]               (float32)   β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+ε)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + output‑gate (produces z = LN(out) * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_fuse_kernel(
    x_ptr,           # const   * [total_rows, H]   (float16)   – out after TriMul
    gate_ptr,        # const   * [total_rows, H]   (float16)   – out_gate
    gamma_ptr,       # const   * [H]               (float32)   – to_out_norm.weight
    beta_ptr,        # const   * [H]               (float32)   – to_out_norm.bias
    y_ptr,           # mutable * [total_rows, H]   (float16)   – normalized * gate
    total_rows,      # int (runtime)
    H,               # int (runtime)
    eps,             # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration (H‑block)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute mean & variance over the hidden dimension H
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptrs = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply LN affine (γ,β), then multiply by out_gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # load raw value and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * H + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # LN affine parameters
        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        norm = ((x_vals - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated = norm * g_vals                                          # (row, H‐block)

        tl.store(y_ptr + offs_m[:, None] * H + offs_d[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : all Linear / LayerNorm parameters
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, C]  (fp32)
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # launch Triton kernel for the first LN
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices needed for the fused GEMM
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],    # [H, C]
            weights["right_proj.weight"],   # [H, C]
            weights["left_gate.weight"],    # [H, C]
            weights["right_gate.weight"],   # [H, C]
        ],
        dim=0,
    ).to(torch.float16)                       # [4·H, C]

    # Half‑precision fused GEMM: [R, C] @ [4·H, C]ᵀ → [R, 4·H]
    fused = F.linear(x_norm_fp16, w_4h)       # fp16 result

    # split fused result
    left_raw        = fused[:, 0:H]                 # [R, H]
    right_raw       = fused[:, H:2*H]                # [R, H]
    left_gate_raw   = fused[:, 2*H:3*H]              # [R, H]
    right_gate_raw  = fused[:, 3*H:4*H]              # [R, H]

    # output gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, C]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, and apply them
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten to [R, 1] (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    # inplace sigmoids
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring the hidden dimension to the batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + output‑gate (produces z)
    # --------------------------------------------------------------
    # flatten to [R, H]  (R = B·N·N)
    out_flat = out_mat.view(R, H).contiguous()
    out_gate_flat = out_gate.view(R, H).contiguous()

    # allocate buffer for the normalized‑and‑gated tensor
    z_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    # launch fused LN + gate kernel
    BLOCK_M2 = 256
    BLOCK_D2 = 32                       # we tile over the hidden dim H
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    ln_gate_fuse_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        z_fp16,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to model dimension
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        z_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim]  fp16

    # reshape to output shape and cast to float32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
