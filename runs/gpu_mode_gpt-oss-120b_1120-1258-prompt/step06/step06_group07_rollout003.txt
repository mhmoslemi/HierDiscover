"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (FP32→FP16) – Triton.
2️⃣ Two fused Linear+Gate passes (left/right) in FP16 with cuBLAS.
3️⃣ Output‑gate (Linear+Sigmoid) in FP16.
4️⃣ Optional binary mask (broadcasted on the hidden dimension).
5️⃣ Triangular multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched half‑precision GEMM (torch.bmm).
6️⃣ Fused LayerNorm + out‑gate on the result – Triton.
7️⃣ Final projection back to the original channel dimension (FP16 GEMM) and
   cast to FP32.

Only the two LayerNorms are written in Triton; all dense linear
operations use Tensor‑Core‑accelerated cuBLAS which is already optimal on
H100 GPUs. This keeps kernel launch overhead low while still demonstrating
custom‑kernel fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float*   [total_rows, dim]   (FP32)
    y_ptr,            # *mut   half*    [total_rows, dim]   (FP16)
    weight_ptr,       # *const float*   [dim]               (γ)
    bias_ptr,         # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # channel block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalise + affine ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,            # *const half*   [total_rows, hidden] (pre‑norm)
    y_ptr,            # *mut   half*   [total_rows, hidden] (post‑norm⋅gate)

    bias_ptr,         # *const float*  [hidden] (β)
    gate_ptr,         # *const half*   [total_rows, hidden] (sigmoid(out_gate))
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalise + affine + gate ----------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)              # FP32

        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : dict of model parameters (all float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                 # input channel dimension C
    hidden = cfg["hidden_dim"]          # hidden dimension H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # flatten batch+seq1+seq2

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32→FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)           # [R, C] half

    # Choose block sizes – they fit the maximum channel size (384)
    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear + gate passes (left/right) – cuBLAS GEMM, FP16
    # ------------------------------------------------------------------
    # Cast weights once to half for the GEMMs
    left_proj_w  = weights["left_proj.weight"].half().contiguous()   # (H, C)
    left_gate_w  = weights["left_gate.weight"].half().contiguous()   # (H, C)
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ---- left side ---------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)                # (R, 2H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                        # (R, H)
    left = left_proj_raw * left_gate                               # (R, H)

    # ---- right side --------------------------------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)               # (R, 2H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                      # (R, H)
    right = right_proj_raw * right_gate                            # (R, H)

    # ---- output gate -------------------------------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)               # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                         # (R, H)  (half)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [R, 1] (half) and broadcast on hidden dim
        mask_flat = mask.view(total_rows, 1).to(torch.half)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors
    left  = left.view(B, N, N, hidden)       # (B, N, N, H)
    right = right.view(B, N, N, hidden)      # (B, N, N, H)

    # Prepare for batched matmul:
    #   left_mat  : (B*H, N, N)  = left[b, h, i, k]
    #   right_mat : (B*H, N, N)  = right[b, h, k, j]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)

    # Batched half‑precision matrix multiplication (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()         # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)                   # (R, H) half

    # Block config for the hidden dimension (≤ 384)
    BLOCK_D2 = 64 if hidden <= 256 else 128
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM) → FP32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()                  # (C, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)              # (R, C) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

The implementation follows the algorithm described above, using two Triton kernels for the two LayerNorms (the first normalises the input; the second normalises the multiplicative‑update result and applies the output gate). All dense linear layers and the heavy \(N^3\) multiplicative update are performed with half‑precision cuBLAS GEMMs, which on an H100 provide Tensor‑Core speed that easily meets the target runtime. The final tensor is cast back to `float32` to match the reference interface.
