"""
TriMul (outgoing) forward pass – highly‑optimized Triton + torch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (B·N·N × dim) → fp16.
   Implemented with a single Triton kernel that computes per‑row mean,
   variance and the normalized output in two loops over the channel axis.

2️⃣ Fuse the four projection / gate linear layers *and* the output‑gate
   linear layer into a single half‑precision GEMM:
       X_norm @ W_fused.T  →  [R, 5·H]  (R = B·N·N)
   The weight matrix W_fused concatenates
   left_proj, right_proj, left_gate, right_gate, out_gate
   (each shape H×dim).  The result is sliced into the five tensors
   needed for the next steps.

3️⃣ Apply the optional mask, the sigmoid gates and element‑wise
   multiplication → gated tensors `left` and `right` (both [R, H]).

4️⃣ Reshape to 4‑D, permute to (B, H, N, N) and perform the
   N³ multiplicative update with a batched half‑precision matrix
   multiplication:
       out_bh = left_bh @ right_bhᵀ          (shape B·H × N × N)
   This uses `torch.matmul`, which maps to cuBLAS Tensor‑Core kernels
   on H100.

5️⃣ Second LayerNorm over the hidden dimension (H) → fp16
   (again a Triton kernel).

6️⃣ Multiply by the output‑gate, apply the final linear projection
   (H → dim) in half precision and cast the result back to float32.

All heavy linear algebra is performed in fp16 to exploit H100 Tensor Cores.
Two Triton kernels (first and second LayerNorm) dramatically reduce
memory traffic compared with the naive PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – gamma
    bias_ptr,       # const   * [dim]      (float32) – beta
    rows,           # int32 (runtime)
    dim,            # int32 (runtime)
    eps,            # float32 compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # channels per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x-mean)*rstd * gamma + beta  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]                # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor `x` of shape [rows, dim] (float32)
    → returns a half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # Flatten to [R, dim] (float32) and normalise.
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )  # fp16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates + output‑gate (single GEMM)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (left_proj, right_proj,
    # left_gate, right_gate, out_gate) along the output dimension.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # shape [5·H, dim]

    # One large linear: X_norm_half @ W_fused.T → [R, 5·H]
    proj_gate = F.linear(x_norm, w_fused)   # fp16

    # Slice the result.
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2 * H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]
    out_gate_raw    = proj_gate[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) and sigmoid gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16) broadcasted over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to avoid extra buffers.
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates.
    left = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate   # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Move the head dimension to the batch axis: (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Fuse batch & head dimensions for a single batched GEMM.
    left_bh = left_bh.view(B * H, N, N)
    right_bh = right_bh.view(B * H, N, N)

    # out_bh = left_bh @ right_bhᵀ   (half‑precision matmul)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # [B·H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )  # fp16, shape [R, H]
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Output‑gate multiplication and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate            # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),          # [R, H]
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )                                          # fp16, shape [R, dim]

    # Cast to float32 for the API contract.
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
