```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl,
    stride_br, stride_ir, stride_kr,
    stride_bo, stride_io, stride_ko,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [i, k]
        left_ptrs = left_ptr \
            + pid_b * stride_bl \
            + offs_i[:, None] * stride_il \
            + offs_k[None, :] * stride_kl
        left_tile = tl.load(left_ptrs,
            mask = mask_i[:, None] & mask_k[None, :],
            other=0.0)

        # right tile transposed [k, j] using strides: row=k (stride_kr), col=j (stride_ir)
        right_ptrs = right_ptr \
            + pid_b * stride_br \
            + offs_k[:, None] * stride_kr \
            + offs_j[None, :] * stride_ir
        right_tile = tl.load(right_ptrs,
            mask = mask_k[:, None] & mask_j[None, :],
            other=0.0)

        # compute dot product
        acc += tl.dot(left_tile, right_tile)  # left_tile [M,K], right_tile [K,N] => [M,N]

    # write output
    out_ptrs = out_ptr \
        + pid_b * stride_bo \
        + offs_i[:, None] * stride_io \
        + offs_j[None, :] * stride_ko

    tl.store(out_ptrs,
        acc.to(tl.float16),
        mask = mask_i[:, None] & mask_j[None, :])
```

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim) – custom Triton
    kernel that reads fp32, accumulates in fp32 and writes fp16.

2️⃣  Fuse the four linear projections (left/right + two gates) into a single
    GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N).

3️⃣  Compute the output‑gate linear layer separately.

4️⃣  Optional mask is applied *before* gating, sigmoid gates are evaluated
    in‑place (fp16) and applied to the left/right projections.

5️⃣  Reshape left/right to [B, H, N, N] (H = hidden_dim) and perform the
    expensive N³ update with a custom Triton batched‑GEMM kernel:
        out_{b·H,i,j} = Σₖ left_{b·H,i,k} * right_{b·H,j,k}
    The kernel works on a fused batch dimension (B·H) and uses fp16
    arithmetic with fp32 accumulation.

6️⃣  Second row‑wise LayerNorm over the hidden dimension (again Triton).

7️⃣  Multiply by the output‑gate, project back to `dim` with a final linear
    layer (cuBLAS), and cast the result to float32.

Only the two LayerNorms and the N³ update are written in Triton;
all dense linear layers use `torch.nn.functional.linear` which already
leverages cuBLAS/Tensor‑Cores on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim]  (float32)
    y_ptr,        # mutable * [total_rows, dim]  (float16)
    w_ptr,        # const   * [dim]              (float32)  γ
    b_ptr,        # const   * [dim]              (float32)  β
    total_rows,   # int (runtime)
    dim,          # int (runtime)
    eps,          # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized result γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched N³ update (outgoing TriMul)
#   out_{b·H,i,j} = Σₖ left_{b·H,i,k} * right_{b·H,j,k}
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # const   * [M, N, N] (fp16)
    right_ptr,         # const   * [M, N, N] (fp16)
    out_ptr,           # mutable * [M, N, N] (fp16)
    N,                 # int (runtime)
    stride_l_batch,    # int
    stride_l_i,        # int
    stride_l_k,        # int
    stride_r_batch,    # int
    stride_r_i,        # int
    stride_r_k,        # int
    stride_o_batch,    # int
    stride_o_i,        # int
    stride_o_k,        # int
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # fused batch index (B·H)
    pid_i = tl.program_id(1)          # output‑row block
    pid_j = tl.program_id(2)          # output‑col block

    # -------------------- tile coordinates --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [i, k]
        left_ptrs = left_ptr + pid_b * stride_l_batch \
                     + offs_i[:, None] * stride_l_i \
                     + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile transposed: we need tile [k, j]
        # (row = k uses stride_r_k, col = j uses stride_r_i)
        right_ptrs = right_ptr + pid_b * stride_r_batch \
                      + offs_k[:, None] * stride_r_k \
                      + offs_j[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) → (M,N)

    # -------------------- write result --------------------
    out_ptrs = out_ptr + pid_b * stride_o_batch \
                + offs_i[:, None] * stride_o_i \
                + offs_j[None, :] * stride_o_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor (float32)  [B, N, N, dim]
        - mask_tensor  : torch.Tensor (bool/float) [B, N, N] (may be ignored)
        - weights_dict : model parameters
        - config_dict  : contains "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                         # total rows for the first LN
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()      # [R, dim], fp32
    x_norm = torch.empty(R, dim,
                         dtype=torch.float16,
                         device=device)                # [R, dim], fp16

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = ((R + BLOCK_M - 1) // BLOCK_M, )
    rowwise_layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (fp32)
        weights["norm.bias"],         # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (no bias) → [4·H, dim]
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16

    # Fused GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, proj_gate_weight)  # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # fp16, shape [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R, 1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (half‑precision)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gating
    left = left_proj_raw * left_gate     # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).contiguous()   # [B, N, N, H]
    right = right.view(B, N, N, H).contiguous() # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H).contiguous()  # [B, N, N, H]

    # Move the hidden dimension to the batch axis for a batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    M = B * H
    left_flat = left_perm.view(M, N, N)   # [M, N, N]  fp16, contiguous
    right_flat = right_perm.view(M, N, N) # [M, N, N]  fp16, contiguous

    # Output buffer for the N³ update
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)   # [M, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Batched N³ update – custom Triton GEMM
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        M,                                 # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,      # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # col tiles
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat2 = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat2, dtype=torch.float16)

    grid_ln2 = ((R + BLOCK_M - 1) // BLOCK_M, )
    rowwise_layernorm_kernel[grid_ln2](
        out_flat2,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)                # fp16
    out_norm = out_norm * out_gate                       # element‑wise gating

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
