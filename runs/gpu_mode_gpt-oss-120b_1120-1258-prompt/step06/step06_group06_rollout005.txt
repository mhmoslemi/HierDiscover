"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32→float16) via a custom
   Triton kernel (two‑pass: mean/variance → normalized output).

2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   half‑precision GEMM:
   [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N, H = hidden_dim).

3️⃣ Compute the output‑gate projection (separate half‑precision GEMM).

4️⃣ Optional mask is broadcast over the hidden dimension and applied before
   gating.  Sigmoid gates are computed in‑place (fp16).

5️⃣ Batched half‑precision matrix multiplication:
   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   – uses cuBLAS Tensor‑Core GEMM.

6️⃣ Fuse the second LayerNorm (over the hidden dimension) with the
   multiplication by the output‑gate in a single Triton kernel.

7️⃣ Final linear projection back to `dim` (half‑precision), cast to float32.

All heavy per‑row normalisations are executed in custom Triton kernels,
while the large matrix multiplications rely on cuBLAS Tensor‑Core kernels.
The implementation works for `dim` ∈ {128, 384}, `hidden_dim` = 128,
batch size 1‑2 and sequence lengths up to 1024, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]          (float32) γ
    bias_ptr,       # const   * [dim]          (float32) β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)              # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (γ·(x−μ)·rstd + β) as fp16
    # ------------------------------------------------------------------
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise multiplication by a gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,            # [total_rows, dim] input/output (fp16)
    weight_ptr, bias_ptr,    # LayerNorm γ, β (float32)
    gate_ptr,                # gating tensor, same shape as x (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------- mean / variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- normalized + bias + gate ----------
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = (norm * w[None, :] + b[None, :]) * gate           # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of model weights (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    # Flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Tune block sizes: keep BLOCK_D a power‑of‑two that covers dim
    BLOCK_M = 256
    BLOCK_D = 64 if dim <= 64 else 128 if dim <= 128 else 256
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single GEMM)
    # --------------------------------------------------------------
    # Cast linear weights once to fp16
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]

    # Concatenate along output dimension → [4·H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (half‑precision)
    proj_gate = F.linear(x_norm, w_concat)        # fp16

    left_proj_raw = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)               # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1]  (broadcast over hidden dim)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate     # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] then bring H into the batch axis
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                 # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out‑gate multiplication
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    # Second LayerNorm parameters (float32)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.view(total_rows, H),    # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
