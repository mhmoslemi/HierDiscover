    x_ptr, gamma_ptr, beta_ptr,
    w_left_ptr, w_right_ptr, w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,
    bias_left_ptr, bias_right_ptr, bias_lgate_ptr, bias_rgate_ptr, bias_outgate_ptr,
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, outgate_ptr,
    total_rows, dim, H, eps,
    BLOCK_D: tl.constexpr
):
    row = tl.program_id(0)
    mask_row = row < total_rows
    # compute prefix offsets:
    x_row_ptr = x_ptr + row * dim

    # sum and sumsq
    sum_val = tl.float32(0)
    sum_sq = tl.float32(0)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_row_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulate outputs
    acc_left = tl.zeros([H], dtype=tl.float32)
    acc_right = tl.zeros([H], dtype=tl.float32)
    acc_lgate = tl.zeros([H], dtype=tl.float32)
    acc_rgate = tl.zeros([H], dtype=tl.float32)
    acc_outgate = tl.zeros([H], dtype=tl.float32)

        mask_d = offs_d < dim
        x = tl.load(x_row_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        x_norm = (x - mean) * rstd * gamma
        # weight loads
        w_left = tl.load(w_left_ptr + offs_d[:, None] * H + tl.arange(0, H)[None, :],

    acc_right += bias_right
    acc_lgate += bias_lgate
    acc_rgate += bias_rgate

    # Store results as half

@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, gamma_ptr, beta_ptr,
    w_left_ptr, w_right_ptr, w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,
    bias_left_ptr, bias_right_ptr, bias_lgate_ptr, bias_rgate_ptr, bias_outgate_ptr,
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, outgate_ptr,
    total_rows, dim, eps,
    BLOCK_D: tl.constexpr,
    H: tl.constexpr
):
    row = tl.program_id(0)
    if row >= total_rows:
        return

    x_row_ptr = x_ptr + row * dim

    # Compute mean and variance
    sum_val = tl.float32(0)
    sum_sq = tl.float32(0)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs_d < dim
        x = tl.load(x_row_ptr + offs_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulators
    acc_left = tl.zeros([H], dtype=tl.float32)
    acc_right = tl.zeros([H], dtype=tl.float32)
    acc_lgate = tl.zeros([H], dtype=tl.float32)
    acc_rgate = tl.zeros([H], dtype=tl.float32)
    acc_outgate = tl.zeros([H], dtype=tl.float32)

    # Loop over d for dot products
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs_d < dim

        x = tl.load(x_row_ptr + offs_d, mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + offs_d, mask=mask, other=0.0).to(tl.float32)

        # normalized and scaled
        x_norm = (x - mean) * rstd * gamma   # shape (BLOCK_D)

        # weight tiles
        # Define column offsets
        offs_h = tl.arange(0, H)
        # left weight
        w_left = tl.load(w_left_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask[:, None] & (offs_h[None, :] < H),
                         other=0.0).to(tl.float32)
        # similarly for others
        w_right = tl.load(w_right_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask[:, None] & (offs_h[None, :] < H),
                         other=0.0).to(tl.float32)
        w_lgate = tl.load(w_lgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask[:, None] & (offs_h[None, :] < H),
                         other=0.0).to(tl.float32)
        w_rgate = tl.load(w_rgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask[:, None] & (offs_h[None, :] < H),
                         other=0.0).to(tl.float32)
        w_outgate = tl.load(w_outgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask[:, None] & (offs_h[None, :] < H),
                         other=0.0).to(tl.float32)

        # accumulate dot product across D dimension (as sum)
        # x_norm[:, None] multiplies broadcast
        # shape (BLOCK_D, H)
        acc_left += tl.sum(x_norm[:, None] * w_left, axis=0)
        acc_right += tl.sum(x_norm[:, None] * w_right, axis=0)
        acc_lgate += tl.sum(x_norm[:, None] * w_lgate, axis=0)
        acc_rgate += tl.sum(x_norm[:, None] * w_rgate, axis=0)
        acc_outgate += tl.sum(x_norm[:, None] * w_outgate, axis=0)

    # Add bias
    bias_left = tl.load(bias_left_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)
    bias_right = tl.load(bias_right_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)
    bias_lgate = tl.load(bias_lgate_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)
    bias_rgate = tl.load(bias_rgate_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)
    bias_outgate = tl.load(bias_outgate_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)

    acc_left += bias_left
    acc_right += bias_right
    acc_lgate += bias_lgate
    acc_rgate += bias_rgate
    acc_outgate += bias_outgate

    # Store results (half)
    dst_left = left_ptr + row * H
    dst_right = right_ptr + row * H
    dst_lgate = lgate_ptr + row * H
    dst_rgate = rgate_ptr + row * H
    dst_outgate = outgate_ptr + row * H

    tl.store(dst_left + tl.arange(0, H), acc_left.to(tl.float16), mask=tl.arange(0, H) < H)
    tl.store(dst_right + tl.arange(0, H), acc_right.to(tl.float16), mask=tl.arange(0, H) < H)
    tl.store(dst_lgate + tl.arange(0, H), acc_lgate.to(tl.float16), mask=tl.arange(0, H) < H)
    tl.store(dst_rgate + tl.arange(0, H), acc_rgate.to(tl.float16), mask=tl.arange(0, H) < H)
    tl.store(dst_outgate + tl.arange(0, H), acc_outgate.to(tl.float16), mask=tl.arange(0, H) < H)

"""
TriMul (outgoing) forward pass – highly optimized Triton implementation.

Algorithm
---------
1. **Fused LayerNorm + 5 linear projections**  
   A single Triton kernel computes, for every row `x∈ℝᵈ` of the
   flattened input `[B·N·N, d]`:
     * per‑row mean/variance → LayerNorm (γ,β)
     * element‑wise scaling by γ·rsqrt(var+eps)
     * five matrix‑vector products in half‑precision:
          left_proj, right_proj, left_gate, right_gate, out_gate
     * adds the pre‑computed bias `β·Wᵀ` for each projection.
   The kernel writes the five raw tensors directly, avoiding the
   intermediate `x_norm` buffer and a second GEMM pass.

2. **Mask & gating** (pure PyTorch, fp16)
   * optional mask is applied to `left` and `right`
   * sigmoid on the three gate tensors
   * element‑wise gating: `left = left_raw * left_gate`,
                         `right = right_raw * right_gate`.

3. **Cubic multiplicative update**  
   The update `out[b,i,j,k] = Σₗ left[b,i,ℓ,k] * right[b,j,ℓ,k]`
   is a batched matrix multiplication:
   `left` and `right` are reshaped to `[B·H, N, N]`,
   then `out = torch.bmm(left, right.transpose(1,2))` (half‑precision).

4. **Second LayerNorm** (Triton) on the hidden dimension `H`.

5. **Output gate & final linear projection**  
   `out_norm = out_norm * out_gate` (fp16) → `torch.nn.functional.linear`
   with `to_out.weight` (half) → cast back to `float32`.

All heavy workloads stay in FP16, enabling full Tensor‑Core utilisation
on an H100. The only CPU‑side work is bias pre‑computation and tensor
reshaping, which is negligible compared to the GPU kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [rows, dim]       (float32)
    y_ptr,               # mutable * [rows, dim]       (float16)
    weight_ptr,          # const   * [dim]             (float32)
    bias_ptr,            # const   * [dim]             (float32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per iteration
):
    row_idx = tl.program_id(0)                         # one program = one row
    mask_row = row_idx < rows
    x_row_ptr = x_ptr + row_idx * dim

    # ----------------------------------------------------------
    # 1️⃣  mean & variance (float32)
    # ----------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_row_ptr + offs_d,
                    mask=mask_row & mask_d,
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------------------------------------------------
    # 2️⃣  Normalise, apply γ, β and write result (fp16)
    # ----------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_row_ptr + offs_d,
                    mask=mask_row & mask_d,
                    other=0.0).to(tl.float32)

        # (x‑μ) * rstd
        norm = (x - mean) * rstd

        gamma = tl.load(weight_ptr + offs_d,
                        mask=mask_d,
                        other=0.0).to(tl.float32)
        beta  = tl.load(bias_ptr   + offs_d,
                        mask=mask_d,
                        other=0.0).to(tl.float32)

        out = norm * gamma + beta            # still fp32
        tl.store(y_ptr + row_idx * dim + offs_d,
                 out.to(tl.float16),
                 mask=mask_row & mask_d)


# --------------------------------------------------------------
# Fused LayerNorm + 5 linear projections (half‑precision)
# --------------------------------------------------------------
@triton.jit
def fused_norm_proj_gate_kernel(
    # inputs
    x_ptr,                # const * [R, D]               (float32)
    gamma_ptr,            # const * [D]                 (float32)
    beta_ptr,             # const * [D]                 (float32)

    # weight matrices (already transposed, shape [D, H])
    w_left_ptr,           # const * [D, H]               (float16)
    w_right_ptr,
    w_lgate_ptr,
    w_rgate_ptr,
    w_outgate_ptr,

    # biases (β·Wᵀ) for the five projections, shape [H]
    b_left_ptr,           # const * [H]                 (float16)
    b_right_ptr,
    b_lgate_ptr,
    b_rgate_ptr,
    b_outgate_ptr,

    # outputs (raw, before sigmoid / gating)
    out_left_ptr,         # mutable * [R, H]             (float16)
    out_right_ptr,
    out_lgate_ptr,
    out_rgate_ptr,
    out_outgate_ptr,

    # scalars
    R, D, eps,
    BLOCK_D: tl.constexpr,    # tile over input dimension
    H: tl.constexpr,          # hidden dimension (compile‑time constant)
):
    row = tl.program_id(0)                    # one program = one input row
    if row >= R:
        return

    # ------------------------------------------------------------------
    # 1️⃣  mean / var over the D dimension
    # ------------------------------------------------------------------
    x_row_ptr = x_ptr + row * D
    sum_ = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < D
        x = tl.load(x_row_ptr + offs_d,
                    mask=mask_d,
                    other=0.0).to(tl.float32)
        sum_   += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_ / D
    var  = sum_sq / D - mean * mean
    inv_std = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  allocate accumulators for the five outputs
    # ------------------------------------------------------------------
    acc_left   = tl.zeros([H], dtype=tl.float32)
    acc_right  = tl.zeros([H], dtype=tl.float32)
    acc_lgate  = tl.zeros([H], dtype=tl.float32)
    acc_rgate  = tl.zeros([H], dtype=tl.float32)
    acc_outgate= tl.zeros([H], dtype=tl.float32)

    # column offsets (compile‑time constant)
    offs_h = tl.arange(0, H)

    # ------------------------------------------------------------------
    # 3️⃣  second pass: compute (x‑μ)*inv_std*γ  and mat‑vec products
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < D

        # load raw input and γ
        x = tl.load(x_row_ptr + offs_d,
                    mask=mask_d,
                    other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + offs_d,
                        mask=mask_d,
                        other=0.0).to(tl.float32)

        # normalised & γ‑scaled value (shape [BLOCK_D])
        x_norm = (x - mean) * inv_std * gamma

        # ---- weight tiles (shape [BLOCK_D, H]) ----
        w_left   = tl.load(w_left_ptr   + offs_d[:, None] * H + offs_h[None, :],
                           mask=mask_d[:, None],
                           other=0.0).to(tl.float32)
        w_right  = tl.load(w_right_ptr  + offs_d[:, None] * H + offs_h[None, :],
                           mask=mask_d[:, None],
                           other=0.0).to(tl.float32)
        w_lgate  = tl.load(w_lgate_ptr  + offs_d[:, None] * H + offs_h[None, :],
                           mask=mask_d[:, None],
                           other=0.0).to(tl.float32)
        w_rgate  = tl.load(w_rgate_ptr  + offs_d[:, None] * H + offs_h[None, :],
                           mask=mask_d[:, None],
                           other=0.0).to(tl.float32)
        w_outgate= tl.load(w_outgate_ptr+ offs_d[:, None] * H + offs_h[None, :],
                           mask=mask_d[:, None],
                           other=0.0).to(tl.float32)

        # accumulate dot products: Σ_d x_norm[d] * w[d, h]
        # (broadcast x_norm to [BLOCK_D, H] and sum over D)
        acc_left    += tl.sum(x_norm[:, None] * w_left,    axis=0)
        acc_right   += tl.sum(x_norm[:, None] * w_right,   axis=0)
        acc_lgate   += tl.sum(x_norm[:, None] * w_lgate,   axis=0)
        acc_rgate   += tl.sum(x_norm[:, None] * w_rgate,   axis=0)
        acc_outgate += tl.sum(x_norm[:, None] * w_outgate, axis=0)

    # ------------------------------------------------------------------
    # 4️⃣  add the pre‑computed biases β·Wᵀ
    # ------------------------------------------------------------------
    b_left    = tl.load(b_left_ptr   + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    b_right   = tl.load(b_right_ptr  + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    b_lgate   = tl.load(b_lgate_ptr  + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    b_rgate   = tl.load(b_rgate_ptr  + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    b_outgate = tl.load(b_outgate_ptr+ offs_h, mask=offs_h < H, other=0.0).to(tl.float32)

    acc_left    += b_left
    acc_right   += b_right
    acc_lgate   += b_lgate
    acc_rgate   += b_rgate
    acc_outgate += b_outgate

    # ------------------------------------------------------------------
    # 5️⃣  write results (fp16)
    # ------------------------------------------------------------------
    dst_left   = out_left_ptr   + row * H
    dst_right  = out_right_ptr  + row * H
    dst_lgate  = out_lgate_ptr  + row * H
    dst_rgate  = out_rgate_ptr  + row * H
    dst_outgate= out_outgate_ptr+ row * H

    tl.store(dst_left    + offs_h, acc_left.to(tl.float16),    mask=offs_h < H)
    tl.store(dst_right   + offs_h, acc_right.to(tl.float16),   mask=offs_h < H)
    tl.store(dst_lgate   + offs_h, acc_lgate.to(tl.float16),   mask=offs_h < H)
    tl.store(dst_rgate   + offs_h, acc_rgate.to(tl.float16),   mask=offs_h < H)
    tl.store(dst_outgate + offs_h, acc_outgate.to(tl.float16), mask=offs_h < H)

# Entry point

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, optional)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]               # H
    eps        = 1e-5
    device     = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                                   # total rows

    # --------------------------------------------------------------
    # 0️⃣  pre‑compute bias terms β·Wᵀ for the five linear layers
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # γ, shape [dim]
    norm_bias   = weights["norm.bias"]              # β, shape [dim]

    # transposed weight matrices (shape: [dim, hidden_dim])
    w_left_T    = weights["left_proj.weight"].to(torch.float16).t().contiguous()
    w_right_T   = weights["right_proj.weight"].to(torch.float16).t().contiguous()
    w_lgate_T   = weights["left_gate.weight"].to(torch.float16).t().contiguous()
    w_rgate_T   = weights["right_gate.weight"].to(torch.float16).t().contiguous()
    w_outgate_T = weights["out_gate.weight"].to(torch.float16).t().contiguous()

    # bias = β @ Wᵀ   (float32 → half)
    with torch.cuda.amp.autocast(enabled=False):
        b_left    = (norm_bias @ w_left_T.float()).to(torch.float16)
        b_right   = (norm_bias @ w_right_T.float()).to(torch.float16)
        b_lgate   = (norm_bias @ w_lgate_T.float()).to(torch.float16)
        b_rgate   = (norm_bias @ w_rgate_T.float()).to(torch.float16)
        b_outgate = (norm_bias @ w_outgate_T.float()).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + all five projections/gates
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()   # [R, D] float32

    left_raw     = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right_raw    = torch.empty_like(left_raw)
    left_gate_raw= torch.empty_like(left_raw)
    right_gate_raw=torch.empty_like(left_raw)
    out_gate_raw = torch.empty_like(left_raw)

    grid = (R,)
    fused_norm_proj_gate_kernel[grid](
        x_flat,
        norm_weight,
        norm_bias,
        w_left_T,
        w_right_T,
        w_lgate_T,
        w_rgate_T,
        w_outgate_T,
        b_left,
        b_right,
        b_lgate,
        b_rgate,
        b_outgate,
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        R, dim, eps,
        BLOCK_D=32,
        H=hidden_dim,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣  Optional masking (applied before gating)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.reshape(R).to(torch.float16)          # [R]
        mask_flat = mask_flat[:, None]                               # [R,1] broadcast
        left_raw  = left_raw  * mask_flat
        right_raw = right_raw * mask_flat

    # --------------------------------------------------------------
    # 3️⃣  Sigmoid gates + element‑wise gating (in‑place)
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)
    right= right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣  Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat   = left_perm.view(B * hidden_dim, N, N)          # [B·H, N, N]
    right_mat  = right_perm.view(B * hidden_dim, N, N)

    # batched mat‑mul, keep everything in fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.reshape(R, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M = 256
    BLOCK_D = 32
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],               # γ
        weights["to_out_norm.bias"],                 # β
        R,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣  Apply the output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    final = F.linear(out_norm.reshape(R, hidden_dim), to_out_w)   # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)    # cast back to fp32

    return final
