"""
TriMul (outgoing) – fast forward pass
------------------------------------
The implementation follows the reference PyTorch module but drastically
reduces memory traffic:

1️⃣ Row‑wise LayerNorm (FP32 → FP16) is performed by a small Triton kernel.
2️⃣ All five linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) are fused into a **single** GEMM
   (rows × dim) × (5·hidden_dim × dim)ᵀ.  The result is sliced into
   proj/gate tensors, the mask (if present) is applied once, and the
   sigmoid gates are computed.
3️⃣ The tri‑multiplicative update
   `out[i,j] = Σ_k left[i,k]·right[j,k]`
   is realized as a batched matrix‑multiply:
   `left` and `right` are reshaped to (B·H, N, N) and a single
   `torch.bmm` call computes all H heads simultaneously.
4️⃣ A second row‑wise LayerNorm (over the hidden dimension) is again
   done with the Triton kernel.
5️⃣ The output gate is applied, then a final FP16 linear projection
   back to the original dimension; the result is cast to FP32.

All heavy arithmetic stays in FP16, only the first/second LayerNorms
(and the final cast) use FP32 statistics.  By loading the input tensor
only **once** for the five projections we cut the memory bandwidth
requirement by ~3×, which is the main source of the speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute mean & variance (FP32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (γ·x̂ + β) in FP16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]         # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor  : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor   : torch.Tensor   shape [B, N, N] (may be omitted)
        weights_dict  : {
            "norm.weight"          : Tensor[dim],
            "norm.bias"            : Tensor[dim],
            "left_proj.weight"     : Tensor[hidden_dim, dim],
            "right_proj.weight"    : Tensor[hidden_dim, dim],
            "left_gate.weight"     : Tensor[hidden_dim, dim],
            "right_gate.weight"    : Tensor[hidden_dim, dim],
            "out_gate.weight"      : Tensor[hidden_dim, dim],
            "to_out_norm.weight"   : Tensor[hidden_dim],
            "to_out_norm.bias"     : Tensor[hidden_dim],
            "to_out.weight"        : Tensor[dim, hidden_dim],
        }
        config_dict   : {"dim": int,
                         "hidden_dim": int,
                         "nomask": bool (optional, default True)}
    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (FP32 → FP16) – Triton
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim]  FP32
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tuned block sizes – dim ≤ 384 ⇒ at most 3 tiles per row
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # x_norm_fp16 now holds the normalized tensor in FP16
    # ------------------------------------------------------------------
    # 3️⃣ fused projection + gating (single GEMM)
    # ------------------------------------------------------------------
    # Assemble the five weight matrices (ordered as described in the
    # forward pass).  All are stored as FP16 for the GEMM.
    left_proj_w = weights["left_proj.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    left_gate_w = weights["left_gate.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    right_proj_w = weights["right_proj.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    right_gate_w = weights["right_gate.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    out_gate_w = weights["out_gate.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    # (5*H, dim)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # shape (5*H, dim)

    # One linear pass (rows, dim) @ (5*H, dim).T → (rows, 5*H)
    fused_out = F.linear(x_norm_fp16.view(total_rows, dim), fused_weight)  # FP16

    # ------------------------------------------------------------------
    # 4️⃣ slice, mask (if any) and apply sigmoid gates
    # ------------------------------------------------------------------
    # Slicing indices
    s0 = 0
    s1 = H
    s2 = 2 * H
    s3 = 3 * H
    s4 = 4 * H
    s5 = 5 * H

    left_proj = fused_out[:, s0:s1]          # (R, H)
    left_gate_pre = fused_out[:, s1:s2]      # (R, H)
    right_proj = fused_out[:, s2:s3]         # (R, H)
    right_gate_pre = fused_out[:, s3:s4]     # (R, H)
    out_gate_pre = fused_out[:, s4:s5]       # (R, H)

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_fp16 = mask.view(total_rows, 1).to(
            dtype=torch.float16, device=device, non_blocking=True
        )
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Multiply projections by their gates
    left_h = left_proj * left_gate          # (R, H)
    right_h = right_proj * right_gate       # (R, H)

    # ------------------------------------------------------------------
    # 5️⃣ tri‑multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Merge H into the batch dimension → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_mat[b*H, i, j] = Σ_k left_mat[b*H, i, k] * right_mat[b*H, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)               # FP16

    # ------------------------------------------------------------------
    # 7️⃣ apply output gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)       # FP16
    gated = out_norm * out_gate                # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(
            dtype=torch.float16, device=device, non_blocking=True
        ),
    )                                          # (rows_h, dim) FP16

    # Cast back to FP32 as required by the reference implementation
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
