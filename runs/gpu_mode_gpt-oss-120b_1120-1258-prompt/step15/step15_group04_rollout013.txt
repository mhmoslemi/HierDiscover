```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const *[R, 5*H] fp16
    left_ptr,          # mutable *[B, H, N, N] fp16
    right_ptr,         # mutable *[B, H, N, N] fp16
    out_gate_ptr,      # mutable *[B, H, N, N] fp16
    mask_ptr,          # const *[R] fp16 – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flattened position
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N          # third index

    # optional mask broadcast
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]   # broadcast across hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # distance between successive H slices

    # base offset for h=0 in a contiguous [B, H, N, N] tensor
    base_offset = b * (H * N * N) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # compute offset into fused proj: each row has 5*H values, layout: [left, right, left_gate, right_gate, out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets for each hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Optimized outgoing TriMul forward pass.

    Steps:
    1️⃣ Row‑wise LayerNorm on the input (float32 → float16) using torch.nn.functional.
    2️⃣ Fuse the five linear maps into a single GEMM.
    3️⃣ Triton kernel splits the fused projection into left/right tensors,
       applies the optional pair‑wise mask and sigmoid gates.
    4️⃣ Batched matrix‑multiply (left @ rightᵀ) across the sequence dimension.
    5️⃣ Second LayerNorm (hidden‑dim) + out‑gate multiplication performed with
       torch.nn.functional for speed.
    6️⃣ Final linear projection to the model dimension.

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)
    # torch.layer_norm works on float32 directly
    x_norm_f32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_f32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*H, dim]

    proj = F.linear(x_norm, fused_w)  # [R, 5*H]  fp16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_fp16,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)  # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate
    # ------------------------------------------------------------------
    # bring hidden dim to last axis for LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out_perm.view(R, H)  # [R, H]

    ln_weight = w["to_out_norm.weight"].to(torch.float16)
    ln_bias = w["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=ln_weight,
        bias=ln_bias,
        eps=eps,
    )  # fp16

    # apply out_gate (same layout as out)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)
    out_mod = out_norm * out_gate_perm  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection to dim
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)  # [dim, H]
    final = F.linear(out_mod, w_out)  # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const *[R, 5*H] (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const *[R] (fp16) – ignored if HAS_MASK==0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flattened index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N          # original “j” index renamed to k

    # broadcast mask (if present) over the hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden slices
    # offset of the element with h==0 for each (b,i,k)
    base_offset = b * (H * N * N) + i * N + k

    # iterate over hidden blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside the fused projection:
        # [left, right, left_gate, right_gate, out_gate] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask only to left/right (as in the reference implementation)
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination pointers: stride between hidden slices = N*N
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Optimised outgoing TriMul forward pass.

    1️⃣  Row‑wise LayerNorm on the input (float32 → fp16) via torch.nn.functional.
    2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
        single GEMM.
    3️⃣  Triton kernel splits the fused projection into left/right tensors,
        applies the optional pair‑wise mask and the sigmoid gates.
    4️⃣  Batched matrix‑multiply (left @ rightᵀ) across the sequence dimension.
    5️⃣  Second LayerNorm (over the hidden dimension) + out‑gate multiplication
        performed with torch.nn.functional (highly optimised CUDA kernels).
    6️⃣  Final linear projection back to the model dimension.

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                                        # [R, dim]
    x_norm_f32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                               # still float32
    x_norm = x_norm_f32.to(torch.float16)                           # fp16 for the GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                             # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                               # [R, 5·H] (fp16)
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask (only to left/right) and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)               # flatten to [R]

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_fp16,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out.view(B, H, N, N)                       # back to (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate
    # ------------------------------------------------------------------
    # bring hidden dimension to the last axis for LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out_perm.view(R, H)                    # [R, H]

    # LayerNorm parameters for the hidden dimension (cast to fp16 for the kernel)
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                                 # fp16

    # multiply by out_gate (same layout as out)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]
    out_mod = out_norm * out_gate_perm                                       # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to model dimension
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)      # [dim, H]
    final = F.linear(out_mod, w_out)                 # [R, dim] (fp16)

    # reshape to original layout and cast to float32 as required by the spec
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
