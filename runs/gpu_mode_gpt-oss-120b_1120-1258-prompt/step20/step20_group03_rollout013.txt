"""
High‑performance forward pass of the *outgoing* TriMul operator (AlphaFold3).

Algorithm (all tensors live on the CUDA device):
1️⃣ Row‑wise LayerNorm over the feature dimension (dim) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (FP16) and keep the result in a flat matrix.
3️⃣ Split the fused matrix, apply the optional pair‑wise mask, compute sigmoid
   gates and store three tensors (left, right, out_gate) with shape
   [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) over the hidden dimension:
   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]   (FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (second LayerNorm statistics, FP32)
   – custom Triton kernel.
6️⃣ Fuse the second LayerNorm, out‑gate multiplication and final projection
   into a single Triton kernel that writes the final [B, N, N, dim] float32
   output.

All heavy per‑element work is performed in Triton; only the large batched
matrix multiply (step 4) uses cuBLAS.  The implementation works for
batch size = 1 or 2, sequence length up to 1024 and arbitrary dim/hidden_dim
(e.g. 128 / 384, hidden = 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,      # contiguous tensors
    rows, dim, eps,                  # runtime args
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate sum & sum‑of‑squares in FP32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # scan the feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized values (γ·(x‑μ)·rstd + β) as FP16
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # gamma (FP32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # beta  (FP32)

        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                # [R, 5·H]   fp16
    left_ptr,                # [B, H, N, N] fp16
    right_ptr,               # [B, H, N, N] fp16
    out_gate_ptr,            # [B, H, N, N] fp16
    mask_ptr,                # [R]        fp16 (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b,i,k)
    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    k  = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]        # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # layout inside proj: [left | right | left_gate | right_gate | out_gate]
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    # iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # offset inside the fused matrix: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32, cast back to fp16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offset for the three output tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset,
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # reduce over hidden dimension in blocks
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (pairwise positions)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # hidden‑plane 0

    # load per‑row mean / rstd (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # 1) load contracted tensor (FP16 → FP32)
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)        # [M, K]

        # 2) load out‑gate (FP16 → FP32)
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)        # [M, K]

        # 3) second LayerNorm (γ,β FP32) + gate multiplication
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
        term = norm * gate_vals                                    # [M, K]

        # 4) final projection (Wᵀ : K×dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, N_tile]

        # dot over hidden dimension (fp16·fp16 → fp32 accumulation)
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # write result [B, N, N, dim] (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    # torch.nn.functional.linear expects (N, in_features) weight matrix
    proj = F.linear(x_norm, fused_w)                             # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate (FP16)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if not nomask and mask is not None:
        # flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        has_mask = 1
    else:
        # dummy buffer – never accessed because HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and perform bmm with transposed right
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                    # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate multiplication + final projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows per program
    BLOCK_K_OUT = 64   # hidden‑dim chunk (fits H=128 nicely)
    BLOCK_N_OUT = 64   # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))

    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```
