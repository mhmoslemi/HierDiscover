"""
TriMul (outgoing) – fast forward pass
-------------------------------------

*   First LayerNorm (row‑wise) is performed by a custom Triton kernel.
*   Five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    are fused into a single GEMM (`torch.nn.functional.linear`).  The result
    is split, masked, and Sigmoid‑gated by a second Triton kernel.
*   The heavy pairwise contraction ``out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]``
    is executed as a batched FP16 GEMM (`torch.bmm`) – cuBLAS handles the
    massive N²×N work efficiently.
*   The second LayerNorm, gating by ``out_gate`` and the final projection
    to the original ``dim`` are done with vectorised PyTorch ops (mean,
    variance, broadcasted scaling and ``torch.nn.functional.linear``).  This
    leverages cuBLAS for the last GEMM and avoids extra Triton‑kernel launches.

The implementation keeps all intermediate tensors on the GPU, uses mixed
precision (FP16 for the bulk of the work, FP32 for the statistics) and
requires only the two Triton kernels shown below.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # input   [rows, dim]  (float32)
    y_ptr,                # output  [rows, dim]  (float16)
    w_ptr,                # γ       [dim]        (float32)
    b_ptr,                # β       [dim]        (float32)
    rows, dim, eps,       # runtime args
    BLOCK_M: tl.constexpr,  # rows processed per program
    BLOCK_D: tl.constexpr,  # feature‑dim chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Accumulate sum and sum‑of‑squares in FP32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values (Γ·(x‑μ)·rstd + β) as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16  fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Decode flat index r → (b,i,k) and write the three tensors
    (left, right, out_gate) after mask & sigmoid.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from the flat index
    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    k  = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # mask already applied before

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset,
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of all required weight tensors (float32)
            - config_dict  : must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # block sizes – larger for big channel dimension
    BLOCK_D = 64 if dim <= 256 else 128
    BLOCK_M = 128
    grid_ln = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused projection for left/right & gates
    # --------------------------------------------------------------
    # concatenated weight: [left_proj; right_proj; left_gate; right_gate; out_gate]  → (5·H, dim)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    # fused GEMM – torch uses cuBLAS (FP16)
    proj = F.linear(x_norm, fused_w)      # (R, 5·H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if not cfg.get("nomask", True):
        # flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Pairwise contraction – batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)                    # (B·H, N, N) fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                            # (B, H, N, N) fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate gating & final projection
    # --------------------------------------------------------------
    # convert to FP32 for statistics (more stable)
    out_f32 = out.to(torch.float32)                       # (B, H, N, N)

    # mean / variance across the hidden dimension H
    mu   = out_f32.mean(dim=1, keepdim=True)              # (B,1,N,N)
    var  = out_f32.var(dim=1, unbiased=False, keepdim=True)  # (B,1,N,N)
    rstd = torch.rsqrt(var + eps)                         # (B,1,N,N)

    # LayerNorm (γ,β) – shape (H,)
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)       # (1,H,1,1)
    ln_b = w["to_out_norm.bias"].view(1, H, 1, 1)         # (1,H,1,1)

    # normalize, apply γ/β and the output‑gate
    out_norm = (out_f32 - mu) * rstd                     # (B,H,N,N)
    out_norm = out_norm * ln_w + ln_b                     # (B,H,N,N)
    out_norm = out_norm * out_gate.to(torch.float32)      # (B,H,N,N)

    # final linear projection: (B·N·N, H) → (B·N·N, dim)
    R_total = B * N * N
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R_total, H)  # (R, H)

    # weight for final projection is (dim, H) in PyTorch – keep it FP32
    final_flat = F.linear(out_flat, w["to_out.weight"])  # (R, dim) float32

    # reshape to (B, N, N, dim)
    final = final_flat.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # cleanup (optional, let Python GC reclaim)
    # ------------------------------------------------------------------
    del left, right, out_gate, out, out_f32, out_norm, mu, var, rstd, out_flat, final_flat

    return final
