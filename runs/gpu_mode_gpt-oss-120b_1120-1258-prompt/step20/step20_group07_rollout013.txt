@triton.jit
def norm_gate_kernel(
    out_ptr,            # const *[B, H, N, N] fp16
    out_gate_ptr,       # const *[B, H, N, N] fp16
    mu_ptr,             # const *[R] fp32
    rstd_ptr,           # const *[R] fp32
    ln_w_ptr,           # const *[H] fp32
    ln_b_ptr,           # const *[H] fp32
    out_norm_gate_ptr,  # mutable *[R, H] fp16
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode (b,i,j) positions
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j

    # load mu & rstd for each flat position
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # accumulate per hidden dimension in blocks
    n_blocks_k = tl.cdiv(H, BLOCK_K)
    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        # pointers for out and out_gate
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = out_gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # layernorm: (out - mu) * rstd * gamma + beta
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        prod = norm * gate_vals
        prod_fp16 = prod.to(tl.float16)

        # store to output flattened [R, H]
        dst_ptr = out_norm_gate_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, prod_fp16, mask=mask_m[:, None] & mask_k[None, :])
```

"""
TriMul (outgoing) – Fusion‐heavy Triton forward pass
===================================================

The implementation follows the exact computation of the “outgoing” TriMul
operator from AlphaFold‑3 while moving the bulk of the element‑wise work into
custom Triton kernels.  The overall algorithm is:

1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × dim) – FP32 → FP16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single FP16 GEMM (`torch.nn.functional.linear`).
3️⃣  Split the fused projection, apply the optional pair‑wise mask,
    evaluate sigmoid gates and store the three tensors (`left`, `right`,
    `out_gate`) in the layout (B, H, N, N).
4️⃣  Batched half‑precision GEMM (`torch.bmm`) – compute
    `out = left @ rightᵀ` → shape (B, H, N, N) FP16.
5️⃣  Compute per‑position mean and rsqrt‑variance of `out` (row‑wise reduction
    over the hidden dimension H) using a Triton kernel.
6️⃣  Fuse the second LayerNorm, the out‑gate and produce the gated hidden
    representation `out_norm_gate` of shape (R, H) (R = B·N·N) via a Triton
    kernel.
7️⃣  Final projection: a cuBLAS FP16 GEMM (`torch.nn.functional.linear`)
    with `to_out.weight` (shape dim×H) → output (R, dim) FP32, reshaped to
    (B, N, N, dim).

All kernels are written for Triton 3.3.1 and target the H100 (Tensor‑Core
accelerated FP16 GEMMs).  The code works for any `dim ∈ {128,384}`,
`hidden_dim = 128`, batch size 1‑2 and sequence lengths up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const  *[rows, dim] (float32)
    y_ptr,               # mutable *[rows, dim] (float16)
    w_ptr,               # const  *[dim]      (float32) γ
    b_ptr,               # const  *[dim]      (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance in FP32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
#    (output layout B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)   fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index (b,i,k) → destination layout (B,H,N,N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask handling (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    # base offset for hidden plane 0 in the destination tensors
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    n_blocks_h = tl.cdiv(H, BLOCK_H)
    for h_off in range(0, n_blocks_h):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                          # not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of `out` (shape B·H·N·N)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # const *[B, H, N, N] (fp16)
    mu_ptr,     # mutable *[R]         (float32)
    rstd_ptr,   # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_blocks_k = tl.cdiv(H, BLOCK_K)
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)        # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and pack (R, H) for final GEMM
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,            # const *[B, H, N, N] (fp16)
    out_gate_ptr,       # const *[B, H, N, N] (fp16)
    mu_ptr,             # const *[R] (float32)
    rstd_ptr,           # const *[R] (float32)
    ln_w_ptr,           # const *[H] (float32)    γ of to_out_norm
    ln_b_ptr,           # const *[H] (float32)    β of to_out_norm
    out_norm_gate_ptr,  # mutable *[R, H] (fp16)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # Decode flat position (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of hidden‑plane 0

    # Load per‑position statistics once
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    n_blocks_k = tl.cdiv(H, BLOCK_K)
    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and out_gate slices
        ptr_out  = out_ptr      + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = out_gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

        # Normalise, apply γ/β and multiply by out_gate
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        prod = norm * gate_vals
        prod_fp16 = prod.to(tl.float16)

        # Store packed (R, H) tensor
        dst_ptr = out_norm_gate_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, prod_fp16,
                 mask=mask_m[:, None] & mask_k[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with Linear / LayerNorm parameters (float32)
                             Linear layers have no bias.
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32 if dim <= 128 else 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],    # γ (float32)
        w["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection of left/right and three gates
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left/right/out_gate (B,H,N,N)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ) – FP16
    # --------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N).contiguous()
    right_mat = right.reshape(B * H, N, N).contiguous()
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                    # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and pack (R, H)
    # --------------------------------------------------------------
    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_NG = 128
    BLOCK_K_NG = 32
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    norm_gate_kernel[grid_ng](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        out_norm_gate,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_K=BLOCK_K_NG,
        num_warps=4,
    )
    # we can release tensors no longer needed
    del out, out_gate, mu, rstd, left, right

    # --------------------------------------------------------------
    # 7️⃣ Final projection (FP16 GEMM → FP32 output)
    # --------------------------------------------------------------
    to_out_w = w["to_out.weight"].to(torch.float16).contiguous()   # [dim, H]
    # torch.nn.functional.linear does (input @ weight.T)
    final_fp16 = F.linear(out_norm_gate, to_out_w)                 # (R, dim) fp16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final
