"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm over the last channel (dim) – Triton kernel that directly
    writes FP16 output (avoids an extra cast).

2️⃣ Fuse *all* linear projections that depend on the normalized tensor:
      left_proj, left_gate, right_proj, right_gate, out_gate
   into a single FP16 GEMM (shape: [B·N·N, 5*hidden_dim]).
   Slicing yields the five required tensors.

3️⃣ Apply optional pair‑wise mask (broadcasted over hidden_dim) and the
   sigmoid gates (left, right, out).

4️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] · right_h[j,k]   (per hidden channel)
   Implemented as a batched FP16 matrix multiplication:
        left (B·H, N, N) @ rightᵀ (B·H, N, N) → out (B·H, N, N).

5️⃣ Second layer‑norm over the hidden dimension – Triton kernel that
   writes FP16 and *also* multiplies by the output‑gate in the same pass.

6️⃣ Final FP16 linear projection back to `dim` and cast to float32.

All heavy reductions run on Triton, all dense linear algebra runs in
FP16 Torch kernels (tensor‑core accelerated).  The implementation works
for every configuration described in the prompt (various batch sizes,
sequence lengths, channel dimensions and with/without a mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm that writes FP16 (and optionally multiplies
# by a gate).  The kernel works on a 2‑D view [rows, dim].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const *float16  [rows, dim]
    y_ptr,          #       *float16  [rows, dim]
    w_ptr,          # const *float32  [dim]   (γ)
    b_ptr,          # const *float32  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static channel size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance per row (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # linear address = row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum‑of‑sq

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values, apply γ/β and cast to FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # float32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # float32

        out_fp32 = norm * w[None, :] + b[None, :]             # float32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel that combines the second LayerNorm with the output‑gate.
# Input:    out (FP16)    – shape [rows, hidden_dim]
#           gate (FP16)   – same shape, already sigmoid‑ed
# Output:   y (FP16)      – LayerNorm(out) * gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_fp16_kernel(
    x_ptr,          # const *float16  [rows, dim]
    gate_ptr,       # const *float16  [rows, dim]   (sigmoid output_gate)
    y_ptr,          #       *float16  [rows, dim]
    w_ptr,          # const *float32  [dim]       (γ)
    b_ptr,          # const *float32  [dim]       (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static channel size (hidden_dim)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean/var per row (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Apply norm, γ/β and multiply by gate, then store as FP16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # γ/β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # float32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # float32

        out_fp32 = norm * w[None, :] + b[None, :]

        # load gate
        gate_vals = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # cast to fp32 for multiplication

        # multiply by gate
        out_fp32 = out_fp32 * gate_vals

        # store as FP16
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp32.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    nomask       = config.get("nomask", True)

    # ------------------- first LayerNorm (FP16 output) -------------------------
    x = input_tensor.contiguous()                      # [B, N, N, dim]
    B, N, _, _ = x.shape
    total_rows = B * N * N                              # rows after flattening

    x_flat    = x.view(total_rows, dim)                 # [rows, dim]  (FP32)
    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16)

    # Tunable block sizes (chosen for H100)
    BLOCK_M = 1024
    BLOCK_D = 128      # covers the most common dim (<=384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------- fused linear: left/right projections + gates + out_gate
    # Combine weight matrices (out_features, in_features)
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, dim]

    # Linear on the normalized tensor (FP16)
    combined = F.linear(x_norm_fp16, combined_w)          # [rows, 5*H]

    # Slice the fused result
    h = hidden_dim
    offset = 0
    left_proj_h      = combined[:, offset: offset + h];            offset += h
    left_gate_pre_h  = combined[:, offset: offset + h];            offset += h
    right_proj_h     = combined[:, offset: offset + h];            offset += h
    right_gate_pre_h = combined[:, offset: offset + h];            offset += h
    out_gate_pre_h   = combined[:, offset: offset + h];            offset += h

    # ------------------- mask (optional) --------------------------------------
    if not nomask:
        # mask shape [B,N,N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------- gates -------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)

    left_h  = left_proj_h * left_gate_h          # [rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------- triangular multiplicative update (batched GEMM) ------
    # reshape to [B, N, N, H]
    left_h  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_h = right_h.view(B, N, N, hidden_dim)

    # bring hidden dim to the second axis: [B, H, N, N] and transpose right → [B, H, N, N]
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()   # (j,k) ↔ (k,j)

    BH = B * hidden_dim                     # batch × hidden
    left_mat  = left_perm.view(BH, N, N)    # [BH, N, N]   (FP16)
    right_mat = right_perm.view(BH, N, N)   # [BH, N, N]

    # batched matrix multiplication in FP16 (Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                           # FP16

    # ------------------- second LayerNorm + output‑gate (fused) -------------
    out_rows = B * N * N
    out_flat_fp16   = out.view(out_rows, hidden_dim)          # [rows, H]
    out_gate_flat_h = out_gate_h.view(out_rows, hidden_dim)  # already FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat_fp16)

    grid2 = (triton.cdiv(out_rows, BLOCK_M),)
    layernorm_mul_fp16_kernel[grid2](
        out_flat_fp16,
        out_gate_flat_h,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm_gate = out_norm_gate_fp16.view(B, N, N, hidden_dim)  # FP16

    # ------------------- final linear projection (FP16) ----------------------
    final_h = F.linear(
        out_norm_gate.view(out_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [rows, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # cast back to FP32

    return final
```
