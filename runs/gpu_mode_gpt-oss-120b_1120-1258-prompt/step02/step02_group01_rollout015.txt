def proj_gate_kernel(
    x_ptr,  # [R, dim]
    mask_ptr,  # [R] or optional
    w_left_proj_ptr,
    w_right_proj_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    left_out_ptr,   # [R, hidden_dim]
    right_out_ptr,
    out_gate_out_ptr,
    total_rows,
    dim,
    hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask = tl.load(mask_ptr + offs_m, mask=offs_m < total_rows, other=tl.full([BLOCK_M], 1.0, dtype=tl.float16))
    # compute block on hidden dimension; we need nested loops for hidden dimension
    for h_start in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # Initialize accumulators
        acc_left = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_right = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_out_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

        for k in range(0, tl.cdiv(dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < dim

            x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_k[None, :])
            x = tl.load(x_ptrs, mask=(offs_m[:, None] < total_rows) & mask_k[None, :], other=0.0).to(tl.float32)

            # Load weight chunks for each matrix
            # each weight shape (hidden_dim, dim)
            w_left = tl.load(w_left_proj_ptr + (offs_h[:, None] * dim + offs_k[None, :]), mask=mask_h[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, N,  # batch = BH
    stride_ab, stride_ai, stride_ak,   # A: [batch, N, K] (K=N)
    stride_bb, stride_bk, stride_bj,   # B: [batch, K, N]
    stride_cb, stride_ci, stride_cj,   # C: [batch, N, N]
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks: ensure we are within bounds
    mask_m = off_m < N
    mask_n = off_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_id * stride_ab + off_m[:, None] * stride_ai + off_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N) because B is transposed compared to A
        b_ptrs = B_ptr + batch_id * stride_bb + off_k[:, None] * stride_bk + off_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write out
    c_ptrs = C_ptr + batch_id * stride_cb + off_m[:, None] * stride_ci + off_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” kernel – highly fused Triton implementation.

Algorithm
---------
1. Row‑wise LayerNorm over the last channel (`dim`) in FP32 using a tiny
   Triton kernel.  The normalized tensor is immediately cast to FP16.
2. A single custom Triton kernel fuses **five** linear projections
   (left_proj, right_proj, left_gate, right_gate, out_gate), the sigmoid
   gates and an optional binary mask.  The kernel computes
        left  = (x @ W_left)   * sigmoid(x @ W_left_gate)   * mask
        right = (x @ W_right)  * sigmoid(x @ W_right_gate)  * mask
        out_gate = sigmoid(x @ W_out_gate)
   All operations are accumulated in FP32 and stored as FP16.
3. Reshape ``left`` and ``right`` to `[B, H, N, N]` (H = hidden_dim) and
   perform a **batched** half‑precision matrix multiplication
        out = left @ rightᵀ
   via a second Triton kernel.  This replaces the expensive
   ``torch.bmm`` call with a kernel that directly works on the
   `[B·H, N, N]` batch.
4. A second LayerNorm over the hidden dimension (still FP16) using the
   built‑in ``torch.nn.functional.layer_norm`` (the cost is negligible).
5. Apply the final gate, project back to ``dim`` with a half‑precision linear
   layer, and cast the result to FP32.

All heavy work is done in mixed‑precision on the GPU; only the final
output is returned in float32.  The implementation works for all
configurations described in the test suite (different batch sizes,
sequence lengths, dimensions and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- first pass: mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 out,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused projection + gating + mask kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,               # [R, dim]   FP16
    mask_ptr,            # [R]        FP16 (1.0 or 0.0)
    w_left_proj_ptr,     # [H, dim]   FP16
    w_right_proj_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    left_out_ptr,        # [R, H]     FP16
    right_out_ptr,
    out_gate_out_ptr,
    total_rows,
    dim,
    hidden_dim,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden tiles per program
    BLOCK_K: tl.constexpr,   # dim tiles
    NUM_H_BLOCKS: tl.constexpr,
    eps: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    row_mask = offs_m < total_rows

    # load mask for the whole row block (1.0 if no masking)
    mask_val = tl.load(mask_ptr + offs_m,
                       mask=row_mask,
                       other=1.0).to(tl.float32)      # [BLOCK_M]

    # iterate over hidden‑dimension tiles
    for hb in range(NUM_H_BLOCKS):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        h_mask = offs_h < hidden_dim

        # accumulators (FP32 for accuracy)
        acc_left      = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_right     = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_right_gate= tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_out_gate  = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

        # K‑loop
        for k in range(0, tl.cdiv(dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            k_mask = offs_k < dim

            # x block  [BLOCK_M, BLOCK_K]
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_k[None, :]
            x = tl.load(x_ptrs,
                        mask=row_mask[:, None] & k_mask[None, :],
                        other=0.0).to(tl.float32)

            # weight blocks are loaded transposed (K, H) to match tl.dot
            w_lp = tl.load(w_left_proj_ptr
                           + offs_k[:, None]
                           + offs_h[None, :] * dim,
                           mask=k_mask[:, None] & h_mask[None, :],
                           other=0.0).to(tl.float32)
            w_rp = tl.load(w_right_proj_ptr
                           + offs_k[:, None]
                           + offs_h[None, :] * dim,
                           mask=k_mask[:, None] & h_mask[None, :],
                           other=0.0).to(tl.float32)
            w_lg = tl.load(w_left_gate_ptr
                           + offs_k[:, None]
                           + offs_h[None, :] * dim,
                           mask=k_mask[:, None] & h_mask[None, :],
                           other=0.0).to(tl.float32)
            w_rg = tl.load(w_right_gate_ptr
                           + offs_k[:, None]
                           + offs_h[None, :] * dim,
                           mask=k_mask[:, None] & h_mask[None, :],
                           other=0.0).to(tl.float32)
            w_og = tl.load(w_out_gate_ptr
                           + offs_k[:, None]
                           + offs_h[None, :] * dim,
                           mask=k_mask[:, None] & h_mask[None, :],
                           other=0.0).to(tl.float32)

            # dot products (M×K)·(K×H) → (M×H)
            acc_left       += tl.dot(x, w_lp)
            acc_right      += tl.dot(x, w_rp)
            acc_left_gate  += tl.dot(x, w_lg)
            acc_right_gate += tl.dot(x, w_rg)
            acc_out_gate   += tl.dot(x, w_og)

        # sigmoid gates (stable formulation)
        left_gate  = 1.0 / (1.0 + tl.exp(-acc_left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-acc_right_gate))
        out_gate   = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        # apply gates
        left  = acc_left  * left_gate
        right = acc_right * right_gate

        # apply the optional pairwise mask (broadcast over hidden dim)
        left  = left  * mask_val[:, None]
        right = right * mask_val[:, None]

        # store results (back to FP16)
        out_ptrs = left_out_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :]
        tl.store(out_ptrs,
                 left.to(tl.float16),
                 mask=row_mask[:, None] & h_mask[None, :])

        out_ptrs = right_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        tl.store(out_ptrs,
                 right.to(tl.float16),
                 mask=row_mask[:, None] & h_mask[None, :])

        out_ptrs = out_gate_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        tl.store(out_ptrs,
                 out_gate.to(tl.float16),
                 mask=row_mask[:, None] & h_mask[None, :])


# ----------------------------------------------------------------------
# 3) Batched GEMM for the triangular multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N,
    stride_aa, stride_ai, stride_ak,   # A: [batch, N, K]  (K == N)
    stride_bb, stride_bk, stride_bj,   # B: [batch, K, N]
    stride_cc, stride_ci, stride_cj,   # C: [batch, N, N]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = off_m < N
    mask_n = off_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        a_ptrs = a_ptr + batch_id * stride_aa + off_m[:, None] * stride_ai + off_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        b_ptrs = b_ptr + batch_id * stride_bb + off_k[:, None] * stride_bk + off_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + batch_id * stride_cc + off_m[:, None] * stride_ci + off_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()                # [B, N, N, dim]  FP32
    x_flat = x.view(total_rows, dim)             # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x_norm_h = x_norm_flat.to(torch.float16)          # [R, dim]

    # ------------------------------------------------------------------
    # 2) Fused projection / gating kernel
    # ------------------------------------------------------------------
    # Prepare half‑precision weight copies (single contiguous tensors)
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)

    # output buffers
    left_h_flat    = torch.empty((total_rows, hidden_dim), dtype=torch.float16, device=device)
    right_h_flat   = torch.empty_like(left_h_flat)
    out_gate_flat  = torch.empty_like(left_h_flat)

    # mask handling – broadcast a tensor of ones when no mask is required
    if nomask:
        mask_half = torch.ones((total_rows,), dtype=torch.float16, device=device)
    else:
        mask_half = mask_tensor.view(total_rows).to(torch.float16)

    # kernel launch parameters
    BLOCK_M = 64          # rows per program
    BLOCK_H = 64          # hidden tiles per program
    BLOCK_K = 32          # dim tiles
    NUM_H_BLOCKS = (hidden_dim + BLOCK_H - 1) // BLOCK_H

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    proj_gate_kernel[grid](
        x_ptr=x_norm_h,
        mask_ptr=mask_half,
        w_left_proj_ptr=w_left_proj,
        w_right_proj_ptr=w_right_proj,
        w_left_gate_ptr=w_left_gate,
        w_right_gate_ptr=w_right_gate,
        w_out_gate_ptr=w_out_gate,
        left_out_ptr=left_h_flat,
        right_out_ptr=right_h_flat,
        out_gate_out_ptr=out_gate_flat,
        total_rows=total_rows,
        dim=dim,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        NUM_H_BLOCKS=NUM_H_BLOCKS,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Reshape to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left_h     = left_h_flat.view(B, N, N, hidden_dim).contiguous()
    right_h    = right_h_flat.view(B, N, N, hidden_dim).contiguous()
    out_gate_h = out_gate_flat.view(B, N, N, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 3) Batched GEMM for the multiplicative update
    # ------------------------------------------------------------------
    # Move hidden dimension to batch axis and transpose the right operand
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_mat.view(BH, N, N)    # [BH, N, N]
    right_mat = right_mat.view(BH, N, N)
    out_mat   = torch.empty_like(left_mat)  # result buffer (FP16)

    # strides for the batched GEMM
    stride_aa = left_mat.stride(0)   # batch stride
    stride_ai = left_mat.stride(1)   # row stride (N)
    stride_ak = left_mat.stride(2)   # col stride (1)

    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(1)  # K dimension stride (N)
    stride_bj = right_mat.stride(2)  # N (col) stride (1)

    stride_cc = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    BLOCK_MM = 64
    BLOCK_NN = 64
    BLOCK_KK = 64

    grid = (BH,
            triton.cdiv(N, BLOCK_MM),
            triton.cdiv(N, BLOCK_NN))
    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        batch=BH,
        N=N,
        stride_aa=stride_aa, stride_ai=stride_ai, stride_ak=stride_ak,
        stride_bb=stride_bb, stride_bk=stride_bk, stride_bj=stride_bj,
        stride_cc=stride_cc, stride_ci=stride_ci, stride_cj=stride_cj,
        BLOCK_M=BLOCK_MM,
        BLOCK_N=BLOCK_NN,
        BLOCK_K=BLOCK_KK,
    )
    torch.cuda.synchronize()

    # Reshape back to original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Second LayerNorm (half‑precision) – torch.nn.functional
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still FP16

    # ------------------------------------------------------------------
    # 5) Final gating + projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_h

    out_norm_flat = out_norm.view(total_rows, hidden_dim)   # [R, H]

    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_flat, w_to_out)         # [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
