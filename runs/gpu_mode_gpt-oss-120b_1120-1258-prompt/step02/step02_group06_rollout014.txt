@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k, stride_h,  # left strides? actually left stride for each dim
    stride_batch_r, stride_j, stride_k_r, stride_h_r,
    stride_batch_o, stride_i_o, stride_j_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_h = tl.program_id(2)  # batch*hidden index

    # compute batch and hidden indices
    batch_idx = pid_h // H
    hidden_idx = pid_h % H

    # offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # Compute base pointers for left and right:

    # left offset: base = left_ptr + batch_idx * stride_batch + hidden_idx * stride_h
    # then plus i*stride_i + k*stride_k (where k is the inner loop)
    left_base = left_ptr + batch_idx * stride_l_batch + hidden_idx * stride_l_h

    # right offset similarly: base = right_ptr + batch_idx * stride_r_batch + hidden_idx * stride_r_h
    # plus k*stride_r_k + j*stride_r_j for right (not transposed)
    right_base = right_ptr + batch_idx * stride_r_batch + hidden_idx * stride_r_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left[i,k]
        # left_ptrs shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right[k,j] (transposed)
        right_ptrs = right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    # write output
    out_ptrs = out_ptr + batch_idx * stride_o_batch + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated cubic contraction

The implementation follows the reference PyTorch module:
1. Layer‑norm on the input channel dimension (dim).
2. Bias‑free linear projections → left, right (half‑precision).
3. Optional pair‑wise mask.
4. Three element‑wise sigmoid gates (left, right, out) applied in‑place.
5. Batched GEMM  C[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   
   performed by a custom Triton kernel (fp16 × fp16 → fp32 accumulator).
   The kernel works on the original layout [B, N, N, H] without a costly
   permute‑copy; it uses the existing strides to address the tensors.
6. Layer‑norm over the hidden dimension, out‑gate multiplication,
   and a final bias‑free projection back to the original dimension.

All heavy arithmetic (steps 2‑5) is done in FP16; the final result is
returned as FP32.  The Triton kernel uses 64×64 tiles, BLOCK_K=64,
8 warps and 3 stages for good H100 utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B : M×N×N,  M = B*hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                # left  tensor  [B, N, N, H]  – fp16
    right_ptr,               # right tensor  [B, N, N, H]  – fp16 (read transposed)
    out_ptr,                 # output tensor [B, N, N, H]  – fp32
    batch_size: tl.constexpr,
    N: tl.constexpr,
    hidden_dim: tl.constexpr,
    # strides of left  : (batch, i, k, h)
    stride_la: tl.constexpr,
    stride_li: tl.constexpr,
    stride_lk: tl.constexpr,
    stride_lh: tl.constexpr,
    # strides of right : (batch, j, k, h)
    stride_ra: tl.constexpr,
    stride_rj: tl.constexpr,
    stride_rk: tl.constexpr,
    stride_rh: tl.constexpr,
    # strides of out   : (batch, i, j, h)
    stride_oa: tl.constexpr,
    stride_oi: tl.constexpr,
    stride_oj: tl.constexpr,
    stride_oh: tl.constexpr,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    #  program identifiers
    # ---------------------------------------------------------------
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # batch * hidden index

    # ---------------------------------------------------------------
    #  decode batch and hidden indices
    # ---------------------------------------------------------------
    batch_idx = pid_hb // hidden_dim   # integer division
    hid_idx   = pid_hb % hidden_dim

    # ---------------------------------------------------------------
    #  tile offsets in the i and j dimensions
    # ---------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ---------------------------------------------------------------
    #  base pointers for the current (batch, hidden) slice
    # ---------------------------------------------------------------
    left_base  = left_ptr  + batch_idx * stride_la + hid_idx * stride_lh
    right_base = right_ptr + batch_idx * stride_ra + hid_idx * stride_rh
    out_base   = out_ptr   + batch_idx * stride_oa + hid_idx * stride_oh

    # accumulator in FP32 (numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    #  reduction over the k dimension
    # ---------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[i, k]  – shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_li + k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                 # fp16 load

        # right[k, j] = right[j, k]  – shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k[:, None] * stride_rk + j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                 # fp16 load

        # Tensor‑core accelerated dot, accumulate in FP32
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    #  write the result tile (FP32)
    # ---------------------------------------------------------------
    c_ptrs = out_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.  Heavy cubic contraction
    is performed by the Triton kernel above; all other ops are carried
    out with standard PyTorch (FP32/FP16) kernels.
    """
    input_tensor, mask, weights, config = data

    dim       = config["dim"]
    hidden    = config["hidden_dim"]
    nomask    = config.get("nomask", True)
    eps       = 1e-5

    # --------------------------- unpack weights ---------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------- 1. layer‑norm ---------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, dim]  float32

    # work in fp16 for the heavy stages
    x_h = x.half()

    # ------------------- 2. bias‑free linear projections -----------------
    left  = F.linear(x_h, left_proj_weight.half())   # [B, N, N, H] fp16
    right = F.linear(x_h, right_proj_weight.half())  # [B, N, N, H] fp16

    # ------------------- 3. optional pair‑wise mask --------------------
    if not nomask:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------- 4. gating (sigmoid) -------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_weight.half()))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.half()))  # [B,N,N,H] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))    # [B,N,N,H] fp16

    left  = left * left_gate
    right = right * right_gate

    # ------------------- 5. cubic contraction via Triton -------------
    B, N, _, _ = left.shape               # [B, N, N, H]

    # allocate output tensor (float32)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=left.device)

    # tile size – 64×64 tiles give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the three tensors (in elements, not bytes)
    stride_la, stride_li, stride_lk, stride_lh = left.stride()
    stride_ra, stride_rj, stride_rk, stride_rh = right.stride()
    stride_oa, stride_oi, stride_oj, stride_oh = out.stride()

    # launch grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_la, stride_li, stride_lk, stride_lh,
        stride_ra, stride_rj, stride_rk, stride_rh,
        stride_oa, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # H100 likes many warps per block
        num_stages=3,              # double‑buffering of loads
    )

    # ------------------- 6. hidden‑dim layer‑norm, gating -------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                   # fp32

    out = out * out_gate.to(out.dtype)   # broadcast out‑gate

    # final projection back to original dimension (float32)
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
