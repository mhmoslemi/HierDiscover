"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16 for the bulk of the work).
2️⃣  Linear projections (left/right) and three gating linear layers.
    All linear ops are performed in FP16 using the native PyTorch kernels.
3️⃣  Optional binary mask (broadcast over the hidden dimension) is applied
    in‑place on the projected tensors.
4️⃣  The N³ contraction
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    is a *batched* matrix multiplication C = A @ Bᵀ for each hidden head.
    This is executed by a custom Triton kernel that:
        • works on 2‑D tiles (BLOCK_M × BLOCK_N)
        • streams K‑tiles (BLOCK_K) through shared memory
        • uses fp16 inputs, fp32 accumulation (Tensor‑cores)
        • adapts the tile size to the sequence length for best occupancy.
5️⃣  Output LayerNorm, out‑gate (sigmoid) and final linear projection back
    to the original `dim`.  The result is returned as FP32.

The kernel is deliberately written to minimise memory traffic:
the tensors are reshaped once (no extra copies) and the launch parameters
are tuned for the H100 (8 warps, 3 staging buffers).  A small
fallback to a larger tile size is provided for very short sequences.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Batched GEMM kernel:  C[b] = A[b] @ B[b].T   (A,B: M×K,  K×N)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr,                # fp16 * (batch, M, K)
    b_ptr,                # fp16 * (batch, K, N)   ← will be read as (K,N)
    c_ptr,                # fp16 * (batch, M, N)
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # strides of A
    stride_bz, stride_bk, stride_bn,   # strides of B
    stride_cz, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A fully‑fused batched GEMM for the outgoing TriMul contraction.
    The first program‑id dimension packs (batch * hidden) and the tile‑M,
    the second program‑id dimension packs tile‑N.
    """
    # ----------------------- decode indices -----------------------
    pid0 = tl.program_id(0)          # batch*hidden * tile_m  + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # number of M‑tiles per matrix
    num_tiles_m = tl.cdiv(M, BLOCK_M)

    batch_hid = pid0 // num_tiles_m          # (batch * hidden) index
    tile_m    = pid0 %  num_tiles_m          # which M‑tile inside this matrix
    tile_n    = pid1                         # which N‑tile

    # ----------------------- tile offsets ------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointer to the start of the current tile (no K‑loop offset yet)
    a_ptrs = a_ptr + batch_hid * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_hid * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # ----------------------- accumulation -----------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------- main K‑loop -----------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # out‑of‑bounds guards
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16×fp16 → fp32 accumulator (Tensor‑cores)
        acc += tl.dot(a, b)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----------------------- write back -----------------------
    c_ptrs = c_ptr + batch_hid * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _launch_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Helper that launches `batched_gemm_outgoing_kernel` with
    tile sizes tuned to the sequence length.
    A, B : torch tensors of shape (batch_hid, N, N) and dtype fp16.
    Returns C of the same shape and dtype.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch_hid, N, K = A.shape
    assert B.shape == (batch_hid, K, N)

    # Choose tile size based on N (the “most important” dimension)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # First launch dimension encodes (batch_hid * tile_m)
    grid = (batch_hid * grid_m, grid_n)

    C = torch.empty_like(A, dtype=torch.float16)

    batched_gemm_outgoing_kernel[grid](
        a_ptr=A,
        b_ptr=B,
        c_ptr=C,
        M=N,
        N=N,
        K=N,
        stride_az=A.stride(0),
        stride_am=A.stride(1),
        stride_ak=A.stride(2),
        stride_bz=B.stride(0),
        stride_bk=B.stride(1),
        stride_bn=B.stride(2),
        stride_cz=C.stride(0),
        stride_cm=C.stride(1),
        stride_cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (bool/float) or None
        weights      : dict of model parameters (float32 tensors)
        config       : dict with keys {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 1️⃣ Unpack and move everything to the appropriate precision
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ---- LayerNorm on the input (FP32 for numerical safety) ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ---- Switch to FP16 for the heavy computation --------------------
    x_fp16 = x.to(torch.float16)

    # ---- Cache half‑precision weights (to avoid repeated casts) ------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # ---- Projections -------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left  = F.linear(x_fp16, left_proj_w)   # left projection
    right = F.linear(x_fp16, right_proj_w)  # right projection

    # ---- Optional mask ------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ---- Gating scalars (sigmoid) ------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ---- Apply element‑wise gates ------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 2️⃣ N³ contraction via the fused batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape          # B = batch size
    #   left   : [B, N, N, hidden]   →  (B*hidden, N, N)
    #   right  : [B, N, N, hidden]   →  (B*hidden, N, N)  (transposed on the 2‑nd N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM (half‑precision inputs, fp32 accumulation)
    out_mat = _launch_batched_gemm(left_mat, right_mat)   # [B*hidden, N, N] fp16

    # Reshape back to the original layout and cast to FP32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 3️⃣ Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate was computed in FP16 – promote for the multiply
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim` (bias=False by design)
    out = F.linear(out, weights["to_out.weight"])

    return out
