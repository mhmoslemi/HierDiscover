"""
TriMul “outgoing” kernel (AlphaFold3)

Forward pass:
1. Layer‑norm over the input feature dimension (dim).
2. Bias‑free linear projections → left, right (hidden_dim features).
3. Optional pair‑wise mask (broadcasted on the hidden dimension).
4. Three sigmoid gates (left, right, out) and element‑wise gating of left/right.
5. Cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM C = A @ Bᵀ where each (batch, hidden) slice
   is an independent matrix multiply.  The heavy work is done in a Triton
   kernel that uses fp16 loads, Tensor‑Core‑accelerated `tl.dot`, and fp32
   accumulation for numerical stability.
6. Layer‑norm over the hidden dimension, element‑wise multiplication with the
   output gate, and a final bias‑free linear projection back to the original
   dimension.

All heavy arithmetic (steps 2‑5) is performed in half‑precision; the final
output is returned in float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : M×N×N,  M = B * hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                      # left  matrix (M, N, N)  – fp16
    b_ptr,                      # right matrix (M, N, N)  – fp16, read transposed
    c_ptr,                      # output matrix (M, N, N) – fp32
    N,                          # sequence length (int)
    stride_am, stride_ai, stride_ak,   # strides of A: (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B: (M, j, k)  (read as B[k, j])
    stride_cm, stride_ci, stride_cj,   # strides of C: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ where each batch dimension encodes (batch, hidden)."""
    pid_m = tl.program_id(0)          # tile along i (rows)
    pid_n = tl.program_id(1)          # tile along j (cols)
    pid_b = tl.program_id(2)          # (batch * hidden) index

    # start coordinates of this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # coordinate vectors inside the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # output pointer for the tile C[i, j]
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k]  (fp16)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 load

        # B[k, j] = right[j, k]  (fp16) – we read transposed
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16 load

        # Tensor‑core dot, accumulate in fp32
        acc += tl.dot(a, b)

    # store the result (fp32)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the input feature dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # [B, N, N, dim]  float32

    # work in fp16 for the heavy ops
    x_h = x.half()

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) → left, right  (fp16)
    # ------------------------------------------------------------------
    left_proj_w_h = left_proj_weight.half()
    right_proj_w_h = right_proj_weight.half()
    left = torch.nn.functional.linear(x_h, left_proj_w_h)   # [B, N, N, H] fp16
    right = torch.nn.functional.linear(x_h, right_proj_w_h) # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate_w_h = left_gate_weight.half()
    right_gate_w_h = right_gate_weight.half()
    out_gate_w_h = out_gate_weight.half()

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w_h))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w_h))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the cubic contraction:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    (i.e. batched GEMM A @ Bᵀ for each hidden slice)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) so that every (b, h) pair becomes a
    # separate batch dimension for the kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] fp16

    B, H, N, _ = left_perm.shape
    M_total = B * H

    # Flatten the first two dimensions -> (M_total, N, N)
    left_flat = left_perm.view(M_total, N, N)
    right_flat = right_perm.view(M_total, N, N)

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # ------- choose a tile size that is a multiple of 16 (Tensor‑Core friendly)
    # For very small seq‑len we fall back to 32‑tile to keep occupancy high.
    if N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 128
    # ---------------------------------------------------------------

    # Strides for the three tensors (Python ints, not tensors)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # second dim (j)
    stride_bk = right_flat.stride(2)   # third dim (k)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)     # i
    stride_cj = out_flat.stride(2)     # j

    # Grid definition: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout: [B, N, N, H] (fp32)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, output gating, final projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    # Apply the output gate (broadcasted)
    out = out * out_gate.to(out.dtype)

    # Final linear projection (bias‑free) to original dimension
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] float32

    return output
```
