import torch.nn.functional as F

import triton.language as tl

# Triton kernel for batched gemm with optional mask
@triton.jit
def outgoing_tri_mul_kernel(
    A_ptr, B_ptr, C_ptr, mask_ptr,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    stride_mm, stride_mi, stride_mk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)  # tile row i
    pid_n = tl.program_id(2)  # tile col j

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K
        k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load A block (i, k)
        a_ptr = A_ptr + batch * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # load B block (k, j) => Bᵀ
        b_ptr = B_ptr + batch * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        if HAS_MASK:
            # mask for A (i,k)
            m_a_ptr = mask_ptr + batch * stride_mm + i[:, None] * stride_mi + k[None, :] * stride_mk
            m_a = tl.load(m_a_ptr,
                          mask=mask_i[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float16)
            # mask for B (j,k) – note the order of indices is swapped
            m_b_ptr = mask_ptr + batch * stride_mm + j[:, None] * stride_mi + k[None, :] * stride_mk
            m_b = tl.load(m_b_ptr,
                          mask=mask_j[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float16)

            a = a * m_a
            b = b * m_b

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    tl.store(c_ptr,
             acc,

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Implements the cubic contraction   out[i,j] = Σ_k left[i,k] * right[j,k]
    using a Triton kernel (batched GEMM with optional mask).
    All other operations (layer‑norms, linear projections, sigmoid gates,
    final projection) stay in PyTorch for simplicity and numerical stability.
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ----------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------------------- #
    # 2. FP16 projections + sigmoid gates
    # ----------------------------------------------------------------- #
    x_h = x.half()   # fp16

    # bias‑free projections (half)
    left = F.linear(x_h, weights["left_proj.weight"].half())     # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # gates (half)
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))  # [B,N,N,H]

    # apply gates (commutes with mask, so we can apply mask later inside kernel)
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 3. Prepare tensors for the batched GEMM kernel
    # ----------------------------------------------------------------- #
    B, N, _, _ = left.shape  # batch and seq len
    # bring hidden dimension forward
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    M_total = B * hidden_dim
    left_flat = left_perm.view(M_total, N, N)   # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N, N) # [M, N, N]  fp16
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # ----------------------------------------------------------------- #
    # 4. Optional mask – expand to (M, N, N) if needed
    # ----------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] -> expand over hidden dim
        mask_exp = mask.unsqueeze(1).expand(-1, hidden_dim, -1, -1).contiguous()
        mask_flat = mask_exp.view(M_total, N, N).half()
        has_mask = 1
    else:
        # dummy placeholder (won't be read)
        mask_flat = torch.empty((0,), dtype=torch.float16, device=left_flat.device)
        has_mask = 0

    # ----------------------------------------------------------------- #
    # 5. Launch Triton kernel
    # ----------------------------------------------------------------- #
    # Tunable tile sizes – chosen to work well on H100 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # strides for A, B, C (int64)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    if has_mask:
        stride_mm = mask_flat.stride(0)
        stride_mi = mask_flat.stride(1)
        stride_mk = mask_flat.stride(2)
    else:
        # dummy strides (won't be used)
        stride_mm = 0
        stride_mi = 0
        stride_mk = 0

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    outgoing_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        mask_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        stride_mm,
        stride_mi,
        stride_mk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=has_mask,
    )

    # ----------------------------------------------------------------- #
    # 6. Reshape back to [B, N, N, hidden_dim] (float32)
    # ----------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------- #
    # 7. Output LayerNorm, output‑gate, final projection
    # ----------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate (casting gate to float32)
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])
    return output

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) forward pass with a Triton‑accelerated cubic contraction.
-----------------------------------------------------------------------
1. Layer‑norm the input (float32).
2. Cast to fp16 and apply bias‑free linear projections for the left/right
   tensors together with three sigmoid gates.
3. (Optional) mask the projected tensors – the mask is broadcasted over the
   hidden dimension and applied inside the Triton kernel.
4. Compute the batched GEMM
        out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a single Triton kernel.  Each (batch × hidden) pair is a separate GEMM
   (A @ Bᵀ) where A and B are fp16, the dot product is accumulated in fp32
   (Tensor‑core friendly) and the result is written as fp32.
5. Reshape back to [B, N, N, H]; apply a second LayerNorm, multiply by the
   output gate and finally a bias‑free linear projection to the original
   feature dimension.
The kernel uses a configurable BLOCK_M/N/K tiling (128×128×64 by default),
works on H100 tensor‑cores and supports an optional mask without any extra
Python‑level post‑processing.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_tri_mul_kernel(
    A_ptr,               # left tensor  (M, N, N)   fp16
    B_ptr,               # right tensor (M, N, N)   fp16 (read as Bᵀ)
    C_ptr,               # output      (M, N, N)   fp32
    mask_ptr,            # mask tensor  (M, N, N)   fp16 (optional)
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bm, stride_bj, stride_bk,   # B strides: batch, j(i), k
    stride_cm, stride_ci, stride_cj,   # C strides: batch, i, j
    stride_mm, stride_mi, stride_mk,   # mask strides (same layout as A)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    pid_m   = tl.program_id(1)           # tile row (i)
    pid_n   = tl.program_id(2)           # tile col (j)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

        k = k0 + tl.arange(0, BLOCK_K)

            other=0.0,

        b = tl.load(

            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,

            m_a_ptr = mask_ptr + batch_id * stride_mm + i[:, None] * stride_mi + k[None, :] * stride_mk
            m_a = tl.load(
                m_a_ptr,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float16)

            # mask for B (j,k) – note swapped axes
            m_b_ptr = mask_ptr + batch_id * stride_mm + j[:, None] * stride_mi + k[None, :] * stride_mk
            m_b = tl.load(
                m_b_ptr,
                mask=mask_j[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float16)

            a = a * m_a
            b = b * m_b

        # dot‑product (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)
        acc += tl.dot(a, b)

    # ---------- write output ----------
    c_ptr = C_ptr + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        mask         : torch.Tensor [B, N, N]     (any dtype, optional)
        weights      : dict with all parameters required by the module
        config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # -----------------------------------------------------------------
    # 2) FP16 projections + sigmoid gates
    # -----------------------------------------------------------------
    x_h = x.half()                                 # [B,N,N,dim]  fp16

    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))   # [B,N,N,H]

    # Apply sigmoid gates (mask will be applied inside the kernel)
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3) Prepare tensors for the batched GEMM kernel
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                # batch size and sequence length
    # bring hidden dimension forward for a clean (M, N, N) view
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = B * hidden_dim                       # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)    # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N, N)   # [M, N, N]  fp16
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # -----------------------------------------------------------------
    # 4) Optional mask – expand to (M, N, N) and cast to fp16
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → expand over hidden dimension → [B, H, N, N]
        mask_exp = mask.unsqueeze(1).expand(-1, hidden_dim, -1, -1).contiguous()
        mask_flat = mask_exp.view(M_total, N, N).half()
        has_mask = 1
    else:
        # dummy placeholder – will never be dereferenced
        mask_flat = torch.empty((0,), dtype=torch.float16, device=left_flat.device)
        has_mask = 0

    # -----------------------------------------------------------------
    # 5) Launch the Triton kernel
    # -----------------------------------------------------------------
    # Tile sizes tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides (int64 scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # corresponds to the “i” axis of right → later used as j
    stride_bk = right_flat.stride(2)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    if has_mask:
        stride_mm = mask_flat.stride(0)
        stride_mi = mask_flat.stride(1)
        stride_mk = mask_flat.stride(2)
    else:
        # dummy strides – never accessed when HAS_MASK == 0
        stride_mm = 0
        stride_mi = 0
        stride_mk = 0

    grid = (
        M_total,                                 # one program per (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,            # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,            # tiles over j
    )

    outgoing_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        mask_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        stride_mm,
        stride_mi,
        stride_mk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=has_mask,
    )

    # -----------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden] (float32)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Multiply by output gate (cast gate to float32 for stability)
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output
