"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. Row‑wise LayerNorm over the last channel (dim) → directly writes
   FP16 results (no extra cast).
2. Fuse the five linear projections (left/right and three gates) into a
   single GEMM by concatenating the weight matrices.  All GEMMs run in
   FP16 (Tensor‑Core) and the result is split into the individual
   tensors.
3. (Optional) apply the binary mask to the left/right projections.
4. Apply sigmoid gates and element‑wise gating (still FP16).
5. Triangular multiplicative update:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batched matrix‑multiply per hidden dimension
   (B·H × N × N) using `torch.bmm`.
6. Fused LayerNorm + final output gate:
   - compute mean/var per row,
   - apply per‑channel weight & bias,
   - multiply by the pre‑computed `out_gate`.
   The kernel writes FP16 values ready for the final projection.
7. Final linear projection `hidden_dim → dim` (FP16 GEMM) and cast to
   FP32 for the returned tensor.

Only the two LayerNorm‑related steps are implemented in Triton; all
dense linear algebra uses cuBLAS/Tensor‑Core via `torch.nn.functional`.
"""

    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: normalize, apply γ, β and store as FP16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused LayerNorm + output‑gate (both in FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                     # *const half* [total_rows, dim] (pre‑norm)
    y_ptr,                     # *mut   half* [total_rows, dim] (post‑norm⋅gate)
    weight_ptr,                # *const float* [dim] (γ)
    bias_ptr,                  # *const float* [dim] (β)
    gate_ptr,                  # *const half*  [total_rows, dim] (out_gate)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # Compute mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalize, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        # ----- multiply by pre‑computed out_gate (FP16 → FP32) -----
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32/bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → no mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for the flattened tensors

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (produces FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)               # [R, C] FP16

    BLOCK_M = 128
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (single GEMM, FP16)
    # ------------------------------------------------------------------
    # Concatenate the five weight tensors (shape (5*hidden, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H) FP16
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden, dim=1
        )  # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional mask (apply before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).half()  # (R,1)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)      # (R, H)

    left_fp16 = left_proj_fp16 * left_gate       # (R, H)
    right_fp16 = right_proj_fp16 * right_gate    # (R, H)

    # ------------------------------------------------------------------
    # 3) Reshape for triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)       # (B,N,N,H)
    right = right_fp16.view(B, N, N, hidden)    # (B,N,N,H)

    # Batched GEMM: out_i_j = Σ_k left_i_k * right_j_k
    # → left as (B,H,N,N), right as (B,H,N,N) with right transposed on k
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat)    # (B*H,N,N) FP16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Second LayerNorm + final output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # gate already sigmoid'ed, FP16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()               # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)       # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)        # cast to FP32

    return final
