```
@triton.jit
def layernorm_kernel_fp16(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load half and convert to fp32
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w + b                          # fp32
        out_half = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half, mask=mask_m[:, None] & mask_d[None, :])
```

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(from AlphaFold‑3).  The algorithm follows the reference PyTorch code but
fuses the two LayerNorms into custom Triton kernels that work in
float16 (the heavy reductions use float32 accumulation for numerical
stability).  All large linear layers are executed in fp16 using Tensor‑Core
GEMMs, and the expensive N³ multiplicative update is performed with a
single batched GEMM (torch.bmm) on tensors of shape (B·H, N, N).

Inputs:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

Outputs:
    Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm: input & output are half‑precision.
# The mean/variance are accumulated in float32.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const half*   [total_rows, dim]
    y_ptr,          # *mut   half*   [total_rows, dim]
    weight_ptr,     # *const float*  [dim]
    bias_ptr,       # *const float*  [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,                # int (compile‑time)
    eps: tl.constexpr,                # float (compile‑time)
    BLOCK_M: tl.constexpr,            # rows per program
    BLOCK_D: tl.constexpr,            # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # First pass – compute mean and variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load half → float32
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0,
                       dtype=tl.float16).to(tl.float32)  # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1/√(var+eps)

    # --------------------------------------------------------------
    # Second pass – write normalized values (half)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0,
                       dtype=tl.float16).to(tl.float32)

        # (x - mean) * rsqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d)          # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d)          # fp32

        out = norm * w + b                                     # fp32
        out_half = out.to(tl.float16)                          # cast

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0) Convert input to fp16 (the whole pipeline works in fp16)
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16).contiguous()   # [B, N, N, C]
    B, N, _, _ = x_fp16.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension (fp16 output)
    # ------------------------------------------------------------------
    x_flat = x_fp16.view(total_rows, dim)                # [total_rows, C]
    x_norm = torch.empty_like(x_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – operations are serialized on the default stream.

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections (left/right + three gates)
    #    – a single GEMM in fp16
    # ------------------------------------------------------------------
    w_left       = weights["left_proj.weight"]
    w_right      = weights["right_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    # Concatenate the five weight matrices (shape: (5*H, C))
    fused_weight = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).half().contiguous()                                   # (5*H, C)

    # Linear projection:  [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm, fused_weight)               # fp16
    left_proj, right_proj, left_gate_proj, right_gate_proj, out_gate_proj = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [total_rows, 1] (half)
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_proj)     # fp16, shape [total_rows, H]
    right_gate = torch.sigmoid(right_gate_proj)   # fp16
    out_gate   = torch.sigmoid(out_gate_proj)     # fp16

    left  = left_proj  * left_gate                # fp16
    right = right_proj * right_gate               # fp16

    # ------------------------------------------------------------------
    # 5) Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden_dim)        # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)       # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    #   left_perm : (B, H, N, N)   – rows = i, cols = k
    #   right_perm: (B, H, N, N)   – rows = j, cols = k  (already transposed)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)     # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)  # fp16, [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [rows, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, hidden_dim)          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Apply final output gate and project back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                            # fp16

    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)      # [rows, H]

    final_fp16 = F.linear(gated_flat, to_out_weight_fp16) # fp16 → [rows, C]

    # Cast to float32 for the official output dtype
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
