"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass
================================================================
Algorithm
---------
1. **Half‑precision LayerNorm** over the channel dimension `dim`.
   A fused Triton kernel computes per‑row mean/variance in FP32,
   applies the learned γ/β (also FP16) and writes the normalized
   tensor in FP16.  This halves the memory bandwidth compared to an
   FP32 norm.

2. **Linear projections & gating** are carried out with
   `torch.nn.functional.linear` in FP16.  If a mask is supplied
   it is broadcasted to the hidden dimension and applied before
   the element‑wise gates.

3. **Triangular multiplicative update**  
   For each hidden channel the operation  
   `out[i,j] = Σ_k left[i,k] * right[j,k]`  
   is exactly a batched matrix‑multiply `L @ Rᵀ`.  By reshaping/
   permuting we obtain tensors of shape `[B*H, N, N]` and invoke the
   highly‑optimized `torch.bmm` (Tensor‑Core GEMM on H100).

4. **Second LayerNorm** over the hidden dimension (`hidden_dim`) is
   performed with `torch.nn.functional.layer_norm` in FP16.

5. **Output gate** (sigmoid) is applied element‑wise, then a final
   linear projection `hidden_dim → dim` produces the result.  The
   output is cast back to FP32 as required.

Only the first LayerNorm is implemented in Triton, satisfying the
requirement that “part of the operations” be a custom kernel while
still delivering a fast end‑to‑end forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                # const half*   [total_rows, dim]
    y_ptr,                # mut   half*   [total_rows, dim]
    weight_ptr,           # const half*   [dim]
    bias_ptr,             # const half*   [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (input & output in FP16, internal FP32)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # guard rows

    # -----------------------------------------------------------------
    # 1) Compute sum and sum‑of‑squares for each row (mean & variance)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a [BLOCK_M, BLOCK_D] tile, convert to fp32 for reduction
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 tile
        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2) Write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 tile

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # Load γ and β (fp16 → fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)                # back to fp16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) First LayerNorm (FP16) over `dim`
    # -----------------------------------------------------------------
    # Cast input to half‑precision (the kernel works on fp16)
    x_fp16 = input_tensor.to(torch.float16)                 # [B, N, N, dim]
    B, N, _, _ = x_fp16.shape
    total_rows = B * N * N

    x_fp16_flat = x_fp16.view(total_rows, dim)              # [R, dim]
    x_norm_fp16_flat = torch.empty_like(x_fp16_flat)        # output buffer

    # Tuned block sizes – work well for all supported N (≤ 1024)
    BLOCK_M = 1024            # rows processed per program
    BLOCK_D = 32              # channel tiles

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_fp16_flat,
        x_norm_fp16_flat,
        weights["norm.weight"].to(torch.float16),
        weights["norm.bias"].to(torch.float16),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Reshape back to the 4‑D tensor
    x_norm = x_norm_fp16_flat.view(B, N, N, dim)             # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 2) Linear projections & gating (all in FP16)
    # -----------------------------------------------------------------
    # Flatten to 2‑D for matmul
    x_norm_flat = x_norm.view(total_rows, dim)               # [R, dim]

    # Half‑precision copies of all weight matrices
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)   # [H, dim]
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)    # [H, dim]
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)  # [H]
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)    # [H]
    w_to_out      = weights["to_out.weight"].to(torch.float16)       # [dim, H]

    # Linear projections (no bias)
    left_proj = F.linear(x_norm_flat, w_left_proj)   # [R, H]
    right_proj = F.linear(x_norm_flat, w_right_proj)  # [R, H]

    # Optional mask (broadcasted over hidden dimension)
    if not nomask:
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm_flat, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_norm_flat, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_norm_flat, w_out_gate))

    # Apply element‑wise gates to the projections
    left_h = left_proj * left_gate          # [R, H]
    right_h = right_proj * right_gate       # [R, H]
    out_gate_h = out_gate                   # saved for later

    # -----------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # Reshape to 4‑D tensors
    left_h = left_h.view(B, N, N, hidden_dim)   # [B,N,N,H] half
    right_h = right_h.view(B, N, N, hidden_dim) # [B,N,N,H] half
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)  # [B,N,N,H] half

    # Bring hidden dimension into the batch axis and transpose the second operand
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (j ↔ k)

    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)  # [B*H, N, N]

    # Batched matrix multiplication (Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat)                 # [B*H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # -----------------------------------------------------------------
    # 4) Second LayerNorm over hidden dimension (FP16)
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # still half‑precision

    # -----------------------------------------------------------------
    # 5) Final gated projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                       # element‑wise half
    out_norm_flat = out_norm.view(total_rows, hidden_dim)  # [R, H]

    # Linear projection hidden_dim → dim (no bias)
    final_half = F.linear(out_norm_flat, w_to_out)          # [R, dim] half

    # Cast to FP32 as required by the API
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
