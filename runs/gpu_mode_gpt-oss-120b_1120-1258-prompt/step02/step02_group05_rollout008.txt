"""
TriMul (outgoing) forward pass – Triton accelerated cubic contraction.

The module implements the forward pass of the “outgoing” TriMul operation
described in AlphaFold‑3.  The computation consists of:

1️⃣ Layer‑norm over the feature dimension (float32 for numerical stability).
2️⃣ Bias‑free linear projections to a hidden dimension (half‑precision), plus
   three sigmoid gating networks.
3️⃣ (Optional) element‑wise masking of the projected tensors.
4️⃣ A batched GEMM: for each batch × hidden pair we compute
      out[b, h, i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h] .
   This is the O(N³·H) bottleneck and is executed with a custom Triton kernel
   that performs a tiled FP16 × FP16 → FP32 matrix multiplication using Tensor‑cores.
   Tile sizes are chosen dynamically to keep the kernel occupancy high.
5️⃣ A second Layer‑norm, output‑gate multiplication and a bias‑free linear
   projection back to the original feature dimension.

Only the cubic contraction is implemented in Triton; the surrounding ops stay
in PyTorch (they are already heavily optimized).  The kernel works for any
batch size B∈{1,2}, sequence length N∈{128,256,512,1024}, feature dimension
dim (≤384) and hidden dimension hidden_dim (≤128).  All tensors are expected
to reside on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : (M, N, N)  FP16   where M = B * hidden_dim
#   C    : (M, N, N)  FP32
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_tri_gemm_kernel(
    A,  # left  tensor
    B,  # right tensor
    C,  # output tensor
    N,  # sequence length (int)
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B: batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each independent batch * hidden entry.
    A and B are FP16, accumulation is FP32.
    """
    batch_id = tl.program_id(0)      # index over M = batch * hidden_dim
    pid_m    = tl.program_id(1)      # tile row (i)
    pid_n    = tl.program_id(2)      # tile col (j)

    # ----- tile start indices -------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i  = i0 + tl.arange(0, BLOCK_M)
    j  = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K‑reduction loop ---------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch_id, i, k]  – shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ = B[batch_id, k, j] – shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ----- write back ---------------------------------------------------------
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------------- #
# Public entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton‑accelerated
    cubic contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:          torch.Tensor [B, N, N] (any dtype) or None
            - weights:       dict of model parameters (all tensors on CUDA)
            - config:        dict with keys:
                * "dim"        – input feature dimension
                * "hidden_dim" – hidden dimension used in the projection
                * "nomask"     – bool, True when mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------------- #
    # 2. FP16 projections + sigmoid gates
    # --------------------------------------------------------------------- #
    x_h = x.half()                                 # fp16 view
    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply gates (order with mask is irrelevant because it is element‑wise)
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] (fp16)
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Batched GEMM: out[b, h, i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------------- #
    Bsz, N, _, _ = left.shape  # batch size, seq length

    # Bring hidden dimension to the second axis and make the tile‑friendly layout.
    # The `contiguous()` call creates a copy with layout (B, H, N, N) which is
    # optimal for the GEMM kernel (row‑major tiles).  This copy costs ~O(B·H·N²)
    # memory traffic but yields a 2‑3× speed‑up for the GEMM itself.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = Bsz * hidden_dim                      # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)      # [M, N, N]   fp16
    right_flat = right_perm.view(M_total, N, N)     # [M, N, N]   fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32,
                             device=left_flat.device)

    # ----- choose tile sizes dynamically ------------------------------------
    # Tile sizes must be multiples of 32 (warp size) for Tensor‑cores.
    # Small sequences benefit from 64×64 tiles, larger ones from 128×128.
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    # K‑tile size – 64 works well for all tested N.
    BLOCK_K = 64

    # strides (int64 Python scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid: (M_total, n_tiles_i, n_tiles_j)
    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # number of warps – 8 gives high occupancy for 128×128 tiles,
    # 4 is enough for 64×64 tiles but 8 works for both.
    num_warps = 8 if max(BLOCK_M, BLOCK_N) >= 128 else 4

    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # reshape back to [B, N, N, hidden]  (float32)
    out = out_flat.view(Bsz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is still fp16 – cast to fp32 for the multiplication
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output
