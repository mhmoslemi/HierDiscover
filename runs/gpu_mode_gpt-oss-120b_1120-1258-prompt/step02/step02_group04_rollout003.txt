"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The operator works on a 4‑D tensor X ∈ R^{B×N×N×D}:

1. Layer‑norm over the last dimension (FP32 → FP16 for heavy work).
2. Two *project+gate* pairs are computed:
   * (left_proj , left_gate)  from the same linear transform.
   * (right_proj, right_gate) from the same linear transform.
   The weights are concatenated so that each pair needs only **one**
   matrix‑multiply instead of two.
3. Optional pairwise mask is applied to the projected tensors.
4. Gated tensors are multiplied element‑wise.
5. The N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]
   is a **batched GEMM** `C = A @ Bᵀ` for each hidden dimension.
   We flatten (B,hidden) → batch and launch a custom Triton kernel that
   computes many independent GEMMs in parallel, using fp16 inputs,
   fp32 accumulation and Tensor‑core‑friendly tiling.
6. Output LayerNorm → out‑gate → final linear projection back to D.

The heavy N³ part is fully fused in Triton; the remaining linear /
layer‑norm work uses PyTorch (still very fast).  Block sizes are
adapted to the sequence length to keep the kernel efficient for all
tested sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C[b] = A[b] @ B[b]ᵀ
#   A : (batch, M, K)   fp16
#   B : (batch, K, N)   fp16   (accessed as (K,N))
#   C : (batch, M, N)   fp16   (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,  # fp16 * (batch, M, K)
    b_ptr,  # fp16 * (batch, K, N)
    c_ptr,  # fp16 * (batch, M, N)
    M, N, K,                         # matrix sizes (int32)
    stride_az, stride_am, stride_ak,  # strides for A
    stride_bz, stride_bk, stride_bn,  # strides for B
    stride_cz, stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    The first program‑id packs (batch, tile_m), the second is tile_n.
    """
    pid0 = tl.program_id(0)       # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)       # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // num_tiles_m
    tile_m   = pid0 %  num_tiles_m
    tile_n   = pid1

    # -------------------------------------------------------------
    # Offsets within the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile (without the K‑loop offset)
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Mask out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16·fp16 → fp32 accumulation (Tensor‑core friendly)
        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Store result (back to fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0) Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] FP32

    # -----------------------------------------------------------------
    # 2) Cast to half‑precision for the bulk of the work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Fuse projection + gate linear layers
    # -----------------------------------------------------------------
    # left_proj + left_gate
    left_combined_w = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),
            weights["left_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )
    # right_proj + right_gate
    right_combined_w = torch.cat(
        [
            weights["right_proj.weight"].to(torch.float16),
            weights["right_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )
    # out_gate (single linear)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # ----- left side -----
    left_combined = F.linear(x_fp16, left_combined_w)          # [B,N,N,2*hidden]
    left_proj = left_combined[..., :hidden_dim]               # [B,N,N,hidden]
    left_gate = left_combined[..., hidden_dim:]               # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_gate)

    # ----- right side -----
    right_combined = F.linear(x_fp16, right_combined_w)       # [B,N,N,2*hidden]
    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate)

    # ----- out‑gate -----
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 4) Apply optional pairwise mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -----------------------------------------------------------------
    # 5) Gated tensors
    # -----------------------------------------------------------------
    left  = left_proj * left_gate   # [B,N,N,hidden] FP16
    right = right_proj * right_gate # [B,N,N,hidden] FP16

    # -----------------------------------------------------------------
    # 6) N³ contraction via fused batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N_seq, _, _ = left.shape   # B, N, N, hidden

    # Rearrange to (batch*hidden, M, K) / (batch*hidden, K, N)
    #   left : (B, hidden, N, N)   → (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)
    #   right : (B, hidden, N, N) with K as first dim → (B*hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N_seq, N_seq)

    # Output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -------------------------------------------------------------
    # Dynamic block size selection (roughly matches seq length)
    # -------------------------------------------------------------
    if N_seq <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N_seq <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 128
        num_warps = 8

    grid_m = triton.cdiv(N_seq, BLOCK_M)
    grid_n = triton.cdiv(N_seq, BLOCK_N)
    # First dimension packs (batch*hidden) and tile_m
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,      # good latency‑hiding for all sizes
    )

    # -------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden] and cast to FP32
    # -------------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N_seq, N_seq)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B,N,N,hidden] FP32

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in FP16; broadcast multiply with FP32 works fine
    out = out * out_gate.to(torch.float32)

    # Final linear projection hidden → dim (no bias)
    out = F.linear(out, weights["to_out.weight"])   # stays FP32

    return out
```

**Explanation of key optimisations**

* **Fused projection+gate** – concatenating projection and gate weight matrices halves the number of `torch.nn.functional.linear` calls (5 → 3), reducing memory traffic and kernel launches.
* **Dynamic tiling** – block sizes adapt to the sequence length, avoiding excessive idle threads for small `N` while keeping large tiles for long sequences.
* **Batched GEMM in Triton** – the N³ contraction is expressed as many independent matrix multiplications (`A @ Bᵀ`). The kernel uses fp16 inputs, fp32 accumulation and Tensor‑core‑friendly `tl.dot`.
* **Minimal data movement** – tensors are made contiguous only once before the Triton launch, and all heavy arithmetic stays on‑device.
* **Mixed‑precision** – the dominant compute (the GEMM) runs in fp16, while layer‑norms and the final projection stay in fp32 for numerical stability.

The function `custom_kernel` follows the required signature and can be used directly as the entry point for the TriMul forward pass.
