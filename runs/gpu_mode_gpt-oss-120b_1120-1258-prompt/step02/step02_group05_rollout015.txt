"""
Optimized outgoing TriMul implementation.
1. LayerNorm (float32)
2. FP16 linear projections + sigmoid gates, optional mask.
3. Batched GEMM for cubic contraction using a tuned Triton kernel.
   The kernel computes out = left @ rightᵀ for each hidden slice.
   Tile sizes: 64×64×64 – a good trade‑off between shared‑memory
   usage and register pressure on H100.
4. Second LayerNorm + out‑gate + final FP32 linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B are FP16, C FP32)
# ----------------------------------------------------------------- #
@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor (M, N, N) – FP16
    B,               # right tensor (M, N, N) – FP16 (will be read as Bᵀ)
    C,               # output (M, N, N)       – FP32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A : batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B : batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C : batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes a batch of matrix‑multiplications C = A @ Bᵀ.
    A, B – FP16, C – FP32.  Each program instance works on one
    ``batch’’ element (i.e. a particular hidden slice of a specific
    batch) and on a (BLOCK_M, BLOCK_N) tile of the output matrix.
    """
    batch_id = tl.program_id(0)      # index over M = batch * hidden
    pid_m   = tl.program_id(1)      # tile‑row index
    pid_n   = tl.program_id(2)      # tile‑col index

    # starting coordinates of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    # row / column indices inside the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (tensor‑core dot returns FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k] – shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Bᵀ[batch, k, j] = B[batch, j, k] – shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # TL.dot internally uses tensor‑cores and accumulates in FP32
        acc += tl.dot(a, b)

    # write the tile back to C
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.
    The heavy cubic contraction is performed by a tuned Triton kernel.
    All other linear layers and layer‑norms are executed with PyTorch
    primitives (they already use highly‑optimized cuBLAS / cuDNN kernels).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # -----------------------------------------------------------------
    # 2. FP16 projection + gating
    # -----------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – fp16

    # bias‑free linear projections (FP16)
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half()) # [B, N, N, H]

    # sigmoid gates (FP16)
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply mask before gating (mask broadcast on hidden)
    if not nomask:
        # mask : [B, N, N] → [B, N, N, 1] (fp16)
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. Batched GEMM: out = left @ rightᵀ for every hidden slice
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                        # B, seq_len, seq_len, hidden
    # rearrange to (batch, hidden, N, N) – makes K‑dimension contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M_total = B * hidden_dim                         # number of independent GEMMs
    left_flat = left_perm.view(M_total, N, N)        # [M, N, N] – fp16
    right_flat = right_perm.view(M_total, N, N)      # [M, N, N] – fp16
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # stride arguments (int64 Python scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid: (batch*hidden, tile_rows, tile_cols)
    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is fp16 – up‑cast to fp32 for the multiplication
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input over the feature dimension (FP32).
2️⃣ Cast to FP16 and apply four bias‑free linear projections:
   * left_proj  → left  (hidden_dim)
   * right_proj → right (hidden_dim)
   * left_gate, right_gate, out_gate → sigmoid‑gated masks (hidden_dim)
   The optional pair‑wise mask is applied after the two projections
   and before the element‑wise gates.
3️⃣ Cubic contraction:
   For each batch‑&‑hidden slice we need
       out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ.  The heavy O(N³·H) work‑horse
   is performed by a tuned Triton kernel (64 × 64 × 64 tiles)
   that uses Tensor‑cores (FP16→FP32 accumulation) and drops the
   extra copies required by a naïve implementation.
4️⃣ Second Layer‑norm on the hidden dimension, multiplication by the
   output gate and a final bias‑free linear projection back to *dim*.
   The last projection is small enough to stay in PyTorch.
All tensors stay on the CUDA device; the final tensor is FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------- #
# Triton kernel – batched GEMM C = A @ Bᵀ
#   A, B : FP16, shape (M, N, N)   (M = batch * hidden_dim)
#   C    : FP32, shape (M, N, N)
# --------------------------------------------------------------------- #
@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor  (M, N, N)  – FP16
    B,               # right tensor  (M, N, N)  – FP16 (read as Bᵀ)
    C,               # output      (M, N, N)  – FP32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # A strides: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # B strides: batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # C strides: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program instance processes a single (batch*hidden) entry
    and a (BLOCK_M, BLOCK_N) tile of the output matrix.
    """
    batch_id = tl.program_id(0)      # 0‑th axis → independent GEMM
    pid_m    = tl.program_id(1)      # tile‑row index
    pid_n    = tl.program_id(2)      # tile‑col index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- reduction over k -------------------------------------- #
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k]  → (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )                           # already FP16

        # Bᵀ[batch, k, j] = B[batch, j, k]  → (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)          # Tensor‑core dot, accumulates in FP32

    # -------- store ------------------------------------------------- #
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask         : [B, N, N]      (any dtype, optional)
            - weights      : dict of model parameters
            - config       : dict with "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, N, N, dim]  FP32

    # -----------------------------------------------------------------
    # 2️⃣ FP16 projections + sigmoid gates
    # -----------------------------------------------------------------
    x_h = x.half()                              # [B, N, N, dim]  FP16

    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # optional mask (broadcast on hidden dim)
    if not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3️⃣ Batched GEMM: out = left @ rightᵀ  (per hidden slice)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                         # B, seq_len, seq_len, hidden

    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M_total = B * hidden_dim                         # independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)       # (M, N, N)  FP16
    right_flat = right_perm.view(M_total, N, N)      # (M, N, N)  FP16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # tile sizes tuned for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (int64 scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]  (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Output LayerNorm → gating → final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                     # broadcasted multiplication
    output = F.linear(out, weights["to_out.weight"])

    return output
