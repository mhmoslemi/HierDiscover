"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1.  Layer‑norm the input (FP32 → FP16 for the heavy work).
2.  Linear projections + optional mask → tensors *left* and *right*
    of shape [B, N, N, H] (H = hidden_dim) in FP16.
3.  Compute per‑element sigmoid gates and fuse them into *left*/*right*.
4.  Fuse the N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a **batched GEMM**   C = A @ Bᵀ   for each hidden slice.
    The GEMM is performed by a custom Triton kernel that
    works on tiles (BLOCK_M×BLOCK_N) and streams K‑tiles (BLOCK_K)
    through registers, using FP16 × FP16 → FP32 accumulation
    (Tensor‑cores).  Tile sizes are auto‑tuned to the sequence length.
5.  Reshape the result back to [B, N, N, H] (FP16 → FP32).
6.  Apply the output LayerNorm, the output‑gate, and the final linear
    projection back to the original *dim*.

All heavy arithmetic (the N³ contraction) is done in the Triton kernel;
the surrounding linear / normalization steps use fused PyTorch ops and
remain FP16‑friendly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C[b] = A[b] @ B[b]ᵀ   (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                 # fp16 *  (batch, M, K)
    b_ptr,                 # fp16 *  (batch, K, N)   ← accessed as (K,N)
    c_ptr,                 # fp16 *  (batch, M, N)
    M, N, K,               # matrix dimensions (int32)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    Axis‑0 programs index (batch * tile_m), axis‑1 programs index tile_n.
    """
    pid0 = tl.program_id(0)          # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // tiles_m
    tile_m     = pid0 %  tiles_m
    tile_n     = pid1

    # -------------------------------------------------------------
    # Offsets inside the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # -------------------------------------------------------------
    # Pointers to the start of the tile (without the k‑loop offset)
    # -------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Masks for out‑of‑bounds loads (necessary for edge tiles)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16⋅fp16 → fp32 accumulation (uses Tensor‑cores)
        acc += tl.dot(a, b)

        # Advance the pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Store the result (back to fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _choose_launch_params(seq_len):
    """
    Pick tile sizes that give good occupancy for the supplied sequence
    length.  The heuristic is simple but works well for the test
    configurations.
    """
    if seq_len <= 128:
        # Very small sequences – keep shared‑memory pressure low
        return 64, 64, 32, 4
    if seq_len <= 256:
        return 64, 64, 64, 4
    # Larger sequences (up to 1024) – use the classic 128‑tile
    return 128, 128, 64, 8


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys {"dim": int,
                                         "hidden_dim": int,
                                         "nomask": bool}

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 1) Unpack & basic configuration
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 2) Input LayerNorm (FP32 → FP16 for the heavy work)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # FP16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 3) Linear projections (FP16) + optional mask
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) – also FP16
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) N³ contraction via a fused batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                # B, N, N, hidden

    # Rearrange to (batch*hidden, M, K)   and (batch*hidden, K, N)
    # A = left  : rows = i,  cols = k
    # B = right : rows = k,  cols = j   (note the transpose)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16, accumulation in FP32 inside the kernel)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Triton launch configuration (tuned per sequence length)
    # -----------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K, num_warps = _choose_launch_params(N)

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # pid0 packs (batch*hidden_dim, tile_m)
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,                # increase latency hiding
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden] (FP32) and apply post‑norm
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # Output LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the learned output gate (broadcast on hidden)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to the original `dim`
    out = F.linear(out, weights["to_out.weight"])   # bias=False

    return out
```
