pid = tl.program_id(0)

sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

for d in range(0, tl.cdiv(dim, BLOCK_D)):
    offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D) # [BLOCK_D]
    mask_d = offs_d < dim

    vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq += tl.sum(vals * vals, axis=1)

mean = sum_val / dim
var = sum_sq / dim - mean * mean
rstd = tl.rsqrt(var + eps)

# second pass
for d in range(0, tl.cdiv(dim, BLOCK_D)):
    offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # gamma and beta
    gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
    beta = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

    out = out * gate
    # store
    out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None,:])

"""
TriMul outgoing kernel – fast forward pass (AlphaFold‑3)

Algorithm
---------
1. **Row‑wise LayerNorm (float32)** on the input tensor `x` `[B,N,N,D]`.
   Implemented in a small Triton kernel that computes mean/variance per row,
   applies the learned γ/β and writes the normalized result (`x_norm`).

2. **Single fused Linear**:
   Concatenate the five weight matrices
   `[left_proj, left_gate, right_proj, right_gate, out_gate]`
   (shape `(5*H, D)`).  A single half‑precision `F.linear` computes
   all projections and gates at once:
   `proj = x_norm @ W_concatᵀ → [R,5*H]` where `R=B·N·N`.

3. **Mask & sigmoid**:
   *Mask* (if present) is applied to the two projection parts only.
   The gate slices are passed through `torch.sigmoid` and combined with
   their corresponding projections → `left` and `right` tensors of shape
   `[R, H]`.

4. **Tri‑multiplicative update**:
   Reshape `left/right` to `[B,N,N,H]`, permute to `[B,H,N,N]`,
   transpose the right operand (`[...,k,j]`) and perform a batched matrix
   multiplication `L @ Rᵀ` (implemented with `torch.bmm`).  This yields the
   intermediate tensor `out` of shape `[B,N,N,H]` (half‑precision).

5. **LayerNorm + output‑gate (half‑precision)**:
   A second Triton kernel fuses the hidden‑dim LayerNorm, the learned
   γ/β (`to_out_norm`) and the element‑wise output gate (`out_gate`).  The
   kernel writes the gated‑normalised tensor directly.

6. **Final linear projection**:
   `out_normed_gate @ to_out.weightᵀ` completes the forward pass.
   The result is reshaped back to `[B,N,N,D]` and cast to `float32`.

Only the two LayerNorm‑related parts are custom Triton kernels; all heavy
matrix‑multiplications are delegated to highly‑optimised cuBLAS/Tensor‑Core
operations.  Mixed‑precision (FP16) is used for the bulk of the work,
while the final output remains FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       float*   [total_rows, dim]  (output)
    weight_ptr,     # const float*   [dim]               (γ)
    bias_ptr,       # const float*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm in float32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – compute per‑row sum and sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq += tl.sum(vals * vals, axis=1)          # row‑wise sum of squares

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass – write normalized output
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*    [total_rows, dim]   (input to LN)
    y_ptr,          #       half*    [total_rows, dim]   (LN output * gate)
    gate_ptr,       # const half*    [total_rows, dim]   (out_gate)
    weight_ptr,     # const half*    [dim]                (γ)
    bias_ptr,       # const half*    [dim]                (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    LayerNorm across the hidden dimension (half‑precision) followed by
    element‑wise multiplication with a gate tensor.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – mean / variance (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass – normalize, apply γ/β, multiply by gate
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_val = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # normalize
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate

        # store back as half
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: torch.Tensor [B, N, N, D]  (float32)
        - mask_tensor : torch.Tensor [B, N, N]   (bool/int, optional)
        - weights_dict: mapping from layer names to torch.Tensor
        - config_dict : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------
    # 1. Unpack inputs & constants
    # ------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ---------------------------------------------------------
    # 2. First LayerNorm over the channel dimension (float32)
    # ---------------------------------------------------------
    x = input_tensor.contiguous()       # [B,N,N,D]  float32
    x_flat = x.view(total_rows, dim)    # [R,D]
    x_norm_flat = torch.empty_like(x_flat)  # output buffer (float32)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],      # γ   (float32, [D])
        weights["norm.bias"],        # β   (float32, [D])
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # (no need to sync explicitly; torch will handle dependencies)

    # ---------------------------------------------------------
    # 3. Cast to half‑precision and compute all projections/gates
    # ---------------------------------------------------------
    x_norm_h = x_norm_flat.to(torch.float16)        # [R,D] half
    # Concatenate weight matrices for a single GEMM
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, D)

    # fused linear: [R,D] @ (5*H,D)ᵀ → [R,5*H]
    proj_all = F.linear(x_norm_h, w_concat)        # half

    # Slice the concatenated result
    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2*H]
    right_proj  = proj_all[:, 2*H:3*H]
    right_gate  = proj_all[:, 3*H:4*H]
    out_gate    = proj_all[:, 4*H:5*H]

    # Optional mask (broadcast over hidden dimension)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)  # [R,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply sigmoid to gate tensors
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # Element‑wise gating of the projections
    left  = left_proj * left_gate          # [R, H] half
    right = right_proj * right_gate        # [R, H] half

    # ---------------------------------------------------------
    # 4. Tri‑multiplicative update via batched GEMM
    # ---------------------------------------------------------
    # reshape to 4‑D tensors
    left  = left.view(B, N, N, H)   # [B,N,N,H]
    right = right.view(B, N, N, H)  # [B,N,N,H]

    # permute so that hidden dimension becomes batch axis
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # transpose (k ↔ j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)   # [B*H, N, N]

    # Batched matrix multiply (Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] half

    # Restore original layout: [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ---------------------------------------------------------
    # 5. Second LayerNorm over hidden dimension + out_gate
    # ---------------------------------------------------------
    out_flat = out.view(total_rows, H)               # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)     # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)       # half, final pre‑projection

    # Use the same block size strategy; H ≤ 128 → BLOCK_D=32 works well
    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ   (half)
        weights["to_out_norm.bias"].to(torch.float16),     # β   (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # ---------------------------------------------------------
    # 6. Final linear projection back to dim
    # ---------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [D, H]
    final_half = F.linear(out_norm_gate, w_to_out)          # [R, D] half
    final = final_half.view(B, N, N, dim).to(torch.float32) # cast back to FP32

    return final
