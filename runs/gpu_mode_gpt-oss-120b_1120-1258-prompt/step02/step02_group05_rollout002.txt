"""
TriMul (outgoing) – optimized Triton implementation
================================================================
The module implements the forward pass of the “outgoing” TriMul operator
described in AlphaFold‑3.  The heavy cubic contraction

    out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is executed by a custom Triton kernel that performs a batched GEMM
C = A @ Bᵀ for each (batch, hidden) pair.  All tensors are kept in
FP16 during the contraction and accumulated in FP32 for numerical
stability.  The kernel uses a simple 3‑dimensional grid:

    axis‑0 → (batch * hidden_dim)    (independent GEMMs)
    axis‑1 → tile rows (i)
    axis‑2 → tile cols (j)

Tile sizes are chosen adaptively:
    N ≤ 512      → BLOCK_M = BLOCK_N = 128
    N  > 512     → BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64 for all cases (fits nicely into H100 tensor‑cores).

The rest of the TriMul pipeline (LayerNorms, linear projections,
sigmoid gates, optional masking, and the final output projection) is
performed with PyTorch – the huge O(N³·H) part is fully off‑loaded to
the Triton kernel.

The function `custom_kernel` follows the required signature:
    (input_tensor, mask, weights, config) -> output_tensor
and returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor (M, N, N)      – FP16
    B,               # right tensor (M, N, N)      – FP16 (read as Bᵀ)
    C,               # output tensor (M, N, N)    – FP32
    N,               # sequence length (int)
    stride_am, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bm, stride_bj, stride_bk,   # B strides: batch, j, k  (Bᵀ view)
    stride_cm, stride_ci, stride_cj,   # C strides: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ for each entry in the first dimension (M = B*H).
    Each thread block computes a BLOCK_M × BLOCK_N tile of the output.
    """
    batch_id = tl.program_id(0)          # 0‑th axis → independent GEMM
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # start indices of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # intra‑tile coordinates
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (dot on FP16 returns FP32 on H100)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch_id, i, k]  → shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ → B[batch_id, k, j]  shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back tile
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    cubic contraction.  All operations are performed in mixed precision
    (FP16 for the heavy GEMM, FP32 for the final output).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask        : torch.Tensor, shape [B, N, N] (ignored if config["nomask"])
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2. Cast to FP16 and linear projections + sigmoid gates
    # -----------------------------------------------------------------
    # Cast once
    x_h = x.half()   # FP16

    # Pre‑cast weight matrices to FP16 (bias‑free)
    w_left   = weights["left_proj.weight"].half()   # (dim, hidden_dim)
    w_right  = weights["right_proj.weight"].half()
    w_lgate  = weights["left_gate.weight"].half()
    w_rgate  = weights["right_gate.weight"].half()
    w_outgate = weights["out_gate.weight"].half()

    # Linear projections (FP16)
    left  = F.linear(x_h, w_left)    # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # Gates (FP16, sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1], cast to FP16
        mask_exp = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Cubic contraction: out[b, h, i, j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → batched GEMM C = A @ Bᵀ where A and B have shape (M=N_batch*H, N, N)
    # -----------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    # bring hidden dim to the second axis for contiguous layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = B * hidden_dim                 # number of independent GEMMs
    # reshape to (M_total, N, N) – still FP16
    left_flat  = left_perm.view(M_total, N_seq, N_seq)
    right_flat = right_perm.view(M_total, N_seq, N_seq)

    # allocate output (FP32 accumulator)
    out_flat = torch.empty((M_total, N_seq, N_seq), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 5. Choose tile sizes (simple heuristic)
    # -----------------------------------------------------------------
    if N_seq <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 64

    # compute strides for the kernel (int64 python scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    # B is accessed as B[batch, k, j] → stride for j is dim‑2, for k dim‑1
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid: (M_total, tiles_i, tiles_j)
    grid = (
        M_total,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_seq,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # a good default for H100
    )

    # reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm + out‑gate scaling + final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()   # out_gate was FP16, cast to FP32 for safety
    output = F.linear(out, weights["to_out.weight"])   # bias‑free, fp32 → fp32

    return output
