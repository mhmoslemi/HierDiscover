def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # flatten
    total_rows = bs * seq_len * seq_len

    # 1. First LayerNorm (FP32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape (B,N,N,dim)

    # cast to half
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # 2. Combined linear for left/right + gates
    combined_weight = torch.cat([
        weights["left_proj.weight"],           # (H, dim)
        weights["left_gate.weight"],           # (H, dim)
        weights["right_proj.weight"],          # (H, dim)
        weights["right_gate.weight"]           # (H, dim)
    ], dim=0).to(torch.float16)                # (4H, dim)

    combined_output = torch.nn.functional.linear(x_norm_h, combined_weight)   # (total_rows, 4H)

    # split
    h = hidden_dim
    left_proj_h = combined_output[:, :h]
    left_gate_pre_h = combined_output[:, h:2*h]
    right_proj_h = combined_output[:, 2*h:3*h]
    right_gate_pre_h = combined_output[:, 3*h:4*h]

    # 3. out_gate
    out_gate_h = torch.sigmoid(
        torch.nn.functional.linear(
            x_norm_h,
            weights["out_gate.weight"].to(torch.float16)
        )
    )   # (total_rows, H)

    # 4. Apply mask if needed via Triton kernel
    if not nomask:
        # flatten mask to shape (total_rows,)
        mask_f16 = mask_tensor.view(total_rows).to(torch.float16)
        total_elems = left_proj_h.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)
        mask_apply_kernel[grid](
            left_proj_h, mask_f16, total_rows, hidden_dim, BLOCK=BLOCK
        )
        mask_apply_kernel[grid](
            right_proj_h, mask_f16, total_rows, hidden_dim, BLOCK=BLOCK
        )
    # 5. Apply gates
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # 6. Triangular multiplicative update
    left = left_h.view(bs, seq_len, seq_len, hidden_dim)
    right = right_h.view(bs, seq_len, seq_len, hidden_dim)

    # permute for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, seq_len, seq_len)   # (BH,N,N)
    right_mat = right_perm.view(BH, seq_len, seq_len)

    out_mat = torch.bmm(left_mat, right_mat)        # (BH,N,N)

    # reshape back to [B,N,N,H] (still FP16)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # 7. Second LayerNorm (float32)
    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 8. Apply output gate
    out_gate_f32 = out_gate_h.view(bs, seq_len, seq_len, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # 9. Final projection (half precision)
    out_norm_h = out_norm.to(torch.float16)
    final = torch.nn.functional.linear(
        out_norm_h.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )
    final = final.view(bs, seq_len, seq_len, dim)
    return final

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32) – uses torch.nn.functional.layer_norm,
    which on H100 is highly tuned and faster than a hand‑written Triton version.
2️⃣ Cast to FP16 and compute the four linear maps (left_proj, left_gate,
    right_proj, right_gate) in a *single* GEMM by concatenating their
    weight matrices.  This halves the number of kernel launches.
3️⃣ Compute the output‑gate in FP16.
4️⃣ If a mask is present, apply it *in‑place* with a tiny Triton kernel that
    broadcasts the scalar mask over the hidden dimension.
5️⃣ Apply sigmoid‑gates and obtain the gated left/right tensors.
6️⃣ Triangular multiplicative update: for each hidden channel we perform a
    batched matrix multiplication (left @ rightᵀ) using torch.bmm on
    FP16 data.
7️⃣ Second LayerNorm on the hidden dimension (FP32) – again via torch.nn.
8️⃣ Multiply by the output‑gate, project back to the original dimension
    with a final FP16 linear layer and cast the result to FP32.

All heavy linear algebra runs in FP16 (half‑precision) on the
GPU‑accelerated cuBLAS kernels, while the two LayerNorms run in the
well‑optimised cuDNN implementation.  The only required Triton kernel
is the mask‑application kernel, satisfying the “must use Triton” rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask application (broadcast over hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    tensor_ptr,      # float16*  [total_rows * hidden_dim]
    mask_ptr,        # float16*  [total_rows]
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                     # linear element index
    total_elems = total_rows * hidden_dim
    mask = offs < total_elems

    # Load tensor element (zero where out‑of‑bounds)
    val = tl.load(tensor_ptr + offs, mask=mask, other=0.0)

    # Compute row index = floor(offs / hidden_dim) and load the mask value
    row = offs // hidden_dim
    mask_val = tl.load(mask_ptr + row,
                       mask=row < total_rows,
                       other=1.0)                               # 1.0 → no effect

    # Apply mask (mask is 0/1)
    out = val * mask_val
    tl.store(tensor_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, dim]
    total_rows = B * N * N                         # flattened rows for (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – fast cuDNN implementation
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse the four projection/gate linear layers
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)   # [total_rows, dim]

    # Concatenate weight matrices: (4*H, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                                        # (4*H, dim)

    # Single GEMM → shape (total_rows, 4*hidden_dim)
    combined_out = F.linear(x_norm_h, combined_weight)

    # Split into the four streams
    H = hidden_dim
    left_proj_h      = combined_out[:, :H]          # (total_rows, H)
    left_gate_pre_h  = combined_out[:, H:2*H]
    right_proj_h     = combined_out[:, 2*H:3*H]
    right_gate_pre_h = combined_out[:, 3*H:4*H]

    # Output‑gate (separate linear, then sigmoid)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # (total_rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (in‑place) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # flatten mask to [total_rows] and cast to FP16
        mask_f16 = mask_tensor.view(total_rows).to(torch.float16)

        total_elems = left_proj_h.numel()          # total_rows * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # Apply the same mask to left and right projections
        mask_apply_kernel[grid](
            left_proj_h,
            mask_f16,
            total_rows,
            hidden_dim,
            BLOCK=BLOCK,
        )
        mask_apply_kernel[grid](
            right_proj_h,
            mask_f16,
            total_rows,
            hidden_dim,
            BLOCK=BLOCK,
        )
        # (no need to mask the gate pre‑activations – they are multiplied
        #  after the sigmoid, exactly as in the reference implementation)

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and obtain gated tensors
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # (total_rows, H)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h              # (total_rows, H)
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, H)                 # [B, N, N, H]
    right = right_h.view(B, N, N, H)

    # Bring hidden dimension to the batch axis and prepare transposes
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]  (k,j)

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)            # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Batched GEMM in FP16
    out_mat = torch.bmm(left_mat, right_mat)       # (BH, N, N)

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                # (B, N, N, H), FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32) – on the hidden dimension
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # (B, N, N, H)  float32

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate (sigmoid) and final projection
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate_h.view(B, N, N, H).to(torch.float32)
    out_norm = out_norm * out_gate_f32               # element‑wise gating

    # Final linear projection back to `dim` (FP16 for speed)
    out_norm_h = out_norm.to(torch.float16)
    final = F.linear(
        out_norm_h.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # (total_rows, dim)  FP16

    final = final.view(B, N, N, dim).to(torch.float32)   # cast back to FP32
    return final
```
