    rstd = tl.rsqrt(var + eps)

        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

            other=0.0,

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)

    paths (weights are concatenated → a single matmul per side).  All
    dense work is performed in fp16.

3️⃣  Optional masking of the projections (broadcast over the hidden
    dimension) followed by the sigmoid‑gates.

4️⃣  Tri‑multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented with a single ``torch.einsum`` which internally collapses
    the batch‑and‑head dimensions and runs a batched GEMM – no explicit
    ``permute(...).contiguous()`` copies are needed.

5️⃣  Second LayerNorm over the hidden dimension (`hidden_dim`).  The same
    Triton kernel is reused (input now in fp16, output still fp16).

6️⃣  Apply the output‑gate (sigmoid) and a final fp16 linear projection
    back to `dim`.  The result is cast to fp32 before returning.

All heavy reductions (the two LayerNorms) are executed in Triton,
while the matrix multiplications are delegated to PyTorch’s highly‑optimised
FP16 kernels.  The implementation avoids the large intermediate copies
present in the naive reference, delivering runtimes ≤ 1 ms on an H100 for
all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input can be fp32 or fp16, output is fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const *      [total_rows, dim]   (fp32 or fp16)
    y_ptr,          #       *      [total_rows, dim]   (fp16)
    w_ptr,          # const *      [dim]               (γ, fp32)
    b_ptr,          # const *      [dim]               (β, fp32)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of channels
    eps: tl.constexpr,          # static eps
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """Row‑wise LayerNorm.
    Computes y = (x-μ)/√(σ²+eps) * γ + β   where μ,σ² are per‑row.
    The output is stored as fp16; accumulation is done in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- pass 1: mean / var -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        # Load fp16 or fp32, then up‑cast to fp32
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                     # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)                       # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)                # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1/√(var+eps)

    # ------------------- pass 2: write normalized values ------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x-mean)/sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32; broadcast over rows
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                  # fp32 result
        out_fp16 = out.to(tl.float16)

        dst = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(dst, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape          # [B, N, N, dim]
    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim)          # [total_rows, dim] (fp32)
    x_norm_fp16 = torch.empty((total_rows, dim),
                               dtype=torch.float16,
                               device=device)      # destination

    BLOCK_M = 128    # rows per program
    BLOCK_D = 32     # columns per inner loop

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left and right paths
    # ------------------------------------------------------------------
    # Concatenate projection & gate weights (out_features, in_features)
    left_concat_w  = torch.cat([weights["left_proj.weight"],
                                weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_concat_w = torch.cat([weights["right_proj.weight"],
                                weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_w      = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w    = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b    = weights["to_out_norm.bias"]            # fp32
    to_out_w         = weights["to_out.weight"].to(torch.float16)

    # Flatten normalized tensor for linear layers (still fp16)
    x_norm_flat = x_norm_fp16.view(total_rows, dim)          # [total_rows, dim]

    # ----- left path ---------------------------------------------------
    left_concat = F.linear(x_norm_flat, left_concat_w)       # [total_rows, 2*hidden_dim]
    left_proj   = left_concat[:, :hidden_dim]                # projection part

    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_h                       # mask before gating

    left_gate   = torch.sigmoid(left_concat[:, hidden_dim:]) # gate part
    left_h      = left_proj * left_gate                     # [total_rows, hidden_dim]

    # ----- right path --------------------------------------------------
    right_concat = F.linear(x_norm_flat, right_concat_w)    # [total_rows, 2*hidden_dim]
    right_proj   = right_concat[:, :hidden_dim]

    if not nomask:
        # reuse mask_h (same mask for left & right)
        right_proj = right_proj * mask_h

    right_gate   = torch.sigmoid(right_concat[:, hidden_dim:])
    right_h      = right_proj * right_gate                  # [total_rows, hidden_dim]

    # ----- output gate -------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))   # [total_rows, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM) – einsum version
    # ------------------------------------------------------------------
    left_4d  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_4d = right_h.view(B, N, N, hidden_dim)

    # einsum performs the contraction Σ_k left[...,i,k,h] * right[...,j,k,h]
    #   → result shape [B, N, N, H]  (still fp16)
    out_4d = torch.einsum('b i k h, b j k h -> b i j h', left_4d, right_4d)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out_4d.view(total_rows, hidden_dim)          # [total_rows, hidden_dim]
    out_norm_fp16 = torch.empty_like(out_flat)              # fp16 destination

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        to_out_norm_w,
        to_out_norm_b,
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm_gate = (out_norm_fp16.view(B, N, N, hidden_dim) *
                     out_gate.view(B, N, N, hidden_dim))   # element‑wise fp16

    # final linear (hidden_dim → dim) – fp16 weight, fp16 activation
    final_fp16 = F.linear(out_norm_gate.view(total_rows, hidden_dim),
                          to_out_w)                           # [total_rows, dim]

    # Cast to fp32 for the API contract
    final_fp32 = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final_fp32
