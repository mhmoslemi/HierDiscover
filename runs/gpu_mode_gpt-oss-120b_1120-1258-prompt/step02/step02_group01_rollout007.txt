```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def rowwise_layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute sum and sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)  # promotes
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])
```

```python
def custom_kernel(data):
    """
    Forward pass of “outgoing” TriMul.
    Optimizations:
    - First LayerNorm implemented in Triton (row‑wise, FP32).
    - Linear projections and three gate projections are fused into
      two matrix‑multiplications (one for left/right, one for all gates).
    - Optional mask applied in FP16.
    - Batched matrix multiplication (left @ rightᵀ) performed with torch.bmm
      on half‑precision tensors.
    - Second LayerNorm (over hidden_dim) implemented in Triton (FP16).
    - Final gated projection back to `dim`.
    All intermediate tensors are half‑precision; the final output is float32.
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    # first layer norm
    B, N, _, D = input_tensor.shape
    assert D == dim, "dim mismatch"

    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty_like(x_flat)
    # launch Triton kernel
    BLOCK_M = 256  # can be tuned
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)

    # cast to half
    x_norm_h = x_norm.to(torch.float16)

    # Flatten for linear ops
    R = total_rows
    x_norm_h_flat = x_norm_h.view(R, dim)

    # combine projection weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0).to(torch.float16)  # [2*H, D]

    proj_out = F.linear(x_norm_h_flat, proj_weight)  # [R, 2*H]
    left_proj_h = proj_out[:, :hidden_dim]
    right_proj_h = proj_out[:, hidden_dim:]

    # mask (optional)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(R, 1)  # broadcast over hidden
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # combine gate weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    gate_weight = torch.cat([left_gate_weight, right_gate_weight, out_gate_weight], dim=0).to(torch.float16)  # [3*H, D]

    gate_out = F.linear(x_norm_h_flat, gate_weight)  # [R, 3*H]
    left_gate_h = torch.sigmoid(gate_out[:, :hidden_dim])
    right_gate_h = torch.sigmoid(gate_out[:, hidden_dim:2*hidden_dim])
    out_gate_h = torch.sigmoid(gate_out[:, 2*hidden_dim:])

    # apply gates
    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # reshape to 4-D
    left_h = left_h.view(B, N, N, hidden_dim)
    right_h = right_h.view(B, N, N, hidden_dim)
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)

    # TriMul: batched GEMM
    # permute to [B, H, N, N]
    left_mat = left_h.permute(0, 3, 1, 2).contiguous()
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # note transpose k<->j

    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)
    right_mat = right_mat.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # [BH, N, N] half

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Second LayerNorm (over hidden_dim) using Triton kernel
    total_rows2 = B * N * N
    out_flat = out.view(total_rows2, hidden_dim).contiguous()
    out_norm_flat = torch.empty_like(out_flat)

    # use half-precision weights/bias (cast)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    # launch Triton
    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows2, BLOCK_M2),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        total_rows2,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # apply out_gate
    out_norm = out_norm * out_gate_h

    # final linear projection back to dim
    out_norm_flat2 = out_norm.view(total_rows2, hidden_dim)
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final_half = F.linear(out_norm_flat2, w_to_out)  # [R, dim]
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (`dim`).
   Implemented in Triton (float‑32 accumulator, fp32 output).  This satisfies
   the requirement that at least one operation is a custom kernel.

2️⃣ Cast the normalized tensor to FP16 – all remaining heavy work is done
   in half‑precision to profit from H100 Tensor‑Cores.

3️⃣ **Fused linear projections**:
   • Left‑ and right‑projections are concatenated into one weight matrix
     of shape `[2·hidden_dim, dim]` → one `F.linear` call.
   • The three gate projections (left‑gate, right‑gate, out‑gate) are
     concatenated into `[3·hidden_dim, dim]` → one more `F.linear` call.
   This halves the number of large GEMM launches and reduces memory traffic.

4️⃣ Optional mask (`mask` of shape `[bs, N, N]`) is applied in FP16 after the
   projections – broadcasting over the hidden dimension.

5️⃣ Gated projections:
   `left  = left_proj  * sigmoid(left_gate)`
   `right = right_proj * sigmoid(right_gate)`

6️⃣ Triangular multiplicative update (the `einsum` in the reference):
   For every hidden channel `h` we need `L_h @ R_hᵀ` where
   `L_h = left[..., :, :, h]` and `R_h = right[..., :, :, h]`.
   By permuting to `[B, H, N, N]` and reshaping we obtain a tensor of shape
   `[B·H, N, N]` and perform a **batched matrix multiplication** with
   `torch.bmm`.  This maps directly to a Tensor‑Core GEMM on the H100.

7️⃣ Row‑wise LayerNorm over the hidden dimension (`hidden_dim`).  The same
   Triton kernel is reused (input is FP16, output is FP32).  The FP32 result
   is cast back to FP16 for the next step.

8️⃣ Apply the output gate (element‑wise), then a final linear projection
   `hidden_dim → dim` (still in FP16).  The result is cast to FP32 before
   returning.

All intermediate tensors are kept in FP16, only the two LayerNorms produce
FP32 buffers.  This design lowers kernel‑launch overhead, halves the number of
large linear GEMMs, and leverages the fast batched GEMM for the O(N³) update.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,                          # const *    [total_rows, dim]   (any dtype)
    y_ptr,                          # mut   *    [total_rows, dim]   (float32 output)
    weight_ptr,                     # const *    [dim]               (float32)
    bias_ptr,                       # const *    [dim]               (float32)
    total_rows,                     # int (runtime)
    dim: tl.constexpr,              # compile‑time constant
    eps: tl.constexpr,              # compile‑time constant
    BLOCK_M: tl.constexpr,          # rows per program
    BLOCK_D: tl.constexpr,          # cols per program
):
    """
    Row‑wise LayerNorm (mean/variance in fp32).  Input may be fp32 or fp16,
    output is always fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: accumulate sum and sum‑of‑squares for mean/var
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        # weight/bias are fp32, promote to fp32 already
        out = norm * w[None, :] + b[None, :]          # fp32

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First (row‑wise) LayerNorm over `dim` – Triton
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # flatten the first three axes
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty_like(x_flat)     # fp32 output

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],               # fp32
        weights["norm.bias"],                 # fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    x_norm = x_norm_flat.view(B, N, N, dim)    # [B,N,N,dim] fp32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy ops in half‑precision
    # -----------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)        # fp16
    R = total_rows
    x_norm_h_flat = x_norm_h.view(R, dim)      # [R, dim] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fused linear projections (left/right) + gating
    # -----------------------------------------------------------------
    # ---- 3a. Projection (left & right) ---------------------------------
    left_proj_w  = weights["left_proj.weight"]   # [hidden_dim, dim] fp32
    right_proj_w = weights["right_proj.weight"]  # same shape
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0).to(torch.float16)
    # One large GEMM: (R, dim) @ (dim, 2*hidden_dim)ᵀ
    proj_out = F.linear(x_norm_h_flat, proj_weight)   # [R, 2*hidden_dim] fp16
    left_proj_h  = proj_out[:, :hidden_dim]            # [R, hidden_dim]
    right_proj_h = proj_out[:, hidden_dim:]            # [R, hidden_dim]

    # ---- 3b. Optional mask (broadcast over hidden_dim) -----------------
    if not nomask:
        # mask shape: [B, N, N] → flatten → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(R, 1)   # fp16
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ---- 3c. Gates (left_gate, right_gate, out_gate) -----------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)

    gate_out = F.linear(x_norm_h_flat, gate_weight)   # [R, 3*hidden_dim] fp16
    left_gate_h  = torch.sigmoid(gate_out[:, :hidden_dim])
    right_gate_h = torch.sigmoid(gate_out[:, hidden_dim:2*hidden_dim])
    out_gate_h   = torch.sigmoid(gate_out[:, 2*hidden_dim:])

    # ---- 3d. Apply element‑wise gates ---------------------------------
    left_h  = left_proj_h  * left_gate_h   # [R, hidden_dim] fp16
    right_h = right_proj_h * right_gate_h  # [R, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 4️⃣ Reshape for the triangular multiplicative update
    # -----------------------------------------------------------------
    left_h  = left_h.view(B, N, N, hidden_dim)    # [B,N,N,H] fp16
    right_h = right_h.view(B, N, N, hidden_dim)   # same
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)  # same shape

    # Move hidden dimension to the batch axis and transpose the second matrix
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]  (j ↔ k)

    BH = B * hidden_dim
    left_mat  = left_mat.view(BH, N, N)    # [B*H, N, N] fp16
    right_mat = right_mat.view(BH, N, N)   # same

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM  (L @ Rᵀ) – O(N³) per head
    # -----------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] fp16

    # Restore the original 4‑D layout: [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                            # fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (still fp16 input)
    # -----------------------------------------------------------------
    total_rows2 = total_rows                       # B*N*N
    out_flat = out.view(total_rows2, hidden_dim).contiguous()
    out_norm_fp32 = torch.empty_like(out_flat, dtype=torch.float32)

    grid2 = (triton.cdiv(total_rows2, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,                                 # fp16 input
        out_norm_fp32,                            # fp32 output
        weights["to_out_norm.weight"],            # fp32 γ
        weights["to_out_norm.bias"],              # fp32 β
        total_rows2,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_fp32.to(torch.float16)      # back to fp16

    # -----------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                # element‑wise fp16
    out_norm_flat = out_norm.view(total_rows2, hidden_dim)

    w_to_out = weights["to_out.weight"].to(torch.float16)  # [dim, hidden_dim]
    final_half = F.linear(out_norm_flat, w_to_out)         # [R, dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
