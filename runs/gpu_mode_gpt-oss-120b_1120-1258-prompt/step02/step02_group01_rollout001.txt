def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1: first LN via Triton.
    total_rows = bs * N * N
    x_flat = x.view(total_rows, dim)
    x_norm_flat = torch.empty_like(x_flat)
    BLOCK_M = 128  # maybe 128 rows per program
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # cast to half
    x_norm_h = x_norm_flat.view(bs, N, N, dim).to(torch.float16)
    # flatten for matmul
    x_norm_h_flat = x_norm_h.view(total_rows, dim)

    # fuse left + left_gate
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    # shape (2*hidden_dim, dim)
    left_concat = F.linear(x_norm_h_flat, left_concat_w)  # shape (R, 2*H)
    left_proj_h = left_concat[:, :hidden_dim]
    left_gate_h = torch.sigmoid(left_concat[:, hidden_dim:])

    # fuse right + right_gate
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    right_concat = F.linear(x_norm_h_flat, right_concat_w)
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_h = torch.sigmoid(right_concat[:, hidden_dim:])

    # out gate separate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_h = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_w))

    # apply mask if needed
    if not config.get("nomask", True):
        mask_h = mask.to(torch.float16).view(total_rows, 1)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # apply gates
    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # reshape to (B,N,N,H)
    left_h = left_h.view(bs, N, N, hidden_dim)
    right_h = right_h.view(bs, N, N, hidden_dim)
    out_gate_h = out_gate_h.view(bs, N, N, hidden_dim)

    # TriMul (batched GEMM)
    # left_mat shape (B*H,N,N), same for right
    left_mat = left_h.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 2, 1).reshape(bs * hidden_dim, N, N)  # note swapped i/j
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second LN using Triton (could reuse kernel)
    # but we need LN over hidden dimension (dim = hidden_dim)
    total_rows_2 = bs * N * N
    out_flat = out.view(total_rows_2, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)
    BLOCK_M2 = 128
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows_2, BLOCK_M2),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_2,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    # cast to half? The kernel outputs float32, but we can cast to half for later ops
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim).to(torch.float16)

    # gating & projection
    out_norm = out_norm * out_gate_h
    # final projection to dim
    w_to_out = weights["to_out.weight"].to(torch.float16)
    out_norm_flat = out_norm.view(total_rows_2, hidden_dim)
    final_h = F.linear(out_norm_flat, w_to_out)  # shape (R, dim)
    final = final_h.view(bs, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fast forward pass.

Algorithm Overview
------------------
1. **First LayerNorm (row‑wise)** – implemented in Triton.
   The tensor is flattened to `[B·N·N, dim]` and normalized in FP32.
2. **Mixed‑precision projection & gating** – the three linear projections
   (`left`, `right`) and the three sigmoid gates are fused:
   * `left_proj + left_gate`  → one matmul of shape `[R, 2·H]`,
   * `right_proj + right_gate` → one matmul of shape `[R, 2·H]`,
   * `out_gate`               → one matmul of shape `[R, H]`.
   All weights are cast once to FP16 and the whole block runs in FP16.
3. **Optional mask** – applied (broadcast) only to the projected tensors.
4. **Triangular multiplicative update** – for each hidden channel the
   operation `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` is a
   batched matrix multiplication:
   ```
   left_mat  = left   .permute(0,3,1,2)   # (B,H,N,N)
   right_mat = right  .permute(0,3,2,1)   # (B,H,N,N)
   out_mat   = torch.bmm(left_mat.reshape(B*H,N,N),
                         right_mat.reshape(B*H,N,N))   # (B*H,N,N)
   ```
   No extra copy is performed – the `reshape` works on the non‑contiguous
   permuted view.
5. **Second LayerNorm (over `hidden_dim`)** – also done with the same
   Triton kernel.
6. **Final gated projection** – element‑wise gating followed by a
   linear projection back to `dim`.  The final result is cast to FP32.

Only the two LayerNorms are custom Triton kernels; all heavy linear
algebra is performed by highly tuned cuBLAS kernels (`torch.bmm`,
`torch.nn.functional.linear`).  The implementation stays fully on GPU,
uses FP16 for the bulk of the work and respects the required API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton Row‑wise LayerNorm (float32)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const float*   [total_rows, dim]
    y_ptr,            #   mut float*   [total_rows, dim]
    weight_ptr,       # const float*   [dim]
    bias_ptr,         # const float*   [dim]
    total_rows,       # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor (float32)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute sum & sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum‑sq

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, residing on CUDA).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Tensor dimensions
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # -----------------------------------------------------------------
    # 1) First LayerNorm over `dim` (float32, Triton)
    # -----------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)                           # output buffer

    # Tuned block sizes (good for all supported N)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Cast to FP16 for the bulk of the work
    x_norm = x_norm_flat.view(B, N, N, dim).to(torch.float16)      # [B,N,N,dim]
    R = total_rows
    x_norm_flat_h = x_norm.view(R, dim)                           # [R, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections + gating (fused, half‑precision)
    # -----------------------------------------------------------------
    # Fuse left_proj + left_gate
    w_left_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                                            # [2H, dim]
    left_concat = F.linear(x_norm_flat_h, w_left_concat)          # [R, 2H]
    left_proj_h = left_concat[:, :hidden_dim]                    # [R, H]
    left_gate_h = torch.sigmoid(left_concat[:, hidden_dim:])     # [R, H]

    # Fuse right_proj + right_gate
    w_right_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                                            # [2H, dim]
    right_concat = F.linear(x_norm_flat_h, w_right_concat)        # [R, 2H]
    right_proj_h = right_concat[:, :hidden_dim]                  # [R, H]
    right_gate_h = torch.sigmoid(right_concat[:, hidden_dim:])   # [R, H]

    # Out gate (stand‑alone)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)    # [H, dim]
    out_gate_h = torch.sigmoid(F.linear(x_norm_flat_h, w_out_gate))  # [R, H]

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dimension)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(R, 1)       # [R,1]
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Apply element‑wise gates
    left_h = left_proj_h * left_gate_h        # [R, H]
    right_h = right_proj_h * right_gate_h     # [R, H]

    # Restore 4‑D layout
    left_h = left_h.view(B, N, N, hidden_dim)          # [B,N,N,H]
    right_h = right_h.view(B, N, N, hidden_dim)        # [B,N,N,H]
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # Transform to (B, H, N, N) without extra copies
    left_mat = left_h.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H,N,N]
    right_mat = right_h.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N) # [B*H,N,N]

    # Batched matrix multiplication – leverages Tensor‑Core GEMM
    out_mat = torch.bmm(left_mat, right_mat)                              # [B*H,N,N]

    # Back to original layout: [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5) Second LayerNorm over hidden_dim (Triton, FP32)
    # -----------------------------------------------------------------
    total_rows2 = B * N * N
    out_flat = out.view(total_rows2, hidden_dim)               # [R, H]
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows2, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows2,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Cast back to half for remaining ops
    out_norm = out_norm_flat.view(B, N, N, hidden_dim).to(torch.float16)

    # -----------------------------------------------------------------
    # 6) Final gated projection to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                         # element‑wise gate
    out_norm_flat = out_norm.view(total_rows2, hidden_dim)   # [R, H]

    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_flat, w_to_out)          # [R, dim]

    # Cast to float32 for the API contract
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
