"""
TriMul outgoing kernel (AlphaFold3, Triton 3.3.1)

Implements the forward pass of the “outgoing” TriMul operation:

1. Layer‑norm over the input feature dimension (dim) – done in FP32.
2. Bias‑free linear projections → left, right (half‑precision).
3. Optional binary mask (broadcast on hidden dimension).
4. Three sigmoid gates (left, right, out) – still half‑precision.
5. Cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realized as a batched GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N}
   for each (batch, hidden) pair.  The heavy part is a Triton kernel
   that loads tiles in fp16, performs Tensor‑Core accelerated `tl.dot`,
   and accumulates in fp32.
6. Hidden‑dim Layer‑norm, element‑wise multiplication with the output gate,
   and a final bias‑free linear projection back to the original dimension.
All heavy arithmetic (steps 2‑5) runs in fp16; the final result is returned
in fp32.  The implementation avoids any extra `contiguous()` copies of the
large tensors – the kernel works directly on the strided tensors produced
by `permute`, which saves several hundred megabytes of memory traffic and
substantially speeds up the forward pass.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # fp16, shape (B, H, N, N) – rows = i, cols = k
    right_ptr,         # fp16, same shape – rows = i, cols = k (we read transposed)
    out_ptr,           # fp32, shape (B, H, N, N) – rows = i, cols = j
    N,                 # sequence length (int)
    stride_am, stride_ah, stride_ai, stride_ak,   # left strides: batch, hidden, i, k
    stride_bm, stride_bh, stride_bi, stride_bk,   # right strides: batch, hidden, i, k
    stride_cm, stride_ch, stride_ci, stride_cj,   # out strides: batch, hidden, i, j
    hidden_dim,                                   # hidden dimension (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]"""

    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid   = tl.program_id(2)          # batch*hidden index

    # decode batch and hidden indices
    batch_idx = pid // hidden_dim
    hidden_idx = pid % hidden_dim

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_am + hidden_idx * stride_ah
    right_base = right_ptr + batch_idx * stride_bm + hidden_idx * stride_bh
    out_base   = out_ptr   + batch_idx * stride_cm + hidden_idx * stride_ch

    # output pointer for the (i,j) tile
    out_ptrs = out_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k‑loop (reduction)
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  – fp16
        a_ptrs = left_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # fp16 load

        # right[j, k] transposed -> we need B[k, j]
        #   row index = k  (stride_bk), column index = j (stride_bi)
        b_ptrs = right_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # fp16 load

        acc += tl.dot(a, b)                # Tensor‑Core, acc in fp32

    # write tile
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  The input tuple must contain:
        - input_tensor : [B, N, N, dim]      (float32)
        - mask         : [B, N, N]           (float32, bool or None)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    # ------------------------------------------------------------------
    # unpack model weights (already on `device`)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the input feature dimension (dim) – FP32
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, dim]  float32

    # convert to half for the heavy ops
    x_h = x.half()

    # ------------------------------------------------------------------
    # cast projection / gate weights to half
    # ------------------------------------------------------------------
    left_proj_w_h  = left_proj_weight.half()
    right_proj_w_h = right_proj_weight.half()
    left_gate_w_h  = left_gate_weight.half()
    right_gate_w_h = right_gate_weight.half()
    out_gate_w_h   = out_gate_weight.half()

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, left_proj_w_h)   # [B,N,N,H]  fp16
    right = torch.nn.functional.linear(x_h, right_proj_w_h)  # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w_h))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w_h))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the cubic contraction
    #    out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    #    We keep tensors in the (B, H, N, N) layout without any copies.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)    # [B, H, N, N]  non‑contiguous (hidden stride = 1)
    right_perm = right.permute(0, 3, 1, 2)   # same layout

    B, N = input_tensor.shape[0], input_tensor.shape[1]

    # output tensor in the same layout as left/right (float32)
    out = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides for the kernel (in units of elements, not bytes)
    stride_am, stride_ah, stride_ai, stride_ak = left_perm.stride()
    stride_bm, stride_bh, stride_bi, stride_bk = right_perm.stride()
    stride_cm, stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # launch parameters – 128×128 tiles give good occupancy on H100.
    # BLOCK_K can be 64 or 128; 64 works well with fp16 Tensor‑Core tiles.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out,
        N,
        stride_am, stride_ah, stride_ai, stride_ak,
        stride_bm, stride_bh, stride_bi, stride_bk,
        stride_cm, stride_ch, stride_ci, stride_cj,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps ≈ 32 threads per warp → 256 threads per block
    )

    # ------------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, output gating and final linear projection
    # ------------------------------------------------------------------
    # out is currently [B, H, N, N] (fp32).  Bring hidden dim to last axis
    # for the LayerNorm + Linear that expect (..., hidden_dim).
    out = out.permute(0, 2, 3, 1)                         # [B, N, N, H] (non‑contiguous)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                     # fp32

    # broadcast output gate (shape [B, N, N, H])
    out_gate = out_gate.permute(0, 2, 3, 1)               # [B, N, N, H] (same layout)
    out = out * out_gate.to(out.dtype)

    # final linear back to original dim (bias‑free)
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] float32

    return output
