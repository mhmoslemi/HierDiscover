import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    # compute mean,var in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        norm = (x - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        # cast to fp16
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

Now `custom_kernel`:

```python
def custom_kernel(data):
    """
    custom Triton+torch implementation of the `outgoing` TriMul operation.
    The algorithm:
        1. Row‑wise LayerNorm on the input (fp32 -> fp16 output).
        2. Fuse the five linear layers (two projections + three gates) into a single
           GEMM; keep the result in fp16.
        3. Apply the optional pair‑wise mask.
        4. Apply sigmoid gates and element‑wise gating.
        5. Batched GEMM (B*H, N, N) implementing
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
        6. Second LayerNorm on the hidden dimension (fp16 → fp16).
        7. Final gating + linear projection back to `dim`; output cast to fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 0. Flatten input and run first LayerNorm (fp32->fp16)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    x = input_tensor.contiguous()  # [B,N,N,dim]
    # view as 2‑D for layernorm
    x_flat = x.view(total_rows, dim)                      # [total_rows, dim]
    # allocate fp16 buffer for normalized result
    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16)  # [total_rows, dim]

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 64 else (128 if dim <= 128 else 256)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # --------------------------------------------------------------
    # 1. Fused linear projections + gates (fp16 GEMM)
    # --------------------------------------------------------------
    # Convert weight matrices to fp16 once
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    # concatenate to single weight for fusion
    concat_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )  # shape (5*hidden_dim, dim)

    # single GEMM
    proj_all = F.linear(x_norm_fp16, concat_weight)  # fp16, shape (total_rows, 5*hidden_dim)

    # split
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # --------------------------------------------------------------
    # 2. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # --------------------------------------------------------------
    # 3. Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate          # [total_rows, H]
    right = right_proj_fp16 * right_gate       # [total_rows, H]

    # reshape to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 4. Triangular multiplicative update (batched GEMM)
    #    out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    # --------------------------------------------------------------
    # Bring hidden dimension to leading axis
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H, N, N]
    # right needs a transpose on the (i,k) axes to have (j,k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # [B*H, N, N]

    # batched matmul (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]

    # restore 4‑D shape
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------
    # 5. Second LayerNorm over hidden_dim (output remains fp16)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)           # [total_rows_h, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6. Final gating + linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16
    # final linear projection (H -> dim)
    to_out_w = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w)    # fp16, shape (total_rows_h, dim)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
custom_kernel: highly‑optimised “outgoing” TriMul (AlphaFold‑3).

Algorithm (all tensors stay on GPU):
1️⃣ Row‑wise LayerNorm on the 4‑D input (float32 → half output).
2️⃣ Fuse the five linear layers (two projections + three gates) into a single
   GEMM:   (B·N·N, dim) @ (dim, 5·hidden_dim)ᵀ  →  (B·N·N, 5·hidden_dim) (fp16).
3️⃣ Optional pairwise mask applied to the left/right projections.
4️⃣ Apply sigmoid‑gates and element‑wise gating (still fp16).
5️⃣ Batched GEMM for the triangular update
      out_h[i,j] = Σ_k left_h[i,k]·right_h[j,k]
   implemented as torch.bmm on tensors of shape (B·hidden_dim, N, N)
   (fp16 Tensor‑Core kernels).
6️⃣ Second LayerNorm over the hidden dimension (fp16 → fp16).
7️⃣ Final gate × linear projection back to `dim` and cast to float32.

Both LayerNorms are custom Triton kernels (float‑32 accumulation, fp16
output).  All heavy linear algebra uses fp16 Tensor‑Core kernels, while
intermediate tensors stay in fp16 to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (float32 accumulation, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const void*   [total_rows, dim]
    y_ptr,          # *mut   void*   [total_rows, dim]  (fp16)
    weight_ptr,     # *const float* [dim]               (fp32)
    bias_ptr,       # *const float* [dim]               (fp32)
    total_rows,     # integer (runtime)
    dim,            # integer (compile‑time)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    IS_INPUT_FP16: tl.constexpr,
):
    """Row‑wise LayerNorm.
    *x_ptr* points to either fp32 or fp16 data (controlled by *IS_INPUT_FP16*).
    The result is always written as fp16, the weight/bias are fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # first pass: mean & variance (float32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load with correct precision
        if IS_INPUT_FP16:
            vals = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)          # promote to fp32
        else:
            vals = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
                dtype=tl.float32,
            )
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # second pass: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        if IS_INPUT_FP16:
            vals = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)
        else:
            vals = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
                dtype=tl.float32,
            )
        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ (weight) and β (bias) – fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]             # fp32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),                          # cast to fp16
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton‑driven TriMul (outgoing) implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Un‑pack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # B·N·N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()              # [B, N, N, C] (fp32)
    x_flat = x.view(total_rows, dim)           # [total_rows, C]

    x_norm_fp16 = torch.empty(
        total_rows, dim, dtype=torch.float16, device=device
    )  # output buffer (fp16)

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 64 else (128 if dim <= 128 else 256)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        IS_INPUT_FP16=False,                     # input is fp32
    )
    # ------------------------------------------------------------------
    # 2) Fuse five linear layers (proj + three gates) in fp16
    # ------------------------------------------------------------------
    # Cast all projection / gate weights once to fp16
    w_left_proj   = weights["left_proj.weight"].half()
    w_right_proj  = weights["right_proj.weight"].half()
    w_left_gate   = weights["left_gate.weight"].half()
    w_right_gate  = weights["right_gate.weight"].half()
    w_out_gate    = weights["out_gate.weight"].half()

    concat_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5·H, C), fp16, contiguous

    # Single GEMM:  (total_rows, C) @ (C, 5·H)ᵀ  -> (total_rows, 5·H)
    proj_all = F.linear(x_norm_fp16, concat_weight)  # fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates and element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)        # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate                # [total_rows, H]
    right = right_proj_fp16 * right_gate             # [total_rows, H]

    # Reshape to 4‑D tensors – these are only views, no copies
    left = left.view(B, N, N, hidden_dim)            # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)          # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Bring H to the batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B·H, N, N)
    # right must be transposed on the two N axes: (j, k) → (k, j)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N) # (B·H, N, N)

    # Batched matrix multiplication (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)        # (B·H, N, N)

    # Back to 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension (fp16 → fp16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        IS_INPUT_FP16=True,                       # input is fp16
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Final gate and projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # still fp16

    # Linear projection H → dim (fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # fp16, shape (total_rows_h, dim)

    # Cast result to fp32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
