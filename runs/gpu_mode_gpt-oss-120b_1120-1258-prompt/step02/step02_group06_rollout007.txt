"""
TriMul outgoing kernel (AlphaFold3)

Forward pass of the “outgoing” TriMul operation:
  1. Layer‑norm over the input feature dimension (dim).
  2. Bias‑free linear projections → left, right (hidden_dim features).
  3. Optional pair‑wise mask.
  4. Three sigmoid gates (left, right, out) and element‑wise gating of left/right.
  5. Cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     realised as a batched GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N} for each
     (batch, hidden) pair.  The heavy part is executed with a Triton kernel
     that loads fp16 tiles, uses Tensor‑Core‑accelerated `tl.dot`,
     and accumulates in fp32.
  6. Layer‑norm over the hidden dimension, element‑wise multiplication
     with the output gate, and a final bias‑free linear projection back to
     the original dimension.

Only the cubic contraction (step 5) is performed in Triton; all other
operations use PyTorch (float32 for the two layer‑norms, fp16 for the
linear/gating work).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A,B : (M_total, N, N)   fp16
#   C   : (M_total, N, N)   fp32
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  matrix (M_total, N, N) – fp16
    b_ptr,               # right matrix (M_total, N, N) – fp16
    c_ptr,               # output   (M_total, N, N) – fp32
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A (M_total, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M_total, j, k) – we read as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C (M_total, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # tile identifiers
    # ---------------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along output‑row (i)
    pid_j = tl.program_id(1)   # tile along output‑col (j)
    pid_m = tl.program_id(2)   # index for (batch * hidden)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ---------------------------------------------------------------
    # coordinate vectors for the current tile
    # ---------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ---------------------------------------------------------------
    # output pointer for this tile
    # ---------------------------------------------------------------
    c_ptrs = c_ptr + pid_m * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # reduction over the third dimension k
    # ---------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k):
        k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k] – fp16
        a_ptrs = a_ptr + pid_m * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # B[k, j] = right[j, k] – we read transposed, fp16
        b_ptrs = b_ptr + pid_m * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Tensor‑core multiplication, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ---------------------------------------------------------------
    # Store the tile (fp32)
    # ---------------------------------------------------------------
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the input channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # Cast small weight matrices to fp16
    # ------------------------------------------------------------------
    left_proj_w_h = left_proj_weight.half()
    right_proj_w_h = right_proj_weight.half()
    left_gate_w_h = left_gate_weight.half()
    right_gate_w_h = right_gate_weight.half()
    out_gate_w_h = out_gate_weight.half()

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) → left, right
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_h, left_proj_w_h)    # [B, N, N, H]  fp16
    right = torch.nn.functional.linear(x_h, right_proj_w_h)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] and cast to fp16
        mask_exp = mask.to(dtype=x_h.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w_h))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w_h))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cubic contraction via a Triton batched GEMM
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    We treat each (batch, hidden) pair as an independent GEMM.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] fp16

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)   # [M_total, N, N] fp16
    right_flat = right_perm.view(M_total, N, N)  # [M_total, N, N] fp16

    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # Block sizes – 64×64 tiles give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (Python ints)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over j (second dim)
    stride_bk = right_flat.stride(2)   # stride over k (third dim)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )

    # Launch kernel – num_warps=8 works well on H100.
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, output gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32, still [B, N, N, H]

    out = out * out_gate.to(out.dtype)      # element‑wise gate (fp32)

    # final linear back to the original dimension (float32)
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
```
