"""
TriMul “outgoing” kernel (AlphaFold3) – fast forward implementation.

Algorithm
---------
1. Layer‑norm over the input feature dimension (dim) – FP32.
2. Fuse the five bias‑free linear projections (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single matmul:
      out_all = x @ W_cat.T
   where W_cat = [left_proj; left_gate; right_proj; right_gate; out_gate].
   The result is split into the five tensors (all FP16).
3. Optional pair‑wise mask (broadcast on the hidden dimension).
4. Apply sigmoid gates and element‑wise gating of *left*/*right*.
5. Batched GEMM C = A @ Bᵀ  (A = left, B = right) for every
   (batch, hidden) pair.  The heavy cubic contraction is executed by a
   Triton kernel that loads tiles in FP16, performs Tensor‑Core `tl.dot`,
   and accumulates in FP32.
6. Hidden‑dim layer‑norm, element‑wise multiplication with the *out*
   gate and the final bias‑free linear projection back to the original
   dimension.  The final result is FP32.

All large‑scale arithmetic is done in half‑precision to maximise
bandwidth/Tensor‑Core utilisation; only the LayerNorms and the final
linear use FP32.  The kernel avoids extra copies by keeping the
intermediate tensors contiguous only where it benefits memory access.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM  C = A @ Bᵀ    (A,B : M×N×N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                      # left  matrix  (M, N, N) – fp16
    b_ptr,                      # right matrix  (M, N, N) – fp16 (accessed transposed)
    c_ptr,                      # output matrix (M, N, N) – fp32
    N,                          # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A : (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B : (M, j, k) – we will read B[k, j]
    stride_cm, stride_ci, stride_cj,   # strides of C : (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_b = tl.program_id(2)          # batch * hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # row / column indices inside the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # pointers to the output tile C[i, j]
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k] – shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # B[k, j] – we read B transposed on‑the‑fly
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16

        # Tensor‑core accelerated dot, accumulate in fp32
        acc += tl.dot(a, b)

    # Store the result (fp32)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    O(N³·hidden) contraction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]      (float32)
            - mask         : [B, N, N]          (float32 / bool). May be ignored.
            - weights      : dict of model parameters (already on the target device)
            - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)       # True → ignore the mask
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack weight tensors (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the input feature dimension (dim) – stays FP32
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, dim]  float32

    # Cast to fp16 for the heavy ops
    x_h = x.half()                     # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2. Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    # Concatenate weight matrices (out_features, in_features) → (5*hidden, dim)
    W_cat = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    ).half()                           # (5*hidden, dim), fp16

    # Single matmul: (B·N·N, dim) @ (dim, 5*hidden)ᵀ  → (B, N, N, 5*hidden)
    all_proj = torch.nn.functional.linear(x_h, W_cat)      # fp16

    # Split the concatenated result
    h = hidden_dim
    left, left_gate, right, right_gate, out_gate = torch.split(
        all_proj, [h, h, h, h, h], dim=-1
    )                                   # each : [B, N, N, h]  fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Treat every (batch, hidden) pair as an independent GEMM.
    # ------------------------------------------------------------------
    # Permute to [B, H, N, N] so that the hidden dimension becomes part
    # of the batch axis for the kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]  fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]  fp16

    B, H, N, _ = left_perm.shape
    M_total = B * H                                    # = batch * hidden

    # View as (M_total, N, N)
    left_flat = left_perm.view(M_total, N, N)          # fp16, contiguous
    right_flat = right_perm.view(M_total, N, N)        # fp16, contiguous

    # Output buffer (FP32)
    out_flat = torch.empty(
        (M_total, N, N), dtype=torch.float32, device=left_flat.device
    )

    # --------------------------------------------------------------
    # Strides (Python integers, note the order: (M, i, k) for A)
    # --------------------------------------------------------------
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)   # batch*hidden stride
    stride_bj = right_flat.stride(1)   # stride over *j* (second N)
    stride_bk = right_flat.stride(2)   # stride over *k* (third N)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # --------------------------------------------------------------
    # Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]  (float32)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                   # [B, N, N, H]  fp32

    # ------------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                   # fp32

    # Broadcast the out‑gate (currently fp16) to fp32
    out = out * out_gate.to(out.dtype)

    # Final linear: (hidden → dim) – bias‑free
    output = torch.nn.functional.linear(out, to_out_weight)   # fp32

    return output
