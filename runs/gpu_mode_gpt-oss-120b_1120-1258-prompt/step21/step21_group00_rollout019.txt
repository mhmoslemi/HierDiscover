pid_n = tl.program_id(1)

row_start = pid_m * BLOCK_M
col_start = pid_n * BLOCK_N

mask_n = offs_n < dim

# Loop over hidden dim blocks
for k_start in range(0, hidden_dim, BLOCK_K):
    offs_k = k_start + tl.arange(0, BLOCK_K)
    mask_k = offs_k < hidden_dim

    # Load out and gate slices: shape [M, K], strides contiguous for hidden dim (K is inner)
    # base pointer: out_ptr + offs_m[:, None] * hidden_dim + offs_k[None,:]
    # Because out is flattened as (R, H) row-major: each row length H, hidden dimension contiguous.
    out_vals = tl.load(out_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

    gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)   # [M, K]

    # Load LayerNorm parameters for this block: gamma, beta (FP32)
    gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
    beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

    # compute normalized hidden values:

    # multiply by gate

                     mask=mask_k[:, None] & mask_n[None, :],
                     other=0.0)  # [K, Ntile] FP16

# Write output
dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    H = hidden_dim

    device = input_tensor.device

        right,

---------
1️⃣  Per‑pair LayerNorm over the feature dimension (float32 → float16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:   proj = x_norm @ W_fusedᵀ   → (R, 5·H)  (R = B·N·N).
3️⃣  Split the fused projection, apply the optional pair‑wise mask,
    compute sigmoid gates and store three tensors `left`, `right`,
    `out_gate`   – layout [B, H, N, N] (FP16).
4️⃣  Batched half‑precision GEMM (cuBLAS):
        out = Σₖ left[:, :, i, k] * right[:, :, j, k]   →   [B, H, N, N] (FP16).
5️⃣  Rearrange `out` and `out_gate` to [B, N, N, H] (contiguous hidden dim).
    Compute per‑row mean / variance with native torch reductions (float32).
6️⃣  Second LayerNorm + out‑gate multiplication + final projection are
    performed with pure torch operations (TensorCore GEMM + element‑wise
    FP16 arithmetic).  The final tensor is returned as float32.

Only the mask‑aware split/gate kernel is written in Triton; the rest uses
high‑performance torch kernels (cuBLAS, layer_norm, reductions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,                     # 1‑D mask of length R (fp16), ignored if HAS_MASK==0
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if present) over the hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # scalar per row
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # hidden stride inside the (B, H, N, N) tensors
    hidden_stride = N * N
    # base address for the first hidden channel of this (b,i,k) element
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              all tensors are float32.
            - config_dict  must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ----------------------------------------------------------------------
    # 1️⃣ per‑pair LayerNorm (float32 → float16)
    # ----------------------------------------------------------------------
    # Use the highly‑optimized torch implementation
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [B, N, N, dim] (FP16)

    x_flat = x_norm.view(R, dim)       # [R, dim] (FP16)

    # ----------------------------------------------------------------------
    # 2️⃣ fused linear projection (5·H outputs)
    # ----------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    proj = F.linear(x_flat, fused_w)   # (R, 5·H) FP16
    del fused_w, x_flat, x_norm

    # ----------------------------------------------------------------------
    # 3️⃣ split, optional mask, sigmoid gates → left / right / out_gate
    # ----------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ----------------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension (cuBLAS)
    # ----------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] (FP16)
    out = out.view(B, H, N, N)                            # [B, H, N, N] (FP16)

    # ----------------------------------------------------------------------
    # 5️⃣ make hidden dimension contiguous (layout [B, N, N, H])
    # ----------------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H] FP16
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H] FP16

    out_flat  = out.view(R, H)        # [R, H] FP16
    gate_flat = out_gate.view(R, H)   # [R, H] FP16

    # ----------------------------------------------------------------------
    # 6️⃣ row‑wise mean & rsqrt‑variance of `out` (float32)
    # ----------------------------------------------------------------------
    out_fp32 = out_flat.float()
    mu = out_fp32.mean(dim=1)                     # [R] float32
    var = out_fp32.var(dim=1, unbiased=False)    # [R] float32
    rstd = torch.rsqrt(var + eps)                # [R] float32

    # ----------------------------------------------------------------------
    # 7️⃣ second LayerNorm + out_gate + final projection (cuBLAS)
    # ----------------------------------------------------------------------
    # cast statistics to fp16 for the element‑wise part
    mu_fp16   = mu.to(torch.float16).unsqueeze(-1)      # [R,1]
    rstd_fp16 = rstd.to(torch.float16).unsqueeze(-1)   # [R,1]

    # (out - μ) * rstd
    norm = (out_flat - mu_fp16) * rstd_fp16            # FP16, shape [R,H]

    # to_out_norm (γ,β) – broadcast over rows
    norm = norm * w["to_out_norm.weight"].to(torch.float16) \
               + w["to_out_norm.bias"].to(torch.float16)

    # multiply by out‑gate
    norm = norm * gate_flat

    # final linear projection (H → dim)
    weight_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = F.linear(norm, weight_out_fp16)            # [R, dim] FP16

    # reshape back to [B, N, N, dim] and cast to FP32 as required by the API
    final = final_fp16.float().view(B, N, N, dim)

    return final
