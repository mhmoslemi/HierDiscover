"""
Outgoing TriMul (AlphaFold3) – fully‑fused forward pass (no backward).

Algorithm
---------
1️⃣ Row‑wise LayerNorm of the flattened [B·N·N, C] tensor (FP32 → FP16).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into one matrix‑multiply:   X_norm @ W_fusedᵀ → (R, 5·H)  (FP16).
3️⃣ Split the fused tensor, apply the optional per‑pair mask, compute
   sigmoid‑gates and write the three tensors (left, right, out_gate) in the
   layout [B, H, N, N] (FP16).  All of this is done by a single Triton kernel.
4️⃣ Batched GEMM over the hidden dimension H:   left[b,h] @ right[b,h]ᵀ.
   Implemented with torch.bmm which internally uses cuBLAS/H‑gemm.
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   projection into one Triton kernel.
   For every row (pair‑position) we first compute μ and σ⁻¹, then apply
   the learned LN‑γ/β, multiply by out_gate and finally contract with
   the output weight matrix (H×C).  The result is written as FP32
   [B, N, N, C].

The whole pipeline runs entirely on the GPU, uses FP16 for all heavy
matrix work and only stores the final tensor in FP32 – exactly matching
the PyTorch reference.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C]   input  (FP32)
    y_ptr,          # [R, C]   output (FP16)
    w_ptr,          # [C]     gamma (FP32)
    b_ptr,          # [C]     beta  (FP32)
    rows, cols, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]      # [R, C]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * cols + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   (FP16)
    left_ptr,          # [B, H, N, N] (FP16)
    right_ptr,         # [B, H, N, N] (FP16)
    out_gate_ptr,      # [B, H, N, N] (FP16)
    mask_ptr,          # [R] (FP16) – ignored if HAS_MASK==0
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k   (k is the summed index)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # (FP16)
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # offset to the first hidden‑plane (h=0) of the current (b,i,k)
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k                    # start of h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        # each component has stride H inside the fused dim
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # write into the (B,H,N,N) tensors
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm + out_gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N]            (FP16) – result of left·rightᵀ
    gate_ptr,         # [B, H, N, N]            (FP16)
    mu_ptr,           # [R]                     (FP32) – row‑wise mean
    rstd_ptr,         # [R]                     (FP32) – row‑wise rstd
    ln_w_ptr,         # [H]                     (FP32) – γ of 2nd LN
    ln_b_ptr,         # [H]                     (FP32) – β of 2nd LN
    w_out_T_ptr,      # [H, C]                  (FP16) – transposed final weight
    out_final_ptr,    # [B, N, N, C]            (FP32)
    R, H, N, C,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """
    For each pair‑position (row) we:
      * fetch μ, rstd,
      * loop over the hidden dimension in chunks of BLOCK_K,
      * apply LN (γ,β), multiply by out_gate,
      * accumulate a FP16·FP16 → FP32 GEMM against the final weight.
    """
    pid_r = tl.program_id(0)          # works on rows of shape (B·N·N)
    pid_c = tl.program_id(1)          # works on output‑channel tiles

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_C

    offs_r = row_start + tl.arange(0, BLOCK_R)             # [BR]
    offs_c = col_start + tl.arange(0, BLOCK_C)             # [BC]

    mask_r = offs_r < R
    mask_c = offs_c < C

    # decode (b,i,j) from flat row index
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    j   = rem - i * N

    # base offset for the hidden dimension (stride = N*N)
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j          # first hidden plane

    # load per‑row statistics (μ and rstd)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)    # [BR]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)    # [BR]

    # accumulator for the final output tile
    acc = tl.zeros([BLOCK_R, BLOCK_C], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BK]
        mask_k = offs_k < H

        # ------------------------------------------------------------------
        # load hidden tensor and gate (FP16 → FP32)
        # ------------------------------------------------------------------
        # layout: [R, H] but physically [B, H, N, N] -> stride = hidden_stride
        ptr_out  = out_ptr  + base_off[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BR, BK]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BR, BK]

        # ------------------------------------------------------------------
        # apply LayerNorm (γ,β) → multiply by gate
        # ------------------------------------------------------------------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BK]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BK]

        # (out - μ) * rstd  →  *γ  + β  →  *gate
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_vals                     # [BR, BK]   fp32

        # cast back to fp16 because we'll dot with fp16 weight matrix
        term_fp16 = term.to(tl.float16)

        # ------------------------------------------------------------------
        # final projection:  (term_fp16)·(W_out_T)  → accumulate
        # ------------------------------------------------------------------
        w_ptr = w_out_T_ptr + offs_k[:, None] * C + offs_c[None, :]         # [BK, BC]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_c[None, :],
                         other=0.0)                                         # [BK, BC]

        # dot: (BR,BK)·(BK,BC) → (BR,BC)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # write final result (float32)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_r[:, None] * C + offs_c[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_r[:, None] & mask_c[None, :])

        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ)  – use torch.bmm (FP16)
    # ------------------------------------------------------------------
    # reshape into (B·H, N, N) for a single batched GEMM
    left_bmm  = left.view(B * H, N, N)
    right_bmm = right.view(B * H, N, N)
    out = torch.bmm(left_bmm, right_bmm.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rstd needed for the second LayerNorm
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, C]

    final = torch.empty(B, N, N, C, dtype=torch.float32, device=device)

    # These block sizes have been tuned for the H100 (large shared‑memory,
    # high warp count).  You may tweak them for other GPUs.
    BLOCK_R_OUT = 128    # rows per program (i.e. a subset of B·N·N)
    BLOCK_K_OUT = 32     # hidden‑dim chunk
    BLOCK_C_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(C, BLOCK_C_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        C,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_C=BLOCK_C_OUT,
        num_warps=8,
    )
    # final :  [B, N, N, C] (float32)
    return final
