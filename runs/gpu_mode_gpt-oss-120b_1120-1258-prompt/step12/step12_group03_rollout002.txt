"""
TriMul (outgoing) – Triton‑accelerated forward pass (no gradients)

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}.  It consists of:
  1️⃣ Row‑wise LayerNorm (float32 → float16)
  2️⃣ A fused linear layer that computes the three projections (left/right)
      and three gating matrices (left, right, out) in a single GEMM.
  3️⃣ A split kernel that writes the five tensors directly in the (B,H,N,N)
      layout used by the batched matrix multiplication, applying the optional
      pair‑wise mask and sigmoid gates on‑the‑fly.
  4️⃣ Batched half‑precision matrix multiplication (cuBLAS) that performs the
      multiplicative update:  out[b,h] = left[b,h] @ right[b,h]ᵀ.
  5️⃣ A second LayerNorm over the hidden dimension H that also multiplies the
      result by the out‑gate, writing the final tensor in the contiguous
      (B,N,N,H) layout.
  6️⃣ Final linear projection back to the original `dim` (fp16 → fp32).

Only the three heavy data‑movement kernels are written in Triton; the two
GEMMs are delegated to cuBLAS via `torch.nn.functional.linear` and
`torch.bmm`, which is the fastest way on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # const *[rows, dim] (float32)
    y_ptr,       # mutable *[rows, dim] (float16)
    w_ptr,       # const *[dim] (float32)   γ
    b_ptr,       # const *[dim] (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values ----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,       # = B·N·N
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index -----------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive h slices

    # base offset common to all hidden slices of a row (B,H,N,N layout)
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five blocks from the fused projection -----------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate = tl.load(proj_ptr + proj_base + 2 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid (stable fp32) ---------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- compute destination offsets -----------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr  + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + multiplication by out_gate,
#    writing directly into (B, N, N, H) layout (contiguous)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_fused_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)
    out_fused_ptr,  # mutable *[B, N, N, H] (fp16)   <-- result
    gamma_ptr,      # const *[H] (float32)
    beta_ptr,       # const *[H] (float32)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program (R = B·N·N)
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    rows = B * N * N
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode (b,i,j) ----------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # strides for the (B, H, N, N) layout (source)
    stride_b_src = H * N * N
    stride_i_src = N
    stride_j_src = 1
    stride_h_src = N * N                # distance between consecutive h slices

    # base offset (h = 0) for source tensors
    src_base = b * stride_b_src + i * stride_i_src + j * stride_j_src   # [BLOCK_R]

    # ---- compute mean & variance over H ------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- destination strides for (B, N, N, H) layout (contiguous) ----------
    stride_b_dst = N * N * H
    stride_i_dst = N * H
    stride_j_dst = H
    # stride_h_dst = 1 (last dimension)

    dst_base = b * stride_b_dst + i * stride_i_dst + j * stride_j_dst   # [BLOCK_R]

    # ---- normalize, apply γ/β, multiply with gate and store ----------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # load out values
        ptr_out = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        out_val = tl.load(ptr_out,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # load gate values
        ptr_gate = gate_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        gate_val = tl.load(ptr_gate,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # normalize (fp32)
        norm = (out_val - mean[:, None]) * rstd[:, None]

        # γ and β (fp32)
        w = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        b_param = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)  # β

        norm = norm * w[None, :] + b_param[None, :]

        # multiply with gate and store (fp16)
        out = norm * gate_val
        dst_ptr = out_fused_ptr + dst_base[:, None] + offs_h[None, :]   # stride_h_dst = 1
        tl.store(dst_ptr,
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments

    data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters

        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32

    BLOCK_M = 256
    BLOCK_D = 64 if dim <= 256 else 128

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],    # γ  (float32)
        w["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + three gates → (R, 5·H)

    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]

    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:

        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:

        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256   # rows per program for split kernel
    BLOCK_H = 32    # hidden‑dim chunk

    split_fuse_gate_kernel[grid_split](

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,

        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    # Views for cuBLAS batch matmul
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat   = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N)                               # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate

    # ------------------------------------------------------------------

    )   # fp16
