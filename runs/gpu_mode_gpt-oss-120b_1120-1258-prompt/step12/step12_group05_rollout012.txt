@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    base_addr = b * (H * nn) + i * N + k

    # mask factor
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # broadcast hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load fused projection slices
        proj_idx = offs_r[:, None] * (5 * H) + offs_h[None, :]
        left_raw   = tl.load(proj_ptr + proj_idx, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw  = tl.load(proj_ptr + proj_idx + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_idx + 2*H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_idx + 3*H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_idx + 4*H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))

        left_val = left_raw * mask_val * left_gate.to(tl.float16)
        right_val = right_raw * mask_val * right_gate.to(tl.float16)
        out_gate_val = out_gate_raw * mask_val * out_gate_val.to(tl.float16)

        dst_offset = base_addr[:, None] + offs_h[None, :] * nn

        tl.store(left_ptr + dst_offset, left_val, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right_val, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate_val, mask=mask_r[:, None] & mask_h[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # config = {"dim":..., "hidden_dim":..., "nomask": ...}
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    B,N,_,_ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1. First layernorm using torch.nn.functional
    x_flat = input_tensor.view(R, dim)
    x_norm_fp32 = F.layer_norm(x_flat, (dim,), weights["norm.weight"], weights["norm.bias"], eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # 2. fused projection
    fused_w = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16) # shape (5*H, dim)
    proj = F.linear(x_norm, fused_w) # shape (R, 5*H) half

    # 3. split + apply mask + gates via Triton
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=input_tensor.device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=input_tensor.device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=input_tensor.device)

    if config.get("nomask", True) or mask_tensor is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=input_tensor.device)
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask_tensor.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    # Launch kernel
    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # 4. Batched GEMM (torch)
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2)) # half
    out = out_mat.view(B, H, N, N) # half

    del left, right

    # 5. Second LayerNorm and out_gate multiplication
    # reshape for layernorm
    out_flat = out.permute(0,2,3,1).contiguous().view(R, H)  # shape [R, H] half
    out_ln_fp32 = F.layer_norm(out_flat, (H,), weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps=eps)
    out_ln = out_ln_fp32.to(torch.float16)
    # gate flatten
    out_gate_flat = out_gate.permute(0,2,3,1).contiguous().view(R, H)
    out_fused = out_ln * out_gate_flat

    del out, out_gate

    # 6. Final linear projection
    final_half = F.linear(out_fused, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
custom_kernel – outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (B·N·N × dim) using
   :func:`torch.nn.functional.layer_norm`.  The result is cast to fp16.

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single half‑precision GEMM:
        X_norm @ W_fusedᵀ   →   (R, 5·H)   (R = B·N·N)

3️⃣ Custom Triton kernel ``split_fuse_gate_kernel``:
   - decodes the flat row index (b,i,k) → writes directly into the
     (B, H, N, N) layout required by the batched GEMM;
   - applies the optional pair‑wise mask,
   - evaluates the three sigmoid gates,
   - stores ``left``, ``right`` and ``out_gate`` (all fp16).

4️⃣ Batched GEMM (cuBLAS/Tensor‑cores) over the hidden dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
   Implemented with ``torch.bmm`` on tensors of shape (B·H, N, N).

5️⃣ Second LayerNorm (over the hidden dimension H) using
   :func:`torch.nn.functional.layer_norm`.  The tensor is viewed as
   (B·N·N, H) by permuting to (B, N, N, H).  After normalization we
   multiply element‑wise by the already‑sigmoided ``out_gate``.

6️⃣ Final linear projection back to ``dim`` (fp16 → fp32).

Only step 3 is implemented in Triton; all heavy matrix‑multiplications and
layer‑norms use highly‑optimized PyTorch/cuBLAS kernels.  This reduces
kernel‑launch overhead while keeping the computation fully fused and
numerically stable.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask and sigmoid gates,
#               write directly into (B, H, N, N) layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) – ignored if HAS_MASK=0
    R, H, N,           # program constants
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Decode (b,i,k) from the flat row index.
    # ------------------------------------------------------------------
    nn = N * N                         # N²  (stride for hidden dim)
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # base address for the first hidden slice (h = 0)
    base_addr = b * (H * nn) + i * N + k          # [BLOCK_R]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]          # shape (BLOCK_R, 1) for broadcasting
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load 5 blocks from the fused projection ------------------
        # proj layout: [R, 5*H]  -> column stride = 1
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ---- sigmoid in fp32 for numerical stability -------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ---------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        out_g = out_gate_raw * mask_val * out_gate

        # ---- compute destination offsets (stride = N²) ------------------
        # dst = base_addr + h * N²
        dst_offset = base_addr[:, None] + offs_h[None, :] * nn

        # ---- store -----------------------------------------------------------------
        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 out_g,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (use highly‑optimized PyTorch kernel)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim)                               # [R, dim]
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        w["norm.weight"],
        w["norm.bias"],
        eps=eps,
    )                                                      # still fp32
    x_norm = x_norm_fp32.to(torch.float16)                # cast to fp16 for GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (proj = X_norm @ W_fusedᵀ)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                       # [R, 5·H]  (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (custom Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)     # [R]
        has_mask = 1
        mask_ptr = mask_fp16

    # launch Triton kernel
    BLOCK_R = 256   # rows per program (tuned for H100)
    BLOCK_H = 64    # hidden dimension chunk
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS/Tensor‑Core)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, [B·H, N, N]
    out = out_mat.view(B, H, N, N)                         # (B, H, N, N)

    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension H) + out‑gate multiplication
    # --------------------------------------------------------------
    # Bring hidden dimension to the last axis: (B, N, N, H)
    out_perm = out.permute(0, 2, 3, 1)                # shape (B, N, N, H)
    out_flat = out_perm.reshape(R, H)                # [R, H]  (fp16)

    # LayerNorm in fp32, then cast back to fp16
    out_ln_fp32 = F.layer_norm(
        out_flat,
        (H,),
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        eps=eps,
    )                                                # fp32
    out_ln = out_ln_fp32.to(torch.float16)           # fp16

    # gate tensor – same layout as out_flat
    gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)   # fp16

    # Element‑wise multiplication (already‑sigmoided)
    out_fused = out_ln * gate_flat                     # fp16, shape [R, H]

    del out, out_gate, out_ln, gate_flat

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),        # [dim, H]
    )                                                # fp16, [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
