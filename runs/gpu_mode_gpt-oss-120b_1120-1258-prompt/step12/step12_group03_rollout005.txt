"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The operator processes a 4‑D tensor x ∈ ℝ^{B×N×N×C} (C = dim) as follows:

1️⃣ **Row‑wise LayerNorm** on the last dimension (fp32 → fp16).  
   This normalises each (b,i,k) row of the input and writes the result
   as half‑precision.

2️⃣ **Fused linear projection** computing the five maps
   (left_proj, right_proj, left_gate, right_gate, out_gate) in a single
   GEMM:   y = x_norm @ W_fusedᵀ   with W_fused ∈ ℝ^{5·H×C}.
   The result y has shape (B·N·N, 5·H) and is kept in fp16.

3️⃣ **Split & gate** – a Triton kernel extracts the five blocks from y,
   applies the optional pair‑wise mask, evaluates the sigmoid gates
   (stable fp32 → fp16), and writes the tensors
   *left*, *right* and *out_gate* directly in the (B, H, N, N) layout
   required for the batched GEMM.

4️⃣ **Multiplicative update** – a batched half‑precision matrix‑multiply
   (cuBLAS) computes
        out[b,h] = left[b,h] @ right[b,h]ᵀ
   yielding a tensor of shape (B, H, N, N).

5️⃣ **Second LayerNorm + gating** – a fused Triton kernel normalises each
   (b,i,j) vector across the hidden dimension H, applies the learned
   γ/β (float32) and multiplies element‑wise with *out_gate*.
   The result is written directly into a contiguous buffer with layout
   (B, N, N, H).

6️⃣ **Final linear projection** – a half‑precision linear layer maps the
   hidden dimension back to the original feature dimension C and the
   output is cast to float32.

Only the two LayerNorms and the split‑gate step are custom Triton kernels;
all heavy matrix multiplications are delegated to cuBLAS (torch.bmm /
torch.nn.functional.linear) for maximum throughput on H100.

The implementation works for both masked and un‑masked inputs, supports
batch sizes 1‑2, sequence lengths up to 1024 and feature dimensions 128‑384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *[rows, dim] (float32)
    y_ptr,            # mutable *[rows, dim] (float16)
    gamma_ptr,        # const   *[dim]      (float32)
    beta_ptr,         # const   *[dim]      (float32)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- mean & variance (fp32) -----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- normalise & apply γ,β (fp32→fp16) -----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * gamma[None, :] + beta[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly into (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (unused if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # -------- decode (b,i,k) from flat row index -------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # -------- broadcast mask (if any) ------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                      # distance between two H‑slices

    # base offset for the (B, H, N, N) layout (h = 0)
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load 5 blocks from the fused projection --------------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid in fp32 (numerically stable) ----------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ---------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- destination offsets -----------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

@triton.jit
def layernorm_mul_fused_kernel(

    N: tl.int32,
    H: tl.int32,

    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):

    rows = B * N * N
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------- decode (b,i,j) ------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    stride_b_src = H * N * N
    stride_i_src = N
    stride_j_src = 1
    stride_h_src = N * N

    src_base = b * stride_b_src + i * stride_i_src + j * stride_j_src   # [BLOCK_R]

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # destination strides for (B, N, N, H) – contiguous
    stride_b_dst = N * N * H
    stride_i_dst = N * H
    stride_j_dst = H
    dst_base = b * stride_b_dst + i * stride_i_dst + j * stride_j_dst   # [BLOCK_R]

    # --------------- normalise, apply γ/β, multiply gate ---------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]

        out_val = tl.load(src_ptr_h,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        gate_ptr_h = gate_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        gate_val = tl.load(gate_ptr_h,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # normalise
        norm = (out_val - mean[:, None]) * rstd[:, None]          # fp32

        # γ / β (fp32)
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)

        norm = norm * gamma[None, :] + beta[None, :]               # fp32
        norm = norm * gate_val                                    # fp32

        # store to contiguous layout
        dst_ptr_h = dst_ptr + dst_base[:, None] + offs_h[None, :]   # stride_h_dst = 1
        tl.store(dst_ptr_h,
                 norm.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

    # --------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]

            w["right_proj.weight"],

            w["right_gate.weight"],

        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # fp16, shape [R, 5·H]
    del fused_w, x_norm                         # free

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # flatten mask and cast to fp16 – same broadcasting logic as reference
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128   # rows per program for split kernel
    BLOCK_H = 32    # hidden‑dim chunk (fits within 128‑thread warp)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM: out = left @ rightᵀ
    # --------------------------------------------------------------
    # Reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # rightᵀ -> (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N)                 # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + gating → contiguous (B, N, N, H)
    # --------------------------------------------------------------
    out_fused = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    layernorm_mul_fused_kernel[grid_ln2](
        out,                 # src (B, H, N, N)
        out_gate,            # gate (B, H, N, N)
        out_fused,           # dst (B, N, N, H)
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        B,
        N,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate, left, right

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    # out_fused is already in (B, N, N, H) – flatten first three dims
    out_fused_flat = out_fused.view(R, H)                     # [R, H]

    final_half = F.linear(
        out_fused_flat,
        w["to_out.weight"].to(torch.float16),                 # [dim, H]
        bias=None,
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
