"""
TriMul (outgoing) – fast Triton + cuBLAS implementation
========================================================
The forward pass consists of six stages:

1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the reshaped input (B·N·N × dim).
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single fp16 GEMM:  X_norm @ W_fusedᵀ → (R, 5·H),  R = B·N·N.
3️⃣ Split the fused result, apply the optional pair‑wise mask, evaluate the
   sigmoid gates and write the three tensors directly in the (B, H, N, N)
   layout required by the batched GEMM.
4️⃣ Batched half‑precision matrix multiplication (cuBLAS) over the hidden
   dimension:
        out[b,h] = left[b,h] @ right[b,h]ᵀ
5️⃣ Fuse the second LayerNorm and element‑wise out‑gate multiplication, writing
   the result into a contiguous (R, H) buffer ready for the final linear.
6️⃣ Final linear projection back to the original `dim` (fp16 → fp32).

Only three custom Triton kernels are needed – the heavy matrix multiplies are
delegated to cuBLAS (torch.bmm / torch.nn.functional.linear) which gives near‑
peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,         # const   *[rows, dim]   (float32)
    y_ptr,         # mutable *[rows, dim]   (float16)
    w_ptr,         # const   *[dim]         (float32)  γ
    b_ptr,         # const   *[dim]         (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # dim chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β  (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly into (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat row index → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Broadcast mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between two hidden slices

    # Base offset for hidden‑dim = 0 in the destination layout
    base_offset = b * (H * nn) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Load the five packed blocks from the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Stable sigmoid in fp32 (then back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offsets for the hidden dimension
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm + out_gate multiplication,
#    write into a contiguous (R, H) buffer for the final linear
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fused_kernel(
    out_ptr,        # const   *[B, H, N, N] (fp16)
    gate_ptr,       # const   *[B, H, N, N] (fp16)
    out_fused_ptr,  # mutable *[R, H]       (fp16)
    w_ptr,          # const   *[H]           (float32) γ
    b_ptr,          # const   *[H]           (float32) β
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N                           # stride for the hidden dimension
    # Decode (b,i,j) from flat row index
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # Base offset for hidden = 0 in the (B, H, N, N) layout
    base_offset = b * (H * nn) + i * N + j                # [BLOCK_R]

    # ---- First pass: compute per‑row mean & variance over H ---------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + base_offset[:, None] +
                       offs_h[None, :] * nn,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- Second pass: normalize, apply γ/β, multiply by out‑gate ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr  + base_offset[:, None] +
                           offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        gate_vals = tl.load(gate_ptr + base_offset[:, None] +
                            offs_h[None, :] * nn,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        w = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0)   # γ (fp32)
        b_ = tl.load(b_ptr + offs_h, mask=mask_h, other=0.0) # β (fp32)

        norm = (out_vals - mean[:, None]) * rstd[:, None]           # fp32
        norm = norm * w[None, :] + b_[None, :]                      # fp32
        out = norm * gate_vals                                      # fp32

        # Store into contiguous (R, H) buffer
        tl.store(out_fused_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

    """

    hidden_dim = cfg["hidden_dim"]          # H

    )

    # 3️⃣ Split, apply optional mask and sigmoid gates (write in B·H·N·N layout)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        # No mask – compile kernel without mask handling
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # Real mask present – flatten to (R,)
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # Collapse (B, H) → batch dimension for cuBLAS
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out_mat[b] = left_mat[b] @ right_mat[b]ᵀ   (half‑precision, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N)          # back to (B, H, N, N)

    del left, right    # free temporary tensors

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate multiplication (produces (R, H))
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_R2),)
    layernorm_gate_fused_kernel[grid_ln_gate](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate   # no longer needed

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 → fp32)
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

**Explanation of the key points**

* **LayerNorm kernels** perform the mean/variance reduction in fp32 and emit fp16 results, eliminating the need for a separate reduction pass.
* **`split_fuse_gate_kernel`** reads the fused projection `[R,5·H]`, optionally multiplies by the pair‑wise mask, evaluates sigmoid gates (stable fp32 → fp16) and writes the three tensors directly into the layout required for the batched GEMM, avoiding extra transposes or copies.
* **Batched GEMM** (`torch.bmm`) leverages cuBLAS Tensor‑Core kernels for the expensive `N³` operation.
* **`layernorm_gate_fused_kernel`** computes the second LayerNorm per `(b,i,j)` position while simultaneously applying the `out_gate` and writing the result into a compact `(R, H)` buffer for the final linear projection.
* All heavy element‑wise work is done inside Triton kernels, while the three large matrix multiplications are delegated to cuBLAS, which is optimal on an H100.

The implementation respects the required function signature (`custom_kernel(data)`) and works for all configurations listed in the problem statement.
