"""
custom_kernel  –  High‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (dim).
   Implemented in Triton (`layernorm_kernel`).  Input is FP32,
   output is FP16 to feed the following GEMMs.

2️⃣ Fuse the four linear projections (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) into a single half‑precision GEMM.
   The result is split into the four tensors.

3️⃣ Compute the output‑gate linear projection (`out_gate`) separately
   (still FP16).

4️⃣ Apply the optional pair‑wise mask, then the sigmoid‑gates for
   left/right and the output‑gate.

5️⃣ Batched N³ multiplication:
   left   : [B,N,N,H]   →   (B·H,N,N)
   right  : [B,N,N,H]   →   (B·H,N,N)
   out    = left @ rightᵀ   (torch.bmm, FP16)

6️⃣ **Fused** second LayerNorm, out‑gate multiplication and the final
   projection back to `dim` in a single Triton kernel
   (`norm_gate_linear_kernel`).  This eliminates the intermediate
   tensor of shape (R,H) and cuts memory traffic dramatically.

7️⃣ Cast the result back to FP32 and reshape to [B,N,N,dim].

All heavy compute is performed in half‑precision on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  *[total_rows, dim]   (fp32)
    y_ptr,          # mutable*[total_rows, dim]   (fp16)
    weight_ptr,     # const  *[dim]               (fp32)   γ
    bias_ptr,       # const  *[dim]               (fp32)   β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6️⃣  Fused LayerNorm + out‑gate multiplication + final Linear
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_linear_kernel(
    x_ptr,          # const  *[total_rows, H]   (fp16)   – result of bmm
    gate_ptr,       # const  *[total_rows, H]   (fp16)   – out_gate
    out_ptr,        # mutable*[total_rows, dim] (fp16)   – final result
    weight_ptr,     # const  *[H, dim]          (fp16)   – to_outᵀ
    total_rows,     # int32   runtime
    H,              # int32   runtime  (hidden_dim)
    dim,            # int32   runtime  (output dim)
    eps,            # float   compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output cols per iteration
    BLOCK_K: tl.constexpr,   # hidden cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulators for mean/var
    sum_x  = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_x2 = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulators for the two dot‑products
    acc1 = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)   # Σ x·gate·W
    acc2 = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)   # Σ gate·W

    # ------------------------------------------------------------------
    # Loop over hidden dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # --- load x and gate ---
        x_ptrs = x_ptr + (offs_m[:, None] * H + offs_k[None, :])
        g_ptrs = gate_ptr + (offs_m[:, None] * H + offs_k[None, :])
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        g = tl.load(g_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # --- update mean/var (on x only) ---
        sum_x  += tl.sum(x, axis=1)
        sum_x2 += tl.sum(x * x, axis=1)

        # --- load weight tile (H × dim) ---
        offs_d = tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        w_ptrs = weight_ptr + (offs_k[:, None] * dim + offs_d[None, :])
        w = tl.load(w_ptrs,
                    mask=mask_k[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)           # [BLOCK_K, BLOCK_D]

        # --- contributions to the two inner products ---
        xg = x * g                                      # [BLOCK_M, BLOCK_K]
        acc1 += tl.dot(xg, w)                           # Σ (x·gate)·W
        acc2 += tl.dot(g,  w)                           # Σ gate·W

    # ------------------------------------------------------------------
    # Final LayerNorm + gate + Linear
    # ------------------------------------------------------------------
    mean = sum_x / H
    var  = sum_x2 / H - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # broadcast to block‑wise shape
    mean_b = mean[:, None]      # [BLOCK_M, 1]
    rstd_b = rstd[:, None]      # [BLOCK_M, 1]

    # out = rstd * (acc1 - mean * acc2)
    out = rstd_b * (acc1 - mean_b * acc2)               # FP32

    # write FP16 result
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim
    out_ptrs = out_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    tl.store(out_ptrs, out.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator

def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ---------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm  (FP32 → FP16)
    # ---------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                # [R, dim] fp16

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    #    Also compute out_gate (still fp16)
    # ---------------------------------------------------------------

        [

            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]

        dim=0,

    right_raw      = fused[:, H:2*H]

    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)      # [H, dim]
    )                                                     # [R, H] fp16

    # ---------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ---------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (left @ rightᵀ)
    # ---------------------------------------------------------------
    # reshape to (B, N, N, H) and bring H into the batch axis
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16
    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # flatten for the fused kernel
    out_flat = out.view(total_rows, H)                # [R, H] fp16
    out_gate_flat = out_gate                          # [R, H] fp16

    # ---------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out_gate multiplication + final Linear
    # ---------------------------------------------------------------
    # transpose the final weight to [H, dim] for easier tiling
    to_out_weight_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # [H, dim]

    final_fp16 = torch.empty(total_rows, dim,
                             dtype=torch.float16,
                             device=device)      # [R, dim] fp16

    BLOCK_M2 = 128
    BLOCK_D2 = 64
    BLOCK_K2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)

    norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        final_fp16,
        to_out_weight_T,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        BLOCK_K=BLOCK_K2,
        num_warps=8,
    )

    # ---------------------------------------------------------------
    # Cast back to fp32 and reshape
    # ---------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
