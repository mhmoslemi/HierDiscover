"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (dim) – uses the
   highly‑optimised torch implementation (FP32 → FP16).

2️⃣ Fuse the four linear projections (left_proj, right_proj,
   left_gate, right_gate) into a **single** FP16 GEMM:
   X_norm @ W_fused.T → [R, 4·H]   (R = B·N·N, H = hidden_dim).

3️⃣ Compute the output‑gate with a second FP16 linear layer.

4️⃣ (Optional) Apply the binary mask to the two projection tensors.
   Apply sigmoid to all three gate tensors and multiply:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)

5️⃣ Batch the N³ multiplicative update:
        left  = left .view(B, H, N, N).reshape(B·H, N, N)
        right = right.view(B, H, N, N).reshape(B·H, N, N)
        out   = bmm(left, right.transpose(1,2))   # (B·H,N,N)

   Reshape back to (B,N,N,H).

6️⃣ Fuse the second LayerNorm (over hidden_dim) with the element‑wise
   multiplication by the output‑gate in a Triton kernel
   (FP32 → FP16).

7️⃣ Final linear projection back to the original dimensionality
   (FP16 GEMM) and cast the result to float32.

Only the two custom Triton kernels below are required – all heavy
matrix‑multiplications use the highly‑optimised torch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise out‑gate (FP32 → FP16)
#   x  : [total_rows, dim]   (float32)    – tensor to normalise
#   g  : [total_rows, dim]   (float16)   – out‑gate (already after sigmoid)
#   y  : [total_rows, dim]   (float16)   – output = LayerNorm(x) * g
#   w  : [dim]               (float32)  – γ
#   b  : [dim]               (float32)  – β
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    g_ptr,          # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    w_ptr,          # const   * [dim]              (float32)
    b_ptr,          # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ---------- normalise, apply γ/β and multiply by gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation (FP32)
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # gate (FP16 → FP32)
        g = tl.load(g_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g           # (x̂·γ+β)·gate
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int,
                           "hidden_dim": int,
                           "nomask": bool (optional)
                          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- unpack --------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (i,j) pairs
    device = x.device

    # -------------------- 1️⃣ Row‑wise LayerNorm (torch) --------------------
    # torch's native LayerNorm is highly optimised; we keep at least one
    # custom Triton kernel later (step 6).
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim]  (float32)
    x_norm_fp16 = x_norm.to(torch.float16)   # cast once

    # -------------------- 2️⃣ Fuse left/right proj + gates (single GEMM) --------------------
    # Concatenate the four projection weights: [left, right, left_gate, right_gate]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H]  (FP16)

    left_raw   = fused[:, 0:hidden_dim]
    right_raw  = fused[:, hidden_dim:2 * hidden_dim]
    left_gate_raw  = fused[:, 2 * hidden_dim:3 * hidden_dim]
    right_gate_raw = fused[:, 3 * hidden_dim:4 * hidden_dim]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] (FP16)

    # -------------------- 3️⃣ Mask & sigmoid gates --------------------
    if (not nomask) and mask is not None:
        # mask shape = [B,N,N] → [R,1]
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] (FP16)

    left  = left_raw * left_gate     # [R, H] FP16
    right = right_raw * right_gate   # [R, H] FP16

    # -------------------- 4️⃣ Batched N³ multiplicative update --------------------
    # reshape to (B, H, N, N) → (B·H, N, N)
    left_mat  = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] (FP16)

    # -------------------- 5️⃣ Fused LayerNorm + out‑gate (Triton) --------------------
    # flatten rows = B·N·N, each row length = hidden_dim
    out_fp16 = out
    out_gate_fp16 = out_gate.view(B, N, N, hidden_dim)

    # allocate output buffer
    out_norm_gate = torch.empty_like(out_fp16)

    # launch Triton kernel
    BLOCK_M = 128          # rows per program (tuned for H100)
    BLOCK_D = 64           # columns per iteration
    total_rows = B * N * N
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_fp16,                         # x_ptr  (float32) – will be cast inside kernel
        out_gate_fp16,                    # g_ptr  (float16)
        out_norm_gate,                    # y_ptr  (float16)
        weights["to_out_norm.weight"],    # γ (float32)
        weights["to_out_norm.bias"],      # β (float32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # -------------------- 6️⃣ Final linear projection (FP16 GEMM) --------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    )                                                # [R, dim] (FP16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
