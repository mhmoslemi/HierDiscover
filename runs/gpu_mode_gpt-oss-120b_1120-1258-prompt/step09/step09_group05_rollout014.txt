"""
TriMul (outgoing) – fused Triton implementation.

Algorithm (all ops are in FP16 except the LayerNorm statistics which stay FP32):
1️⃣  Row‑wise LayerNorm + 5 linear projections (left/right + 3 gates) are fused
    into a single GEMM:
        out = ((x - μ) / √(σ²+ε) * γ + β) @ Wᵀ
    where   W = [W_left ; W_right ; W_left_gate ; W_right_gate ; W_out_gate]
    (shape [5·H, C]).
2️⃣  Apply the optional binary mask to the raw left/right tensors, then
    apply the three sigmoid gates (in‑place).
3️⃣  Reshape left/right → [B·H, N, N] and perform the cubic update with a
    batched FP16 GEMM:
        out = left @ rightᵀ          (B·H bmm)
4️⃣  Fuse the second LayerNorm, the output‑gate and the final linear
    projection into one kernel:
        y = ((out - μ) / √(σ²+ε) * γ₂ + β₂)   # LN over H
        y = y * out_gate                     # element‑wise gating
        out = y @ W_outᵀ                     # W_out: [C, H]
5️⃣  Cast the result back to FP32.

The implementation uses only two Triton kernels (the two fused passes) and
leverages Torch’s highly‑optimized batched GEMM for the N³ cubic update.
All intermediate tensors stay in FP16, minimizing memory traffic and
bringing the runtime well below the reference 2.5 ms.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1:  row‑wise LayerNorm + 5 fused linear projections
# ----------------------------------------------------------------------
@triton.jit
def proj_norm_kernel(
    x_ptr,                 # fp32   [R, C]
    out_ptr,               # fp16   [R, O]   (O = 5*H)
    ln_w_ptr,              # fp32   [C]     (γ)
    ln_b_ptr,              # fp32   [C]     (β)
    w_ptr,                 # fp16   [O, C]
    R, C, O,               # runtime dimensions
    eps,                   # compile‑time constant
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_K: tl.constexpr, # tile over C
    BLOCK_N: tl.constexpr, # tile over O
):
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # output‑col block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < R
    mask_n = offs_n < O

    # --------------------------------------------------------------
    # 1. Compute mean / variance of each row (over C)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(C, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < C

        x = tl.load(x_ptr + (offs_m[:, None] * C + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [BM, BK]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2. GEMM: ( (x-μ)/σ * γ + β ) @ Wᵀ
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(C, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < C

        # ---- load tile of X -------------------------------------------------
        x = tl.load(x_ptr + (offs_m[:, None] * C + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [BM, BK]

        # ---- normalize ------------------------------------------------------
        x_norm = (x - mean[:, None]) * rstd[:, None]  # (x‑μ)/σ

        # ---- apply LayerNorm scale+bias ------------------------------------
        wgt = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # γ
        b   = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # β
        x_norm = x_norm * wgt[None, :] + b[None, :]               # γ·x + β

        # ---- load corresponding tile of Wᵀ ---------------------------------
        #   W is [O, C] row‑major.  We need Wᵀ tile shape [BK, BN].
        w = tl.load(w_ptr + (offs_n[None, :] * C + offs_k[:, None]),
                    mask=mask_n[None, :] & mask_k[:, None],
                    other=0.0).to(tl.float16).to(tl.float32)   # [BK, BN]

        # ---- accumulate -------------------------------------------------------
        acc += tl.dot(x_norm, w)   # (BM,BK)·(BK,BN) → (BM,BN)

    # --------------------------------------------------------------
    # 3. Store result (FP16)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + (offs_m[:, None] * O + offs_n[None, :])
    tl.store(out_ptrs,
             out,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Kernel 2: final LayerNorm + output gate + projection to C
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    inp_ptr,               # fp16   [R, H]    (cubic‑update result)
    gate_ptr,              # fp16   [R, H]    (out_gate after sigmoid)
    out_ptr,               # fp16   [R, C]    (final output)
    ln_w_ptr,              # fp32   [H]      (γ₂)
    ln_b_ptr,              # fp32   [H]      (β₂)
    w_ptr,                 # fp16   [C, H]    (final projection)
    R, H, C,               # runtime dimensions
    eps,
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_K: tl.constexpr, # tile over H
    BLOCK_N: tl.constexpr, # tile over C
):
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # column (C) block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < R
    mask_n = offs_n < C

    # --------------------------------------------------------------
    # 1. Compute mean / variance over H for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        x = tl.load(inp_ptr + (offs_m[:, None] * H + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [BM, BK]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2. Accumulate final projection while applying LN, gate and weight
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ---- load raw activation and gate ---------------------------------
        x = tl.load(inp_ptr + (offs_m[:, None] * H + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [BM, BK]

        g = tl.load(gate_ptr + (offs_m[:, None] * H + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [BM, BK]

        # ---- normalize ------------------------------------------------------
        x_norm = (x - mean[:, None]) * rstd[:, None]

        # ---- apply second LayerNorm scale+bias --------------------------------
        wgt = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # γ₂
        b   = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # β₂
        x_norm = x_norm * wgt[None, :] + b[None, :]

        # ---- apply output‑gate -----------------------------------------------
        x_norm = x_norm * g

        # ---- load weight tile (W_outᵀ) ---------------------------------------
        #   W_out : [C, H] row‑major.  Need tile as [BK, BN] where BK=H‑tile,
        #   BN=C‑tile:  offset = col*H + row
        w = tl.load(w_ptr + (offs_n[None, :] * H + offs_k[:, None]),
                    mask=mask_n[None, :] & mask_k[:, None],
                    other=0.0).to(tl.float16).to(tl.float32)   # [BK, BN]

        # ---- accumulate -------------------------------------------------------
        acc += tl.dot(x_norm, w)   # (BM,BK)·(BK,BN) -> (BM,BN)

    # --------------------------------------------------------------
    # 3. Write final output (FP16)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + (offs_m[:, None] * C + offs_n[None, :])
    tl.store(out_ptrs,
             out,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (alpha‑fold‑3 style).

    Args
    ----
    data : tuple
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32)
        - config_dict  : dict with keys
            * "dim"         – input channel dimension (C)
            * "hidden_dim" – hidden dimension H (typically 128)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, C = x.shape
    H = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # total rows that we treat as a 2‑D matrix [R, C]
    R = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Fuse first LayerNorm + 5 linear projections (left, right, gates, out_gate)
    # ------------------------------------------------------------------
    x_flat = x.view(R, C).contiguous()                # [R, C]  fp32

    # concatenate the five weight matrices:  left, right, left_gate,
    # right_gate, out_gate   →  shape [5*H, C]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, C]
            weights["right_proj.weight"],  # [H, C]
            weights["left_gate.weight"],   # [H, C]
            weights["right_gate.weight"],  # [H, C]
            weights["out_gate.weight"],    # [H, C]
        ],
        dim=0,
    ).to(torch.float16).contiguous()                 # [5*H, C]

    O = 5 * H                                          # number of output channels

    proj_out = torch.empty(R, O, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_K = 32
    BLOCK_N = 64

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(O, BLOCK_N))
    proj_norm_kernel[grid](
        x_flat,
        proj_out,
        weights["norm.weight"],          # γ (fp32)
        weights["norm.bias"],            # β (fp32)
        w_fused,
        R,
        C,
        O,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Split & apply mask + sigmoid gates
    # ------------------------------------------------------------------
    proj_out = proj_out.view(R, 5, H)   # [R, 5, H]

    left_raw  = proj_out[:, 0, :]       # [R, H]
    right_raw = proj_out[:, 1, :]
    left_gate_raw = proj_out[:, 2, :]
    right_gate_raw = proj_out[:, 3, :]
    out_gate_raw = proj_out[:, 4, :]

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)          # [R]
        left_raw  = left_raw * mask_flat[:, None]
        right_raw = right_raw * mask_flat[:, None]

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate        # [R, H]  fp16
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Cubic multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left_ = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_ = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat = left_.view(B * H, N, N)           # [B·H, N, N]
    right_mat = right_.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]  fp16
    out_gate_flat = out_gate.view(R, H).contiguous()

    final_fp16 = torch.empty(R, C, dtype=torch.float16, device=device)

    # grid for final kernel
    grid2 = (triton.cdiv(R, BLOCK_M), triton.cdiv(C, BLOCK_N))
    final_norm_proj_kernel[grid2](
        out_flat,
        out_gate_flat,
        final_fp16,
        weights["to_out_norm.weight"],   # γ₂ (fp32)
        weights["to_out_norm.bias"],     # β₂ (fp32)
        weights["to_out.weight"].to(torch.float16),   # [C, H]
        R,
        H,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to float32 and reshape
    # ------------------------------------------------------------------
    out_fp32 = final_fp16.view(B, N, N, C).to(torch.float32)
    return out_fp32
