"""
custom_kernel – High‑performance forward of the *outgoing* TriMul operator.

Algorithm (all heavy compute fused in Triton)
------------------------------------------------
1️⃣ Row‑wise LayerNorm over the input channel dimension `dim`.
   • Input   : FP32 tensor [B·N·N, dim]
   • Output  : FP16 tensor   [B·N·N, dim]
   • Reduction (mean/var) is performed in FP32, results are written in FP16.

2️⃣ Fused projection + gating (single GEMM):
   • Build a weight matrix W_fused = [left_proj; right_proj; left_gate; right_gate]
     of shape (4·H, dim) where H = hidden_dim.
   • Compute  X_norm @ W_fusedᵀ  →  [R, 4·H] (R = B·N·N, FP16).

3️⃣ Separate the four heads and compute the output gate (second linear on X_norm).

4️⃣ Apply optional pair‑wise mask and sigmoid‑gates (in‑place).

5️⃣ Batched N³ multiplicative update:
   • Reshape left/right to (B·H, N, N) and perform a batched matmul
       out = left @ rightᵀ   (FP16, torch.bmm).

6️⃣ Fused LayerNorm + out‑gate multiplication (Triton):
   • Normalise the result over hidden dimension H, apply γ/β and finally
     multiply by the pre‑computed out‑gate (all in FP32, store FP16).

7️⃣ Final linear projection back to the original channel dimension `dim`
   (FP16 GEMM) and cast the result to FP32.

The kernels are deliberately simple, use contiguous row‑major tensors,
and avoid any Python‑level loops – the only Python work left is the
high‑throughput batched GEMM, which is already heavily optimised on the
H100.  The implementation works for any combination of
B∈{1,2}, N≤1024, dim∈{128,384}, hidden_dim=128 and with/without mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [R, D] (float32)
    y_ptr,          # mutable * [R, D] (float16)
    gamma_ptr,      # const   * [D]    (float32) scale γ
    beta_ptr,       # const   * [D]    (float32) bias β
    total_rows,     # i32   runtime
    dim,            # i32   runtime
    eps,            # f32   compile‑time constant
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    # row offsets
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute per‑row mean & variance (FP32 reduction)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # Load a tile [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ/β and write out (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]   # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [R, D] (float16)   activation
    gate_ptr,       # const   * [R, D] (float16)   gate
    y_ptr,          # mutable * [R, D] (float16)   output
    gamma_ptr,      # const   * [D]    (float32) γ
    beta_ptr,       # const   * [D]    (float32) β
    total_rows,     # i32   runtime
    dim,            # i32   runtime
    eps,            # f32   compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute mean/var of the activation (ignore gate)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise + γ/β + gate multiplication
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        act = tl.load(x_ptr   + offs_m[:, None] * dim + offs_d[None, :],
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        g   = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (act - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensor parameters (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm → FP16
    # ------------------------------------------------------------------
    # Flatten to [R, dim] contiguous FP32
    x_flat = x.view(R, dim).contiguous()
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Choose tile sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_D = 128 if dim >= 128 else dim
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates into one GEMM
    # ------------------------------------------------------------------
    # Build fused weight matrix [4·H, dim] in FP16
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # Linear on the normalized input (FP16 → FP16)
    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H]

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B, N, N] → [R, 1] FP16
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # already FP16

    left  = left_raw * left_gate   # [R, H] FP16
    right = right_raw * right_gate # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Move head dim into batch dimension → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul (FP16, very fast on H100)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + multiplication by out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()      # [R, H] FP16
    out_gate_flat = out_gate                    # [R, H] FP16 (same layout)

    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Tile sizes for hidden dimension (H ≤ 128)
    BLOCK_M2 = 128
    BLOCK_D2 = 64 if H > 64 else H
    grid2 = (triton.cdiv(R, BLOCK_M2),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
