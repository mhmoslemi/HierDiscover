   This is a set of B·H independent matrix‑multiplications of size N×N.
   We realise it with a single call to ``torch.bmm`` on a view of shape
   (B·H, N, N).  No data copy is performed – the view is built with
   ``transpose`` (which is a stride‑only operation).

5️⃣ Fuse the second LayerNorm and the element‑wise out‑gate into a
   second Triton kernel, again working in FP32 internally and writing
   FP16 results.

6️⃣ Final FP16 linear projection back to the original “dim’’ channel.
   The output is cast to FP32 to match the reference implementation.

Only the two LayerNorm‑related kernels are written in Triton; all heavy
linear algebra (the N³ contraction and the final projection) uses the
high‑performance cuBLAS kernels accessed via ``torch.nn.functional.linear``
and ``torch.bmm``.  This balance gives excellent performance on an H100
while satisfying the requirement that “part of the operation” runs on
Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim]    (float32)
    y_ptr,          # mutable * [rows, dim]    (float16)
    weight_ptr,     # const   * [dim]          (float32)
    bias_ptr,       # const   * [dim]          (float32)
    rows,           # int32   total rows
    dim,            # int32   trailing dim
    eps,            # fp32    epsilon (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / var -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ---- normalise + apply γ/β --------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]           # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise out‑gate (float32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_kernel(
    x_ptr,          # const   * [rows, dim]    (float32)
    gate_ptr,       # const   * [rows, dim]    (float16)
    y_ptr,          # mutable * [rows, dim]    (float16)
    weight_ptr,     # const   * [dim]          (float32)
    bias_ptr,       # const   * [dim]          (float32)
    rows,           # int32   total rows
    dim,            # int32   trailing dim
    eps,            # fp32    epsilon (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- mean / var -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalise, apply γ/β and multiply by gate --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor  [B, N, N]       (bool/float) or None
        weights_dict : mapping of all Linear/LayerNorm parameters
        config_dict  : {"dim": int, "hidden_dim": int,
                       "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & basic constants
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                    # x: [B, N, N, dim]
    R = B * N * N                            # total rows (= B·N·N)
    H = hidden_dim                           # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # Choose block sizes based on dim (BLOCK_D must divide dim efficiently)
    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 128 else 128

    grid = (triton.cdiv(R, BLOCK_M),)
    _layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # FP16, shape [R, 4·H]

    # split the fused result
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2 * H]           # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw= fused[:, 3 * H:4 * H]       # [R, H]

    # out‑gate (separate linear, still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape [B,N,N] → [R,1] (FP16)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # inplace sigmoid -> stays FP16
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()        # [R, H] FP16

    # gated projections
    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    #    No extra memory copies – only stride‑only transposes.
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) then view as (B, H, N, N) via transpose
    left  = left.view(B, N, N, H).transpose(1, 3)   # [B, H, N, N] (non‑contig)
    right = right.view(B, N, N, H).transpose(1, 3)   # [B, H, N, N]

    # fuse batch & head dimensions for a single bmm call
    left_mat  = left.reshape(B * H, N, N)            # (B·H, N, N)
    right_mat = right.reshape(B * H, N, N)           # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).transpose(1, 3).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + element‑wise out‑gate (over hidden dim H)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] FP16
    out_gate_flat = out_gate                         # already FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    _layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim (FP16 GEMM)
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    # cast back to float32 as the reference model does
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
