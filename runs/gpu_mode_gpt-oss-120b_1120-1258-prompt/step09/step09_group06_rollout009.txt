@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for batch
    a_ptr = A_ptr + pid_batch * stride_a_batch
    b_ptr = B_ptr + pid_batch * stride_b_batch
    c_ptr = C_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a = tl.load(a_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

        # Increment pointers? Not needed as we compute addresses dynamically using offsets.

    # Write back
    c = acc.to(tl.float16)
    tl.store(c_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

"""
High‑performance forward pass for the *outgoing* TriMul operator (AlphaFold3).

Algorithm (mixed‑precision, FP16 compute):
1️⃣ Row‑wise LayerNorm on the flattened input (FP32 → FP16) via a Triton kernel.
2️⃣ Fuse the four linear projections/gates into a single FP16 GEMM:
   X_norm @ [W_left;W_right;W_left_gate;W_right_gate]ᵀ   →   left_raw,
   right_raw, left_gate_raw, right_gate_raw.
3️⃣ Compute the output‑gate with a second FP16 GEMM.
4️⃣ Optional binary mask, sigmoid‑gates and element‑wise gating (in‑place).
5️⃣ Batched cubic update – a GEMM for each head:
   (B·H, N, N) @ (B·H, N, N)ᵀ  using a custom Triton kernel (no extra copies).
6️⃣ Second LayerNorm over the hidden dimension (FP16) via the same Triton kernel.
7️⃣ Apply the output‑gate, final linear projection and cast back to FP32.

All heavy compute runs in FP16 (Tensor‑Core friendly); only the two LayerNorms
need the FP32 statistics.  The implementation works for B∈{1,2},
N∈{128,256,512,1024}, dim∈{128,384,768}, hidden_dim=128 and supports optional
masking.  At least one operation (both LayerNorms) is hand‑written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const * [total_rows, dim]   (float32)
    y_ptr,          # mut   * [total_rows, dim]   (float16)
    w_ptr,          # const * [dim]               (float32) – γ
    b_ptr,          # const * [dim]               (float32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per tile
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # program index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1. mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/σ

    # ---------- 2. write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        # γ and β stay in FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the cubic update
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,          # const * [batch, M, K]   (fp16)
    b_ptr,          # const * [batch, N, K]   (fp16)   (already transposed)
    c_ptr,          # mut   * [batch, M, N]   (fp16)
    M, N, K,        # matrix dimensions (runtime)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch entry.
    A : (batch, M, K)
    B : (batch, N, K)   (note: K is the inner dimension; we will treat B as transposed)
    C : (batch, M, N)
    """
    pid_batch = tl.program_id(0)   # which head (B*H)
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a = tl.load(a_batch_ptr + offs_m[:, None] * stride_a_m
                                 + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        b = tl.load(b_batch_ptr + offs_n[:, None] * stride_b_n
                                 + offs_k[None, :] * stride_b_k,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # Write back result in FP16
    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + offs_m[:, None] * stride_c_m
                              + offs_n[None, :] * stride_c_n,
              c,
              mask=mask_m[:, None] & mask_n[None, :])

    # 4️⃣ Optional mask & sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → flatten → broadcast over hidden dim
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate            # [R, H]
    right = right_raw * right_gate          # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Batched cubic update  (left @ rightᵀ)  – custom Triton GEMM
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # permute to bring H into batch dimension: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # now we have shape (B*H, N, N) for each operand
    left_mat = left_perm.view(B * H, N, N)                     # [B·H, N, N]
    # we need rightᵀ → transpose the last two dimensions and make it contiguous
    right_mat_t = right_perm.view(B * H, N, N).transpose(1, 2).contiguous()

    # allocate output buffer (fp16)
    out_mat = torch.empty_like(left_mat)                       # [B·H, N, N]

    # launch Triton batched GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        N, N, N,                                 # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_t.stride(0), right_mat_t.stride(2), right_mat_t.stride(1),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final linear projection, cast back to fp32
    # ------------------------------------------------------------------
    # Element‑wise gating (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # Final projection: [R, H] @ [dim, H]ᵀ → [R, dim] (fp16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    # Cast to fp32 and restore original shape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
