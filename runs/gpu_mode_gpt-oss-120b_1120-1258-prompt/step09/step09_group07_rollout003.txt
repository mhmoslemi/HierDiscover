"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (float32 → float16) using
    torch.nn.functional.layer_norm (highly‑optimised cuBLAS kernel).

2️⃣  Fuse the four linear projections (left, right, left_gate, right_gate)
    into a single matmul with a weight matrix of shape [4·H, dim] (H = hidden_dim).
    The result is split into the four tensors.

3️⃣  Compute the output‑gate (out_gate) with a second linear on the same
    normalized input.

4️⃣  Apply the optional mask (broadcasted on the hidden dim) and the sigmoid
    gates. All tensors stay in float16.

5️⃣  Reshape to (B·H, N, N) and perform a batched matrix‑multiply:
        out = left @ rightᵀ                (tensor‑core FP16 GEMM)

6️⃣  Fuse the second LayerNorm (over the hidden dim H) with the multiplication
    by out_gate in a custom Triton kernel (`layernorm_gate_kernel`).

7️⃣  Final linear projection back to the original channel dimension `dim`
    (FP16 GEMM).  Cast the result back to float32.

Only the two LayerNorm kernels are written in Triton; all heavy GEMMs are
delegated to cuBLAS tensor‑core kernels for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const  * [total_rows, dim] (float16)
    gate_ptr,       # const  * [total_rows, dim] (float16)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    weight_ptr,     # const  * [dim]            (float32) γ
    bias_ptr,       # const  * [dim]            (float32) β
    total_rows,     # int32 runtime
    dim,            # int32 runtime
    eps,            # float compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- normalise + γ/β + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load activation & gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_vals          # FP32
        out_fp16 = out.to(tl.float16)

        ptr_out = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(ptr_out, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : Mapping with all linear / layer‑norm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # total rows for (B, N, N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    # Use the highly‑optimised torch.nn.functional.layer_norm
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim], float32
    x_norm_fp16 = x_norm.to(torch.float16).reshape(R, dim)   # [R, dim], FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections (left/right + gates)
    # ------------------------------------------------------------------
    # Build fused weight [4*H, dim] for left_proj, right_proj, left_gate, right_gate
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                    # [4*H, dim]

    # Compute all four tensors in one matmul
    fused = F.linear(x_norm_fp16, w_fused)          # [R, 4*H], FP16

    left_raw   = fused[:, 0:H]               # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]        # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]        # [R, H]

    # Output‑gate (still using the normalized input)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H], FP16

    # ------------------------------------------------------------------
    # 3️⃣ Mask & sigmoid gates
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1]
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H], FP16
    right = right_raw * right_gate  # [R, H], FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    # Tensor‑core FP16 GEMM
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate multiplication (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                 # [R, H] FP16
    out_gate_flat = out_gate                               # [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_D = 32   # hidden_dim ≤ 128 → fits nicely
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim (FP16 → FP32)
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim], FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
