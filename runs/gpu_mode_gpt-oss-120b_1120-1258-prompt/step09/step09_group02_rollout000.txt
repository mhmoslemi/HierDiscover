"""
custom_kernel – high‑performance “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim)   – Triton.
2️⃣ Fuse the five linear projections (left, right, left_gate,
   right_gate, out_gate) into a single GEMM:  X_norm @ W_fusedᵀ.
   The result has shape (R, 5·H) where R = B·N·N.
3️⃣ One Triton kernel splits the fused tensor, applies the optional
   pair‑wise mask, evaluates sigmoid gates (fp32 → fp16) and writes
   three tensors (left, right, out_gate) of shape (R, H).
4️⃣ Batched matrix‑multiply over the hidden dim (H) using torch.bmm:
   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N).  This is the N³ “TriMul”.
5️⃣ Second row‑wise LayerNorm (again Triton) on the hidden‑dim result.
   Multiply element‑wise with the out‑gate.
6️⃣ Final linear projection back to the original dim (fp16 → fp32).

Only the two LayerNorms and the split‑gate kernel are written in Triton.
All heavy matrix‑multiplications use cuBLAS via PyTorch for maximal
throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # -------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32 accumulation)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32
        sum_val += tl.sum(vals, axis=1)                     # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # 2️⃣  Normalize, apply γ/β and write as fp16
    # -------------------------------------------------
    for d_iter in range(num_d_iters):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r ∈ [0,R):
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    All ops are fp16; sigmoid is evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # -------------------------------------------------
    # Load scalar mask per row (if present)
    # -------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                         # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # -------------------------------------------------
    # Iterate over hidden dimension
    # -------------------------------------------------
    num_h_iters = tl.cdiv(H, BLOCK_H)
    for h_iter in range(num_h_iters):
        offs_h = h_iter * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout in fused tensor:
        #   0:H          -> left
        #   H:2H         -> right
        #   2H:3H        -> left_gate
        #   3H:4H        -> right_gate
        #   4H:5H        -> out_gate
        row_offset = offs_r[:, None] * (5 * H)               # [R,1]
        col_offset = offs_h[None, :]                         # [1,H]

        left_raw   = tl.load(proj_ptr + row_offset + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        right_raw  = tl.load(proj_ptr + row_offset + H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        left_g_raw = tl.load(proj_ptr + row_offset + 2 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        right_g_raw= tl.load(proj_ptr + row_offset + 3 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        out_g_raw  = tl.load(proj_ptr + row_offset + 4 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for better accuracy
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Store
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                                   # rows for row‑wise ops
    H = hidden_dim

    # -------------------------------------------------
    # 1️⃣ First LayerNorm  (float32 → float16)
    # -------------------------------------------------
    x_flat = x.view(R, dim).contiguous()            # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64                                    # more columns per iteration
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -------------------------------------------------
    # 2️⃣ Fused projection: left, right, left_gate, right_gate, out_gate
    # -------------------------------------------------
    # Concatenate weight matrices → shape (5*H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, w_fused)        # fp16
    del w_fused, x_norm                     # free early

    # -------------------------------------------------
    # 3️⃣ Split & apply mask + sigmoid gates
    # -------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # No mask – kernel will ignore mask
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
        has_mask = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # -------------------------------------------------
    # Reshape to [B, H, N, N] for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # Each head (and batch) performs: (N,N) @ (N,N)ᵀ → (N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # -------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out

    # Element‑wise gate multiplication (fp16)
    out_norm = out_norm * out_gate
    del out_gate

    # -------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # -------------------------------------------------
    final_half = F.linear(
        out_norm,                                 # [R, H]
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )                                            # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
