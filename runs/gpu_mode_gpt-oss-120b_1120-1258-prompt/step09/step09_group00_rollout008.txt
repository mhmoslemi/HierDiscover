"""
Tri‑Multiplicative (outgoing) update – optimized Triton + Torch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) via torch.nn.functional.layer_norm.
2️⃣  Fused linear projection for the three tensors (left, right, left_gate, right_gate, out_gate)
    using a single weight matrix (5·H × dim) – computed with torch.nn.functional.linear (FP16).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
        * splits the fused projection into left/right tensors,
        * applies the optional pair‑wise mask,
        * computes sigmoid gates (left_gate, right_gate, out_gate),
        * stores left, right and out_gate (FP16).
4️⃣  Batched GEMM over the hidden dimension:
        left[b,i,k,d] @ right[b,j,k,d]ᵀ → out[b,i,j,d]   (`torch.bmm` on (B·H,N,N) matrices).
5️⃣  Row‑wise LayerNorm + out‑gate multiplication (fused with torch.nn.functional.layer_norm).
6️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to FP32 output.

Only the mask/gate split is implemented in Triton – the rest uses highly‑optimized
PyTorch kernels (LayerNorm, GEMM, Linear). This meets the “must use a kernel”
requirement while keeping memory traffic minimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # optional mask (broadcasted over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] contiguous
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R,H] offset for seg0

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_dst_ptr     = left_ptr   + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr  + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total rows for the pairwise tensors
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm operates on the last dimension
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim], float32
    x_norm = x_norm_fp32.half().contiguous()  # convert to fp16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right & gates (5·H outputs)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear projection in fp16
    proj = F.linear(x_norm, w_concat)       # [R, 5*H], fp16
    del w_concat, x_norm                    # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # No mask – pass a dummy empty tensor, kernel will ignore it
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        # mask: [B,N,N] → flatten → fp16 (1.0 for True/keep, 0.0 for False)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256   # rows per program
    BLOCK_H = 32    # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr  # free

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H to batch dimension
    left_4d  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_4d = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_4d.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right_4d.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()   # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused via torch)
    # ------------------------------------------------------------------
    # LayerNorm expects weight/bias of the same dtype as the input
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # still fp16

    gated = out_norm * out_gate  # element‑wise gating, fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # [dim, H]
    final_half = F.linear(gated, to_out_w)                # [R, dim], fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
