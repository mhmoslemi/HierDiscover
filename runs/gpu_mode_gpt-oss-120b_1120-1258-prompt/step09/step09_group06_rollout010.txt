High‑performance implementation of the *outgoing* TriMul operator (AlphaFold3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) – Triton kernel.
2️⃣ Fuse the four projection/gate linear layers **and** the output‑gate linear
   into a *single* batched GEMM (FP16).
3️⃣ In‑place masking + sigmoid‑gates.
4️⃣ Batched cubic update:   (B·H, N, N) @ (B·H, N, N)ᵀ   – torch.bmm (Tensor‑Core).
5️⃣ Fuse the second LayerNorm, the output‑gate element‑wise multiplication and
   the final linear projection into one Triton kernel that works on FP16.
6️⃣ Cast the result back to FP32 and return.

Only the two LayerNorm‑related kernels are written in Triton; the heavy
matrix‑multiplications use cuBLAS (via torch.nn.functional.linear / torch.bmm),
which is optimal on an H100.  The implementation supports batch sizes 1‑2,
sequence lengths up to 1024 and hidden dimensions 128 (dim may be 128/384/768).

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm for FP32 input → FP16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1. mean & variance (accumulate in FP32)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -------------------------------------------------
    # 2. write normalized result (FP16)
    # -------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + out‑gate + final projection (FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp16) – hidden tensor
    y_ptr,          # mutable* [total_rows, dim] (fp16) – scaled output
    w_ptr,          # const  * [dim]          (fp32) – γ (to_out_norm)
    b_ptr,          # const  * [dim]          (fp32) – β (to_out_norm)
    gate_ptr,       # const  * [total_rows, dim] (fp16) – out_gate
    total_rows,     # int (runtime)
    dim,            # int (runtime) = hidden_dim
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm on FP16 values, multiply by a per‑element gate,
       then store the result (still FP16)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1. mean & variance (fp32 accumulation)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # 2. normalize, apply γ/β, multiply by gate and store
    # -------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw hidden value
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x‑μ)/σ

        # γ/β (fp32)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        # gate (fp16 → fp32) – out_gate ∈ (0, 1)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=1.0)                               # default = 1
        out = out * g.to(tl.float32)

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (bool/float) or None
        - weights_dict : dict with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) – row‑wise Triton kernel
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp32_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ  (fp32)
        weights["norm.bias"],        # β  (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four projections/gates **and** the output‑gate into a single GEMM
    # ------------------------------------------------------------------
    # All projection/gate weights are of shape [hidden_dim, dim] (no bias)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                         # [5·H, dim]

    # One batched GEMM:  [R, dim] @ [5·H, dim]ᵀ  →  [R, 5·H]
    fused = F.linear(x_norm_fp16, w_fused)

    # slice the fused result (views, no copy)
    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]
    out_gate_raw    = fused[:, 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (in‑place) and sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] → broadcast over H
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_raw.mul_(mask_fp16)
        right_raw.mul_(mask_fp16)

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # multiply by gates (in‑place, reuse memory)
    left_raw.mul_(left_gate)
    right_raw.mul_(right_gate)

    # rename for clarity – after this point left/right are the final tensors
    left  = left_raw          # [R, H]   fp16
    right = right_raw         # [R, H]   fp16

    # ------------------------------------------------------------------
    # 4️⃣ Batched cubic update  (left @ rightᵀ)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H into the batch dimension → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate + final projection (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()                # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous()      # [R, H] fp16

    scaled_hidden_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        scaled_hidden_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        out_gate_flat,                   # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (FP16 GEMM) and cast back to FP32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        scaled_hidden_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim]  (FP16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
