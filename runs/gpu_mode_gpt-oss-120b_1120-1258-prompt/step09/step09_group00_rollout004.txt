"""
Tri‑Multiplicative (outgoing) update – fast Triton‑accelerated implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N×dim) – performed with the
    highly‑optimised Torch `layer_norm` kernel (FP32 → FP32) and a cheap
    cast to FP16.

2️⃣  Single fused Linear that produces the five hidden‑dim tensors
    *(left, right, left_gate, right_gate, out_gate)* in one matmul
    (`R × dim` × `(5·H)`).

3️⃣  Triton kernel `split_fuse_gate_kernel`  
    – splits the fused result, applies the optional pair‑wise mask,  
      computes the three sigmoid gates and stores `left`, `right` and
      `out_gate` (all FP16).

4️⃣  Batched GEMM over the hidden dimension:  
    `left` and `right` are reshaped to `[B·H, N, N]` and multiplied with
    `torch.bmm`, which uses cuBLAS tensor‑cores.

5️⃣  Second LayerNorm + element‑wise multiplication by `out_gate` – done
    with Torch `layer_norm` (FP16) followed by a single broadcasted mul.

6️⃣  Final Linear projection back to `dim` (FP16 matmul) and cast to
    FP32 for the output.

Only step 3 uses a custom Triton kernel, satisfying the “must contain a
kernel” requirement while keeping the heavy matrix‑multiplications in
cuBLAS where they are already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]      (fp16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Optional mask (broadcasted to hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]          # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside the fused tensor:
        # [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # Load raw values
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # --------------------------------------------------------------
        # Sigmoid (computed in fp32 for better accuracy)
        # --------------------------------------------------------------
        left_gate = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate= (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate  = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # --------------------------------------------------------------
        # Apply mask and gates
        # --------------------------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # --------------------------------------------------------------
        # Store results
        # --------------------------------------------------------------
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows (B·N·N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------
    # flatten to (R, dim) and apply torch's highly optimized layer_norm
    x_flat = x.view(R, dim).contiguous()          # (R, dim) float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],            # gamma (float32)
        bias=weights["norm.bias"],                # beta  (float32)
        eps=eps,
    )
    # cast to fp16 for the following matmul
    x_norm = x_norm_fp32.half()                    # (R, dim) fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5·H hidden tensors
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each shape H×dim) → (5·H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)      # (5·H, dim)

    # Linear: (R, dim) @ (dim, 5·H)ᵀ  →  (R, 5·H)  (FP16 GEMM, tensor‑cores)
    proj = F.linear(x_norm, w_fused)               # (R, 5·H) fp16
    del w_fused, x_norm, x_norm_fp32                # free intermediate tensors

    # ------------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates (Triton kernel)
    # ------------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        # No mask → provide dummy pointer, compile‑time constant disables loading
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        # mask: [B, N, N] → [R]  (1.0 where allowed, 0.0 where masked)
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128      # rows per program
    BLOCK_H = 32       # hidden dimension tiles
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H to batch dimension: [B·H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right.view(B * H, N, N)  # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ   →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (torch kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # (R, H)
    # LayerNorm (FP16) – Torch implementation works with FP16 input
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],      # gamma (float32)
        bias=weights["to_out_norm.bias"],          # beta  (float32)
        eps=eps,
    ).to(dtype=torch.float16)                      # FP16 result

    # Element‑wise multiplication with the out‑gate (both FP16)
    gated = out_norm * out_gate                      # (R, H) fp16
    del out_flat, out_norm, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),  # (dim, H)
        bias=None,
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(dtype=torch.float32)
    return final
```
