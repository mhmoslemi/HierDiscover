    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]

"""
high‑performance forward pass for the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input channel dimension (dim).  Implemented in
   Triton (`layernorm_kernel`).  Input is FP32, output is FP16 to keep the
   subsequent GEMMs in half‑precision.

2️⃣ A single fused GEMM computes the four linear projections (left, right,
   left‑gate, right‑gate) from the normalized tensor.  The result is split
   into the four [batch·seq·seq, hidden] tensors.

3️⃣ A second linear layer produces the output‑gate (`out_gate`) from the same
   normalized input.

4️⃣ Optional pair‑wise mask is applied to the left/right projections, then
   the sigmoid‑gates are applied.

5️⃣ The pairwise multiplicative update is a batched matrix‑multiply:
   `out = left @ rightᵀ`  (shape (B·H, N, N)).  This uses cuBLAS (`torch.bmm`)
   which is optimal on H100.

6️⃣ **Fused** second LayerNorm + out‑gate multiplication + the final projection
   back to the original channel dimension are performed in a single Triton
   kernel (`layernorm_gate_linear_kernel`).  This removes an extra
   intermediate tensor and one matrix multiplication.

The heavy work is done in two Triton kernels; all other steps use cuBLAS
which is already highly tuned for the target hardware.  The implementation
supports the full range of shapes required by the test suite (B∈{1,2},
N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128) and works with or
without a mask.  The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # const   * [total_rows, dim] (float32)
    y_ptr,           # mutable * [total_rows, dim] (float16)
    weight_ptr,      # const   * [dim] (float32)   γ
    bias_ptr,        # const   * [dim] (float32)   β
    total_rows,      # int32   runtime
    dim,             # int32   runtime
    eps,             # float   compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalised values: out = γ·(x−μ)·rstd + β   (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                       # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (over hidden) + out‑gate multiplication
#                + final linear projection (to original dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,            # const   * [total_rows, hidden] (float16) — raw `out`
    gate_ptr,         # const   * [total_rows, hidden] (float16) — `out_gate`
    y_ptr,            # mutable * [total_rows, dim]    (float16) — final result
    ln_weight_ptr,    # const   * [hidden] (float32)     — to_out_norm.weight
    ln_bias_ptr,      # const   * [hidden] (float32)     — to_out_norm.bias
    out_weight_ptr,   # const   * [dim, hidden] (float16) — to_out.weight
    total_rows,       # int32   runtime
    hidden,           # int32   runtime
    dim,              # int32   runtime
    eps,              # float   compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑dim chunk
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance over hidden dimension (ignore gate)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        vals = tl.load(
            x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ for each dim‑chunk, apply LN, gate and accumulate linear product
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # load raw out and gate
            out_raw = tl.load(
                x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # [M, Hc]

            gate = tl.load(
                gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # [M, Hc]

            # layer‑norm (γ, β) on the hidden vector
            norm = (out_raw - mean[:, None]) * rstd[:, None]   # [M, Hc]

            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # [Hc]
            beta  = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)  # [Hc]

            norm = norm * gamma[None, :] + beta[None, :]         # [M, Hc]

            # apply out‑gate
            gated = norm * gate                                 # [M, Hc]

            # weight slice for final projection
            w = tl.load(
                out_weight_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # [Dblk, Hc]

            # accumulate dot(gated, wᵀ) → [M, Dblk]
            acc += tl.dot(gated, w.t())

        # store result
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor   : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict  : mapping with all Linear/LayerNorm parameters
        - config_dict   : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 32           # cols per iteration (fits dim≤384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ fused projection for left/right + their gates
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                    # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)                 # [R, 4·H]

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # out‑gate (still on the normalized input)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)      # [H, dim]
    )                                                      # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ mask (if any) + sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()       # FP16, shape [R, H]

    left  = left_raw  * left_gate    # [R, H] FP16
    right = right_raw * right_gate   # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ batched N³ multiplicative update (left @ rightᵀ)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move hidden dim into batch axis -> (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ flatten for the fused LN + gate + final linear
    # ------------------------------------------------------------------
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).reshape(total_rows, H).contiguous()
    out_gate_flat = out_gate.reshape(total_rows, H).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ fused second LayerNorm + out‑gate * LN + final linear (to dim)
    # ------------------------------------------------------------------
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M2 = 128
    BLOCK_D2 = 32          # output‑dim chunk
    BLOCK_H = 32           # hidden‑dim chunk (hidden_dim=128 → 4 iterations)

    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)

    layernorm_gate_linear_kernel[grid2](
        out_flat,                 # raw output of bmm   (FP16)
        out_gate_flat,            # out‑gate            (FP16)
        final_fp16,               # destination         (FP16)
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        weights["to_out.weight"].to(torch.float16),   # [dim, hidden] (FP16)
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ reshape to original layout and cast back to fp32
    # ------------------------------------------------------------------
    out = final_fp16.view(B, N, N, dim).to(torch.float32)
    return out
