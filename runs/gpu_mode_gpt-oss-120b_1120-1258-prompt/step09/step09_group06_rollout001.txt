"""
High‑performance forward pass for the *outgoing* TriMul operator (AlphaFold3).

Algorithm (mixed FP16 / FP32):
  1️⃣ Row‑wise LayerNorm (FP32 → FP16) implemented in Triton.
  2️⃣ Fuse the four linear projections (left/right + left/right‑gate) into a
     single FP16 GEMM:   X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ.
  3️⃣ Compute the output‑gate with a second FP16 GEMM.
  4️⃣ Optional binary mask, sigmoid‑gates and element‑wise gating (in‑place).
  5️⃣ Reshape to [B, N, N, H] and perform a strided‑batched matrix‑multiply
     (N×N) @ (N×N)ᵀ for each head → the cubic multiplicative update.
  6️⃣ Second LayerNorm over the hidden dimension (FP16) – same Triton kernel.
  7️⃣ Apply the output‑gate, final linear projection and cast back to FP32.

Only the two LayerNorms are handwritten Triton kernels; the remaining
operations leverage highly‑optimized PyTorch kernels (FP16 GEMM,
torch.bmm).  This implementation works for:
  * batch size   B ∈ {1,2}
  * seq‑len      N ∈ {128,256,512,768,1024}
  * dim          D ∈ {128,384,768}
  * hidden_dim   H = 128
and optional per‑pair masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # const  * [total_rows, dim] (fp32)
    y_ptr,           # mutable* [total_rows, dim] (fp16)
    w_ptr,           # const  * [dim]          (fp32) – gamma
    b_ptr,           # const  * [dim]          (fp32) – beta
    total_rows,      # int (runtime)
    dim,             # int (runtime)
    eps,             # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel iterates over the channel dimension in tiles of size BLOCK_D.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        # γ,β are stored in fp32 – keep the computation in fp32, cast at end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        out = norm * w[None, :] + b[None, :]                 # fp32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32) with keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight", "to_out_norm.weight",
              "to_out_norm.bias", "to_out.weight"
        - config_dict  : dict with keys:
              "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N           # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, D] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tile sizes tuned for H100 (empirically good)
    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # gamma, fp32
        weights["norm.bias"],        # beta,  fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right proj + gate into ONE GEMM (FP16)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (no bias in the reference model)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],    # [H, D]
            weights["right_proj.weight"],   # [H, D]
            weights["left_gate.weight"],    # [H, D]
            weights["right_gate.weight"],   # [H, D]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, D]

    fused = F.linear(x_norm_fp16, fused_w)   # [R, 4·H]   (FP16)

    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM, FP16)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, D]
    )   # [R, H] (FP16)

    # ------------------------------------------------------------------
    # 4️⃣ Mask + sigmoid gates (in‑place, FP16)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] -> [R, 1] broadcast
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]
    right = right_raw * right_gate        # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update: (left @ rightᵀ) per head
    #    using a strided‑batched half‑precision GEMM.
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Permute to bring H into the batch dimension: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & head → (B·H, N, N)
    left_mat  = left_perm.view(B * H, N, N)               # FP16
    right_mat = right_perm.view(B * H, N, N)              # FP16

    # (N×N) @ (N×N)ᵀ  →  (B·H, N, N)   (still FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim (FP16) – Triton reuse
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma, fp32
        weights["to_out_norm.bias"],     # beta,  fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gating, final projection, cast to fp32
    # ------------------------------------------------------------------
    # Apply the output‑gate (still FP16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # Final linear: [R, H] @ [D, H]ᵀ → [R, D] (FP16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [D, H]
    )   # [R, D] (FP16)

    # Reshape back to original [B, N, N, D] and convert to float32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
