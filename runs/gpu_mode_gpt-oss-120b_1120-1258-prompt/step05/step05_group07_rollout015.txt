@triton.jit
def trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    b_idx = pid_h // H
    h_idx = pid_h % H
    batch_offset = b_idx * stride_batch + h_idx   # base pointer for this (b,h)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (i,k)
        a_ptrs = a_ptr + batch_offset + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (k,j)  (note transposed)
        b_ptrs = b_ptr + batch_offset + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store C tile (i,j)
    c_ptrs = c_ptr + batch_offset + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Input LayerNorm → fp16 (torch.nn.functional.layer_norm, fast cuDNN kernel).
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM (torch.nn.functional.linear).
3️⃣  (Optional) apply the pair‑wise mask before the left/right gates.
4️⃣  Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣  **Outgoing triangular update** – custom Triton kernel that computes
    out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h] without any
    intermediate copies.  The kernel is a tiled fp16 GEMM with
    `tl.dot` → Tensor‑Core‑accelerated, processing each head (h) as a separate
    batch dimension (BH = B·H).
6️⃣  Second LayerNorm over the hidden dimension H (custom Triton kernel).
7️⃣  Multiply by the output gate and apply the final linear projection back to
    the original channel dimension C.
All heavy 𝒪(N³·H) work is performed by the custom kernel using Tensor Cores,
while the two LayerNorms use a fast Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,           # *const  fp16*   [total_rows, dim]
    y_ptr,           # *mut    fp16*   [total_rows, dim]
    weight_ptr,      # *const  fp32*   [dim]   (γ)
    bias_ptr,        # *const  fp32*   [dim]   (β)
    total_rows,      # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton outgoing TriMul kernel (batched GEMM on (B,N,N,H) tensors)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,          # fp16 pointers: [B,N,N,H]
    B, N, H,                      # runtime numbers
    stride_batch, stride_i, stride_k,   # strides for the tensors
    BLOCK_M: tl.constexpr,        # tile rows (i dimension)
    BLOCK_N: tl.constexpr,        # tile cols (j dimension)
    BLOCK_K: tl.constexpr,        # reduction tile (k dimension)
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_h = tl.program_id(2)          # combined (batch, head)

    # decode batch & hidden‑head index
    b_idx = pid_h // H
    h_idx = pid_h % H
    batch_offset = b_idx * stride_batch + h_idx          # base address for (b,h)

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile : left[b,i,k,h]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_offset + \
                 offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile : right[b,j,k,h] transposed -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_offset + \
                 offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # block‑wise GEMM
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # store result  out[b,i,j,h]
    c_ptrs = c_ptr + batch_offset + \
             offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]        (float32)
        - mask_tensor  : torch.Tensor [B, N, N]          (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm – use cuDNN implementation (fast)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # (B,N,N,C)  fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Each weight: [hidden_dim, dim] (no bias)
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).half().contiguous()   # (5*H, C)

    # fused projection (accepts input with arbitrary leading dims)
    proj_all = F.linear(x_norm, concat_w)   # (B,N,N,5*H)  fp16

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=-1
    )                                        # each (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    left  = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate        # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing version)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton GEMM kernel → no extra copies.
    # --------------------------------------------------------------
    out = torch.empty_like(left)           # (B,N,N,H) fp16

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),           # i‑tiles
        triton.cdiv(N, BLOCK_N),           # j‑tiles
        B * hidden_dim,                    # (batch, head) tiles
    )

    # strides for the tensor layout: (B,N,N,H) is contiguous → (B,N,N,H)
    stride_batch = left.stride(0)          # N*N*H
    stride_i     = left.stride(1)          # N*H
    stride_k     = left.stride(2)          # H

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_batch, stride_i, stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # (total_rows, H)
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, 256),)  # reuse BLOCK_M=256 from the kernel
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=256,
        BLOCK_D=64,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)       # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                           # (B,N,N,H) fp16
    gated_flat = gated.view(total_rows, hidden_dim)       # (total_rows, H)

    to_out_w = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated_flat, to_out_w)           # (total_rows, C) fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
