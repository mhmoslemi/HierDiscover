"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “tri‑multiplicative update”).

Algorithm
---------
1️⃣  Input‑wise LayerNorm (FP32 → FP16) using the native torch kernel.
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM by concatenating their weight matrices.
3️⃣  (Optional) apply the binary mask to the left/right projections.
4️⃣  Apply sigmoid to the three gate tensors and multiply them element‑wise.
5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision matrix‑multiply (`torch.bmm`);
    each head (hidden dimension) is an independent batch.
6️⃣  Fused LayerNorm + output‑gate (both FP16) performed by a tiny Triton kernel.
7️⃣  Final linear projection back to the model dimension (`dim`) – half‑precision
    GEMM (cuBLAS/Tensor‑Core).
8️⃣  Cast the result back to float32 (the original PyTorch module returns FP32).

Only the second LayerNorm + gate step is written in Triton; the remaining
operations use highly‑optimized PyTorch/cuBLAS kernels, yielding a large
speed‑up over the pure‑Python reference while keeping the implementation
correct for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,            # const half  [rows, dim]   (input to be normalized)
    y_ptr,            # mut   half  [rows, dim]   (output after LN·gate)
    weight_ptr,       # const float [dim]        (γ)
    bias_ptr,         # const float [dim]        (β)
    gate_ptr,         # const half  [rows, dim] (out_gate, already sigmoid‑ed)
    rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- 1st pass: compute mean & variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # ----- 2nd pass: normalize, apply γ,β and multiply by gate -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]           # FP32

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)          # gate (half → FP32)
        out = out * g                                   # apply out‑gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored
                         if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    inp, mask, weights, cfg = data
    device = inp.device

    dim   = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]       # H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N          # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use the highly‑optimized torch kernel
    # ------------------------------------------------------------------
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)                # FP32
    x = x.half()                               # FP16 for the subsequent GEMMs

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single half‑precision GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices: each is [hidden, dim]
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear: input shape [..., C] → [..., 5*H]
    proj_all = F.linear(x, concat_w)          # (B, N, N, 5*H)

    # Split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=-1
    )                                          # each (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Apply binary mask (if present) to the *projections* only
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)    # (B,N,N,1) in FP16
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)     # (B,N,N,H) half
    right_gate = torch.sigmoid(right_gate)    # (B,N,N,H) half
    out_gate   = torch.sigmoid(out_gate)      # (B,N,N,H) half – used later

    left  = left_proj * left_gate             # (B,N,N,H) half
    right = right_proj * right_gate           # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat)                        # (B*H,N,N)

    # Back to (B,N,N,H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out‑gate (Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden).contiguous()          # (total_rows, H)
    out_gate_flat = out_gate.view(-1, hidden).contiguous()

    out_norm_gate = torch.empty_like(out_flat)            # (total_rows, H) half

    # Triton launch configuration (tuned for H≤128)
    BLOCK_M = 64
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_flat,
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()       # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w_fp16)   # (total_rows, dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 and reshape to original layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
