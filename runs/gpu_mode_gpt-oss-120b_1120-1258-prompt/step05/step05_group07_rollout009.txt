    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # Half conversion of input
    x_fp16 = input_tensor.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    total_rows = B * N * N

    # layernorm first
    x_flat = x_fp16.view(total_rows, dim)
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # concat linear weights
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0
    ).half().contiguous()  # shape (5*H, dim)

    # fused projection
    proj_all = torch.nn.functional.linear(x_norm_fp16, concat_w)  # (total_rows, 5*H)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # apply mask if required
    if not nomask:
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    # elementwise gating
    left = left_proj_fp16 * left_gate
    right = right_proj_fp16 * right_gate

    # reshape to 4D for triangular update
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # perform batched GEMM per head
    # bring head dimension into batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batch matmul left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm using torch.nn.functional.layer_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply out gate (already of shape B,N,N,H)
    out = out * out_gate

    # final linear projection
    final = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # cast to float32
    return final.to(torch.float32)

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Convert the input tensor to fp16 and perform a row‑wise LayerNorm
   (custom Triton kernel, fp32 accumulation, fp16 output).

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   fp16 GEMM by concatenating the corresponding weight matrices.

3️⃣ (Optional) apply the pair‑wise mask before the left/right gates.

4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).

5️⃣ Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   – realized as a batched GEMM `left @ rightᵀ` where the head dimension
     `hidden_dim` is folded into the batch dimension (B·H, N, N).

6️⃣ Second LayerNorm over the hidden dimension – performed with the
   highly‑optimised cuDNN `torch.nn.functional.layer_norm` (no extra copy).

7️⃣ Multiply by the output gate and apply the final linear projection back
   to the original channel dimension.

All heavy linear algebra uses fp16 Tensor‑Core kernels; the only Triton
kernel is the first LayerNorm.  This implementation works for the full
range of test configurations (B∈{1,2}, N∈{128…1024}, C∈{128,384},
H=128) and returns a float‑32 tensor as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,      # *const fp16*  [total_rows, dim]  (input, fp16 or fp32)
    y_ptr,      # *mut    fp16*  [total_rows, dim]  (output)
    w_ptr,      # *const fp32*  [dim]               (γ)
    b_ptr,      # *const fp32*  [dim]               (β)
    total_rows, # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack & basic config
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Shape preparation
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ Input → fp16 and first LayerNorm (Triton)
    # --------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)                # [B,N,N,C] fp16
    x_flat = x_fp16.view(total_rows, dim)                  # (total_rows, C)

    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)          # (total_rows, C)

    # tuning: dim ≤ 384 → BLOCK_D = 64 works well; rows are large → BLOCK_M = 256
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # each weight: [hidden_dim, dim]  (no bias in the reference)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()   # shape (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )                                            # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)      # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape to 4‑D layout for the triangular update
    left = left.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Move the hidden dimension into the batch axis (contiguous copy)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: left @ rightᵀ   (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore 4‑D layout (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (cuDNN)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection back to C
    # --------------------------------------------------------------
    gated = out * out_gate                     # (B, N, N, H) fp16
    final = F.linear(gated, weights["to_out.weight"].half())   # (B, N, N, C) fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    return final.to(torch.float32)
