5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision matrix‑multiply (`torch.bmm`);
   each head (`hidden_dim`) is treated as an independent batch.
6️⃣  Fuse the second LayerNorm, the output‑gate and the final linear projection
    back to `dim` into a single Triton kernel (normalisation → γ/β → *out_gate
    → linear).  This eliminates an extra memory write/read.
7️⃣  Cast the result back to FP32 (as required by the reference module).

Only the two LayerNorm kernels are written in Triton; all dense linear algebra
uses cuBLAS/Tensor‑Core (already optimal on an H100).  The implementation works
for the full range of test configurations (B∈{1,2}, N∈{128,…,1024},
dim∈{128,384}, hidden_dim=128) and supports both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # channel dimension (≤ 384)
    eps: tl.constexpr,   # LayerNorm epsilon
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 256)
    BLOCK_D: tl.constexpr,   # columns processed per iteration (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused LayerNorm + out‑gate + final linear (half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    out_ptr,             # *mut   half*  [total_rows, dim]    (final output)
    weight_ptr,          # *const float* [hidden, dim]       (γ of norm)  -> note shape transposed for convenience
    bias_ptr,            # *const float* [hidden]           (β of norm)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate already sigmoid‑ed)
    linear_w_ptr,        # *const half*  [dim, hidden]        (to_out weight)
    linear_b_ptr,        # *const half*  [dim]                (to_out bias, not used in original code → zero)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (r = b * N * N + i * N + j):
        1) Compute LayerNorm on the hidden vector x[r, :]
        2) Multiply by out_gate[r, :] (already sigmoid‑ed)
        3) Apply final linear projection: y = (norm * gate) @ W_T + bias
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply γ/β, multiply by out_gate and
    #                perform final linear projection (FMA over dim)
    # -----------------------------------------------------------------
    # Initialise accumulator for the output (FP32 for accumulation)
    acc = tl.zeros([BLOCK_M, dim], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # Load the hidden slice
        x_vals = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]

        # Normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]       # [BLOCK_M, BLOCK_H]

        # Apply γ/β (γ and β are FP32 scalars per channel)
        w_norm = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ  (shape hidden)
        b_norm = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β
        norm = norm * w_norm[None, :] + b_norm[None, :]                # FP32

        # Multiply by out_gate (half -> FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)                  # [BLOCK_M, BLOCK_H]
        norm = norm * gate_vals                                      # FP32

        # ----- final linear: y = norm @ W_T (dim, hidden) -----
        # Load a slice of the weight matrix: shape [dim, BLOCK_H]
        w_linear = tl.load(linear_w_ptr + offs_h[None, :] * dim + tl.arange(0, dim)[:, None],
                           mask=mask_h[None, :], other=0.0).to(tl.float32)   # [dim, BLOCK_H]

        # Accumulate: (norm) [BLOCK_M, BLOCK_H] dot (w_linear)^T [BLOCK_H, dim]
        # We use tl.dot for each row of norm against the whole w_linear
        # Loop over rows of BLOCK_M to keep registers modest.
        for i in range(0, BLOCK_M):
            if mask_m[i] == 0:
                continue
            n_i = norm[i]                     # [BLOCK_H]
            # Compute dot for all dim at once using tl.dot (requires contiguous)
            acc_i = tl.dot(n_i, w_linear)    # [dim]
            acc[i, :] += acc_i

    # Optional bias addition (original model has no bias; we keep it zero)
    # Load bias if provided; here we assume zero so we skip.

    # Store final output (FP16)
    for i in range(0, BLOCK_M):
        if mask_m[i] == 0:
            continue
        tl.store(out_ptr + offs_m[i] * dim + tl.arange(0, dim),
                 acc[i, :].to(tl.float16))


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]     # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    # Flatten to [R, C] where R = B*N*N
    inp_flat = inp.reshape(total_rows, dim).contiguous()   # FP32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    BLOCK_M = 256
    BLOCK_D = 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                 # shape (5*H, C)

    # Linear projection – FP16, no bias
    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present) directly on the projected tensors
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)  # (R,1)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)    # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)        # (R, H) – used later

    # Apply gates
    left_fp16 = left_proj_fp16 * left_gate          # (R, H)
    right_fp16 = right_proj_fp16 * right_gate       # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left_fp16.view(B, N, N, hidden)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # Prepare batched matrices for bmm:
    #   left_mat  : (B*H, N, N)   rows = i, cols = k
    #   right_mat : (B*H, N, N)   rows = k, cols = j
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiply (Tensor‑Core, FP16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H)
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)  # (R, dim)

    # Prepare weight pointer for the second LayerNorm (γ,β are float32)
    # For the final linear we need weight matrix W_T of shape (dim, hidden)
    to_out_w_fp16 = weights["to_out.weight"].half().contiguous()  # (dim, hidden)

    # Triton kernel grid
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid2](
        out_flat,
        final_fp16,
        weights["to_out_norm.weight"],   # γ (float32, size hidden)
        weights["to_out_norm.bias"],     # β (float32, size hidden)
        out_gate,                        # out_gate already sigmoid‑ed, half
        to_out_w_fp16,                   # linear weight (dim, hidden)
        torch.empty(dim, dtype=torch.half, device=device),  # bias (zero)
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_D,   # reuse BLOCK_D (128) for hidden slicing
    )

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to FP32 as required by the reference module
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
