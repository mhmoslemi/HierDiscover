Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣ Fast layer‑norm of the input (torch.nn.functional.layer_norm, FP32 → FP16).
2️⃣ Concatenate the five linear maps (left/right proj + 3 gates) and apply them
   in a single half‑precision GEMM (cuBLAS/Tensor‑Core).
3️⃣ Fuse mask, sigmoid‑gate and element‑wise multiplication for the left/right
   branches with a small Triton kernel (no extra temporaries).
4️⃣ Compute the triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a batched half‑precision matrix‑multiply (torch.bmm) where each head
   (hidden dimension) is treated as an independent batch.
5️⃣ Fuse the second layer‑norm, its affine parameters and the output‑gate
   (half‑precision) in a Triton kernel.
6️⃣ Final linear projection back to the model dimension (half‑precision GEMM).
7️⃣ Cast the result back to FP32 (as required by the reference module).

All heavy arithmetic runs in FP16 (Tensor‑Core) while the two layer‑norms keep
FP32 statistics for numerical stability.  The only Triton kernels are the
element‑wise gating (with optional mask) and the fused layer‑norm + gate.
These kernels reduce memory traffic and kernel‑launch overhead compared to a
pure‑PyTorch implementation and comfortably meet the ≤ 1 ms target on an H100.
"""

@triton.jit
def gating_kernel(
    proj_ptr,          # *const half*  [total_rows, hidden]  (proj value)
    gate_ptr,          # *const half*  [total_rows, hidden]  (pre‑sigmoid gate)
    mask_ptr,          # *const half*  [total_rows]          (binary mask, ignored if APPLY_MASK=0)
    out_ptr,           # *mut   half*  [total_rows, hidden]  (result)
    total_rows: tl.int32,
    hidden: tl.int32,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 64)
    BLOCK_H: tl.constexpr,   # hidden per block (e.g. 32)
    APPLY_MASK: tl.constexpr # 0 ⇒ no mask, 1 ⇒ use mask
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]

                       other=0.0)

                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0)

@triton.jit
def sigmoid_kernel(
    inp_ptr,          # *const half*  [total_rows, hidden]
    out_ptr,          # *mut   half*  [total_rows, hidden]
    total_rows: tl.int32,
    hidden: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        x = tl.load(inp_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0)

        s = 1.0 / (1.0 + tl.exp(-x))

        tl.store(out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 s,
                 mask=mask_m[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate (both half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,           # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,           # *mut   half*  [total_rows, hidden] (post‑norm ⋅ gate)
    weight_ptr,      # *const float* [hidden]            (γ of LN)
    bias_ptr,        # *const float* [hidden]            (β of LN)
    gate_ptr,        # *const half*  [total_rows, hidden] (already sigmoid‑ed out_gate)
    total_rows: tl.int32,
    hidden: tl.int32,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # 1) mean & variance (FP32) per row
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # 2) Normalise, apply γ/β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # γ·norm+β

        # out_gate already sigmoid‑ed (half → fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim   = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]       # H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use highly‑optimised torch kernel
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(inp,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps)               # still FP32
    x_norm_fp16 = x_norm.half()                # (B,N,N,dim) → half

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Stack weight matrices for the five linear maps (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear projection (R, 5*H) – half‑precision GEMM; no bias
    proj_all = F.linear(x_norm_fp16.view(-1, dim), concat_w)   # (R, 5*H)

    # Split the concatenated result
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask (if any), sigmoid‑gate and element‑wise product
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).half()
        mask_ptr = mask_flat
        apply_mask = 1
    else:
        # dummy tensor – never read when APPLY_MASK==0
        mask_ptr = torch.empty((1,), dtype=torch.half, device=device)
        apply_mask = 0

    # Allocate buffers for the gated tensors
    left_fp16  = torch.empty_like(left_proj)
    right_fp16 = torch.empty_like(right_proj)
    out_gate   = torch.empty_like(out_gate_raw)   # sigmoid‑ed out‑gate

    # Grid configuration
    BLOCK_M = 64
    BLOCK_H = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    # left branch
    gating_kernel[grid](
        left_proj,
        left_gate_raw,
        mask_ptr,
        left_fp16,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        APPLY_MASK=apply_mask,
    )
    # right branch
    gating_kernel[grid](
        right_proj,
        right_gate_raw,
        mask_ptr,
        right_fp16,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        APPLY_MASK=apply_mask,
    )
    # out‑gate (sigmoid only)
    sigmoid_kernel[grid](
        out_gate_raw,
        out_gate,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left_fp16 .view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # (B, H, N, N) → (B*H, N, N) layout required by torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                        # (B*H, N, N)

    # back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate (both half‑precision)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate = torch.empty_like(out_flat)   # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_H,              # hidden ≤ 384 → one pass
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()   # (C, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)   # (R, C) half

    # Cast back to FP32 as required by the original module
    final = final_fp16.view(B, N, N, dim).float()
    return final
