@triton.jit
def row_reduce_mean_var_kernel(
    x_ptr, # [total_rows, dim]
    mean_ptr, inv_std_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d in range(0, tl.cdiv(dim, BLOCK_D)):
      offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
      mask_d = offs_d < dim
      ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
      vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
      sum_val += tl.sum(vals, axis=1)
      sum_sq += tl.sum(vals * vals, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   inv_std = tl.rsqrt(var + eps)

   # store
   tl.store(mean_ptr + offs_m, mean, mask=mask_m)
   tl.store(inv_std_ptr + offs_m, inv_std, mask=mask_m)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B,N,_,C = input_tensor.shape
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5
    nomask = config.get('nomask', True)

    total_rows = B * N * N

    # 1. Compute mean and inv_std per row using Triton
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    inv_std = torch.empty(total_rows, dtype=torch.float32, device=device)

    x_flat_fp32 = input_tensor.reshape(total_rows, dim).contiguous()
    # Launch kernel
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_reduce_mean_var_kernel[grid](
        x_flat_fp32,
        mean,
        inv_std,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2. Prepare weight matrix for 5 projections
    w_left = weights["left_proj.weight"]   # shape (H, C)
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0).half().contiguous()  # (5*H, C)

    # 3. Compute raw linear output
    x_fp16 = x_flat_fp32.half()
    proj_raw = torch.nn.functional.linear(x_fp16, concat_w)   # (total_rows, 5*H) fp16

    # 4. Apply LayerNorm scaling to projection outputs
    sum_w = concat_w.sum(dim=1).half()  # shape (5*H,)
    mu_fp16 = mean.to(torch.float16)
    inv_std_fp16 = inv_std.to(torch.float16)

    proj = (proj_raw - mu_fp16[:, None] * sum_w[None, :]) * inv_std_fp16[:, None]   # (total_rows,5*H) fp16

    # 5. Split projections
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(proj, hidden_dim, dim=1)

    # 6. Apply mask if needed (only to left/right)
    if not nomask:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # 7. Compute gate sigmoids
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # 8. Gated left and right
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape to (B,N,N,H)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # 9. Triangular multiplicative update (outgoing)
    # permute to (B, H, N, N), then bmm.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (BH,N,N) fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 10. Second LayerNorm (fp16) using Triton
    out_flat = out.view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)
    # kernel for LN already exists: layernorm_fp16_kernel
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    # 11. Gating and final projection
    gated = out_norm * out_gate  # fp16
    # final linear
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = torch.nn.functional.linear(gated_flat, to_out_w_fp16)   # (total_rows, C) fp16

    # 12. Cast back to FP32 and reshape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

1️⃣  Compute per‑row mean and inverse‑std of the raw input (FP32) with a tiny
    Triton reduction kernel – no intermediate LayerNorm output is stored.

2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM `x @ Wᵀ`.  The result is corrected for the
    LayerNorm statistics using the pre‑computed `mean`, `inv_std` and the
    per‑output‑channel sums `sum_W`.  This yields the same values as
    applying `LayerNorm → Linear` but avoids an extra memory pass.

3️⃣  Apply the optional pair‑wise mask (only to `left`/`right`).

4️⃣  Apply sigmoid gates and element‑wise gating:
       `left  = left_proj  * sigmoid(left_gate )`
       `right = right_proj * sigmoid(right_gate)`

5️⃣  Triangular multiplicative update (outgoing version):
       `out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]`
    Implemented as a batched GEMM `left @ rightᵀ` (Tensor‑Core).

6️⃣  Second LayerNorm over the hidden dimension (Trion kernel).

7️⃣  Multiply by the output gate, apply the final linear projection back to
    the original channel dimension, and cast the result to FP32.

All heavy linear algebra uses FP16 Tensor‑Core kernels.  Two Triton kernels
are used (first reduction and second LayerNorm).  The implementation works
for the full range of test cases (B∈{1,2}, N∈{128…1024},
C∈{128,384}, H=128) and returns a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise mean / inv‑std reduction (no output transform)
# ----------------------------------------------------------------------
@triton.jit
def row_mean_std_kernel(
    x_ptr,            # *const  float*   [total_rows, dim]   (input, FP32)
    mean_ptr,         # *mut    float*   [total_rows]      (output)
    inv_std_ptr,      # *mut    float*   [total_rows]      (output)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load a block of the input
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    inv_std = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean,    mask=mask_m)
    tl.store(inv_std_ptr + offs_m, inv_std, mask=mask_m)


# ----------------------------------------------------------------------
# 2️⃣  Row‑wise LayerNorm (fp16 output) – reused for the second LN
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const  float*   [total_rows, dim]   (input, fp16 or fp32)
    y_ptr,            # *mut    float16* [total_rows, dim]   (output)
    weight_ptr,       # *const  float*   [dim]              (γ)
    bias_ptr,         # *const  float*   [dim]              (β)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]        (float32)
        - mask_tensor  : torch.Tensor [B, N, N]          (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N                 # flattened batch * pairwise positions

    # --------------------------------------------------------------
    # 1️⃣  Compute mean and inv‑std per row (Triton)
    # --------------------------------------------------------------
    # make input contiguous for the kernel
    x_fp32 = input_tensor.reshape(total_rows, dim).contiguous()
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    inv_std = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_mean_std_kernel[grid](
        x_fp32,
        mean,
        inv_std,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣  Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices
    w_left    = weights["left_proj.weight"]   # (H, C)
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).half().contiguous()                     # (5*H, C)

    # Half‑precision GEMM on the raw input
    x_fp16 = x_fp32.half()
    proj_raw = F.linear(x_fp16, concat_w)    # (total_rows, 5*H) fp16

    # --------------------------------------------------------------
    # 3️⃣  Apply the LayerNorm correction to the GEMM output
    # --------------------------------------------------------------
    # sum of each weight row (used to subtract μ·ΣW)
    sum_w = concat_w.sum(dim=1).half()       # (5*H,)

    mu_fp16      = mean.to(torch.float16)    # (total_rows,)
    inv_std_fp16 = inv_std.to(torch.float16) # (total_rows,)

    # Broadcasting: (total_rows, 5*H)
    proj = (proj_raw - mu_fp16[:, None] * sum_w[None, :]) * inv_std_fp16[:, None]

    # --------------------------------------------------------------
    # 4️⃣  Split projections, apply mask & sigmoid gates
    # --------------------------------------------------------------
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )                                          # each (total_rows, H)

    if not nomask:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj  = left_proj  * mask_fp16
        right_proj = right_proj * mask_fp16

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_proj  * left_gate   # (total_rows, H)
    right = right_proj * right_gate  # (total_rows, H)

    # --------------------------------------------------------------
    # 5️⃣  Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # (B, H, N, N) → (BH, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # left @ rightᵀ  (Tensor‑Core batch GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N) fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension (Triton)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)                # (total_rows, H)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣  Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                 # (B, N, N, H) fp16
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)            # (total_rows, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)          # (total_rows, C) fp16

    # --------------------------------------------------------------
    # 8️⃣  Cast to fp32 and reshape
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
