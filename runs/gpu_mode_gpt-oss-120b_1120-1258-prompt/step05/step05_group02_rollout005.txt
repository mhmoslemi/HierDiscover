"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.
-----------------------------------------------------------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣  Fuse projection and gate weights → a single linear per side.
    * (dim → hidden) + (dim → hidden) → (dim → 2·hidden)  (no bias)
3️⃣  Optional pair‑wise mask (broadcast over hidden dim).
4️⃣  Apply sigmoid gates.
5️⃣  Tri‑multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]
    performed with a single `torch.einsum`, which on CUDA uses a
    CuTensor‑backed kernel (highly tuned for this pattern).
6️⃣  Second LayerNorm (over the hidden dimension) → FP16.
7️⃣  Multiply by the output‑gate, final linear projection (hidden → dim)
    and cast the result back to FP32.

All heavy element‑wise work (layer‑norms) is done with a tiny Triton
kernel; the dominant O(N³·H) contraction uses the fast PyTorch
einsum implementation.  The code works for any batch ≤2, any
sequence length up to 1024 and hidden/feature dimensions used in the
test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance per row (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]       # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    input_tensor : torch.Tensor[float32]  [B, N, N, dim]
    mask_tensor  : torch.Tensor[float32] [B, N, N]   (may be unused)
    weights_dict : {
        "norm.weight":            Tensor[dim],
        "norm.bias":              Tensor[dim],
        "left_proj.weight":       Tensor[hidden_dim, dim],
        "right_proj.weight":      Tensor[hidden_dim, dim],
        "left_gate.weight":       Tensor[hidden_dim, dim],
        "right_gate.weight":      Tensor[hidden_dim, dim],
        "out_gate.weight":        Tensor[hidden_dim, dim],
        "to_out_norm.weight":     Tensor[hidden_dim],
        "to_out_norm.bias":       Tensor[hidden_dim],
        "to_out.weight":          Tensor[dim, hidden_dim],
    }
    config_dict : {
        "dim": int,
        "hidden_dim": int,
        "nomask": bool (optional, default True),
    }

    Returns
    -------
    torch.Tensor[float32]  [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack and sanity‑check inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [rows, dim]

    # allocate FP16 buffer for normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # launch parameters – good for dim up to 384
    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # column block
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # back to 4‑D shape, still FP16
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (two linear layers → one concat‑linear)
    # ------------------------------------------------------------------
    # concatenate projection and gate weights (dim → 2·hidden)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, dim]

    rows = total_rows
    # Linear maps (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat = F.linear(x_norm_h.view(rows, dim), left_concat_w)   # [rows, 2·H]
    right_concat = F.linear(x_norm_h.view(rows, dim), right_concat_w)

    # split into projection & pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (shared across positions)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_pre_h = F.linear(x_norm_h.view(rows, dim), out_gate_w)   # [rows, H]
    out_gate_h = torch.sigmoid(out_gate_pre_h)                        # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [rows, 1]
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h   # [rows, H]
    right_h = right_proj_h * right_gate_h # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via a single einsum
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Einstein contraction: Σ_k left(b,i,k,h) * right(b,j,k,h)
    # on CUDA this dispatches to the cuTensor / cublasLt path (very fast)
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final linear projection, cast to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, hidden_dim)   # FP16

    final_h = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, dim] (FP16)

    # result back to FP32 as required by the reference implementation
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
