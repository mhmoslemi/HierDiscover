"""
custom_kernel
=============

Fast Triton‑accelerated forward pass for the *outgoing* TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm (FP16‑centric)
------------------------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣ One fused linear projection that produces the five hidden tensors
   (left_proj, right_proj, left_gate, right_gate, out_gate) in a single
   GEMM.
3️⃣ Optional pair‑wise mask applied only to the two projections.
4️⃣ Apply sigmoid gates → gated tensors `left` and `right`.
5️⃣ Batched matrix multiplication:
      out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
   (implemented with a single torch.matmul on strided tensors).
6️⃣ Fused second LayerNorm + output‑gate (Triton kernel).  The kernel:
   * reads the non‑contiguous result `out_perm`,
   * computes per‑row mean/variance over the hidden dimension,
   * applies γ/β,
   * multiplies element‑wise by `out_gate`,
   * writes the result as a contiguous FP16 buffer.
7️⃣ Final linear projection back to the model dimension (FP16 GEMM) and
   cast to FP32.

Only two small Triton kernels are needed (both fused), all heavy
computations run in FP16 on Tensor‑Cores, yielding a large speed‑up
over the reference PyTorch implementation while preserving numerical
stability after the first LayerNorm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   *  [total_rows, dim]   (fp32)
    y_ptr,                # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,           # const   *  [dim]               (fp32)
    bias_ptr,             # const   *  [dim]               (fp32)
    total_rows,           # int     (runtime)
    dim,                  # int     (compile‑time)
    eps,                  # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1) compute mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) write normalized + affine (γ/β) as fp16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (fp32)

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Fused second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    src_ptr,            # const   *  [B, H, N, N] (fp16)
    gate_ptr,           # const   *  [R, H]      (fp16)
    dst_ptr,            # mutable *  [R, H]      (fp16)
    ln_weight_ptr,      # const   *  [H]         (fp32)
    ln_bias_ptr,        # const   *  [H]         (fp32)
    total_rows,         # int (runtime)  = B·N·N
    hidden_dim,         # int (compile‑time)
    seq_len,            # int (runtime)  = N
    stride_b,           # int (runtime)  stride on dim 0 (B) of src
    stride_h,           # int (runtime)  stride on dim 1 (H) of src
    stride_i,           # int (runtime)  stride on dim 2 (N) of src
    stride_j,           # int (runtime)  stride on dim 3 (N) of src
    eps,                # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # decode linear row index → (b,i,j)
    # --------------------------------------------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # 1) compute mean & variance over hidden dim
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # load a tile of hidden dim (strided by stride_h)
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) normalize, apply γ/β, multiply with out_gate, store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # raw hidden values
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        # out_gate (fp16 → fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # write back as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # usually 128
    eps = 1e-5

    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 64                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) One fused linear projection that yields 5·hidden_dim values
    # --------------------------------------------------------------
    # concatenate weight matrices (no bias) and cast once to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    proj_out = F.linear(x_norm, w_concat)   # [R, 5*H]  (fp16)

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate   = proj_out[:, 2*H:3*H]
    right_gate  = proj_out[:, 3*H:4*H]
    out_gate    = proj_out[:, 4*H:5*H]       # [R, H]

    # --------------------------------------------------------------
    # 3) Sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 4) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    # reshape back to (B, N, N, H) for the batched matmul
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # --------------------------------------------------------------
    # 6) Batched matrix multiplication: out_perm[b, h, i, j] =
    #    Σₖ left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N)

    out_perm = torch.matmul(left_perm,
                            right_perm.transpose(-1, -2))  # (B, H, N, N)

    # --------------------------------------------------------------
    # 7) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Results are stored contiguously as (R, H) in fp16.
    # --------------------------------------------------------------
    stride_b = out_perm.stride(0)   # B‑stride = H·N·N
    stride_h = out_perm.stride(1)   # H‑stride = N·N
    stride_i = out_perm.stride(2)   # i‑stride = N
    stride_j = out_perm.stride(3)   # j‑stride = 1

    hidden_out = torch.empty(total_rows, H,
                             dtype=torch.float16,
                             device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid2](
        out_perm,                               # src (non‑contig)
        out_gate,                               # gate (contig, [R, H])
        hidden_out,                             # dst (contig)
        weights["to_out_norm.weight"],          # γ (fp32)
        weights["to_out_norm.bias"],            # β (fp32)
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 8) Final linear projection back to model dimension
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(hidden_out, w_out)             # [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
