"""
custom_kernel
==============

Fast forward‑pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (mixed‑precision):
1. Row‑wise LayerNorm over the input channel dimension C (FP32 reduction,
   FP16 output) – custom Triton kernel.
2. Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM by concatenating their weight matrices.
3. (Optional) apply the pair‑wise mask to the left/right projections.
4. Apply sigmoid gates and element‑wise gating (still FP16).
5. Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   Implemented as a batched half‑precision GEMM `left @ rightᵀ` on tensors of
   shape (B·H, N, N) using torch.bmm (cuBLAS Tensor‑Core kernels).
6. Row‑wise LayerNorm over the hidden dimension H (fp16 output, fp32
   reduction) – same Triton kernel as #1.
7. Multiply by the output gate, then a final half‑precision linear projection
   back to C.
8. Cast the result to FP32 and return.

All heavy reductions (LayerNorm) are executed in Triton; dense linear algebra
uses cuBLAS which delivers the best performance on an H100.  The implementation
supports both masked / un‑masked variants and works for any
`dim ∈ {128,384}` and `hidden_dim = 128` as required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  void*   [total_rows, dim]   (float16 or float32)
    y_ptr,          # *mut    float16* [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const  float*   [dim]               (γ, fp32)
    bias_ptr,       # *const  float*   [dim]               (β, fp32)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per loop iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # 1. Compute mean & variance (fp32) – reduction over dim
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in chunks of BLOCK_D
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim

        # 2‑D pointer: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load (fp16 or fp32) and up‑cast to fp32
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                        # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                 # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # -------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]         # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict  : mapping of weight names to tensors (float32)
            - config_dict   : {"dim": C, "hidden_dim": H,
                               "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    # Flatten to [total_rows, dim] – no copy, just a view
    x_flat = input_tensor.view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Launch kernel – tuned block sizes
    BLOCK_M = 1024          # rows per program (larger grid‑size reduction)
    BLOCK_D = 64           # columns per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (fp32)
        weights["norm.bias"],            # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    # Cast weight matrices once to half‑precision
    w_left    = weights["left_proj.weight"].half()
    w_right   = weights["right_proj.weight"].half()
    w_lgate   = weights["left_gate.weight"].half()
    w_rgate   = weights["right_gate.weight"].half()
    w_outgate = weights["out_gate.weight"].half()

    # Concatenate along the output dimension: shape [(5*H), C]
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).contiguous()

    # Fused GEMM:  [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]  (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # (half)

    # Split back into the five tensors (each [total_rows, H])
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask → half and broadcast to [total_rows, 1]
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj  *= mask_fp16
        right_proj *= mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_proj   *= torch.sigmoid(left_gate)
    right_proj  *= torch.sigmoid(right_gate)
    out_gate_fp = torch.sigmoid(out_gate)          # keep for later

    # Reshape back to 4‑D layout: [B, N, N, H]
    left  = left_proj.view(B, N, N, hidden_dim)
    right = right_proj.view(B, N, N, hidden_dim)
    out_gate = out_gate_fp.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Move hidden dimension into batch axis to obtain shape (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul: left @ rightᵀ   →  [B·H, N, N] (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [total_rows, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                         # fp16 input (cast to fp32 inside)
        out_norm_fp16,
        weights["to_out_norm.weight"],    # γ (fp32)
        weights["to_out_norm.bias"],      # β (fp32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # element‑wise FP16

    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # (half) → [total_rows, C]

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
