def custom_kernel(data):
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]; hidden = cfg["hidden_dim"]; eps = cfg.get("eps",1e-5); nomask = cfg.get("nomask", True)
    B,N,_,_ = inp.shape
    total_rows = B * N * N

    # 1 Input LN using torch
    x_norm_fp16 = torch.nn.functional.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).half()
    # flatten
    x_norm_fp16 = x_norm_fp16.view(total_rows, dim)

    # 2 fused Linear for projections and gates
    concat_w = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"], weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).half()
    proj_all = torch.nn.functional.linear(x_norm_fp16, concat_w)  # note: F.linear call is fine (torch.nn.functional.linear)
    # It's also provided as torch.randn but we want no bias
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(proj_all, hidden, dim=1)

    # mask & gate
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)  # broadcast across hidden
        left_gate = torch.sigmoid(left_gate_fp16)  # (R, H)
        right_gate = torch.sigmoid(right_gate_fp16)
        left = left_proj_fp16 * (mask_flat * left_gate)
        right = right_proj_fp16 * (mask_flat * right_gate)
    else:
        left_gate = torch.sigmoid(left_gate_fp16)
        right_gate = torch.sigmoid(right_gate_fp16)
        left = left_proj_fp16 * left_gate
        right = right_proj_fp16 * right_gate

    out_gate = torch.sigmoid(out_gate_fp16)  # (R, H)

    # reshape to batched matrices
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right = right.view(B, N, N, hidden).permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # batched matrix multiply
    out_mat = torch.bmm(left, right)  # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6. fused layernorm + out_gate using Triton
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate_fp16 = torch.empty_like(out_flat)  # half

    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # final linear
    final_fp16 = torch.nn.functional.linear(out_norm_gate_fp16, weights["to_out.weight"].half())
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16) – use the native torch.layer_norm kernel.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single half‑precision GEMM by concatenating the weight matrices.
3️⃣  Apply binary mask (if present) *and* the left/right sigmoid gates in a
    single element‑wise multiplication, thus cutting one pass over the data.
4️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision matrix multiply (`torch.bmm`);
    each head (hidden dimension) is treated as an independent batch.
5️⃣  Fused LayerNorm + output‑gate (both FP16) – custom Triton kernel.
6️⃣  Final linear projection back to the model dimension (`dim`) – half‑precision
    GEMM through `torch.nn.functional.linear`.
7️⃣  Cast back to FP32 (the original PyTorch module returns float32).

Only the two LayerNorm steps are written in Triton; all dense linear algebra
uses cuBLAS/Tensor‑Core which is already optimal on an H100.  The implementation
handles the presence/absence of a mask and works for the full range of test
configurations (B∈{1,2}, N∈{128,…,1024}, dim∈{128,384}, hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # const half  [total_rows, dim]  (pre‑norm)
    y_ptr,          # mut   half  [total_rows, dim]  (post‑norm·gate)
    weight_ptr,     # const float [dim]            (γ)
    bias_ptr,       # const float [dim]            (β)
    gate_ptr,       # const half  [total_rows, dim] (out_gate, already sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_D: tl.constexpr,   # columns per iteration (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        # (x‑µ)/σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (FP32)
        out = norm * w[None, :] + b[None, :]                      # FP32

        # multiply by the already‑sigmoided out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim  = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]      # H (always 128 in the tests)
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                         # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use the highly‑optimised torch kernel
    # ------------------------------------------------------------------
    # layer_norm returns FP32; we immediately cast to FP16 for the rest of the pipeline.
    x_norm_fp16 = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                     # (B, N, N, dim) FP16
    x_norm_fp16 = x_norm_fp16.view(total_rows, dim)   # (R, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left/right proj + three gates)
    # ------------------------------------------------------------------
    # Concatenate weight matrices (each [hidden, dim]) → [5*hidden, dim]
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    # Linear projection (R, 5*H) – half precision, no bias
    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present) + sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    # Sigmoid on gates (still half)
    left_gate  = torch.sigmoid(left_gate_fp16)    # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)   # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)    # (R, H) – used later

    if not nomask:
        # mask: [B, N, N] → [R, 1] (broadcast over hidden dim)
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left  = left_proj_fp16  * (mask_flat * left_gate)   # (R, H)
        right = right_proj_fp16 * (mask_flat * right_gate)  # (R, H)
    else:
        left  = left_proj_fp16  * left_gate
        right = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) then permute so that each head is a separate batch.
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right = right.view(B, N, N, hidden).permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiplication using Tensor‑Core FP16 (cuBLAS)
    out_mat = torch.bmm(left, right)                # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + output‑gate (both FP16) – Triton
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()          # (R, H) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)               # (R, H) half

    # Tune block sizes for the small hidden dimension (≤384)
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                 # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(out_norm_gate_fp16, weights["to_out.weight"].half())  # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)                      # (B, N, N, dim) FP32

    return final
