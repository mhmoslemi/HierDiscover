"""
custom_kernel
==============

Fast forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (mixed FP16/FP32):
1️⃣ Row‑wise LayerNorm over the input channel dimension C (FP32 accumulator,
   FP16 output) – executed by a custom Triton kernel.
2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   GEMM by concatenating their weight matrices.
3️⃣ (Optional) apply the pair‑wise mask to the left/right projections.
4️⃣ Apply sigmoid gates and element‑wise gating (still FP16).
5️⃣ Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   Implemented as a batched GEMM `left @ rightᵀ` (B·H independent GEMMs
   of size N×N) using `torch.bmm` which maps to cuBLAS Tensor‑Core kernels.
6️⃣ Row‑wise LayerNorm over the hidden dimension H (same Triton kernel).
7️⃣ Multiply by the output gate and a final linear projection back to C.
8️⃣ Cast the result to FP32 and return.

All heavy reductions (the two LayerNorms) run on Triton; dense linear algebra
uses cuBLAS which is optimal on an H100.  The implementation works for
any `dim ∈ {128,384}` and `hidden_dim = 128` as required by the test suite,
with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const   *  [total_rows, dim]   (float16 or float32 input)
    y_ptr,          # mut       *  [total_rows, dim]   (fp16 output)
    weight_ptr,     # const     *  [dim]               (γ, fp32)
    bias_ptr,       # const     *  [dim]               (β, fp32)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per loop iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean & variance (fp32) over the channel dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a BLOCK_M × BLOCK_D tile, up‑cast to fp32
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])



# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                               # flatten B×N×N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256          # rows per program – fits well in shared memory
    BLOCK_D = 64           # columns per tile  – 64 is a multiple of 32 (warp)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and three gates into a single GEMM
    # --------------------------------------------------------------
    # All linear layers have weight shape [hidden_dim, dim] and no bias
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate on the output dimension → [(5*H), C]
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()   # FP16

    # Fused linear: [R, C] @ [C, 5*H]^T → [R, 5*H]  (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)           # shape [R, 5*H]

    # Split back
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )                                                    # each [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating, same as reference)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B, N, N] → flatten → [R, 1]
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to 4‑D layout required for the multiplicative update
    left  = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)       # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Permute to [B, H, N, N] then view as (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                    # [BH, N, N] FP16
    right_mat = right_perm.view(BH, N, N)                   # [BH, N, N] FP16

    # Batched matrix multiplication (cuBLAS Tensor‑Core)
    out_mat = torch.empty_like(left_mat)
    torch.bmm(left_mat, right_mat.transpose(1, 2), out=out_mat)   # [BH, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                               # FP16 element‑wise

    # Linear projection: (R, H) @ (C, H)^T → (R, C)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)          # FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
