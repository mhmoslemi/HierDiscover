            mask_h = offs_h < hidden_dim

            out_raw = tl.load(out_ptr + row[:, None] * hidden_dim + offs_h[None, :],
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            # load gate
            gate = tl.load(out_gate_ptr + row[:, None] * hidden_dim + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # fp16 -> fp32
            # load gamma, beta
            gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
            beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)    # fp32

            # apply LN weight and bias: norm_scaled = norm * gamma + beta
            norm_scaled = norm * gamma[None, :] + beta[None, :] # fp32
            # apply gate
            gated = norm_scaled * gate
            # load weight block for current out_dim block
            w = tl.load(weight_ptr + offs_o[:, None] * hidden_dim + offs_h[None, :],
                        mask=mask_o[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)  # weight shape [out_dim, hidden_dim]

            # accumulate dot product: gated (BLOCK_R, BLOCK_H) * w (BLOCK_O, BLOCK_H).T?
            # We need sum_h gated * w[d, h]. So compute tl.dot(gated, w.T) but w has shape [BLOCK_O, BLOCK_H];
            # We can compute using tl.dot(gated, w.T) => shape [BLOCK_R, BLOCK_O].
            acc += tl.dot(gated, w, trans_b=True)  # w^T, proper orientation
        # store acc to result
        result_ptrs = result_ptr + row[:, None] * out_dim + offs_o[None, :]
        tl.store(result_ptrs, acc.to(tl.float16), mask=mask_r[:, None] & mask_o[None, :])

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    device = input_tensor.device

    # 1. First LayerNorm
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)
    BLOCK_M=256; BLOCK_D=32

    layernorm_kernel[grid](x_flat, x_norm,
                           weights["norm.weight"], weights["norm.bias"],
                           total_rows, dim, eps,
                           BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
                           num_warps=4)

    # 2. Fuse projection+gate+out_gate
    w_concat = torch.cat([weights["left_proj.weight"],
                          weights["right_proj.weight"],
                          weights["left_gate.weight"],
                          weights["right_gate.weight"],
                          weights["out_gate.weight"]], dim=0).to(torch.float16)  # [5*H, dim]

    H = hidden_dim
    left_proj_raw  = proj[:, 0:H]
    right_proj_raw = proj[:, H:2*H]
    left_gate_raw  = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw   = proj[:, 4*H:5*H]

    # 3. Optional mask (applies only to left/right)
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # 4. Sigmoid gates (in-place)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # 5. Reshape for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Perform batched matmul using torch.bmm (or custom kernel)
    out_raw_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_raw = out_raw_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_raw_flat = out_raw.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()

    # 6. Fused second LN + gating + final linear
    final_half = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Determine block sizes
    BLOCK_R = 128
    BLOCK_H = 32
    BLOCK_O = 32

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    fused_ln_linear_kernel[grid2](

        out_gate_flat,
        weights["to_out.weight"].to(torch.float16),

        final_half,

        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    final = final_half.view(B, N, N, dim).to(torch.float32)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (B·N·N × dim) → fp16.
2️⃣ Fuse *left_proj*, *right_proj*, *left_gate*, *right_gate* and *out_gate*
   linear layers into a single GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H].
3️⃣ Apply optional mask only to the first two H‑sized chunks (left/right).
   In‑place sigmoid on the three H‑sized gate chunks.
4️⃣ Compute the outgoing multiplicative update
   `out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]` via a batched GEMM
   (torch.bmm on (B·H, N, N) @ (B·H, N, N)ᵀ).
5️⃣ Fuse the second LayerNorm, the output‑gate multiplication and the final
   projection `to_out` into a single Triton kernel:
   for each row r (shape H) we compute
        µ, σ² → rstd
        y = ((out_raw - µ)·rstd)·γ + β               (LayerNorm)
        y = y·out_gate                               (output gate)
        out = y @ W_outᵀ                               (linear)
   The fused kernel eliminates intermediate tensors and reduces memory traffic.
6️⃣ Reshape to [B, N, N, dim] and cast to float32 (the reference dtype).

The implementation reduces the number of GEMMs from three to two
(5·H‑projection replaces 4·H + H) and replaces the second LayerNorm+linear
by a single Triton kernel, delivering a significant speed‑up on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # const   * [total_rows, dim] (float32)
    y_ptr,       # mutable * [total_rows, dim] (float16)
    weight_ptr,  # const   * [dim] (float32)   -- γ
    bias_ptr,    # const   * [dim] (float32)   -- β
    total_rows,  # int (runtime)
    dim,         # int (runtime)
    eps,         # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- 1️⃣ mean / variance -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ---- 2️⃣ write normalized values (γ·(x‑μ)·rstd + β) --------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)·rstd

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LN (hidden dim) + out‑gate + final linear
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_out_linear_kernel(
    inp_ptr,          # const   * [total_rows, H]  (fp16)   -- raw output of TriMul
    gate_ptr,         # const   * [total_rows, H]  (fp16)   -- out_gate
    w_out_ptr,        # const   * [dim, H]        (fp16)   -- to_out.weight
    gamma_ptr,        # const   * [H]             (fp32)   -- to_out_norm.weight
    beta_ptr,         # const   * [H]             (fp32)   -- to_out_norm.bias
    out_ptr,          # mutable * [total_rows, dim] (fp16) -- final result
    total_rows,       # int (runtime)
    H,                # int (hidden_dim)
    dim,              # int (output dim)
    eps,              # float (compile‑time constexpr)
    BLOCK_R: tl.constexpr,   # rows per block
    BLOCK_H: tl.constexpr,   # hidden‑dim block
    BLOCK_O: tl.constexpr,   # output‑dim block
):
    """
    For each row r:
        y = ((inp - µ) / sqrt(var+eps)) * γ + β        # LayerNorm
        y = y * gate                                   # output gate
        out = y @ W_outᵀ                               # final linear
    All arithmetic is in fp32, result stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------------------------------------------------------
    # 1️⃣ Compute mean & variance of the raw TriMul output
    # ---------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H
        ptr = inp_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ---------------------------------------------------------
    # 2️⃣ For each output‑dim block, accumulate dot products
    # ---------------------------------------------------------
    for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)   # (rows, out_block)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
            mask_h = offs_h < H

            # ----- load raw TriMul values and corresponding gate -----
            ptr_inp = inp_ptr + offs_r[:, None] * H + offs_h[None, :]
            raw = tl.load(ptr_inp,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)            # [BR, BH]

            ptr_gate = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            gate = tl.load(ptr_gate,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)           # [BR, BH]

            # ----- LayerNorm scaling (γ,β) -----
            gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # [BH] fp32
            beta  = tl.load(beta_ptr   + offs_h, mask=mask_h, other=0.0)   # [BH] fp32

            # normalized = (raw - mean) * rstd
            norm = (raw - mean[:, None]) * rstd[:, None]

            # LN: norm * γ + β
            ln_val = norm * gamma[None, :] + beta[None, :]               # [BR, BH]

            # apply output gate
            gated = ln_val * gate                                        # [BR, BH]

            # ----- load weight block of to_out (dim × H) -----
            # weight layout: [dim, H] row‑major → offset = out_id * H + hidden_id
            w_ptr = w_out_ptr + offs_o[:, None] * H + offs_h[None, :]   # [BO, BH]
            w = tl.load(w_ptr,
                        mask=mask_o[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)                        # fp32

            # ----- accumulate dot product (gated @ wᵀ) -----
            # gated: [BR, BH] ; wᵀ : [BH, BO] → result [BR, BO]
            acc += tl.dot(gated, w, trans_b=True)                        # [BR, BO]
        # ----- store completed output block -----
        out_ptrs = out_ptr + offs_r[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim", "nomask" flag

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, their gates and the out‑gate
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the five weight matrices (no bias) -> [5·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # fp16

    # One big GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]
    proj = F.linear(x_norm, w_concat)                       # fp16, shape [R,5·H]

    # Split the concatenated result
    left_proj_raw  = proj[:, 0:H]                # [R, H]
    right_proj_raw = proj[:, H:2*H]               # [R, H]
    left_gate_raw  = proj[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj[:, 3*H:4*H]            # [R, H]
    out_gate_raw   = proj[:, 4*H:5*H]            # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to left/right)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)  # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates (in‑place) and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_raw_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to [B, N, N, H]
    out_raw = out_raw_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_raw_flat = out_raw.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Fused second LayerNorm + output‑gate + final linear
    # --------------------------------------------------------------
    final_half = torch.empty(total_rows, dim,
                             dtype=torch.float16,
                             device=device)

    # Prepare constant tensors for the fused kernel
    gamma_out = weights["to_out_norm.weight"]      # fp32, shape [H]
    beta_out  = weights["to_out_norm.bias"]        # fp32, shape [H]
    w_out_fp16 = weights["to_out.weight"].to(torch.float16)   # [dim, H] fp16

    BLOCK_R = 128      # rows per program (tune for the hardware)
    BLOCK_H = 32       # hidden‑dim tile
    BLOCK_O = 32       # output‑dim tile
    grid2 = (triton.cdiv(total_rows, BLOCK_R),)

    fused_ln_out_linear_kernel[grid2](
        out_raw_flat,                     # inp_ptr   [R, H] fp16
        out_gate_flat,                    # gate_ptr  [R, H] fp16
        w_out_fp16,                       # w_out_ptr [dim, H] fp16
        gamma_out,                        # gamma_ptr [H]   fp32
        beta_out,                         # beta_ptr  [H]   fp32
        final_half,                       # out_ptr   [R, dim] fp16
        total_rows,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Cast to float32 and return
    # --------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
