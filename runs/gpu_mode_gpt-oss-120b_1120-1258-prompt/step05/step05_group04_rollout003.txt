@triton.jit
def fused_layernorm_gate_linear_kernel(
    x_ptr,          # *const half*  [total_rows, hidden]
    gate_ptr,       # *const half*  [total_rows, hidden]
    norm_w_ptr,     # *const float* [hidden]
    norm_b_ptr,     # *const float* [hidden]
    out_w_ptr,      # *const half*  [dim, hidden] // stored as row-major (dim, hidden)
    out_ptr,        # *mut float*   [total_rows, dim]
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_d = tl.program_id(1)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # load hidden dim chunk
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    # Load x and gate as half, then cast to fp32 for stats
    x_half = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :],
                     other=0.0)
    g_half = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :],
                     other=0.0)

    x_fp32 = x_half.to(tl.float32)
    g_fp32 = g_half.to(tl.float32)

    # compute mean and var
    sum_val = tl.sum(x_fp32, axis=1)
    sum_sq = tl.sum(x_fp32 * x_fp32, axis=1)
    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # load norm weight and bias
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)

    # apply layernorm, then gate
    norm_fp32 = (x_fp32 - mean[:, None]) * rstd[:, None] # (BLOCK_M, hidden)
    norm_fp32 = norm_fp32 * w_norm[None, :] + b_norm[None, :]
    gated_fp32 = norm_fp32 * g_fp32

    # cast to half for dot
    gated_half = gated_fp32.to(tl.float16)

    # output dim block
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load weight slice (hidden, block_d) as half -> transpose because weight is [dim, hidden]
    # out_w_ptr is pointer to [dim, hidden]; we want hidden x block_d
    # row-major, stride for dim dimension = hidden, for hidden dimension = 1?
    # Actually memory layout: out_w[dim][hidden], row-major: offset = i * hidden + j.
    # We need to load hidden rows (hidden) and block_d columns (d). 
    # Use two-dim indexing: base = offs_h[:, None] + offs_d[None, :] * hidden
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0)  # shape (hidden, BLOCK_D), half

    # dot product (BLOCK_M, hidden) x (hidden, BLOCK_D) -> (BLOCK_M, BLOCK_D)
    out_chunk_fp16 = tl.dot(gated_half, out_w_T)  # half result

    # cast to fp32 and store
    out_chunk_fp32 = out_chunk_fp16.to(tl.float32)

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk_fp32,
             mask=mask_m[:, None] & mask_d[None, :])
```

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm – use the fast PyTorch `F.layer_norm` (FP32
    reduction → FP16 output).  This removes a custom Triton kernel
    and its memory traffic.

2️⃣  Fuse the five linear projections (left/right + three gates) into a
    single FP16 GEMM (`F.linear`).  The weight matrix has shape
    (5·H, C) and is cast to half once.

3️⃣  (Optional) apply the binary mask to the left/right projections.

4️⃣  Sigmoid on the three gate tensors, element‑wise gating and a final
    reshape to (B,H,N,N).

5️⃣  Triangular multiplicative update – a batched matrix multiply
    `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` which is
    `left @ rightᵀ` for each hidden channel.  Implemented with an
    `torch.bmm` on FP16 tensors.

6️⃣  **Fused** LayerNorm + out‑gate + final linear projection:
    a custom Triton kernel reads the flattened `(B·N·N, H)` tensor,
    computes per‑row mean/variance (FP32), applies `to_out_norm`,
    multiplies by the sigmoid‑gated `out_gate` and projects with the
    `to_out` weight (stored in FP16).  The result is written directly
    as FP32, avoiding two separate kernels and extra memory copies.

The whole pipeline stays on‑device, returns a `float32` tensor of shape
`[B, N, N, C]` and satisfies all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final projection (all FP16 where possible)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_linear_kernel(
    x_ptr,            # const half*  [total_rows, hidden]   (pre‑norm)
    gate_ptr,         # const half*  [total_rows, hidden]   (out_gate, sigmoid’ed)
    norm_w_ptr,       # const float* [hidden]              (γ of to_out_norm)
    norm_b_ptr,       # const float* [hidden]              (β of to_out_norm)
    out_w_ptr,        # const half*  [dim, hidden]         (to_out weight)
    out_ptr,          # *mut float*  [total_rows, dim]    (final FP32 output)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_D: tl.constexpr,   # output‑channel chunk (e.g. 64)
):
    pid = tl.program_id(0)          # row block index
    pid_d = tl.program_id(1)        # output‑channel block index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the whole hidden vector for each selected row (BLOCK_M × hidden)
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    # Load x and gate as half, cast to fp32 for statistics
    x_h = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                  mask=mask_m[:, None] & mask_h[None, :],
                  other=0.0)                      # (BLOCK_M, hidden) half
    g_h = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                  mask=mask_m[:, None] & mask_h[None, :],
                  other=0.0)                      # (BLOCK_M, hidden) half

    x = x_h.to(tl.float32)
    g = g_h.to(tl.float32)

    # -----------------------------------------------------------------
    # Compute per‑row mean / variance (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                       # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                   # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    # (x - μ) / σ
    norm = (x - mean[:, None]) * rstd[:, None]
    # γ,β
    norm = norm * w_norm[None, :] + b_norm[None, :]
    # gate
    gated = norm * g                                   # (BLOCK_M, hidden)

    # Cast to half for the final matmul (Tensor‑Core friendly)
    gated_h = gated.to(tl.float16)

    # -----------------------------------------------------------------
    # Output‑channel block handling
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load the slice of the final weight matrix (hidden × BLOCK_D)
    # to_out weight has shape [dim, hidden] → we need its transpose.
    out_w_T = tl.load(
        out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
        mask=mask_h[:, None] & mask_d[None, :],
        other=0.0)                                     # (hidden, BLOCK_D) half

    # Compute mat‑mul: (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk_h = tl.dot(gated_h, out_w_T)            # half

    # Cast back to FP32 and store
    out_chunk = out_chunk_h.to(tl.float32)

    tl.store(
        out_ptr + offs_m[:, None] * dim + offs_d[None, :],
        out_chunk,
        mask=mask_m[:, None] & mask_d[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use PyTorch for speed)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # (R, C) float32
    x_norm_fp32 = F.layer_norm(
        inp_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm_fp32.to(torch.half)                  # (R, C) half

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate (single FP16 GEMM)
    # --------------------------------------------------------------
    # Concatenate projection and gate weight matrices (each [H, C])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                     # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)                 # (R, 5*H) half

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                                          # each (R, H) half

    # --------------------------------------------------------------
    # 2a) Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N) -> (R, 1) broadcast
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)                 # (R, H) half
    right_gate = torch.sigmoid(right_gate_fp16)               # (R, H) half
    out_gate = torch.sigmoid(out_gate_fp16)                   # (R, H) half (used later)

    left = left_proj_fp16 * left_gate                         # (R, H) half
    right = right_proj_fp16 * right_gate                      # (R, H) half

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H)
    left = left.view(B, N, N, hidden)                         # (B, N, N, H)
    right = right.view(B, N, N, hidden)                       # (B, N, N, H)

    # prepare for batched matmul: left @ rightᵀ per hidden channel
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N) half

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) half
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    # Cast final linear weight to half once
    to_out_weight_fp16 = weights["to_out.weight"].half().contiguous()

    BLOCK_M = 128                     # rows per program (tuned)
    BLOCK_D = 64                      # output‑channel chunk
    grid = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    fused_layernorm_gate_linear_kernel[grid](
        out_flat,                                   # x_ptr
        out_gate,                                   # gate_ptr (sigmoid’ed)
        weights["to_out_norm.weight"],              # γ
        weights["to_out_norm.bias"],                # β
        to_out_weight_fp16,                         # to_out weight (dim, hidden) half
        final_fp32,                                 # output buffer (FP32)
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)               # [B, N, N, dim] float32
    return final
