def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Steps:
    1. Layer‑norm (FP32) using torch.nn.functional.layer_norm, then cast to FP16.
    2. Fuse projection and gate linear layers for left/right and compute sigmoid gates.
    3. Optional mask multiplication.
    4. Tri‑multiplicative update via batched GEMM (torch.bmm).
    5. Second layer‑norm (FP16 → FP16) implemented in a small Triton kernel.
    6. Apply output gate, final linear projection, and cast back to FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # 1️⃣ First LayerNorm in FP32 (torch) and cast to FP16
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half()   # [B,N,N,dim] FP16

    rows = B * N * N

    # Cast weights required for Linear operations to FP16 once
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"].half()

    # 2️⃣ Linear projections (fused proj+gate) for left and right
    x_flat = x_norm.view(rows, dim)
    left_concat = F.linear(x_flat, left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_flat, right_concat_w) # [rows, 2*H]

    left_proj_h = left_concat[:, :hidden_dim]
    left_gate_pre_h = left_concat[:, hidden_dim:]
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # 3️⃣ Output gate (shared)
    out_gate_h = torch.sigmoid(F.linear(x_flat, out_gate_w))  # [rows, H]

    # 4️⃣ Optional mask application
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).half()
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # 5️⃣ Apply sigmoid gates
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # 6️⃣ Tri‑multiplicative update via batched GEMM
    # reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [BH, N, N]
    right_mat = right_perm.view(BH, N, N) # [BH, N, N]

    # (N×N) @ (N×N)^T  → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # 7️⃣ Second LayerNorm (FP16) via Triton kernel
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M2 = 128
    BLOCK_D2 = hidden_dim  # hidden_dim is 128, fits into a single block
    grid2 = (triton.cdiv(rows_h, BLOCK_M2),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        to_out_norm_w,
        to_out_norm_b,
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    # 8️⃣ Apply output gate and final projection
    gated = out_norm * out_gate_h.view(B, N, N, hidden_dim)
    final_fp16 = F.linear(gated.view(rows_h, hidden_dim), to_out_w)  # [rows, dim] FP16
    final = final_fp16.view(B, N, N, dim).float()  # cast back to FP32

    return final

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton + PyTorch.

Algorithm (FP16‑first, FP32‑output)
-----------------------------------
1️⃣  Layer‑norm on the input tensor (FP32) using torch.nn.functional.layer_norm,
    then cast the normalized tensor to FP16.
2️⃣  Fuse each projection with its gate (dim→hidden) via a single Linear
    (weights are pre‑concatenated).  Apply the sigmoid gates.
3️⃣  (Optional) multiply the projected tensors by the pair‑wise mask.
4️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    realised as a batched GEMM (torch.bmm) after reshaping to
    [B·H, N, N] matrices.
5️⃣  Second Layer‑norm (over the hidden dimension) performed by a small
    Triton kernel that works in FP16 → FP16.
6️⃣  Multiply by the output gate, apply the final linear projection
    (hidden→dim) and cast the result back to FP32.

All heavy arithmetic is in FP16 to exploit Tensor‑Cores on an H100.
The Triton kernel is used for the second LayerNorm, satisfying the
“must use a kernel” requirement while the first LayerNorm is delegated
to the highly‑optimised PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,     # *float  [total_rows, dim] – input (FP32)
    y_ptr,     # *half   [total_rows, dim] – output (FP16)
    gamma_ptr, # *float  [dim]               – scale
    beta_ptr,  # *float  [dim]               – bias
    total_rows,               # int (runtime)
    dim: tl.constexpr,        # static number of columns
    eps: tl.constexpr,        # static epsilon
    BLOCK_M: tl.constexpr,    # rows processed per program
    BLOCK_D: tl.constexpr,    # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- pass 1 : compute mean and variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---- pass 2 : write normalized values (γ·x̂+β) ---------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32
        out_fp16 = out.to(tl.float16)                # cast to FP16

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor[float32/bool] shape [B, N, N]
        - weights_dict : mapping from weight name to torch.Tensor (FP32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – use the highly‑optimised PyTorch impl
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half()                 # [B, N, N, dim]  FP16

    rows = B * N * N

    # ------------------------------------------------------------------
    # Prepare FP16‑cast weights (performed once)
    # ------------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                                    # [2·H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                                    # [2·H, dim]

    out_gate_w = weights["out_gate.weight"].half()   # [H, dim]

    to_out_norm_w = weights["to_out_norm.weight"]    # FP32 (used in Triton kernel)
    to_out_norm_b = weights["to_out_norm.bias"]      # FP32
    to_out_w = weights["to_out.weight"].half()      # [dim, H]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left & right paths
    # ------------------------------------------------------------------
    x_flat = x_norm.view(rows, dim)                 # [rows, dim]

    left_concat = F.linear(x_flat, left_concat_w)   # [rows, 2·H]
    right_concat = F.linear(x_flat, right_concat_w) # [rows, 2·H]

    # Split projection and gate pre‑activations
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Out‑gate (shared for all positions)
    out_gate_h = torch.sigmoid(F.linear(x_flat, out_gate_w))  # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).half()          # [rows, 1]
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h       # [rows, H]
    right_h = right_proj_h * right_gate_h     # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # bring H into the batch dimension → [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)            # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)

    # (N×N) @ (N×N)^T  → sum over the k‑dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) – Triton implementation
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M2 = 128                     # rows per program
    BLOCK_D2 = hidden_dim              # fits the whole hidden dimension (128)
    grid2 = (triton.cdiv(rows_h, BLOCK_M2),)

    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        to_out_norm_w,
        to_out_norm_b,
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection and cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, hidden_dim)   # element‑wise FP16

    # Final linear projection hidden → dim (no bias)
    final_fp16 = F.linear(gated.view(rows_h, hidden_dim), to_out_w)  # [rows, dim]

    # Cast to float32 to match the original module's output dtype
    final = final_fp16.view(B, N, N, dim).float()

    return final
