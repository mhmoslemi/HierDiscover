#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass
===================================================================

Algorithm (forward only)
------------------------
1️⃣  Row‑wise LayerNorm on the input tensor (torch.nn.functional.layer_norm,
    FP32 → FP16 conversion done afterwards).

2️⃣  One fused linear projection that simultaneously produces the five
    required tensors (left_proj, right_proj, left_gate, right_gate, out_gate)
    by a single GEMM (FP16 Tensor‑Core).

3️⃣  Optional pair‑wise mask applied only to the two projection tensors.

4️⃣  Apply sigmoid gates and obtain gated tensors *left* and *right*.

5️⃣  Batched matrix multiplication
        out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    (Torch matmul on the (B,H,N,N) view – no extra copies).

6️⃣  Second LayerNorm + out‑gate fused in a custom Triton kernel.
    The kernel:
        • reads the non‑contiguous result of the matmul,
        • reduces over the hidden dimension (H) to compute mean/var,
        • normalises, applies γ/β,
        • multiplies by the output gate,
        • writes the hidden representation (R = B·N·N rows, H columns) as FP16.

7️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only the fused second LayerNorm + gate is implemented in Triton – the
first LayerNorm uses the highly‑optimised PyTorch kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣  Fused second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    # Input tensor: out_perm shape (B, H, N, N) – FP16, non‑contiguous
    src_ptr,                     # const   *  [B, H, N, N]
    # Gate tensor: out_gate shape (R, H) – contiguous FP16
    gate_ptr,                    # const   *  [R, H]
    # Output buffer: hidden after LN & gate, shape (R, H) – contiguous FP16
    dst_ptr,                     # mutable *  [R, H]
    # LayerNorm parameters (γ/β) over the hidden dimension H
    ln_weight_ptr,               # const   *  [H]   (fp32)
    ln_bias_ptr,                 # const   *  [H]   (fp32)
    # Shape / stride information
    total_rows,                  # int (runtime) = B·N·N
    hidden_dim,                  # int (compile‑time) = H
    seq_len,                     # int (runtime) = N
    stride_b,                    # int (runtime) stride on dim 0  of src
    stride_h,                    # int (runtime) stride on dim 1  of src
    stride_i,                    # int (runtime) stride on dim 2  of src
    stride_j,                    # int (runtime) stride on dim 3  of src
    eps,                         # float (compile‑time)
    BLOCK_M: tl.constexpr,       # rows per program
    BLOCK_D: tl.constexpr,       # tile over hidden dimension
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Decode (b, i, j) from the linear row index
    # --------------------------------------------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # Base offset of the first hidden element for each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # Pass 1 – compute mean & variance across the hidden dimension H
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # Load a tile (stride_h accounts for the hidden‑dim stride)
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)          # sum over hidden dim tile
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – normalise, apply γ/β, multiply with out_gate, store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # Load raw hidden values again
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        # Multiply with out_gate (already fp16 → promote to fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (AlphaFold‑3) operator.
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
          - input_tensor : [B, N, N, dim] (float32, CUDA)
          - mask_tensor  : [B, N, N]       (float32, CUDA) – may be unused
          - weights_dict : mapping str → torch.Tensor (fp32)
          - config_dict  : contains "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # e.g. 128 or 384
    hidden_dim = config["hidden_dim"] # always 128 in the supplied tests
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N           # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output directly in fp16
    # --------------------------------------------------------------
    # PyTorch’s LayerNorm is highly tuned; we use it and cast to FP16.
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)    # shape (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) One fused linear (5 * hidden_dim outputs)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices once (FP16) and perform a single GEMM.
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)

    # Linear projection: result shape (B, N, N, 5*H) in FP16
    proj_out = F.linear(x_norm, w_concat)       # no bias

    H = hidden_dim
    # Slice the five tensors
    left_proj   = proj_out[..., 0 : H]
    right_proj  = proj_out[..., H : 2 * H]
    left_gate   = proj_out[..., 2 * H : 3 * H]
    right_gate  = proj_out[..., 3 * H : 4 * H]
    out_gate    = proj_out[..., 4 * H : 5 * H]

    # --------------------------------------------------------------
    # 3) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N, 1) – broadcast over hidden dimension
        mask_f16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 4) Apply sigmoid gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # fp16
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate       # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Batched matrix multiplication: out_perm shape (B, H, N, N)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N) – view
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N) – view
    # (B*H) independent matmuls of size (N, N) @ (N, N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # --------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate (fused Triton kernel)
    # --------------------------------------------------------------
    # Prepare stride information for the source tensor `out_perm`
    stride_b = out_perm.stride(0)   # stride over batch dimension B
    stride_h = out_perm.stride(1)   # stride over hidden dimension H
    stride_i = out_perm.stride(2)   # stride over i (N)
    stride_j = out_perm.stride(3)   # stride over j (N)

    # Allocate the contiguous hidden buffer (R, H)
    hidden_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    # Triton launch configuration -------------------------------------------------
    BLOCK_M = 1024          # rows per program – up to 1024 threads (max on H100)
    BLOCK_D = 32            # tile over hidden dim (covers 128‑dim with 4 iterations)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid](
        out_perm,                                 # src (non‑contig)
        out_gate.view(-1, H).contiguous(),       # gate (contig)
        hidden_out,                               # dst (contig)
        weights["to_out_norm.weight"],            # LN γ (fp32)
        weights["to_out_norm.bias"],              # LN β (fp32)
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 7) Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # shape (dim, H)
    final_half = F.linear(hidden_out, w_out)             # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
