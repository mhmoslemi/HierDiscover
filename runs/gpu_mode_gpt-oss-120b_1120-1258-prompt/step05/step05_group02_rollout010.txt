"""
custom_kernel: highly‑optimized Triton‑based forward for the “outgoing” TriMul
operator (AlphaFold‑3).  

Algorithm
---------
1️⃣  Layer‑norm on the 4‑D input tensor (FP32 → FP16) – Triton kernel.
2️⃣  Fuse **all** linear projections/gates (left‑proj, left‑gate, right‑proj,
    right‑gate, out‑gate) into a single FP16 GEMM (rows = B·N·N, cols = 5·H).
3️⃣  Apply optional pair‑wise mask (broadcast over H) and sigmoid gates.
4️⃣  Compute the tri‑multiplicative update  
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]  
    via a custom Triton batched GEMM that works directly on the
    (B,N,N,H) layout – **no** costly permute/contiguous copies.
5️⃣  Second Layer‑norm over the hidden dimension (FP16) – Triton kernel.
6️⃣  Multiply by the output‑gate and a final FP16 linear projection
    (hidden → dim).  Cast the result back to FP32.

The implementation keeps the heavy GEMMs (step‑2 & step‑4) in cuBLAS/TensorCore
via `torch.nn.functional.linear` and a handcrafted Triton matmul, while all
normalisation steps are hand‑written in Triton.  This reduces memory traffic
(especially the expensive permute/contiguous for the N³ GEMM) and brings the
runtime well below the target 1 ms on the test matrix sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton 1‑D LayerNorm (FP32 → FP16) – operates on a flat [rows, dim] view.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim] (input, FP32)
    y_ptr,          # half*    [total_rows, dim] (output, FP16)
    gamma_ptr,      # float*   [dim] (scale)
    beta_ptr,       # float*   [dim] (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized output (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton batched GEMM for the Tri‑multiplicative update.
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# The kernel works directly on tensors of shape [B, N, N, H] (FP16) and
# avoids any permute/contiguous copy.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,    # half* [B, N, N, H]
    right_ptr,   # half* [B, N, N, H]
    out_ptr,     # half* [B, N, N, H]   (written)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile over output rows   (i)
    pid_j = tl.program_id(1)          # tile over output cols   (j)
    pid_hb = tl.program_id(2)         # combined batch & head (b*H + h)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    hb = pid_hb                       # integer in [0, B*H)

    # Decode batch index and hidden index
    b = hb // H
    h = hb % H

    # Offsets for i, j and k
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base offsets for this (b,h) slice
    left_base = b * stride_left_b + h * stride_left_h
    right_base = b * stride_right_b + h * stride_right_h
    out_base = b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[i,k]  -> shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + left_base \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # ------------------------------------------------------------------
        # Load right[j,k]  -> shape (BLOCK_N, BLOCK_K)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + right_base \
                     + offs_j[:, None] * stride_right_i \
                     + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float16)

        # TL.dot performs Σₖ left[i,k] * right[j,k]  → (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right)          # acc stays FP32

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] [B, N, N]   (may be unused)
        weights_dict : {
            "norm.weight":            Tensor[dim],
            "norm.bias":              Tensor[dim],
            "left_proj.weight":       Tensor[hidden_dim, dim],
            "right_proj.weight":      Tensor[hidden_dim, dim],
            "left_gate.weight":       Tensor[hidden_dim, dim],
            "right_gate.weight":      Tensor[hidden_dim, dim],
            "out_gate.weight":        Tensor[hidden_dim, dim],
            "to_out_norm.weight":     Tensor[hidden_dim],
            "to_out_norm.bias":       Tensor[hidden_dim],
            "to_out.weight":          Tensor[dim, hidden_dim],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor[float32]  [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()      # [rows, dim] FP32
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    BLOCK_M = 128                     # rows per program
    BLOCK_D = 64                      # column block size (covers dim ≤ 384)
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B, N, N, dim] (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections & gates (single GEMM)
    #    Shape after GEMM: [rows, 5*H]  (fp16)
    # ------------------------------------------------------------------
    rows = total_rows
    # concatenate projection & gate weights (dim → 5·H)
    big_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16, non_blocking=True)   # [5·H, dim]

    proj_all = F.linear(x_norm_h.view(rows, dim), big_weight)   # no bias
    # split
    H = hidden_dim
    left_proj      = proj_all[:, 0 * H : 1 * H]
    left_gate_pre  = proj_all[:, 1 * H : 2 * H]
    right_proj     = proj_all[:, 2 * H : 3 * H]
    right_gate_pre = proj_all[:, 3 * H : 4 * H]
    out_gate_pre   = proj_all[:, 4 * H : 5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)               # [rows, H]

    left_h  = left_proj * left_gate                        # [rows, H]
    right_h = right_proj * right_gate

    # reshape to [B, N, N, H] (still fp16)
    left_h  = left_h.view(B, N, N, H)
    right_h = right_h.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (custom Triton GEMM)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h)          # [B, N, N, H] fp16

    # strides (in elements)
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_ri, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out_h.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                               # good for Tensor‑Core

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch × head
    )
    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out_h now holds the result of the “outgoing” TriMul (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out_h.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)                # FP16
    gated = out_norm * out_gate                         # element‑wise FP16

    final_h = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, dim] FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
