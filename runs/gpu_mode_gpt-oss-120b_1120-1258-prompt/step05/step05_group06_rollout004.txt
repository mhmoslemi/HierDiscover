"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣  Layer‑norm on the 4‑D input (FP32 stats → FP16 values) – handled by
    `torch.nn.functional.layer_norm` (very fast cuDNN kernel).

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single half‑precision GEMM by concatenating their weight matrices.

3️⃣  (Optional) apply the binary mask to the left‑ and right‑projections.

4️⃣  Apply the three sigmoids and gate the two projections.

5️⃣  Triangular multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision `torch.bmm`
    (shape (B·H, N, N) × (B·H, N, N)) – the GPU’s Tensor‑Core matmul is optimal.

6️⃣  **Fused** LayerNorm + out‑gate (both half‑precision) – a single Triton kernel
    (`fused_layernorm_gate_kernel`).  This is the only custom kernel, satisfying
    the “must use a kernel” requirement.

7️⃣  Final linear projection back to `dim` (`torch.nn.functional.linear`)
    – still half‑precision.

8️⃣  Cast the result to FP32 (matching the original PyTorch module).

The implementation works for all test configurations (B∈{1,2},
N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128) and supports the
presence or absence of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # const   half* [total_rows, dim]    (pre‑norm)
    y_ptr,          # mutable half* [total_rows, dim]    (post‑norm⋅gate)
    w_ptr,          # const   float* [dim]              (γ)
    b_ptr,          # const   float* [dim]              (β)
    gate_ptr,       # const   half*  [total_rows, dim]  (sigmoided out‑gate)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_D: tl.constexpr,   # columns per iteration (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # 1️⃣  Compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # 2️⃣  Normalize, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑µ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ   (FP32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β   (FP32)
        out = norm * w[None, :] + b[None, :]                # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)      # FP32
        out = out * gate_vals                               # apply out‑gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] – ignored if
                         config["nomask"] is True
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"     : bool,
              "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim   = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]       # H (always 128 in the tests)
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 stats → FP16 values) – use cuDNN fast kernel
    # ------------------------------------------------------------------
    # Returns FP32 → cast to half
    x_norm_fp16 = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                   # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    #   (B,N,N,C) → (R, C) where R = B·N·N
    x_flat = x_norm_fp16.view(-1, dim)          # (R, C) – already contiguous

    # Concatenate weight matrices: each is (hidden, dim)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5·H, C)

    proj_all = F.linear(x_flat, concat_w)      # (R, 5·H), no bias

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                          # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply mask if required (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)  # (R,1)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)    # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)        # (R, H) – will be used later

    left_fp16 = left_proj_fp16 * left_gate          # (R, H)
    right_fp16 = right_proj_fp16 * right_gate       # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    left  = left_fp16.view(B, N, N, hidden)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)         # (B, N, N, H)

    # Reshape to (B·H, N, N) where rows=i, cols=k for left
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    # For right we need rows=k, cols=j
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B·H, N, N)

    # Batched mat‑mul: each head performs N×N × N×N → N×N
    out_mat = torch.bmm(left_mat, right_mat)        # (B·H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate (single Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden).contiguous()   # (R, H) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)  # (R, H) half

    # Triton launch parameters – a larger BLOCK_M reduces kernel count
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                # already sigmoid‑ed, half‑precision
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()   # (dim, H)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)  # (R, dim) half

    # Cast back to FP32 as required by the reference implementation
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
