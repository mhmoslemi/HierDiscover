Optimized Triton‑based implementation of the *outgoing* TriMul operator.

Algorithm Overview
------------------
1. Row‑wise LayerNorm on the input (C‑dim) – Triton kernel with
   fp32 accumulation and fp16 output.
2. Fuse the five linear projections (left, right, three gates) into a
   single half‑precision GEMM (torch.nn.functional.linear on the
   concatenated weight matrix).
3. Optional mask is applied before gating.
4. Apply sigmoid gates and element‑wise gating (still fp16).
5. Triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with `torch.einsum` which avoids the large
   `permute+contiguous` copy required by the batched‑GEMM formulation.
6. Second LayerNorm over the hidden dimension – same Triton kernel as (1).
7. Multiply by the output gate and project back to the original channel
   dimension with a final half‑precision linear layer.
8. Cast the result to float32.

All heavy linear algebra (GEMMs, einsum) are executed by cuBLAS /
Tensor‑Core kernels; the only Triton kernels are the two LayerNorms.
"""

import torch.nn.functional as F
import triton

    bias_ptr,        # const * fp32   [dim]               (β)
    total_rows,      # i32

    eps: tl.constexpr,

    BLOCK_D: tl.constexpr,   # columns per iteration
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):

            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    out_flat = out.contiguous().view(total_rows, hidden_dim)   # (M, H)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                         # input (fp16)
        out_norm_fp16,
        weights["to_out_norm.weight"],    # γ (float32, [H])
        weights["to_out_norm.bias"],      # β (float32, [H])
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Multiply by the output gate and final linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                           # (B, N, N, H), fp16
    gated_flat = gated.view(total_rows, hidden_dim)       # (M, H)

    to_out_w_fp16 = weights["to_out.weight"].half().contiguous()  # (C, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)              # (M, C), fp16

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

    # Reshape back to 4‑D tensors for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)         # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)       # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing version)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    #    Implemented with torch.einsum to avoid materialising the
    #    (B, H, N, N) tensors required by a batched GEMM.
    # --------------------------------------------------------------
    # Result shape: (B, N, N, H), dtype = fp16
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.contiguous().view(total_rows, hidden_dim)   # (M, H)
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                         # input (fp16)
        out_norm_fp16,
        weights["to_out_norm.weight"],    # γ (float32, [H])
        weights["to_out_norm.bias"],      # β (float32, [H])
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Multiply by the output gate and final linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                           # (B, N, N, H), fp16
    gated_flat = gated.view(total_rows, hidden_dim)       # (M, H)

    to_out_w_fp16 = weights["to_out.weight"].half().contiguous()  # (C, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)              # (M, C), fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast to float32 and reshape to the original shape
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
