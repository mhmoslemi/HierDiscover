    left_ptr, right_ptr, out_ptr,
    B: tl.int32, N: tl.int32, H: tl.int32,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):

    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    b_idx = pid_b

"""
Highly‑optimized Triton forward for the **outgoing** TriMul op (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – same row‑wise kernel as before.
2️⃣ A single FP16 GEMM (torch.nn.functional.linear) computes the five
   linear projections (left/right + 3 gates) in one call.
3️⃣ Optional binary mask is applied to the left/right projections.
4️⃣ Sigmoid on the three gate tensors and element‑wise gating.
5️⃣ **Fused triangular multiplicative update** – a custom Triton kernel
   computes
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   in a single launch.  The kernel tiles the i‑ and j‑axes (BLOCK_I×BLOCK_J)
   and processes the hidden dimension in chunks (BLOCK_H).  For each (i,j)
   tile it loops over *k* and multiplies the i‑vector with the j‑vector
   (broadcasted over the hidden chunk), accumulating in FP32.  This removes
   the many small cuBLAS batched‑gemm launches of the reference
   implementation and drastically cuts launch‑overhead.
6️⃣ Fused LayerNorm + out‑gate + final linear projection – identical to the
   previous kernel, but now operates on the output of the fused TriMul.
The final tensor is returned in float32 and matches the PyTorch reference
to within torch.float16 tolerances.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # const float*  [total_rows, dim]
    y_ptr,               # *mut half*    [total_rows, dim]
    weight_ptr,          # const float*  [dim]   (γ)
    bias_ptr,            # const float*  [dim]   (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------- first pass: mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- second pass: normalise and write --------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # const half*  [total_rows, hidden]
    gate_ptr,            # const half*  [total_rows, hidden] (sigmoid‑ed out_gate)
    norm_w_ptr,          # const float* [hidden]            (γ)
    norm_b_ptr,          # const float* [hidden]            (β)
    out_w_ptr,           # const half*  [dim, hidden]       (to_out weight)
    out_ptr,             # *mut float*  [total_rows, dim]
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑channel chunk
):
    pid = tl.program_id(0)          # row block
    pid_d = tl.program_id(1)        # output‑channel block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------- read a whole hidden vector for each selected row ----------
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # (BLOCK_M, hidden)

    # ---------------- mean / rstd (FP32) ----------------
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------- LayerNorm weight / bias ----------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # (BLOCK_M, hidden)
    gated = norm * g                                                # (BLOCK_M, hidden)

    # ------------------- linear projection -------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # out_w has shape [dim, hidden]; we need its transpose (hidden, dim)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                # (hidden, BLOCK_D)

    out_chunk = tl.dot(gated, out_w_T)                              # (BLOCK_M, BLOCK_D)

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------

@triton.jit

    stride_b_r: tl.int32,
    stride_i_r: tl.int32,
    stride_k_r: tl.int32,
    stride_h_r: tl.int32,

    stride_i_o: tl.int32,
    stride_j_o: tl.int32,
    stride_h_o: tl.int32,

    pid_j = tl.program_id(1)          # j‑tile
    pid_b = tl.program_id(2)          # batch index

    j_start = pid_j * BLOCK_J
    b_idx   = pid_b

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over the hidden dimension in blocks
    # ------------------------------------------------------------------

        h_start = h_tile * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

                + b_idx * stride_b_l \
                + offs_i[:, None] * stride_i_l \
                + k * stride_k_l \
                + offs_h[None, :] * stride_h_l

                                mask=mask_i[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)   # (BLOCK_I, BLOCK_H)

            # right: shape (BLOCK_J, BLOCK_H)   (note j dimension uses stride_i_r)
            right_ptrs = right_ptr \
                + b_idx * stride_b_r \
                + offs_j[:, None] * stride_i_r \
                + k * stride_k_r \
                + offs_h[None, :] * stride_h_r
            right_vals = tl.load(right_ptrs,
                                 mask=mask_j[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)  # (BLOCK_J, BLOCK_H)

            # outer product over i and j, broadcast over hidden chunk
            # left_vals[:,None,:]  -> (BLOCK_I, 1, BLOCK_H)
            # right_vals[None,:,:] -> (1, BLOCK_J, BLOCK_H)
            acc += left_vals[:, None, :] * right_vals[None, :, :]

        # ------------------------------------------------------------------
        # Store the result (cast back to FP16)
        # ------------------------------------------------------------------
        out_ptrs = out_ptr \
            + b_idx * stride_b_o \
            + offs_i[:, None, None] * stride_i_o \
            + offs_j[None, :, None] * stride_j_o \
            + offs_h[None, None, :] * stride_h_o

        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_i[:, None, None] &
                      mask_j[None, :, None] &
                      mask_h[None, None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                         # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (single FP16 GEMM)
    # --------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                   # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)               # (R, 5*H) FP16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                                        # each (R, H)

    # --------------------------------------------------------------
    # 2a) Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16  * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)               # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)              # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)                # (R, H) – used later

    left = left_proj_fp16 * left_gate                        # (R, H)
    right = right_proj_fp16 * right_gate                     # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update – fused Triton kernel
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hidden).contiguous()          # (B, N, N, H)
    right = right.view(B, N, N, hidden).contiguous()         # (B, N, N, H)

    out = torch.empty_like(left)                             # (B, N, N, H)

    # strides for the 4‑D tensors (batch, i, k, h)
    s_b_l, s_i_l, s_k_l, s_h_l = left.stride()
    s_b_r, s_i_r, s_k_r, s_h_r = right.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out.stride()  # identical layout

    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 32

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B)
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_b_l=s_b_l,
        stride_i_l=s_i_l,
        stride_k_l=s_k_l,
        stride_h_l=s_h_l,
        stride_b_r=s_b_r,
        stride_i_r=s_i_r,
        stride_k_r=s_k_r,
        stride_h_r=s_h_r,
        stride_b_o=s_b_o,
        stride_i_o=s_i_o,
        stride_j_o=s_j_o,
        stride_h_o=s_h_o,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) FP16
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,                                      # x_ptr
        out_gate,                                      # gate_ptr (already sigmoid)
        weights["to_out_norm.weight"],                 # γ
        weights["to_out_norm.bias"],                   # β
        weights["to_out.weight"].half().contiguous(), # to_out weight (dim, hidden)
        final_fp32,                                    # output buffer
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)                 # [B, N, N, dim] float32
    return final
