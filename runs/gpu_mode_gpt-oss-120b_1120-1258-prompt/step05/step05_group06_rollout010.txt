def custom_kernel(data):
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]; hidden = cfg["hidden_dim"]; eps = cfg.get("eps", 1e-5); nomask = cfg.get("nomask", True)

    # First layernorm
    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps) # fp32
    x_norm = x_norm.to(torch.half)

    # Concatenate weights (half)
    w_cat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.half)

    # Linear projection on flat tensor
    B,N,_,_ = inp.shape
    R = B*N*N
    proj = F.linear(x_norm.view(R, dim), w_cat) # shape (R, 5*hidden)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(proj, hidden, dim=1)

    # mask
    if not nomask:
        mask_flat = mask.view(R, 1).to(torch.half)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # Gate sigmoids (half)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # N^3 contraction using einsum
    out = torch.einsum('bikc,bjkc->bijc', left, right)

    # second layernorm
    out_norm = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out_norm = out_norm.to(torch.half)

    # apply out_gate and final linear
    out_norm = out_norm * out_gate
    final = F.linear(out_norm.view(R, hidden), weights["to_out.weight"].to(torch.half))
    final = final.view(B, N, N, dim).to(torch.float32)
    return final

Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “tri‑multiplicative update”).

Algorithm
---------
1️⃣  FP32 LayerNorm on the input (torch.nn.functional.layer_norm) – the
    result is immediately cast to float16 to keep the whole pipeline in FP16.

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a

3️⃣  Apply the optional binary mask (broadcast on the hidden dimension) and
    the three sigmoid gates – all in FP16.

4️⃣  **Tri‑multiplicative update**  
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

5️⃣  A fused Triton kernel performs the second LayerNorm (FP32 statistics →

6️⃣  Final half‑precision linear projection back to the model dimension

    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- mean / variance (FP32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------- normalisation, γ, β, gate -------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * gamma[None, :] + beta[None, :]      # FP32

        # multiply by the already‑sigmoided out_gate
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor            [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor            [B, N, N]   (bool/float)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim   = cfg["dim"]                # model dimension C
    hidden = cfg["hidden_dim"]        # hidden dimension H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N            # rows when flattening a [B,N,N,·] tensor

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimised (cuDNN) and
    # performs the mean/var reduction in FP32.
    x_norm_fp32 = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.half)                     # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gates
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (each: [hidden, dim]) once
    # and execute a single GEMM in half precision.
    w_cat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.half).contiguous()                         # (5*H, C)

    # Flatten the normalized tensor to [R, C] for a single linear call
    proj = F.linear(x_norm.view(total_rows, dim), w_cat)   # (R, 5*H)

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden, dim=1
    )  # each (R, H)

    # -----------------------------------------------------------------
    # 2a) Optional binary mask (broadcast on the hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] → half
        mask_flat = mask.reshape(total_rows, 1).to(torch.half)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # -----------------------------------------------------------------
    # 2b) Sigmoid gates (still half) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)   # (R, H)
    right_gate = torch.sigmoid(right_gate)  # (R, H)
    out_gate   = torch.sigmoid(out_gate)    # (R, H), used later

    left  = left_proj  * left_gate          # (R, H)
    right = right_proj * right_gate         # (R, H)

    # -----------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update – O(N³·H)
    # -----------------------------------------------------------------
    # Reshape to [B, N, N, H] (still half) and use a single einsum.
    # ``einsum`` avoids the expensive view/permute/contiguous dance required
    # for a batched ``torch.bmm``.
    left  = left.view(B, N, N, hidden)   # (B, N, N, H)
    right = right.view(B, N, N, hidden)  # (B, N, N, H)

    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("bikc,bjkc->bijc", left, right)  # (B, N, N, H), FP16

    # -----------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (single fused Triton kernel)
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)             # (R, H) half

    # Triton launch parameters – a few thousand rows per program is ideal.
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                               # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.half)    # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half

    # Cast back to FP32 to match the original module’s output dtype
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
