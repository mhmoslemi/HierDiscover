"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
-------------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣  Row‑wise LayerNorm over the last channel (C) – computed in FP32
    but written in FP16 (triton).

2️⃣  Two *fused* linear‑+‑gate passes:
    * left  side  →  proj (H) + gate (H)   (2·H rows)
    * right side + out‑gate → proj (H) + gate (H) + out_gate (H)   (3·H rows)
    The three weight matrices are concatenated and a single
    `torch.nn.functional.linear` is executed per side, so the
    normalised input is read only **twice** instead of three times.

3️⃣  Optional binary mask – broadcast on the hidden dimension after
    gating.

4️⃣  Triangular multiplicative update – a batched matrix‑multiply:
    `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.
    Implemented with `torch.bmm` on tensors shaped
    `(B·H, N, N)` → leverages Tensor‑Core cuBLAS GEMM.

5️⃣  Fused LayerNorm + output‑gate – a second triton kernel that
    computes the per‑row mean/var (FP32), applies γ/β, multiplies the
    sigmoid‑gate and stores the result in FP16.

6️⃣  Final linear projection back to the original channel dimension
    (`hidden → dim`) – FP16 GEMM (`torch.nn.functional.linear`).

Resulting tensor is cast back to FP32 to match the reference module.

The kernels are deliberately small (row‑wise) to maximise occupancy
on an H100.  All heavy arithmetic stays in FP16/FP32 (Tensor‑Core),
while memory traffic is cut by fusing the three linear layers into
only two passes over the normalised tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) LayerNorm over the input channel dimension (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16)
    weight_ptr,          # *const float*  [dim]           (γ)
    bias_ptr,            # *const float*  [dim]           (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance in FP32
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ/β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm over the hidden dimension + out‑gate (FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ/β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, ignored if config["nomask"])
        - weights_dict : model weights (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                   # C
    hidden = cfg["hidden_dim"]            # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N               # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                            device=device)                     # [R, C] half

    BLOCK_M = 128
    BLOCK_D = 128 if dim > 256 else 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Two fused linear + gate passes (left side, right+out_gate side)
    # ------------------------------------------------------------------
    # Cast weights once to half
    left_proj_w  = weights["left_proj.weight"].half().contiguous()   # (H, C)
    left_gate_w  = weights["left_gate.weight"].half().contiguous()   # (H, C)
    right_proj_w = weights["right_proj.weight"].half().contiguous() # (H, C)
    right_gate_w = weights["right_gate.weight"].half().contiguous() # (H, C)
    out_gate_w   = weights["out_gate.weight"].half().contiguous()   # (H, C)

    # Left side: proj + gate (2·H rows)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2H, C)
    left_comb   = F.linear(norm_fp16, left_comb_w)                  # (R, 2H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                        # (R, H)
    left = left_proj_raw * left_gate                                # (R, H)

    # Right side + out_gate: proj + gate + out_gate (3·H rows)
    right_out_comb_w = torch.cat([right_proj_w, right_gate_w, out_gate_w],
                                 dim=0)                                 # (3H, C)
    right_out_comb   = F.linear(norm_fp16, right_out_comb_w)            # (R, 3H)
    right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        right_out_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                        # (R, H)
    out_gate   = torch.sigmoid(out_gate_raw)                          # (R, H) (half)
    right = right_proj_raw * right_gate                              # (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional binary mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1) half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] for easier permutation
    left = left.view(B, N, N, hidden)      # (B, N, N, H)
    right = right.view(B, N, N, hidden)    # (B, N, N, H)

    # (B*H, N, N) matrices for cuBLAS batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # Batched matrix‑multiply → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()        # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)                  # (R, H) half

    BLOCK_D_H = 128 if hidden > 256 else 64
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                               # (R, H) half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_H,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half().contiguous()    # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)             # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)    # cast back to FP32

    return final
