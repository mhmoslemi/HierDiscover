@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    # strides
    stride_aba, stride_abm, stride_abk,
    stride_bba, stride_bbn, stride_bbk,
    stride_cca, stride_ccm, stride_ccn,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # calculate batch index and tile indices
    # We create a grid: (batch * ceil(M/BLOCK_M) * ceil(N/BLOCK_N))
    # decode program_id into batch_idx, pid_m, pid_n
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = rem // tl.cdiv(N, BLOCK_N)
    pid_n = rem % tl.cdiv(N, BLOCK_N)

    # Compute offsets for block of C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to start of block in each tensor
    a_ptrs = a_ptr + batch_idx * stride_aba + offs_m[:, None] * stride_abm + tl.arange(0, BLOCK_K)[None, :] * stride_abk
    b_ptrs = b_ptr + batch_idx * stride_bba + tl.arange(0, BLOCK_K)[:, None] * stride_bbk + offs_n[None, :] * stride_bbn

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_b, stride_h, stride_i, stride_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch dimensions
    b_idx = batch_idx // H
    h_idx = batch_idx % H
    base = b_idx * stride_b + h_idx * stride_h

    # offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (i, k)
        a_ptrs = a_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load B tile (k, j) from transposed view
        b_ptrs = b_ptr + base + offs_k[:, None] * stride_j + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    dim = config["dim"]

    eps = 1e-5
    device = input_tensor.device

    # 1) LayerNorm

    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_D = 64

        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2) Linear projection (5 * hidden_dim)
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)
    proj_out = F.linear(x_norm, w_concat)   # shape [R, 5*hidden_dim]

    H = hidden_dim
    left_proj = proj_out[:, 0:H]
    right_proj = proj_out[:, H:2*H]
    left_gate = proj_out[:, 2*H:3*H]
    right_gate = proj_out[:, 3*H:4*H]
    out_gate = proj_out[:, 4*H:5*H]

    # Apply sigmoid
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Optional mask
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Permute to (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)

    # 5) Batched GEMM via Triton
    out_perm = torch.empty_like(left_perm)  # fp16

    stride_i = left_perm.stride(2)
    stride_j = left_perm.stride(3)

    BLOCK_N_GEMM = 128
    BLOCK_K_GEMM = 32
    grid_gemm = (B * H,
                 triton.cdiv(N, BLOCK_M_GEMM),

        B,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        BLOCK_M=BLOCK_M_GEMM,
        BLOCK_N=BLOCK_N_GEMM,
        BLOCK_K=BLOCK_K_GEMM,
    )

    # free left/right tensors to save memory
    del left, right, left_perm, right_perm

        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    w_out = weights["to_out.weight"].to(torch.float16)
    final_half = F.linear(hidden_out, w_out)   # (R, dim)
    final = final_half.view(B, N, N, dim).to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
----------------------------------------------------------------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣  One fused Linear that produces the five hidden‑dim tensors
    (left_proj, right_proj, left_gate, right_gate, out_gate).
3️⃣  Optional mask applied only to the two projections.
4️⃣  Apply sigmoid gates → gated tensors *left* and *right*.
5️⃣  Batched matrix multiplication (B·H independent GEMMs of size N×N)
    implemented as a single Triton kernel, avoiding the many cuBLAS
    launches of torch.matmul.
6️⃣  Fused LayerNorm + out‑gate (still a Triton kernel).  The result is
    a contiguous (B·N·N, hidden_dim) FP16 buffer.
7️⃣  Final Linear projection back to `dim` (FP16 GEMM) and cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   *  [total_rows, dim]   (fp32)
    y_ptr,                # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,           # const   *  [dim]               (fp32)
    bias_ptr,             # const   *  [dim]               (fp32)
    total_rows,           # int     (runtime)
    dim,                  # int     (compile‑time)
    eps,                  # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # pass 1 – compute mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # pass 2 – write normalized data (γ/β) as fp16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣  Batched GEMM for the outgoing multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # fp16 tensors (B,H,N,N)
    B, H, N,                           # runtime sizes
    stride_b, stride_h, stride_i, stride_j,  # strides of the 4‑D tensors
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes
        C[b,h,i,j] = Σₖ A[b,h,i,k] * B[b,h,j,k]   (B×H independent GEMMs)
    A and B are stored as (B, H, N, N) with the last two dims being (i, k)
    and (j, k) respectively, i.e. we multiply A by the transpose of B.
    """
    pid_batch = tl.program_id(0)        # linear index over B·H
    pid_m     = tl.program_id(1)        # tile over i
    pid_n     = tl.program_id(2)        # tile over j

    # decode (b, h) from the linear batch index
    b_idx = pid_batch // H
    h_idx = pid_batch %  H
    base  = b_idx * stride_b + h_idx * stride_h      # start of this (b,h) slice

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A slice: rows i, cols k
        a_ptrs = a_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B slice (transposed): rows k, cols j
        b_ptrs = b_ptr + base + offs_k[:, None] * stride_j + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 6️⃣  Fused second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    src_ptr,                 # const   *  [B, H, N, N] (fp16)
    gate_ptr,                # const   *  [R, H]    (fp16)   R = B·N·N
    dst_ptr,                 # mutable *  [R, H]    (fp16)
    ln_weight_ptr,           # const   *  [H]       (fp32)
    ln_bias_ptr,             # const   *  [H]       (fp32)
    total_rows,              # int (runtime)  = B·N·N
    hidden_dim,              # int (compile‑time)
    seq_len,                 # int (runtime)  = N
    stride_b,                # int (runtime) stride on dim‑0 (B) of src
    stride_h,                # int (runtime) stride on dim‑1 (H) of src
    stride_i,                # int (runtime) stride on dim‑2 (N) of src
    stride_j,                # int (runtime) stride on dim‑3 (N) of src
    eps,                     # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- decode (b,i,j) from linear row index -----------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # pass 1 – mean & variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        ptr = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # pass 2 – normalize, apply γ/β, multiply with out_gate and store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # raw hidden values
        ptr = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        # Multiply with out_gate (fp16 → promote)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]            # model dimension (e.g. 128 or 384)
    hidden_dim = config["hidden_dim"]    # hidden dimension (always 128 in tests)
    eps        = 1e-5
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim)                # contiguous view
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 256
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # --------------------------------------------------------------
    # 2) One fused linear projection (5 × hidden_dim outputs)
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, dim)

    proj_out = F.linear(x_norm, w_concat)   # (R, 5*H), fp16

    H = hidden_dim
    left_proj  = proj_out[:, 0:H]
    right_proj = proj_out[:, H:2*H]
    left_gate  = proj_out[:, 2*H:3*H]
    right_gate = proj_out[:, 3*H:4*H]
    out_gate   = proj_out[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4) Apply sigmoid gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate  = torch.sigmoid(out_gate)

    left  = left_proj  * left_gate   # (R, H)
    right = right_proj * right_gate  # (R, H)

    # --------------------------------------------------------------
    # 5) Reshape and permute to (B, H, N, N)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2)   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2)  # (B, H, N, N)

    # --------------------------------------------------------------
    # 6) Batched GEMM (B·H independent N×N matmuls) – Triton kernel
    # --------------------------------------------------------------
    out_perm = torch.empty_like(left)   # fp16 buffer for the result

    stride_b = left.stride(0)   # stride on batch dim
    stride_h = left.stride(1)   # stride on hidden dim (contiguous ⇒ 1)
    stride_i = left.stride(2)   # stride on i  (N*H)
    stride_j = left.stride(3)   # stride on j  (H)

    BLOCK_M_GEMM = 128
    BLOCK_N_GEMM = 128
    BLOCK_K_GEMM = 32
    grid_gemm = (
        B * H,
        triton.cdiv(N, BLOCK_M_GEMM),
        triton.cdiv(N, BLOCK_N_GEMM),
    )
    batched_gemm_kernel[grid_gemm](
        left,
        right,
        out_perm,
        B,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        BLOCK_M=BLOCK_M_GEMM,
        BLOCK_N=BLOCK_N_GEMM,
        BLOCK_K=BLOCK_K_GEMM,
    )
    # free large temporaries early
    del left, right

    # --------------------------------------------------------------
    # 7) Second LayerNorm + out‑gate (fused Triton kernel)
    # --------------------------------------------------------------
    hidden_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN),)
    fused_ln_gate_kernel[grid_ln2](
        out_perm,
        out_gate,
        hidden_out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # --------------------------------------------------------------
    # 8) Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final_half = F.linear(hidden_out, w_out)            # (R, dim), fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
