"""
Optimized Triton + PyTorch forward implementation of the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP16 output).  The kernel computes the mean/variance in
   FP32 and writes the normalized tensor as half‑precision.

2️⃣ **One** fused GEMM replaces three separate linear layers:
   * Concatenate the five weight matrices
     (left_proj, left_gate, right_proj, right_gate, out_gate) → (5·H, C).
   * Perform a single half‑precision GEMM:  X_norm @ Wᵀ  →  (R, 5·H).
   * Split the result, apply sigmoid to the three gate slices and multiply
     the projection slices → `left`, `right`.  The `out_gate` slice is also
     sigmoid‑ed and kept for later use.
   This cuts the number of large GEMMs from three to one.

3️⃣ Optional binary mask – applied after the two gated projections.

4️⃣ Tri‑multiplicative update:  For every hidden channel `h`,
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as a batched half‑precision matrix‑multiply
   (`torch.bmm`) on tensors of shape `(B·H, N, N)`.  This uses Tensor‑Core
   cuBLAS and is the dominant compute cost.

5️⃣ Fused LayerNorm + output‑gate.  The Triton kernel computes per‑row
   mean/variance, applies learned scale/bias and finally multiplies by the
   already‑sigmoid’ed `out_gate`.  The result is stored as half.

6️⃣ Final linear projection back to `dim` (FP16 GEMM).  The result is cast
   to float32 before returning.

All heavy reductions (both LayerNorms) are performed with Triton kernels,
while the large GEMMs use cuBLAS via `torch.nn.functional.linear` /
`torch.bmm`, which is optimal on H100.  The implementation works for all
test configurations (different `N`, `B`, `dim`, `hidden_dim` and optional
mask) and returns a float32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim]  (FP32 input)
    y_ptr,               # *mut   half*   [total_rows, dim]  (FP16 output)
    weight_ptr,          # *const float*  [dim]             (γ)
    bias_ptr,            # *const float*  [dim]             (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]               # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, sigmoid‑ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]                # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)    # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)    # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)               # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float/bool) – ignored if config["nomask"]=True
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]                # C
    hidden = cfg["hidden_dim"]      # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()            # [R, C] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half, device=device)  # [R, C] FP16

    # Choose block sizes – tuned for dim ≤ 384
    BLOCK_M = 1024                     # rows per program (few launches)
    BLOCK_D = 128 if dim > 256 else 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the three linear layers (left_proj, left_gate,
    #     right_proj, right_gate, out_gate) into a single GEMM.
    # --------------------------------------------------------------
    # Cast all projection / gate weights to FP16 once.
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # Combine them → (5·H, C)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0
    ).contiguous()   # shape (5*H, C)

    # Single half‑precision GEMM
    combined = F.linear(x_norm_fp16, combined_w)   # (R, 5*H), half

    # Split the combined result
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(combined, hidden, dim=1)

    # Apply sigmoid to gate slices
    left_gate = torch.sigmoid(left_gate_raw)      # (R, H)
    right_gate = torch.sigmoid(right_gate_raw)    # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)        # (R, H) – used later

    # Gated projections
    left = left_proj_raw * left_gate              # (R, H) half
    right = right_proj_raw * right_gate           # (R, H) half

    # --------------------------------------------------------------
    # 2a) Optional binary mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N) → (R, 1)
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left.view(B, N, N, hidden)    # (B, N, N, H) half
    right = right.view(B, N, N, hidden)  # (B, N, N, H) half

    # Convert to batched matrices of shape (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # Batched half‑precision GEMM → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)

    # Back to tensor shape [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) half

    # --------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    # Re‑use BLOCK_M / BLOCK_D (they fit hidden ≤ 384)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                 # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection (hidden → dim)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w)   # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
