    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    # linear:
    rows = total_rows
    x_norm_flat = x_norm.view(rows, dim)

    right_concat = F.linear(x_norm_flat, right_concat_w)

    right_proj_h = right_concat[:, :hidden_dim]

    right_h = right_proj_h * right_gate_h

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch * hidden id
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    total_batch = B * H
    batch_id = pid % total_batch   # Actually pid is already in range total_batch if we set grid[0] = total_batch
    # compute b and h
    b = batch_id // H
    h = batch_id - b * H

    # base pointers for selected slice
    left_base = left_ptr + b * stride_left_batch + h * 1
    right_base = right_ptr + b * stride_right_batch + h * 1
    out_base = out_ptr + b * stride_out_batch + h * 1

    # compute offsets for i, j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A: left[i, k]
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load B_T: right[k, j] (original right[j,k])
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write out
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel  –  Triton‑accelerated “outgoing” TriMul (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the input tensor (shape [B,N,N,C]) in FP16.
2️⃣  Fuse the projection (dim → hidden) and the gate (dim → hidden) into a single
    linear layer per side (left / right) → shape [B·N·N, 2·hidden].
3️⃣  Optionally apply the pair‑wise mask (broadcast over the hidden dimension)
    to the projected tensors.
4️⃣  Apply sigmoid gates → left_h, right_h  (FP16).
5️⃣  **Tri‑multiplicative update** – a custom Triton GEMM that computes, for each
    batch‑hidden slice (b,h),

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    i.e. a batched mat‑mul “left @ rightᵀ” without any extra memory copies.
6️⃣  Second Layer‑norm over the hidden dimension (FP16).
7️⃣  Multiply by the output‑gate (sigmoid) and a final linear projection
    (hidden → dim) → cast to float32 for the module output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*  [total_rows, dim] (input)
    y_ptr,          # half*   [total_rows, dim] (output)
    gamma_ptr,      # float*  [dim]            (scale)
    beta_ptr,       # float*  [dim]            (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- pass 1: mean & variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: write normalized values -----------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # half*  [B, N, N, H]
    right_ptr,         # half*  [B, N, N, H]
    out_ptr,           # half*  [B, N, N, H] (result)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_batch: tl.constexpr,   # stride for the batch dimension (N*N*H)
    stride_i: tl.constexpr,       # stride for the first N dimension (N*H)
    stride_k: tl.constexpr,       # stride for the second N dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)   # combined (b, h) index
    pid_i     = tl.program_id(1)   # tile over i
    pid_j     = tl.program_id(2)   # tile over j

    # decode (b, h)
    b = pid_batch // H
    h = pid_batch - b * H

    # base pointers for this (b, h) slice (h is the innermost dim → stride 1)
    left_base  = left_ptr  + b * stride_batch + h
    right_base = right_ptr + b * stride_batch + h
    out_base   = out_ptr   + b * stride_batch + h

    # i / j coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  – shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right_T[k, j] = right[j, k]  – shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write result (FP16)
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor[float32]   [B, N, N, C]
            mask_tensor  : torch.Tensor[float32]   [B, N, N]  (may be ignored)
            weights_dict : dict of module parameters (see code)
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt)}

    Returns:
        torch.Tensor[float32]  [B, N, N, C]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [rows, C]
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm = x_norm_fp16.view(B, N, N, dim)          # (B,N,N,C) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (dim → H) for left/right
    # ------------------------------------------------------------------
    # concatenate projection & gate weights → shape (2·H, dim)
    left_w  = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
                        ).to(torch.float16, non_blocking=True)
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
                        ).to(torch.float16, non_blocking=True)

    rows = total_rows
    x_norm_flat = x_norm.view(rows, dim)

    left_cat  = F.linear(x_norm_flat, left_w)   # [rows, 2·H]
    right_cat = F.linear(x_norm_flat, right_w)

    left_proj   = left_cat[:, :H]                 # [rows, H]
    left_gate_z = left_cat[:, H:]                 # [rows, H]
    right_proj  = right_cat[:, :H]
    right_gate_z = right_cat[:, H:]

    # out‑gate (shared across positions)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate   = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))   # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)

    left_h  = left_proj * left_gate   # [rows, H]
    right_h = right_proj * right_gate

    # reshape to (B, N, N, H)
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (custom Triton GEMM)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # FP16, shape (B,N,N,H)

    # strides of the contiguous (B,N,N,H) tensors
    stride_batch = left.stride(0)   # N*N*H
    stride_i     = left.stride(1)   # N*H
    stride_k     = left.stride(2)   # H   (innermost)

    total_batch = B * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    rows_h = total_rows
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)   # FP16
    gated = out_norm * out_gate             # element‑wise (FP16)

    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    gated_flat = gated.view(rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w)   # [rows, dim] FP16

    # Cast back to FP32 (module output dtype)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
