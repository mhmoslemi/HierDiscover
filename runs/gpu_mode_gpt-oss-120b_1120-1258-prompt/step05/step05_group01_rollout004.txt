"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input.
2️⃣  One fused linear projection that produces the five hidden tensors
    (left, right, left‑gate, right‑gate, out‑gate) in FP16.
3️⃣  Optional pair‑wise mask applied only to the two projections.
4️⃣  Apply sigmoid gates and element‑wise gating.
5️⃣  Batched matrix multiplication:
        out[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    performed with a single `torch.matmul` on the strided tensors.
6️⃣  Fuse the second LayerNorm, the output‑gate and the write‑out to a
    contiguous buffer with a Triton kernel (`fused_ln_gate_kernel`).
7️⃣  Final linear projection back to the model dimension and cast to FP32.

Only the two LayerNorm‑related kernels are written in Triton; all heavy
linear algebra (the two GEMMs) is delegated to cuBLAS/Tensor‑Cores via
PyTorch.  The implementation works for all configurations listed in the
statement and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *  [total_rows, dim]   (fp32)
    y_ptr,            # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,       # const   *  [dim]               (fp32)
    bias_ptr,         # const   *  [dim]               (fp32)
    total_rows,       # int     (runtime)
    dim,              # int     (compile‑time)
    eps,              # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1st pass – compute mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2nd pass – write normalized data (γ/β) as fp16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6️⃣  Fuse second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    src_ptr,          # const   *  [B, H, N, N] (fp16) – non‑contiguous view
    gate_ptr,         # const   *  [R, H]        (fp16)
    dst_ptr,          # mutable *  [R, H]        (fp16)
    ln_weight_ptr,    # const   *  [H]           (fp32)
    ln_bias_ptr,      # const   *  [H]           (fp32)
    total_rows,       # int (runtime) = B·N·N
    hidden_dim,       # int (compile‑time)
    seq_len,          # int (runtime) = N
    stride_b,         # int (runtime) – stride on dim 0 (B) of src
    stride_h,         # int (runtime) – stride on dim 1 (H) of src
    stride_i,         # int (runtime) – stride on dim 2 (N) of src
    stride_j,         # int (runtime) – stride on dim 3 (N) of src
    eps,              # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- decode (b,i,j) from linear row index -----------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # Base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # Pass 1 – mean & variance across hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – normalize, apply γ/β, multiply with out_gate and store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # Load the raw hidden vector
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm (fp32)
        norm = (raw - mean[:, None]) * rstd[:, None]

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                         # fp32

        # Multiply with out_gate (already fp16 → promote to fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # Store final hidden representation as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # model dimension (e.g., 128 or 384)
    hidden_dim   = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps          = 1e-5
    device       = input_tensor.device
    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N                 # R = B·N·N

    # -----------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # -----------------------------------------------------------------
    # View as (R, dim). Input is float32 → keep that for accurate LN.
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256          # rows per program (tuned)
    BLOCK_D = 64           # dim tile (covers up to dim=384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 2) One fused linear producing left/right + 3 gates (5*H outputs)
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices once and cast to fp16.
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Linear: [R, dim] @ [dim, 5*H]^T → [R, 5*H]   (fp16, no bias)
    proj_out = F.linear(x_norm, w_concat)   # shape (R, 5*hidden_dim)

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate   = proj_out[:, 2*H:3*H]
    right_gate  = proj_out[:, 3*H:4*H]
    out_gate    = proj_out[:, 4*H:5*H]

    # Sigmoid gates (still fp16)
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 3) Optional mask (applied only to the two projections)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------------
    # 4) Apply gates → gated tensors `left` and `right`
    # -----------------------------------------------------------------
    left  = left_proj * left_gate      # [R, H]
    right = right_proj * right_gate    # [R, H]

    # Reshape back to (B, N, N, H) – still contiguous because H is the last dim
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # -----------------------------------------------------------------
    # 5) Batched matrix multiplication (no extra copies)
    #    out_perm: (B, H, N, N) = left @ rightᵀ
    # -----------------------------------------------------------------
    # Both tensors are non‑contiguous views; torch.matmul can handle this.
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N)
    out_perm = torch.matmul(
        left_perm,
        right_perm.transpose(-1, -2)
    )                                             # (B, H, N, N)

    # -----------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Result is a contiguous (R, H) fp16 tensor `hidden_out`.
    # -----------------------------------------------------------------
    # Strides of the source tensor (non‑contiguous view)
    stride_b = out_perm.stride(0)   # batch stride
    stride_h = out_perm.stride(1)   # hidden‑dim stride (N*N)
    stride_i = out_perm.stride(2)   # i‑axis stride (N)
    stride_j = out_perm.stride(3)   # j‑axis stride (1)

    hidden_out = torch.empty(total_rows, H,
                             dtype=torch.float16,
                             device=device)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid_ln2](
        out_perm,                               # src (non‑contig)
        out_gate,                               # gate (contig, [R,H])
        hidden_out,                             # dst (contig)
        weights["to_out_norm.weight"],          # γ (fp32)
        weights["to_out_norm.bias"],            # β (fp32)
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 7) Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # -----------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(hidden_out, w_out)            # [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
