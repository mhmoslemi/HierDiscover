@triton.jit
def final_preproj_kernel(out_ptr, out_gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr, out_norm_gate_ptr, total_rows, H, N, eps, BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask = offs_m < total_rows
    # compute i,j similarly as before
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j
    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        out_vals = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)  # [M, K]
        gate_vals = tl.load(out_gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        mu = tl.load(mu_ptr + offs_m, mask=mask, other=0.0)[:, None]  # [M,1]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask, other=0.0)[:, None]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)[None, :]  # [1,K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)[None, :]

        # Compute normalized
        norm = ((out_vals - mu) * rstd) * gamma + beta
        # Multiply gate
        final = norm * gate_vals

        # store to out_norm_gate_ptr
        tl.store(out_norm_gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                 final.to(tl.float16),
                 mask=mask[:, None] & mask_k[None, :])

@triton.jit
def rowwise_stats_flat_kernel(
    out_ptr, mu_ptr, rstd_ptr, rows, hidden_dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, hidden_dim, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        ptr = out_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation in Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor  (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (FP16) → tensor `[R, 5·H]`   (R = B·N·N, H = hidden_dim).
3️⃣  Split the fused projection, apply the (optional) pair mask and the three
    sigmoid gates in a Triton kernel → tensors `left`, `right`, `out_gate`
    with layout `[B, H, N, N]` (FP16).
4️⃣  Core “einsum’’  `out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]`  is a
    batched GEMM and is performed with cuBLAS (`torch.bmm`).
5️⃣  Row‑wise mean & rsqrt‑variance of `out` (Triton, on the flattened
    `[R, H]` view).
6️⃣  Fuse the second LayerNorm and the `out_gate` multiplication in a
    Triton kernel → a contiguous tensor `[R, H]` ready for the final
    linear projection.
7️⃣  Final projection (`R×H → R×dim`) with cuBLAS (`F.linear`).  The result
    is reshaped to `[B, N, N, dim]` (float32).

All kernels run in mixed‑precision (FP32 for statistics, FP16 for the bulk)
and target an H100 (Triton 3.3.1).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm  (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C]   FP32 input
    y_ptr,      # [R, C]   FP16 output
    w_ptr,      # [C]      γ (FP32)
    b_ptr,      # [C]      β (FP32)
    rows,       # scalar   R
    dim,        # scalar   C
    eps,        # scalar   ε
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass : mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, dim, BLOCK_D):
        offs_d = d_off + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass : write normalized values ----------
    for d_off in range(0, dim, BLOCK_D):
        offs_d = d_off + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]               # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]   FP16
    right_ptr,     # [B, H, N, N]   FP16
    out_gate_ptr,  # [B, H, N, N]   FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    # start offset of hidden‑plane 0 inside the [B, H, N, N] tensor
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)                  # [BLOCK_H]
        mask_h = offs_h < H

        # fused proj layout: [R, 5·H] → contiguous per‑row
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride        # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance on the flattened [R, H] tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_flat_kernel(
    out_ptr,   # [R, H]   FP16
    mu_ptr,    # [R]      FP32
    rstd_ptr,  # [R]      FP32
    rows,      # scalar   R
    hidden,    # scalar   H
    eps,       # scalar   ε
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, hidden, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < hidden

        ptr = out_ptr + offs_m[:, None] * hidden + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm + out_gate (write to contiguous [R, H])
# --------------------------------------------------------------
@triton.jit
def final_preproj_kernel(
    out_ptr,        # [R, H]   FP16 (contracted tensor)
    gate_ptr,       # [R, H]   FP16 (out_gate)
    mu_ptr,         # [R]      FP32
    rstd_ptr,       # [R]      FP32
    ln_w_ptr,       # [H]      FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]      FP32 (to_out_norm.bias)
    out_pre_ptr,    # [R, H]   FP16 (pre‑projection tensor)
    rows, hidden, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]

    for k_start in range(0, hidden, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                    # [BLOCK_K]
        mask_k = offs_k < hidden

        out_vals = tl.load(out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)                # [M,K]
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)[None, :]   # [1,K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)[None, :]   # [1,K]

        # (out - µ) * rstd → LayerNorm → * out_gate
        norm = ((out_vals - mu) * rstd) * gamma + beta
        out_pre = norm * gate_vals

        tl.store(out_pre_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                 out_pre.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float)   or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                # flattened pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse 5 linear maps (left, right, gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                     # [R, 5·H]  FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # left/right : [B, H, N, N] → view as [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (flattened view)
    # --------------------------------------------------------------
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)      # [R, H] FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                         # H ≤ 128 → one tile suffices
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_flat_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate (produces tensor ready for final proj)
    # --------------------------------------------------------------
    out_preproj = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_PRE = 128
    BLOCK_K_PRE = 128               # full hidden dim in one iteration
    grid_pre = (triton.cdiv(R, BLOCK_M_PRE),)
    final_preproj_kernel[grid_pre](
        out_flat,
        out_gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],    # γ   FP32
        w["to_out_norm.bias"],      # β   FP32
        out_preproj,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_PRE,
        BLOCK_K=BLOCK_K_PRE,
        num_warps=8,
    )
    # release tensors that are no longer needed
    del out, out_gate, out_flat, out_gate_flat, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS) and reshape
    # --------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)          # [dim, H]
    final = F.linear(out_preproj, w_out)                 # [R, dim]  FP16
    final = final.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim]  FP32

    return final
