"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  First LayerNorm on the flattened pair tensor (FP32 → FP16) – torch.nn.functional.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
    single GEMM (FP16) → tensor `[R, 5·H]` where `R = B·N·N`, `H = hidden_dim`.
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection, applies the
    optional pair mask, computes three sigmoids and writes three tensors
    `left`, `right`, `out_gate` with layout `[B, H, N, N]` (FP16).
4️⃣  Core einsum is a batched GEMM (`torch.bmm`) on the reshaped tensors:
    `out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]` → shape `[B, H, N, N]` (FP16).
5️⃣  Second LayerNorm + out‑gate are performed with PyTorch (`torch.nn.functional.layer_norm`
    + element‑wise multiply) on the view `[R, H]`.
6️⃣  Final projection (`to_out.weight`) is a single large GEMM (`torch.nn.functional.linear`).
    The result is cast back to FP32 and reshaped to `[B, N, N, dim]`.

Only the split‑and‑gate step is custom‑written in Triton; all heavy GEMMs use
cuBLAS / torch's highly‑optimized kernels, dramatically reducing launch overhead
and memory traffic compared with the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates (Triton)
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N]  FP16 output
    right_ptr,       # [B, H, N, N]  FP16 output
    out_gate_ptr,    # [B, H, N, N]  FP16 output
    mask_ptr,        # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows of flattened pair tensor processed per program
    BLOCK_H: tl.constexpr,   # hidden dimension tile per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # ----- decode flat index r = b·N·N + i·N + k -----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ----- optional mask (broadcast over H) -----
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    # base offset for hidden‑plane h == 0 (contiguous layout of torch tensor)
    base = b * (H * hidden_stride) + i * N + k

    # ----- iterate over hidden‑dimension tiles -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: consecutive H‑slices for each of the 5 groups
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ---------- sigmoid (computed in FP32 for numerical stability) ----------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---------- apply mask and gates ----------
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # destination offsets (row‑major, hidden stride = N·N)
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = inp.shape
    R = B * N * N                     # flattened pair dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    inp_flat = inp.reshape(R, dim)

    # native torch LayerNorm – operates in FP32 for stability
    x_norm_fp32 = F.layer_norm(
        inp_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    # cast to FP16 for the remaining GEMMs
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    # (R, dim) @ (5·H, dim).t()  →  (R, 5·H)
    proj = F.linear(x_norm, fused_w)            # FP16, shape [R, 5·H]

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask broadcasted over hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free memory we no longer need
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # view as [B·H, N, N] → (B·H) independent matrix multiplications
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out = left @ rightᵀ   (fp16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                     # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (pure PyTorch)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H)                   # [R, H]
    # Cast LN weights/bias to fp16 for speed (the op will promote to the input dtype)
    ln_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    ln_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    # LayerNorm over the hidden dimension H
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=ln_w_fp16,
        bias=ln_b_fp16,
        eps=eps,
    )                                              # fp16

    # Apply out‑gate (element‑wise)
    out_gate_flat = out_gate.reshape(R, H)
    gated = out_norm * out_gate_flat                # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (cuBLAS) → FP32 output
    # ------------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = F.linear(gated, w_out_fp16)            # (R, dim), fp16

    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim] FP32

    return final
