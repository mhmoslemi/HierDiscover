"""
Highly‑optimized forward‑only Triton implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 → float16) over the last dimension `dim`.
2️⃣ Fused linear projection that produces the five matrices
   (left, right, left‑gate, right‑gate, out‑gate) in FP16.
   The projection is done with a single `torch.nn.functional.linear` call
   (cuBLAS) after the LayerNorm; the intermediate normalized tensor is
   materialised only once.
3️⃣ `split_fuse_gate_kernel` extracts the five slices, applies the optional
   pair‑wise mask and the three sigmoid gates, writing `left`, `right`
   and `out_gate` in layout `[B, H, N, N]` (H = hidden_dim).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]    →  `[B, H, N, N]` (FP16)
5️⃣ `final_fused_kernel` merges the *row‑wise* mean/variance computation,
   the second `LayerNorm` (over the hidden dimension), the out‑gate
   multiplication and the final linear projection into a **single** Triton
   kernel.  This eliminates an extra pass over the large tensor `out`,
   cutting memory traffic and runtime.
6️⃣ The kernel writes the result in float32 with shape `[B, N, N, dim]`.

The implementation works for all test configurations listed in the prompt.

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 input (flattened)
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim] γ (FP32)
    b_ptr,        # [dim] β (FP32)
    rows,         # total rows R (scalar)
    dim,          # channel dimension `dim` (scalar)
    eps,          # epsilon (scalar)
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean and variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                         # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                  # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional) – 0/1
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional broadcast mask (float16 0/1)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                       # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k   # offset of hidden‑plane 0

    # loop over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # flat offsets into the fused projection matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],

            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,

        # sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate                                   # <-- NOT masked!

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N] FP16   (result of left·right)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,         # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (tiling over R)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)       # [BLOCK_N]

    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j      # offset of hidden‑plane 0

    # ---- accumulators -------------------------------------------------------
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)      # Σ out_h
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)      # Σ out_h²

    acc_S_out  = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out_h·gate_h·γ·W
    acc_S_gate = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ gate_h·γ·W
    acc_S_beta = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β·gate_h·W

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < H

        # load out and gate for this hidden tile
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # update row‑wise moments
        sum_out += tl.sum(vals_out, axis=1)
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)

        # layer‑norm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k,  mask=mask_k, other=0.0)   # [K] FP32

        # broadcast to rows
        gamma_b = gamma[None, :]     # [1,K]
        beta_b  = beta[None, :]      # [1,K]

        # contributions for the final linear projection
        tmp_out_gate = vals_out * vals_gate * gamma_b               # [M,K]
        tmp_gate     = vals_gate * gamma_b                         # [M,K]
        tmp_beta     = vals_gate * beta_b                          # [M,K]

        # load slice of final weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)                      # [K, TILE_N]

        # accumulate three dot‑products
        acc_S_out  += tl.dot(tmp_out_gate, w_tile)   # [M,N]
        acc_S_gate += tl.dot(tmp_gate,     w_tile)
        acc_S_beta += tl.dot(tmp_beta,     w_tile)

    # ---- compute μ, rsqrt‑variance ------------------------------------------------
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                       # [M]

    # broadcast μ & rstd across output dim
    mu_b   = mu[:, None]      # [M,1]
    rstd_b = rstd[:, None]    # [M,1]

    # final formula:
    #   out = rstd * (S_out - mu * S_gate) + S_beta
    out = rstd_b * (acc_S_out - mu_b * acc_S_gate) + acc_S_beta   # [M,N]

    # ---- store --------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, out, mask=mask_m[:, None] & mask_n[None, :])

            - mask_tensor  : torch.Tensor  [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",

                "to_out_norm.weight", "to_out_norm.bias",

              (all float32)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # flatten over pair‑wise dimension
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 128          # one iteration for dim=128, 3 for dim=384
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,

    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    )
    return final
