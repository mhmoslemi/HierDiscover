"""
Outgoing TriMul (AlphaFold‑3 style) – forward only

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16)  
2️⃣  Fuse the 5 linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
    single `torch.nn.functional.linear` → (R, 5·H) FP16.
3️⃣  Triton kernel `split_fuse_gate_kernel`  
    * decodes the flat index (b,i,k) → (b,i,k) layout,  
    * applies the optional mask,  
    * computes the sigmoids for the three gates (FP32 → FP16),  
    * writes `left`, `right` and `out_gate` in layout `[B,H,N,N]` (FP16).
4️⃣  Batched GEMM (`torch.bmm`) implements the core einsum  
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k] →  [B,H,N,N] (FP16)
5️⃣  Triton kernel `rowwise_stats_kernel` computes per‑row mean & rsqrt‑variance
    over the hidden dimension H (needed for the second LayerNorm).
6️⃣  The remaining post‑processing – second LayerNorm, element‑wise out‑gate
    multiplication and the final projection – is carried out with highly
    optimized PyTorch kernels (`torch.nn.functional.layer_norm`‑like ops and a
    single `torch.nn.functional.linear`).  This avoids a second custom kernel
    and leverages cuBLAS for the final matrix‑multiply.

All kernels use mixed‑precision: FP16 for storage/computation, FP32 for
reductions.  The implementation works for any batch size ≤2, any sequence
length in {128,256,512,1024} and hidden dimensions 128 (c) with `dim` ∈
{128,384}.  The mask is optional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, D]  FP32 input
    y_ptr,               # [R, D]  FP16 output
    w_ptr,               # [D]    LN weight (γ)   FP32
    b_ptr,               # [D]    LN bias  (β)   FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, optional mask, and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R,5·H] FP16 fused projection
    left_ptr,       # [B,H,N,N] FP16
    right_ptr,      # [B,H,N,N] FP16
    out_gate_ptr,   # [B,H,N,N] FP16
    mask_ptr,       # [R]    FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]          # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Base offset for tensors of shape [B,H,N,N] (h = 0)
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k        # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # View of the fused projection: 5 blocks of size H each
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # address of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): either (input, mask, weights, config) or (input, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, on the correct cuda device)
            - config : dict containing at least "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments (support both 3‑ and 4‑element signatures)
    # ------------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, cfg = data
    elif len(data) == 3:
        x, weights, cfg = data
        mask = None
    else:
        raise ValueError("expected data tuple of length 3 or 4")

    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_ln = 128
    BLOCK_D_ln = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_ln),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_ln,
        BLOCK_D=BLOCK_D_ln,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (5·H) – no bias
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R,5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, and sigmoid → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are [B,H,N,N] FP16 → reshape for bmm
    left_mat = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)              # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_stats = 256
    BLOCK_K_stats = 64          # H ≤ 128 ⇒ two tiles at most
    grid_stats = (triton.cdiv(R, BLOCK_M_stats),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_stats,
        BLOCK_K=BLOCK_K_stats,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (all in PyTorch)
    # ------------------------------------------------------------------
    # Permute to [B,N,N,H] → flatten to (R, H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B,N,N,H] FP16
    out_flat = out_perm.view(R, H).to(torch.float32)         # FP32 for stats

    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP16
    out_gate_flat = out_gate_perm.view(R, H).to(torch.float32)   # FP32

    # Apply the second LayerNorm (γ,β are to_out_norm params)
    gamma = weights["to_out_norm.weight"]   # [H] FP32
    beta  = weights["to_out_norm.bias"]     # [H] FP32

    # (out - μ) * rstd
    normed = ((out_flat - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

    # Gating
    gated = normed * out_gate_flat

    # Final linear projection (hidden_dim → dim)
    # weight shape [dim, hidden_dim]; F.linear expects [out_features, in_features]
    final_fp32 = F.linear(gated, weights["to_out.weight"])   # [R, dim] FP32

    # Reshape back to [B, N, N, dim]
    final = final_fp32.view(B, N, N, dim)

    return final
