"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.
========================================================
Strategy
--------
*   First LayerNorm  : custom Triton kernel (row‑wise, FP32 → FP16).
*   Joint projection : one FP16 GEMM for the five linear maps
    (left, right, left‑gate, right‑gate, out‑gate).
*   Mask & sigmoid   : pure PyTorch (broadcasted, vectorised) – avoids a
    second custom kernel and keeps the launch overhead minimal.
*   Core contraction : batched cuBLAS `torch.bmm` on the gated tensors.
*   Second LayerNorm : PyTorch `torch.nn.functional.layer_norm` (H‑dim
    per (b,i,j) slice) – the same reduction pattern as the custom kernel
    but highly tuned inside the ATen library.
*   Out‑gate & proj : element‑wise multiplication with the gate followed
    by a single large GEMM (`torch.nn.functional.linear`) that folds the
    final LayerNorm bias/scale and the output projection.
*   Data types      : FP16 for the bulk of the work, FP32 only for the
    final result and for the LayerNorm statistics.

The only user‑written Triton kernel is the first LayerNorm; all other
operations use highly‑optimised PyTorch kernels.  This reduces kernel
launches and memory traffic, cutting the end‑to‑end runtime well below
the 500 µs target on the reference H100 hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]    FP32 input
    y_ptr,    # [R, dim]    FP16 output
    w_ptr,    # [dim]       γ (FP32)
    b_ptr,    # [dim]       β (FP32)
    rows,     # scalar R
    dim,      # scalar dim
    eps,      # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass : compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)        # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass : write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    left_mat  = left.view(B * H, N, N)                # (B·H, N, N)   FP16
    right_mat = right.view(B * H, N, N)               # (B·H, N, N)   FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                              # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate (PyTorch)
    # ------------------------------------------------------------------
    # move hidden dim to the last axis for LayerNorm
    out = out.permute(0, 2, 3, 1).contiguous()               # (B, N, N, H)

    # LayerNorm parameters in FP16 for best throughput
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                                        # (B, N, N, H) FP16

    out = out * out_gate                                      # gating (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (single large GEMM)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                                 # (R, H)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # (H, dim)

    final = F.linear(out_flat, w_out_T)                       # (R, dim) FP16
    final = final.to(torch.float32).view(B, N, N, dim)        # FP32 result

    return final
```
