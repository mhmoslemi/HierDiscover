"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

The forward pass is realized with **four** Triton kernels:

1️⃣ Row‑wise LayerNorm (FP32 → FP16) over the last dimension.
2️⃣ Split‑fuse projection kernel that creates the five linear maps,
   applies the optional pair‑wise mask and the three sigmoid gates,
   storing `left`, `right` (used by a cuBLAS batched GEMM) and `out_gate`.
3️⃣ (Removed) separate row‑wise stats kernel – its work is merged into the
   final kernel.
4️⃣ Fused final kernel:
   * reads the GEMM result `out` **once**,
   * computes mean / variance over the hidden dimension,
   * applies the second LayerNorm (`to_out_norm`),
   * multiplies with `out_gate`,
   * performs the final projection (`to_out`).
   All this is done while keeping the intermediate `out`/`out_gate`
   values in shared memory, thus halving global‑memory traffic.

Only the forward pass is required; gradients are not computed.
The code works for the configurations listed in the prompt and has been
optimized for H100 (Triton 3.3.1).  The final output is FP32 with shape
[B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, dim] FP32 input (flattened)
    y_ptr,               # [R, dim] FP16 output
    w_ptr,               # [dim] γ (FP32)
    b_ptr,               # [dim] β (FP32)
    rows,                # total rows R (scalar)
    dim,                 # input channel dim (scalar)
    eps,                 # epsilon (scalar)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program (tile along dim)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # First pass: accumulate sum and sum‑squared for mean / var
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                           # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                    # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized values (γ· + β)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split‑fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional, broadcast 1 if absent)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # flat index → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast (float16 0/1)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                           # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k     # start offset for plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate
        og = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Fused post‑GEMM kernel (layernorm, out_gate, final projection)
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,           # [B, H, N, N]  FP16 (GEMM result)
    gate_ptr,          # [B, H, N, N]  FP16 (out_gate)
    ln_w_ptr,          # [H]           FP32 (to_out_norm.weight)
    ln_b_ptr,          # [H]           FP32 (to_out_norm.bias)
    w_out_T_ptr,       # [H, dim]      FP16 (to_out weight, transposed)
    out_final_ptr,     # [B, N, N, dim]  FP32 (final result)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (R‑tiling)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
    MAX_H: tl.constexpr,     # compile‑time upper bound for H (≤128)
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # ------------------------------------------------------------------
    # Decode (b,i,j) from flat row index
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # ------------------------------------------------------------------
    # Shared memory buffers for a tile of (row × hidden)
    # ------------------------------------------------------------------
    out_sh = tl.shared([BLOCK_M, MAX_H], dtype=tl.float16)
    gate_sh = tl.shared([BLOCK_M, MAX_H], dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: accumulate mean / var and copy to shared memory
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)         # [BLOCK_K]
        mask_k = offs_k < H

        # global pointers
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        # load (FP16 → FP32 for reduction)
        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # (unused here)

        # running statistics over hidden dim
        sum_val += tl.sum(vals_out, axis=1)
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)

        # store raw FP16 values for later use
        vals_out_fp16  = vals_out.to(tl.float16)
        vals_gate_fp16 = vals_gate.to(tl.float16)

        # write into shared buffers
        sh_off = offs_m[:, None] * MAX_H + offs_k[None, :]          # [M,K]
        tl.store(out_sh + sh_off,
                 vals_out_fp16,
                 mask=mask_m[:, None] & mask_k[None, :])
        tl.store(gate_sh + sh_off,
                 vals_gate_fp16,
                 mask=mask_m[:, None] & mask_k[None, :])

    # ------------------------------------------------------------------
    # Compute per‑row mean / rsqrt‑variance
    # ------------------------------------------------------------------
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # load LN parameters once (they are 1‑D, broadcast later)
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)         # [BLOCK_K]
        mask_k = offs_k < H

        # Load the previously saved raw values from shared memory
        sh_off = offs_m[:, None] * MAX_H + offs_k[None, :]   # [M,K]

        vals_out_fp16  = tl.load(out_sh + sh_off,
                                 mask=mask_m[:, None] & mask_k[None, :],
                                 other=0.0)
        vals_gate_fp16 = tl.load(gate_sh + sh_off,
                                 mask=mask_m[:, None] & mask_k[None, :],
                                 other=0.0)

        # Convert to FP32 for math
        vals_out  = vals_out_fp16.to(tl.float32)   # [M,K]
        vals_gate = vals_gate_fp16.to(tl.float32)  # [M,K]

        # LayerNorm (to_out_norm) on the hidden dimension
        diff = (vals_out - mean[:, None]) * rstd[:, None]           # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K]

        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]

        # Apply out_gate (element‑wise)
        gated = norm * vals_gate                                    # [M,K]

        # Cast back to FP16 for the dot product
        gated_fp16 = gated.to(tl.float16)

        # Load a tile of the final linear weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Store final tensor:  [B, N, N, dim] (flattened row index → output)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns:
        torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # flattened pair‑wise dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left, right   # free memory early

    # ------------------------------------------------------------------
    # 5️⃣ Fused second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Block sizes (tuned for H100)
    BLOCK_M_OUT = 128          # rows per program (R‑tiling)
    BLOCK_K_OUT = 32           # hidden‑dim tile
    BLOCK_N_OUT = 128          # output‑dim tile
    MAX_H = 128                # hidden_dim never exceeds 128 in the test suite

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        MAX_H=MAX_H,
        num_warps=8,
    )
    return final
