def layernorm_kernel(
    x_ptr,            # [R, D] fp32 input
    y_ptr,            # [R, D] fp16 output
    w_ptr,            # [D]   γ (fp32)
    b_ptr,            # [D]   β (fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of pairwise tokens (flattened)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32) shape [dim]
        weights["norm.bias"],     # β (fp32) shape [dim]
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory early

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    # Concatenate the five projection matrices once (fp16)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                     # (5·H, dim)
    ).to(torch.float16).contiguous()

    # Linear projection – pure torch (cuBLAS)
    proj = F.linear(x_norm, fused_weight)          # [R, 5·H] fp16
    del fused_weight, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & sigmoid gates (pure torch)
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                      # [R, 5, H]

    left_raw   = proj[:, 0, :]                     # [R, H]
    right_raw  = proj[:, 1, :]                     # [R, H]
    left_gate_raw  = proj[:, 2, :]                 # [R, H]
    right_gate_raw = proj[:, 3, :]                 # [R, H]
    out_gate_raw   = proj[:, 4, :]                 # [R, H]

    # sigmoid gates (fp16)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)      # [R, H] (no mask)

    # optional mask handling
    if (mask is not None) and (not nomask):
        mask_flat = mask.to(torch.float16).reshape(R).unsqueeze(-1)   # [R,1]
        left  = left_raw  * left_gate  * mask_flat
        right = right_raw * right_gate * mask_flat
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (core einsum) – torch.bmm
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] -> view to [B·H, N, N] for torch.bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N] fp16

    left_mat  = left.reshape(B * H, N, N)      # [B·H, N, N] fp16
    right_mat = right.reshape(B * H, N, N)     # [B·H, N, N] fp16

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16

    del left, right, left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣  LayerNorm over hidden dimension + out_gate multiplication
    # ------------------------------------------------------------------
    # Flatten to shape [R, H] with ordering (b, i, j)
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)   # fp16
    out_gate_flat = out_gate                               # fp16, already flat

    # 2‑pass LayerNorm (torch – fp32) over H
    out_norm = F.layer_norm(
        out_flat.float(),                     # cast to fp32
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                        # [R, H] fp32

    # element‑wise gating (convert gate to fp32)
    out_norm = out_norm * out_gate_flat.float()

    # ------------------------------------------------------------------
    # 6️⃣  Final projection (to_out) – torch.linear
    # ------------------------------------------------------------------
    # weight shape: [dim, H] (fp32)
    final = F.linear(out_norm, weights["to_out.weight"])   # [R, dim] fp32

    # reshape back to [B, N, N, dim]
    final = final.view(B, N, N, dim)

    return final
