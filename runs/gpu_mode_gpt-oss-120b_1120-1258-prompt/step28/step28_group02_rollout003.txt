@triton.jit
def final_norm_output_kernel(
    out_ptr,          # [R, H] FP16 (flattened out_perm)
    gate_ptr,         # [R, H] FP16 (flattened out_gate_perm)
    ln_w_ptr,         # [H] FP32
    ln_b_ptr,         # [H] FP32
    w_out_T_ptr,      # [H, dim] FP16
    out_final_ptr,    # [R, dim] FP32
    R, H, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # ---------------------------------
    # Compute sum and sumsq per row
    # ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------
    # Second pass: compute final output
    # ---------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate
        out_ptr_tile = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        out_vals = tl.load(out_ptr_tile, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32) # [M,K]

        gate_ptr_tile = gate_ptr + offs_m[:, None] * H + offs_k[None, :]
        gate_vals = tl.load(gate_ptr_tile, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32) # [M,K]

        # Normalize
        # (out - mu) * rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]  # [M,K]

        # Apply LN gamma and beta
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = diff * gamma[None, :] + beta[None, :]   # [M,K]

        gated = norm * gate_vals  # [M,K]

        # Cast to fp16 for dot with weight
        gated_fp16 = gated.to(tl.float16)

        # Load weight tile
        w_ptr_tile = w_out_T_ptr + offs_k[:, None] * dim + tl.arange(0, BLOCK_N)[None, :]   # [K, N]
        # but need mask for N as well
        offs_n = tl.arange(0, BLOCK_N)
        mask_n = offs_n < dim
        w_tile = tl.load(w_ptr_tile,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)  # [K, BLOCK_N]

        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # Store
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + tl.arange(0, BLOCK_N)[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  

The forward pass is divided into a handful of fused kernels:

1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor `[B,N,N,C]`.  
2️⃣  One fused linear projection that produces the five matrices
    (left, right, left‑gate, right‑gate, out‑gate) in FP16.  
3️⃣  `split_fuse_gate_kernel` extracts the five slices, applies the optional
    pair‑wise mask and the three sigmoid gates, writing `left`/`right` in the
    layout `[B, H, N, N]` (required for the subsequent batched GEMM) and
    `out_gate` in the layout `[B, N, N, H]` (contiguous over the hidden dim).  
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum
    `out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]` → tensor `[B, H, N, N]`.  
5️⃣  The tensor is permuted to `[B, N, N, H]` (contiguous over `H`).  
6️⃣  `final_norm_output_kernel` fuses **both** the second LayerNorm,
    the `out_gate` multiplication and the final linear projection.
    It first reduces over the hidden dimension to obtain mean/variance,
    then normalises, applies the LayerNorm γ/β, multiplies by the gate,
    and finally performs a fused dot‑product with the transposed output
    weight `W_outᵀ`.  The result is written as FP32 `[B,N,N,C]`.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, C] FP32 input (flattened)
    y_ptr,    # [R, C] FP16 output
    w_ptr,    # [C] γ (FP32)
    b_ptr,    # [C] β (FP32)
    rows,     # total rows R (scalar)
    dim,      # channel dimension C (scalar)
    eps,      # epsilon (scalar)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                    # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)             # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Split & gate – writes left/right in [B,H,N,N] and out_gate in [B,N,N,H]
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, N, N, H] FP16 (contiguous over H)
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask (float16 0/1) or constant 1.0
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes for left/right
    gate_stride = N * H                      # stride inside out_gate layout (per i)

    # base offset for left/right (start of hidden plane 0)
    base_lr = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # base offset for out_gate (start of hidden dim 0 in [B,N,N,H])
    base_gate = b * (N * N * H) + i * (N * H) + k * H          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside the fused projection (row-major)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for numerical stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        # write left/right (strided over hidden dimension)
        dst_lr = base_lr[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst_lr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_lr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # write out_gate (contiguous over H)
        dst_gate = base_gate[:, None] + offs_h[None, :]               # [BLOCK_R, BLOCK_H]
        tl.store(out_gate_ptr + dst_gate,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Fuse second LayerNorm, out_gate and final projection (no extra stats kernel)
# ----------------------------------------------------------------------
@triton.jit
def final_norm_output_kernel(
    out_ptr,        # [R, H] FP16  (out after permute)
    gate_ptr,       # [R, H] FP16  (out_gate after permute)
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out.weight.T)
    out_final_ptr,  # [R, dim] FP32
    R, H, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)           # row tile
    pid_n = tl.program_id(1)         # output‑dim tile

    row_start = pid * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)           # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        # out slice
        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: LayerNorm, gate, projection ----------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load out and gate vectors for this hidden tile
        out_tile = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        gate_tile = gate_ptr + offs_m[:, None] * H + offs_k[None, :]

        out_vals  = tl.load(out_tile,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)        # [M,K]
        gate_vals = tl.load(gate_tile,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)        # [M,K]

        # (out - mu) * rstd   →   LayerNorm γ/β   →   * gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]          # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]
        gated = norm * gate_vals                                    # [M,K]
        gated_fp16 = gated.to(tl.float16)                           # back to FP16 for dot

        # weight tile (H x dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, BLOCK_N]

        # accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # store final result
    dst = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim" (int) and "hidden_dim" (int);
              optionally "nomask" (bool).

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # flattened pair‑wise dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5·H outputs (left, right, gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, N, N, H, dtype=torch.float16, device=device)  # contiguous H

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ) → out [B, H, N, N]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                        # FP16
    right_mat = right.view(B * H, N, N)                       # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Permute tensors so hidden dimension is contiguous
    # ------------------------------------------------------------------
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] FP16
    gate_perm = out_gate.permute(0, 1, 2, 3).contiguous()    # already [B, N, N, H] FP16
    del out, out_gate, left, right

    # flatten pair‑wise dimension for the final kernel
    out_flat = out_perm.view(-1, H)        # [R, H] FP16
    gate_flat = gate_perm.view(-1, H)      # [R, H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gating + projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program
    BLOCK_K_OUT = 32     # hidden‑dim tile (fits in registers)
    BLOCK_N_OUT = 128    # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_output_kernel[grid_out](
        out_flat,
        gate_flat,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final.view(-1, dim),       # flatten final tensor for easy indexing
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # final already has shape [B, N, N, dim]
    return final
