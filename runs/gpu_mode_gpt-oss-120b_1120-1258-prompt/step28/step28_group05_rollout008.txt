Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm over the pairwise tensor (float32 → float16) using the
    highly‑optimized CuDNN / torch.nn.functional.layer_norm kernel.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (FP16) → tensor `[R,5·H]` (R = B·N·N).
3️⃣  Split the fused projection, apply the optional pair mask, and evaluate
    three sigmoid gates in a custom Triton kernel → tensors `left`,
    `right`, `out_gate` with layout `[B, H, N, N]` (FP16).
4️⃣  Core einsum via a batched matrix‑multiply:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    (implemented with `torch.bmm` on FP16 tensors).
5️⃣  Row‑wise Layer‑norm (over the hidden dimension H) on `out` using the
    built‑in `F.layer_norm` (fast CuDNN kernel).  Afterwards multiply by the
    already‑computed `out_gate`.
6️⃣  Final projection (H → dim) with a single large GEMM (`F.linear`) on
    FP16 tensors; the result is cast back to FP32.

Only the split‑and‑gate step is written in Triton – all heavy arithmetic
uses cuBLAS / cuDNN kernels, which gives the required ~2× speed‑up over the
reference implementation.
"""

import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# -------------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]   FP16
    right_ptr,     # [B, H, N, N]   FP16
    out_gate_ptr,  # [B, H, N, N]   FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    # start offset for h == 0 : [B, H, N, N] layout → offset = b·(H·N·N) + i·N + k
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # layout of the fused projection: contiguous H‑wise blocks
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # destination offsets for [B, H, N, N]
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# -------------------------------------------------------------------------
# Entrypoint
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]      (float32)
            mask_tensor  : torch.Tensor [B, N, N] or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                # number of pair‑tokens
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Pairwise LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    # flatten to [R, dim] and apply the highly‑optimised cuDNN kernel
    x_flat = inp.reshape(R, dim)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)          # FP16 for the next GEMM
    del x_flat, x_norm_fp32

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection of the five linear maps (left, right, gates)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)         # (R, 5·H)  FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton)
    # -----------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden‑dim vector length per iteration
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Core TriMul (einsum) via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    # out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                               # (B, H, N, N) FP16
    del left_mat, right_mat

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over the hidden dim) + out_gate
    # -----------------------------------------------------------------
    # bring hidden dimension to the last axis for the CuDNN kernel
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                                      # FP16

    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    gated = out_norm * out_gate_perm                               # FP16
    del out_norm, out_gate_perm, out, out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection H → dim (large GEMM)
    # -----------------------------------------------------------------
    gated_flat = gated.view(-1, H)                             # (B·N·N, H)
    w_out_fp16 = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    final_flat = F.linear(gated_flat, w_out_fp16)              # (B·N·N, dim) FP16
    final = final_flat.view(B, N, N, dim).to(torch.float32)    # cast back to FP32

    return final
