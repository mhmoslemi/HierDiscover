"""
Outgoing TriMul (AlphaFold‑3) – fused row‑wise LayerNorm + five linear
projections + gating.  The implementation follows the reference PyTorch
module but fuses the first three stages (LayerNorm, left/right projection,
and three gates) into a single Triton kernel, dramatically reducing memory
traffic.  The core ∑ₖ left·right contraction is performed with a batched
matrix‑multiply (torch.bmm).  Row‑wise statistics of the contracted tensor
and the final LayerNorm + out‑gate + output projection are also fused in
Triton.  All kernels target Triton 3.3.1 on an H100 and the final result is
returned as float‑32 of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fused LayerNorm + 5‑way projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    x_ptr,           # [R, D]   FP32 input (flattened pair tensor)
    left_ptr,        # [B, H, N, N]  FP16 output (left)
    right_ptr,       # [B, H, N, N]  FP16 output (right)
    out_gate_ptr,    # [B, H, N, N]  FP16 output (out_gate)
    mask_ptr,        # [R] FP16 mask (optional, broadcasted over H)
    w_ptr,           # [5*H, D] FP16 fused weight matrix
    norm_w_ptr,      # [D] FP32 LayerNorm gamma
    norm_b_ptr,      # [D] FP32 LayerNorm beta
    R, N, D, H,
    EPS: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim tile
    BLOCK_D: tl.constexpr,   # input‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # decode flat row index  r = b·N·N + i·N + k   (pair token)
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        m_val = m_val[:, None]                                       # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # ------------------------------------------------------------------
    # first pass – row‑wise mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < D

        # load a tile of the raw input
        x_tile = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x_tile, axis=1)
        sum_sq  += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ------------------------------------------------------------------
    # second pass – compute the five projections (FP16 → FP32 acc)
    # ------------------------------------------------------------------
    # accumulators for the five sub‑projections
    left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    hidden_stride = N * N
    # base offset for the three output tensors (before adding hidden offset)
    base = b * (H * hidden_stride) + i * N + k                # [BLOCK_R]

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < D

        # load raw input tile, normalize, and apply LN affine (γ,β)
        x_tile = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_R, BLOCK_D]

        # (x - μ) * rstd
        norm = (x_tile - mean[:, None]) * rstd[:, None]       # [BLOCK_R, BLOCK_D]

        # LN affine transformation
        gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        norm = norm * gamma[None, :] + beta[None, :]           # still FP32

        # cast to fp16 for tensor‑core dot
        norm_fp16 = norm.to(tl.float16)                       # [BLOCK_R, BLOCK_D]

        # loop over hidden‑dim tiles
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
            mask_h = offs_h < H

            # ------------------------------------------------------------------
            # weight loading – we load the transposed tile so that its shape is
            #   [BLOCK_D, BLOCK_H] (dim‑major) which matches tl.dot(A,B)
            # ------------------------------------------------------------------
            for proj_idx in range(5):
                # pointer = d (row) + (proj_offset + h) * D   (col‑major)
                w_tile = tl.load(
                    w_ptr
                    + offs_d[:, None]                      # [BLOCK_D, 1]
                    + (proj_idx * H + offs_h)[None, :] * D, # [1, BLOCK_H] * D
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0,
                )                                             # [BLOCK_D, BLOCK_H]

                # dot: [BLOCK_R, BLOCK_D] @ [BLOCK_D, BLOCK_H] -> [BLOCK_R, BLOCK_H]
                if proj_idx == 0:
                    left_acc += tl.dot(norm_fp16, w_tile)
                elif proj_idx == 1:
                    right_acc += tl.dot(norm_fp16, w_tile)
                elif proj_idx == 2:
                    left_gate_acc += tl.dot(norm_fp16, w_tile)
                elif proj_idx == 3:
                    right_gate_acc += tl.dot(norm_fp16, w_tile)
                else:  # proj_idx == 4
                    out_gate_acc += tl.dot(norm_fp16, w_tile)

    # ------------------------------------------------------------------
    # apply sigmoid to gates, mask to left/right and write results
    # ------------------------------------------------------------------
    # sigmoid in FP32, then cast back to FP16
    left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
    right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

    # finalize left/right (apply gates and optional mask)
    left_fp16  = left_acc.to(tl.float16) * left_gate  * m_val
    right_fp16 = right_acc.to(tl.float16) * right_gate * m_val

    # destination offsets (layout [B, H, N, N] → flat)
    dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

    tl.store(left_ptr      + dst, left_fp16,      mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr     + dst, right_fp16,     mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr  + dst, out_gate,       mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Row‑wise statistics of the contracted tensor (mean & rsqrt‑var)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows,   # R = B·N·N
    H, N,
    EPS,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + EPS)

    tl.store(mu_ptr + offs_m,   mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# 3️⃣ Final LayerNorm + out_gate + projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    EPS,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden values and gating tensor
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (x - μ) * rstd → normalize → affine → gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term   = norm * vals_gate                                 # [M,K]
        term_fp16 = term.to(tl.float16)                           # fp16 for dot

        # load slice of final projection weight (H×dim) – already transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])
