"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm (GPU‑only, no grads)
--------------------------------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) via a
   small Triton kernel.

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single cuBLAS GEMM:
       (R, dim) × (dim, 5·H)ᵀ → (R, 5·H)   where R = B·N·N , H = hidden_dim.
   A second Triton kernel `split_fuse_gate_kernel` un‑flattens the result,
   applies the optional pair mask, evaluates the three sigmoids and writes
   three tensors:
       left   : [B, H, N, N]   FP16
       right  : [B, H, N, N]   FP16
       out_g  : [B, H, N, N]   FP16   (out_gate)

3️⃣ Batched GEMM (Tensor‑cores) computes the core pairwise contraction:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   Implemented with `torch.bmm`, the cost is O(B·H·N³) but cuBLAS/Tensor‑cores
   are already optimal for this size.

4️⃣ Second LayerNorm, out_gate multiplication and the final projection
   are performed entirely with PyTorch primitives (highly‑optimised
   reductions + GEMM).  The whole computation after step‑3 looks like:
       term    = (LayerNorm(out) ⊙ out_gate)                # [B,H,N,N]
       final   = term reshaped to (R,H)  @  to_out.weight   # [R,dim]
   The reduction for LayerNorm (mean & rsqrt‑var) is done with FP32
   accumulation for numerical stability; the subsequent mat‑mul uses
   FP16/Tensor‑cores and the result is cast back to FP32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required; no autograd hooks are used.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim]  FP32 input
    y_ptr,           # [R, dim]  FP16 output
    w_ptr,           # [dim]    γ  (FP32)
    b_ptr,           # [dim]    β  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 (fused output of the big GEMM)
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16   (optional, values 0/1)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode (b,i,k) from the flat pair index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)[:, None]          # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                      # distance between two hidden slices

    # base offset of the (b,i,k) position in the flattened [B,H,N,N] layout
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load the five projected slices (packed as [5·H])
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offset for this hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

    BLOCK_D = 64

        w["norm.bias"],

    # ------------------------------------------------------------------

    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16

    # ------------------------------------------------------------------

    # variance using E[x²] - μ²
    var = out_fp32.pow(2).mean(dim=1, keepdim=True) - mean.pow(2) # [B,1,N,N] FP32
    rstd = torch.rsqrt(var + eps)                                 # [B,1,N,N] FP32

    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1,H,1,1) FP32
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)     # (1,H,1,1) FP32

    # Normalized tensor, still FP32
    out_norm = ((out_fp32 - mean) * rstd) * gamma + beta

    # Apply out_gate (converted to FP32) – still FP32
    term_fp32 = out_norm * out_gate.float()           # [B,H,N,N] FP32

    # Reshape to (R, H) for the final linear projection
    term_fp16 = term_fp32.to(torch.float16)           # cast to FP16 for the GEMM
    term_reshaped = term_fp16.permute(0, 2, 3, 1).reshape(R, H)   # [R, H]

    # Final projection: (R, H) @ (H, dim) → (R, dim)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)          # [dim, H]
    final_fp16 = F.linear(term_reshaped, w_out_fp16)          # [R, dim] FP16

    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim] FP32

    return final
```
