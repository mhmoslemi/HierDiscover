```
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H] fp16
    left_ptr,    # [R, H] fp16
    right_ptr,   # [R, H] fp16
    out_gate_ptr,# [R, H] fp16
    mask_ptr,    # [R] fp16 optional
    R, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode for each row? Not needed for left/right computations other than flatten.
    # mask broadcast:
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)
    # iterate over hidden in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # base index in proj: row * (5*H) + offs_h
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # Apply sigmoid in float32 for stability
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offsets
        dst_base = offs_r[:, None] * H + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_base, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_base, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_base, out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

    multiplication and the final projection (`to_out`).  It reads the
    per‑row statistics, normalises `out`, multiplies by `out_gate`,
    and performs a dot‑product with the transposed `to_out` weight,
    writing the final `[B, N, N, dim]` tensor in FP32.

All kernels are written for Triton 3.3.1 and target mixed‑precision
(FP16 for bulk work, FP32 for statistics).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Split fused projection, apply mask and sigmoid gates.
# Output layout: [R, H] (contiguous) for left, right and out_gate.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]   FP16
    left_ptr,          # [R, H]     FP16
    right_ptr,         # [R, H]     FP16
    out_gate_ptr,      # [R, H]     FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # broadcast mask (if present) to hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base index for the projected tensor of shape [R, 5*H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # load the five chunks
        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 (more stable) and cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets in flat [R, H] layout
        dst = offs_r[:, None] * H + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,     left,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,    right,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Fuse second LayerNorm, out_gate and final projection.
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]  FP16
    gate_ptr,       # [B, H, N, N]  FP16 (out_gate)
    mu_ptr,         # [R]           FP32
    rstd_ptr,       # [R]           FP32
    ln_w_ptr,       # [H]           FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]           FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]      FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows over R
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N                      # stride for a single hidden slice
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # offset of the first hidden entry for (b,i,j)
    base_offset = b * (H * nn) + i * N + j        # scalar offset for h=0

    # load per‑row statistics (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # iterate over hidden dimension in tiles of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load slices of out and out_gate (FP16 → FP32)
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * nn
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * nn

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)    # [M, K]

        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # Layer‑norm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → *γ + β
        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out_gate
        term = norm * vals_gate

        # cast back to FP16 for the GEMM with the final projection
        term_fp16 = term.to(tl.float16)

        # load a tile of the transposed output weight [K, BLOCK_N]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)   (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final tensor (flattened view of [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use the highly‑optimized cuDNN kernel
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()   # [R, dim]   FP32
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps
    )
    x_norm = x_norm_fp32.to(torch.float16)     # [R, dim]   FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection producing (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # [R, 5·H]   FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split / mask / sigmoid gates (Triton)
    #    Store tensors in flat [R, H] layout for optimal memory access.
    # ------------------------------------------------------------------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise product via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # reshape flat tensors → [B, N, N, H] → permute → [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # batch gemm over the hidden dimension
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N]
    out = out.view(B, H, N, N)                            # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance over the hidden dim (torch reductions)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1)                      # [B, N, N]
    var = out_fp32.pow(2).mean(dim=1) - mu.pow(2)  # [B, N, N]
    rstd = torch.rsqrt(var + eps)                  # [B, N, N]

    mu_flat = mu.reshape(R)       # [R]  FP32
    rstd_flat = rstd.reshape(R)   # [R]  FP32
    del out_fp32, mu, var

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection (Triton)
    # ------------------------------------------------------------------
    # to_out.weight has shape [dim, hidden_dim]; we need its transpose.
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
