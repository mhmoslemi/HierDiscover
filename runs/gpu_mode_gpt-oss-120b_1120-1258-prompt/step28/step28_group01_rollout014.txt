"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) producing a tensor of shape (R, 5·H)
    where R = B·N·N and H = hidden_dim.
3️⃣  Triton kernel splits the fused projection, applies the optional mask,
    computes the three sigmoid gates and writes three tensors
    `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).
5️⃣  LayerNorm over the hidden dimension (H) is performed with the
    highly‑optimised PyTorch implementation, then multiplied by `out_gate`.
6️⃣  Final linear projection (hidden → dim) is a single `torch.nn.functional.linear`
    (FP16 → FP32).  The function returns a float32 tensor.

The only Triton‑implemented portion is step 3, which fuses the mask, the
three gate sigmoids and the split of the fused projection, providing a large
speed‑up over a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16   fused projection
    left_ptr,       # [B, H, N, N]  FP16
    right_ptr,      # [B, H, N, N]  FP16
    out_gate_ptr,   # [B, H, N, N]  FP16
    mask_ptr,       # [R] FP16   (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Broadcast mask (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices
    # Base offset for (b, i, k) in layout [B, H, N, N] (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Process hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection (5·H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five slices
        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # already sigmoid‑ed

        # Destination offsets for the flat view of [B, H, N, N]
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim], keep FP32 for statistics, cast to half afterwards
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                         # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # [5·H, dim]
    # (R, 5·H) FP16
    proj = F.linear(x_norm, fused_w)
    del fused_w, x_norm  # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()   # [R]
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    # out: [B·H, N, N]   (left ⨉ rightᵀ)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                            # [B, H, N, N]  FP16
    del left_mat, right_mat                               # free

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension + gating
    # ------------------------------------------------------------------
    # Move hidden dimension to the last axis for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H] FP16
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                                   # keep FP16

    # Apply out_gate (also need to permute)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    gated = out_norm * out_gate_perm                             # FP16
    del out_norm, out_gate, out_gate_perm, out                # free

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (hidden → dim)
    # ------------------------------------------------------------------
    # to_out.weight has shape [dim, hidden_dim]; use it directly in F.linear
    # Input shape: [..., hidden_dim]  → output shape [..., dim]
    final = F.linear(gated, weights["to_out.weight"].to(torch.float16))
    final = final.float()                                   # cast to FP32 as required
    return final
