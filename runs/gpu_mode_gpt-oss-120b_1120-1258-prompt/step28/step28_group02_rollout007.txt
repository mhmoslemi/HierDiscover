"""
Optimized outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣  Input LayerNorm  (FP32 → FP16) using PyTorch (`torch.nn.functional.layer_norm`),
    which is highly tuned on NVIDIA GPUs.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single `torch.nn.functional.linear` call.
3️⃣  Triton kernel `split_fuse_gate_kernel` extracts the three tensors we need:
    * left   – multiplied by mask and the *left* sigmoid gate,
    * right  – multiplied by mask and the *right* sigmoid gate,
    * out_gate – only the sigmoid gate (no mask).
   All tensors are stored in FP16 layout ``[B, H, N, N]``.
4️⃣  Batched GEMM (`torch.bmm`) computes the core contraction
    ``out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]``   →   ``[B, H, N, N]``.
5️⃣  Second LayerNorm over the hidden dim ``H`` is performed with
    `torch.nn.functional.layer_norm` (FP16 → FP16).  The normalized tensor
    is multiplied *in‑place* by the previously computed ``out_gate``.
6️⃣  Final linear projection (hidden → model dimension) is a single
    `torch.nn.functional.linear` on the flattened ``[B·N·N, H]`` tensor.
    The result is cast back to FP32 with shape ``[B, N, N, dim]``.

Only the split‑&‑gate kernel is custom Triton code; all heavy‑weight
matrix‑multiply / layer‑norm steps use the highly‑optimised CuBLAS/PyTorch
implementations.  This dramatically reduces the total number of kernel
launches and memory traffic while keeping the semantics identical to the
reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – split the fused projection, apply optional pair‑wise mask
# and the three sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] fused projection (FP16)
    left_ptr,      # [B, H, N, N] → FP16
    right_ptr,     # [B, H, N, N] → FP16
    out_gate_ptr,  # [B, H, N, N] → FP16
    mask_ptr,      # [R] (optional) FP16 0/1 mask
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (float16 0/1)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N          # stride between consecutive hidden planes
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden‑plane 0

    # Process hidden dim in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets into the fused projection matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32, cast back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate    # already gated

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all stored in float32 as in a regular nn.Module)
        - config_dict must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)
    Returns
    -------
    torch.Tensor: shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # flatten over the pair‑wise dimension
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # Use the highly‑optimised torch implementation (FP32 input, FP16 output)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # [B, N, N, dim]  FP16
    x_norm_flat = x_norm.view(R, dim).contiguous()   # [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim outputs)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm_flat, fused_w)      # [R, 5·H]  FP16
    del fused_w, x_norm, x_norm_flat          # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (FP16 matmul)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim H, then apply out_gate
    # --------------------------------------------------------------
    # permute to [B, N, N, H] → LayerNorm works on the last dim (H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # FP16

    # multiply by out_gate (also need to permute)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    out_norm = out_norm * out_gate_perm                         # in‑place style; still FP16
    del out, out_gate, out_gate_perm, out_norm  # we will keep `out_norm` for projection

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection: hidden → model dim
    # --------------------------------------------------------------
    # flatten the first three dims → [B·N·N, H]
    out_flat = out_norm.view(B * N * N, H)                # FP16
    # weight for projection is stored as (dim, hidden); we keep FP16 for speed
    w_out = w["to_out.weight"].to(torch.float16)          # [dim, H]
    out_fp16 = F.linear(out_flat, w_out)                  # [B·N·N, dim] FP16

    # reshape to final layout and cast to FP32
    final = out_fp16.view(B, N, N, dim).to(torch.float32)
    return final
