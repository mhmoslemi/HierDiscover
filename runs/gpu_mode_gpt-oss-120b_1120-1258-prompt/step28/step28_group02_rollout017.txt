    # 2️⃣ Fused projection
    fused_w = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"],
                         weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).to(torch.float16)  # (5*H, dim)
    # flatten input to (R, dim)
    x_flat = x_norm.reshape(R, dim)
    proj = F.linear(x_flat, fused_w)   # (R, 5*H) FP16
    del fused_w, x_norm, x_flat

    # 3️⃣ Split and apply mask and gates (torch ops)
    # reshape to (R, 5, H)
    proj = proj.view(R, 5, H)
    left_proj = proj[:,0,:]      # (R, H)
    right_proj = proj[:,1,:]     # (R, H)
    left_gate = torch.sigmoid(proj[:,2,:])   # (R, H)
    right_gate = torch.sigmoid(proj[:,3,:])  # (R, H)
    out_gate = torch.sigmoid(proj[:,4,:])    # (R, H)
    del proj

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    out_gate = out_gate  # already sigmoid

    # reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # 4️⃣ Batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # free left/right to save memory
    del left, right

    # 5️⃣ Compute per‑row mean & rsqrt‑var (torch reduction)
    mu = out.mean(dim=1)  # shape [B, N, N]
    var = out.var(dim=1, unbiased=False)  # shape [B, N, N]
    rstd = torch.rsqrt(var + eps)  # shape [B, N, N]

    mu_flat = mu.reshape(R)
    rstd_flat = rstd.reshape(R)

    # 6️⃣ Final projection + LN + out_gate (Triton)
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 64   # increased tile size for hidden dim
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],

        H,
        N,

        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

The forward pass is realized as

1️⃣ LayerNorm on the last dimension (torch.nn.functional.layer_norm, FP32→FP16).
2️⃣ One fused linear projection that produces the five hidden‑dim tensors
   (left, right, left_gate, right_gate, out_gate) in FP16.
3️⃣ Simple PyTorch ops to apply the (optional) pair‑wise mask and the three sigmoid gates,
   then reshape the tensors to [B, H, N, N].
4️⃣ Batched GEMM (torch.bmm) computes the core einsum
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   →   [B, H, N, N] (FP16).
5️⃣ Per‑row mean / variance of `out` are obtained with torch.mean / torch.var
   (still on‑GPU, FP32).
6️⃣ A Triton kernel fuses the second LayerNorm, the element‑wise `out_gate`,
   and the final linear projection (to_out) while reading the intermediate
   `out` only once.  The kernel works on a flattened view of the
   (b,i,j) triples and accumulates a mat‑mul with the weight matrix
   `to_out.weight.T` (H×dim) using TL‑dot.

All heavy work stays on the GPU; the final result is a float32 tensor
[B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection (single pass)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]    FP16
    gate_ptr,       # [B, H, N, N]    FP16
    mu_ptr,         # [R]             FP32  (row‑wise mean)
    rstd_ptr,       # [R]             FP32  (row‑wise rsqrt‑var)
    ln_w_ptr,       # [H]             FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H]             FP32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]        FP16  (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim]  FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (tiling over R)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j    # offset of hidden‑plane 0

    # per‑row µ and rsqrt‑var
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden slice of the contracted tensor and the gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - µ)·rstd → γ· + β → * gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]           # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]              # [M,K]
        gated  = norm * vals_gate                                   # [M,K]

        # Cast back to FP16 for the dot (Tensor‑core friendly)
        gated_fp16 = gated.to(tl.float16)

        # Load a tile of the final linear weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)  (FP16·FP16 → FP32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # Store final tensor (flattened row index → [B,N,N,dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # flattened pair‑wise dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (torch, FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.layer_norm is highly tuned (uses cuDNN); we keep the result in FP16.
    x_norm = F.layer_norm(x,
                          (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).to(torch.float16)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing the five hidden tensors
    # ------------------------------------------------------------------
    # concat the five weight matrices → shape (5·H, dim)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # -> FP16

    # flatten the input for a single GEMM
    x_flat = x_norm.view(R, dim)                       # [R, dim] FP16
    proj   = F.linear(x_flat, fused_w)                  # [R, 5·H] FP16
    del fused_w, x_norm, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask (if any) and apply sigmoid gates
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                           # [R,5,H]

    left_proj  = proj[:, 0, :]                          # [R, H]
    right_proj = proj[:, 1, :]                          # [R, H]
    left_gate  = torch.sigmoid(proj[:, 2, :])           # [R, H]
    right_gate = torch.sigmoid(proj[:, 3, :])           # [R, H]
    out_gate   = torch.sigmoid(proj[:, 4, :])           # [R, H]
    del proj

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R, 1)   # [R,1]
        left_proj  = left_proj  * mask_flat
        right_proj = right_proj * mask_flat

    left  = left_proj  * left_gate
    right = right_proj * right_gate
    # out_gate already holds sigmoid; no mask applied to it

    # reshape to [B, H, N, N]
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication (core einsum)
    # ------------------------------------------------------------------
    left_mat  = left .view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                 # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                          # [B, H, N, N] FP16

    # free memory that is no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & variance of `out` (torch reductions)
    # ------------------------------------------------------------------
    mu   = out.mean(dim=1)                 # [B,N,N] FP32
    var  = out.var(dim=1, unbiased=False) # [B,N,N] FP32
    rstd = torch.rsqrt(var + eps)          # [B,N,N] FP32
    mu_flat   = mu   .reshape(R)           # [R]
    rstd_flat = rstd .reshape(R)           # [R]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final linear projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128               # rows per program (tiling over R)
    BLOCK_K_OUT = 64                # hidden‑dim tile (H ≤ 128)
    BLOCK_N_OUT = 128               # output‑dim tile (dim ≤ 384)

    grid = (
        triton.cdiv(R,   BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
