"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣ Pairwise LayerNorm on the input tensor (torch.nn.functional.layer_norm, FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
   single GEMM (FP16) → shape (R, 5·H) where R = B·N·N.
3️⃣ Triton kernel “split_fuse_gate_kernel”:
   * splits the fused projection into left / right tensors (shape [B,H,N,N]),
   * applies the optional pair‑mask,
   * applies three sigmoid gates (left‑gate, right‑gate, out‑gate).
4️⃣ Batched GEMM (cuBLAS) over the hidden dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →   (B,H,N,N)  (FP16).
5️⃣ Second LayerNorm (torch.nn.functional.layer_norm) on the contracted tensor,
   multiplication by the already‑sigmoided “out_gate”.
6️⃣ Final linear projection (FP16 GEMM) → output shape [B,N,N,dim] (float32).

Only the splitting/gating step is written in Triton; the remaining heavy
operations use highly‑optimised PyTorch / cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    One program processes a tile of rows (size BLOCK_R) and a tile of hidden
    channels (size BLOCK_H).  It extracts the five fused linear maps, applies an
    optional pair‑mask and the three sigmoid gates, and writes the three tensors
    needed by the rest of the pipeline.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N                         # stride for one hidden‑plane
    # base offset of the element (b, 0, i, k) inside left/right/out_gate
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside the fused projection: 5 blocks of size H each
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        # sigmoid in FP32 → back to FP16 (numerically stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination offsets for the three outputs
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                  mask=mask_r[:, None] & mask_h[None, :])

# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm  (torch, FP32 → FP16)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # shape [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # (5·H, dim)

    # flatten to [R, dim] for the GEMM
    x_flat = x_norm.view(R, dim).contiguous()  # [R, dim]  FP16
    proj = F.linear(x_flat, fused_w)           # (R, 5·H)  FP16
    del fused_w, x_flat, x_norm                # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, and apply sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask: [B, N, N] → flatten → FP16 (0.0 or 1.0)
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128   # rows per program (tuneable)
    BLOCK_H = 64    # hidden‑dim tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # shape for cuBLAS: (B·H, N, N)
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = contracted.view(B, H, N, N)                     # [B, H, N, N] FP16
    del left_mat, right_mat, contracted, left, right     # free memory

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate multiplication (torch)
    # ------------------------------------------------------------------
    R_total = R                                            # reuse variable name
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R_total, H)       # (R, H)
    gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R_total, H)  # (R, H)

    # LayerNorm over the hidden dimension
    out_ln = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the already‑sigmoided gate
    gated = out_ln * gate_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (FP16 GEMM) → float32 output
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    final_fp16 = torch.matmul(gated, w_out_T)                         # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)           # (B,N,N,dim) FP32

    return final
```
