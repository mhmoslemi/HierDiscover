    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_ptr_final,
    R, H, dim,
    eps: tl.constexpr,

):

    pid_n = tl.program_id(1)  # output dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_n = offs_n < dim

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # ------------------------------------------------------------------
    # First pass: compute mean and rsqrt‑variance per row (over hidden)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, hidden, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < hidden

        # load a tile of out (FP16 → FP32)
        out_tile = tl.load(
            out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [M, K]

        sum_val += tl.sum(out_tile, axis=1)
        sum_sq  += tl.sum(out_tile * out_tile, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                           # [M]

    # ------------------------------------------------------------------
    # Second pass: normalize, apply out_gate, and project to `dim`
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, hidden, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # load out and gate tiles (FP16 → FP32)
        out_tile = tl.load(
            out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [M, K]

        gate_tile = tl.load(
            gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [M, K]

        # LayerNorm parameters (γ, β)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # LN: (out - μ) * rstd → *γ + β
        norm = ((out_tile - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # Apply out_gate
        term = norm * gate_tile                                        # [M, K] (FP32)

        # Cast to FP16 for the mat‑vec product
        term_fp16 = term.to(tl.float16)

        # Load a tile of the final weight matrix (H×dim) – transposed already
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [K, N_tile] FP16

        # Accumulate (M,K)·(K,N) → (M,N)   (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Store final output (flattened view of [B, N, N, dim])
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N] (bool/float) or None
        - weights_dict : mapping containing the model weights:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use PyTorch implementation
    # ------------------------------------------------------------------
    # Output is float16 to keep memory traffic low.
    x_norm_fp32 = F.layer_norm(x,
                               normalized_shape=(dim,),
                               weight=w["norm.weight"],
                               bias=w["norm.bias"],
                               eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # (5·H, dim)

    x_flat = x_norm.view(R, dim)                           # (R, dim)
    proj = F.linear(x_flat, fused_w)                       # (R, 5·H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (pure PyTorch)
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, hidden_dim)                     # (R, 5, H)
    left      = proj[:, 0, :]                             # (R, H)
    right     = proj[:, 1, :]                             # (R, H)
    left_gate = proj[:, 2, :]                             # (R, H)
    right_gate= proj[:, 3, :]                             # (R, H)
    out_gate  = proj[:, 4, :]                             # (R, H)

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)        # (R,)
        left  = left  * mask_flat[:, None]
        right = right * mask_flat[:, None]

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (equivalent to the einsum)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the bmm
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    left_mat  = left.view(B * hidden_dim, N, N)            # (B·H, N, N)
    right_mat = right.view(B * hidden_dim, N, N)           # (B·H, N, N)

    # core contraction: out = left @ rightᵀ   (all FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out_flat = out.view(R, hidden_dim)                     # (R, H)

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + linear projection (single Triton kernel)
    # ------------------------------------------------------------------
    # reshape out_gate similarly
    out_gate = out_gate.view(B, N, N, hidden_dim).permute(0, 2, 3, 1).contiguous()
    out_gate_flat = out_gate.view(R, hidden_dim)           # (R, H)

    # transpose to_out weight so that we can access it as [H, dim]
    w_out_T = w["to_out.weight"].t().to(torch.float16).contiguous()  # (H, dim)

    final_flat = torch.empty(R, dim, dtype=torch.float32, device=device)

    # kernel launch configuration
    BLOCK_M = 128                     # rows per program (tuneable)
    BLOCK_K = 32                      # hidden‑dim tiles (H ≤ 128)
    BLOCK_N = 128                     # output‑dim tiles (dim ≤ 384)

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(dim, BLOCK_N))

    final_output_kernel[grid](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],        # γ (FP32)
        w["to_out_norm.bias"],          # β (FP32)
        w_out_T,
        final_flat,
        R,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # Reshape to the original 4‑D layout and return
    # ------------------------------------------------------------------
    final = final_flat.view(B, N, N, dim)                # (B, N, N, dim)
    return final
