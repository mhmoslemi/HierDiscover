"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm over the last dimension of the input tensor (float‑32) using
    ``torch.nn.functional.layer_norm`` and cast the result to FP16.

2️⃣  Fuse the five linear maps
       left_proj, right_proj, left_gate, right_gate, out_gate
    into a single GEMM (cuBLAS) producing a tensor of shape (R, 5·H) where
       R = B·N·N   (flattened pairwise positions)
       H = hidden_dim.

3️⃣  Triton kernel (``split_fuse_gate_kernel``) splits the fused projection,
    applies the optional mask, evaluates the sigmoid gates and writes three
    tensors ``left``, ``right`` and ``out_gate`` in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core contraction
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]            →  FP16.

5️⃣  Second Layer‑Norm (over the hidden dimension) is performed with
    ``torch.nn.functional.layer_norm`` on the tensor of shape [B,N,N,H] (FP16).

6️⃣  Multiply the normalized tensor by ``out_gate`` (in‑place).

7️⃣  Final linear projection (hidden → model dimension) is a single GEMM
    ``torch.nn.functional.linear`` (FP16 → FP32).

Only step 3 uses a custom Triton kernel – the rest relies on highly‑optimised
PyTorch/cuBLAS kernels, giving a fast end‑to‑end execution while keeping the
implementation easy to understand.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional mask (broadcasted over the hidden tile)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N               # distance between successive hidden slices

    # Base offset for the (b,i,k) entry in the flattened [B,H,N,N] layout (h==0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Iterate over the hidden dimension in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection for each of the five heads
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for stability, then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left      = left_raw  * mask_val * left_gate
        right     = right_raw * mask_val * right_gate
        out_gate  = out_gate                                            # no mask or extra scaling

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim]  (float32)
    # torch's layer_norm works on the last dimension; we keep the FP32
    # stats (weight/bias are FP32) and only cast the result to FP16.
    x_norm = F.layer_norm(x_flat, (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).to(torch.float16)        # [R, dim]  (FP16)

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                       # (R, 5·H)  (FP16)
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # flatten mask to match the (b,i,k) ordering used in the kernel
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) → [batch, rows, cols]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # core contraction: out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (FP16)
    out = out.view(B, H, N, N)                             # [B, H, N, N] (FP16)

    # free memory we no longer need
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension) + out_gate
    # --------------------------------------------------------------
    # bring hidden dimension to the last axis → [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()             # [B, N, N, H] (FP16)
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] (FP16)

    # LayerNorm on the hidden channel; we cast the weight/bias to FP16
    # (tiny numerical difference is acceptable for the forward‑only test)
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)                             # (FP16)

    # multiply by the out‑gate (in‑place to save one allocation)
    out.mul_(out_gate)                                      # (FP16)
    del out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection (hidden → model dim)
    # --------------------------------------------------------------
    out_flat = out.view(R, H)                               # [R, H] (FP16)
    w_out = w["to_out.weight"].to(torch.float16)            # [dim, H]
    final = F.linear(out_flat, w_out)                       # [R, dim] (FP16)
    final = final.view(B, N, N, dim).to(torch.float32)      # back to FP32

    return final
