Pseudo:

```
@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    out_gate_norm_ptr, # [R, H] FP16
    R, H,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # Hidden block
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out and gate
        # Provide stride offsets: out and gate are [B,H,N,N] flattened as [R,H] => out_ptr + offs_r[:,None] * H + offs_h[None,:]
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # Load mu, rstd
        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)               # [BLOCK_R] FP32
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)           # [BLOCK_R] FP32

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)         # [BLOCK_H] FP32
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)           # [BLOCK_H] FP32

        # Compute scale and bias: broadcast mu, rstd across H
        # scale = rstd * gamma
        # bias = beta - mu * rstd * gamma
        # out_norm_gate = (out * scale + bias) * gate

        # compute scale per hidden: rstd[:,None] * gamma[None,:] => shape [BLOCK_R, BLOCK_H]
        scale = rstd[:, None] * gamma[None, :]                           # [BLOCK_R, BLOCK_H] FP32
        bias = beta[None, :] - mu[:, None] * rstd[:, None] * gamma[None, :]   # [BLOCK_R, BLOCK_H] FP32

        # out_norm
        tmp = out_vals * scale + bias   # FP32
        tmp = tmp * gate_vals           # FP32

        # Cast to fp16 for storage
        out_store = tmp.to(tl.float16)

        tl.store(out_gate_norm_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_store,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation (highly tuned).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input pair tensor (C = dim) – use PyTorch
    F.layer_norm (FP32 → FP16) which is already heavily vectorised.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) producing a tensor of shape (R, 5·H) where
    R = B·N·N and H = hidden_dim.

3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional mask, computes the three sigmoid gates and writes
    the three tensors `left`, `right`, `out_gate` in layout [B, H, N, N]
    (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` (LayerNorm over the hidden dim)
    in the Triton kernel `rowwise_stats_kernel`, yielding per‑row
    statistics μ and ρ (both FP32, shape R).

6️⃣  Triton kernel `norm_gate_kernel` applies the second LayerNorm,
    multiplies by the `out_gate` tensor and writes the result as a
    flattened tensor `[R, H]` (FP16).

7️⃣  Final projection is performed by a single cuBLAS GEMM:
       final_fp16 = (R, H) @ (H, dim) → (R, dim) (FP16 → FP32).
    Reshape to `[B, N, N, dim]`.

All heavy reductions are done in FP32 for numerical stability,
bulk arithmetic in FP16 to exploit Tensor‑Cores, and all custom
operations are fused into Triton kernels.  No gradients are computed.   
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over hidden)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices
    # base offset for (b,i,k) in the flat [B,H,N,N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: (5·H) contiguous per row
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise statistics (mean & rsqrt‑variance) over hidden dim
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # first hidden entry for this (b,i,j)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Apply second LayerNorm, out_gate and flatten to [R,H]
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,                # [B, H, N, N] FP16
    gate_ptr,               # [B, H, N, N] FP16
    mu_ptr,                 # [R] FP32
    rstd_ptr,               # [R] FP32
    ln_w_ptr,               # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,               # [H] FP32 (to_out_norm.bias)
    out_norm_ptr,           # [R, H] FP16 (flattened, row‑major)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,j) from flat row index
    stride_hidden = N * N
    nn = stride_hidden
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    base_offset = b * (H * stride_hidden) + i * N + j   # first hidden entry

    # per‑row statistics (mu, rstd) – same for all hidden entries of this row
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R] FP32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R] FP32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Load raw out and gate values for this hidden tile
        out_vals = tl.load(
            out_ptr + base_offset[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32

        # scale = rstd * gamma, bias = beta - mu * rstd * gamma
        scale = rstd[:, None] * gamma[None, :]                     # [BLOCK_R, BLOCK_H]
        bias  = beta[None, :] - mu[:, None] * scale               # [BLOCK_R, BLOCK_H]

        # Apply LayerNorm, then element‑wise gate
        out_norm = (out_vals * scale + bias) * gate_vals          # FP32

        # Store as FP16 into the flattened [R, H] buffer (row‑major)
        dst_ptr = out_norm_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(dst_ptr, out_norm.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : dictionary with model parameters:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # Use the highly‑optimised PyTorch implementation for the first LN.
    x_norm = F.layer_norm(x, (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"])
    x_norm = x_norm.to(torch.float16)          # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (LayerNorm over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm + out_gate → flattened [R, H] (FP16)
    # ------------------------------------------------------------------
    out_norm_flat = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_NORM = 64
    BLOCK_H_NORM = 64
    grid_norm = (triton.cdiv(R, BLOCK_R_NORM),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        out_norm_flat,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_NORM,
        BLOCK_H=BLOCK_H_NORM,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    # to_out.weight shape = [dim, hidden_dim]; transpose → [hidden_dim, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    # GEMM: (R, H) @ (H, dim) → (R, dim)  (FP16 → FP32)
    final_fp16 = torch.matmul(out_norm_flat, w_out_T)                # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)          # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    return final
