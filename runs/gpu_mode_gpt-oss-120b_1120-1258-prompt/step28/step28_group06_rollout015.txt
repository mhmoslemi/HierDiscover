Pseudo code:

```
@triton.jit
def proj_and_gate_kernel(
    x_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    mask_ptr,
    left_proj_ptr,
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode row index to b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # maybe load mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]  # broadcast across hidden block later
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]
    # We'll write for each hidden block later.

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for each projection/gate
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # Loop over dim
        for d_off in range(0, dim, BLOCK_D):
            offs_d = d_off + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load x tile
            x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
            x_tile = tl.load(x_ptrs,
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_D]

            # Load weight tiles for each matrix
            # shape [BLOCK_H, BLOCK_D]
            left_w = tl.load(left_proj_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            right_w = tl.load(right_proj_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                                   mask=mask_h[:, None] & mask_d[None, :],
                                   other=0.0).to(tl.float32)
            out_gate_w = tl.load(out_gate_w_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                                 mask=mask_h[:, None] & mask_d[None, :],
                                 other=0.0).to(tl.float32)

            # dot product accumulate
            left_acc += tl.dot(x_tile, left_w)
            right_acc += tl.dot(x_tile, right_w)
            left_gate_acc += tl.dot(x_tile, left_gate_w)
            right_gate_acc += tl.dot(x_tile, right_gate_w)
            out_gate_acc += tl.dot(x_tile, out_gate_w)

        # After dim loop: apply sigmoid to gates:
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # apply mask and gates to left/right
        # left_raw and right_raw are in float32, but we want to store as fp16.
        left_raw = left_acc.to(tl.float16)
        right_raw = right_acc.to(tl.float16)

        left_val = left_raw * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # For out_gate, no mask
        out_gate_val = out_gate

        # Compute destination offsets
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset, left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])
```

        BLOCK_D=BLOCK_D_ln,
        num_warps=8,
    )
    del x_flat

    # transpose weight matrices for projection
    left_proj_T = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    right_proj_T = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_T = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_T = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_T = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    # allocate output tensors
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # launch projection+gate kernel
    BLOCK_R = 32
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_and_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_proj_T,
        right_proj_T,
        left_gate_T,
        right_gate_T,
        out_gate_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr

    # core einsum via batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # rowwise stats
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)
    BLOCK_M_stats = 256
    BLOCK_K_stats = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_stats),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_stats,
        BLOCK_K=BLOCK_K_stats,
        num_warps=4,
    )
    # final projection
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)
    BLOCK_M_out = 128
    BLOCK_K_out = 32
    BLOCK_N_out = 128
    grid_out = (triton.cdiv(R, BLOCK_M_out), triton.cdiv(dim, BLOCK_N_out))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_out,
        BLOCK_K=BLOCK_K_out,
        BLOCK_N=BLOCK_N_out,
        num_warps=8,
    )

    return final

"""
Outgoing TriMul (AlphaFold‑3 style) – forward only, Triton‑accelerated

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – triton kernel.
2️⃣ Fused projection + gating:
   for every pair token (b,i,k) compute the five linear maps
   (left_proj, right_proj, left_gate, right_gate, out_gate) directly
   and apply the optional mask and sigmoid gates.  The result is written to
   three tensors `left`, `right`, `out_gate` of shape [B, H, N, N] (FP16).
   This replaces the separate `torch.nn.functional.linear` + split kernel
   and saves ~⅓ of the memory traffic.
3️⃣ Core einsum:   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   performed as a batched GEMM with `torch.bmm` (FP16, tensor‑cores).
4️⃣ Row‑wise (over H) mean & rsqrt‑variance of `out` – triton kernel.
5️⃣ Final LayerNorm + out‑gate + projection to `dim` – one fused triton kernel.
   The kernel reads `out`, `out_gate`, the pre‑computed mean/rstd, the
   to_out_norm parameters and the final projection weight (transposed) and
   writes the result `[B,N,N,dim]` in FP32.

All kernels use mixed precision (FP16 compute, FP32 reductions) and are
tuned for H100 (Triton 3.3.1).  The entry‑point `custom_kernel` follows the
required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, D]  FP32 input
    y_ptr,               # [R, D]  FP16 output
    w_ptr,               # [D]    γ        FP32
    b_ptr,               # [D]    β        FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating (no intermediate “proj” tensor)
# ----------------------------------------------------------------------
@triton.jit
def proj_and_gate_kernel(
    x_ptr,               # [R, D]                FP16 (layer‑normed input)
    left_ptr,            # [B, H, N, N]          FP16
    right_ptr,           # [B, H, N, N]          FP16
    out_gate_ptr,        # [B, H, N, N]          FP16
    mask_ptr,            # [R] (optional)        FP16
    left_proj_ptr,       # [D, H]                FP16 (transposed weight)
    right_proj_ptr,      # [D, H]                FP16
    left_gate_ptr,       # [D, H]                FP16
    right_gate_ptr,      # [D, H]                FP16
    out_gate_w_ptr,      # [D, H]                FP16
    R, H, N, D,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R] FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                         # stride of one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # ------------------------------------------------------------------
    # Process hidden dimension in chunks
    # ------------------------------------------------------------------
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators – use FP32 for numerical safety
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # iterate over the feature dimension D
        for d_off in range(0, D, BLOCK_D):
            offs_d = d_off + tl.arange(0, BLOCK_D)            # [BLOCK_D]
            mask_d = offs_d < D

            # ---- load X ------------------------------------------------
            x_ptrs = x_ptr + (offs_r[:, None] * D + offs_d[None, :])
            x_tile = tl.load(
                x_ptrs,
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

            # ---- load weight tiles (already transposed, shape [D, H]) ---
            # offset = d * H + h  because stride0 = H, stride1 = 1
            w_off = offs_d[:, None] * H + offs_h[None, :]

            left_w = tl.load(left_proj_ptr + w_off,
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)        # [BLOCK_D, BLOCK_H]
            right_w = tl.load(right_proj_ptr + w_off,
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + w_off,
                                  mask=mask_d[:, None] & mask_h[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_ptr + w_off,
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            out_gate_w = tl.load(out_gate_w_ptr + w_off,
                                 mask=mask_d[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)

            # ---- dot products -----------------------------------------
            left_acc      += tl.dot(x_tile, left_w)          # [BLOCK_R, BLOCK_H]
            right_acc     += tl.dot(x_tile, right_w)
            left_gate_acc += tl.dot(x_tile, left_gate_w)
            right_gate_acc+= tl.dot(x_tile, right_gate_w)
            out_gate_acc  += tl.dot(x_tile, out_gate_w)

        # ---- apply sigmoid to gate values (FP32 → FP16) -------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # ---- apply mask and write results ---------------------------
        left_raw  = left_acc.to(tl.float16)
        right_raw = right_acc.to(tl.float16)

        left_val  = left_raw * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate
        out_gate_val = out_gate                       # no mask

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B,H,N,N] FP16
    gate_ptr,       # [B,H,N,N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B,N,N,dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden‑plane values
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm over hidden dim H
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (vals_out - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]                # LN

        gated = norm * vals_gate                                    # apply out_gate

        # projection to output dim
        gated_fp16 = gated.to(tl.float16)

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 style).
    Args:
        data (tuple):
            - input : torch.Tensor [B, N, N, dim]      (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights : dict containing model weights
            - config : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments (support both 3‑ and 4‑element signatures)
    # --------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, cfg = data
    elif len(data) == 3:
        x, weights, cfg = data
        mask = None
    else:
        raise ValueError("expected a tuple of length 3 or 4")

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_ln = 256
    BLOCK_D_ln = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_ln),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_ln,
        BLOCK_D=BLOCK_D_ln,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps + optional mask + sigmoid gates
    # --------------------------------------------------------------
    # Transpose weights so that they have shape [dim, H] (better for tl.dot)
    left_proj_T  = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    right_proj_T = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_T  = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_T = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_T   = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 32
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_and_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_proj_T,
        right_proj_T,
        left_gate_T,
        right_gate_T,
        out_gate_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Core einsum via batched GEMM (FP16, tensor cores)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_stats = 256
    BLOCK_K_stats = 64   # H ≤ 128 → two tiles are enough
    grid_stats = (triton.cdiv(R, BLOCK_M_stats),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_stats,
        BLOCK_K=BLOCK_K_stats,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + projection to dim
    # --------------------------------------------------------------
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim] FP16
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_out = 128    # rows per program
    BLOCK_K_out = 32     # hidden‑dim chunk
    BLOCK_N_out = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_out),
        triton.cdiv(dim, BLOCK_N_out),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ  (FP32)
        weights["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_out,
        BLOCK_K=BLOCK_K_out,
        BLOCK_N=BLOCK_N_out,
        num_warps=8,
    )

    return final
```
