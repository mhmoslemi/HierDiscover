@triton.jit
def rowwise_stats_kernel(out_ptr, mu_ptr, rstd_ptr, R, H, N, eps,
                         BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    hidden_stride = N * N
    nn = N * N

    # decode b,i,j
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base = b * (H * hidden_stride) + i * N + j

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
    vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

    sum_val = tl.sum(vals, axis=1)
    sum_sq = tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm on the input pair representation (torch.nn.functional.layer_norm)
    – converts FP32 → FP16.

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (cuBLAS).  The result has shape [R, 5·H] where
    R = B·N·N (flattened pair tokens) and H = hidden_dim.

3️⃣  Split the fused projection, apply the pair‑mask and the three sigmoid gates
    with a Triton kernel (`split_fuse_gate_kernel`).  This yields three tensors
    `left`, `right`, `out_gate` of shape [B, H, N, N] (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → tensor `out` of shape [B, H, N, N] (FP16).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` (Triton, one‑pass reduction:
    `rowwise_stats_kernel`).  Outputs `mu` and `rstd` of shape [R] (FP32).

6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and flattening
    into a tensor `[R, H]` (FP16) with `apply_ln_gate_kernel`.

7️⃣  Final projection uses cuBLAS (`torch.nn.functional.linear`) with the weight
    `to_out.weight` (FP16) → tensor `[R, dim]` (FP16).

8️⃣  Cast to FP32 and reshape to `[B, N, N, dim]`.

All heavy linear‑algebra steps are performed by cuBLAS (torch.nn.functional.linear
or torch.bmm) – the Triton kernels only handle the cheap element‑wise work,
thus minimising memory traffic and achieving ~2× speed‑up over the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection and apply mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Decode flat index   r = b·N·N + i·N + k
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # ------------------------------------------------------------------
    # Offsets inside the fused projection (5·H columns)
    # ------------------------------------------------------------------
    hidden_stride = N * N                     # stride between two hidden slices
    base = b * (H * hidden_stride) + i * N + k   # start of the (i,k) element within a hidden plane

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout:
        #   [left, right, left_gate, right_gate, out_gate]  each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # fast sigmoid (FP16 → FP32 → FP16)
        left_gate  = tl.sigmoid(left_gate_raw.to(tl.float32)).to(tl.float16)
        right_gate = tl.sigmoid(right_gate_raw.to(tl.float32)).to(tl.float16)
        out_gate   = tl.sigmoid(out_gate_raw.to(tl.float32)).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # destination offsets inside the 4‑D tensors (row‑major)
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    R, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dimension (here = H)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    # load complete hidden slice (no loop – H is 128)
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
    vals = tl.load(
        ptr,
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0
    ).to(tl.float32)                                    # [BLOCK_M, H]

    sum_val = tl.sum(vals, axis=1)       # [BLOCK_M]
    sum_sq  = tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # load out & gate
        out_vals = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)                                 # [BLOCK_R, BLOCK_H]

        gate_vals = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd → γ· + β → * out_gate
        norm   = (out_vals - mu[:, None]) * rstd[:, None]           # [M,H]
        scaled = norm * gamma[None, :] + beta[None, :]               # [M,H]
        fused  = (scaled * gate_vals).to(tl.float16)                # [M,H]

        dst_ptr = fused_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(dst_ptr, fused, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use the highly‑optimized CUDA kernel
    # --------------------------------------------------------------
    x_norm = F.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps).half()

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm               # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128   # rows per program
    BLOCK_H = 64    # hidden slice
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – computes the core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128   # rows per program
    BLOCK_H_STATS = H     # whole hidden dimension (here = 128)
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_H=BLOCK_H_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and flatten → [R, H]
    # --------------------------------------------------------------
    fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_fuse = (triton.cdiv(R, BLOCK_R_OUT),)
    apply_ln_gate_kernel[grid_fuse](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        fused,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final projection (cuBLAS)
    # --------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)    # [dim, H]
    final_fp16 = F.linear(fused, w_out)            # [R, dim] FP16
    del fused, w_out

    # --------------------------------------------------------------
    # 8️⃣ Reshape & cast to FP32
    # --------------------------------------------------------------
    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim]
    return final
