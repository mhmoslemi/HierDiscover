   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

       sum_sq += tl.sum(vals * vals, axis=1)

   var = sum_sq / H - mean * mean
   rstd = tl.rsqrt(var + eps)

   # Now accumulate final output
   acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

def norm_gate_kernel(...):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode row indices
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # First pass: compute sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_h = k_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        ptr = out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: apply LN and gate, store
    for k_start in range(0, H, BLOCK_K):
        offs_h = k_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # Load out
        ptr_out = out_ptr + base_offset[:, None] + offs_h[None,:] * hidden_stride
        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_h[None,:], other=0.0).to(tl.float32)
        # Load gate
        ptr_gate = gate_ptr + base_offset[:, None] + offs_h[None,:] * hidden_stride
        vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_h[None,:], other=0.0).to(tl.float32)
        # LN weight and bias
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)    # FP32

        # Normalize and gate
        diff = (vals_out - mean[:, None]) * rstd[:, None]       # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]
        gated = norm * vals_gate                                # [M,K]

        # Store in out_normed
        dst_ptr = out_normed_ptr + offs_m[:, None] * H + offs_h[None, :]  # [M,K]
        tl.store(dst_ptr, gated.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – forward only
--------------------------------
This implementation follows the AlphaFold‑3 “outgoing” TriMul but replaces the
slow Python loops and several torch calls with three highly‑optimised Triton
kernels:

1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣  Split‑fuse kernel: decodes the flat pair index (b,i,k) → (b,h,i,k) layout,
    applies the optional mask, computes the sigmoids for the four gates and
    writes the three tensors `left`, `right` and `out_gate` in layout
    `[B, H, N, N]` (FP16).  This fuses the five linear projections of the
    original module.
3️⃣  `norm_gate_kernel`: a single kernel that (a) computes per‑row mean & rsqrt‑
    variance of the contracted tensor, (b) applies the second LayerNorm
    (`to_out_norm`), (c) multiplies by the `out_gate` and (d) stores the
    normalised‑gated values into a compact `[R, H]` buffer (`R = B·N·N`).

After the kernels the remaining work consists of:
* a batched matrix multiplication (`torch.bmm`) that performs the core
  Einsum `out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`;
* a single dense linear projection (`torch.nn.functional.linear`) that maps
  the `[R, H]` tensor to the final `[R, dim]` output.  The final tensor is
  reshaped to `[B, N, N, dim]` (float32).

All kernels use mixed‑precision (FP16 compute, FP32 reductions) and are tuned
for the H100 (Triton 3.3.1).  The entry point `custom_kernel` matches the
required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D]   FP32 input
    y_ptr,            # [R, D]   FP16 output
    w_ptr,            # [D]     γ (FP32)
    b_ptr,            # [D]     β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- reduction for mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalised tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split‑fuse projection, optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R,5·H]  FP16 fused projection
    left_ptr,        # [B,H,N,N] FP16
    right_ptr,       # [B,H,N,N] FP16
    out_gate_ptr,    # [B,H,N,N] FP16
    mask_ptr,        # [R]      FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for the [B,H,N,N] tensors (h = 0)
    hidden_stride = N * N                                   # stride of one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k      # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # out_gate is *not* masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise stats + LN + out‑gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,         # [B,H,N,N] FP16 (contracted tensor)
    gate_ptr,        # [B,H,N,N] FP16 (out_gate)
    ln_w_ptr,        # [H]      FP32 (to_out_norm.weight)
    ln_b_ptr,        # [H]      FP32 (to_out_norm.bias)
    out_normed_ptr,  # [R,H]    FP16 (final normalised & gated values)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # address of hidden plane 0

    # -------------------------------------------------
    # first pass: compute mean & variance across H
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M,K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # second pass: LayerNorm, gate and store
    # -------------------------------------------------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        # load contracted values and out_gate
        ptr_out  = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M,K]
        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M,K]

        # LN parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # (out - μ) * rstd  →  LN  →  gate
        diff = (vals_out - mean[:, None]) * rstd[:, None]            # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                # LN
        gated = norm * vals_gate                                    # apply out_gate

        # write compact [R, H] buffer (row‑major)
        dst_ptr = out_normed_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.
    Args:
        data (tuple): (input, mask?, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights : dict with the model parameters (all float32, on CUDA)
            - config  : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments (support 3‑ or 4‑tuple signatures)
    # ------------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, cfg = data
    else:
        x, weights, cfg = data
        mask = None

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                      # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_ln = 128
    BLOCK_D_ln = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_ln),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_ln,
        BLOCK_D=BLOCK_D_ln,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single matmul (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R,5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        # dummy tensor – never dereferenced when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 128
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (core einsum)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean/var, to_out_norm + out_gate (fused)
    # ------------------------------------------------------------------
    out_normed = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_norm = 256
    BLOCK_K_norm = 32
    grid_norm = (triton.cdiv(R, BLOCK_M_norm),)
    norm_gate_kernel[grid_norm](

        weights["to_out_norm.bias"],     # β  (FP32)
        out_normed,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_norm,
        BLOCK_K=BLOCK_K_norm,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (dense GEMM) → [R, dim] → reshape
    # ------------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16).contiguous()   # [dim, H] FP16
    final = F.linear(out_normed, w_out)       # [R, dim] (FP16 -> FP32 accumulation)
    final = final.to(torch.float32)            # ensure FP32 output
    final = final.view(B, N, N, dim)           # [B, N, N, dim]

    return final
