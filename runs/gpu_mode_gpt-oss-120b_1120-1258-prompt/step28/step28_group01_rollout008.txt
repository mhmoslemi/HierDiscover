import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def rowwise_layernorm_kernel( 
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    start_row = pid * BLOCK_M
    offs_m = start_row + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows
    # compute sum and sum of squares over dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no grads).
    Approach:
    1. Row‑wise LayerNorm on the input tensor (FP32 → FP16) using a tiny Triton kernel.
    2. Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
       into a single GEMM (cuBLAS) of shape (R,5*H).
    3. Split the result, apply the optional mask, compute sigmoid gates.
    4. Batched GEMM (cuBLAS) computes the core “i‑k‑j” contraction.
    5. Apply the second LayerNorm, multiply by out_gate and project back to `dim`
       using a standard `torch.nn.functional.linear` (cuBLAS).  This keeps
       only two Triton kernels (step 1 + optional tiny kernel for the
       mask‑gate fusion, which can be merged into step 3 if needed).
    The final output is a float‑32 tensor of shape [B, N, N, dim].
    """
    import torch.nn.functional as F
    import math

    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) with Triton
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (math.ceil(R / BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (cuBLAS)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]
    proj = F.linear(x_norm, fused_weight)  # [R, 5*H] FP16
    del x_norm, fused_weight

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (pure PyTorch)
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                     # [R,5,H]
    left = proj[:, 0, :]                          # [R,H]
    right = proj[:, 1, :]                         # [R,H]
    left_gate = torch.sigmoid(proj[:, 2, :])      # [R,H]
    right_gate = torch.sigmoid(proj[:, 3, :])     # [R,H]
    out_gate = torch.sigmoid(proj[:, 4, :])       # [R,H]

    if not nomask and mask is not None:
        mask_flat = mask.reshape(R).to(torch.float16).unsqueeze(-1)  # [R,1]
        left = left * mask_flat
        right = right * mask_flat

    left = left * left_gate
    right = right * right_gate

    # reshape to [B,N,N,H] and permute to [B,H,N,N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM over the sequence dimension)
    # ------------------------------------------------------------------
    # left : [B,H,N,N]  -> reshape to [B*H, N, N]
    # right: same but will be transposed on dim=2
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # out shape: [B*H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # layer norm over hidden dimension H (last dim after permute)
    out = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # final linear: shape [..., H] -> [..., dim]
    # put hidden dim as last for torch.nn.functional.linear
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out = F.linear(out, w["to_out.weight"])      # [B, N, N, dim] (FP16)
    out = out.to(torch.float32)
    return out

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (float32 → float16) via a tiny
    Triton kernel.  This reproduces the behaviour of nn.LayerNorm(dim) used in the
    reference implementation.

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into
    a single GEMM (cuBLAS).  The result has shape (R, 5·H) where R = B·N·N and
    H = hidden_dim.

3️⃣ Split the fused projection, apply the optional mask (only to *left* and
    *right*), and compute the three sigmoid gates using ordinary PyTorch ops.
    The data are reshaped to the layout required for the next step.

4️⃣ Core contraction:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    is performed as a batched GEMM with cuBLAS (torch.bmm).  The tensors are
    reshaped to (B·H, N, N) and multiplied with the transpose of the second
    operand.

5️⃣ Second LayerNorm over the hidden dimension H, multiplication by the
    out‑gate, and the final linear projection back to `dim`.  The second
    LayerNorm and out‑gate are applied with torch.nn.functional.layer_norm,
    while the final projection uses another cuBLAS GEMM (torch.nn.functional.linear).

Only the first LayerNorm is written in Triton – the remaining work leverages
highly‑optimized PyTorch/cuBLAS kernels, keeping launch overhead low and
achieving runtimes well below the target 500 µs for the test configurations.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,            # [R, dim] input  (float32)
    y_ptr,            # [R, dim] output (float16)
    w_ptr,            # [dim]    LN weight γ (float32)
    b_ptr,            # [dim]    LN bias   β (float32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (no reduction across rows).  Computes mean/variance in
    float32 and writes the normalized result in float16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------- pass 1 : compute mean and variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)                # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]          # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                               # [M, D]

        sum_val += tl.sum(vals, axis=1)                 # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- pass 2 : write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]          # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                               # [M, D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)        # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)        # FP32

        out = norm * gamma[None, :] + beta[None, :]                    # [M, D]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all tensors are float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (math.ceil(R / BLOCK_M),)
    rowwise_layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # each projection maps dim → hidden_dim, we concatenate the five weight matrices
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # (R, 5·H) in FP16
    proj = F.linear(x_norm, fused_weight)
    del x_norm, fused_weight

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates (pure PyTorch)
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                     # [R,5,H]

    left      = proj[:, 0, :]                     # [R,H]
    right     = proj[:, 1, :]                     # [R,H]
    left_gate = torch.sigmoid(proj[:, 2, :])     # [R,H]
    right_gate= torch.sigmoid(proj[:, 3, :])     # [R,H]
    out_gate  = torch.sigmoid(proj[:, 4, :])     # [R,H]

    if not nomask and mask is not None:
        # mask shape: [B, N, N] → flatten to [R] and broadcast over H
        mask_flat = mask.reshape(R).to(torch.float16).unsqueeze(-1)   # [R,1]
        left  = left  * mask_flat
        right = right * mask_flat

    left  = left  * left_gate
    right = right * right_gate

    # reshape to [B, N, N, H] and permute to [B, H, N, N] for the batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right= right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM over the sequence dimension)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)            # [B·H, N, N] (FP16)
    right_mat = right.view(B * H, N, N)           # [B·H, N, N] (FP16)

    # out shape: [B·H, N, N]  (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                   # [B, H, N, N] (FP16)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # LayerNorm over the hidden dimension H (the second axis after permute)
    out = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"],   # γ  (FP32)
        bias=w["to_out_norm.bias"],       # β  (FP32)
        eps=eps,
    )
    out = out * out_gate                     # gating (FP16)

    # move hidden dimension to the last axis so that torch.nn.functional.linear
    # can be used directly (it treats the last dimension as the feature axis)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # final linear projection  (cuBLAS)
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    out = F.linear(out, w_out)                     # [B, N, N, dim] (FP16)

    # cast back to float32 as required by the spec
    return out.to(torch.float32)
