@triton.jit
def ln_gate_kernel(
    out_ptr, out_gate_ptr, mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    out2_ptr,
    R, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Load mu and rstd broadcast
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)[:, None]  # [M,1]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]  # [M,1]

    # Load gamma and beta per hidden dimension
    gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :]  # [1,H]
    beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1,H]

    # Load out and gate
    out = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                  mask=mask_m[:, None] & mask_h[None, :],
                  other=0.0).to(tl.float32)
    gate = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                  mask=mask_m[:, None] & mask_h[None, :],
                  other=0.0).to(tl.float32)

    # LN: (out - mu) * rstd * gamma + beta
    norm = (out - mu) * rstd
    norm = norm * gamma + beta
    # Multiply gate
    res = norm * gate

    # store as FP16
    tl.store(out2_ptr + offs_m[:, None] * H + offs_h[None, :],
             res.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
```

@triton.jit
def apply_ln_gate_kernel(
    out_ptr,                # [R, H] FP16 permuted version (flattened)
    out_gate_ptr,           # [R, H] FP16 permuted version
    mu_ptr,                 # [R] FP32
    rstd_ptr,               # [R] FP32
    gamma_ptr,              # [H] FP32
    beta_ptr,               # [H] FP32
    out_norm_gate_ptr,      # [R, H] FP16 output
    R, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Broadcast mu, rstd across hidden dim
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)[:, None]    # [M,1]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]    # [M,1]

    # Load gamma and beta
    gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :]    # [1,H]
    beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)[None, :]      # [1,H]

    # Load out and out_gate
    base_out = out_ptr + offs_m[:, None] * H + offs_h[None, :]  # pointer
    out_val = tl.load(base_out, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
    out_gate_val = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

    # LN + gate
    norm = (out_val - mu) * rstd
    norm = norm * gamma + beta
    res = norm * out_gate_val

    # Store
    tl.store(out_norm_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
             res.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
```

def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N; H = hidden_dim

    # 1 LayerNorm
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # 2 fused projection
    fused_w = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"]
    ], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # (R,5*H)
    del x_norm, fused_w

    # 3 split/gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:

    out = out.view(B, H, N, N)  # [B, H, N, N] FP16

        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # 6 apply LN + out_gate, then final projection

    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_H_OUT = 32

        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the channel dimension (dim) → FP16.
2️⃣  Fuse five linear maps (left, right, left_gate, right_gate, out_gate) into a
    single GEMM (R × dim) × (5·H × dim)ᵀ → (R,5·H)   (R = B·N·N).
3️⃣  Split the fused projection, apply the optional mask and sigmoid gates,
    write three tensors `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → [B, H, N, N] (FP16).
5️⃣  Compute per‑pair (b,i,j) mean and rsqrt‑variance over the hidden dimension H
    (row‑wise LayerNorm statistics).
6️⃣  Apply the second LayerNorm (γ,β), multiply by `out_gate`,
    and reshape to a dense [R, H] matrix (R = B·N·N, H = hidden_dim).
7️⃣  Final projection `to_out` is a single GEMM (R × H) × (H × dim)ᵀ
    performed by torch.matmul (FP16 → FP32).  The result is reshaped to
    [B, N, N, dim] (float32).

All kernels are written in Triton 3.3.1 and run on an H100.  Only the forward
pass is required; no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim] FP32 input
    y_ptr,          # [R, dim] FP16 output
    w_ptr,          # [dim]   LN weight (γ) FP32
    b_ptr,          # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices
    base_offset = b * (H * stride_hidden) + i * N + k   # start offset for hidden=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (LN over H)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # first hidden entry for (b,i,j)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Apply second LayerNorm, out_gate and flatten to [R, H]
# ----------------------------------------------------------------------
@triton.jit
def apply_ln_gate_kernel(
    out_ptr,          # [R, H] FP16 (contiguous)
    gate_ptr,         # [R, H] FP16 (contiguous)
    mu_ptr,           # [R] FP32
    rstd_ptr,         # [R] FP32
    gamma_ptr,        # [H] FP32  (to_out_norm.weight)
    beta_ptr,         # [H] FP32  (to_out_norm.bias)
    out2_ptr,         # [R, H] FP16
    R, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # broadcast μ & rstd over hidden dim
    mu   = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None] # [M,1]

    gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :]  # [1,H]
    beta  = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1,H]

    # offsets inside the flattened [R, H] tensor
    base = offs_m[:, None] * H + offs_h[None, :]

    out_val  = tl.load(out_ptr  + base,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
    gate_val = tl.load(gate_ptr + base,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

    # LayerNorm over H, then multiply by gate
    norm = (out_val - mu) * rstd
    norm = norm * gamma + beta
    res  = norm * gate_val

    tl.store(out2_ptr + base,
             res.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Arguments
    ---------
    data : tuple
        (input_tensor   : torch.Tensor  [B, N, N, dim]  (float32)
         mask_tensor    : torch.Tensor  [B, N, N]      (bool/float) or None
         weights_dict   : dict with the model weights
         config_dict    : must contain "dim", "hidden_dim", "nomask" (bool))

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over `dim`  →  FP16
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, compute sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ  (the core einsum)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance over H  (LayerNorm statistics)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply second LayerNorm, multiply by out_gate, flatten to [R, H]
    # ------------------------------------------------------------------
    # Make hidden dimension innermost for a dense [R, H] view
    out_perm      = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_H_OUT = 32
    grid_out = (triton.cdiv(R, BLOCK_M_OUT),)
    apply_ln_gate_kernel[grid_out](
        out_perm,
        out_gate_perm,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        out_norm_gate,
        R,
        H,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out_perm, out_gate_perm, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (to_out) – one GEMM using torch.matmul
    # ------------------------------------------------------------------
    # to_out.weight has shape [dim, hidden_dim]; we need its transpose [hidden_dim, dim]
    w_out_T = w["to_out.weight"].t().to(torch.float16).contiguous()   # [H, dim]

    # (R, H) @ (H, dim) → (R, dim)  (FP16 → cast back to FP32)
    final_fp16 = torch.matmul(out_norm_gate, w_out_T)                # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)          # [B, N, N, dim]

    return final
