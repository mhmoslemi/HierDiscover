    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]      # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * D + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights
            - cfg_dict     : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 64       # channel tiles
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],          # FP32
        w["norm.bias"],            # FP32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free original FP32 tensor

    # ------------------------------------------------------------------
    # 2️⃣ Five independent linear projections (FP16)
    # ------------------------------------------------------------------
    # cast weights once to FP16
    left_w      = w["left_proj.weight"].to(torch.float16)
    right_w     = w["right_proj.weight"].to(torch.float16)
    left_gate_w = w["left_gate.weight"].to(torch.float16)
    right_gate_w= w["right_gate.weight"].to(torch.float16)
    out_gate_w  = w["out_gate.weight"].to(torch.float16)

    # proj → [R, H] (FP16)
    left      = F.linear(x_norm, left_w)          # (R, H)
    right     = F.linear(x_norm, right_w)         # (R, H)
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate= torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate  = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # optional mask (broadcast over H)
    if (mask is not None) and (not nomask):
        mask_flat = mask.reshape(R).to(torch.float16)          # [R]
        left  = left  * mask_flat[:, None]
        right = right * mask_flat[:, None]

    # apply element‑wise gates
    left  = left  * left_gate
    right = right * right_gate

    # reshape to [B, H, N, N] – make the layout contiguous for the GEMM
    left      = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right     = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate  = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del x_norm, left_gate, right_gate, out_gate_w, left_w, right_w, left_gate_w, right_gate_w

    # ------------------------------------------------------------------
    # 3️⃣ Core einsum via a single batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # shape: (B·H, N, N) * (B·H, N, N)ᵀ  →  (B·H, N, N)
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out_mat   = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16
    out = out_mat.view(B, H, N, N)                               # [B, H, N, N]

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final projection (all FP16)
    # ------------------------------------------------------------------
    # flatten spatial axes – each row corresponds to one (b,i,j) pair
    out_flat      = out.reshape(R, H)            # [R, H]
    out_gate_flat = out_gate.reshape(R, H)       # [R, H]

    # LayerNorm over hidden dimension H (cublas/cuDNN)
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(out_flat,
                            (H,),
                            weight=ln_w,
                            bias=ln_b,
                            eps=eps)            # FP16

    # gating
    gated = out_norm * out_gate_flat               # FP16

    # final linear projection to original dim
    w_out = w["to_out.weight"].to(torch.float16)   # shape (dim, H)
    final_fp16 = F.linear(gated, w_out)           # [R, dim]  FP16

    # reshape to [B, N, N, dim] and cast back to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
