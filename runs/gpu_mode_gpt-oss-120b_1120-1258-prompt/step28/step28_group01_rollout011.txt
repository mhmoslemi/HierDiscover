```
@triton.jit
def ln_gate_kernel(out_ptr, gate_ptr, out_buf_ptr, ln_w_ptr, ln_b_ptr, R, H, eps,
                   BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # Hidden dimension indices
    offs_h = tl.arange(0, H)  # H constant at compile time = 128

    # Compute per-row sum and sum_sq
    # Load out row
    out = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                  mask=mask_m[:, None], other=0.0).to(tl.float32)   # [BLOCK_M, H]
    # Now compute sum and sum_sq using tl.sum across dim=1
    sum_val = tl.sum(out, axis=1)  # [BLOCK_M]
    sum_sq = tl.sum(out * out, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Normalize
    # Load gamma and beta
    gamma = tl.load(ln_w_ptr + offs_h, mask=offs_h < H, other=0.0)
    beta = tl.load(ln_b_ptr + offs_h, mask=offs_h < H, other=0.0)
    # Broadcast
    # Need to compute (out - mean) * rstd * gamma + beta
    # Subtract mean: broadcast mean[:, None]
    norm = (out - mean[:, None]) * rstd[:, None]  # FP32
    norm = norm * gamma[None, :] + beta[None, :]   # FP32
    # Cast to fp16
    norm_fp16 = norm.to(tl.float16)

    # Load gate
    gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                  mask=mask_m[:, None], other=0.0)   # FP16

    # Multiply
    out_buf = norm_fp16 * gate

    # Store output buffer for final projection
    tl.store(out_buf_ptr + offs_m[:, None] * H + offs_h[None, :],
             out_buf, mask=mask_m[:, None])
```

def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    x_norm = F.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2. Fused projection
    fused_w = torch.cat([w["left_proj.weight"], w["right_proj.weight"], w["left_gate.weight"],
                         w["right_gate.weight"], w["out_gate.weight"]], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_w)  # (R, 5*H) FP16

    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    else:

        left,
        right,
        out_gate,

        BLOCK_H=H,

@triton.jit
def ln_gate_kernel(
    out_ptr, gate_ptr, out_buf_ptr,
    ln_w_ptr, ln_b_ptr,
    R, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # Hidden indices
    offs_h = tl.arange(0, BLOCK_H)

    # Load out rows (FP16 -> FP32)
    out = tl.load(out_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
                  mask=mask_m[:, None], other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_H]

    # Compute mean and variance per row
    sum_val = tl.sum(out, axis=1)  # [BLOCK_M]
    sum_sq = tl.sum(out * out, axis=1)

    mean = sum_val / BLOCK_H
    var = sum_sq / BLOCK_H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Load LN gamma & beta (FP32)
    gamma = tl.load(ln_w_ptr + offs_h, mask=offs_h < BLOCK_H, other=0.0)  # [BLOCK_H]
    beta = tl.load(ln_b_ptr + offs_h, mask=offs_h < BLOCK_H, other=0.0)   # [BLOCK_H]

    # Normalize
    norm = (out - mean[:, None]) * rstd[:, None]          # FP32
    norm = norm * gamma[None, :] + beta[None, :]           # FP32
    norm_fp16 = norm.to(tl.float16)

    # Load gate
    gate = tl.load(gate_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
                   mask=mask_m[:, None], other=0.0)  # FP16

    # Multiply
    out_buf = norm_fp16 * gate

    tl.store(out_buf_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
             out_buf,
             mask=mask_m[:, None])

Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Input‑wise LayerNorm (FP32 → FP16) via torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)

    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # distance between two consecutive hidden slices in the destination layout
    stride_hidden = N * N                     # = nn

    # base offset for (b,i,k) in the flattened [B,H,N,N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # iterate over hidden dimension in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets to the fused projection for this hidden tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ LayerNorm over hidden dim + out‑gate multiplication (single load)
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_kernel(
    out_ptr,          # [R, H] FP16  (contracted tensor)
    gate_ptr,         # [R, H] FP16  (out_gate)
    out_buf_ptr,      # [R, H] FP16  (result after LN & gate)
    ln_w_ptr,         # [H]   FP32  (to_out_norm.weight)
    ln_b_ptr,         # [H]   FP32  (to_out_norm.bias)
    R, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]

    # ---- load one whole hidden vector per row ------------------------
    # layout after permute is contiguous: stride = H
    row_fp16 = tl.load(
        out_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
        mask=mask_m[:, None],
        other=0.0,
    ).to(tl.float32)                                   # [BLOCK_M, BLOCK_H] in FP32

    # ---- mean / variance (row‑wise) ---------------------------------
    sum_val = tl.sum(row_fp16, axis=1)                 # [BLOCK_M]
    sum_sq  = tl.sum(row_fp16 * row_fp16, axis=1)      # [BLOCK_M]

    mean = sum_val / BLOCK_H
    var  = sum_sq / BLOCK_H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- load LayerNorm parameters (γ, β) ----------------------------
    gamma = tl.load(ln_w_ptr + offs_h, mask=offs_h < BLOCK_H, other=0.0)   # FP32
    beta  = tl.load(ln_b_ptr + offs_h, mask=offs_h < BLOCK_H, other=0.0)   # FP32

    # ---- normalize ---------------------------------------------------
    norm = (row_fp16 - mean[:, None]) * rstd[:, None]                  # FP32
    norm = norm * gamma[None, :] + beta[None, :]                        # FP32
    norm_fp16 = norm.to(tl.float16)

    # ---- load gate ----------------------------------------------------
    gate = tl.load(
        gate_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
        mask=mask_m[:, None],
        other=0.0,
    )                                                                   # FP16

    # ---- final element‑wise product -----------------------------------
    out_buf = norm_fp16 * gate

    # ---- store ---------------------------------------------------------
    tl.store(
        out_buf_ptr + offs_m[:, None] * BLOCK_H + offs_h[None, :],
        out_buf,
        mask=mask_m[:, None],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack & constants
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]          # = H (always 128 in the tests)
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input layer‑norm (FP32 → FP16)
    # --------------------------------------------------------------
    # torch's highly‑optimized LayerNorm
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)        # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    # R = B·N·N; flatten input for the GEMM
    x_flat = x_norm.view(R, dim)       # [R, dim]  (contiguous)
    proj = F.linear(x_flat, fused_w)   # [R, 5·H]  FP16
    del x_norm, x_flat, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64                     # rows per program
    BLOCK_H = H                      # hidden‑dim tile (H = 128)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: out = left @ rightᵀ   (core einsum)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over hidden dim + out_gate multiplication
    # --------------------------------------------------------------
    # move hidden dim to the last position -> contiguous rows of size H
    out_perm      = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]
    del out, out_gate

    out_flat  = out_perm.view(R, H)        # [R, H]  FP16
    gate_flat = out_gate_perm.view(R, H)   # [R, H]  FP16
    del out_perm, out_gate_perm

    gated = torch.empty(R, H, dtype=torch.float16, device=device)   # [R, H]

    BLOCK_M = 128                     # rows per program for LN+gate
    grid_ln_gate = (triton.cdiv(R, BLOCK_M),)
    ln_gate_kernel[grid_ln_gate](
        out_flat,
        gate_flat,
        gated,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        R,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=H,
        num_warps=8,
    )
    del out_flat, gate_flat

    # --------------------------------------------------------------
    # 6️⃣ Final projection (hidden → dim) – cuBLAS GEMM
    # --------------------------------------------------------------
    to_out_w = w["to_out.weight"].to(torch.float16)   # [dim, H]  FP16, no bias
    final_fp16 = F.linear(gated, to_out_w)           # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim] FP32

    return final
