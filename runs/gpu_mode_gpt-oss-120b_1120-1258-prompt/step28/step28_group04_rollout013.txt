def apply_ln_gate_kernel(out_ptr, gate_ptr, mu_ptr, rstd_ptr, gamma_ptr, beta_ptr, out_gated_ptr, R, H, BLOCK_R, BLOCK_H):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0,BLOCK_R)
    mask_r = offs_r < R

    # Determine hidden offsets inside loop
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0,BLOCK_H)
        mask_h = offs_h < H

        # load out values: shape [BLOCK_R,BLOCK_H]
        out_vals = tl.load(out_ptr + offs_r[:,None] * H + offs_h[None,:], mask=mask_r[:,None] & mask_h[None,:], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_r[:,None] * H + offs_h[None,:], mask=mask_r[:,None] & mask_h[None,:], other=0.0).to(tl.float32)
        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0).to(tl.float32)[:,None] # shape [BLOCK_R,1]
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0).to(tl.float32)[:,None]

        # LN: (out - mu) * rstd
        norm = (out_vals - mu) * rstd

        # apply to_out_norm weight and bias
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

        norm = norm * gamma[None,:] + beta[None,:]

        # apply out_gate
        gated = norm * gate_vals

        # store to out_gated
        tl.store(out_gated_ptr + offs_r[:,None] * H + offs_h[None,:], gated.to(tl.float16), mask=mask_r[:,None] & mask_h[None,:])
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (torch.nn.functional.linear) producing a tensor
   of shape (R, 5·H) where R = B·N·N and H = hidden_dim.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * converts the fused projection back to the three tensors
     (left, right, out_gate) in layout [B, H, N, N] (FP16),
   * optionally applies the pair mask and sigmoid gates.
4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → tensor `[B, H, N, N]` (FP16).
5️⃣ Triton kernel `rowwise_stats_kernel` computes per‑position mean and
   rsqrt‑variance of `out` (LayerNorm over the hidden dimension H).
6️⃣ Triton kernel `apply_ln_gate_kernel` fuses:
   * the second LayerNorm (`to_out_norm`),
   * the element‑wise `out_gate` multiplication,
   * and writes the result into a compact `[R, H]` buffer `out_gated`.
7️⃣ Final projection is a single dense matmul using Torch’s
   highly‑optimized GEMM: `out_gated @ to_out.weight.T`,
   yielding the result `[B, N, N, dim]` (float32).

All kernels use mixed precision (FP32 for statistics, FP16 for bulk
data) and are tuned for an H100 (Triton 3.3.1).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 input
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim]   LN weight (γ) FP32
    b_ptr,        # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)      # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for (b,i,k) in the hidden‑major layout
    # hidden slices are spaced by N·N elements
    base_offset = b * (H * nn) + i * N + k   # offset of hidden‑slice 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # element indexes inside the fused projection [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * nn   # stride across hidden dimension

        tl.store(left_ptr   + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr  + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise statistics (mean & rsqrt‑var) of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base_offset = b * (H * nn) + i * N + j   # start of hidden‑slice 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over hidden dimension in chunks of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * nn
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Apply second LayerNorm, out_gate & store compact [R, H] buffer
# ----------------------------------------------------------------------
@triton.jit
def apply_ln_gate_kernel(
    out_ptr,       # [B, H, N, N] FP16
    gate_ptr,      # [B, H, N, N] FP16 (out_gate)
    mu_ptr,        # [R] FP32
    rstd_ptr,      # [R] FP32
    gamma_ptr,     # [H] FP32   (to_out_norm.weight)
    beta_ptr,      # [H] FP32   (to_out_norm.bias)
    outgated_ptr,  # [R, H] FP16   (output buffer)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # start of hidden‑slice 0 for this (b,i,j) location
    base_offset = b * (H * nn) + i * N + j

    # load per‑row statistics once (broadcast later)
    mu   = tl.load(mu_ptr   + offs_r,   mask=mask_r,   other=0.0).to(tl.float32)[:, None]   # [BLOCK_R,1]
    rstd = tl.load(rstd_ptr + offs_r,   mask=mask_r,   other=0.0).to(tl.float32)[:, None]   # [BLOCK_R,1]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a tile of the contracted tensor and the gate
        out_tile = tl.load(
            out_ptr  + base_offset[:, None] + offs_h[None, :] * nn,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_h[None, :] * nn,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm weight / bias (per‑hidden)
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)   # [BLOCK_H]
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0).to(tl.float32)   # [BLOCK_H]

        # (out - μ) * rstd
        norm = (out_tile - mu) * rstd

        # Apply to_out_norm (γ, β) then element‑wise out_gate
        norm = norm * gamma[None, :] + beta[None, :]
        gated = norm * gate_tile

        # Store compact result
        tl.store(
            outgated_ptr + offs_r[:, None] * H + offs_h[None, :],
            gated.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :]
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into one GEMM (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)               # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask & sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy buffer

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr  # free

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (torch.bmm)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left, right  # free memory (no longer needed)

    # ------------------------------------------------------------------
    # 5️⃣ Per‑position mean & rsqrt‑variance of `out` (LayerNorm over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128   # H ≤ 128 → fits in a single tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate → compact [R, H] buffer
    # ------------------------------------------------------------------
    out_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 32
    grid_outln = (triton.cdiv(R, BLOCK_R_OUT),)
    apply_ln_gate_kernel[grid_outln](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        out_gated,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    # No longer need the raw `out` or `out_gate` tensors
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final dense projection (torch.nn.functional.linear, highly‑optimized GEMM)
    # ------------------------------------------------------------------
    # Cast to FP32 for the final matmul (the weight is kept FP16 → matmul in FP32)
    final = torch.nn.functional.linear(
        out_gated.float(),                     # [R, H] → FP32
        w["to_out.weight"].float(),            # [dim, H] → FP32, interpreted as weight.T inside linear
        bias=None,
    )                                          # [R, dim] FP32

    final = final.view(B, N, N, dim)            # reshape to [B, N, N, dim]

    return final
