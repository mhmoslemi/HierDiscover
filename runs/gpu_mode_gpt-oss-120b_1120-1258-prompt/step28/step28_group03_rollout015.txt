def norm_gate_kernel(out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr, outgated_ptr, total_rows, H, N, eps, BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr):

tid = program_id(0).
Process rows per block. Use offsets for hidden dimension similarly as final_output.

Pseudo:

pid_m = tl.program_id(0)
row_start = pid_m * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)
mask_m = offs_m < total_rows

nn = N * N
b = offs_m // nn
rem = offs_m - b * nn
i = rem // N
j = rem - i * N

hidden_stride = N * N

base = b * (H * hidden_stride) + i * N + j

mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)  # [M]
rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

for k_start in range(0, H, BLOCK_K):
   offs_k = k_start + tl.arange(0, BLOCK_K)
   mask_k = offs_k < H

   out_vals = tl.load(out_ptr + base[:, None] + offs_k[None,:] * hidden_stride,
                      mask=mask_m[:,None] & mask_k[None,:],
                      other=0.0).to(tl.float32)
   gate_vals = tl.load(gate_ptr + base[:, None] + offs_k[None,:] * hidden_stride,
                      mask=mask_m[:,None] & mask_k[None,:],
                      other=0.0).to(tl.float32)

   gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
   beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

   diff = (out_vals - mu[:, None]) * rstd[:, None]
   norm = diff * gamma[None,:] + beta[None,:]
   final = norm * gate_vals

   # store final as FP16
   # outgated_ptr shape [R, H] layout row-major contiguous: offset = offs_m[:,None] * H + offs_k[None,:]
   dst_ptr = outgated_ptr + offs_m[:,None] * H + offs_k[None,:]
   tl.store(dst_ptr, final.to(tl.float16),
            mask=mask_m[:,None] & mask_k[None,:])

def norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr, outgated_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)

    nn = N * N

    rem = offs_m - b * nn

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  LayerNorm on the flattened pair tensor (B·N·N, C) – use the highly‑optimised
    torch.nn.functional.layer_norm (FP32 → FP16 conversion afterwards).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single GEMM (FP16) → tensor [R, 5·H] where R = B·N·N.

3️⃣  Split the fused projection, apply the optional pair‑mask and three sigmoid
    gates in a Triton kernel → tensors `left`, `right`, `out_gate` with layout
    [B, H, N, N] (FP16).

4️⃣  Core einsum `out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]` is performed by a
    batched GEMM (torch.bmm) on the reshaped tensors → `out` shape [B, H, N, N] (FP16).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` (over the hidden dimension H) in a
    small Triton kernel → `mu`, `rstd` (FP32, length R).

6️⃣  Normalise `out`, apply the second LayerNorm (`to_out_norm`), the output‑gate
    (`out_gate`) and store the result as a dense [R, H] matrix `out_gated`
    (FP16) in a second Triton kernel.

7️⃣  Final projection `to_out` is a large GEMM `out_gated @ to_out.weight.T`
    (cuBLAS) yielding `[R, dim]` (FP32).  Reshape to `[B, N, N, dim]`.

All heavy arithmetic uses FP16/Tensor‑cores; only the few reductions for statistics
are done in FP32.  The implementation launches only three custom Triton kernels
(split‑gate, rowwise‑stats, norm‑gate) and relies on cuBLAS for the two large
matrix multiplications, giving a very low kernel‑launch overhead.
"""

import torch.nn.functional as F

import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16  (fused projection)
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 mask (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program (over R)
    BLOCK_H: tl.constexpr,   # hidden dim tiles
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # Base offset into the (B·H·N·N) tensors
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,

            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination pointers in the 4‑D tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile (use H to get a single pass)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Since BLOCK_K == H we loop only once
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 3️⃣  Normalise `out`, apply the second LayerNorm and the output‑gate.
#     Store a dense [R, H] matrix ready for the final linear layer.
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    outgated_ptr,   # [R, H] FP16  ← result
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile (use H for a single pass)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # Load contracted tensor and gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M,K]

        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)           # [M,K]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out - μ) * rstd) * γ + β   →   normalized
        diff   = (vals_out - mu[:, None]) * rstd[:, None]           # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]              # [M,K]

        # Apply the output‑gate
        gated  = norm * vals_gate                                   # [M,K]

        # Store to the dense [R, H] buffer (row‑major)
        dst_ptr = outgated_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask_tensor  : torch.Tensor [B, N, N]      (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 2️⃣ LayerNorm over the last dimension (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim)                           # [R, dim] FP32
    x_norm = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                               # [R, dim] FP16

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5·H, dim)

    # (R, 5·H) = (R, dim) @ (5·H, dim).T   →   FP16 output
    proj = F.linear(x_norm, fused_w)                  # [R, 5·H] FP16
    del x_norm, fused_w                               # free memory

    # --------------------------------------------------------------
    # 4️⃣ Split, apply optional mask & sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask: [B, N, N] → flatten → FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows per program (tunable)
    BLOCK_H = 64   # hidden‑dim tile (tunable)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 5️⃣ Core pairwise multiplication – batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right                                         # free memory

    # --------------------------------------------------------------
    # 6️⃣ Row‑wise statistics (mean & rsqrt‑variance) over hidden dim
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = H                     # 128 → single pass over H
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Normalise, apply second LayerNorm + out_gate → dense [R, H]
    # --------------------------------------------------------------
    out_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_NG = 128
    BLOCK_K_NG = H                       # single pass over hidden dim
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    norm_gate_kernel[grid_ng](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],      # γ (FP32)
        w["to_out_norm.bias"],        # β (FP32)
        out_gated,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_K=BLOCK_K_NG,
        num_warps=8,
    )
    del out, out_gate, mu, rstd      # free memory we no longer need

    # --------------------------------------------------------------
    # 8️⃣ Final projection (cuBLAS) – produces [R, dim] FP32
    # --------------------------------------------------------------
    # to_out.weight shape: (dim, H)
    final = F.linear(out_gated.to(torch.float32), w["to_out.weight"], bias=None)  # [R, dim] FP32
    del out_gated

    # --------------------------------------------------------------
    # 9️⃣ Reshape to the expected output shape
    # --------------------------------------------------------------
    final = final.view(B, N, N, dim)                 # [B, N, N, dim]
    return final
