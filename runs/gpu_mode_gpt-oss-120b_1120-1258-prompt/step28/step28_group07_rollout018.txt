@triton.jit
def fused_linear_gate_kernel(
    x_ptr,
    w_proj_ptr,
    w_gate_ptr,
    out_ptr,
    mask_ptr,
    R, H, D,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute mask values if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # init accumulators
    acc_proj = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    for d_start in range(0, D, BLOCK_D):
        # offsets
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # load X tile: shape (BLOCK_R, BLOCK_D)
        x_ptrs = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # weight proj tile: shape (BLOCK_D, BLOCK_H) from transposed weight
        w_proj_ptrs = w_proj_ptr + offs_d[:, None] * H + tl.arange(0, BLOCK_H)[None, :]  # but need offs_h.
        # Actually we need to compute offs_h similarly:
        offs_h = tl.arange(0, BLOCK_H)
        w_proj = tl.load(w_proj_ptr + offs_d[:, None] * H + offs_h[None, :], mask=mask_d[:, None] & (offs_h < H), other=0.0).to(tl.float32)

        # same for w_gate
        w_gate = tl.load(w_gate_ptr + offs_d[:, None] * H + offs_h[None, :], mask=mask_d[:, None] & (offs_h < H), other=0.0).to(tl.float32)

        # accumulate
        acc_proj += tl.dot(x, w_proj)    # dot returns BLOCK_R x BLOCK_H
        acc_gate += tl.dot(x, w_gate)

    # compute gate via sigmoid
    gate = (1.0 / (1.0 + tl.exp(-acc_gate)))   # FP32
    # apply mask: left = acc_proj * gate * mask_val
    result = acc_proj * gate * mask_val

    # store result to out_ptr as FP16
    out_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
    tl.store(out_ptrs, result.to(tl.float16), mask=mask_r[:, None] & (offs_h < H))

"""
Outgoing TriMul – highly‑fused forward implementation (no grads).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ Three fused GEMM+gate kernels:
   • left   = (x·W_left_proj) * sigmoid(x·W_left_gate) * mask
   • right  = (x·W_right_proj) * sigmoid(x·W_right_gate) * mask
   • out_gate = sigmoid(x·W_out_gate)
   All kernels work on the flattened [R, D] tensor (R = B·N·N, D = dim) and
   write directly the final tensors of shape [R, H] (H = hidden_dim);
   the large intermediate [R,5·H] matrix used in the reference
   implementation is completely avoided.
3️⃣ Batched GEMM (core einsum) – torch.bmm:
   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   (FP16, tensor cores).
4️⃣ Row‑wise mean & rsqrt‑variance of `out` (Triton kernel).
5️⃣ Final fused kernel:
   – second LayerNorm on `out`,
   – element‑wise multiply by `out_gate`,
   – projection back to `dim` using the transposed “to_out” weight.
   Result‑type is float32 → `[B, N, N, dim]`.

The implementation keeps the memory footprint low (≈3·R·H fp16) and
exposes the dominant work to tensor‑core GEMMs, achieving a large
speed‑up over the naïve reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, D]   FP32 input
    y_ptr,   # [R, D]   FP16 output
    w_ptr,   # [D]      γ (FP32)
    b_ptr,   # [D]      β (FP32)
    rows,    # scalar R
    dim,     # scalar D
    eps,     # scalar
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # first pass – compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]
    # compute statistics
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # second pass – write normalized + affine transformed values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # affine γ, β
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gate (left & right)
# ----------------------------------------------------------------------
@triton.jit
def fused_linear_gate_kernel(
    x_ptr,           # [R, D]           FP16
    w_proj_ptr,      # [D, H] (proj)    FP16   – transposed weight
    w_gate_ptr,      # [D, H] (gate)    FP16   – transposed weight
    out_ptr,         # [R, H]           FP16   – final left / right
    mask_ptr,        # [R] (optional)   FP16
    R, H, D,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)   # [BLOCK_R]
        mask_row = mask_row[:, None]                       # [BLOCK_R,1]
    else:
        mask_row = tl.full([BLOCK_R, 1], 1.0, tl.float32)

    # loop over hidden dimension tiles
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for this tile
        acc_proj = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # loop over the input dimension D
        for d_start in range(0, D, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < D

            # X tile  [BLOCK_R, BLOCK_D]
            x = tl.load(
                x_ptr + offs_r[:, None] * D + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # weight tiles  [BLOCK_D, BLOCK_H]
            w_proj = tl.load(
                w_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_gate = tl.load(
                w_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # dot products
            acc_proj += tl.dot(x, w_proj)   # [BLOCK_R, BLOCK_H]
            acc_gate += tl.dot(x, w_gate)

        # sigmoid on gate
        gate = 1.0 / (1.0 + tl.exp(-acc_gate))      # FP32

        # apply mask and write result
        out = acc_proj * gate * mask_row            # FP32
        out_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_ptrs, out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Linear + sigmoid (out_gate)
# ----------------------------------------------------------------------
@triton.jit
def linear_sigmoid_kernel(
    x_ptr,           # [R, D] FP16
    w_ptr,           # [D, H] FP16 (transposed)
    out_ptr,         # [R, H] FP16
    R, H, D,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_start in range(0, D, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < D

            x = tl.load(
                x_ptr + offs_r[:, None] * D + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            w = tl.load(
                w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            acc += tl.dot(x, w)   # [BLOCK_R, BLOCK_H]

        gate = 1.0 / (1.0 + tl.exp(-acc))   # sigmoid, FP32
        out_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_ptrs, gate.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Row‑wise mean & rsqrt‑variance of contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows,   # R = B·N·N
    H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m,   mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 5️⃣ Final LayerNorm + out_gate + projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden values and gating tensor
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (x - μ) * rstd → normalize → affine → gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term   = norm * vals_gate                                 # [M,K]
        term_fp16 = term.to(tl.float16)                           # fp16 for dot

        # load slice of final projection weight (H×dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    """

    # --------------------------------------------------------------

    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate for left, right, out_gate
    # --------------------------------------------------------------
    # transposed weight matrices (shape [dim, hidden])
    left_proj_T  = w["left_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_T  = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_proj_T = w["right_proj.weight"].to(torch.float16).t().contiguous()
    right_gate_T = w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_T   = w["out_gate.weight"].to(torch.float16).t().contiguous()

    # allocate outputs (flattened [R, H])
    left_flat     = torch.empty(R, H, dtype=torch.float16, device=device)
    right_flat    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate_flat = torch.empty(R, H, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # kernel launch parameters
    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32

    grid_proj = (triton.cdiv(R, BLOCK_R),)

    # left
    fused_linear_gate_kernel[grid_proj](
        x_norm,
        left_proj_T,
        left_gate_T,
        left_flat,
        mask_ptr,
        R,
        H,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # right
    fused_linear_gate_kernel[grid_proj](
        x_norm,
        right_proj_T,
        right_gate_T,
        right_flat,
        mask_ptr,
        R,
        H,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # out_gate (no mask)
    linear_sigmoid_kernel[grid_proj](
        x_norm,
        out_gate_T,
        out_gate_flat,
        R,
        H,
        dim,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_norm, left_proj_T, left_gate_T, right_proj_T, right_gate_T, out_gate_T, mask_ptr

    # --------------------------------------------------------------
    # reshape to [B, H, N, N] layout
    # --------------------------------------------------------------
    left = left_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del left_flat, right_flat, out_gate_flat

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM (core einsum)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)        # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 ⇒ one tile suffices
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + projection (fused)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows processed per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```
