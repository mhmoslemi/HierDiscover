1️⃣  Input LayerNorm (torch.nn.functional.layer_norm) – [B,N,N,dim] → FP16
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into one GEMM (R×dim)·(dim×5·H) → (R×5·H)   (R = B·N·N)
3️⃣  Custom Triton kernel `split_fuse_gate_kernel`:
    * splits the fused projection,
    * applies the optional pair‑mask,
    * evaluates the three sigmoid gates,
    * writes `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → FP16
5️⃣  Row‑wise LayerNorm on the contracted tensor (torch.nn.functional.layer_norm)
    – the hidden dimension H becomes the normalized axis.
6️⃣  Element‑wise gating with `out_gate`.
7️⃣  Final linear projection back to the original channel dimension `dim`
    (single GEMM: (R×H)·(H×dim) → (R×dim)).

Only step 3 needs a handwritten Triton kernel; all other stages use
high‑throughput cuBLAS/cuDNN kernels.  The code works for
B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128 and
handles the presence or absence of a pair mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: split the fused projection and apply mask/gates
# ------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  (FP16) fused projection
    left_ptr,      # [B, H, N, N] (FP16) output
    right_ptr,     # [B, H, N, N] (FP16) output
    out_gate_ptr,  # [B, H, N, N] (FP16) output
    mask_ptr,      # [R] (FP16) – only read if HAS_MASK==1
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows of the flattened pair tensor
    BLOCK_H: tl.constexpr,   # hidden‑dim tile (≤ H)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat row index  r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        m_val = m_val[:, None]                         # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = nn                               # distance between two hidden planes
    # base offset for (b, i, k) in the flattened [B, H, N, N] tensors (h==0)
    base = b * (H * hidden_stride) + i * N + k      # [BLOCK_R]

    # --------------------------------------------------------
    # iterate over hidden dimension tiles
    # --------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside the fused projection: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # load the five slices (left, right, left_gate, right_gate, out_gate)
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate   # shape [BLOCK_R, BLOCK_H]

        # destination offsets in the flat output tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ------------------------------------------------------------
# Entrypoint
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch, FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim]  float32
    x_norm = x_norm.to(torch.float16)  # cast to half for the GEMMs

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five Linear maps into a single GEMM (R×dim)·(dim×5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # [R, 5·H]  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates (custom Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64                     # rows per program
    BLOCK_H = 128                    # we can load the whole hidden dim at once (H≤128)

    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM: out = left @ rightᵀ
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                       # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                      # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over the hidden dimension H
    #    (need H as the last dimension → permute)
    # ------------------------------------------------------------------
    # bring H to the trailing axis so that torch.layer_norm can work on it
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                                                        # FP16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply out_gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_perm                         # element‑wise, FP16
    gated_flat = gated.view(R, H)                            # [R, H]

    w_out = w["to_out.weight"].to(torch.float16)             # [dim, H]
    out_final_half = F.linear(gated_flat, w_out)             # [R, dim], FP16
    out_final = out_final_half.view(B, N, N, dim).to(torch.float32)

    return out_final
```
