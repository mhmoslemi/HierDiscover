    3. Triton kernel splits the projection, applies the pair mask and the three sigmoid gates.

    5. Torch's built‑in LayerNorm is used on the contracted tensor (H‑dim) plus the

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) – Triton.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
    single GEMM (FP16) → tensor [R, 5·H] (R = B·N·N, H = hidden_dim) – Torch.
3️⃣  Split the fused projection, apply optional pair mask and three sigmoid gates –
    Triton (splits into left, right, out_gate tensors of shape [B, H, N, N]).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k] → [B, H, N, N] (FP16).
5️⃣  Second LayerNorm on the hidden dimension, followed by element‑wise out‑gate
    multiplication – `torch.nn.functional.layer_norm`.
6️⃣  Final linear projection (H → dim) – `torch.nn.functional.linear`.
   The result is cast to float32 and reshaped back to [B, N, N, dim].

Only the element‑wise parts (LayerNorm, mask handling, sigmoid) are written in
Triton; all large dense GEMMs use cuBLAS/torch kernels, giving a fast
end‑to‑end implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm  (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]    FP32 input
    y_ptr,    # [R, dim]    FP16 output
    w_ptr,    # [dim]       γ   FP32
    b_ptr,    # [dim]       β   FP32
    rows,     # scalar R
    dim,      # scalar dim
    eps,      # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass : compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)        # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass : write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, mask and sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]   FP16
    right_ptr,     # [B, H, N, N]   FP16
    out_gate_ptr,  # [B, H, N, N]   FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    # start offset for h == 0 (layout: [B, H, N, N] → contiguous in H)
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H)   FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_f
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # Shape: [B, H, N, N] → view as [B·H, N, N]
    left_mat  = left.view(B * H, N, N)           # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (torch kernels)
    # --------------------------------------------------------------
    # reshape to [R, H] where R = B·N·N
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # [R, H] FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H] FP16

    # LayerNorm over the hidden dimension (uses FP32 accumulator internally)
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # [R, H] FP16

    # Apply the out‑gate element‑wise
    gated = out_norm * out_gate_flat      # FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection (H → dim) – linear + cast to float32
    # --------------------------------------------------------------
    final = F.linear(gated, w["to_out.weight"])          # [R, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)   # float32 output

    return final
