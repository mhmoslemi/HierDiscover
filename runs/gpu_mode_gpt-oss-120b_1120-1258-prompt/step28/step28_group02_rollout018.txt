Implementation:

```
pid = tl.program_id(0)
row_start = pid * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)
mask_m = offs_m < rows

sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
    offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
    vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

    sum_val += tl.sum(vals, axis=1)
    sum_sq += tl.sum(vals * vals, axis=1)

mean = sum_val / dim
var = sum_sq / dim - mean * mean
rstd = tl.rsqrt(var + eps)

# Second pass: apply normalization, gamma, beta, multiply gate
for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
    offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
    vals = tl.load(ptr_x, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

    normalized = (vals - mean[:, None]) * rstd[:, None]

    gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
    beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

    out_val = normalized * gamma[None, :] + beta[None, :]

    # multiply by gate
    ptr_gate = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
    gate_val = tl.load(ptr_gate, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

    out_final = out_val * gate_val

    tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :], out_final.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

```python
"""
TriMul outgoing kernel (forward only) – Triton implementation.

Algorithm
---------
1. Row‑wise LayerNorm on the input (dim) → FP16.
2. One fused linear projection creates five tensors:
   left, right, left_gate, right_gate, out_gate.  The projection is
   done with a cuBLAS GEMM (torch.nn.functional.linear) on FP16.
3. split_fuse_gate_kernel extracts the five slices, applies the optional
   pair‑wise mask, and folds the three sigmoid gates, writing `left`,
   `right` and `out_gate` in layout [B, H, N, N] (H = hidden_dim).
4. Batched GEMM (`torch.bmm`) computes the core einsum:
   out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   →  shape [B, H, N, N] (FP16).
5. layernorm_gate_kernel performs a second LayerNorm over the hidden
   dimension and multiplies by `out_gate` in‑place, writing the result
   into a flat tensor of shape [R, H] where R = B·N·N.
6. The final linear projection (hidden_dim → dim) is performed with
   cuBLAS (`torch.nn.functional.linear`) on FP16.  The output is cast
   back to FP32 and reshaped to [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (generic, works for any dim)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 (or FP16 – it will be cast to FP32)
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim] γ (FP32)
    b_ptr,        # [dim] β (FP32)
    rows,         # total rows R
    dim,          # feature dimension
    eps,          # epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: apply γ,β ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# 2️⃣ Split‑fuse‑gate kernel (creates left, right, out_gate)
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16 out
    right_ptr,      # [B, H, N, N] FP16 out
    out_gate_ptr,   # [B, H, N, N] FP16 out
    mask_ptr,       # [R] FP16 mask (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)     # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r -> b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn   # stride between hidden planes

    # base offset of the first hidden plane (h=0) for each row
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (FP32 for accuracy, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])

# --------------------------------------------------------------
# 3️⃣ Row‑wise LayerNorm over hidden dimension + out_gate multiplication.
#    Output is written as a flat [R, H] tensor (row‑major).
# --------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    out_ptr,            # [B, H, N, N] FP16 (raw “out” from GEMM)
    gate_ptr,           # [B, H, N, N] FP16 (out_gate)
    out_flat_ptr,       # [R, H] FP16 output (normed * gate) -- can be same as out_ptr
    w_ptr,              # [H] γ (FP32)  – to_out_norm.weight
    b_ptr,              # [H] β (FP32)  – to_out_norm.bias
    rows,               # R = B·N·N
    hidden_dim,         # H
    N,                  # sequence length
    eps,                # epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # tile along hidden dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    # offset of the first hidden component (h=0) for each row
    base_in = b * (hidden_dim * hidden_stride) + i * N + j     # [BLOCK_M]

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        ptr = out_ptr + base_in[:, None] + offs_d[None, :] * hidden_stride
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: normalize, apply LN params, multiply gate, write flat ----------
    for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load raw out
        ptr_out = out_ptr + base_in[:, None] + offs_d[None, :] * hidden_stride
        val_out = tl.load(ptr_out,
                          mask=mask_m[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)

        # load gate
        ptr_gate = gate_ptr + base_in[:, None] + offs_d[None, :] * hidden_stride
        val_gate = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        # layer‑norm
        norm = (val_out - mean[:, None]) * rstd[:, None]
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        norm = norm * gamma[None, :] + beta[None, :]

        # gate
        gated = norm * val_gate

        # write to flat output
        dst_ptr = out_flat_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

            w["right_gate.weight"],

"""
TriMul outgoing kernel (forward only) – Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (dim) → FP16.
2️⃣ One fused linear projection creates five tensors:
   left, right, left_gate, right_gate, out_gate.  The projection is
   done with a cuBLAS GEMM (torch.nn.functional.linear) on FP16.
3️⃣ split_fuse_gate_kernel extracts the five slices, applies the optional
   pair‑wise mask, and folds the three sigmoid gates, writing `left`,
   `right` and `out_gate` in layout [B, H, N, N] (H = hidden_dim).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
   out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   →  shape [B, H, N, N] (FP16).
5️⃣ layernorm_gate_kernel performs a second LayerNorm over the hidden
   dimension and multiplies by `out_gate` in‑place, writing the result
   into a flat tensor of shape [R, H] where R = B·N·N.
6️⃣ The final linear projection (hidden_dim → dim) is performed with
   cuBLAS (`torch.nn.functional.linear`) on FP16.  The output is cast
   back to FP32 and reshaped to [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Generic row‑wise LayerNorm (FP32→FP16 or FP16→FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 (or FP16 – casted inside)
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim] γ (FP32)
    b_ptr,        # [dim] β (FP32)
    rows,         # total rows R
    dim,          # feature dimension
    eps,          # epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: apply γ,β ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split‑fuse‑gate kernel (left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 mask (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)     # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn

    base = b * (H * hidden_stride) + i * N + k   # start offset of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])

    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M2 = 128
    BLOCK_D2 = 32                    # hidden‑dim tile (H ≤ 128)
    grid_ln2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_gate_kernel[grid_ln2](
        out,
        out_gate,
        out_norm_gate,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    del out, out_gate   # free memory

    # --------------------------------------------------------------
    # 6️⃣ Final projection (hidden_dim → dim) using cuBLAS
    # --------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    final_fp16 = F.linear(out_norm_gate, w_out_fp16)    # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
