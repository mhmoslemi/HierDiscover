"""
Optimized outgoing TriMul (AlphaFold‑3) – forward only.
----------------------------------------------------
Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16)
   – implemented with a custom Triton kernel (statistics in FP32,
     normalization in FP16).

2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
   into one GEMM:  proj = X_norm @ W_fused.T   (FP16).

3️⃣ Split the fused projection, apply the optional pair mask and the three
   sigmoid gates **with pure Torch ops** (high‑throughput elementwise kernels).

4️⃣ Core cubic einsum reduced to a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   – executed by a single torch.bmm over the batch (B·H matrices).

5️⃣ Row‑wise mean / variance of `out` (FP32), then LayerNorm over the hidden
   dimension, multiply by `out_gate` and finally project to the model
   dimension with another GEMM (FP16 → FP32).

Only the first LayerNorm is a Triton kernel; the rest relies on highly
optimized cuBLAS / Torch kernels, which on an H100 gives a large speed‑up
over the naïve PyTorch implementation while keeping the code simple and
correct.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # [R, dim]   FP32 input
    y_ptr,                # [R, dim]   FP16 output
    w_ptr,                # [dim]      γ   FP32
    b_ptr,                # [dim]      β   FP32
    rows,                 # scalar R
    dim,                  # scalar dim
    eps,                  # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = bool(cfg.get("nomask", True))
    eps         = 1e-5
    device      = inp.device

    B, N, _, _ = inp.shape          # input shape: [B, N, N, dim]
    R = B * N * N                    # number of pair tokens (flattened)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the pair tensor (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = inp.reshape(R, dim).contiguous()                        # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device) # [R, dim] FP16

    BLOCK_M = 256          # rows per program – a good compromise for all cases
    BLOCK_D = 64           # columns per program – fits most dims (≤384)
    grid_ln = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fuse 5 linear maps into a single GEMM (FP16)
    # ------------------------------------------------------------------
    #   (left, right, left_gate, right_gate, out_gate) → [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                # (R, 5·H)   FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, apply mask and sigmoid gates (pure PyTorch)
    # ------------------------------------------------------------------
    # view as [R, 5, H] – the 5 groups are contiguous
    proj = proj.view(R, 5, H)

    left_raw      = proj[:, 0, :]   # (R, H)
    right_raw     = proj[:, 1, :]   # (R, H)
    left_gate_raw = proj[:, 2, :]   # (R, H)
    right_gate_raw= proj[:, 3, :]   # (R, H)
    out_gate_raw  = proj[:, 4, :]   # (R, H)

    # sigmoid gates (stable FP32 -> FP16)
    left_gate  = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw).to(torch.float16)

    # optional mask – broadcast over the hidden dimension
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R)               # [R]
        mask_exp  = mask_flat[:, None]                              # [R,1]
        left  = left_raw  * mask_exp * left_gate
        right = right_raw * mask_exp * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # reshape to [B, N, N, H] and permute to [B, H, N, N] for the batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣  Core einsum → batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    #   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    left_mat  = left.view(B * H, N, N)                         # (B·H, N, N)  FP16
    right_mat = right.view(B * H, N, N)                        # (B·H, N, N)  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # (B·H, N, N)  FP16
    out = out.view(B, H, N, N)                                 # (B, H, N, N) FP16
    del left_mat, right_mat, left, right                         # free memory

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise mean / variance (FP32) + 2nd LayerNorm + out_gate
    # ------------------------------------------------------------------
    #   (a) statistics
    out_fp32 = out.to(torch.float32)
    mu  = out_fp32.mean(dim=1, keepdim=True)                     # (B,1,N,N) FP32
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)     # (B,1,N,N) FP32
    rstd = torch.rsqrt(var + eps)                               # (B,1,N,N) FP32

    #   (b) cast statistics to FP16 for the elementwise ops
    mu_fp16   = mu.to(torch.float16)
    rstd_fp16 = rstd.to(torch.float16)

    #   (c) LayerNorm parameters (γ,β) – cast once to FP16
    gamma = w["to_out_norm.weight"].to(torch.float16).view(1, H, 1, 1)   # (1,H,1,1)
    beta  = w["to_out_norm.bias"  ].to(torch.float16).view(1, H, 1, 1)   # (1,H,1,1)

    #   (d) normalize → apply 2nd LN → multiply with out_gate
    out_norm = ((out - mu_fp16) * rstd_fp16)           # (B,H,N,N) FP16
    out_norm = out_norm * gamma + beta                # (B,H,N,N) FP16
    out_norm = out_norm * out_gate                    # (B,H,N,N) FP16
    del out, out_gate, mu, rstd, mu_fp16, rstd_fp16   # free memory

    # ------------------------------------------------------------------
    # 6️⃣  Final projection (H → dim) – another GEMM (FP16 → FP32)
    # ------------------------------------------------------------------
    # flatten to (R, H) where R = B·N·N
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16
    w_out_fp16 = w["to_out.weight"].to(torch.float16)                  # (dim, H)
    final_fp16 = F.linear(out_flat, w_out_fp16)                        # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)            # (B,N,N,dim) FP32

    return final
