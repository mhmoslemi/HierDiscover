Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  LayerNorm over the flattened pair tensor with the official
    `torch.nn.functional.layer_norm`.  The result is cast to FP16 for the
    remaining pipeline.

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM (FP16) → tensor `[R, 5·H]` where
    `R = B·N·N` and `H = hidden_dim`.

3️⃣  Split the fused projection, apply the optional pair mask and the three
    sigmoid gates *inside a tiny Triton kernel* → tensors
    `left`, `right`, `out_gate` with layout `[B, H, N, N]` (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:

        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]

    → tensor `[B, H, N, N]` (FP16).

5️⃣  Second LayerNorm + out‑gate are performed with the built‑in
    `torch.nn.functional.layer_norm` (FP16 → FP16).

6️⃣  The final linear projection `H → dim` is a single large matrix‑multiply:

        (out_norm * out_gate).view(R, H) @ W_outᵀ   (FP16)

    The result is reshaped to `[B, N, N, dim]` and cast to FP32.

All heavy arithmetic is done in FP16 (Tensor‑cores on H100), while the
statistics (means, variances) are accumulated in FP32 for numerical
stability.  The only custom Triton kernel is the split‑&‑gate kernel,
which avoids an extra memory pass compared with a pure‑PyTorch
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates (Triton)
# -------------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]   FP16
    right_ptr,     # [B, H, N, N]   FP16
    out_gate_ptr,  # [B, H, N, N]   FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N                     # stride between consecutive hidden slices
    base = b * (H * hidden_stride) + i * N + k   # start offset for h == 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # layout of the fused projection: [R, 5·H] (contiguous in H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]     # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# -------------------------------------------------------------------------
# Entrypoint
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                     # flattened pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.view(R, dim)                                   # [R, dim]  FP32
    x_norm = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                         # [R, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣  Fuse 5 linear maps (left, right, gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                              # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣  Split, apply mask and sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)           # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm + out_gate (torch)
    # --------------------------------------------------------------
    # move hidden dim to the last axis so that torch.layer_norm can operate on it
    out_perm = out.permute(0, 2, 3, 1).contiguous()           # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]

    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                          # FP16

    gated = out_norm * out_gate_perm                            # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 6️⃣  Final projection H → dim
    # --------------------------------------------------------------
    gated_flat = gated.view(R, H)                               # [R, H] FP16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim] FP16

    final_fp16 = torch.mm(gated_flat, w_out_T)                  # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)    # FP32

    return final
