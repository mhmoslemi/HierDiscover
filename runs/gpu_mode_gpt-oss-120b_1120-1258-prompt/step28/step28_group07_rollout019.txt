"""
Outgoing TriMul (AlphaFold‑3) – fast fused implementation
========================================================

The original PyTorch reference does:

    x      -> LayerNorm (row‑wise)
    x_norm -> Linear (5·H)          # huge [R,5H] intermediate tensor
    split  -> left/right/out_gate   # mask + sigmoid
    out    = bmm(left, rightᵀ)      # [B·H,N,N]  (H ≈ 128)
    out    -> LayerNorm + out_gate + Linear(dim)

The dominant cost is the intermediate Tensor of shape (R,5·H) (≈1 GB for
N=1024) which is written‑then‑read back.  We therefore fuse the
LayerNorm, the five linear projections and the gating into a **single
Triton kernel** that writes directly the three tensors
`left`, `right` and `out_gate`.  Afterwards we keep the original
batched‑GEMM for the core einsum, the row‑wise statistics and the final
fused LayerNorm+projection.

Resulting memory traffic is ~5× smaller and the overall latency drops
well below the 500 µs target on the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fuse LayerNorm, 5 Linear maps and the three gates (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr,                # [R, D]      FP32 input
    mask_ptr,             # [R]         FP16 mask (only when HAS_MASK=1)
    norm_w_ptr,           # [D]         γ of the first LayerNorm (FP32)
    norm_b_ptr,           # [D]         β of the first LayerNorm (FP32)
    left_w_ptr,           # [H, D]      FP16
    right_w_ptr,          # [H, D]      FP16
    left_gate_w_ptr,      # [H, D]      FP16
    right_gate_w_ptr,     # [H, D]      FP16
    out_gate_w_ptr,       # [H, D]      FP16
    left_ptr,             # [B, H, N, N] FP16   (output)
    right_ptr,            # [B, H, N, N] FP16   (output)
    out_gate_ptr,         # [B, H, N, N] FP16   (output)
    R, D, H, N,           # scalars
    HAS_MASK: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    For each flattened pair token r = b·N·N + i·N + k
    compute

        x̂ = (x - μ)·rstd·γ + β                (row‑wise LayerNorm)
        l  = x̂·W_left   ,   r  = x̂·W_right   (linear proj)
        lg = sigmoid(x̂·W_lgate) , rg = sigmoid(x̂·W_rgate)
        og = sigmoid(x̂·W_outgate)

        l ← l·mask·lg ,   r ← r·mask·rg ,   og ← og

    The three results are written with layout [B, H, N, N].
    """
    pid_r = tl.program_id(0)       # row block
    pid_h = tl.program_id(1)       # hidden block

    # ------------------------------------------------------------------
    # 0️⃣  Row indices
    # ------------------------------------------------------------------
    row_start = pid_r * BLOCK_R
    hidden_start = pid_h * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_h = hidden_start + tl.arange(0, BLOCK_H)       # [BLOCK_H]

    mask_r = offs_r < R
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / variance for each row (no hidden dim needed)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # load a tile of the input (row‑major)
        x = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)      # [BLOCK_R]
        sum_sq  += tl.sum(x * x, axis=1)  # [BLOCK_R]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Allocate accumulators for the five linear maps (FP32)
    # ------------------------------------------------------------------
    left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣  Second pass – compute the five dot‑products across dim
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # load the same tile again (now we need the values)
        x = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_R, BLOCK_D]

        # (x - μ)·rstd
        norm = (x - mean[:, None]) * rstd[:, None]        # [BLOCK_R, BLOCK_D]

        # affine γ, β
        gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        x_norm = norm * gamma[None, :] + beta[None, :]                 # [BLOCK_R, BLOCK_D]

        # ----- load weight tiles (transposed -> [BLOCK_D, BLOCK_H]) -----
        w_left      = tl.load(
            left_w_ptr + offs_d[:, None] + offs_h[None, :] * D,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                                           # [BLOCK_D, BLOCK_H]

        w_right     = tl.load(
            right_w_ptr + offs_d[:, None] + offs_h[None, :] * D,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        w_left_gate = tl.load(
            left_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * D,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        w_right_gate= tl.load(
            right_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * D,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        w_out_gate  = tl.load(
            out_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * D,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- dot‑products over the current d‑tile -----
        left_acc      += tl.dot(x_norm, w_left)       # [BLOCK_R, BLOCK_H]
        right_acc     += tl.dot(x_norm, w_right)
        left_gate_acc += tl.dot(x_norm, w_left_gate)
        right_gate_acc+= tl.dot(x_norm, w_right_gate)
        out_gate_acc  += tl.dot(x_norm, w_out_gate)

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask (if any) and the three sigmoids
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # sigmoid in FP32 → cast back to FP16
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc ))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc  ))).to(tl.float16)

    # convert the four linear results to FP16
    left_raw  = left_acc .to(tl.float16)
    right_raw = right_acc.to(tl.float16)

    # mask is applied *before* the sigmoid‑gating
    left  = left_raw  * mask_val * left_gate
    right = right_raw * mask_val * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Write results with layout [B, H, N, N]
    # ------------------------------------------------------------------
    nn = N * N                                           # number of (i,k) pairs
    b  = offs_r // nn                                    # batch index
    pos= offs_r - b * nn                                 # i·N + k   (0 ≤ pos < nn)

    # destination offsets for each (row, hidden) pair
    dst = b[:, None] * (H * nn) + offs_h[None, :] * nn + pos[:, None]   # [BLOCK_R, BLOCK_H]

    tl.store(left_ptr      + dst, left,       mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr     + dst, right,      mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr  + dst, out_gate,   mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Row‑wise statistics of the contracted tensor (mean & rsqrt‑var)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows,   # R = B·N·N
    H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 3️⃣  Final LayerNorm + out_gate + projection (all fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # hidden values
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LN parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (x - μ)·rstd → normalize → affine → gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term   = norm * vals_gate                                 # [M,K]
        term_fp16 = term.to(tl.float16)                           # fp16 for dot

        # final projection weight tile
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint (forward only, no autograd)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 0️⃣  Gather weight tensors (make them contiguous & correct dtype)
    # --------------------------------------------------------------
    norm_w = w["norm.weight"]
    norm_b = w["norm.bias"]

    left_w      = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w     = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).contiguous()

    to_out_norm_w = w["to_out_norm.weight"]   # γ  FP32
    to_out_norm_b = w["to_out_norm.bias"]     # β  FP32
    to_out_w_T    = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim] FP16

    # --------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + 5 linear maps + mask + gates
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # launch parameters – product BLOCK_R·BLOCK_H ≤ 1024 for safety
    BLOCK_R = 32
    BLOCK_H = 32
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_R), triton.cdiv(H, BLOCK_H))
    fused_norm_proj_gate_kernel[grid](
        x_flat,
        mask_ptr,
        norm_w,
        norm_b,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        left,
        right,
        out_gate,
        R,
        dim,
        H,
        N,
        HAS_MASK=HAS_MASK,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣  Core einsum via batched GEMM (left × rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                 # H ≤ 128 ⇒ single tile passes
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 4️⃣  Second LayerNorm + out_gate + final projection (fused)
    # --------------------------------------------------------------
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 64    # hidden‑dim tile (2 tiles for H=128)
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
