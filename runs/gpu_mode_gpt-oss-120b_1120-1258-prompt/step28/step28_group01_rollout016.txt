"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm (mixed‑precision, no gradients)
---------------------------------------------------------------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM (cuBLAS) → tensor (R, 5·H) where
   R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply the optional mask and sigmoid
   gates – all via vectorised torch kernels (no extra Triton kernel needed).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).
5️⃣ Row‑wise LayerNorm over the hidden dimension (H) using torch.mean/var.
6️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection (`to_out`) into a single torch linear call.
   The final output has shape `[B, N, N, dim]` (float32).

Only the first LayerNorm is written as a Triton kernel; all other
operations use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim]  FP32 input
    y_ptr,       # [R, dim]  FP16 output
    w_ptr,       # [dim]    LN weight γ (FP32)
    b_ptr,       # [dim]    LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                * "norm.weight" , "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5 × hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H)  FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid‑gates (pure torch, no extra kernel)
    # --------------------------------------------------------------
    # reshape to (R, 5, H)
    proj = proj.view(R, 5, H)

    # separate the five streams
    left_raw      = proj[:, 0, :]   # (R, H)
    right_raw     = proj[:, 1, :]   # (R, H)
    left_gate_raw = proj[:, 2, :]   # (R, H)
    right_gate_raw= proj[:, 3, :]   # (R, H)
    out_gate_raw  = proj[:, 4, :]   # (R, H)

    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_fp = mask.to(torch.float16).reshape(R, 1)   # (R,1)
        left_raw  = left_raw  * mask_fp
        right_raw = right_raw * mask_fp
    # otherwise mask is effectively 1

    # sigmoid gates (computed in FP32 for stability)
    left_gate  = torch.sigmoid(left_gate_raw.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.float()).to(torch.float16)

    # apply gates
    left    = left_raw  * left_gate   # (R, H)
    right   = right_raw * right_gate  # (R, H)
    # out_gate stays as‑is (no mask applied)

    # clean up large intermediate
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B, H, N, N)
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] FP16
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] FP16
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] FP16

    # merge batch & hidden for bmm
    left_mat  = left .view(B * H, N, N)     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # core contraction: (N×N)·(N×N)^T  →  (N×N) for each hidden slice
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over H (torch.mean / torch.var)
    # --------------------------------------------------------------
    # cast to FP32 for statistics (more stable, negligible cost)
    out_fp32 = out.float()                         # [B, H, N, N] FP32

    mu   = out_fp32.mean(dim=1, keepdim=True)      # [B,1,N,N] FP32
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N] FP32
    rstd = torch.rsqrt(var + eps)                 # [B,1,N,N] FP32

    # LayerNorm parameters (γ,β) – shape [H]
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1] FP32
    beta  = w["to_out_norm.bias"  ].view(1, H, 1, 1)   # [1,H,1,1] FP32

    # Normalise, apply γ/β and out‑gate (all in FP32)
    norm   = (out_fp32 - mu) * rstd               # [B,H,N,N]
    norm   = norm * gamma + beta                  # [B,H,N,N]
    gated  = norm * out_gate.float()              # [B,H,N,N]

    # --------------------------------------------------------------
    # 6️⃣ Final projection (H → dim) – fused into a single GEMM
    # --------------------------------------------------------------
    # reshape to (B·N·N, H)
    gated_flat = gated.permute(0, 2, 3, 1).reshape(B * N * N, H)   # [R, H] FP32

    # final linear: weight shape [dim, H]; use FP16 weight for tensor‑cores
    w_out = w["to_out.weight"].to(torch.float16)                   # [dim, H]

    # torch.nn.functional.linear expects weight in (out_features, in_features)
    final_fp16 = F.linear(gated_flat.half(), w_out)                # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)        # [B,N,N,dim] FP32

    return final
```
