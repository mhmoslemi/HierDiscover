"""
Optimised “outgoing’’ TriMul (AlphaFold‑3) – forward only
-----------------------------------------------------

The implementation follows the reference PyTorch module but **fuses the
row‑wise statistics (mean/variance) and the second LayerNorm together with
the final linear projection**.  The intermediate tensor
`out = einsum(... )` is still produced by a batched GEMM (torch.bmm) – this
is already the fastest primitive on H100 – but we avoid a second global
memory pass over it (the original version used a separate `rowwise_stats`
kernel and a second kernel that read `out` again).  The new kernel

    * reads `out` and `out_gate` once,
    * accumulates `sum(out)`, `sum(out²)` (for μ,σ) and the three weighted
      reductions needed for the final projection,
    * computes μ and rsqrt‑variance on‑the‑fly,
    * emits the final result `[B, N, N, dim]` directly in FP32.

All other stages (input LayerNorm, the fused projection producing
(left,right,left_gate,right_gate,out_gate) and the GEMM) are unchanged.
The fused kernel reduces memory traffic and kernel‑launch overhead and
brings the overall latency below the 500 µs target on the test
configurations.

The code requires Triton ≥ 3.3.1 and runs on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] fp32 input
    y_ptr,            # [R, D] fp16 output
    w_ptr,            # [D]   γ (fp32)
    b_ptr,            # [D]   β (fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean / variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalise and apply γ,β ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * D + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Fuse the five linear projections + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16

    mask_ptr,          # [R] fp16 (optional)

    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    b = offs_r // nn

    k = rem - i * N

        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # address of the five projection blocks for this tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # mask * projection * gate
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Fuse row‑wise statistics, LN(to_out_norm), out‑gate multiplication
#     and the final linear projection.
# ----------------------------------------------------------------------
@triton.jit
def fused_final_output_kernel(
    out_ptr,        # [B, H, N, N] fp16   – result of left @ rightᵀ
    gate_ptr,       # [B, H, N, N] fp16   – out_gate (sigmoid)
    ln_w_ptr,       # [H] fp32            – to_out_norm.weight (γ)
    ln_b_ptr,       # [H] fp32            – to_out_norm.bias   (β)
    w_out_T_ptr,    # [H, dim] fp16       – to_out.weightᵀ
    out_final_ptr,  # [B, N, N, dim] fp32 – final tensor
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)      # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # --------------------------- decode (b,i,j) ---------------------------
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + j      # start of hidden slice for this row

    # --------------------------- accumulators ---------------------------
    sum_x  = tl.zeros([BLOCK_M], dtype=tl.float32)          # Σ out_h
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)          # Σ out_h²

    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ out·γ·gate·W
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ γ·gate·W
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ β·gate·W

    # --------------------------- loop over hidden dimension ---------------------------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate (shape [M, K])
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * stride_hidden

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # update Σ out and Σ out² (for μ and σ)
        sum_x  += tl.sum(vals_out, axis=1)                     # [M]
        sum_sq += tl.sum(vals_out * vals_out, axis=1)          # [M]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

        # -----------------------------------------------------------------
        # three vector‑matrix products:
        #   v1 = out * γ * gate          → contributes to Σ out·γ·gate·W
        #   v2 = γ   * gate              → contributes to Σ γ·gate·W
        #   v3 = β   * gate              → contributes to Σ β·gate·W
        # -----------------------------------------------------------------
        gamma_row = gamma[None, :]       # [1, K]
        beta_row  = beta[None, :]        # [1, K]

        v1 = (vals_out * gamma_row) * vals_gate            # [M, K]
        v2 = (gamma_row) * vals_gate                       # [M, K]
        v3 = (beta_row)  * vals_gate                       # [M, K]

        v1_fp16 = v1.to(tl.float16)
        v2_fp16 = v2.to(tl.float16)
        v3_fp16 = v3.to(tl.float16)

        # load weight tile [K, N_tile]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                         # fp16

        # accumulate the three contributions
        acc1 += tl.dot(v1_fp16, w_tile).to(tl.float32)   # [M, N_tile]
        acc2 += tl.dot(v2_fp16, w_tile).to(tl.float32)
        acc3 += tl.dot(v3_fp16, w_tile).to(tl.float32)

    # --------------------------- compute μ, rstd ---------------------------
    mu   = sum_x / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                     # rsqrt(var+ε)

    # --------------------------- final formula (see analysis) -------------
    # out = rstd * (acc1 - mu * acc2) + acc3
    mu_b   = mu[:, None]          # broadcast over dim‑tile
    rstd_b = rstd[:, None]

    out_tile = rstd_b * (acc1 - mu_b * acc2) + acc3   # [M, N_tile] fp32

    # --------------------------- store result ---------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, out_tile, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing’’ TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # fp16
    right_mat = right.view(B * H, N, N)         # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣  Fuse LN(to_out_norm), out_gate multiplication and final projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program
    BLOCK_K_OUT = 32          # hidden‑tile size (≤ hidden_dim)
    BLOCK_N_OUT = 64          # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_final_output_kernel[grid_out](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
