    left_mat = left.view(B*H, N, N)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.
================================================================
Algorithm (all heavy Tensor cores use FP16, statistics use FP32):
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a single
    GEMM → tensor [R, 5·H]   (R = B·N·N, H = hidden_dim).
3️⃣  Split the fused projection, apply the optional pair‑mask and the three sigmoid gates
    (left, right, out_gate) with a tiny Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   →   [B, H, N, N] (FP16).
5️⃣  Compute per‑row mean & variance with torch.var_mean (GPU‑accelerated reduction).
6️⃣  Apply the second LayerNorm, multiply by out_gate and perform the final projection
    (norm·gate)·W_out using a single FP16 GEMM (torch.nn.functional.linear).
All kernels use mixed‑precision (FP32 for reductions, FP16 for bulk work) and
run on an H100.  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]         FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows processed per program
    BLOCK_H: tl.constexpr,   # hidden dimension processed per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # Offsets for the hidden dimension stride N·N
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside fused projection: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five fused linear results
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (performed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all stored as float32 tensors)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    # torch.nn.functional.layer_norm is highly‑optimized on GPU
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                    # [R, dim] FP16
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # (5·H, dim)
    proj = F.linear(x_norm, fused_w)                        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask: [B, N, N] → flat FP16, broadcast over hidden dim
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden dim tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are currently [B, H, N, N]; reshape to (B·H, N, N) for bmm
    left_mat  = left.view(B * H, N, N)            # FP16
    right_mat = right.view(B * H, N, N)           # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                    # [B, H, N, N] FP16
    # Release left/right as they are no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Prepare for per‑row LayerNorm & final projection
    # ------------------------------------------------------------------
    # Bring hidden dimension to the innermost position for efficient reductions
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H)                     # [R, H] FP16

    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    gate_flat = out_gate.view(R, H)                         # [R, H] FP16
    del out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Row‑wise mean & variance (FP32) – use torch.var_mean which fuses the two reductions
    # ------------------------------------------------------------------
    out_fp32 = out_flat.float()                     # [R, H] FP32
    var, mu = torch.var_mean(out_fp32, dim=1, unbiased=False)   # both [R]
    rstd = torch.rsqrt(var + eps)                   # [R]

    # ------------------------------------------------------------------
    # 7️⃣ Apply second LayerNorm, multiply by out_gate and project
    # ------------------------------------------------------------------
    # Cast statistics & LayerNorm params to FP16 for the element‑wise pass
    mu_fp16   = mu.to(torch.float16).unsqueeze(1)    # [R,1]
    rstd_fp16 = rstd.to(torch.float16).unsqueeze(1)# [R,1]

    gamma = w["to_out_norm.weight"].to(torch.float16).unsqueeze(0)   # [1, H]
    beta  = w["to_out_norm.bias"  ].to(torch.float16).unsqueeze(0)   # [1, H]

    # Normalisation → γ· + β → gate
    norm = (out_flat - mu_fp16) * rstd_fp16          # [R, H] FP16
    norm = norm * gamma + beta                      # [R, H] FP16
    norm = norm * gate_flat                         # [R, H] FP16
    del out_flat, gate_flat, mu_fp16, rstd_fp16, gamma, beta

    # Final linear projection (FP16 GEMM → FP32 result)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)          # [dim, H]
    final_fp16 = F.linear(norm, w_out_fp16)                    # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)    # [B, N, N, dim]

    return final
