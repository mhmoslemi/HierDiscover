@triton.jit
def norm_gate_kernel(
    out_ptr,         # [R, H] FP16 raw out
    out_gate_ptr,    # [R, H] FP16
    ln_w_ptr,        # [H] FP32
    ln_b_ptr,        # [H] FP32
    out_norm_gate_ptr, # [R, H] FP16
    R, H, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < R

    # Phase 1: compute per-row sum and sum_sq for mean and var.
    # We'll accumulate across hidden dim in blocks.

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # load block of out values
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # shape [M, BH]

        sum_val += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)   # [M]

    # Phase 2: compute normalized + gated and store
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # shape [M, BH]

        gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]

        # broadcast mu, rstd
        mu = mean[:, None]
        r = rstd[:, None]

        # Compute: ((out - mu) * r) * gamma + beta   [M, BH]
        # Wait we want LN: (out - mu) * rstd * gamma + beta
        # Then out_gate multiplication
        normalized = ((out_vals - mu) * r) * gamma[None, :] + beta[None, :]
        gated = normalized * gate_vals

        # store as fp16
        tl.store(out_norm_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

@triton.jit
def norm_gate_kernel(
    out_ptr,      # [R, H] FP16 raw out
    gate_ptr,     # [R, H] FP16 out_gate
    ln_w_ptr,     # [H] FP32
    ln_b_ptr,     # [H] FP32
    out_ptr_norm, # [R, H] FP16 result
    R, H,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # first pass compute mean & rstd per row
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < R

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over H blocks
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)   # [BM, BH]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)   # [BM]

    # second pass: compute normalized, apply LN weight/bias and out_gate
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # [BM, BH]

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]

        # broadcast mean/rstd
        mu = mean[:, None]   # [BM,1]
        rs = rstd[:, None]   # [BM,1]

        # LayerNorm + out_gate
        norm = ((out_vals - mu) * rs) * gamma[None, :] + beta[None, :]   # [BM,BH]
        gated = norm * gate_vals

        tl.store(out_ptr_norm + offs_m[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start offset for h==0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [R, 5·H] (contiguous)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate & store result
# --------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,          # [R, H]   FP16  (raw einsum result)
    gate_ptr,         # [R, H]   FP16  (out_gate)
    ln_w_ptr,         # [H]      FP32  (to_out_norm.weight)
    ln_b_ptr,         # [H]      FP32  (to_out_norm.bias)
    out_norm_ptr,     # [R, H]   FP16  (output of LN + gate)
    R, H,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    # ----------- first pass : mean & variance ------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(
            out_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                                    # [M, BH]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                               # [M]

    # ----------- second pass : LN + out_gate ------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(
            out_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                                    # [M, BH]

        gate_vals = tl.load(
            gate_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]

        mu = mean[:, None]            # [M,1]
        rs = rstd[:, None]            # [M,1]

        # LayerNorm: ((out - μ) * rstd) * γ + β
        norm = ((out_vals - mu) * rs) * gamma[None, :] + beta[None, :]

        gated = norm * gate_vals

        tl.store(
            out_norm_ptr + offs_m[:, None] * H + offs_h[None, :],
            gated.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = inp.shape
    R = B * N * N               # flattened pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.reshape(R, dim)                                 # [R, dim] FP32
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                            # FP32
    x_norm = x_norm_fp32.to(torch.float16)                       # FP16
    del x_flat, x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣  Fuse 5 linear maps (left, right, gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                             # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣  Split, apply mask and sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    right = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, H, N, N), dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Core einsum via batched GEMM (fp16, tensor‑cores)
    # --------------------------------------------------------------
    # view as [B·H, N, N]   (contiguous per hidden slice)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                                    # [B, H, N, N] fp16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣  Row‑wise LayerNorm + out_gate (fused) → hidden_norm [R, H]
    # --------------------------------------------------------------
    # bring hidden dimension to the last axis to get a flat [R, H] layout
    out = out.permute(0, 2, 3, 1).contiguous()                    # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()          # same layout
    out_flat = out.view(R, H)
    out_gate_flat = out_gate.view(R, H)

    hidden_norm = torch.empty_like(out_flat)                      # FP16

    BLOCK_M = 128
    BLOCK_H = 64
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    norm_gate_kernel[grid_norm](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        hidden_norm,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del out, out_gate, out_flat, out_gate_flat

    # --------------------------------------------------------------
    # 6️⃣  Final projection (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16)            # (dim, H)
    final_fp16 = F.linear(hidden_norm, w_out_fp16)               # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)      # FP32

    return final
