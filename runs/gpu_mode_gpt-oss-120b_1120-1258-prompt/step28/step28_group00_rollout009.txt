"""
TriMul “outgoing” forward pass – highly‑fused Triton implementation.
===================================================================

The reference PyTorch module performs:

1️⃣  LayerNorm on the flattened pairwise tensor `[B,N,N,Dim]`.
2️⃣  Five linear projections (Left, Right, Left‑gate, Right‑gate, Out‑gate)
    – all sharing the same weight matrix.
3️⃣  Mask & sigmoid gating on Left/Right and Out‑gate.
4️⃣  Batched GEMM  (Left · Rightᵀ)   →  tensor of shape `[B, H, N, N]`.
5️⃣  Per‑row LayerNorm over the hidden dimension `H`,
    multiplied by the Out‑gate,
    followed by a final linear projection back to `Dim`.

The heavy parts are the three LayerNorms and the two GEMMs.
The implementation below **fuses** most of the work in Triton kernels:
* `layernorm_kernel` – row‑wise LayerNorm (FP32 → FP16).
* `split_fuse_gate_kernel` – one‑shot projection, mask application and sigmoid.
* `final_norm_proj_kernel` – computes the *second* LayerNorm,
  multiplies by the Out‑gate and finishes with the final projection
  (no intermediate “mu/rstd” tensors are stored).

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Mixed‑precision (FP16 for the bulk of the work, FP32 for reductions)
keeps numerical stability while fully exploiting Tensor Cores.

The public entry point is `custom_kernel(data)` with the signature required
by the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,        # [R, D] input FP32, output FP16
    w_ptr, b_ptr,        # LayerNorm γ, β  (FP32)  shape [D]
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * D + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Projection + mask + sigmoid gating (single fused kernel)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,               # [R, 5·H]   FP16 fused projection
    left_ptr, right_ptr,    # [B, H, N, N]  FP16
    out_gate_ptr,           # [B, H, N, N]  FP16
    mask_ptr,               # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r,
                       mask=mask_r,
                       other=1.0)                     # [BLOCK_R]
        mask_val = mval[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                      # distance between two hidden slices
    base_offset = b * (H * stride_hidden) + i * N + k   # start for hidden==0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [R, 5·H] → stride = 5·H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 for stability, then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # store (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Fuse second LayerNorm, Out‑gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    out_ptr,        # [B, H, N, N]  FP16   (contracted tensor)
    gate_ptr,       # [B, H, N, N]  FP16   (out_gate)
    ln_w_ptr, ln_b_ptr,                     # to_out_norm weight / bias  (FP32)  shape [H]
    w_out_T_ptr,                            # to_out weight transposed  (FP16)  shape [H, Dim]
    out_final_ptr,                          # [B, N, N, Dim]   FP32 result
    total_rows, H, N, Dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """ 
    For each (b,i,j) pair:
        μ, σ²  ←  mean & variance of out over hidden dim H
        out̂   ←  (out - μ) * rsqrt(σ²+eps) * γ + β          (γ,β = LN params)
        out̂   ←  out̂ * out_gate
        out    ←  out̂ @ W_out   (W_out : H → Dim)
    """

    pid_m = tl.program_id(0)   # rows (R = B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < Dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + j   # start of hidden slice for this (b,i,j)

    # ------------------------------------------------------------------
    # Phase‑1 : compute mean & variance over hidden dim (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Phase‑2 : normalized out, gate, and final projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        # -------- load raw tensors ----------
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * stride_hidden

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M,K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M,K]

        # -------- LayerNorm parameters (γ,β) ----------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  FP32

        # ------------ normalize & apply out_gate --------------
        diff = (out_vals - mean[:, None]) * rstd[:, None]          # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M,K]
        term = norm * gate_vals                                    # [M,K]
        term_fp16 = term.to(tl.float16)                           # FP16 for TensorCore

        # ------------ dot with final weight matrix ------------
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * Dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K,N_tile]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)           # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Store final result (flattened view of [B, N, N, Dim])
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * Dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, Dim]   (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]       (bool/float) or None
        - weights_dict : dict of model parameters (torch.Tensor)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, Dim]  (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, Dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)  shape [Dim]
        weights["norm.bias"],     # β (fp32)  shape [Dim]
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣  Fused projection + mask + sigmoid gates
    # --------------------------------------------------------------
    # Stack the five projection matrices once (FP16)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],      # (H, Dim)
            weights["right_proj.weight"],     # (H, Dim)
            weights["left_gate.weight"],      # (H, Dim)
            weights["right_gate.weight"],     # (H, Dim)
            weights["out_gate.weight"],       # (H, Dim)
        ],
        dim=0,
    ).to(torch.float16).contiguous()   # shape [5·H, Dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H)   FP16
    del fused_w, x_norm

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # flatten mask to [R] and cast to FP16 for the kernel
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – Triton still expects a pointer
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣  Core einsum  (batched GEMM)  →  out = left @ rightᵀ
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] and use cuBLAS (torch.bmm) which runs on Tensor Cores
    left_mat  = left.view(B * H, N, N)                 # fp16
    right_mat = right.view(B * H, N, N)                # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16

    # --------------------------------------------------------------
    # 4️⃣  Fuse second LayerNorm, out_gate multiplication and final linear projection
    # --------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, Dim]  fp16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128                     # rows per program (R‑tiles)
    BLOCK_K_OUT = 128                     # process the whole hidden dim in one go
    BLOCK_N_OUT = 128                     # output‑dim tile
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_proj_kernel[grid_out](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ (fp32) shape [H]
        weights["to_out_norm.bias"],     # β (fp32) shape [H]
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # --------------------------------------------------------------
    return final
```
