@triton.jit
def norm_gate_kernel(
    out_ptr,      # [B, H, N, N] (FP16)
    gate_ptr,    # same shape (FP16)
    mu_ptr,  # not needed if compute on the fly?
    rstd_ptr, # not needed if compute on the fly?
    ln_w_ptr,   # [H] (FP32)
    ln_b_ptr,   # [H] (FP32)
    out_flat_ptr, # [R, H] (FP16) where R = B*N*N
    total_rows, H, N,
    eps: float,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # rows
    mask_m = offs_m < total_rows

    # compute hidden stride
    hidden_stride = N * N

    # decode (b, i, j) as before
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base pointer offset for h=0 index
    base = b * (H * hidden_stride) + i * N + j   # offset for first hidden

    # sum and sumsq across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # We'll load hidden dimension using stride hidden_stride. We could use loop over k in steps of BLOCK_K.
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate for this slice
        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals_out, axis=1)  # sum across hidden dimension? Wait sum across hidden dimension for each row: we need sum of out across channels, not gating. Actually we need mean of out before normalization, not gating; So we need to compute sum of out across H: That we do with vals_out.

        sum_sq += tl.sum(vals_out * vals_out, axis=1)

    # compute mean, var, rstd
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute normalized_out * out_gate * gamma + beta, write to flat out
    # out_flat layout: each row r has H entries contiguous; need offset r*H + h
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate
        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load layernorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # FP32
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # normalized
        norm = (vals_out - mean[:, None]) * rstd[:, None]

        norm = norm * gamma[None, :] + beta[None, :]

        # gating
        out_val = norm * vals_gate

        # store to flat out
        # compute dest offset: offs_m * H + offs_k
        dst_ptr = out_flat_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, out_val.to(tl.float16), mask=mask_m[:, None] & mask_k[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) – Triton.
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
   single GEMM (FP16) → (R, 5·H)  (R = B·N·N, H = hidden_dim).
3️⃣ Split the fused projection, apply optional mask and three sigmoid gates – Triton.
4️⃣ Core bilinear “einsum” via a batched GEMM (cuBLAS):
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → FP16.
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and flattening into one
   Triton kernel.  For each pair token (b,i,j) we compute the mean/variance over
   the hidden dimension, apply η·γ+β, multiply by out_gate, and write the result
   into a flat buffer `[R, H]`.
6️⃣ A final GEMM (cuBLAS, FP16) projects the normalized gated tensor to the
   original channel dimension:  (R, H) × (H, dim) → (R, dim).
7️⃣ Reshape to `[B, N, N, dim]` and cast to float32.

All kernels run on H100 (Triton 3.3.1).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm  (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]    FP32 input
    y_ptr,    # [R, dim]    FP16 output
    w_ptr,    # [dim]       γ (FP32)
    b_ptr,    # [dim]       β (FP32)
    rows,     # scalar R
    dim,      # scalar dim
    eps,      # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass : compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass : write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]   FP16
    right_ptr,     # [B, H, N, N]   FP16
    out_gate_ptr,  # [B, H, N, N]   FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start offset for h==0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate and flatten into [R, H]
# --------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,        # [B, H, N, N]   FP16   (bilinear result)
    gate_ptr,       # [B, H, N, N]   FP16   (out_gate)
    ln_w_ptr,       # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H]            FP32   (to_out_norm.bias)
    out_flat_ptr,   # [R, H]         FP16   (output buffer)
    total_rows, H, N,
    eps,            # scalar
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # offset of hidden‑plane 0

    # -------- first pass : compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr_out,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass : normalize, apply gate and write ----------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]  FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]  FP32

        # LayerNorm: (x-µ)*rstd*γ + β
        norm = (vals_out - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply out‑gate
        out_val = norm * vals_gate                               # FP32

        # Write into flat buffer [R, H] as FP16
        dst_ptr = out_flat_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr,
                 out_val.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N          # flattened pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣  Fuse 5 linear maps (left, right, gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H)   FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣  Split, apply mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)           # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left, right   # free memory

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm + out_gate, flatten to [R, H]
    # --------------------------------------------------------------
    out_norm_gate_flat = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    grid_norm = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        out_norm_gate_flat,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del out, out_gate   # free

    # --------------------------------------------------------------
    # 6️⃣  Final projection (FP16 GEMM) → [R, dim]
    # --------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16).contiguous()   # [dim, H]
    final_flat = F.linear(out_norm_gate_flat, w_out_fp16)            # [R, dim] FP16
    del out_norm_gate_flat, w_out_fp16

    # --------------------------------------------------------------
    # 7️⃣  Reshape and cast to float32
    # --------------------------------------------------------------
    final = final_flat.to(torch.float32).view(B, N, N, dim)
    return final
