"""
TriMul (outgoing) – highly‑fused forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (dim) using a small Triton
    kernel (FP32 → FP16).  This implements the “norm” sub‑layer.

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (FP16) with weight shape (5·H, dim).

3️⃣  Split the fused projection, apply optional pair mask and three sigmoid
    gates (pure PyTorch – cheap elementwise ops).

4️⃣  Core “einsum” is performed as a batched GEMM:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    using torch.bmm on tensors of shape (B·H, N, N).

5️⃣  LayerNorm over the hidden dimension H of the contracted tensor,
    multiply by the out‑gate, and project back to the original dim with a
    single linear projection (FP16).  The final tensor is cast to float32.

All heavy reductions are done in the Triton kernel (step 1).  The rest of
the pipeline uses highly‑optimized PyTorch kernels (linear, bmm, layer_norm,
sigmoid).  This balance yields a fast implementation while still meeting
the requirement of having a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, D] FP32 input
    y_ptr,      # [R, D] FP16 output
    w_ptr,      # [D]   gamma (FP32)
    b_ptr,      # [D]   beta  (FP32)
    rows,       # scalar R
    dim,        # scalar D
    eps,        # scalar epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance per row -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile: stride = dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values with affine transform -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]    # [BLOCK_M, BLOCK_D]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]      # FP32 → FP16 store
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape                # x: [B, N, N, dim]
    H = cfg["hidden_dim"]                 # hidden dimension (called hidden_dim in config)
    nomask = bool(cfg.get("nomask", True))

    R = B * N * N                         # total number of pair tokens

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()                       # FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 64    # columns per tile (covers dim = 128 or 384)
    grid_norm = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (FP16)
    # --------------------------------------------------------------
    # Stack the five projection weights: each [H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H)   FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    # reshape to (R,5,H) for easy slicing
    proj5 = proj.view(R, 5, H)                     # [R,5,H]

    left_raw      = proj5[:, 0, :]                 # [R,H]
    right_raw     = proj5[:, 1, :]                 # [R,H]
    left_gate_raw = proj5[:, 2, :]                 # [R,H]
    right_gate_raw= proj5[:, 3, :]                 # [R,H]
    out_gate_raw  = proj5[:, 4, :]                 # [R,H]

    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R, 1)   # [R,1], 0/1
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat
    # else: mask is all‑ones => no multiplication needed

    # sigmoid gates (computed in FP32 for stability)
    left_gate  = torch.sigmoid(left_gate_raw.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.float()).to(torch.float16)

    left  = left_raw  * left_gate
    right = right_raw * right_gate
    # out_gate stays as is (already sigmoided)

    # reshape to [B,N,N,H] and permute to [B,H,N,N] for the batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (FP16)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16

    # out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H,N,N] FP16
    out = out_mat.view(B, H, N, N)                              # [B,H,N,N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H, gate, final projection
    # --------------------------------------------------------------
    # bring tensors to [B,N,N,H] layout for layer_norm
    out = out.permute(0, 2, 3, 1).contiguous()          # [B,N,N,H] FP16
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()# [B,N,N,H] FP16

    # LayerNorm over last dimension H (torch's implementation is highly optimized)
    out_ln = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # returns same dtype (FP16)

    # Apply out‑gate
    out_ln_gate = out_ln * out_gate                     # [B,N,N,H] FP16

    # Final linear projection: (H → dim)
    # reshape to (R, H) where R = B·N·N
    out_ln_gate_flat = out_ln_gate.view(R, H)           # FP16
    w_out = w["to_out.weight"].to(torch.float16)       # [dim, H]

    out_final = F.linear(out_ln_gate_flat, w_out)      # (R, dim) FP16
    out_final = out_final.view(B, N, N, dim).float()   # cast back to float32

    return out_final
