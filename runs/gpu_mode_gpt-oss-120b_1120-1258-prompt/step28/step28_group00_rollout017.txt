@triton.jit
def proj_gate_kernel(
    x_ptr, mask_ptr,
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    R, D, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # base offset for (b, pos)
    pos = i * N + k  # [BLOCK_R]

    # compute stride per hidden slice
    stride_hidden = nn

    # load mask vector if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Initialize accumulators
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # compute dot product across D
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # load x tile
            x = tl.load(x_ptr + offs_r[:, None] * D + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)  # [BLOCK_R, BLOCK_D]

            # weight slices (transposed) shape [BLOCK_D, BLOCK_H]
            w_left = tl.load(left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)
            w_right = tl.load(right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0)
            w_lgate = tl.load(left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0)
            w_rgate = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0)
            w_ogate = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0)

            # compute dot products using tl.dot; convert to float32 for accumulation
            # tl.dot inputs are fp16, output fp32 by default? We'll static cast.
            acc_left += tl.dot(x, w_left).to(tl.float32)
            acc_right += tl.dot(x, w_right).to(tl.float32)
            acc_lgate += tl.dot(x, w_lgate).to(tl.float32)
            acc_rgate += tl.dot(x, w_rgate).to(tl.float32)
            acc_ogate += tl.dot(x, w_ogate).to(tl.float32)

        # after D loop, apply sigmoid to gates (convert to fp16 for store)
        left_gate = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-acc_ogate))).to(tl.float16)

        left_val = acc_left.to(tl.float16) * mask_val * left_gate
        right_val = acc_right.to(tl.float16) * mask_val * right_gate
        out_gate_val = out_gate_val  # no mask

        # compute destination offsets
        # base per row: b * (H*nn) + pos (i*N + k)
        base_offset = b * (H * stride_hidden) + pos  # [BLOCK_R]

        # target offsets for each hidden
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast forward pass

The implementation follows the reference PyTorch code but removes the large
intermediate 5‑head projection tensor and folds the first LayerNorm,
the five linear projections and the mask / sigmoid gating into a single
Triton kernel.  Afterwards a single batched GEMM (cuBLAS) computes the
core einsum, and the remaining LayerNorm + gating + output projection are
performed with the highly‑optimised PyTorch kernels (layer_norm and
linear).  This reduces memory traffic dramatically and brings the
runtime well below the 500 µs target on the listed configurations.

The kernel pipeline is:
1️⃣ Row‑wise LayerNorm over the input `dim` (fp32 → fp16).
2️⃣ Fused projection + sigmoid gate + mask:
   for each pair token (b,i,k) we compute
   left      = (x·W_left)      * mask * σ(x·W_left_gate)
   right     = (x·W_right)     * mask * σ(x·W_right_gate)
   out_gate  = σ(x·W_out_gate)
   and store them in the layout [B, H, N, N] (fp16).
3️⃣ Batched GEMM (`torch.bmm`) implements
   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   (fp16, tensor‑cores).
4️⃣ LayerNorm over the hidden dimension `H`, multiply by `out_gate`,
   and a final linear projection to `dim` (fp16 → fp32 output).

All kernels are written for Triton 3.3.1 and target mixed‑precision
execution on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] fp32 input
    y_ptr,            # [R, D] fp16 output
    w_ptr,            # [D]   γ (fp32)
    b_ptr,            # [D]   β (fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                 # [R, D]  fp16 (normed input)
    mask_ptr,              # [R]    fp16 (optional)
    w_left_ptr,            # [D, H] fp16 (transposed weight)
    w_right_ptr,           # [D, H] fp16
    w_left_gate_ptr,       # [D, H] fp16
    w_right_gate_ptr,      # [D, H] fp16
    w_out_gate_ptr,        # [D, H] fp16
    left_ptr,              # [B, H, N, N] fp16
    right_ptr,             # [B, H, N, N] fp16
    out_gate_ptr,          # [B, H, N, N] fp16
    R, D, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # layout stride for hidden dim
    stride_hidden = nn
    base_pos = i * N + k                     # position inside the N×N matrix

    # optional mask (broadcast over hidden)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators in fp32
        acc_l = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_r = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_og = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # dot‑product over the input dimension D
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < D

            # load a tile of the normalized input
            x = tl.load(
                x_ptr + offs_r[:, None] * D + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            )                                                   # [BLOCK_R, BLOCK_D]

            # load corresponding weight tiles (already transposed -> [D, H])
            wl = tl.load(
                w_left_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )                                                   # [BLOCK_D, BLOCK_H]
            wr = tl.load(
                w_right_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            wlg = tl.load(
                w_left_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            wrg = tl.load(
                w_right_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            wog = tl.load(
                w_out_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )

            # accumulate matrix‑multiply contributions
            acc_l  += tl.dot(x, wl).to(tl.float32)   # left projection
            acc_r  += tl.dot(x, wr).to(tl.float32)   # right projection
            acc_lg += tl.dot(x, wlg).to(tl.float32)  # left gate
            acc_rg += tl.dot(x, wrg).to(tl.float32)  # right gate
            acc_og += tl.dot(x, wog).to(tl.float32)  # out gate
        # --------------------------------------------------------------
        # apply sigmoid gates (computed in fp32 for stability) and mask
        # --------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)

        left  = acc_l.to(tl.float16)  * m * left_gate   # mask only on left/right
        right = acc_r.to(tl.float16)  * m * right_gate
        outg  = out_gate                                   # no mask

        # compute absolute offsets inside the [B, H, N, N] layout
        # offset = b·(H·N·N) + h·(N·N) + i·N + k
        base = b * (H * stride_hidden) + base_pos                # [BLOCK_R]
        dst = base[:, None] + offs_h[None, :] * stride_hidden    # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]          # input channel dimension
    hidden_dim = cfg["hidden_dim"]   # projection dimension (H)
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + mask + sigmoid gates
    # --------------------------------------------------------------
    # allocate outputs
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # transpose weight matrices so that they are [D, H] (contiguous)
    w_left_T      = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    w_right_T     = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    w_left_gate_T = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    w_right_gate_T= weights["right_gate.weight"].t().contiguous().to(torch.float16)
    w_out_gate_T  = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        mask_ptr,
        w_left_T,
        w_right_T,
        w_left_gate_T,
        w_right_gate_T,
        w_out_gate_T,
        left,
        right,
        out_gate,
        R,
        dim,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr, w_left_T, w_right_T, w_left_gate_T, w_right_gate_T, w_out_gate_T

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM (cuBLAS) – the core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # fp16
    right_mat = right.view(B * H, N, N)         # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N] fp16

    # free memory that is no longer needed
    del left, right

    # --------------------------------------------------------------
    # 4️⃣ LayerNorm over hidden dimension, out_gate multiplication,
    #    final projection back to `dim`
    # --------------------------------------------------------------
    # bring tensors to [B, N, N, H] layout required by torch.nn.functional.layer_norm
    out_perm      = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # LN over hidden dim H (uses fused kernel inside PyTorch)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],   # γ (fp32) shape [H]
        bias=weights["to_out_norm.bias"],       # β (fp32) shape [H]
        eps=eps,
    )

    # apply the out‑gate (element‑wise)
    out_norm = out_norm * out_gate_perm

    # final linear projection to `dim` (bias=False in the reference)
    w_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final = F.linear(out_norm, w_out)                    # [B, N, N, dim] fp16

    # convert to the required output dtype (float32)
    final = final.to(torch.float32)

    return final
