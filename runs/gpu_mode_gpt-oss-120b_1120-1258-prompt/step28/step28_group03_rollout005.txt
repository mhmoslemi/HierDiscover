"""
Outgoing TriMul (AlphaFold‑3) – forward‑only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) via a Triton
    kernel.  Statistics are accumulated in FP32, then the normalized values are
    written as FP16.

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into
    a single GEMM (FP16).  The result is reshaped to (R,5,hidden) and split with
    pure PyTorch code; this avoids extra kernel launches.

3️⃣  Apply the optional pair‑mask (broadcast over the hidden dimension) and the
    three sigmoid gates.  The tensors are then permuted to (B,hidden,N,N).

4️⃣  Batched GEMM (torch.bmm) implements the core einsum
        out[b,i,j,h] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →   (B,hidden,N,N) FP16.

5️⃣  Second LayerNorm (per‑row across the hidden dimension) + out‑gate
    multiplication.  This uses the highly‑optimised torch.nn.functional.layer_norm
    (operates in FP32).

6️⃣  Final projection back to the original dimensionality with a single linear
    layer (FP32).  The final tensor has shape (B,N,N,dim).

Only the first LayerNorm is written in Triton; the remaining operations rely on
fast PyTorch kernels.  All heavy arithmetic is performed in FP16, while
statistics are kept in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]     gamma (FP32)
    b_ptr,          # [dim]     beta  (FP32)
    rows,           # scalar R
    dim,            # scalar dim
    eps,            # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (FP32→FP16)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict: must contain "dim" (int), "hidden_dim" (int),
                optional "nomask" (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim                    # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # shape [5*H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5*H)  FP16
    del fused_w, x_norm

    # reshape to (R, 5, H) for easy slicing
    proj = proj.view(R, 5, H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_raw   = proj[..., 0, :]                     # (R, H)
    right_raw  = proj[..., 1, :]                     # (R, H)
    left_gate  = torch.sigmoid(left_raw.float()).to(torch.float16)   # (R, H)
    right_gate = torch.sigmoid(right_raw.float()).to(torch.float16)  # (R, H)
    out_gate   = torch.sigmoid(proj[..., 4, :].float()).to(torch.float16)  # (R, H)

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)   # (R,1)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # apply sigmoid gates
    left  = left_raw * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate, right_gate, out_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (core einsum)
    # ------------------------------------------------------------------
    # (B, H, N, N) → (B*H, N, N)
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm across the hidden dimension + out_gate
    # ------------------------------------------------------------------
    # bring hidden dim to the last axis to use torch.nn.functional.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous().float()       # (B, N, N, H) FP32
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous().float()  # (B, N, N, H) FP32
    del out, out_gate

    # LayerNorm over H with learned γ/β (to_out_norm)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # (B, N, N, H) FP32

    # Apply out‑gate (still in FP32)
    gated = out_norm * out_gate_perm                         # (B, N, N, H) FP32
    del out_norm, out_gate_perm

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    gated_flat = gated.view(B * N * N, H)                     # (B·N·N, H)
    final = F.linear(gated_flat, w["to_out.weight"])        # (B·N·N, dim) FP32
    final = final.view(B, N, N, dim)                         # (B, N, N, dim)

    return final
