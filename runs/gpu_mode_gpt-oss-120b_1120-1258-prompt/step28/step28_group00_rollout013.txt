def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat, x_norm,
        weights["norm.weight"], weights["norm.bias"],
        R, dim, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D, num_warps=8)

    # fused projection
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ], dim=0).to(torch.float16)

    proj = torch.nn.functional.linear(x_norm, fused_weight) # (R,5H) fp16

    # split & gating
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj, left, right, out_gate, mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8)

    # core einsum via batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    out = out.view(B, H, N, N)  # (B,H,N,N)

    # LayerNorm on out (over hidden dim)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R,H)
    out_norm = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_out_ln = (triton.cdiv(R, BLOCK_M),)

        R, H, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D, num_warps=8)

    out_norm_gate = out_norm * out_gate_flat

    final = torch.nn.functional.linear(out_norm_gate, w_out)  # (R, dim) fp16

    final = final.to(torch.float32).view(B, N, N, dim)

    return final

"""
TriMul “outgoing” forward pass – highly fused Triton implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (float32 → float16) on the flattened pairwise tensor.
2️⃣  One fused linear projection (5×hidden_dim) produces
    [left, right, left_gate, right_gate, out_gate] in a single matmul.
3️⃣  Split‑fuse kernel:
        * applies the optional mask (broadcast over the hidden dim),
        * evaluates sigmoid gates,
        * stores gated tensors `left`, `right` and the gate `out_gate`
          in layout [B, H, N, N] (fp16).
4️⃣  Core contraction – a batched GEMM (cuBLAS):
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k].
5️⃣  Row‑wise LayerNorm on `out` over the hidden dimension H (γ/β = to_out_norm).
6️⃣  Element‑wise multiplication with `out_gate`.
7️⃣  Final linear projection (to_out) using torch.nn.functional.linear
        (high‑performance Tensor‑Core GEMM).  The result is cast to float32.

Only the LayerNorm and the split‑gate steps are written in Triton; the
remaining dense linear algebra is delegated to cuBLAS which is optimal on
NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (used twice: on the input and on the contracted tensor)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D]   (fp32 or fp16)
    y_ptr,            # [R, D]   (fp16) -- normalized output
    w_ptr,            # [D]      (γ, fp32)
    b_ptr,            # [D]      (β, fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance (per row) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                         # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Split / fuse + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   (fp16)
    left_ptr,          # [B, H, N, N] (fp16)
    right_ptr,         # [B, H, N, N] (fp16)
    out_gate_ptr,      # [B, H, N, N] (fp16)
    mask_ptr,          # [R] (fp16) – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mval[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                         # size of one hidden slice

    # base offset for (b,i,k) at hidden‑index 0 → layout [B, H, N, N]
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # iterate over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # address of the five projection blocks for this hidden tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # load the five fused projection results
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # mask * gate * projection
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                 # not masked

        # destination offset for this hidden tile
        dst = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward only “outgoing” TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]          # input feature dimension C
    hidden_dim = cfg["hidden_dim"]   # hidden dimension H
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm on the input (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] (fp32)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M   = 128
    BLOCK_D   = 64
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)   shape [dim]
        weights["norm.bias"],     # β (fp32)   shape [dim]
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fused projection producing 5×hidden_dim channels
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                     # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Core contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)  fp16
    right_mat = right.view(B * H, N, N)         # (B·H, N, N)  fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)  fp16
    out = out.view(B, H, N, N)                              # (B, H, N, N) fp16

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise LayerNorm on the contracted tensor (over H)
    # ------------------------------------------------------------------
    # reshape to [R, H] where each row corresponds to (b,i,j)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) fp16
    out_norm = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_out_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_out_ln](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32) shape [H]
        weights["to_out_norm.bias"],     # β (fp32) shape [H]
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out

    # ------------------------------------------------------------------
    # 6️⃣  Apply out_gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) fp16
    gated = out_norm * out_gate_flat   # fp16 element‑wise product

    # final projection: hidden -> dim (Tensor‑Core GEMM)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    final_fp16 = F.linear(gated, to_out_w)                 # (R, dim) fp16

    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # cast to fp32

    # ------------------------------------------------------------------
    return final
```
