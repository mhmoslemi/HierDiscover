"""
Highly‑optimized forward‑only “outgoing” TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Pair‑wise LayerNorm (over the channel dimension) – use the
    native torch F.layer_norm (FP32 → FP16 conversion).

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into one matmul:       proj = x_norm @ W_fused.T   (R×5H, FP16).

3️⃣  Triton kernel `split_fuse_gate_kernel`:
      * decodes the flat pair index (b,i,k) → (b,h,i,k) layout,
      * applies optional mask,
      * computes sigmoid gates (FP32 → FP16),
      * writes three tensors `left`, `right`, `out_gate` in layout
        [B, H, N, N] (FP16).

4️⃣  Core “einsum” is a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
      performed by `torch.bmm` on the view (B·H, N, N).

5️⃣  Second LayerNorm over the hidden dimension H, followed by the
    out‑gate multiplication – performed with native torch reductions
    (mean / var) which are far faster than a custom Triton reduction.

6️⃣  Final linear projection (`to_out`) – a single matmul from
    (R, H) → (R, dim) using `F.linear`.  The result is reshaped to
    `[B, N, N, dim]` (float32) and returned.

Only the mask‑aware split/fuse kernel is written in Triton; everything
else leverages highly‑optimized cuBLAS / torch kernels.  This reduces
kernel launch overhead and brings the overall latency below the
target 500 µs on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split‑fused projection, optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R,5·H] FP16 fused projection
    left_ptr,       # [B,H,N,N] FP16
    right_ptr,      # [B,H,N,N] FP16
    out_gate_ptr,   # [B,H,N,N] FP16
    mask_ptr,       # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset inside a [B,H,N,N] tensor (position of h=0)
    hidden_stride = N * N                     # stride of one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                               # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): either (input, mask, weights, cfg) or (input, weights, cfg)
            - input  : torch.Tensor [B, N, N, dim] (float32)
            - mask   : torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the following keys (all float32, on the correct device)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - cfg    : dict with at least "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, cfg = data
    else:
        x, weights, cfg = data
        mask = None

    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (over the channel dimension dim)
    # --------------------------------------------------------------
    # Torch’s native LayerNorm is highly tuned; we keep the computation in FP32
    # and cast the result to FP16 for the rest of the pipeline.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B,N,N,dim], FP32
    x_norm = x_norm.view(R, dim).to(torch.float16)   # [R,dim] FP16, contiguous

    # --------------------------------------------------------------
    # 3️⃣ fused linear projection to (5·H) channels
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim], FP16
    proj = F.linear(x_norm, fused_w)    # (R,5·H) FP16
    del fused_w, x_norm                 # free memory early

    # --------------------------------------------------------------
    # 4️⃣ split & apply mask / sigmoid gates (custom Triton kernel)
    # --------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask is broadcasted over H and stored as FP16 scalar per pair token
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 5️⃣ core einsum via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm over hidden dimension H + out_gate multiplication
    # --------------------------------------------------------------
    # Perform the mean / variance reduction with native torch ops (fast on H100)
    out_fp32 = out.to(torch.float32)                  # [B, H, N, N] FP32
    mu   = out_fp32.mean(dim=1, keepdim=True)        # [B,1,N,N]
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                    # [B,1,N,N]

    # LayerNorm parameters for the hidden dimension (to_out_norm)
    ln_w = weights["to_out_norm.weight"].to(torch.float16).view(1, H, 1, 1)   # [1,H,1,1] FP16
    ln_b = weights["to_out_norm.bias"].to(torch.float16).view(1, H, 1, 1)     # [1,H,1,1] FP16

    # Apply LayerNorm, then the out‑gate (cast out_gate to FP32 for the multiply)
    norm = ((out_fp32 - mu) * rstd)                     # [B,H,N,N] FP32
    norm = (norm * ln_w.to(torch.float32) + ln_b.to(torch.float32))   # LN
    norm = norm * out_gate.to(torch.float32)            # gating
    # `norm` now holds the hidden‑dim tensor ready for the final projection.

    # --------------------------------------------------------------
    # 7️⃣ final linear projection to original channel dimension (dim)
    # --------------------------------------------------------------
    # reshape to (R, H) and apply a single matmul
    norm_flat = norm.permute(0, 2, 3, 1).reshape(R, H)         # [R, H] FP32
    final_flat = F.linear(norm_flat, weights["to_out.weight"])  # [R, dim] FP32

    final = final_flat.view(B, N, N, dim)                     # [B,N,N,dim] FP32
    return final
```
