def fused_projection_norm_kernel(
    x_ptr, # [R, dim] float32
    left_ptr, right_ptr, out_gate_ptr, # [B, H, N, N] fp16 output
    mask_ptr, # [R] fp16 (optional) or None
    # weight pointers: R w_norm_weight (dim), bias (dim), left_proj (H, dim), right_proj (H, dim), left_gate (H, dim), right_gate (H, dim), out_gate (H, dim) all fp16
    # intensively pass config: R H N dim
    HAS_MASK,
    BLOCK_R, BLOCK_H, BLOCK_D,
):
    pid = program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute per row mean and var
    sum_ = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        # load x values
        x_val = tl.load(x_ptr + offs_r[:, None] * DIM + offs_d[None, :], mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32) # [BLOCK_R, BLOCK_D]

        sum_ += tl.sum(x_val, axis=1)
        sum_sq += tl.sum(x_val * x_val, axis=1)

    mu = sum_ / DIM
    var = sum_sq / DIM - mu * mu
    rstd = tl.rsqrt(var + eps)

    # Now second pass compute projection outputs. Use registers for accumulators for each hidden tile.
    # We'll compute for each hidden tile using block loops.
    # Compute base offsets for outputs.
    # Need to map offs_r -> (b,i,k):
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    # Output position base for hidden=0:
    base_output = b * (H * hidden_stride) + i * N + k

    # We'll iterate over hidden dimension tiles.
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Accumulators for left, right, out_gate (AND also left_gate, right_gate for gating)
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32) # maybe use float32 accumulation
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # for each dim tile where we will compute dot product
        for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < DIM

            # load x slice again (maybe we loaded it in first pass; we have to reload).
            x_val = tl.load(x_ptr + offs_r[:, None] * DIM + offs_d[None, :], mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # normalize x: (x - mu)*rstd*gamma + beta (norm weight & bias)
            # Need gamma (norm_weight) and beta (norm_bias) of length dim.
            gamma = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0) # fp32
            beta = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0) # fp32

            # normalized x value: (x - mu_i) * rstd_i * gamma[d] + beta[d]
            # Should broadcast mu/rstd across dim
            # mu: [BLOCK_R] -> broadcast
            x_norm = ((x_val - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

            # Now multiply with weight slices for each of the five weight matrices:
            # left_proj_weight: shape (H, dim) -> we need slice for hidden tile.

@triton.jit
def fused_proj_norm_gate_kernel(
    x_ptr,           # [R, dim] fp32
    norm_w_ptr,      # [dim] fp32
    norm_b_ptr,      # [dim] fp32
    left_proj_ptr,   # [H, dim] fp16
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    out_gate_ptr_w,
    left_out_ptr,    # [B, H, N, N] fp16
    right_out_ptr,
    out_gate_out_ptr,
    mask_ptr,        # [R] fp16 (optional)
    R, H, N, dim,
    eps: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute mean and variance per row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_val = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
        sum_val += tl.sum(x_val, axis=1)
        sum_sq += tl.sum(x_val * x_val, axis=1)

    mu = sum_val / dim
    var = sum_sq / dim - mu * mu
    rstd = tl.rsqrt(var + eps)

    # decode (b,i,k) for each row
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k    # start of hidden plane 0

    # optional mask broadcast value per row
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m_val = m_val[:, None]   # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x tile (float32)
            x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],

                             other=0.0).to(tl.float32)

            # layernorm weight/bias
            gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

            # normalized x (float32)
            norm_x = ((x_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

            # load weight slices (fp16 -> fp32)
            left_w = tl.load(left_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            right_w = tl.load(right_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            out_gate_w = tl.load(out_gate_ptr_w + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)

            # dot products (BLOCK_R, BLOCK_D) x (BLOCK_H, BLOCK_D)^T -> (BLOCK_R, BLOCK_H)
            left_acc += tl.dot(norm_x, left_w, trans_b=True)
            right_acc += tl.dot(norm_x, right_w, trans_b=True)
            left_gate_acc += tl.dot(norm_x, left_gate_w, trans_b=True)

        left_gate_sig = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)

        tl.store(out_gate_out_ptr + dst,
                 out_gate_sig,
                 mask=mask_r[:, None] & mask_h[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# fused kernel for first layernorm + five linear projections + gates + optional mask
@triton.jit
def fused_proj_norm_gate_kernel(
    x_ptr,          # [R, dim] FP32
    ln_w_ptr,       # [dim] FP32 (norm weight)
    ln_b_ptr,       # [dim] FP32 (norm bias)
    left_proj_ptr,  # [H, dim] FP16
    right_proj_ptr, # [H, dim] FP16
    left_gate_ptr,  # [H, dim] FP16
    right_gate_ptr, # [H, dim] FP16
    out_gate_ptr,   # [H, dim] FP16   (weight for out gate)
    left_out_ptr,   # [B, H, N, N] FP16
    right_out_ptr,  # [B, H, N, N] FP16
    out_gate_out_ptr, # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N, dim,
    eps: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---------- Compute mean and variance of each row ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mu  = sum_val / dim
    var = sum_sq / dim - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----------- Decode (b,i,k) for each flattened row ----------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k           # offset of hidden‑dim 0

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        m_val = m_val[:, None]    # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---------- Loop over hidden dimension ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators (FP32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ---------- dot over the feature dimension ----------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # load x tile (FP32)
            x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_D]

            # layer‑norm affine transformation (γ,β) – both FP32
            gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]
            beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]

            # normalized input (float32)
            norm_x = ((x_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

            # weight slices for the five linear maps (FP16 → FP32)
            left_w      = tl.load(left_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            right_w     = tl.load(right_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w= tl.load(right_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            out_gate_w  = tl.load(out_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)

            # accumulate dot products (M=BLOCK_R, K=BLOCK_D, N=BLOCK_H)
            left_acc      += tl.dot(norm_x,      left_w,      trans_b=True)
            right_acc     += tl.dot(norm_x,      right_w,     trans_b=True)
            left_gate_acc += tl.dot(norm_x,      left_gate_w, trans_b=True)
            right_gate_acc+= tl.dot(norm_x,      right_gate_w,trans_b=True)
            out_gate_acc  += tl.dot(norm_x,      out_gate_w,  trans_b=True)

        # ---------- Apply sigmoid gates ----------
        left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # ---------- Apply mask (only to the left/right tensors) ----------
        left_val  = (left_acc.to(tl.float16)  * left_gate)  * m_val
        right_val = (right_acc.to(tl.float16) * right_gate) * m_val

        # ---------- Store results ----------
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_out_ptr + dst,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + dst,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (unchanged)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (R‑tiling)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row µ and rsqrt‑var
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # Load hidden slice of the contracted tensor and the gate
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [M,K]

        # LayerNorm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - µ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M,K]
        gated = norm * vals_gate                                      # [M,K]
        gated_fp16 = gated.to(tl.float16)                             # back to FP16 for dot

        # Load slice of final projection weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # Store final tensor (flattened row index → [B,N,N,dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    The heavy part (first LayerNorm + 5 linear projections + three sigmoids)
    is fused into a single Triton kernel, eliminating the large temporary
    (R × 5 · hidden_dim) tensor used in the reference implementation.
    The remaining steps (pairwise contraction, second LayerNorm and final
    projection) reuse the highly‑optimized kernels defined above.
    """
    torch.set_grad_enabled(False)

    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Fuse LayerNorm + the five linear maps + sigmoid gates
    # ------------------------------------------------------------------
    # flatten input to [R, dim] – view is enough, no copy
    x_flat = input_tensor.reshape(R, dim).contiguous()   # FP32

    # layer‑norm parameters (γ,β) – keep them FP32
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias   = weights["norm.bias"].contiguous()

    # projection / gate weights – FP16 for compute efficiency
    left_proj_w  = weights["left_proj.weight"].to(torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(torch.float16).contiguous()

    # allocate outputs
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional mask handling
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # block sizes – chosen to stay inside the register budget on H100
    BLOCK_R = 32          # rows per program (≈ 1 % of R for N=1024)
    BLOCK_H = 32          # hidden‑dim tile
    BLOCK_D = 64          # feature‑dim tile (covers dim up to 384)

    grid = (triton.cdiv(R, BLOCK_R),)
    fused_proj_norm_gate_kernel[grid](
        x_flat,
        norm_weight,
        norm_bias,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        dim,
        eps,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # cleanup
    del x_flat, mask_ptr

    # ------------------------------------------------------------------
    # 2️⃣ Pairwise contraction (einsum) – expressed as batched GEMM
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Compute per‑pair mean / rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128    # H ≤ 128 → whole hidden dim fits in one tile

    rowwise_stats_kernel[grid_stats](

    # ------------------------------------------------------------------

    BLOCK_N_OUT = 128   # output‑dim chunk

        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),

        eps,

        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only.

Key ideas
---------
* The first LayerNorm and the five linear maps (left/right, three gates) are
  fused into **one** Triton kernel.  This removes the large temporary
  (R × 5·hidden) tensor required by the reference implementation and cuts
  memory traffic dramatically.
* The kernel works on a flattened pair‑wise dimension R = B·N·N.
  For each row it
      – computes mean/variance → normalises the input,
      – evaluates the five linear projections on the normalised input,
      – applies sigmoid gates,
      – (optionally) multiplies the left/right results with the pair mask.
* The pair‑wise contraction `out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]` is
  performed with a batched GEMM (`torch.bmm`), which cuBLAS handles very well.
* A second row‑wise LayerNorm, the out‑gate, and the final linear projection
  are performed by two more Triton kernels (statistics + fused final step).

The implementation keeps the heavy‑compute GEMM unchanged, but the
memory‑bound parts (LayerNorm + projections + mask) are now fully fused,
bringing the end‑to‑end runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Fuse first LayerNorm, five projections and three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_norm_gate_kernel(
    x_ptr,          # [R, dim] FP32
    ln_w_ptr,       # [dim] FP32  (LayerNorm weight γ)
    ln_b_ptr,       # [dim] FP32  (LayerNorm bias β)
    left_proj_ptr,  # [H, dim] FP16
    right_proj_ptr, # [H, dim] FP16
    left_gate_ptr,  # [H, dim] FP16
    right_gate_ptr, # [H, dim] FP16
    out_gate_ptr,   # [H, dim] FP16   (weight for out‑gate)
    left_out_ptr,   # [B, H, N, N] FP16
    right_out_ptr,  # [B, H, N, N] FP16
    out_gate_out_ptr, # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N, dim,
    eps: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    #   a) Row‑wise mean and variance of the raw input (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mu  = sum_val / dim
    var = sum_sq / dim - mu * mu
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    #   b) Decode (b,i,k) from the flattened row index
    # --------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N
    hidden_stride = N * N                         # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden‑dim 0

    # --------------------------------------------------------------
    #   c) Optional pair‑wise mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    #   d) Loop over hidden tiles and accumulate the five dot‑products
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators (FP32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # ----- load a tile of the input -----
            x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_D]

            # ----- LayerNorm affine (γ,β) -----
            gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

            # ----- normalised input -----
            norm_x = ((x_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

            # ----- weight slices (FP16 → FP32) -----
            left_w      = tl.load(left_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            right_w     = tl.load(right_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w= tl.load(right_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)
            out_gate_w  = tl.load(out_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)

            # ----- accumulate dot‑products (M×D)·(N×D)ᵀ → (M×N) -----
            left_acc      += tl.dot(norm_x,      left_w,      trans_b=True)
            right_acc     += tl.dot(norm_x,      right_w,     trans_b=True)
            left_gate_acc += tl.dot(norm_x,      left_gate_w, trans_b=True)
            right_gate_acc+= tl.dot(norm_x,      right_gate_w,trans_b=True)
            out_gate_acc  += tl.dot(norm_x,      out_gate_w,  trans_b=True)

        # --------------------------------------------------------------
        #   e) Apply sigmoid gates (computed in FP32 → cast back to FP16)
        # --------------------------------------------------------------
        left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # --------------------------------------------------------------
        #   f) Apply mask (only to left/right) and write the three tensors
        # --------------------------------------------------------------
        left_val  = (left_acc.to(tl.float16)  * left_gate)  * m_val
        right_val = (right_acc.to(tl.float16) * right_gate) * m_val

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_out_ptr + dst,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + dst,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (unchanged)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out‑gate and final projection (unchanged)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (R‑tiling)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M,K]
        gated = norm * vals_gate                                      # [M,K]
        gated_fp16 = gated.to(tl.float16)                             # back to fp16 for dot

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict must contain "dim" (int) and "hidden_dim" (int);
          optional "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    torch.set_grad_enabled(False)

    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Fuse first LayerNorm + the five linear maps + gates
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # FP32, no copy
    norm_weight = w["norm.weight"].contiguous()      # FP32
    norm_bias   = w["norm.bias"].contiguous()        # FP32

    left_proj_w  = w["left_proj.weight"].to(torch.float16).contiguous()
    right_proj_w = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w  = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w = w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w   = w["out_gate.weight"].to(torch.float16).contiguous()

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 32          # rows per program
    BLOCK_H = 32          # hidden‑dim tile
    BLOCK_D = 64          # feature‑dim tile (covers dim ≤ 384)

    grid = (triton.cdiv(R, BLOCK_R),)
    fused_proj_norm_gate_kernel[grid](
        x_flat,
        norm_weight,
        norm_bias,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        dim,
        eps,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat, mask_ptr

    # --------------------------------------------------------------
    # 2️⃣ Pairwise contraction (einsum) via batched GEMM
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 3️⃣ Row‑wise statistics (mean & rsqrt‑variance) of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                # H ≤ 128 → fits in a single tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (R‑tiling)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32, shape [H]
        w["to_out_norm.bias"],     # β  FP32, shape [H]
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
