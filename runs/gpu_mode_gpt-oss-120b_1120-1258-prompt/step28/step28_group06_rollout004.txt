"""
Outgoing TriMul (AlphaFold‑3 style) – forward only

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ One fused linear projection that produces the five tensors
   (left, right, left_gate, right_gate, out_gate) in a single matmul.
3️⃣ Triton kernel ``split_fuse_gate_kernel``:
   * decodes a flat index (b,i,k) → (b,i,k) layout,
   * optionally masks the left/right projections,
   * evaluates sigmoid gates (FP32 → FP16),
   * writes the three tensors in layout [B, H, N, N] (FP16).
4️⃣ Batched half‑precision GEMM (cuBLAS) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →  [B,H,N,N] (FP16)
5️⃣ Triton kernel ``rowwise_stats_kernel`` computes per‑row mean
   and rsqrt‑variance over the hidden dimension H (needed for the second
   LayerNorm).
6️⃣ Triton kernel ``final_output_kernel`` fuses:
   * the second LayerNorm (γ,β on the hidden dim H),
   * multiplication with the out_gate,
   * the final projection (to_out weight) → output tensor `[B,N,N,dim]`
   (FP32).

All kernels use mixed‑precision (FP16 compute, FP32 reductions) and are
tuned for the H100 (Triton 3.3.1).  The entry point ``custom_kernel`` follows
the required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ LayerNorm (FP32 → FP16) – use PyTorch's highly‑optimised kernel
# ----------------------------------------------------------------------
def _layernorm_fp32_to_fp16(x: torch.Tensor, weight: torch.Tensor,
                           bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Applies LayerNorm over the last dimension of ``x`` (which is expected to be
    a 2‑D tensor ``[R, dim]``) and returns a FP16 tensor.
    """
    # torch.nn.functional.layer_norm works in FP32 and is heavily tuned.
    x_norm = F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=eps)
    return x_norm.to(torch.float16)


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R,5·H]            FP16 fused projection
    left_ptr,       # [B,H,N,N]          FP16
    right_ptr,      # [B,H,N,N]          FP16
    out_gate_ptr,   # [B,H,N,N]          FP16
    mask_ptr,       # [R]                FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride inside the [B,H,N,N] tensors for h = 0
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (do the exponent in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # out_gate is not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # address of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B,H,N,N] FP16
    gate_ptr,       # [B,H,N,N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B,N,N,dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # Load hidden values and gate (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm on hidden dimension: (out - μ) * rstd
        diff = (vals_out - mu[:, None]) * rstd[:, None]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = diff * gamma[None, :] + beta[None, :]   # LN
        gated = norm * vals_gate                       # apply out_gate

        # Convert to FP16 for the dot with the projection weight
        gated_fp16 = gated.to(tl.float16)

        # Load slice of final‑projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, BLOCK_N]

        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # Store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): either (input, mask, weights, config) or
                      (input, weights, config).
        - input  : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask   : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights: dict with the following keys (all torch.float32 tensors):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config : dict containing at least
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments (support both 3‑ and 4‑element signatures)
    # ------------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, cfg = data
    elif len(data) == 3:
        x, weights, cfg = data
        mask = None
    else:
        raise ValueError("expected data tuple of length 3 or 4")

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()           # [R, dim] FP32
    x_norm = _layernorm_fp32_to_fp16(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – produces [R,5·H] tensor
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R,5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(
        left_mat,
        right_mat.transpose(1, 2),
    )                                          # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_stats = 256
    BLOCK_K_stats = 128   # H ≤ 128 → one iteration
    grid_stats = (triton.cdiv(R, BLOCK_M_stats),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_stats,
        BLOCK_K=BLOCK_K_stats,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + gating + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_out = 128
    BLOCK_K_out = 64    # hidden‑dim chunk (covers H=128 in two steps)
    BLOCK_N_out = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_out),
        triton.cdiv(dim, BLOCK_N_out),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ FP32
        weights["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_out,
        BLOCK_K=BLOCK_K_out,
        BLOCK_N=BLOCK_N_out,
        num_warps=8,
    )

    return final
