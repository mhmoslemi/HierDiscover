def ln_gate_permute_kernel(
    out_ptr, # original out [B, H, N, N] FP16
    gate_ptr, # out_gate [B, H, N, N] FP16
    mu_ptr, # [R] FP32
    rstd_ptr, # [R] FP32
    gamma_ptr, # [H] FP32
    beta_ptr, # [H] FP32
    out_perm_ptr, # [R, H] FP16 output
    R, H, N,
    BLOCK_R,
    BLOCK_H,
    eps // not needed
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R) # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,j)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N  # stride between rows in original out for next h

    for h_off in tl.range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate: value for each (r,h)
        # original out layout: offset = b * (H*nn) + h*nn + i*N + j
        # Let base = b * (H*nn) + i*N + j   # offset for h=0
        base = b * (H * nn) + i * N + j
        # load out slice: base + offs_h * nn
        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        mu_vals = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)[:, None]  # [BLOCK_R,1]
        rstd_vals = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :] # [1, BLOCK_H]
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)[None, :]

        norm = (out_vals - mu_vals) * rstd_vals
        norm = norm * gamma + beta
        gated = norm * gate_vals

        # store to out_perm: offset = offs_r[:, None] * H + offs_h[None, :]
        out_perm_ptr + offs_r[:, None] * H + offs_h[None, :]

4️⃣  Core “einsum’’ is a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]      (FP16)
5️⃣  Row‑wise mean / rsqrt‑variance of `out` (Triton).
6️⃣  Apply the second LayerNorm, multiply by `out_gate`, and **permute** the tensor to a
    contiguous layout `[R, H]` (R = B·N·N, H = hidden_dim) – all in one Triton kernel.
7️⃣  Final linear projection (FP32) with cuBLAS:
        out_final = (out_norm_gated) @ W_outᵀ                (FP32)
   Reshape to `[B, N, N, dim]`.

All kernels use mixed precision (FP32 for statistics, FP16 for bulk data) and run on an
H100 with Triton 3.3.1.  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm over the channel dimension (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]     γ (FP32)
    b_ptr,      # [dim]     β (FP32)
    rows,       # scalar R
    dim,        # scalar dim
    eps,        # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]               # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + k               # start offset for h==0

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [R, 5·H] → each hidden block contiguous
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R]          FP32
    rstd_ptr,  # [R]          FP32
    rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j          # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# --------------------------------------------------------------
# 4️⃣ Second LayerNorm + out_gate + permutation to [R, H] (FP32)
# --------------------------------------------------------------
@triton.jit
def ln_gate_permute_kernel(
    out_ptr,       # [B, H, N, N] FP16
    gate_ptr,      # [B, H, N, N] FP16
    mu_ptr,        # [R]          FP32
    rstd_ptr,      # [R]          FP32
    gamma_ptr,     # [H]          FP32 (to_out_norm.weight)
    beta_ptr,      # [H]          FP32 (to_out_norm.bias)
    out_perm_ptr,  # [R, H]       FP32   (output)
    R, H, N,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # offset of (b, h=0, i, j) in the original layout
    base = b * (H * nn) + i * N + j

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # load left/right? no – just the tensor that needs LN (out) and its gate
        idx = base[:, None] + offs_h[None, :] * nn        # [BLOCK_R, BLOCK_H]

        out_val = tl.load(out_ptr + idx,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)       # FP32
        gate_val = tl.load(gate_ptr + idx,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)      # FP32

        mu_val   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)[:, None]   # [BLOCK_R,1]
        rstd_val = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :]      # [1, BLOCK_H]
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)[None, :]

        # (out - μ) * rstd --> LayerNorm --> * out_gate
        norm = (out_val - mu_val) * rstd_val
        norm = norm * gamma + beta
        gated = norm * gate_val                                      # still FP32

        dst = offs_r[:, None] * H + offs_h[None, :]                  # [BLOCK_R, BLOCK_H]
        tl.store(out_perm_ptr + dst,
                 gated,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weight_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse 5 linear maps (left, right, gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H)   FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)           # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                      # H ≤ 128 → one tile sufficient
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + permute to [R, H] (FP32)
    # --------------------------------------------------------------
    out_perm = torch.empty(R, H, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_ln_gate = (triton.cdiv(R, BLOCK_R_OUT),)
    ln_gate_permute_kernel[grid_ln_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        out_perm,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate, mu, rstd  # free intermediate memory

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection (FP32) via cuBLAS
    # --------------------------------------------------------------
    final = torch.nn.functional.linear(out_perm,
                                       w["to_out.weight"],   # [dim, H] FP32
                                       bias=None)            # [R, dim] FP32
    final = final.view(B, N, N, dim)       # [B, N, N, dim] FP32

    return final
```
