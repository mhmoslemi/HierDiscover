@triton.jit
def final_fused_kernel(
    out_ptr,           # [B, H, N, N] fp16
    gate_ptr,          # [B, H, N, N] fp16
    ln_w_ptr,          # [H] fp32
    ln_b_ptr,          # [H] fp32
    w_out_T_ptr,       # [H, dim] fp16 (transposed)
    final_ptr,         # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)  # rows
    pid_n = tl.program_id(1)  # output dim tiles

    ridx = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    cidx = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = ridx < total_rows
    mask_n = cidx < dim

    nn = N * N
    b = ridx // nn
    rem = ridx - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + j

    # allocate shared memory for hidden values
    sh_out = tl.shared_memory([BLOCK_M, H], dtype=tl.float16)
    sh_gate = tl.shared_memory([BLOCK_M, H], dtype=tl.float16)

    # first pass: compute sum, sumsq, store into shared memory
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < H

        # Global pointers
        out_ptrs = out_ptr + base[:, None] + cur_k[None, :] * stride_hidden
        gate_ptrs = gate_ptr + base[:, None] + cur_k[None, :] * stride_hidden

        # Load fp16 values
        out_vals = tl.load(out_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        gate_vals = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Store into shared memory
        sh_out_off = ridx[:, None] * H + cur_k[None, :]   # shape [BLOCK_M, BLOCK_K]
        tl.store(sh_out + sh_out_off, out_vals, mask=mask_m[:, None] & mask_k[None, :])
        tl.store(sh_gate + sh_out_off, gate_vals, mask=mask_m[:, None] & mask_k[None, :])

        # Accumulate sum for stats
        out_fp32 = out_vals.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq += tl.sum(out_fp32 * out_fp32, axis=1)

    # compute mean and rstd
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute normalized term and dot with weight
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < H

        # Load from shared memory
        vals_out = tl.load(sh_out + ridx[:, None] * H + cur_k[None, :], mask=mask_m[:, None] & mask_k[None, :])
        vals_gate = tl.load(sh_gate + ridx[:, None] * H + cur_k[None, :], mask=mask_m[:, None] & mask_k[None, :])

        # Convert to fp32 for arithmetic
        out_fp32 = vals_out.to(tl.float32)
        gate_fp32 = vals_gate.to(tl.float32)

"""
Optimised “outgoing” TriMul (AlphaFold‑3) forward pass – Triton version.
--------------------------------------------------------------------
The implementation follows the reference module but aggressively fuses
memory‑bound work:

1️⃣  Row‑wise LayerNorm (float32 → float16) – independent kernel.
2️⃣  A single fused linear projection produces the concatenated
    tensors `[left, right, left_gate, right_gate, out_gate]`.
3️⃣  `split_fuse_gate_kernel` applies the optional mask, computes the
    sigmoid gates and stores the three tensors (`left`, `right`,
    `out_gate`) in fp16 layout `[B, H, N, N]`.
4️⃣  Batched GEMM (cuBLAS) contracts the two pair‑wise tensors:
    `out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]`.
5️⃣  **Fused statistics + final projection kernel**  
    `final_fused_kernel` loads `out` and `out_gate` once, computes per‑row
    mean/variance over the hidden dimension (`H`), stores the raw values
    in shared memory, and then (still inside the same kernel) applies the
    second LayerNorm (`to_out_norm`), the element‑wise `out_gate`,
    and the final linear projection (`to_out`).  This removes a separate
    row‑wise‑stats pass and halves the global‑memory traffic for the
    `out`/`out_gate` tensors.

All kernels target Triton 3.3.1 and run efficiently on an NVIDIA H100.
The final output is a float32 tensor `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, D] fp32 input
    y_ptr,      # [R, D] fp16 output
    w_ptr,      # [D]    γ (fp32)
    b_ptr,      # [D]    β (fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: accumulate sum / sum‑sq ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Split/fuse + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mval[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between consecutive hidden planes

    # base offset for (b,i,k) at hidden == 0   layout: [B, H, N, N]
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # iterate over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # address of the five projection blocks for this tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 for stability, then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets for the flattened view [B, H, N, N]
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Fuse per‑row statistics, LN(to_out_norm), out_gate multiplication,
#     and final projection into a single kernel.
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,           # [B, H, N, N] fp16 (contracted tensor)
    gate_ptr,          # [B, H, N, N] fp16 (out_gate)
    ln_w_ptr,          # [H] fp32 (to_out_norm.weight)
    ln_b_ptr,          # [H] fp32 (to_out_norm.bias)
    w_out_T_ptr,       # [H, dim] fp16 (to_out weight transposed)
    final_ptr,         # [B, N, N, dim] fp32
    total_rows, H: tl.constexpr, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)        # rows (flattened B·N·N)
    pid_n = tl.program_id(1)        # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)       # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N                       # stride between hidden planes
    base = b * (H * stride_hidden) + i * N + j  # start of hidden slice for this (b,i,j)

    # ------------------------------------------------------------------
    # Shared memory buffers to hold a whole hidden vector for each row
    # ------------------------------------------------------------------
    sh_out = tl.shared_memory([BLOCK_M, H], dtype=tl.float16)
    sh_gate = tl.shared_memory([BLOCK_M, H], dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean / variance and store raw values
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = cur_k < H

        # global pointers for this hidden tile
        out_ptrs = out_ptr + base[:, None] + cur_k[None, :] * stride_hidden
        gate_ptrs = gate_ptr + base[:, None] + cur_k[None, :] * stride_hidden

        # load fp16 values from global memory
        out_vals = tl.load(out_ptrs,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0)
        gate_vals = tl.load(gate_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        # store the raw values into shared memory (for the second pass)
        smem_off = offs_m[:, None] * H + cur_k[None, :]   # [BLOCK_M, BLOCK_K]
        tl.store(sh_out + smem_off, out_vals, mask=mask_m[:, None] & mask_k[None, :])
        tl.store(sh_gate + smem_off, gate_vals, mask=mask_m[:, None] & mask_k[None, :])

        # accumulate statistics (use fp32)
        out_fp32 = out_vals.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: apply LN(to_out_norm), out_gate, and the final projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < H

        # load the previously stored tensors from shared memory
        vals_out = tl.load(sh_out + offs_m[:, None] * H + cur_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :])
        vals_gate = tl.load(sh_gate + offs_m[:, None] * H + cur_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :])

        out_fp32  = vals_out.to(tl.float32)   # [M, K]
        gate_fp32 = vals_gate.to(tl.float32)  # [M, K]

        # LayerNorm over hidden dimension
        diff = (out_fp32 - mean[:, None]) * rstd[:, None]          # [M, K]

        gamma = tl.load(ln_w_ptr + cur_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + cur_k, mask=mask_k, other=0.0)   # [K] fp32

        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]

        # multiply by out_gate
        term = norm * gate_fp32                                      # [M, K]
        term_fp16 = term.to(tl.float16)

        # load a tile of the final weight matrix (H × dim)
        w_ptr = w_out_T_ptr + cur_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # fp16

        # accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Store final tensor (flattened view of [B, N, N, dim])
    # ------------------------------------------------------------------
    dst_ptr = final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS) → core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # fp16
    right_mat = right.view(B * H, N, N)         # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣  Fuse final LN, out_gate multiplication & projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_FUSED = 64          # fits the shared‑memory budget (≈32 KB per buffer)
    BLOCK_K_FUSED = 32
    BLOCK_N_FUSED = 128
    grid_fused = (
        triton.cdiv(R, BLOCK_M_FUSED),
        triton.cdiv(dim, BLOCK_N_FUSED),
    )
    final_fused_kernel[grid_fused](
        out,                     # fused tensor
        out_gate,
        weights["to_out_norm.weight"],   # γ (fp32) shape [H]
        weights["to_out_norm.bias"],     # β (fp32) shape [H]
        w_out_T,
        final,
        R,
        hidden_dim,   # H is a compile‑time constant for shared‑memory allocation
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_FUSED,
        BLOCK_K=BLOCK_K_FUSED,
        BLOCK_N=BLOCK_N_FUSED,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    return final
