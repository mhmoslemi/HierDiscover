@triton.jit
def rowwise_stats_kernel(
    out_ptr,
    mu_ptr,
    rstd_ptr,
    R, H,
    eps,
    BLOCK_R: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)     # [BLOCK_R]
    mask_m = offs_m < R

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]   # [BLOCK_R, BLOCK_K]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

@triton.jit
def final_output_kernel(
    out_ptr,
    gate_ptr,
    mu_ptr,
    rstd_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    R, H, dim,
    eps,
    BLOCK_R: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_c = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # load mu and rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)  # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        # load out and gate values
        ptr_out = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_k[None, :]

        vals_out = tl.load(ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        vals_gate = tl.load(ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        # LN: (out - mu) * rstd
        diff = (vals_out - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]

        # Apply out_gate
        term = norm * vals_gate
        term_fp16 = term.to(tl.float16)

        # load weight tile
        # w_out_T_ptr shape [H, dim], contiguous hidden stride = dim.
        ptr_w = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(ptr_w,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :])
```

Note: At each iteration we recompute gamma and beta for the hidden slice; we should multiply gamma and beta are fp32; but term_fp16 is fp16; but tl.dot expects both arguments same type. It could accept mixed? Usually tl.dot expects both FP16 or both FP32. Here term_fp16 is fp16, w_tile is fp16 so okay; the term is produced by term_fp16 already. However term_fp16 is computed from term = norm * vals_gate; norm is fp32; vals_gate is fp32; term .to(fp16) after multiplication. The multiplication earlier does term = norm * vals_gate; this product is fp32; then term_fp16 = term.to(fp16).

Alternatively we could compute term_fp16 directly by casting both to fp16 earlier: But we may want to maintain precision.

Potential difference: The term after LN and gating should use fp32 for better accuracy but output is float32 after final projection; but converting to fp16 before dot may lose precision; but final dot accum is fp32. That may be acceptable.

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The forward pass consists of six fused stages:

1️⃣  Row‑wise LayerNorm on the flattened pairwise tensor (fp32 → fp16).
2️⃣  One fused linear projection that produces the concatenated tensor
    [left, right, left_gate, right_gate, out_gate] (5 × hidden_dim).
3️⃣  Split + mask + sigmoid‑gate kernel:
    * left/right are stored in layout   [B, H, N, N]   (needed for the
      batched GEMM).
    * out_gate is stored in layout       [R, H]     (R = B·N·N) with
      the hidden dimension contiguous → ideal for the final reduction.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
      out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k].
5️⃣  Row‑wise mean / rsqrt‑variance (layer‑norm over the hidden dimension)
    on the contracted tensor stored as [R, H].
6️⃣  Final kernel fuses the second LayerNorm, the element‑wise multiplication
    with out_gate and the linear projection to the model dimension.
    It produces the output tensor in shape [B, N, N, dim] (float32).

All kernels target Triton 3.3.1 and run on an NVIDIA H100.  The
implementation works for both masked and un‑masked inputs and for all
configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, D] fp32
    y_ptr,        # [R, D] fp16
    w_ptr,        # [D]   γ (fp32)
    b_ptr,        # [D]   β (fp32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split/Fuse + mask + sigmoid gates
#    left/right → [B, H, N, N]   (contiguous hidden stride = N²)
#    out_gate → [R, H]          (hidden stride = 1)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [R, H] fp16
    mask_ptr,          # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flattened index r = b·N·N + i·N + k ----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim, applied only to left/right)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mval[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # stride of one hidden slice in [B,H,N,N]

    # base offset for (b,i,k) at hidden==0 → layout [B, H, N, N]
    base_lr = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # base offset for out_gate (flattened [R, H])
    base_gate = offs_r * H                          # [BLOCK_R]

    # iterate over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # five projection blocks for this tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # mask * gate * projection
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                    # no mask on out_gate

        # store left/right → [B, H, N, N]
        dst_lr = base_lr[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst_lr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_lr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate → [R, H] (contiguous hidden)
        dst_gate = base_gate[:, None] + offs_h[None, :]               # [BLOCK_R, BLOCK_H]
        tl.store(out_gate_ptr + dst_gate,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor (over H)
#    Input layout: [R, H] (contiguous hidden)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [R, H] fp16
    mu_ptr,       # [R] fp32
    rstd_ptr,     # [R] fp32
    R, H,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_m = offs_m < R

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]   # stride_hidden = 1
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_R, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse LN(to_out_norm), out_gate multiplication and final projection
#    Layouts:
#      out      → [R, H] fp16          (contracted tensor)
#      out_gate → [R, H] fp16
#      mu,rstd  → [R]   fp32
#      weight   → [H, dim] fp16  (transposed to‑out weight)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [R, H] fp16
    gate_ptr,       # [R, H] fp16
    mu_ptr,         # [R]   fp32
    rstd_ptr,       # [R]   fp32
    ln_w_ptr,       # [H]   fp32  (to_out_norm.weight)
    ln_b_ptr,       # [H]   fp32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] fp16 (to_out weight transposed)
    out_final_ptr,  # [R, dim] fp32
    R, H, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(0)        # rows
    pid_c = tl.program_id(1)        # output‑dim tiles

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # per‑row statistics (fp32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load out and gate values (fp16 → fp32)
        ptr_out  = out_ptr  + offs_m[:, None] * H + offs_k[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_k[None, :]

        vals_out  = tl.load(ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)     # [BLOCK_R, BLOCK_K]

        vals_gate = tl.load(ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)     # [BLOCK_R, BLOCK_K]

        # LayerNorm parameters for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]

        # LN over hidden dim using the pre‑computed μ, rstd
        diff = (vals_out - mu[:, None]) * rstd[:, None]          # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]             # [M, K]

        # Apply out_gate (element‑wise) → term in fp16 for the dot
        term_fp16 = (norm * vals_gate).to(tl.float16)            # [M, K]

        # Load a tile of the final weight matrix (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [K, TILE_N] fp16

        # Accumulate: (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor (flattened view of [R, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (fp32)  shape [dim]
        weights["norm.bias"],      # β (fp32)  shape [dim]
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # [R, 5·H] fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    # out_gate as flattened [R, H]
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) → core einsum
    # ------------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)          # fp16
    right_mat = right.reshape(B * H, N, N)         # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N]

    # Permute to [B, N, N, H] and flatten → [R, H] (contiguous hidden)
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out.view(R, H)                         # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑variance (LayerNorm over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 64                 # H ≤ 128 → fits in two tiles
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LN, out_gate multiplication & final projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128
    BLOCK_K_OUT = 64
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (fp32) shape [H]
        weights["to_out_norm.bias"],     # β (fp32) shape [H]
        w_out_T,
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # Reshape back to [B, N, N, dim]
    final = final.view(B, N, N, dim)
    return final
