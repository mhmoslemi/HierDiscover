def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1. Input layer norm (float32)
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] float32

    x_norm = x_norm_fp32.to(torch.float16)

    # 2. Fused linear projection to 5*H dims
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]

    proj = torch.nn.functional.linear(x_norm.view(R, dim), fused_weight)  # [R, 5*H] FP16

    # 3. Split, apply mask, sigmoids via Triton kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # configure block sizes
    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat if not nomask else None

    # 4. Batched matrix multiplication (left @ right^T)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)  # [B, H, N, N] FP16
    del left_mat, right_mat, left, right  # free memory

    # 5. LayerNorm over hidden dimension, apply out_gate, final projection
    # reshape out and out_gate to [B,N,N,H] (non‑contiguous)
    out_perm = out.permute(0, 2, 3, 1)  # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1)  # [B, N, N, H]
    del out, out_gate  # free

    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] FP16

    term = out_norm * out_gate_perm  # [B, N, N, H] FP16

    # final linear projection
    w_out = weights["to_out.weight"].to(torch.float16)  # [dim, H]
    term_flat = term.reshape(R, H)  # [R, H]
    final_fp16 = torch.nn.functional.linear(term_flat, w_out)  # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
0️⃣ Input LayerNorm on the last dimension (FP32 → FP16).
1️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM (cuBLAS) yielding a tensor (R, 5·H) with R = B·N·N.
2️⃣ Triton kernel `split_fuse_gate_kernel`:
   * Decodes the flat “pair‑index’’ (b,i,k) from R.
   * Splits the fused projection into left, right and out_gate.
   * Applies the optional pair mask (only to left/right) and the three sigmoid gates.
   * Writes the three tensors in layout [B, H, N, N] (FP16) – optimal for the next batched GEMM.
3️⃣ Batched GEMM: `out = left @ rightᵀ` for each hidden slice (cuBLAS/Tensor‑cores).
4️⃣ LayerNorm over the hidden dimension H (built‑in `torch.nn.functional.layer_norm`,
   operating on a view `[B,N,N,H]` without a copy).
5️⃣ Element‑wise multiplication with `out_gate` (still FP16).
6️⃣ Final linear projection (`to_out`) – a second GEMM (cuBLAS) producing the
   output `[B,N,N,dim]` (cast back to FP32).

Only one custom Triton kernel is required (step 2); all heavy matrix‑multiplications
use cuBLAS/Tensor‑cores, and the two LayerNorms use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional, broadcasted)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # -----------------------------------------------------------------
    # Decode (b, i, k) from the flat row index r = b·N·N + i·N + k
    # -----------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # -----------------------------------------------------------------
    # Optional mask (applied only to left/right)
    # -----------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between two hidden slices

    # base offset for (b, i, k) in the flattened [B, H, N, N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # -----------------------------------------------------------------
    # Loop over hidden dimension in tiles of size BLOCK_H
    # -----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Offsets inside the fused projection tensor
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for stability, then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate  # mask is *not* applied to the out‑gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), fused_w)      # [R, 5·H] FP16
    del fused_w, x_norm, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Split the fused projection, apply mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask  = 1
        mask_ptr  = mask_flat
    else:
        has_mask  = 0
        mask_ptr  = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr, mask_flat if (not nomask) and (mask is not None) else None

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM: left @ rightᵀ  (cuBLAS/Tensor‑cores)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 4️⃣ LayerNorm over hidden dimension H, apply out_gate, final projection
    # ------------------------------------------------------------------
    # move the hidden axis to the last dimension (no copy, just a view)
    out_perm      = out.permute(0, 2, 3, 1)          # [B, N, N, H] (non‑contiguous)
    out_gate_perm = out_gate.permute(0, 2, 3, 1)    # [B, N, N, H]
    del out, out_gate

    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                               # [B, N, N, H] FP16

    term = out_norm * out_gate_perm                  # element‑wise gate (FP16)

    # final linear projection:  (R, H) @ (dim, H)ᵀ → (R, dim)
    w_out = w["to_out.weight"].to(torch.float16)    # [dim, H]
    term_flat = term.reshape(R, H)                  # [R, H]
    final_fp16 = F.linear(term_flat, w_out)        # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
