@triton.jit
def final_norm_output_kernel(
    out_ptr, gate_ptr, ln_w_ptr, ln_b_ptr, w_out_T_ptr,
    out_final_ptr,
    B, H, N, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0) # rows (R = B*N*N)
    pid_n = tl.program_id(1) # columns (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)
    mask_m = offs_m < R
    mask_n = offs_n < dim

    # decode (b,i,j) as before
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j

    # First pass: compute sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # Initialize accumulator for final output
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Second pass: compute normalized and accumulate projection
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        # load out and gate
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden

        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # LN for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * vals_gate

        # projection weight tile (K, TILE_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # store final
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation (high‑performance).

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) producing a tensor (R, 5·H) where
   R = B·N·N, H = hidden_dim.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * un‑flatten the fused projection,
   * optionally apply the pair‑mask (only to left/right),
   * compute sigmoid gates,
   * write three tensors `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → tensor `[B, H, N, N]` (FP16).
5️⃣ Triton kernel `final_norm_output_kernel` fuses
   * per‑row statistics (mean / rsqrt‑var) over the hidden dimension,
   * the second LayerNorm (`to_out_norm`),
   * the out‑gate multiplication,
   * the final projection (`to_out`).
   The kernel reads `out` and `out_gate` twice (once for the statistics,
   once for the projection) but avoids an extra kernel launch and global
   temporary buffers.
6️⃣ The result is returned as a float32 tensor of shape `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is implemented – no gradients are needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim]  FP32 input
    y_ptr,           # [R, dim]  FP16 output
    w_ptr,           # [dim]    LN weight (γ) FP32
    b_ptr,           # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)      # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional, broadcasted)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Mask handling – only applied to left/right
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                        # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between consecutive hidden slices

    # base offset for (b, i, k) at hidden‑slice 0 in layout [B, H, N, N]
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Process the hidden dimension in blocks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Index into the fused projection matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for better numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fusion of second LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (result of left·right)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,         # [H] FP32            (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32            (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16       (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32 (final result)
    B, H, N, dim, R,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """
    For each output position (b, i, j) we:
        1️⃣ Compute mean & variance across the hidden dimension H (row‑wise).
        2️⃣ Apply LayerNorm (to_out_norm) to the hidden vector.
        3️⃣ Multiply by out_gate.
        4️⃣ Project to output dimension `dim` with `to_out`.
    All in one kernel to avoid extra memory traffic.
    """
    pid_m = tl.program_id(0)   # rows over R = B·N·N
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # Decode (b, i, j) from the flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # start of first hidden slice

    # ---------- 1️⃣ Compute mean & variance over H ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---------- 2‑4️⃣ Normalization, out_gate, projection ----------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden‑slice values
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden

        vals_out  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)      # [BLOCK_M, BLOCK_K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)      # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # LN: (out - μ) * rstd → *γ + β
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]

        # Apply out_gate
        term = norm * vals_gate                                       # [M, K]

        # Projection weight tile (K, TILE_N) – weight is stored transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (FP16·FP16 → FP32)
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # Store final tensor (flattened view of [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with the following keys (all torch.Tensor):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (five heads → 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                  # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # flatten mask to [R] and cast to FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128   # more rows per program – better occupancy
    BLOCK_H = 64    # hidden‑dim block (covers hidden_dim≤128 in two passes)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (core einsum)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)           # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                    # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows (R) processed per program
    BLOCK_K_OUT = 32           # hidden‑dim tiles
    BLOCK_N_OUT = 128          # output‑dim tiles (covers dim up to 384)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],      # γ, FP32
        w["to_out_norm.bias"],        # β, FP32
        w_out_T,
        final,
        B, H, N, dim, R,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
