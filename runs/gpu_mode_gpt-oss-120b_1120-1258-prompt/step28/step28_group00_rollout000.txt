"""
TriMul (outgoing) – fast Triton implementation
================================================
This file implements the forward‑only “outgoing” Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 and related protein‑structure models.
The algorithm follows the reference PyTorch code but fuses the heaviest
memory‑bound operations with Triton:

1️⃣  Row‑wise LayerNorm on the flattened (B·N·N, dim) tensor.
2️⃣  A single fused linear projection that produces the five tensors
    [left, right, left_gate, right_gate, out_gate] in one pass.
3️⃣  Split‑fuse kernel: applies the optional mask, computes the two
    sigmoid gates and stores left, right and out_gate in the layout
    [B, H, N, N] (fp16).
4️⃣  Batched GEMM (torch.bmm) computes the core einsum
    ``out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]``.
5️⃣  Row‑wise mean / rsqrt‑variance of *out* over the hidden dimension H.
6️⃣  Final kernel fuses the second LayerNorm (to_out_norm), the
    element‑wise multiplication with out_gate, and the output linear
    projection (to_out).  The result has shape (B,N,N,dim) and dtype
    ``float32``.

The implementation works for all configurations listed in the problem
statement (masked / un‑masked, dim∈{128,384}, hidden_dim=128, N up to
1024, B∈{1,2}).  It runs on Triton 3.3.1 and targets mixed‑precision
execution on an NVIDIA H100.

Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, D] fp32 input
    y_ptr,       # [R, D] fp16 output
    w_ptr,       # [D] γ (fp32)
    b_ptr,       # [D] β (fp32)
    rows, D, eps,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- first pass: compute mean and variance ----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)                       # [BLOCK_R]
        sum_sq  += tl.sum(x * x, axis=1)                   # [BLOCK_R]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

        # (x - μ) / σ
        norm = (x - mean[:, None]) * rstd[:, None]

        # scale+bias (fp32 → fp16)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # fp32

        y = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_r[:, None] * D + offs_d[None, :],
            y.to(tl.float16),
            mask=mask_r[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Split‑fuse + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for one hidden channel in layout [B, H, N, N]
    stride_h = N * N

    # base offset for (b,i,k) at hidden = 0
    base_offset = b * (H * stride_h) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointer into fused projection: each hidden dim occupies a contiguous
        # segment of length 5·H after flattening R dimension.
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R,BLOCK_H]

        # Load the five projection sub‑vectors
        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid (computed in fp32 for numerical stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst = base_offset[:, None] + offs_h[None, :] * stride_h   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Row‑wise μ, rstd over hidden dimension (H) of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] fp16
    mu_ptr,           # [R] fp32
    rstd_ptr,         # [R] fp32
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    j = rem - i * N

    stride_h = N * N
    base = b * (H * stride_h) + i * N + j                # start of hidden slice for this (b,i,j)

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * stride_h
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_R, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mu,   mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣  Fuse LN(to_out_norm), out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] fp16 (contracted tensor)
    gate_ptr,       # [B, H, N, N] fp16 (out_gate)
    mu_ptr,         # [R] fp32
    rstd_ptr,       # [R] fp32
    ln_w_ptr,       # [H] fp32 (to_out_norm.weight)
    ln_b_ptr,       # [H] fp32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] fp16 (weight of to_out transposed)
    out_ptr_final,  # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(0)      # row dimension (R)
    pid_n = tl.program_id(1)      # output dim tile

    row_start = pid_r * BLOCK_R
    col_start = pid_n * BLOCK_N

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_r = offs_r < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    j = rem - i * N

    stride_h = N * N
    base = b * (H * stride_h) + i * N + j                # start of hidden slice for (b,i,j)

    # Load per‑row statistics (fp32)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # hidden‑plane values (fp16 → fp32)
        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * stride_h
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * stride_h

        val_out = tl.load(
            ptr_out,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [BLOCK_R, BLOCK_K]
        val_gate = tl.load(
            ptr_gate,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [BLOCK_R, BLOCK_K]

        # LayerNorm parameters (fp32)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # LN over hidden dim using pre‑computed μ, rstd
        norm = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # Apply out_gate (element‑wise)
        gated = norm * val_gate                                  # [BLOCK_R, BLOCK_K]
        gated_fp16 = gated.to(tl.float16)

        # Load a tile of the final weight matrix (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [BLOCK_K, BLOCK_N]

        # Accumulate: (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # Store final tensor (flattened view of [B, N, N, dim])
    dst_ptr = out_ptr_final + offs_r[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_r[:, None] & mask_n[None, :])

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """

    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]      # denoted H in the kernels
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 64

        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (fp32) shape [dim]
        weights["norm.bias"],        # β (fp32) shape [dim]
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # 2️⃣  Fused linear projection that yields 5·H outputs
    # ------------------------------------------------------------------

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_weight)   # (R, 5·H) fp16
    del fused_weight, x_norm

    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):

        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,

    )

    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for bmm
    left_mat  = left.view(B * H, N, N)          # fp16
    right_mat = right.view(B * H, N, N)         # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise mean / rsqrt‑variance over H (for LN on out)
    # ------------------------------------------------------------------

    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_K_STATS = 32                     # H=128 → 4 tiles of 32

    )

    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],    # γ (fp32) shape [H]
        weights["to_out_norm.bias"],      # β (fp32) shape [H]
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    return final
