custom_kernel – Outgoing TriMul (AlphaFold‑3) fused implementation

The forward pass consists of six stages:
1️⃣  Row‑wise LayerNorm on the flattened (B·N·N, dim) tensor
    – mean/variance accumulated in FP32, normalized output in FP16.
2️⃣  One fused linear projection that produces the five weight matrices
    (left_proj, right_proj, left_gate, right_gate, out_gate) in a single
    matmul.
3️⃣  Split the fused projection into the three tensors required for the
    batched GEMM, apply the optional pairwise mask and sigmoid gates.
4️⃣  Batched half‑precision GEMM (left @ rightᵀ) using torch.bmm (tensor cores).
5️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor (required for the
    second LayerNorm).
6️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    linear projection into a single Triton kernel.  Each output position
    (b,i,j) is normalised across the hidden dimension, gated, then projected
    to the model dimension (dim) with a dot‑product over the hidden dimension.

The implementation works for any batch size B∈{1,2}, sequence length N≤1024,
model dimension dim∈{128,384} and hidden dimension hidden_dim∈{128}.
All heavy element‑wise work is done by Triton kernels; the large GEMM
(step 4) is handled by cuBLAS (torch.bmm) for maximum tensor‑core throughput.
The final output is returned as float32.

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # const   *[rows, dim]   (float32)
    y_ptr,   # mutable *[rows, dim]   (float16)
    w_ptr,   # const   *[dim]        (float32) γ
    b_ptr,   # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----------- accumulate mean & var (fp32) -----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        src = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(src,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------- write normalised output (fp16) -----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        src = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(src,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * gamma[None, :] + beta[None, :]
        dst = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(dst, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]      (fp16)  (ignored if HAS_MASK=0)

    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask per (b,i,k)
    if HAS_MASK:

        mask_val = m[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                       # stride of one hidden slice
    base_offset = b * (H * N * N) + i * N + k   # offset of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # address into the fused projection: [R, 5*H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,

                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                     # not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr    + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

def rowwise_stats_kernel(

    rstd_ptr,   # mutable *[R]         (float32)

    mask_m = offs_m < total_rows

    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

    gate_ptr,       # const *[B, H, N, N] (fp16)   (out_gate)
    mu_ptr,         # const *[R]            (float32)
    rstd_ptr,       # const *[R]            (float32)
    ln_w_ptr,       # const *[H]            (float32) γ of to_out_norm
    ln_b_ptr,       # const *[H]            (float32) β of to_out_norm
    w_out_T_ptr,    # const *[H, dim]       (fp16)   weight transposed (H×dim)

    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,

    pid_n = tl.program_id(1)

    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

        mask_k = offs_k < H

        src_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        src_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
