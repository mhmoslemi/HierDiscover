"""
TriMul (outgoing) – fused Triton implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×D} and computes:

1. LayerNorm over the last dimension D (FP32 → FP16).
2. A fused linear projection that produces five matrices
   (left, right, left‑gate, right‑gate, out‑gate) of shape
   (B, H, N, N). Optional mask is applied to left/right.
3. Batched GEMM  left @ rightᵀ   (sum over the second N index)
   → out ∈ ℝ^{B×H×N×N} (FP16).
4. Row‑wise mean / variance of *out* over the hidden dimension H.
5. Fuse the second LayerNorm, the out‑gate and the final linear projection
   (H → D weight) into a single Triton kernel.

All intermediate tensors are kept in FP16, statistical moments are accumulated
in FP32 and the final output is FP32.  The kernels work for any batch size
(1‑2), sequence length (up to 1024) and channel dimensions
(D ∈ {128,384,768}, H ∈ {128,384,768}).

The entry point is `custom_kernel(data)`, where `data` is a tuple
(input_tensor, mask_tensor, weights_dict, config_dict).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,               # input, output and LN parameters
    rows: tl.int32, dim: tl.int32, eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """LayerNorm over the last dimension (row‑wise)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance in FP32 --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (FP32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (FP32)
        out = norm * w[None, :] + b[None, :]                # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                # [R, 5*H]   (FP16)
    left_ptr,                # [B, H, N, N] (FP16)
    right_ptr,               # [B, H, N, N] (FP16)
    out_gate_ptr,            # [B, H, N, N] (FP16)
    mask_ptr,                # [R] (FP16) – used only if HAS_MASK==1
    R: tl.int32, H: tl.int32, N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Decode the fused projection, apply optional mask & sigmoid gates."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index (b,i,k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # Offset to the first hidden plane for (b,i,k)
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # Layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]     # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates (mask only on left/right)
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask on the out‑gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (over H)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute mean and rsqrt‑variance of `out` over the hidden axis H."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # start of hidden slice 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,          # to_out_norm (γ,β) – FP32
    w_out_T_ptr,                # [H, D] – FP16
    out_final_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32, dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """Apply LN over H, multiply by out‑gate and project to D."""
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden slice 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm parameters (γ,β) for the hidden axis
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # Normalize, apply γ/β and the out‑gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [M,K]
        term = norm * gate_vals                                        # [M,K]
        term_fp16 = term.to(tl.float16)                               # fp16 for dot

        # Load a tile of the final weight matrix (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]      # [K,N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                         # [K,N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # [M,N]

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            input_tensor : torch.Tensor [B, N, N, D] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict  : must contain "dim" (D), "hidden_dim" (H) and
                           "nomask" (bool) – whether the mask is omitted.
    Returns:
        torch.Tensor : [B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                 # total number of (b,i,k) positions
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()                # [R, D] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # [R, D] FP16

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection: 5 × hidden_dim rows
    # -----------------------------------------------------------------
    # Concatenate all five projection weights (shape: [5*H, D])
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, D]

    proj = F.linear(x_norm, fused_w)         # (R, 5*H) FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional mask and sigmoid gates → left / right / out_gate
    # -----------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask_tensor is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ) over the hidden dimension
    # -----------------------------------------------------------------
    # Reshape to (B*H, N, N) for efficient bmm
    left_mat  = left.view(B * H, N, N)               # [B*H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B*H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # -----------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # -----------------------------------------------------------------
    # Transpose final weight (shape: [H, D]) for easier loading in kernel
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, D]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64    # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 64    # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
