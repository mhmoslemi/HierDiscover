"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (`C = dim`).
The forward pass consists of six logical stages, but only the first
LayerNorm is implemented in Triton – the remaining steps are expressed
with highly‑optimised PyTorch primitives (Tensor‑Core bmm, fused
Linear, LayerNorm, element‑wise ops).  This satisfies the “at least one
kernel” requirement while keeping the overall runtime competitive.

1️⃣  Row‑wise LayerNorm (float32 → float16) – custom Triton kernel.
2️⃣  Fuse the five linear projections (left, right, three gates) in a
    single `torch.nn.functional.linear` call.
3️⃣  Split the fused result, apply the optional pair‑wise mask and the
    three sigmoid gates.
4️⃣  Batched GEMM: `out[b, h] = left[b, h] @ right[b, h].T`
5️⃣  Row‑wise LayerNorm over the hidden dimension (`out_norm`).
6️⃣  Multiply by the `out_gate`, then a final linear projection back to
    `dim`.  The result is returned as a contiguous `float32` tensor
    `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32)   γ
    b_ptr,    # const   *[dim]      (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)                      # one program per row tile
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32) for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over feature dimension in blocks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ/β and write result as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]              # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the *outgoing* TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with Linear / LayerNorm parameters
                         (all float32, Linear layers have no bias)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]           # == H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ (float32)
        w["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left, right, three gates)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, dim]

    proj = F.linear(x_norm, fused_w)                       # (R, 5*H) fp16
    # clean up intermediate tensors no longer needed
    del x_norm, fused_w, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    # Split the fused projection
    left_raw      = proj[:, :H]                     # (R, H)
    right_raw     = proj[:, H:2 * H]
    left_gate_raw = proj[:, 2 * H:3 * H]
    right_gate_raw = proj[:, 3 * H:4 * H]
    out_gate_raw  = proj[:, 4 * H:5 * H]

    # Reshape to [B, N, N, H]
    left_raw      = left_raw.view(B, N, N, H)
    right_raw     = right_raw.view(B, N, N, H)
    left_gate_raw = left_gate_raw.view(B, N, N, H)
    right_gate_raw = right_gate_raw.view(B, N, N, H)
    out_gate_raw  = out_gate_raw.view(B, N, N, H)

    # Optional mask handling
    use_mask = (mask is not None) and (not cfg.get("nomask", True))
    if use_mask:
        # mask shape -> [B, N, N, 1] in fp16 for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
    else:
        mask_fp16 = None

    # Sigmoid gates (stay in fp16 for speed)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # (B,N,N,H) fp16

    if use_mask:
        left  = left_raw  * mask_fp16 * left_gate   # (B,N,N,H) fp16
        right = right_raw * mask_fp16 * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # Rearrange to [B, H, N, N] for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) fp16

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (Tensor‑Core) : left @ rightᵀ
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                             # (B, H, N, N) fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # --------------------------------------------------------------
    # Move H to the trailing dimension so that torch.layer_norm works
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # (B, N, N, H) fp32
    out_fp32 = out_perm.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=w["to_out_norm.weight"],   # γ  (float32, shape H)
        bias=w["to_out_norm.bias"],       # β  (float32, shape H)
        eps=eps,
    )                                   # (B, N, N, H) float32

    # Apply the out‑gate (broadcasted)
    out_gate_fp32 = out_gate.permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,H)
    out_norm_gate = out_norm * out_gate_fp32                       # (B,N,N,H) fp32

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    # `to_out.weight` has shape (dim, H).  torch.nn.functional.linear
    # expects weight of shape (out_features, in_features).
    final = F.linear(out_norm_gate, w["to_out.weight"])   # (B, N, N, dim) float32

    return final
