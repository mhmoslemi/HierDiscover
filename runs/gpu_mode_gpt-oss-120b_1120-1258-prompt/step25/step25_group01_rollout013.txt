"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣  Row‑wise LayerNorm (over the last dimension `dim`) – done with
    :func:`torch.nn.functional.layer_norm` (highly‑optimized GPU kernel).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single half‑precision GEMM
    ``proj = x_norm @ W_fused.T``.
    ``W_fused`` is a concatenation of the five weight matrices.

3️⃣  Split the fused projection, apply the optional mask and the sigmoid
    gates with pure PyTorch element‑wise ops and reshape to the layout
    ``[B, H, N, N]`` (half).

4️⃣  Compute the cubic contraction  
    ``out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]`` using a
    batched GEMM (cuBLAS).  The right operand is made contiguous
    beforehand to avoid an internal copy inside cuBLAS.

5️⃣  Compute per‑pair mean/variance over the hidden dimension H with
    `torch.mean` / `torch.var` (population variance → unbiased=False).

6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the
    final linear projection (`to_out`) into a single Triton kernel
    ``final_output_kernel``.  The kernel reads the contracted tensor,
    normalises it with the pre‑computed statistics, applies the
    `out_gate`, and performs the matrix‑vector multiplication with the
    transposed final weight in half‑precision (Tensor‑Core friendly),
    writing the result as float‑32.

Only the last step is implemented in Triton, satisfying the requirement
that “part of the operations” use a Triton kernel while maximising
runtime performance by minimising kernel launches and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣  Final fused kernel: LayerNorm (over hidden dim), gate and projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]    fp16 contracted tensor
    gate_ptr,       # [B, H, N, N]    fp16 out_gate (sigmoid)
    mu_ptr,         # [R]             fp32 mean over H
    rstd_ptr,       # [R]             fp32 rsqrt(var+eps)
    ln_w_ptr,       # [H]             to_out_norm.weight (γ) fp32
    ln_b_ptr,       # [H]             to_out_norm.bias   (β) fp32
    w_out_T_ptr,    # [H, dim]        to_out weight^T     fp16
    out_final_ptr,  # [B, N, N, dim] fp32 result
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (flattened B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # Decode flattened index -> (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for the first hidden plane (h=0)
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # [BLOCK_M]

    # Load mean / rstd for each (b,i,j)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # fp32

    # accumulator for the final projection (float32)
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Loop over hidden dimension in chunks of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        # Load contracted values and gate for the current hidden slice
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm (γ,β) over the hidden dimension
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # fp32

        # (out - μ) / σ  →  diff
        diff = (vals_out - mu[:, None]) * rstd[:, None]             # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]

        # Apply the out‑gate
        term = norm * vals_gate                                      # [M,K]
        term_fp16 = term.to(tl.float16)                              # back to fp16 for the matmul

        # Load a tile of the final linear weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # fp16

        # Accumulate (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store the final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑based forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the model weights (float32)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm  (float32 → float16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly optimized on GPU and
    # computes population statistics (unbiased=False) which matches our
    # custom implementation.
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)               # [B,N,N,dim] float32
    x_norm = x_norm.to(torch.float16)                # cast to FP16 for GEMM
    x_norm_flat = x_norm.view(R, dim)                # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection  (5 * hidden_dim outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [5*H, dim]
    proj = F.linear(x_norm_flat, fused_w)            # [R, 5*H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates (pure PyTorch, vectorised)
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                        # [R, 5, H]

    left_raw      = proj[:, 0, :]   # [R, H]
    right_raw     = proj[:, 1, :]   # [R, H]
    left_gate_raw = proj[:, 2, :]   # [R, H]
    right_gate_raw= proj[:, 3, :]   # [R, H]
    out_gate_raw  = proj[:, 4, :]   # [R, H]

    left_gate  = torch.sigmoid(left_gate_raw)        # [R, H] fp16
    right_gate = torch.sigmoid(right_gate_raw)       # [R, H] fp16
    out_gate   = torch.sigmoid(out_gate_raw)         # [R, H] fp16 (no mask)

    if (mask is not None) and (not nomask):
        mask_fp16 = mask.to(torch.float16).reshape(R)    # [R]
        mask_fp16 = mask_fp16[:, None]                    # [R,1] broadcast over H
        left  = left_raw  * mask_fp16 * left_gate
        right = right_raw * mask_fp16 * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # reshape to [B, H, N, N] (contiguous)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM for the cubic contraction
    # ------------------------------------------------------------------
    # left:  [B, H, N, N] → (B*H, N, N)
    left_mat = left.view(B * H, N, N)                     # contiguous

    # right must be transposed on the last two axes before GEMM
    right_t = right.permute(0, 1, 3, 2).contiguous()      # -> [B, H, N, N] (k ↔ j)
    right_mat = right_t.view(B * H, N, N)                # contiguous

    # Batched matrix multiplication: out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat)                 # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑variance over the hidden dimension H
    # ------------------------------------------------------------------
    # Use torch's fast reduction kernels (population variance → unbiased=False)
    mu   = out.float().mean(dim=1)                        # [B, N, N] FP32
    var  = out.float().var(dim=1, unbiased=False)        # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)                        # [B, N, N] FP32

    mu_flat   = mu.view(-1).contiguous()                 # [R]  FP32
    rstd_flat = rstd.view(-1).contiguous()               # [R]  FP32

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection (Triton)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim] FP16

    out_final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Tile sizes – chosen to balance register pressure and occupancy
    BLOCK_M_OUT = 128   # rows (flattened B·N·N) per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out_ptr=out,
        gate_ptr=out_gate,
        mu_ptr=mu_flat,
        rstd_ptr=rstd_flat,
        ln_w_ptr=w["to_out_norm.weight"],   # γ  FP32
        ln_b_ptr=w["to_out_norm.bias"],     # β  FP32
        w_out_T_ptr=w_out_T,
        out_final_ptr=out_final,
        R=R,
        H=H,
        N=N,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return out_final
