"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm the flattened pair representation (FP32 → FP16) with the
    highly‑optimised torch.nn.functional.layer_norm.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16) that yields a tensor of shape (R, 5·H),
    where R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional mask, compute the three
    sigmoid gates and store the three tensors *left*, *right* and
    *out_gate* in layout ``[B, H, N, N]`` (FP16) via a small Triton kernel.
    This kernel fuses mask‑multiplication and sigmoid‑gating,
    avoiding three separate PyTorch kernels.

4️⃣  Batched GEMM (cuBLAS) computes the core einsum
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).

5️⃣  Second LayerNorm (row‑wise over the hidden dimension) is performed with
    ``torch.nn.functional.layer_norm`` which internally uses a fast
    reduction kernel.

6️⃣  Apply the out‑gate (broadcasted) and the final projection (FP16 GEMM)
    with ``torch.nn.functional.linear``.  The final result is cast back to
    FP32 as required by the model.

Only the split‑&‑gate step lives in Triton; all other heavy‑weight
operations are delegated to highly‑optimised PyTorch kernels, giving a
significant speed‑up over the fully‑Triton reference while keeping the
implementation numerically identical.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of the hidden dimension inside the [B, H, N, N] tensor
    hidden_stride = N * N
    # base offset for the (b,i,k) slice of the first hidden plane (h=0)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather the 5·H fused projection (left | right | left_gate | right_gate | out_gate)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid computed in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets: layout [B, H, N, N] → flatten as B·H·N·N
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,   mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor  : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor   : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all torch.float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  Fast LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    # torch.layer_norm returns FP32; cast to FP16 for the rest
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                         # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣  Fused linear projection producing (R, 5·H)
    # --------------------------------------------------------------
    # Concatenate the five projection matrices (each [H, dim]) → [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                             # (R, 5·H) FP16
    del x_norm, fused_w                                           # free memory

    # --------------------------------------------------------------
    # 3️⃣  Split, mask and sigmoid gates – Triton kernel
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()   # [R]
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right : [B, H, N, N] → view as [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm over hidden dimension (row‑wise)
    # --------------------------------------------------------------
    # Permute to [B, N, N, H] so that the hidden dim is the last one
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()# [B, N, N, H] FP16

    out_norm = F.layer_norm(
        out_perm,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                        # [B,N,N,H] FP16

    # --------------------------------------------------------------
    # 6️⃣  Apply out‑gate and final projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate_perm               # broadcasting, still FP16
    # Linear projection: hidden_dim → dim
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    gated_flat = gated.view(R, hidden_dim)         # [R, H] FP16
    final_fp16 = F.linear(gated_flat, w_out)      # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B,N,N,dim] FP32

    return final
