import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr,
                           mask_ptr, R, H, N,
                           HAS_MASK: tl.constexpr,
                           BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for left (i,k)
    base_offset_left = b * (H * hidden_stride) + i * N + k
    # base offset for right transposed (k,i)
    base_offset_right = b * (H * hidden_stride) + k * N + i

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        # load 5 blocks at once
        proj_block = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )  # shape [BLOCK_R, BLOCK_H]

        # split
        left_raw = proj_block
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_offset_right = base_offset_right[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset_left,

        tl.store(right_ptr + dst_offset_right,

        tl.store(out_gate_ptr + base_offset_left[:, None] + offs_h[None, :] * hidden_stride,

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1) Transform input with first LayerNorm (FP32 -> FP16)
    x_flat = x.reshape(R, dim)
    x_norm = F.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # 2) Fused linear projection to 5*H hidden dim
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]
    proj = F.linear(x_norm, fused_weight)  # (R, 5*H) half

    # 3) Split and apply mask & gates, store left/right/out_gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 256
    BLOCK_H = 64  # 2 blocks for H=128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4) Batched GEMM over hidden dim (no transpose needed)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)  # already transposed
    out = torch.bmm(left_mat, right_mat)  # (B*H, N, N) half
    out = out.view(B, H, N, N)

    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H)

"""
Optimized outgoing TriMul kernel (AlphaFold3) – forward only.
Algorithm:
1️⃣  First LayerNorm on the pair representation (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps into a single GEMM (R × dim → R × 5·H) in FP16.
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional pair‑mask, evaluates sigmoid gates and writes
    the three tensors `left`, `right_T` (already transposed) and `out_gate`
    in layout [B, H, N, N] (H is the 2‑nd fastest dimension).
    Storing `right_T` removes the costly `transpose`‑copy that appears in
    the reference implementation.
4️⃣  Batched half‑precision GEMM (`torch.bmm`) evaluates the core einsum:
    out[b,h,i,j] = Σₖ left[b,h,i,k]·right_T[b,h,k,j].
5️⃣  Second LayerNorm, multiplication with `out_gate` and the final linear
    projection are performed with PyTorch (highly‑optimised kernels).
   The whole pipeline stays in FP16 until the last cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] fused projection (FP16)
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16 (stored transposed, i.e. right_T)
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] optional mask (FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes

    # base offsets for the three output tensors
    # left and out_gate are stored as (i,k) (the natural orientation)
    base_off_left = b * (H * hidden_stride) + i * N + k
    # right is stored already transposed as (k,i)
    base_off_right = b * (H * hidden_stride) + k * N + i

    # loop over hidden dimension in chunks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five projection slices --------------------------------
        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ----- sigmoid in FP32 for better numerical stability ------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -------------------------------------------
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate_val

        # write left
        dst_off_left = base_off_left[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_off_left,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        # write right (already transposed)
        dst_off_right = base_off_right[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            right_ptr + dst_off_right,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        # write out_gate (same orientation as left)
        dst_off_gate = base_off_left[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            out_gate_ptr + dst_off_gate,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )

            - weights_dict : mapping with the keys required by the reference
                             implementation (all float32 tensors)
            - config_dict  : must contain "dim", "hidden_dim", and "nomask"
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ---------------------------- unpack ------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # number of (i,k) tokens
    H = hidden_dim

    # ----------------- 1️⃣ first LayerNorm (FP32 → FP16) --------------------
    x_flat = x.reshape(R, dim)                 # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                        # [R, dim] FP16

    # ----------------- 2️⃣ fused projection (5·H) --------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16

    # ----------------- 3️⃣ split, mask, sigmoid gates --------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 256
    BLOCK_H = 64   # will cover hidden_dim=128 in two steps
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ----------------- 4️⃣ batched GEMM (left @ right_T) -------------------
    # right is already stored transposed, so we can directly use bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)   # transposed layout
    out = torch.bmm(left_mat, right_mat)  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)            # [B, H, N, N] FP16

    # ----------------- 5️⃣ second LayerNorm + out_gate + final proj -----
    # reshape to (R, H) where each row corresponds to a (i,j) pair
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)       # (R, H) FP16
    gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H) # (R, H) FP16

    # LayerNorm over hidden dimension
    out_ln = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    # gating
    out_gated = out_ln * gate_flat

    # final linear projection back to `dim`
    final = F.linear(out_gated, w["to_out.weight"].to(torch.float16))   # (R, dim) FP16
    final = final.view(B, N, N, dim).float()                           # FP32

    return final

"""
Optimized outgoing TriMul kernel (AlphaFold3) – forward only.

Algorithm
---------
1️⃣ First LayerNorm on the pair representation (FP32 → FP16) using
   torch.nn.functional.layer_norm (vectorised, no custom kernel).

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM:   proj = X_norm @ W_fused.T   (shape [R, 5·H]).

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * splits the fused projection,
   * applies the optional mask,
   * evaluates sigmoid gates (FP32 → FP16),
   * stores **left**, **right_T** (right already transposed) and **out_gate**
     in layout [B, H, N, N] (H is the fastest‑moving dimension).
   Storing the transposed right‑matrix removes the costly `transpose`
   copy that appears in the reference implementation.

4️⃣ Batched half‑precision GEMM (`torch.bmm`) evaluates the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right_T[b,h,k,j].

5️⃣ Second LayerNorm, multiplication with `out_gate` and the final linear
   projection back to `dim` are carried out with PyTorch’s highly‑optimised
   kernels (layer_norm + linear).  All heavy work stays in float16; the
   final tensor is cast back to float32.

The whole pipeline respects the original semantics while cutting down
memory traffic (no transpose copy) and reducing kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] fused projection (FP16)
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16 (stored transposed, i.e. right_T)
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] optional mask (FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # -------- decode flat index r = b·N·N + i·N + k --------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # -------- optional mask (broadcast over hidden dim) -------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes

    # ----- base offsets for the three output tensors -----------------
    # left and out_gate keep the natural (i,k) layout
    base_off_left = b * (H * hidden_stride) + i * N + k
    # right is stored **transposed** as (k,i)
    base_off_right = b * (H * hidden_stride) + k * N + i

    # -------- iterate over the hidden dimension ----------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five projection slices -----------------------
        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ----- sigmoid in FP32 (more accurate) -----------------------
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate_val

        # ----- write left --------------------------------------------
        dst_off_left = base_off_left[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_off_left,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        # ----- write right (already transposed) ----------------------
        dst_off_right = base_off_right[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            right_ptr + dst_off_right,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        # ----- write out_gate (same layout as left) -----------------
        dst_off_gate = base_off_left[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            out_gate_ptr + dst_off_gate,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping with keys required by the reference
                              implementation (all float32)
            - config_dict   : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # number of pair tokens (i,k)
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim)                              # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                     # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                         # (R, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left, right_T, out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)   # transposed
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 256
    BLOCK_H = 64          # H=128 fits in two blocks
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left and right are already in the correct layout; no extra transpose
    left_mat = left.view(B * H, N, N)           # (B·H, N, N) FP16
    right_mat = right.view(B * H, N, N)         # (B·H, N, N) FP16 (already transposed)
    out = torch.bmm(left_mat, right_mat)       # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                 # (B, H, N, N) FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication, final projection
    # --------------------------------------------------------------
    # flatten to (R, H) where each row corresponds to a pair (i, j)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)        # (R, H) FP16
    gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H) FP16

    out_ln = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    out_gated = out_ln * gate_flat

    final = F.linear(
        out_gated,
        w["to_out.weight"].to(torch.float16),  # [dim, H] → FP16
    )  # (R, dim) FP16

    final = final.view(B, N, N, dim).float()  # back to FP32
    return final
