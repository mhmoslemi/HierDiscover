@triton.jit
def ln_gate_kernel(
    out_ptr,      # [R, H] FP16 (output of bmm)
    gate_ptr,     # [R, H] FP16 (out_gate)
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    ln_w_ptr,     # [H] FP32
    ln_b_ptr,     # [H] FP32
    out_gated_ptr, # [R, H] FP16 result
    R, H,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # tile hidden
    # we can loop over hidden dimension in chunks
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)[:, None]  # [BLOCK_R, 1]
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :] # [1, BLOCK_H]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]

        # Normalize
        norm = (out_val - mu) * rstd   # [BLOCK_R, BLOCK_H]
        norm = norm * gamma + beta    # apply LN weight/bias
        gated = norm * gate_val

        # store as fp16
        tl.store(out_gated_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
```

@triton.jit
def rowwise_stats_contig_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    R, H, eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # load block of rows
    # accumulate sum and sumsq over hidden dimension in chunks.
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load [BLOCK_R, BLOCK_H] block
        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_r, mean, mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)
```

TriMul (outgoing) – fast forward‑only Triton implementation
================================================================
Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input tensor  (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) producing a tensor of shape (R, 5·H) where
    R = B·N·N and H = hidden_dim.
3️⃣  Split the fused projection, apply the optional pairwise mask,
    compute sigmoid gates and store three tensors *left*, *right*,
    *out_gate* in layout [B, H, N, N] (FP16) – Triton kernel.
4️⃣  Batched GEMM (cuBLAS) over the hidden dimension:
        out = left · rightᵀ      → shape [B, H, N, N] (FP16).
5️⃣  Permute *out* and *out_gate* to [B, N, N, H] and flatten to (R, H)
    – this makes the hidden dimension contiguous.
6️⃣  Row‑wise mean / rsqrt‑variance of *out* → μ, rstd (FP32) – Triton kernel.
7️⃣  Fuse the second LayerNorm, the *out_gate* multiplication and the
    final projection into two kernels:
        • row‑wise stats (step 6)
        • LN + gate (FP32 arithmetic, write FP16)
   The gated hidden representation is then projected to the model dimension
   with a single cuBLAS GEMM (FP16 × FP16 → FP16) and cast back to FP32.
All kernels target Triton 3.3.1 on an H100 and use mixed precision.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]    LN weight (γ) FP32
    b_ptr,      # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass : compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass : write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = m[:, None]    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                       # offset between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k     # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability, convert back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance (contiguous hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_contig_kernel(
    out_ptr,         # [R, H] FP16
    mu_ptr,          # [R] FP32
    rstd_ptr,        # [R] FP32
    R, H, eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_R]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_R]

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and write gated hidden state
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_contig_kernel(
    out_ptr,          # [R, H] FP16   (post‑BMM)
    gate_ptr,         # [R, H] FP16   (out_gate)
    mu_ptr,           # [R]    FP32
    rstd_ptr,         # [R]    FP32
    ln_w_ptr,         # [H]    FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]    FP32 (to_out_norm.bias)
    gated_ptr,        # [R, H] FP16  (output)
    R, H,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # load per‑row statistics once (broadcast later)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    mu   = mu[:, None]      # [BLOCK_R,1]
    rstd = rstd[:, None]    # [BLOCK_R,1]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw tensors
        out_vals  = tl.load(out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_H]
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1, BLOCK_H]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1, BLOCK_H]

        # LayerNorm over hidden dim → normalize → gate
        norm = (out_vals - mu) * rstd          # [BLOCK_R, BLOCK_H]
        norm = norm * gamma + beta            # apply LN weight & bias
        gated = norm * gate_vals               # element‑wise gating

        tl.store(gated_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point – forward only
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # left/right are [B, H, N, N]; view as [B·H, N, N] for bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    # No longer needed
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Make hidden dimension contiguous (permute + flatten)
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    out_flat = out.view(R, H)          # [R, H] FP16
    out_gate_flat = out_gate.view(R, H)  # [R, H] FP16
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_H_STATS = 128                      # H ≤ 128 ⇒ single iteration
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_contig_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_H=BLOCK_H_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_LN = 128
    BLOCK_H_LN = 128
    grid_ln_gate = (triton.cdiv(R, BLOCK_R_LN),)
    ln_gate_contig_kernel[grid_ln_gate](
        out_flat,
        out_gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ, FP32
        w["to_out_norm.bias"],     # β, FP32
        gated,
        R,
        H,
        BLOCK_R=BLOCK_R_LN,
        BLOCK_H=BLOCK_H_LN,
        num_warps=4,
    )
    # clean up intermediate tensors
    del out_flat, out_gate_flat, mu, rstd

    # --------------------------------------------------------------
    # 8️⃣ Final projection (FP16 GEMM) → FP32 result
    # --------------------------------------------------------------
    # weight shape for torch.nn.functional.linear: (out_features, in_features) = (dim, H)
    w_out = w["to_out.weight"].to(torch.float16).contiguous()  # [dim, H]

    final_fp16 = torch.nn.functional.linear(gated, w_out)      # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
