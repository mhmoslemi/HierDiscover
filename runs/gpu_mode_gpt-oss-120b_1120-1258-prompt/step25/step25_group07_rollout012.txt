@triton.jit
def ln_proj_gate_kernel(
    x_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    ln_w_ptr, ln_b_ptr,
    proj_w_ptr,
    R, H, N, dim, eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]  # [B_R,1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # Compute row mean and rstd
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now iterate over hidden tiles
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k  # start of hidden h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # allocate accumulator for 5*BLOCK_H outputs
        acc = tl.zeros((BLOCK_R, 5 * BLOCK_H), dtype=tl.float32)

        # Process dim tiles
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load x and normalize
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
            beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

            norm = ((x - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
            norm_fp16 = norm.to(tl.float16)

            # Load projection weight rows for this hidden tile, 5 groups
            # weight rows index: start = offs_h (for left), but need offset for each group:
            # left rows: offs_h
            # right rows: H + offs_h
            # left_gate rows: 2*H + offs_h, etc.
            # combine into a single offset with stride dim
            # one way: load each group separately and accumulate; but we can load concatenated using tl.load with for each group offset.
            # Simpler: load groups separately and compute dot individually per group but we want a single dot for all.
            # We'll concatenate manually: create a view of size (5*BLOCK_H, BLOCK_D)
            # For each group we compute separate tl.dot with norm_fp16 and weight slice, then stack? Could be easier to compute dot one group at a time and assign into acc.
            # However we can use tl.dot multiple times per group inside loop; overhead may be acceptable (5 times per dim iteration).
            # To simplify correctness, compute each group separately.

            # Left
            w_left = tl.load(proj_w_ptr + (offs_h[:, None]) * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0)   # shape (BLOCK_H, BLOCK_D)
            acc[:, 0:BLOCK_H] += tl.dot(norm_fp16, w_left).to(tl.float32)   # [BLOCK_R, BLOCK_H]

            # Right
            w_right = tl.load(proj_w_ptr + ((H + offs_h)[:, None]) * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            acc[:, BLOCK_H:2*BLOCK_H] += tl.dot(norm_fp16, w_right).to(tl.float32)

            # Left gate
            w_lgate = tl.load(proj_w_ptr + ((2*H + offs_h)[:, None]) * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            acc[:, 2*BLOCK_H:3*BLOCK_H] += tl.dot(norm_fp16, w_lgate).to(tl.float32)

            # Right gate
            w_rgate = tl.load(proj_w_ptr + ((3*H + offs_h)[:, None]) * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            acc[:, 3*BLOCK_H:4*BLOCK_H] += tl.dot(norm_fp16, w_rgate).to(tl.float32)

            # Out gate
            w_ogate = tl.load(proj_w_ptr + ((4*H + offs_h)[:, None]) * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            acc[:, 4*BLOCK_H:5*BLOCK_H] += tl.dot(norm_fp16, w_ogate).to(tl.float32)

        # After dim loops, we have raw projections for hidden tile
        # Apply sigmoid for gate groups
        # left_raw = acc[:, 0:BLOCK_H]
        left_raw = acc[:, 0:BLOCK_H]
        right_raw = acc[:, BLOCK_H:2*BLOCK_H]
        left_gate_raw = acc[:, 2*BLOCK_H:3*BLOCK_H]
        right_gate_raw = acc[:, 3*BLOCK_H:4*BLOCK_H]
        out_gate_raw = acc[:, 4*BLOCK_H:5*BLOCK_H]

        # sigmoid in FP32 then cast to fp16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw))).to(tl.float16)

        left = left_raw.to(tl.float16) * mask_val * left_gate
        right = right_raw.to(tl.float16) * mask_val * right_gate
        outg = out_gate

        # store
        dst_off = base_offset[:, None] + (offs_h[None, :] * hidden_stride)   # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst_off, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, outg, mask=mask_r[:, None] & mask_h[None, :])
```

"""
Optimized outgoing Triangle‑Multiplicative (TriMul) forward kernel.

Algorithm
---------
1️⃣  Fuse the first LayerNorm, the five linear projections (left, right,
    left‑gate, right‑gate, out‑gate) and the optional pairwise mask into a
    single Triton kernel.
    → Produces three tensors `left`, `right` and `out_gate` in the layout
      [B, H, N, N] (FP16) ready for the batched GEMM.

2️⃣  Batched half‑precision GEMM (cuBLAS `torch.bmm`) computes
    out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]   (FP16).

3️⃣  Row‑wise mean / rsqrt‑variance of `out` over the hidden dimension H
    (kernel `rowwise_stats_kernel`).

4️⃣  Fuse the second LayerNorm, multiplication by `out_gate` and the final
    projection (to_out) into one Triton kernel (`final_output_kernel`).

All kernels are written for Triton 3.3.1, target an NVIDIA H100 and use
mixed‑precision (FP16 for the bulk of the work, FP32 for reductions).
The implementation obeys the signature `custom_kernel(data)` required by the
grader.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm + fused projection + optional mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def ln_proj_gate_kernel(
    x_ptr,                # [R, dim]          FP32 input
    mask_ptr,             # [R]               FP16 mask (optional)
    left_ptr,             # [B, H, N, N]      FP16 output
    right_ptr,            # [B, H, N, N]      FP16 output
    out_gate_ptr,         # [B, H, N, N]      FP16 output
    ln_w_ptr,             # [dim]             LN weight γ  (FP32)
    ln_b_ptr,             # [dim]             LN bias   β  (FP32)
    proj_w_ptr,           # [5·H, dim]        fused projection weight (FP16)
    R, H, N, dim, eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]          # [BLOCK_R, 1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # ------------------------------------------------------------------
    # mean / variance of the row (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)        # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # hidden‑wise tile loop (process BLOCK_H hidden channels at a time)
    # ------------------------------------------------------------------
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for the five raw projections  (FP32)
        acc = tl.zeros((BLOCK_R, 5 * BLOCK_H), dtype=tl.float32)

        # --------------------------------------------------------------
        # loop over the input dimension `dim`
        # --------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load and normalize a tile of the input row
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)        # [R, D]

            gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

            # (x‑mean)·rstd·γ + β   →   FP16 for the matmul
            norm = ((x - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
            norm_fp16 = norm.to(tl.float16)                               # [R, D]

            # ------- five independent GEMV’s (one per projection) --------
            # left projection
            w_left = tl.load(
                proj_w_ptr + (offs_h[:, None]) * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc[:, 0:BLOCK_H]          += tl.dot(norm_fp16, w_left).to(tl.float32)

            # right projection
            w_right = tl.load(
                proj_w_ptr + ((H + offs_h)[:, None]) * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc[:, BLOCK_H:2*BLOCK_H]   += tl.dot(norm_fp16, w_right).to(tl.float32)

            # left‑gate projection
            w_lgate = tl.load(
                proj_w_ptr + ((2*H + offs_h)[:, None]) * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc[:, 2*BLOCK_H:3*BLOCK_H] += tl.dot(norm_fp16, w_lgate).to(tl.float32)

            # right‑gate projection
            w_rgate = tl.load(
                proj_w_ptr + ((3*H + offs_h)[:, None]) * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc[:, 3*BLOCK_H:4*BLOCK_H] += tl.dot(norm_fp16, w_rgate).to(tl.float32)

            # out‑gate projection
            w_ogate = tl.load(
                proj_w_ptr + ((4*H + offs_h)[:, None]) * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            )
            acc[:, 4*BLOCK_H:5*BLOCK_H] += tl.dot(norm_fp16, w_ogate).to(tl.float32)

        # --------------------------------------------------------------
        # post‑process the five raw tensors (sigmoid + mask)
        # --------------------------------------------------------------
        left_raw      = acc[:, 0:BLOCK_H]
        right_raw     = acc[:, BLOCK_H:2*BLOCK_H]
        left_gate_raw = acc[:, 2*BLOCK_H:3*BLOCK_H]
        right_gate_raw= acc[:, 3*BLOCK_H:4*BLOCK_H]
        out_gate_raw  = acc[:, 4*BLOCK_H:5*BLOCK_H]

        # sigmoid in FP32 → cast back to FP16
        left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_raw))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_raw))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_raw))).to(tl.float16)

        left  = left_raw .to(tl.float16) * mask_val * left_gate
        right = right_raw.to(tl.float16) * mask_val * right_gate
        outg  = out_gate

        # store results; layout = [B, H, N, N]
        dst_off = base_offset[:, None] + (offs_h[None, :] * hidden_stride)   # [R, BLOCK_H]
        tl.store(left_ptr     + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣  Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_m = tl.program_id(0)   # rows  (R)
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M, K]

        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm (per‑row) → γ· + β
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]             # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]

        # apply out‑gate
        term = norm * gate_vals                                     # [M, K]

        # cast to FP16 for the final projection (tensor‑core friendly)
        term_fp16 = term.to(tl.float16)

        # slice of final projection weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N_out]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, N_out]

        # accumulate (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store result [B, N, N, dim] (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the keys used in the reference model
                ("norm.weight", "norm.bias",
                 "left_proj.weight", "right_proj.weight",
                 "left_gate.weight", "right_gate.weight", "out_gate.weight",
                 "to_out_norm.weight", "to_out_norm.bias",
                 "to_out.weight")
                All weights are float32.
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  fused LN + projection + mask + sigmoid gates
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32

    # allocate outputs of the fused kernel
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # fused projection weight: 5·H rows, each of length dim
    proj_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()           # shape [5·H, dim]

    # optional mask
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # kernel launch configuration
    BLOCK_R = 64          # rows per program
    BLOCK_D = 64          # input‑dim tile
    BLOCK_H = 16          # hidden‑tile (5·BLOCK_H ≤ 80 ≤ 128)
    grid = (triton.cdiv(R, BLOCK_R),)

    ln_proj_gate_kernel[grid](
        x_flat,
        mask_ptr,
        left,
        right,
        out_gate,
        w["norm.weight"],
        w["norm.bias"],
        proj_w,
        R,
        H,
        N,
        dim,
        eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del x_flat, proj_w, mask_ptr

    # --------------------------------------------------------------
    # 2️⃣  batched GEMM: left @ rightᵀ   (FP16)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 3️⃣  row‑wise stats (mean / rsqrt‑var over hidden dim)
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32            # hidden ≤ 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 4️⃣  final LayerNorm + out_gate + projection to `dim`
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
