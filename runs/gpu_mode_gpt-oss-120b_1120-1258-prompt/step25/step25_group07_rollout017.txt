```
@triton.jit
def norm_proj_kernel(
    x_ptr,                # [R, dim] FP32
    out_ptr,              # [R, out_dim] FP16
    weight_ptr,           # [dim, out_dim] FP16 (scaled by gamma)
    bias_ptr,             # [out_dim] FP16
    rows, dim, out_dim, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)
    mask_m = offs_m < rows

    # Compute per-row mean and variance
    # 1st pass over K tiles to compute sum and sum_sq
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for k_off in range(0, dim, BLOCK_K):
        k_range = k_off + tl.arange(0, BLOCK_K)
        mask_k = k_range < dim
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + k_range[None, :], mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute GEMM: out = ((x - mean)*rstd) @ weight + bias
    # Initialize accumulator
    acc = tl.zeros((BLOCK_R, BLOCK_C), dtype=tl.float32)

    for k_off in range(0, dim, BLOCK_K):
        k_range = k_off + tl.arange(0, BLOCK_K)
        mask_k = k_range < dim

        # Load X tile
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + k_range[None, :], mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Center and scale
        x_norm = (x_vals - mean[:, None]) * rstd[:, None]   # shape [BLOCK_R, BLOCK_K]
        # Cast to fp16 for multiply with weight tile
        x_norm_fp16 = x_norm.to(tl.float16)

        # Load weight tile
        w_vals = tl.load(weight_ptr + k_range[:, None] * out_dim + tl.arange(0, BLOCK_C)[None, :], mask=mask_k[:, None] & (tl.arange(0, BLOCK_C) < out_dim)[None, :], other=0.0)  # shape [BLOCK_K, BLOCK_C] FP16

        # Dot product
        acc += tl.dot(x_norm_fp16, w_vals).to(tl.float32)

    # Add bias
    bias = tl.load(bias_ptr + tl.arange(0, BLOCK_C), mask=(tl.arange(0, BLOCK_C) < out_dim), other=0.0)
    acc = acc + bias[None, :]

    # Store
    tl.store(out_ptr + offs_m[:, None] * out_dim + tl.arange(0, BLOCK_C)[None, :], acc.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, BLOCK_C) < out_dim)[None, :])
```

    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)

@triton.jit
def lnorm_gate_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    out_buf_ptr,
    rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j

    # load mu and rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for h_off in range(0, H, BLOCK_K):
        offs_h = h_off + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        diff = (out_vals - mu[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        norm = diff * gamma[None, :] + beta[None, :]
        gated = norm * gate_vals

        # store into flattened [rows, H]
        tl.store(out_buf_ptr + offs_m[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

"""
Optimized “outgoing” TriMul (AlphaFold3) forward‑only implementation.

Algorithm
---------
1️⃣ Fuse the first LayerNorm (over the last dimension) with the five linear
   projections (left, right, left_gate, right_gate, out_gate) into a single
   Triton kernel.  The LayerNorm weight/bias are folded into the projection
   weights, eliminating the intermediate FP16 tensor that the reference
   implementation stores.

2️⃣ Split the fused projection, apply the optional pair‑wise mask and the
   sigmoid gates (left_gate/right_gate/out_gate) – still in Triton, writing
   three tensors `left`, `right` and `out_gate` in layout `[B, H, N, N]` (FP16).

3️⃣ Batched GEMM (`torch.bmm`) computes the core tensor‑wise contraction
   `out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]` (FP16).

4️⃣ Row‑wise mean & rsqrt‑variance of `out` are obtained with a small Triton
   kernel (`rowwise_stats_kernel`).

5️⃣ Apply the second LayerNorm (weight = to_out_norm.weight, bias = to_out_norm.bias)
   and the `out_gate` in a fused kernel (`lnorm_gate_kernel`), producing a
   flattened tensor `[R, H]` (`R = B·N·N`).

6️⃣ The final linear projection (`to_out`) is a standard FP16‑FP16 GEMM
   (`torch.matmul`) that runs on Tensor‑cores.  The result is cast back to
   FP32 and reshaped to `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and target NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fused LayerNorm + five linear projections
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_kernel(
    x_ptr,                 # [R, dim]   FP32 input
    out_ptr,               # [R, O]     FP16 output (O = 5*H)
    w_ptr,                 # [dim, O]   FP16 fused weight (already *norm.weight)
    b_ptr,                 # [O]        FP16 fused bias   (=norm.bias @ W)
    rows, dim, out_dim,
    eps,
    BLOCK_R: tl.constexpr,  # rows per program
    BLOCK_C: tl.constexpr,  # output‑dim tile
    BLOCK_K: tl.constexpr,  # reduction‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # -------------------------------------------------
    # 1. Compute mean & variance of each row (two passes)
    # -------------------------------------------------
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for k_off in range(0, dim, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # -------------------------------------------------
    # 2. GEMM:  out = ((x - mean) * rstd) @ w + b
    # -------------------------------------------------
    for out_start in range(0, out_dim, BLOCK_C):
        offs_o = out_start + tl.arange(0, BLOCK_C)
        mask_o = offs_o < out_dim

        acc = tl.zeros((BLOCK_R, BLOCK_C), dtype=tl.float32)

        for k_off in range(0, dim, BLOCK_K):
            offs_k = k_off + tl.arange(0, BLOCK_K)
            mask_k = offs_k < dim

            # load a tile of X and normalize it
            x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_k[None, :],
                             mask=mask_r[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)          # [BR, BK]
            x_norm = (x_tile - mean[:, None]) * rstd[:, None]   # [BR, BK]
            x_norm_fp16 = x_norm.to(tl.float16)

            # load the corresponding weight tile
            w_tile = tl.load(w_ptr + offs_k[:, None] * out_dim + offs_o[None, :],
                             mask=mask_k[:, None] & mask_o[None, :],
                             other=0.0)                        # [BK, BC]

            # accumulate
            acc += tl.dot(x_norm_fp16, w_tile).to(tl.float32)

        # add bias
        bias_tile = tl.load(b_ptr + offs_o, mask=mask_o, other=0.0)  # FP16
        acc = acc + bias_tile[None, :]

        # write result
        tl.store(out_ptr + offs_r[:, None] * out_dim + offs_o[None, :],
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused tensor, apply optional mask & sigmoids
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                  # [BR]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 scalar
        m_val = m_val[:, None]                                      # broadcast over H
    else:
        m_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # layout helpers
    hidden_stride = N * N                          # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for hidden h = 0

    # iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BR, H_tile]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 -> cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Second LayerNorm + out_gate (produces flattened [R, H] tensor)
# ----------------------------------------------------------------------
@triton.jit
def lnorm_gate_kernel(
    out_ptr,            # [B, H, N, N] FP16
    gate_ptr,           # [B, H, N, N] FP16   (out_gate)
    mu_ptr,             # [R] FP32
    rstd_ptr,           # [R] FP32
    gamma_ptr,          # [H] FP32 (to_out_norm.weight)
    beta_ptr,           # [H] FP32 (to_out_norm.bias)
    out_buf_ptr,        # [R, H] FP16   (gated & normalized)
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BR]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BR]

    for h_off in range(0, H, BLOCK_K):
        offs_h = h_off + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # (out - μ) * rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # FP32

        norm = diff * gamma[None, :] + beta[None, :]
        gated = norm * gate_vals

        # store flattened [R, H]
        tl.store(out_buf_ptr + offs_m[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise positions
    H = hidden_dim
    O = 5 * H                     # concatenated projection dimension

    # ------------------------------------------------------------------
    # Pre‑compute fused projection weights (fold LayerNorm γ & β)
    # ------------------------------------------------------------------
    norm_weight = w["norm.weight"].float()      # [dim]
    norm_bias   = w["norm.bias"].float()        # [dim]

    def prepare_proj(proj_weight):
        # proj_weight : [H, dim] (float32)
        # → weight_T : [dim, H] (FP16)
        w_t = proj_weight.to(torch.float16).t()               # (dim, H)
        # fold γ (norm.weight) into the weight matrix
        w_scaled = w_t * norm_weight[:, None].to(torch.float16)   # (dim, H)
        # fused bias = β @ W  (β shape dim)
        b = torch.matmul(norm_bias, proj_weight.to(torch.float32))    # (H,)
        return w_scaled, b.to(torch.float16)

    left_w,   left_b   = prepare_proj(w["left_proj.weight"])
    right_w,  right_b  = prepare_proj(w["right_proj.weight"])
    lgate_w,  lgate_b  = prepare_proj(w["left_gate.weight"])
    rgate_w,  rgate_b  = prepare_proj(w["right_gate.weight"])
    ogate_w,  ogate_b  = prepare_proj(w["out_gate.weight"])

    fused_weight = torch.cat([left_w, right_w, lgate_w, rgate_w, ogate_w], dim=1).contiguous()
    fused_bias   = torch.cat([left_b, right_b, lgate_b, rgate_b, ogate_b], dim=0).contiguous()

    # ------------------------------------------------------------------
    # 1️⃣ Fused LayerNorm + 5‑proj kernel → [R, O] FP16
    # ------------------------------------------------------------------
    proj = torch.empty(R, O, dtype=torch.float16, device=device)

    BLOCK_R = 32
    BLOCK_C = 64
    BLOCK_K = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    norm_proj_kernel[grid_proj](
        x,
        proj,
        fused_weight,
        fused_bias,
        R,
        dim,
        O,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    del fused_weight, fused_bias  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Split + mask + sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM (core pairwise contraction)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate → flattened gated tensor [R, H]
    # ------------------------------------------------------------------
    gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_NORM = 128
    BLOCK_K_NORM = 32
    grid_norm = (triton.cdiv(R, BLOCK_R_NORM),)
    lnorm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        gated,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_NORM,
        BLOCK_K=BLOCK_K_NORM,
        num_warps=8,
    )
    # free large temporaries no longer needed
    del out, out_gate, mu, rstd, left, right

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (TensorCore GEMM) + reshape
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]
    final_fp16 = torch.matmul(gated, w_out_T)                         # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # [B, N, N, dim]

    return final
