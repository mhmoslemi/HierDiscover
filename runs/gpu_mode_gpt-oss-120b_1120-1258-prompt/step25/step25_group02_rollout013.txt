"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ LayerNorm on the input pair representation (GPU‑builtin, FP32→FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * splits the fused projection,
   * applies the optional mask,
   * applies sigmoid gates,
   * stores `left`, `right` and `out_gate` in layout [B, N, N, H]
     (H = hidden_dim) – hidden dim is contiguous for the next steps.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   producing a tensor `[B, H, N, N]` (half‑precision).
5️⃣ Row‑wise LayerNorm, multiplication by the output gate and the final
   linear projection are expressed with Torch’s highly‑optimized kernels:
   * mean / var reduction over `H`,
   * element‑wise LN (γ,β) and out‑gate multiplication,
   * a final half‑precision GEMM `out_gate_norm @ to_out.weightᵀ`.
   The result is cast back to float‑32.

Only the split‑&‑gate step is handwritten in Triton – the rest
leverages cuBLAS / Torch kernels which on H100 outperform a pure
Triton implementation for the large batched matrix multiplies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,               # [R, 5·H]   fused projection (FP16)
    left_ptr,               # [B, N, N, H]  output left   (FP16)
    right_ptr,              # [B, N, N, H]  output right  (FP16)
    out_gate_ptr,           # [B, N, N, H]  out‑gate     (FP16)
    mask_ptr,               # [R] (optional, FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from the flat index r ----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # base offset in the destination tensors: each (b,i,k) stores H values contiguously
    base = ((b * N + i) * N + k) * H                     # [BLOCK_R]

    # ---- optional mask (broadcast over H) ----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                     # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- iterate over hidden dimension blocks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of fused projection: [ left | right | left_gate | right_gate | out_gate ]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 for stability, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate    # no mask on out_gate

        # destination offsets for the contiguous [B,N,N,H] layout
        dst_off = base[:, None] + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_off, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, outg,  mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim       = cfg["dim"]
    hidden    = cfg["hidden_dim"]
    nomask    = cfg.get("nomask", True)
    eps       = 1e-5
    device    = x.device
    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (GPU‑builtin) + fused 5‑project linear
    # ------------------------------------------------------------------
    # Row‑wise LN over the last dimension (dim) – stays in FP32 for stability
    x_norm = F.layer_norm(x,
                          (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)                     # [B, N, N, dim] FP32
    x_norm = x_norm.to(torch.float16)

    # concatenate the five weight matrices (shape [5·H, dim])
    fused_w = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"],
    ], dim=0).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                # [B, N, N, 5·H] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Split, mask & sigmoid gates  → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, N, N, H, dtype=torch.float16, device=device)
    right     = torch.empty(B, N, N, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)               # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)              # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16
    del left_perm, right_perm, left_mat, right_mat

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise LN (over H) + out_gate + final linear
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1, keepdim=True)                # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                          # [B,1,N,N]

    # LN affine parameters (γ,β) for the hidden dim
    gamma = w["to_out_norm.weight"].to(torch.float16).view(1, H, 1, 1)
    beta  = w["to_out_norm.bias"  ].to(torch.float16).view(1, H, 1, 1)

    norm = ((out - mu) * rstd).to(torch.float16)           # [B,H,N,N] FP16
    norm = norm * gamma + beta                           # affine LN

    # bring hidden dim back to last axis to match out_gate layout
    norm = norm.permute(0, 2, 3, 1).contiguous()           # [B,N,N,H] FP16
    gated = norm * out_gate                               # element‑wise gate

    # final projection: (R, H) @ (H, dim) → (R, dim)
    final = F.linear(gated.view(R, H), w["to_out.weight"].to(torch.float16))  # FP16
    final = final.to(torch.float32).view(B, N, N, dim)  # back to FP32

    return final
