def norm_gate_kernel(
    out_ptr, gate_ptr,  # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,  # [R] FP32
    gamma_ptr, beta_ptr, # [H] FP32
    out_gated_ptr,     # [R, H] FP16
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute hidden stride etc.
    hidden_stride = N * N
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base_offset = b * (H * hidden_stride) + i * N + j  # start offset for hidden plane 0

    # load mu,rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # Load gamma & beta for hidden - we'll broadcast across rows
    # It might be loaded per block of K.

    # accumulator for output gated values (FP16) shape [BLOCK_M, BLOCK_K]
    out_gated = tl.zeros([BLOCK_M, BLOCK_K], dtype=tl.float16)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out values (FP16)
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load gate values
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load LN gamma,beta (FP32)
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)    # [K]

        # compute norm: (out - mu) * rstd * gamma + beta
        # convert to FP32 for high accuracy, then back to FP16
        out_fp32 = out_vals.to(tl.float32)
        gate_fp32 = gate_vals.to(tl.float32)

        diff = (out_fp32 - mu[:, None]) * rstd[:, None]  # [M,K] FP32
        norm = diff * gamma[None, :] + beta[None, :]     # [M,K] FP32
        gated = norm * gate_fp32                         # [M,K] FP32

        out_gated[:, :] = gated.to(tl.float16)   # store
        # store to output pointer stride: out_gated_ptr [R, H] row-major.
        # pointer: out_gated_ptr + offs_m[:, None] * H + offs_k[None, :]
        dst_ptr = out_gated_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, out_gated,
                 mask=mask_m[:, None] & mask_k[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated implementation

Algorithm
---------
1️⃣  LayerNorm on the input pair‑tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM:  (R, dim) × (5·H, dim)ᵀ  → (R, 5·H)   (FP16).
    R = B·N·N  (number of pair positions),  H = hidden_dim.
3️⃣  Split the fused projection, apply the optional mask and sigmoid gates
    in a Triton kernel, producing three tensors
    `left`, `right` and `out_gate` of shape [B, H, N, N] (FP16).
4️⃣  Batched GEMM (`torch.bmm`) computes the core Einsum
    out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]  →  FP16.
5️⃣  Row‑wise mean / variance of `out` over the hidden dimension H
    (another Triton kernel) → μ, rstd (FP32, one per pair position).
6️⃣  A second Triton kernel normalises `out` with the learned LN γ/β,
    multiplies by `out_gate` and writes the result into a flat
    [R, H] buffer (`out_norm_gate`).
7️⃣  The final projection is a huge GEMM performed by cuBLAS:
    (R, H) × (dim, H)ᵀ  →  (R, dim) (FP16).  The result is reshaped to
    [B, N, N, dim] and cast to FP32.

Only the mask‑aware split, the statistics kernel and the
normalisation+gate kernel are written in Triton; the large matrix
multiplications are delegated to cuBLAS (torch.nn.functional.linear /
torch.bmm), which is significantly faster than a pure‑Triton
implementation for the dimensions used in AlphaFold‑3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2) Row‑wise mean & rsqrt‑variance over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 3) Normalise `out` with the second LayerNorm, apply `out_gate`,
#    and write the gated hidden vector into a flat [R, H] buffer.
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,      # [B, H, N, N] FP16
    gate_ptr,     # [B, H, N, N] FP16  (out_gate)
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    ln_w_ptr,     # [H]          FP32  (to_out_norm.weight)
    ln_b_ptr,     # [H]          FP32  (to_out_norm.bias)
    out_norm_gate_ptr,   # [R, H] FP16
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw values and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ / β) – FP32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β  → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
        gated = norm * gate_vals                                    # [M, K]

        # write into flat buffer [R, H]
        dst_ptr = out_norm_gate_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# 4) Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (FP32 → FP16)
    # ------------------------------------------------------------------
    # use the highly‑optimised cuDNN LayerNorm and then cast to FP16
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 · hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # (R, dim) × (5·H, dim)ᵀ → (R, 5·H)
    proj = F.linear(x_norm.view(R, dim), fused_w)   # FP16
    del x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and compute sigmoid gates
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Normalise + out‑gate → flat [R, H] buffer
    # ------------------------------------------------------------------
    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    grid_norm = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        out_norm_gate,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)      # [dim, H]
    final = F.linear(out_norm_gate, w_out)           # (R, dim) FP16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
