@triton.jit
def ln_gate_kernel(
    out_ptr, gate_ptr, ln_w_ptr, ln_b_ptr, out_ptr_norm, R, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # compute per row sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass compute sum and sum_sq over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out chunk
        out_chunk = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # shape [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(out_chunk, axis=1)
        sum_sq += tl.sum(out_chunk * out_chunk, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # second pass: compute normalized and gated
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        out_chunk = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)
        gate_chunk = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # load ln weight & bias
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # FP32
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # broadcast mu,rstd per row dimension
        norm = (out_chunk - mu[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]
        gated = norm * gate_chunk   # still FP32

        # store as FP16
        tl.store(out_ptr_norm + offs_m[:, None] * H + offs_k[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])
```

        weights["norm.bias"],
        R,

        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,

    # free

        [

            weights["right_proj.weight"],

    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

"""
TriMul (outgoing) – forward‑only implementation.
------------------------------------------------
The kernel implements the “outgoing” TriMul operation from AlphaFold‑3.
It fuses most elementwise work in Triton kernels and uses cuBLAS (torch.bmm /
torch.nn.functional.linear) for the two dense matrix multiplies:

1️⃣ Row‑wise LayerNorm over the last channel (dim) → FP16.
2️⃣ One fused linear projection (5 × hidden_dim) for
   left/right projections, left/right gates and the out‑gate.
3️⃣ Split‑and‑gate Triton kernel produces left, right and out_gate
   tensors (shape [B, H, N, N]).
4️⃣ Batched GEMM (cuBLAS) contracts left & right:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
5️⃣ Triton kernel reads the contracted tensor (now [B,H,N,N]),
   computes the row‑wise mean/variance, applies the second LayerNorm
   (γ/β), multiplies by the out‑gate and writes a gated tensor of shape
   [R, H] (R = B·N·N).  Hidden‑dimension accesses are made contiguous by
   a permute+contiguous copy.
6️⃣ Final projection (cuBLAS) maps the gated tensor to the model dimension:
       out_final = gated @ to_out.weight.T   (fp16·fp32 → fp32)
   and reshapes to [B, N, N, dim].

All kernels run on Triton 3.3.1 (H100) and use mixed precision
(FP16 for the bulk of the data, FP32 for reductions and LayerNorm
statistics).  The implementation works with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] input FP32
    y_ptr,            # [R, dim] output FP16
    w_ptr,            # [dim]   LayerNorm weight γ (FP32)
    b_ptr,            # [dim]   LayerNorm bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # offset of the element (h = 0) for each row
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view into the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ LayerNorm + out‑gate (row‑wise) → gated tensor [R, H] (FP16)
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_kernel(
    out_ptr,          # [R, H]  contracted tensor (FP16)
    gate_ptr,         # [R, H]  out_gate   (FP16)
    ln_w_ptr,         # [H]    γ (FP32)
    ln_b_ptr,         # [H]    β (FP32)
    gated_ptr,        # [R, H] gated output (FP16)
    rows, hidden, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, hidden, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = offs_k < hidden

        ptr = out_ptr + offs_m[:, None] * hidden + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / hidden
    var  = sum_sq / hidden - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: LayerNorm + gate -----
    for k_off in range(0, hidden, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # load raw tensors
        out_vals  = tl.load(out_ptr  + offs_m[:, None] * hidden + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M, K]
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M, K]

        # LayerNorm affine parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # (out - μ) * rstd -> normalized
        norm = (out_vals - mu[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]                # apply γ,β

        gated = norm * gate_vals                                   # multiply by out_gate

        tl.store(gated_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    device = torch.device('cuda')
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim]  FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)               # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)             # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                     # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm + out_gate (row‑wise) → gated tensor [R, H] (FP16)
    # ------------------------------------------------------------------
    # make hidden dimension contiguous for fast Triton loads
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]
    del out, out_gate

    out_flat = out_perm.view(R, H)      # [R, H] FP16, contiguous in H
    gate_flat = gate_perm.view(R, H)    # [R, H] FP16
    del out_perm, gate_perm

    gated = torch.empty(R, H, dtype=torch.float16, device=device)  # [R, H]

    BLOCK_M_G = 128
    BLOCK_K_G = 128   # hidden_dim ≤ 128 ⇒ fits in one tile
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_G),)
    ln_gate_kernel[grid_ln_gate](
        out_flat,
        gate_flat,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        gated,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_G,
        BLOCK_K=BLOCK_K_G,
        num_warps=4,
    )
    del out_flat, gate_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (cuBLAS) → [B, N, N, dim] (FP32)
    # ------------------------------------------------------------------
    final = F.linear(gated, w["to_out.weight"])      # [R, dim]  FP32
    final = final.view(B, N, N, dim)                 # [B, N, N, dim] FP32
    return final
