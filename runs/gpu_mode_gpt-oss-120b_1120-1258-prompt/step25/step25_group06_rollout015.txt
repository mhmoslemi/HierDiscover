    left_raw = proj[:, o0:o1]
    right_raw = proj[:, o1:o2]
    left_gate_raw = proj[:, o2:o3]
    right_gate_raw = proj[:, o3:o4]
    out_gate_raw = proj[:, o4:o5]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_raw * mask_flat * left_gate
    right = right_raw * mask_flat * right_gate
    # out_gate stays as is (no mask)

    # reshape to [B,H,N,N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_gate_raw, right_gate_raw, out_gate_raw, left_gate, right_gate, left_raw, right_raw

    # 4. Batched GEMM over hidden dimension
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) FP16
    out = out.view(B, H, N, N)  # (B, H, N, N) FP16

    # 5. Second LayerNorm, out_gate, final projection
    # flatten for LayerNorm across hidden dim
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)

    # LayerNorm over hidden dimension
    ln_w = weights["to_out_norm.weight"].to(torch.float16)
    ln_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(out_flat, (H,), weight=ln_w, bias=ln_b, eps=eps)  # FP16

    # Gating
    gated = out_norm * out_gate_flat

    # Final linear projection
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    final_flat = torch.matmul(gated, w_out_T)  # (R, dim) FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.
-----------------------------------------------------------------
Algorithm (all heavy work is done in FP16, only reductions use FP32):
1️⃣  Row‑wise LayerNorm on the flattened pair representation (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
    single GEMM → tensor (R,5·H) where R = B·N·N.
3️⃣  Split that fused tensor, apply the optional pair mask, compute the sigmoid gates
    and store three tensors (left, right, out_gate) in layout [B, H, N, N] (FP16) via a
    tiny Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B, H, N, N] (FP16)
5️⃣  Row‑wise LayerNorm on the contracted tensor (cuDNN), element‑wise out‑gate
    multiplication, and the final projection (cuBLAS) → [B, N, N, dim] (float32).

All kernels target Triton 3.3.1 and an NVIDIA H100 (mixed‑precision, 8 warps).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        # mask scalar per pair (broadcast over H)
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between hidden planes
    hidden_stride = N * N
    # base offset inside the [B,H,N,N] tensors (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32, back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate                     # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"      (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) pairs
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (cuDNN)  –  FP32 → FP16
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                    # [R, dim] FP32
    # torch layer‑norm computes mean/var in FP32 internally
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                        # [R, dim] FP16
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]
    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate (Triton)
    # --------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate, final projection (cuDNN + cuBLAS)
    # --------------------------------------------------------------
    # flatten to [R, H] where rows correspond to (b,i,j)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)        # [R, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]

    # LayerNorm over the hidden dimension (cuDNN, weights cast to FP16)
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                             # FP16

    # Apply out‑gate
    gated = out_norm * out_gate_flat                               # [R, H] FP16

    # Final linear projection (FP16 × FP16 → FP32)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]
    final_flat = torch.matmul(gated, w_out_T)                         # [R, dim] FP16

    # reshape to the original 4‑D layout and cast to float32
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
