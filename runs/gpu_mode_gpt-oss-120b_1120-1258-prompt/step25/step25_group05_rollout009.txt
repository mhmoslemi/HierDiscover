def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ Input LayerNorm (FP32)
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    # Cast to FP16 for linear operations
    x_norm = x_norm_fp32.half()

    # 2️⃣ Fuse linear projections
    # Build fused weight matrix (5*H, dim) in half
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # (R, 5*H) half

    # 3️⃣ split fusing
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # 4️⃣ batched GEMM
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H, N, N] half
    out = out.view(B, H, N, N)  # [B, H, N, N] half

    # 5️⃣ LayerNorm over hidden dim H and gating + final projection using torch functions
    # flatten to (R, H)
    out_flat = out.reshape(R, H)
    out_gate_flat = out_gate.reshape(R, H)

    # Apply layer norm (float32 weight/bias)
    # using torch.nn.functional.layer_norm on half tensor => internal upcast to float32
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # returns half (same dtype as input). Internally compute in fp32.

    # Apply gating
    gated = out_norm * out_gate_flat

    # Final projection: (R, H) * (dim, H).T
    final_flat = torch.nn.functional.linear(gated, w["to_out.weight"].to(torch.float16))  # (R, dim) half

    final = final_flat.view(B, N, N, dim).float()  # convert to float32 output

    return final

"""
Highly‑optimized forward‑only "outgoing" TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣ Input LayerNorm (float32 → float16) using PyTorch (fast fused kernel).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single GEMM:   proj = X_norm @ W_fused.T   (R × dim) × (dim × 5·H) → (R × 5·H)
3️⃣ Split the fused projection, apply the optional pair mask, compute the three
   sigmoid gates and store the tensors *left*, *right* and *out_gate* in the
   layout [B, H, N, N] (FP16) – a single Triton kernel.
4️⃣ Batched GEMM across the hidden dimension:
   out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   (implemented with torch.bmm, i.e. cuBLAS).
5️⃣ LayerNorm over the hidden dimension *H*, multiply by the *out_gate*,
   and a final linear projection back to *dim*.
   The two operations are done with native PyTorch kernels (layer_norm + linear)
   which internally run in FP32 for statistics but keep the bulk data in FP16.
6️⃣ Return the result in float32 shape [B, N, N, dim].

Only the split‑&‑gate step is written in Triton; all heavy matrix‑multiplications
are delegated to cuBLAS / Torch for maximum throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcasted over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim chunk per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k   (i = row‑index, k = column‑index)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
        m_val = m_val[:, None]                                         # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride for one hidden slice in the output tensors
    hidden_stride = N * N                # stride between successive H planes

    # Base offset for (b,i,k) in the output tensors (H‑plane 0)
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: layout is [left | right | lg | rg | og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Half‑precision sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate                     # gate is NOT masked

        # Destination offsets for the three output tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape               # x is [B, N, N, dim]
    R = B * N * N                       # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly tuned; it works directly on fp32.
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half()                     # [R, dim] after flattening later

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # shape [5·H, dim]

    # reshape to 2‑D (R, dim) for the GEMM
    x_norm_flat = x_norm.reshape(R, dim).contiguous()
    proj = F.linear(x_norm_flat, fused_w)            # (R, 5·H)  FP16
    del x_norm, x_norm_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates (Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask is broadcasted over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128        # rows processed per program
    BLOCK_H = 64         # hidden‑dim chunk per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension, gate, final projection
    # ------------------------------------------------------------------
    # Move hidden dimension to the last axis so we can use torch.nn.functional.layer_norm
    out_perm      = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H] FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16

    # Fuse LayerNorm (γ,β) and the out‑gate
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                 # FP16, statistics computed in FP32

    gated = out_norm * out_gate_perm   # (B, N, N, H) FP16

    # Final projection back to `dim`
    gated_flat = gated.view(R, H)                               # [R, H] FP16
    final_flat = F.linear(gated_flat, w["to_out.weight"].to(torch.float16))  # [R, dim] FP16
    final = final_flat.view(B, N, N, dim).float()               # back to FP32

    return final
