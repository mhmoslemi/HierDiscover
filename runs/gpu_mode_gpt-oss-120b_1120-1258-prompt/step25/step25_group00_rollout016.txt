        weights["right_proj.weight"],

    rstd = torch.empty(R, dtype=torch.float32, device=device)

        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,

    # Permute out to (B,N,N,H) contiguous

    rstd_fp32 = rstd.view(B, N, N).unsqueeze(-1)

    beta = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    # Broadcast: shape (B,N,N,H)
    norm = norm * gamma + beta

    # Apply out_gate
    gated = norm * out_gate_perm  # fp16 * fp16 => fp16? norm is fp32? Actually norm is FP32; we can cast to fp16 before multiply:
    gated = (norm.half()) * out_gate_perm  # shape (B,N,N,H) fp16

    # Flatten to (R,H)
    gated_flat = gated.view(R, H)

    # Final linear projection
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # (H, dim)
    final_fp16 = torch.nn.functional.linear(gated_flat, w_out_T)  # (R, dim) fp16
    final = final_fp16.view(B, N, N, dim).float()

    # Clean up
    del out_perm, out_gate_perm, out_fp32, norm, gated, gated_flat, w_out_T, mu, rstd, out_gate_tensor

    return final

"""
Implementation of the outgoing TriMul operator (AlphaFold3) using
- torch.nn.functional.layer_norm for the first LayerNorm,
- a single fused linear projection (left, right, gates) in FP16,
- element‑wise mask / sigmoid gating in PyTorch,
- batched GEMM via torch.bmm for the core N³ operation,
- a Triton kernel to compute per‑pair mean/rsqrt‑variance over the hidden dimension,
- a final FP16‑FP32 GEMM (to_out) performed with torch.nn.functional.linear.

All heavy matrix multiplications are delegated to cuBLAS/Tensor Cores; the only
custom kernel is the row‑wise statistics reduction. The implementation works
for arbitrary batch size (1‑2), sequence lengths up to 1024, and hidden sizes
fixed to 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise mean and rsqrt‑variance over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32) – use torch built‑in for speed
    # --------------------------------------------------------------
    x_flat = x.view(R, dim)   # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                         # FP32
    x_norm = x_norm_fp32.half()                 # FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection to (left, right, gates) – single matmul
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                   # [R,5,H]

    left_raw   = proj[:, 0, :]   # [R, H]
    right_raw  = proj[:, 1, :]   # [R, H]
    left_gate_raw  = proj[:, 2, :]   # [R, H]
    right_gate_raw = proj[:, 3, :]   # [R, H]
    out_gate_raw   = proj[:, 4, :]   # [R, H]

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)           # [R]
        mask_exp = mask_flat.unsqueeze(1)                    # [R,1]
        left_raw  = left_raw  * mask_exp
        right_raw = right_raw * mask_exp

    left_gate = torch.sigmoid(left_gate_raw)   # [R, H]
    right_gate = torch.sigmoid(right_gate_raw) # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)     # [R, H]

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N] (contiguous layout)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate_tensor = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Core N³ multiplication (batched GEMM) – torch.bmm uses cuBLAS
    # -----------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.reshape(B * H, N, N)                # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # free memory we no longer need
    del left, right, left_mat, right_mat, proj, left_raw, right_raw
    del left_gate_raw, right_gate_raw, out_gate_raw, left_gate, right_gate

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (Triton)
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64                     # H = 128 → two tiles
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + projection (torch + cuBLAS)
    # --------------------------------------------------------------
    # Permute to (B, N, N, H) so H is the innermost/contiguous dim
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] fp16
    out_gate_perm = out_gate_tensor.permute(0, 2, 3, 1).contiguous()  # same layout

    # reshape statistics to broadcast over H
    mu_view = mu.view(B, N, N).unsqueeze(-1)                # [B, N, N, 1] fp32
    rstd_view = rstd.view(B, N, N).unsqueeze(-1)            # [B, N, N, 1] fp32

    # to_out_norm parameters (γ, β)
    gamma = w["to_out_norm.weight"].to(torch.float16)      # [H]
    beta  = w["to_out_norm.bias"].to(torch.float16)        # [H]

    # LayerNorm + scaling + bias (still in FP16)
    norm = ((out_perm.float() - mu_view) * rstd_view)      # FP32
    norm = norm * gamma + beta                             # FP16 after broadcast

    # Apply out gate
    gated = (norm.half()) * out_gate_perm                  # FP16, shape (B,N,N,H)

    # Flatten to (R, H) for final projection
    gated_flat = gated.view(R, H)                           # [R, H] FP16

    # Final linear projection (to_out) – weight shape (dim, H)
    w_out = w["to_out.weight"].to(torch.float16)            # [dim, H]
    final_fp16 = F.linear(gated_flat, w_out)                # [R, dim] FP16

    # Reshape to (B, N, N, dim) and cast to float32 (required output dtype)
    final = final_fp16.view(B, N, N, dim).float()

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return final

"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  First LayerNorm is performed with the AT‑native `torch.nn.functional.layer_norm`
    (FP32 → FP16) – this is heavily optimized on H100.
2️⃣  The five linear maps (left, right, left_gate, right_gate, out_gate) are
    fused into a single FP16 GEMM (`torch.nn.functional.linear`) that produces a
    tensor of shape (R, 5·H) where R = B·N·N.
3️⃣  A pure‑PyTorch pass splits the fused projection, applies the optional mask,
    evaluates the three sigmoid gates and reshapes the results to
    `[B, H, N, N]`.
4️⃣  The core N³ operation is a batched GEMM (`torch.bmm`) computing
    `out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`.
5️⃣  A compact Triton kernel computes the per‑pair (B·N·N) mean and rsqrt‑variance
    of `out` across the hidden dimension H.
6️⃣  The second LayerNorm, the `out_gate` multiplication and the final projection
    (`to_out`) are fused in PyTorch:
    * the statistics from (5) are broadcast to the hidden axis,
    * `(out‑μ)·rstd·γ + β` is applied,
    * the result is multiplied by `out_gate`,
    * the resulting `[R, H]` tensor is multiplied by the `to_out` weight
      (`torch.nn.functional.linear`), yielding the final `[B, N, N, dim]` tensor.

All heavy matrix multiplications (steps 2, 4, 6) are executed by cuBLAS/Tensor
Cores (FP16).  The only custom Triton code is the reduction kernel for the
row‑wise statistics.  This design satisfies the requirement of having at
least one kernel while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise mean and rsqrt‑variance over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"          (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) – torch native implementation
    # --------------------------------------------------------------
    x_flat = x.view(R, dim)           # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                 # still FP32
    x_norm = x_norm_fp32.half()       # FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into one FP16 GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]
    proj = F.linear(x_norm, fused_w)  # (R, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask and sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)         # [R, 5, H]

    left_raw   = proj[:, 0, :]        # [R, H]
    right_raw  = proj[:, 1, :]        # [R, H]
    left_gate_raw  = proj[:, 2, :]    # [R, H]
    right_gate_raw = proj[:, 3, :]    # [R, H]
    out_gate_raw   = proj[:, 4, :]    # [R, H]

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)  # [R]
        mask_exp  = mask_flat.unsqueeze(1)          # [R, 1]
        left_raw  = left_raw * mask_exp
        right_raw = right_raw * mask_exp

    left_gate  = torch.sigmoid(left_gate_raw)   # [R, H]
    right_gate = torch.sigmoid(right_gate_raw)  # [R, H]
    out_gate   = torch.sigmoid(out_gate_raw)    # [R, H]

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N] (contiguous layout required for the bmm)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate_tensor = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Core N³ multiplication via batched GEMM (torch.bmm → cuBLAS)
    # -----------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.reshape(B * H, N, N)                # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # free temporary tensors we no longer need
    del left, right, left_mat, right_mat, proj, left_raw, right_raw
    del left_gate_raw, right_gate_raw, out_gate_raw, left_gate, right_gate

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (Triton)
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64                 # H = 128 → two tiles of 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, `out_gate` multiplication, final projection
    # --------------------------------------------------------------
    # Bring hidden dimension to innermost position for contiguous memory access
    out_perm      = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] fp16
    out_gate_perm = out_gate_tensor.permute(0, 2, 3, 1).contiguous()  # same layout

    # Broadcast statistics over the hidden axis
    mu_view   = mu.view(B, N, N).unsqueeze(-1)   # [B, N, N, 1] fp32
    rstd_view = rstd.view(B, N, N).unsqueeze(-1) # [B, N, N, 1] fp32

    # to_out_norm parameters (γ, β)
    gamma = w["to_out_norm.weight"].to(torch.float16)   # [H]
    beta  = w["to_out_norm.bias"].to(torch.float16)     # [H]

    # LayerNorm + scale + bias (still in FP16)
    norm = ((out_perm.float() - mu_view) * rstd_view)   # FP32
    norm = norm * gamma + beta                          # FP16 after broadcast

    # Apply out_gate (both FP16)
    gated = norm.half() * out_gate_perm                 # [B, N, N, H] FP16

    # Flatten to (R, H) for the final linear projection
    gated_flat = gated.view(R, H)                       # [R, H] FP16

    # Final projection: weight shape (dim, H)
    w_out = w["to_out.weight"].to(torch.float16)        # [dim, H]
    final_fp16 = F.linear(gated_flat, w_out)           # [R, dim] FP16

    # Reshape to original layout and cast to float32 (output dtype)
    final = final_fp16.view(B, N, N, dim).float()       # [B, N, N, dim] FP32

    return final
