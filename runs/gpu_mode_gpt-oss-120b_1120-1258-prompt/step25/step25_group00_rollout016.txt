    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm (custom kernel)
    # flatten
    x_flat = x.view(R, dim).contiguous() # FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    # launch custom layernorm kernel
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # 2. fused projection
    fused_w = torch.cat([w["left_proj.weight"], w["right_proj.weight"], w["left_gate.weight"], w["right_gate.weight"], w["out_gate.weight"]], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # shape [R, 5*H] fp16
    del x_norm, fused_w

    # 3. Split, mask, gates
    # mask flatten
    if (not nomask) and mask is not None:
        mask_flat = mask.to(torch.float16).view(R)  # shape [R]
    else:
        mask_flat = None

    # Raw slices
    left_raw = proj[:, :H]
    right_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:]

    # apply mask and sigmoid gates
    if mask_flat is not None:
        mask_flat_unsq = mask_flat.unsqueeze(-1)  # [R,1]
        left = left_raw * mask_flat_unsq * torch.sigmoid(left_gate_raw)
        right = right_raw * mask_flat_unsq * torch.sigmoid(right_gate_raw)
    else:
        left = left_raw * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    out_gate = torch.sigmoid(out_gate_raw)  # no mask
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # reshape to [B,H,N,N]
    # using view and permute
    left = left.view(B, N, N, H).permute(0,3,1,2).contiguous()
    right = right.view(B, N, N, H).permute(0,3,1,2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0,3,1,2).contiguous()
    # optional free memory
    # 4. batched matmul
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H,N,N] fp16
    out = out.view(B, H, N, N)
    del left_mat, right_mat, left, right

    # 5. second LayerNorm over H dimension (row‑wise)

    out_norm = torch.nn.functional.layer_norm(

        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    del out_view

    out_gate_flat = out_gate.permute(0,2,3,1).contiguous().view(R, H)
    out_norm_gate = out_norm * out_gate_flat
    del out_norm, out_gate, out_gate_flat

    # 6. final linear projection

    final = torch.nn.functional.linear(out_norm_gate, w_out_T)  # [R, dim] fp16
    final = final.view(B, N, N, cfg["dim"]).to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last channel (dim) of the 4‑D tensor.
    Implemented with a tiny Triton kernel (FP32 accumulation → FP16 output).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single FP16 GEMM:  y = x_norm @ W_fused.T
    with   W_fused ∈ ℝ^{5·H×dim}.

3️⃣  Split the fused projection, apply the optional mask and the three
    sigmoid gates, and reshape to layout [B, H, N, N] (FP16).

4️⃣  Core TriMul contraction:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    Realised as a batched GEMM using torch.bmm on FP16 tensors
    (cuBLAS tensor‑cores).

5️⃣  Second LayerNorm over the hidden dimension H.  We cast the
    contraction result to FP32, apply Torch’s fused LayerNorm
    (weights are FP32), multiply by the out‑gate (also cast to FP32).

6️⃣  Final linear projection (FP32 weight, FP16→FP32 matmul) and reshape
    to the required output shape `[B, N, N, dim]` (FP32).

Only the first LayerNorm is written in Triton; everything else
leverages highly‑optimised Torch/CUDA kernels, giving a fast forward
pass while keeping the final result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim] FP32 input
    y_ptr,       # [R, dim] FP16 output
    w_ptr,       # [dim]    LN weight γ (FP32)
    b_ptr,       # [dim]    LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)         # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradient).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # flat pair count
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten the pair dimension
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 64       # channels per program
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free original FP32 tensor early

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (5·H × dim) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                         # [R, 5·H] FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates
    # --------------------------------------------------------------
    # slice the fused projection
    left_raw      = proj[:, :H]                     # [R, H]
    right_raw     = proj[:, H:2 * H]
    left_gate_raw = proj[:, 2 * H:3 * H]
    right_gate_raw = proj[:, 3 * H:4 * H]
    out_gate_raw  = proj[:, 4 * H:]                # [R, H]
    del proj

    # mask handling (optional)
    if (not nomask) and mask is not None:
        mask_flat = mask.to(torch.float16).view(R)          # [R]
        mask_flat = mask_flat.unsqueeze(-1)                 # [R, 1] for broadcasting
        left  = left_raw  * mask_flat * torch.sigmoid(left_gate_raw)
        right = right_raw * mask_flat * torch.sigmoid(right_gate_raw)
    else:
        left  = left_raw  * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    out_gate = torch.sigmoid(out_gate_raw)                 # no mask applied
    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # reshape to [B, H, N, N] (contiguous)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM – core TriMul contraction (FP16 tensor cores)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                # [B·H, N, N] FP16
    # out = left_mat @ right_matᵀ   → shape [B·H, N, N] (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16
    del left_mat, right_mat, left, right               # free large temporaries

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise over H) – keep in FP32 for accuracy
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                  # materialise FP32 copy
    # reshape to [R, H] where each row corresponds to (b,i,j)
    out_view = out_fp32.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    out_norm = F.layer_norm(
        out_view,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP32, shape [R, H]
    del out_fp32, out_view, out

    # --------------------------------------------------------------
    # 6️⃣ Apply out_gate, final projection, reshape
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] FP32
    del out_gate

    gated = out_norm * out_gate_fp32                                        # [R, H] FP32
    del out_norm, out_gate_fp32

    # final linear projection: weight shape [dim, H] (FP32)
    final = F.linear(gated, w["to_out.weight"])    # [R, dim] FP32
    final = final.view(B, N, N, dim)                # [B, N, N, dim]
    return final
