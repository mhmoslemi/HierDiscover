```
@triton.jit
def ln_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr,
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j    # start index for hidden=0

    # load per-row mu, rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)  # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # For each hidden block
    for h_off in range(0, H, BLOCK_K):
        offs_h = h_off + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # load out and gate values
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # Load ln_w and ln_b
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # Compute: ((out - mu) * rstd) * gamma + beta
        # Convert to FP32 for arithmetic
        out_fp32 = out_vals.to(tl.float32)
        diff = (out_fp32 - mu[:, None]) * rstd[:, None]          # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]             # [M,K]
        gated = norm * gate_vals.to(tl.float32)                 # [M,K]

        # Store back as FP16
        tl.store(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

@triton.jit
def ln_gate_kernel(
    out_ptr,      # [B, H, N, N] FP16
    gate_ptr,     # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    ln_w_ptr,     # [H] FP32
    ln_b_ptr,     # [H] FP32
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode flat index into b,i,j
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # offset of hidden=0

    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # Loop over hidden dimension
    for h_off in range(0, H, BLOCK_K):
        offs_h = h_off + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # Load out values and gate values
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)

        # Load LN parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # Convert to FP32
        out_fp32 = out_vals.to(tl.float32)
        gate_fp32 = gate_vals.to(tl.float32)

        # Compute normalized and gated value
        diff = (out_fp32 - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]
        gated = norm * gate_fp32

        # Store result back to out_ptr (overwrite)
        tl.store(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – fused‑kernel implementation

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32→FP16) – Triton.
2️⃣ Fuse the five linear maps (left, right, three gates) into one GEMM:
   (R, dim) × (5·H, dim)ᵀ → (R, 5·H)  (FP16, torch.linear).
3️⃣ Split the fused projection, apply the optional pair‑mask, compute sigmoid
   gates and store three tensors *left*, *right* and *out_gate* in
   layout [B, H, N, N] (FP16) – Triton.
4️⃣ Batched GEMM (cuBLAS) implements the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
5️⃣ Row‑wise mean / rsqrt‑variance of *out* over the hidden dimension H –
   Triton (produces μ, ρ).
6️⃣ In‑place LayerNorm + out‑gate on *out* (using μ,ρ, LN γ/β) – Triton.
7️⃣ Final projection (H → dim) with a regular torch.linear on the now
   contiguous tensor [B·N·N, H] (FP16) → [B·N·N, dim] (FP16) and cast to FP32.

All heavy reductions are done on‑GPU with Triton, the two large GEMMs
(left‑right and final projection) use cuBLAS which is far more efficient
than a handcrafted dot‑loop.  The code works for the full test matrix of
shapes (B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384}, H=128) and supports
both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    γ (FP32)
    b_ptr,          # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – mean & variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalized values --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # hidden‑plane 0 offset

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# --------------------------------------------------------------
# 5️⃣ In‑place LayerNorm + out‑gate (uses μ,ρ)
# --------------------------------------------------------------
@triton.jit
def ln_gate_inplace_kernel(
    out_ptr,      # [B, H, N, N] FP16   (will be overwritten)
    gate_ptr,     # [B, H, N, N] FP16   (out_gate)
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    ln_w_ptr,     # [H]          FP32   (to_out_norm.weight)
    ln_b_ptr,     # [H]          FP32   (to_out_norm.bias)
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode flat index (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # hidden‑plane 0 offset

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for h_off in range(0, H, BLOCK_K):
        offs_h = h_off + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # Load raw tensors
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)

        # LN parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # FP32 math
        out_fp = out_vals.to(tl.float32)
        gate_fp = gate_vals.to(tl.float32)

        diff = (out_fp - mu[:, None]) * rstd[:, None]           # (out - μ) * ρ
        norm = diff * gamma[None, :] + beta[None, :]            # LN
        gated = norm * gate_fp                                   # * out_gate

        # Store back (overwrite) in FP16
        tl.store(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 6️⃣ Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # unpack inputs
    # -------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total pair positions
    H = hidden_dim

    # -------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # -------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                              # FP16, shape [R, 5·H]
    del fused_w, x_norm

    # -------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # -------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # -------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # FP16, shape [B·H, N, N]
    out = out.view(B, H, N, N)                                    # [B, H, N, N] FP16

    # Release left/right as they are no longer needed
    del left, right

    # -------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # -------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # -------------------------------------------------
    # 6️⃣ In‑place LayerNorm + out_gate
    # -------------------------------------------------
    BLOCK_M_G = 128
    BLOCK_K_G = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_G),)
    ln_gate_inplace_kernel[grid_ln_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_G,
        BLOCK_K=BLOCK_K_G,
        num_warps=8,
    )
    # Release the gate tensor – no longer needed
    del out_gate, mu, rstd

    # -------------------------------------------------
    # 7️⃣ Final projection (H → dim) using cuBLAS
    # -------------------------------------------------
    # bring hidden dimension to the innermost axis
    out_perm = out.permute(0, 2, 3, 1).contiguous()               # [B, N, N, H]
    out_flat = out_perm.view(R, H)                               # [R, H]

    w_out = w["to_out.weight"].to(torch.float16)                 # [dim, H]
    proj_out = F.linear(out_flat, w_out)                         # [R, dim] FP16

    final = proj_out.view(B, N, N, dim).to(torch.float32)        # cast to FP32
    return final
