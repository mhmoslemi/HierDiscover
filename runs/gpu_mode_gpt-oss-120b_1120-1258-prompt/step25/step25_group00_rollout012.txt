"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣  Layer‑Norm the pair representation (FP32 → FP16) with Torch’s
    highly‑optimised `torch.nn.functional.layer_norm`.
2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into a single GEMM (`torch.nn.functional.linear`).
3️⃣  Split the fused result, apply the optional mask, compute the three
    sigmoid gates and store `left`, `right` and `out_gate` in the
    layout ``[B, H, N, N]`` (FP16) via a tiny Triton kernel.  This kernel
    is the only custom Triton piece, satisfying the “must use a kernel”
    requirement while keeping the hot‑path in native Torch.
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum
    ``out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`` → ``out`` of shape
    ``[B, H, N, N]`` (FP16).
5️⃣  Second Layer‑Norm (over the hidden dimension `H`) is performed with
    Torch’s `layer_norm`, followed by element‑wise multiplication with the
    pre‑computed `out_gate`.
6️⃣  Final projection from `H → dim` uses another `torch.nn.functional.linear`
    (Tensor‑core GEMM).  The output is cast back to FP32 with shape
    ``[B, N, N, dim]``.

All heavy linear algebra runs in FP16 (Tensor‑cores) while the LayerNorms
are computed in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask (if present) and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]  FP16 fused projection
    left_ptr,          # [B, H, N, N]  FP16
    right_ptr,         # [B, H, N, N]  FP16
    out_gate_ptr,      # [B, H, N, N]  FP16
    mask_ptr,          # [R]   FP16 (optional)
    R,                 # total number of (b,i,k) triples
    H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """One program works on a tile of rows (R) and a tile of hidden units (H).

    The fused projection layout is interpreted as:
        [ left | right | left_gate | right_gate | out_gate ]   (each of size H)

    The kernel writes the three tensors in the layout [B, H, N, N] where the
    hidden dimension H is the fastest‑moving axis (contiguous in memory).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode the flat index r = b·N·N + i·N + k  --------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ---- optional mask -------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        m = m[:, None]                                          # broadcast over H
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- strides for the destination tensors ---------------------------
    hidden_stride = N * N                # stride for one hidden slice in the flat view
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden slice 0 for this (b,i,k)

    # ---- loop over hidden tiles ----------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # gather the fused projection slice for the current hidden tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 then cast back to FP16 (more accurate)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        # destination offsets in the flattened [B, H, N, N] tensors
        # layout: (b, h, i, k) -> offset = b*(H*N*N) + h*(N*N) + i*N + k
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,      left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,     right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,  outg,      mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float) or None
        - weights_dict : mapping with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict must contain:
            "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] and dtype float32
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16) – use torch's highly‑optimised kernel
    # --------------------------------------------------------------
    # Normalise over the last dimension (dim)
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                 # still float32
    x_norm = x_norm_fp32.to(torch.float16).contiguous()
    del x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·H outputs) – single GEMM in FP16
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape [5·H, dim]

    x_flat = x_norm.view(R, dim)                 # [R, dim]
    proj = F.linear(x_flat, fused_w)             # [R, 5·H]  FP16
    del fused_w, x_flat

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton kernel)
    # --------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        HAS_MASK = 1
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0

    BLOCK_R = 64      # rows per program (tunable)
    BLOCK_H = 64      # hidden units per program (tunable)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum – batched GEMM over the hidden dimension
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over H), apply out_gate, final projection
    # --------------------------------------------------------------
    # bring H to the last dimension for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H] (FP16)

    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                      # still FP16

    # back to [B, H, N, N]
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()

    # element‑wise gate
    out_gated = out_norm * out_gate                         # FP16

    # final linear projection H → dim
    out_flat = out_gated.permute(0, 2, 3, 1).reshape(R, H)   # [R, H] FP16
    final = F.linear(out_flat, w["to_out.weight"].to(torch.float16))   # [R, dim] FP16
    final = final.to(torch.float32).view(B, N, N, dim)       # back to FP32

    return final
