def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim_cfg = config["dim"]
    assert dim == dim_cfg
    R = bs * N * N
    eps = 1e-5

    # LayerNorm
    # blank shape: [R, dim] flatten
    x_flat = input_tensor.reshape(R, dim).contiguous()
    # allocate output dtype half
    x_norm = torch.empty_like(x_flat, dtype=torch.float16)
    # get weight and bias tensors: shape [dim]
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w_norm,
        b_norm,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Fused linear projection
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,

    # split
    H = hidden_dim
    left_proj = proj[:, 0:H]
    right_proj = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:5*H]

    # mask
    if not config.get("nomask", True) and mask is not None:
        mask_flat = mask.reshape(R).to(dtype=torch.float16, device=device).unsqueeze(-1)  # (R,1)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat
    # else mask no effect (multiply by 1)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)  # shape (R, H)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape to [B, H, N, N]
    left = left.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()

    right_mat = right.view(bs * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(bs, H, N, N)

    # second LayerNorm across H dimension per (b,i,j)
    # reshape to (R, H)
    out_reshaped = out.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    out_normed = torch.nn.functional.layer_norm(

        (H,),

        bias=weights["to_out_norm.bias"],
        eps=eps,

    final = final_fp16.to(torch.float32).view(bs, N, N, dim)

    return final

import triton
import triton.language as tl
import torch
import torch.nn.functional as F

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr,
                     rows, dim, eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_vals = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass over channels: compute mean & var
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_vals += tl.sum(vals, axis=1)
        sum_sq   += tl.sum(vals * vals, axis=1)

    mean = sum_vals / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized tensor
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])
```

Outgoing TriMul (AlphaFold‑3) – fast Triton + PyTorch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (last channel) on the 4‑D input.
2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:  (R,dim) @ (dim,5·H).T
   → (R,5·H) where R = B·N·N.
3️⃣ Split the fused projection, optionally mask the two “value” streams,
   apply sigmoid‑gates and reshape to [B,H,N,N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
      out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]      (FP16)
5️⃣ Row‑wise LayerNorm over the hidden dimension H, followed by the
   out‑gate multiplication.
6️⃣ Final linear projection (FP16 GEMM) back to the model dimension
   and cast to float32.

Only the first LayerNorm is written in Triton; the remaining heavy
operations are delegated to highly‑optimized cuBLAS kernels via
torch.nn.functional.linear / torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,               # pointers
    rows, dim, eps,                            # scalars
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Normalises each of the ``rows`` rows of length ``dim``.
    Input  : FP32  [rows, dim]
    Output : FP16  [rows, dim]   (γ, β are FP32)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # pass 1 – compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # pass 2 – write normalised values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]                (float32)
        mask_tensor  : torch.Tensor [B, N, N]  (bool/float) or None
        weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"                              (float32)
        config_dict : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    R = B * N * N                # number of pair positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the last channel (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection to obtain left/right values + three gates
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Slice, (optional) mask and apply sigmoid gates
    # ------------------------------------------------------------------
    # split the fused projection
    left_proj   = proj[:, 0:H]
    right_proj  = proj[:, H:2 * H]
    left_gate_r = proj[:, 2 * H:3 * H]
    right_gate_r= proj[:, 3 * H:4 * H]
    out_gate_r  = proj[:, 4 * H:5 * H]

    # optional mask – broadcast over the hidden dimension
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.reshape(R).to(dtype=torch.float16, device=device).unsqueeze(-1)  # (R,1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # else: implicit mask == 1, nothing to do

    # sigmoid gates (computed in fp16 – sufficiently stable for our use‑case)
    left_gate  = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)            # (R, H)

    # apply gates
    left  = left_proj * left_gate
    right = right_proj * right_gate
    del left_proj, right_proj, left_gate_r, right_gate_r, out_gate_r, left_gate, right_gate

    # reshape to [B, H, N, N] (contiguous layout required by torch.bmm)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # merge batch and hidden dimensions for the bmm call
    left_mat  = left.view(B * H, N, N)            # [B·H, N, N]
    right_mat = right.view(B * H, N, N)           # [B·H, N, N]

    # (B·H, N, N) × (B·H, N, N)ᵀ  →  (B·H, N, N)   (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                    # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (row‑wise)
    #    then multiply by the out‑gate
    # ------------------------------------------------------------------
    # place hidden dimension as the last axis so that torch.layer_norm can be used
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # reshape back to [B, H, N, N]
    out_norm = out_norm.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # element‑wise multiplication with the already‑sigmoided out‑gate
    gated = out_norm * out_gate
    del out, out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to ``dim`` and cast to float32
    # ------------------------------------------------------------------
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]

    # weight: (dim, H) → fp16 for speed
    w_out = w["to_out.weight"].to(dtype=torch.float16, device=device)

    final_fp16 = F.linear(gated_flat, w_out)        # (R, dim)  FP16
    final = final_fp16.to(dtype=torch.float32).view(B, N, N, dim)   # FP32 output

    return final
