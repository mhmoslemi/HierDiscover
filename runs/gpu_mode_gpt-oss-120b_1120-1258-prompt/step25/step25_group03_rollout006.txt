"""
Outgoing TriMul (AlphaFold‑3) – fused Triton implementation

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a single
   GEMM: (R, dim) × (5·H, dim)ᵀ → (R, 5·H) (FP16).  R = B·N·N.
3️⃣ Split the fused projection, apply optional mask, compute sigmoid gates and store three
   buffers **left**, **right**, **out_gate** in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, k, j]
   →  out ∈ FP16 of shape [B, H, N, N].
5️⃣ Row‑wise mean & rsqrt‑variance of ``out`` over the hidden dimension H.
6️⃣ Fuse second LayerNorm, the ``out_gate`` multiplication and the final projection
   into a single Triton kernel, writing the final result `[B, N, N, dim]` in float32.

All kernels target Triton 3.3.1 on an NVIDIA H‑100, use mixed‑precision (FP16 for bulk
computations, FP32 for reductions) and avoid any synchronization points.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      γ (FP32)
    b_ptr,          # [dim]      β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – compute mean & variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalized values --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16  (1.0 for no‑mask)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # flat index → b,i,k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # broadcast mask (if any) over hidden dim
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | lgate | rgate | ogate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # H is always 128 → we can take the whole hidden dim in one iteration
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row LN stats
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm γ/β for hidden channels
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
        term = norm * gate_vals                                    # [M, K]

        term_fp16 = term.to(tl.float16)   # back to FP16 for dot

        # load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        # accumulate dot product (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])

# 5️⃣ Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                # total pair positions
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                         # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # FP16 result

    BLOCK_M = 256
    BLOCK_D = 128
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    # (R, dim) × (5·H, dim)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, fused_w)          # FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Always pass a mask tensor; for the “no‑mask” case we use all‑ones.
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
    else:
        mask_fp16 = torch.ones(R, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_fp16,
        R,
        H,
        N,
        HAS_MASK=1,               # mask is always present (all‑ones when unused)
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_fp16

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # -----------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)          # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16
    out = out.view(B, H, N, N)                     # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # -----------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128        # hidden_dim is 128 → single‑iteration reduction
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # -----------------------------------------------------------------
    # transpose final weight for easier indexing: [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (tiles over R)
    BLOCK_K_OUT = 32       # hidden‑dim chunk (trade‑off between regs & occupancy)
    BLOCK_N_OUT = 128      # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
