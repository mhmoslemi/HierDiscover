```python
@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    out_norm_gate_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode indices
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N
    hidden_stride = N * N
    base_out_offset = b * (H * hidden_stride) + i * N + j  # offset for h=0
    # out_norm_gate base offset for storage: each row start * H
    out_gate_base = offs_m * H

    # --- first pass: compute mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + base_out_offset[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_H]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals*vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # second pass: apply LN, gate and store
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # Load out and gate values
        out_vals = tl.load(
            out_ptr + base_out_offset[:, None] + offs_h[None,:] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None,:],
            other=0.0
        ).to(tl.float32)  # [BLOCK_M, BLOCK_H]

        gate_vals = tl.load(
            gate_ptr + base_out_offset[:, None] + offs_h[None,:] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None,:],
            other=0.0
        ).to(tl.float32)  # [BLOCK_M, BLOCK_H]

        # LN parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

        # Normalize
        norm = (out_vals - mu[:, None]) * rstd[:, None]  # [M, H]
        ln_out = norm * gamma[None, :] + beta[None, :]    # [M, H]
        gated = ln_out * gate_vals  # [M, H]

        # store to contiguous layout
        dst_ptr = out_norm_gate_ptr + out_gate_base[:, None] + offs_h[None, :]
        tl.store(dst_ptr, gated.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair representation.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM producing a tensor of shape (R, 5·H) where
   R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply the optional mask, compute the three
   sigmoid gates and store the three tensors `left`, `right` and `out_gate`
   in layout [B, H, N, N] (FP16) via a Triton kernel.
4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   yielding a tensor `[B, H, N, N]` (FP16).
5️⃣ Row‑wise LayerNorm + out_gate (both in half‑precision) → a contiguous
   `[R, H]` buffer (`norm_gate`) using a single Triton kernel.
   This kernel also computes per‑row statistics (mean/rsqrt‑variance) on‑the‑fly.
6️⃣ Final projection is a large GEMM:
       result = norm_gate @ (to_out.weight)ᵀ          (FP16·FP16 → FP16)
   and is cast back to FP32 and reshaped to `[B, N, N, dim]`.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,          # [R, D]  FP32 → FP16
    w_ptr, b_ptr,          # [D]     LN γ,β  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                     # [R, 5·H]  FP16 fused projection
    left_ptr, right_ptr,          # [B, H, N, N]  FP16
    out_gate_ptr,                 # [B, H, N, N]  FP16
    mask_ptr,                     # [R]  FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask (broadcast over H) – 1.0 when no mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)                # [BLOCK_R]
        m_val = m_val[:, None]                       # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for H‑dimension slice (h = 0)
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the concatenated weight matrix (left | right | left_gate | right_gate | out_gate)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (FP32 for stability → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate                     # already sigmoid‑ed

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise LN + out_gate (produces contiguous [R, H] buffer)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr,          # [B, H, N, N] FP16
    ln_w_ptr, ln_b_ptr,        # [H] FP32
    dst_ptr,                    # [R, H] FP16  (R = B·N·N)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # decode (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j       # offset for h = 0
    dst_base = offs_m * H                                 # start of a row in dst

    # ---- first pass: mean & variance across hidden dimension ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: apply LN, multiply by out_gate and write contiguous ----
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw values
        out_vals  = tl.load(
            out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0).to(tl.float32)          # [M, BLOCK_H]

        gate_vals = tl.load(
            gate_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0).to(tl.float32)          # [M, BLOCK_H]

        # LN parameters (γ,β) for this slice
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

        # (out - μ) * rstd → γ· + β → * out_gate
        norm = (out_vals - mu[:, None]) * rstd[:, None]            # [M, BLOCK_H]
        ln_out = norm * gamma[None, :] + beta[None, :]             # [M, BLOCK_H]
        gated = ln_out * gate_vals                                 # [M, BLOCK_H]

        # store to contiguous buffer
        dst = dst_ptr + dst_base[:, None] + offs_h[None, :]        # [M, BLOCK_H]
        tl.store(dst,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of (b,i,j) entries
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection ( 5·hidden_dim )
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LN + out_gate → contiguous [R, H] buffer
    # --------------------------------------------------------------
    norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_H_OUT = 32
    grid_norm = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        norm_gate,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate          # free memory before the big GEMM

    # --------------------------------------------------------------
    # 6️⃣ Final projection (large GEMM) and reshape
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16
    final_fp16 = torch.matmul(norm_gate, w_out_T)                     # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # FP32 result

    return final
