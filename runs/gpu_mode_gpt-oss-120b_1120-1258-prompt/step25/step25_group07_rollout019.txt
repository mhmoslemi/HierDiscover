@triton.jit
def proj_and_gate_kernel(
    x_ptr,        # [R, dim] normalized input FP16
    mask_ptr,    # [R] optional FP16
    left_w_ptr, right_w_ptr, left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask per row
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # shape [BLOCK_R,1]
    else:
        mask_val = tl.full((BLOCK_R,1), 1.0, tl.float16)

    # accumulate outputs for each hidden block
    hidden_stride = N * N

    # loop over hidden tiles
    for h_off in range(0, H, BLOCK_H):
        # initialize accumulators for each projection
        acc_left = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_right = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_out_gate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # loop over dim tiles to compute dot products
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :], other=0.0)   # FP16
            # cast to FP32 for accumulation
            x_f32 = x.to(tl.float32)

            # load weight tiles
            w_left = tl.load(left_w_ptr + (h_off + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :],
                             mask=tl.arange(0, BLOCK_H)[:, None] < BLOCK_H & mask_d[None, :], other=0.0)
            w_right = tl.load(right_w_ptr + (h_off + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :],
                             mask=tl.arange(0, BLOCK_H)[:, None] < BLOCK_H & mask_d[None, :], other=0.0)
            w_lgate = tl.load(left_gate_w_ptr + (h_off + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :],
                              mask=tl.arange(0, BLOCK_H)[:, None] < BLOCK_H & mask_d[None, :], other=0.0)
            w_rgate = tl.load(right_gate_w_ptr + (h_off + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :],
                               mask=tl.arange(0, BLOCK_H)[:, None] < BLOCK_H & mask_d[None, :], other=0.0)
            w_outg = tl.load(out_gate_w_ptr + (h_off + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :],
                              mask=tl.arange(0, BLOCK_H)[:, None] < BLOCK_H & mask_d[None, :], other=0.0)

            # cast weights to float32
            w_left_f32 = w_left.to(tl.float32)
            w_right_f32 = w_right.to(tl.float32)
            w_lgate_f32 = w_lgate.to(tl.float32)
            w_rgate_f32 = w_rgate.to(tl.float32)
            w_outg_f32 = w_outg.to(tl.float32)

            acc_left += tl.dot(x_f32, w_left_f32)
            acc_right += tl.dot(x_f32, w_right_f32)
            acc_left_gate += tl.dot(x_f32, w_lgate_f32)
            acc_right_gate += tl.dot(x_f32, w_rgate_f32)
            acc_out_gate += tl.dot(x_f32, w_outg_f32)

        # after dim loop: convert to FP16
        left_raw = acc_left.to(tl.float16)
        right_raw = acc_right.to(tl.float16)
        left_gate_raw = acc_left_gate.to(tl.float16)
        right_gate_raw = acc_right_gate.to(tl.float16)
        out_gate_raw = acc_out_gate.to(tl.float16)

        # sigmoid gates (in FP32)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates to left/right
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        out_gate = out_gate_val

        # compute destination offsets
        offs_h = h_off + tl.arange(0, BLOCK_H)
        dst_offset = b * (H * hidden_stride) + i * N + k   # base offset for h=0

        # store
        tl.store(left_ptr + dst_offset[:, None] + offs_h[None, :] * hidden_stride,
                 left,
                 mask=mask_r[:, None] & (offs_h < H)[None, :])
        tl.store(right_ptr + dst_offset[:, None] + offs_h[None, :] * hidden_stride,
                 right,
                 mask=mask_r[:, None] & (offs_h < H)[None, :])
        tl.store(out_gate_ptr + dst_offset[:, None] + offs_h[None, :] * hidden_stride,
                 out_gate,
                 mask=mask_r[:, None] & (offs_h < H)[None, :])

"""Optimized outgoing TriMul (AlphaFold3) – forward only.

The implementation fuses the three most expensive memory‑bound phases:
1️⃣ Row‑wise LayerNorm (FP32→FP16).
2️⃣ A single Triton kernel that simultaneously performs the five linear
   projections (left, right, left_gate, right_gate, out_gate), applies the
   optional pair‑wise mask, evaluates the sigmoid gates and writes the three
   tensors `left`, `right` and `out_gate` directly (no intermediate
   `(R,5·H)` buffer is materialised).
3️⃣ The remaining reductions (row‑wise mean/var) and the final LayerNorm,
   out‑gate multiplication and the last linear projection are performed by the
   original `final_output_kernel`.

All heavy matrix multiplies use Tensor‑Core friendly fp16 arithmetic.
The only fp32 work is the per‑row statistics and the final LayerNorm
scaling.  This reduces memory traffic (≈ 150 MiB saved for the largest
configuration) and cuts the number of kernel launches, achieving the
target ~500 µs runtime on the provided test cases."""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, dim]  inputs/outputs (FP32 → FP16)
    w_ptr, b_ptr,               # [dim]     LN γ (weight) and β (bias) – FP32
    rows, dim, eps,             # scalar arguments
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized FP16 values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating (no intermediate (R,5·H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                          # [R, dim]                FP16 (LN output)
    mask_ptr,                       # [R]                     FP16 (optional)

    left_ptr, right_ptr, out_gate_ptr,       # [B, H, N, N] FP16 outputs
    R, H, N, dim,                  # scalar arguments

):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    b   = offs_r // nn

        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for hidden h = 0

    # ------------------------------------------------------------------
    # Loop over hidden tiles
    # ------------------------------------------------------------------

        mask_h = offs_h < H

        acc_left = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_right = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_lgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_rgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_ogate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # ----- dot‑product over the feature dimension -----
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):

            mask_d = offs_d < dim

            x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],

            # weight tiles: transposed layout [dim, H]

                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wR = tl.load(w_right_ptr  + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wLG = tl.load(w_lgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wRG = tl.load(w_rgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wOG = tl.load(w_ogate_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

            # accumulate the five GEMMs
            acc_left += tl.dot(x_tile, wL)   # (M,D)·(D,H) → (M,H)
            acc_right += tl.dot(x_tile, wR)
            acc_lgate += tl.dot(x_tile, wLG)

        left_raw  = acc_left .to(tl.float16)
        right_raw = acc_right.to(tl.float16)
        lgate_raw = acc_lgate.to(tl.float16)
        rgate_raw = acc_rgate.to(tl.float16)
        ogate_raw = acc_ogate.to(tl.float16)

        lgate = (1.0 / (1.0 + tl.exp(-lgate_raw.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate_raw.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * lgate
        right = right_raw * mask_val * rgate
        out_gate = ogate                                 # already sigmoid‑ed

        # ----- write out the three tensors -----
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr      + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Second LayerNorm, out_gate multiplication & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden‑plane values
        out_vals  = tl.load(out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm (γ,β are per‑hidden‑dim)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]                # (out‑μ)·rstd
        norm = diff * gamma[None, :] + beta[None, :]                   # LN
        term = norm * gate_vals                                        # out_gate

        # cast to fp16 for the tensor‑core dot
        term_fp16 = term.to(tl.float16)   # [M, K]

        # weight tile [K, dim_out]
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                  # [K, N_out]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (no intermediate tensor)
    # --------------------------------------------------------------
    # transpose weight matrices once (dim x hidden) for efficient dot
    left_w_T      = w["left_proj.weight"].t().contiguous().to(torch.float16)
    right_w_T     = w["right_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_w_T = w["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_w_T= w["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_w_T  = w["out_gate.weight"].t().contiguous().to(torch.float16)

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D_proj = 64
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        mask_flat,
        left_w_T, right_w_T,
        left_gate_w_T, right_gate_w_T, out_gate_w_T,
        left, right, out_gate,
        R, H, N, dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D_proj,
        num_warps=8,
    )
    del x_norm, mask_flat

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)
    out = out.view(B, H, N, N)                               # [B, H, N, N] (fp16)

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128        # H ≤ 128 for all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
