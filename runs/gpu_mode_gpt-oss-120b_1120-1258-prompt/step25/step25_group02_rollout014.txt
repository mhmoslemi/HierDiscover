"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward implementation.
--------------------------------------------------------------------------------
Algorithm
---------
1️⃣  Pairwise LayerNorm (float16) – torch.nn.functional.layer_norm.
2️⃣  One fused GEMM projects the normalized tensor to the five required
    heads (left, right, left_gate, right_gate, out_gate).  A small Triton
    kernel (`split_fuse_gate_kernel`) splits the fused result, applies the
    optional mask, and evaluates the sigmoid gates – all in FP16.
3️⃣  Batched GEMM (cuBLAS) contracts the left/right tensors:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
4️⃣  Row‑wise LayerNorm on the contracted tensor, element‑wise
    multiplication with `out_gate`, and the final linear projection are
    performed with torch.nn.functional.layer_norm and torch.nn.functional.linear.
    The final tensor is returned in float32.

Only the splitting/gating step is custom Triton code – all heavy
matrix‑multiply work (steps 2‑4) uses highly‑tuned cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣  Split/fuse the 5‑head projection, apply mask and sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16   (fused projection)
    left_ptr,          # [B, H, N, N] FP16   – output
    right_ptr,         # [B, H, N, N] FP16   – output
    out_gate_ptr,      # [B, H, N, N] FP16   – output
    mask_ptr,          # [R]    FP16   – optional (broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode the flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional per‑pair mask (broadcast along H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N               # stride between successive H planes
    # base offset for h==0 in the output tensors
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: (left | right | left_gate | right_gate | out_gate)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Stable sigmoid in FP32, result back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask + gating
        left   = left_raw   * mask_val * left_gate
        right  = right_raw  * mask_val * right_gate
        outg   = out_gate                         # out_gate already sigmoided

        # Destination address for the current hidden slice
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and cast
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # flattened pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (float16) – use torch's highly‑optimized kernel
    # ------------------------------------------------------------------
    norm_w = w["norm.weight"].to(torch.float16)
    norm_b = w["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five linear maps in a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices → shape [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    # GEMM: (R, dim) @ (dim, 5·H) → (R, 5·H)
    proj = F.linear(x_norm, fused_w)            # FP16, shape [B, N, N, 5·H]
    proj = proj.reshape(R, 5 * H).contiguous() # Flat for the kernel

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask, sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_f16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # Choose tile sizes that fit nicely in shared memory (B, H ≤ 128)
    BLOCK_R = 64      # rows per program
    BLOCK_H = 128     # process a whole hidden slice at once

    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Release the large fused projection early
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS) – core einsum
    #     out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm, out_gate, final projection
    # ------------------------------------------------------------------
    # Flatten to (R, H) for the row‑wise LayerNorm
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # [R, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    # LayerNorm over the hidden dimension (float16)
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out_flat, (H,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # Apply the out_gate (element‑wise)
    out_norm_gate = out_norm * out_gate_flat

    # Final linear projection back to `dim`
    to_out_w = w["to_out.weight"].to(torch.float16)   # shape [dim, H]
    final_flat = F.linear(out_norm_gate, to_out_w)    # [R, dim]  FP16

    # Reshape to the original 4‑D layout and cast to float32 (required output type)
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
