):

    j = rem - i * N

@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    dst_ptr, # [R, H] FP16
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0,BLOCK_R)
   mask_r = offs_r < R

   nn = N * N
   b = offs_r // nn
   rem = offs_r - b * nn
   i = rem // N
   j = rem - i * N
   hidden_stride = N * N
   base_off = b * (H * hidden_stride) + i * N + j

   # load mu,rstd per row
   mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0) # [BLOCK_R]
   rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)
   mu = mu[:, None]   # broadcast
   rstd = rstd[:, None]

   for h_off in range(0, tl.cdiv(H, BLOCK_H)):
       offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H

       # load out_vals and gate_vals
       out_vals = tl.load(out_ptr + base_off[:, None] + offs_h[None,:] * hidden_stride,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

       gate_vals = tl.load(gate_ptr + base_off[:, None] + offs_h[None,:] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

       # layernorm scale and bias
       gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
       beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

       # compute norm and gating
       norm = (out_vals - mu) * rstd   # [BLOCK_R, BLOCK_H]
       norm = norm * gamma[None, :] + beta[None, :] # LN
       gated = norm * gate_vals
       # store
       dst_offset = offs_r[:, None] * H + offs_h[None, :]
       tl.store(dst_ptr + dst_offset,
                gated.to(tl.float16),
                mask=mask_r[:, None] & mask_h[None, :])

@triton.jit
def fused_final_output_kernel(
    out_ptr, gate_ptr, ln_w_ptr, ln_b_ptr, w_out_T_ptr, out_final_ptr,
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)  # rows
    pid_n = tl.program_id(1)  # dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)
    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # load per-row mu/rstd computed later
    # accumulators
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # out*gamma*gate term
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # gamma*gate term
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # beta*gate term

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate values
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_K]

        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # update sum_out and sum_sq
        sum_out += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

        # load gamma and beta
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]

        # broadcast gamma,beta across rows
        # compute term contributions
        # term2 = gamma * gate
        term2 = gamma[None, :] * gate_vals   # [BLOCK_M, BLOCK_K]
        term3 = beta[None, :] * gate_vals    # [BLOCK_M, BLOCK_K]
        term1 = out_vals * term2             # out * gamma * gate

        # load weight tile
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                     # [BLOCK_K, BLOCK_N]

        # dot products
        # Convert term1, term2, term3 to fp16
        t1_fp16 = term1.to(tl.float16)
        t2_fp16 = term2.to(tl.float16)
        t3_fp16 = term3.to(tl.float16)

        acc1 += tl.dot(t1_fp16, w_tile).to(tl.float32)
        acc2 += tl.dot(t2_fp16, w_tile).to(tl.float32)
        acc3 += tl.dot(t3_fp16, w_tile).to(tl.float32)

    # after loop compute mu and rstd
    mu = sum_out / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # final combine
    # compute mu broadcasted across dim
    mu_exp = mu[:, None]  # shape [BLOCK_M, 1]
    rstd_exp = rstd[:, None]

    # compute final tile: rstd*(acc1 - mu*acc2) + acc3
    out_tile = rstd_exp * (acc1 - mu_exp * acc2) + acc3

    # store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             out_tile,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fully fused forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – one Triton kernel.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) producing a (R,5·H) matrix, R = B·N·N.
3️⃣ Split the fused projection, apply the optional mask, compute the
   sigmoid gates and store `left`, `right` and `out_gate` (Triton kernel).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → tensor `[B, H, N, N]` (FP16).
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   linear projection into a single Triton kernel.  While iterating over the
   hidden dimension we also accumulate the per‑token mean/variance, thus
   we *do not* need a separate statistics kernel.
6️⃣ Reshape the result to `[B, N, N, dim]` (FP32).

All kernels target Triton‑3.3.1 on an H100 and use mixed‑precision
(fp16 for most tensors, fp32 for reductions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # first pass: mean & variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [L | R | LG | RG | OG]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate, and final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_final_output_kernel(
    out_ptr,               # [B, H, N, N] FP16  (post‑einsum)
    gate_ptr,              # [B, H, N, N] FP16  (out_gate)
    ln_w_ptr,              # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,              # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,           # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,         # [R, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row accumulators
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out * γ * g
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ γ * g
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β * g

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate
        out_vals = tl.load(
            out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)               # [M, K]

        gate_vals = tl.load(
            gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)               # [M, K]

        # update running statistics
        sum_out += tl.sum(out_vals, axis=1)          # Σ out
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)  # Σ out²

        # layer‑norm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # compute the three term families
        g_gamma = gamma[None, :] * gate_vals               # γ * g
        term2 = g_gamma                                    # γ * g
        term3 = beta[None, :] * gate_vals                  # β * g
        term1 = out_vals * term2                           # out * γ * g

        # load weight tile (H, dim) → [K, TILE_N]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0)                                      # [K, TILE_N]

        # dot products (FP16·FP16 → FP32)
        acc1 += tl.dot(term1.to(tl.float16), w_tile).to(tl.float32)
        acc2 += tl.dot(term2.to(tl.float16), w_tile).to(tl.float32)
        acc3 += tl.dot(term3.to(tl.float16), w_tile).to(tl.float32)

    # finalize statistics
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # combine everything:
    # out = rstd * (acc1 - mu * acc2) + acc3
    mu_exp   = mu[:, None]      # [M,1]
    rstd_exp = rstd[:, None]    # [M,1]

    out_tile = rstd_exp * (acc1 - mu_exp * acc2) + acc3

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             out_tile,
             mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    # free memory that is no longer needed
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (tokens)
    BLOCK_K_OUT = 32       # hidden‑dim chunk
    BLOCK_N_OUT = 128      # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_final_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # reshape to the expected output layout
    final = final.view(B, N, N, dim)

    return final
