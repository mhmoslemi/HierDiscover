"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized Triton+cuBLAS implementation

Algorithm (forward)
--------------------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) – custom Triton
   kernel (single‑pass for mean/var → write normalized values).

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM (R‑by‑dim) × (5·H‑by‑dim)ᵀ → (R‑by‑5·H).  The GEMM runs in cuBLAS
   (FP16).

3️⃣ Split the fused projection, optionally apply the pair‑mask, and apply the
   sigmoid gates – custom Triton kernel.  Produces three tensors
   `left`, `right`, `out_gate` of shape [B, H, N, N] (FP16).

4️⃣ Batched matrix multiplication `out = left @ rightᵀ` (FP16) – cuBLAS
   `torch.bmm`.  Shape of `out` = [B, H, N, N].

5️⃣ Row‑wise moments of `out` (mean & rsqrt‑variance) – computed with fast
   PyTorch reductions (`mean` / `var`).  Results `mu`, `rstd` have shape [R]
   (R = B·N·N).

6️⃣ LayerNorm on the contracted tensor, element‑wise multiply by `out_gate`,
   and final linear projection to `dim` – all performed in FP16 with standard
   PyTorch operators (leveraging cuBLAS for the huge (R,H)×(H,dim) GEMM).

The heavy compute (steps 4 and 6) is off‑loaded to cuBLAS, while the
memory‑bound parts (layer‑norm, splitting + gating) are handled by two small
Triton kernels.  This design reduces kernel launch overhead and maximises
throughput on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    γ (FP32)
    b_ptr,          # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (scalar per (b,i,k), broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    # process the full hidden dimension in a single iteration (BLOCK_H = H)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Entry point exposing the full forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pair positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()                         # FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # FP16 result

    BLOCK_M = 1024            # threads per block (covers many rows)
    BLOCK_D = 64              # processed dimension chunk
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]  (FP16)

    # (R, dim) × (5·H, dim)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, fused_w)          # FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 256   # number of rows processed per program
    BLOCK_H = H     # process the whole hidden dimension in one go
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Rowwise moments of `out` (mean & rsqrt‑variance)
    # --------------------------------------------------------------
    # flatten to [R, H] for reductions
    out_flat = out.view(R, H)      # FP16
    mu_fp32 = out_flat.float().mean(dim=1)                     # [R] float32
    var_fp32 = out_flat.float().var(dim=1, unbiased=False)    # [R] float32
    rstd_fp32 = torch.rsqrt(var_fp32 + eps)                   # [R] float32

    # convert to half for the remaining FP16 pipeline
    mu = mu_fp32.to(torch.float16)    # [R]
    rstd = rstd_fp32.to(torch.float16)  # [R]

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm (hidden), gate, final projection (all FP16, cuBLAS)
    # --------------------------------------------------------------
    # broadcast mu/rstd over hidden dimension
    mu_b = mu.unsqueeze(1)          # [R,1]
    rstd_b = rstd.unsqueeze(1)      # [R,1]

    # to_out_norm parameters (γ,β) – cast once to FP16
    ln_w = w["to_out_norm.weight"].to(torch.float16)  # [H]
    ln_b = w["to_out_norm.bias"].to(torch.float16)    # [H]

    # LN + gate (still FP16)
    norm = (out_flat - mu_b) * rstd_b               # (out - μ) * rstd
    norm = norm * ln_w + ln_b                       # γ· + β
    gated = norm * out_gate.view(R, H)               # element‑wise gate

    # final linear projection  (R, H) → (R, dim)  – cuBLAS
    final_fp16 = F.linear(gated, w["to_out.weight"].to(torch.float16))  # FP16

    # reshape to the original 4‑D layout and cast to FP32 as required
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
