"""
Optimized outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm (high‑level)
----------------------
1️⃣  Input LayerNorm  (torch.nn.functional.layer_norm) → FP16.
2️⃣  Fuse 5 linear maps (left, right, left_gate, right_gate, out_gate) into
    a single GEMM (R × dim)·(dim × 5·H) (cuBLAS, FP16).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
        * splits the fused projection,
        * applies the optional pair mask,
        * evaluates sigmoid gates,
        * writes three tensors `left`, `right`, `out_gate` (shape [B,H,N,N]).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B,H,N,N] (FP16).
5️⃣  Output LayerNorm (torch.nn.functional.layer_norm) over the hidden
    dimension H, then element‑wise multiplication with `out_gate`.
6️⃣  Final projection to the model dimension `dim` with a GEMM
    (FP16·FP16 → FP32).  The result is reshaped to `[B,N,N,dim]` (FP32).

Only the split‑and‑gate step is written in Triton; the rest uses
high‑performance cuBLAS/DNN kernels.  This reduces kernel‑launch overhead
and lets the GPU spend most of the time in well‑optimised matrix‑multiply
routines, meeting the 500 µs target on the reference configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, optionally mask, apply sigmoid
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fused projection (FP16)
    left_ptr,          # [B, H, N, N] (FP16)
    right_ptr,         # [B, H, N, N] (FP16)
    out_gate_ptr,      # [B, H, N, N] (FP16)
    mask_ptr,          # [R] (FP16) – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                 # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride to walk the hidden dimension inside each (b,i,k) slice
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k                     # [BLOCK_R]

    # Process H in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection (5 blocks of size H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (FP32 -> FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate                                            # already gated only by mask

        # Destination offsets for left/right/out_gate (layout: B,H,N,N)
        dst = base[:, None] + offs_h[None, :] * hidden_stride      # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,      left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,     right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,  og,    mask=mask_r[:, None] & mask_h[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use cuDNN for speed)
    # ------------------------------------------------------------------
    # Normalise over the last dimension `dim` → keep FP32 for mean/var,
    # cast result to FP16 for the next GEMM.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                         # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM (R, dim)·(dim, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape [5·H, dim]

    x_flat = x_norm.reshape(R, dim)             # [R, dim] FP16, contiguous
    proj = F.linear(x_flat, fused_w)            # [R, 5·H] FP16
    del x_norm, x_flat, fused_w                  # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates (Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)   # dummy
        HAS_MASK = 0

    # Tile sizes – tuned for the typical H (=128) and R (up to ~1e6)
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Core TriMul contraction: batched GEMM (left @ rightᵀ)
    #    shapes: left/right → [B·H, N, N] (FP16)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16
    del left_mat, right_mat, left, right        # free

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm over hidden dimension H (cuDNN) + out_gate
    # ------------------------------------------------------------------
    out_ln = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # still FP16

    out_gated = out_ln * out_gate                # element‑wise, FP16
    del out_ln, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection to `dim` (FP16·FP16 → FP32)
    # ------------------------------------------------------------------
    out_gated_flat = out_gated.reshape(R, H)     # [R, H] FP16
    w_out = weights["to_out.weight"].to(torch.float16, device=device)   # [dim, H]
    out_final = F.linear(out_gated_flat, w_out)  # [R, dim] FP16
    out_final = out_final.view(B, N, N, dim).to(torch.float32)  # FP32 result

    return out_final
