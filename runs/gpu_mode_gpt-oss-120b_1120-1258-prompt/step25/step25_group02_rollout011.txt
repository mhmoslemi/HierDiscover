@triton.jit
def norm_gate_kernel(out_ptr, out_gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr, out_norm_gated_ptr,
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # offsets for hidden stride
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # starting location for hidden idx 0

    # Load mu and rstd scalars per row
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # compute for each hidden channel
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values and gate
        out_val = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(out_gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # load LN parameters gamma, beta for these channels
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # Normalize: (out - mu) * rstd
        norm = (out_val - mu[:, None]) * rstd[:, None]

        # Apply LN
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply gate
        gated = norm * gate_val

        # store back as fp16
        tl.store(out_norm_gated_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

@triton.jit
def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr, R, H, N, HAS_MASK: tl.constexpr, BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute pointer offsets for left/right/out_gate writes
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    # mask value
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16) # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute base index into proj: each row has 5*H elements; slice offset per hidden channel
        # reading "left", "right", "left_gate", "right_gate", "out_gate"
        # For row r and hidden h, we need at index = r * (5*H) + h + offset*H for each group.
        # Idea: compute proj_idx = offs_r[:, None] * (5*H) + offs_h[None, :] + group_offset*H
        # We'll load each group.

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # shape [BLOCK_R,BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        # compute sigmoid in FP32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left_val = left_raw * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device
    B,N,_,_ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm
    x_flat = input_tensor.reshape(R, dim)
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2. Fuse linear projection
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, dim)
    proj = torch.nn.functional.linear(x_norm, fused_weight)  # (R,5*H) fp16
    del x_norm, fused_weight

    # 3. Split and apply mask/gates using Triton
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Batched GEMM (left @ right^T)
    left_mat = left.view(B * H, N, N)  # (B*H,N,N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    out = out.view(B, H, N, N)

    # 5. Compute per-position mean/std over hidden dim
    mu = out.mean(dim=1)  # (B,N,N) float32
    var = out.var(dim=1, unbiased=False)
    rstd = torch.rsqrt(var + eps)

    # 6. Apply second LN + out_gate using Triton
    out_norm_gate = torch.empty_like(out)  # (B,H,N,N) fp16
    total_rows = R
    BLOCK_R2 = 256
    BLOCK_H2 = 64
    grid_norm = (triton.cdiv(total_rows, BLOCK_R2),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_norm_gate,
        total_rows,
        H,
        N,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # 7. Final projection to dim (B,N,N,dim) using cuBLAS
    # reshape to (R, H)
    out_norm_gate_flat = out_norm_gate.permute(0,2,3,1).contiguous().view(R, H)  # (R,H) fp16
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # (H, dim)
    out_final_fp16 = torch.matmul(out_norm_gate_flat, w_out_T)  # (R, dim) fp16
    out_final = out_final_fp16.float().view(B, N, N, dim)

    return out_final

"""
Outgoing TriMul (AlphaFold3) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32) on the flattened pair tensor – torch .nn.functional.layer_norm.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a single
   GEMM (R×dim)·(dim×5·H) → (R×5·H) using torch.nn.functional.linear (FP16).
3️⃣ Triton kernel “split_fuse_gate_kernel”:
      * splits the fused projection into the three tensors left, right, out_gate,
      * applies the optional mask,
      * computes the three sigmoid gates,
      * writes the results in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum
      out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → (B, H, N, N) (FP16).
5️⃣ Row‑wise mean / variance of `out` over the hidden dimension H – torch.mean / torch.var.
6️⃣ Triton kernel “norm_gate_kernel”:
      * performs the second LayerNorm (γ,β) using the statistics from (5),
      * multiplies by the out‑gate,
      * writes the normalised‑gated tensor (FP16).
7️⃣ Final linear projection (H → dim) via a cuBLAS matmul on the
   reshaped tensor (R, H)·(H, dim) → (R, dim) (FP16 → FP32).

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # -------- decode flat row index r = b·N·N + i·N + k ----------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # stride for the hidden dimension inside the 4‑D tensors
    hidden_stride = N * N                     # distance between two consecutive H‑planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    # -------- optional mask (broadcast over H) ---------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # -------- iterate over the hidden dimension in chunks -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for numerical stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Second LayerNorm + out‑gate (no final projection)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,       # [B, H, N, N] FP16 (raw contracted tensor)
    gate_ptr,      # [B, H, N, N] FP16 (out_gate)
    mu_ptr,        # [R]           FP32 (row‑wise mean)
    rstd_ptr,      # [R]           FP32 (row‑wise rsqrt‑var)
    ln_w_ptr,      # [H]           FP32 (to_out_norm.weight)
    ln_b_ptr,      # [H]           FP32 (to_out_norm.bias)
    out_ptr_norm,  # [B, H, N, N] FP16 (normalised & gated result)
    total_rows, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_m = offs_m < total_rows

    # decode positions
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # offset for h = 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # FP32

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # load raw values (FP16 → FP32)
        out_val = tl.load(
            out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        gate_val = tl.load(
            gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd  →  γ· + β  →  * out_gate
        norm = (out_val - mu[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]
        gated = norm * gate_val

        # store back as FP16
        tl.store(
            out_ptr_norm + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            gated.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold3).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on (R, dim) – FP32 → FP16
    # ------------------------------------------------------------------
    inp_flat = inp.view(R, dim).contiguous()                       # [R, dim] FP32
    # torch.layer_norm is highly tuned; we keep it in FP32 and cast later
    inp_norm = torch.nn.functional.layer_norm(
        inp_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                           # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                           # [5·H, dim]
    proj = F.linear(inp_norm, fused_w)                             # (R, 5·H)  FP16
    del inp_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split & gate (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 256       # rows per program
    BLOCK_H = 64        # hidden tiles (covers H ≤ 128 in two passes)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction  – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance over hidden dim (H)
    # ------------------------------------------------------------------
    mu = out.mean(dim=1)                       # (B, N, N)  FP32
    var = out.var(dim=1, unbiased=False)      # (B, N, N)  FP32
    rstd = torch.rsqrt(var + eps)              # (B, N, N)  FP32

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate (Triton)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out)           # (B, H, N, N) FP16
    total_rows = R
    BLOCK_R2 = 256
    BLOCK_H2 = 64
    grid_norm = (triton.cdiv(total_rows, BLOCK_R2),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        out_norm,
        total_rows,
        H,
        N,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (H → dim) – cuBLAS matmul
    # ------------------------------------------------------------------
    # reshape to (R, H)  (contiguous layout)
    out_norm_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)      # (H, dim) FP16
    out_fp16 = torch.matmul(out_norm_flat, w_out_T)                      # (R, dim) FP16
    out_fp32 = out_fp16.float().view(B, N, N, dim)                       # (B, N, N, dim) FP32

    return out_fp32
