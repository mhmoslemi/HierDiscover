"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.

Algorithm
---------
1️⃣  Input pair tensor  x ∈ [B, N, N, C]  →  row‑wise LayerNorm (C‑dim) using
    torch.nn.functional.layer_norm (FP32 → FP16).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:
        proj = x_norm @ W_fused.T          # shape [R, 5·H]  (FP16)
    where R = B·N·N and H = hidden_dim.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * splits `proj` into left, right and the three gates,
    * optionally applies the pair‑wise mask,
    * applies sigmoid to the three gates,
    * stores `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core “TriMul” contraction:
        out[b, h, i, j] = Σ_k left[b, h, i, k]·right[b, h, j, k]
    → tensor `[B, H, N, N]` (FP16).

5️⃣  Final processing (torch kernels, no custom Triton):
    * permute to `[B, N, N, H]`,
    * LayerNorm over the hidden dimension H (to_out_norm),
    * element‑wise multiply by `out_gate`,
    * linear projection to the original channel dimension `C`
      (`to_out.weight`), producing the final `[B, N, N, C]` tensor (float32).

All heavy data‑parallel work is done in FP16; reductions for LayerNorm
are performed in FP32 internally by the PyTorch kernels.  This design
removes two custom kernels (row‑wise stats + final output) and relies
on highly‑optimised PyTorch kernels for those stages, reducing launch
overhead and improving overall latency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask and
# compute the three sigmoid gates.  Stores left/right/out_gate tensors.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride for one hidden plane (N·N)
    hidden_stride = N * N
    # Base offset in the destination tensors for the current (b,i,k)
    base = b * (H * hidden_stride) + i * N + k                 # [BLOCK_R]

    # Loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Address of the fused projection slice for this tile:
        #   [left | right | left_gate | right_gate | out_gate] each size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate  # no mask for the out‑gate

        # Destination offset for each hidden entry
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # Use PyTorch's highly‑optimised layer_norm kernel.
    # It reduces to FP32 statistics internally and writes FP16 results.
    norm_weight = w["norm.weight"].to(torch.float16)
    norm_bias   = w["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x_norm = x_norm.to(torch.float16)                     # [B, N, N, C] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate,
    #     out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # shape [5·H, C]

    # Flatten to [R, C] for the GEMM
    x_flat = x_norm.reshape(R, dim).contiguous()
    proj = F.linear(x_flat, fused_w)                  # (R, 5·H) FP16
    del fused_w, x_flat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask broadcasted over the hidden dimension, stored as FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right: [B, H, N, N] → reshape to [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm (over hidden dim) → gate → output projection
    # ------------------------------------------------------------------
    # Bring hidden dimension to the trailing axis for LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H] FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H] FP16

    # Convert to_out_norm parameters to FP16 (torch expects same dtype as input)
    ln_weight = w["to_out_norm.weight"].to(torch.float16)
    ln_bias   = w["to_out_norm.bias"].to(torch.float16)

    # LayerNorm over the hidden dimension H
    out_ln = F.layer_norm(
        out_perm,
        (H,),
        weight=ln_weight,
        bias=ln_bias,
        eps=eps,
    )                                                        # [B, N, N, H] FP16

    # Apply the out_gate
    out_gated = out_ln * out_gate_perm                       # [B, N, N, H] FP16

    # Final linear projection to the original channel dimension C (dim)
    w_out = w["to_out.weight"].to(torch.float16)            # shape [dim, H]
    out_flat = out_gated.reshape(R, H)                      # [R, H]
    final_fp16 = F.linear(out_flat, w_out)                  # [R, dim] FP16

    # Cast back to float32 and restore the original layout
    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # [B, N, N, dim]

    return final
