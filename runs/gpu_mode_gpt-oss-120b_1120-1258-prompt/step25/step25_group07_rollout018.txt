def proj_gate_kernel(
    inp_ptr,         # [R, dim] FP16
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # same
    out_gate_ptr,    # same
    mask_ptr,        # [R] FP16 (optional)
    # weight pointers:
    left_proj_w_ptr,
    right_proj_w_ptr,
    left_gate_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    # scalars
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_r = tl.program_id(0)
    pid_h = tl.program_id(1)
    row_start = pid_r * BLOCK_R
    hid_start = pid_h * BLOCK_H
    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_h = hid_start + tl.arange(0, BLOCK_H)
    mask_r = offs_r < R
    mask_h = offs_h < H

    # need decode b,i,k from offs_r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute destination base indices for left/right/out_gate
    hidden_stride = N * N
    # dest base index (flattened) for each row and hidden
    # left_ptr layout: [B, H, N, N]
    # offset = ((b * H + h) * N + i) * N + k
    # Compute linear offsets: for each row r and each hidden h:
    # base = (b * H + offs_h) * hidden_stride + i * N + k

    # We'll compute base offsets per (r) and per h.
    # However we can compute vectorized: for each row r, have base_r = b*H*hidden_stride + i*N + k
    base_r = b * (H * hidden_stride) + i * N + k   # shape [BLOCK_R]
    # For each hidden tile:
    # offset = base_r[:, None] + offs_h[None, :] * hidden_stride

    # accumulators
    acc_left = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_ogate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

    for d_start in range(0, dim, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input tile
        x_ptr = inp_ptr + offs_r[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0)

        # weight tiles (transposed: [BLOCK_D, BLOCK_H])
        # left_proj
        w_left = tl.load(left_proj_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        # right_proj
        w_right = tl.load(right_proj_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        # left_gate
        w_lgate = tl.load(left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        # right_gate
        w_rgate = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        # out_gate
        w_ogate = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)

        # dot products: (BLOCK_R, BLOCK_D) * (BLOCK_D, BLOCK_H) => (BLOCK_R, BLOCK_H)
        acc_left += tl.dot(x, w_left)
        acc_right += tl.dot(x, w_right)
        acc_lgate += tl.dot(x, w_lgate)
        acc_rgate += tl.dot(x, w_rgate)
        acc_ogate += tl.dot(x, w_ogate)

    # after loop, apply mask and sigmoid gates
    # mask per row (scalar if HAS_MASK)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # shape [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # sigmoid for gates
    lgate = tl.sigmoid(acc_lgate.to(tl.float16))
    rgate = tl.sigmoid(acc_rgate.to(tl.float16))
    ogate = tl.sigmoid(acc_ogate.to(tl.float16))

    left_out = (acc_left.to(tl.float16) * mask_val * lgate).to(tl.float16)
    right_out = (acc_right.to(tl.float16) * mask_val * rgate).to(tl.float16)

    # destination offsets:
    dst_offset = base_r[:, None] + offs_h[None, :] * hidden_stride
    tl.store(left_ptr + dst_offset,
             left_out,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset,
             right_out,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset,
             ogate,
             mask=mask_r[:, None] & mask_h[None, :])

General design:

```
@triton.jit
def proj_gate_fused_kernel(
    inp_ptr,               # [R, dim] float32
    left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,               # [R] fp16 optional
    ln_weight_ptr, ln_bias_ptr,
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr_weight,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_r = tl.program_id(0)
    pid_h = tl.program_id(1)
    row_start = pid_r * BLOCK_R
    hid_start = pid_h * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_h = hid_start + tl.arange(0, BLOCK_H)

    mask_r = offs_r < R
    mask_h = offs_h < H

    # decode b,i,k positions
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    # accumulators
    acc_left = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_right = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    acc_ogate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

    # B and C accumulators (per hidden)
    B_left = tl.zeros((BLOCK_H,), dtype=tl.float32)
    C_left = tl.zeros((BLOCK_H,), dtype=tl.float32)
    B_right = tl.zeros((BLOCK_H,), dtype=tl.float32)
    C_right = tl.zeros((BLOCK_H,), dtype=tl.float32)
    B_lgate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    C_lgate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    B_rgate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    C_rgate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    B_ogate = tl.zeros((BLOCK_H,), dtype=tl.float32)
    C_ogate = tl.zeros((BLOCK_H,), dtype=tl.float32)

    sum_x = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_x2 = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for d_start in range(0, dim, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load input values
        x_ptr = inp_ptr + offs_r[:, None] * dim + offs_d[None, :]
        x_fp32 = tl.load(x_ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0)
        x_fp16 = x_fp32.to(tl.float16)

        # Update sums for mean/std
        sum_x += tl.sum(x_fp32, axis=1)
        sum_x2 += tl.sum(x_fp32 * x_fp32, axis=1)

        # Load LN weight and bias (fp16)
        gn = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float16)  # gamma
        bn = tl.load(ln_bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float16)   # beta

        x_scaled = x_fp16 * gn   # (x - not yet subtract mean) * gamma

        # Load weight tiles (transposed)
        # for each weight matrix
        w_left = tl.load(left_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_right = tl.load(right_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_lgate = tl.load(left_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_rgate = tl.load(right_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_ogate = tl.load(out_gate_ptr_weight + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :], other=0.0)

        # Accumulate A terms
        acc_left += tl.dot(x_scaled, w_left)
        acc_right += tl.dot(x_scaled, w_right)
        acc_lgate += tl.dot(x_scaled, w_lgate)
        acc_rgate += tl.dot(x_scaled, w_rgate)
        acc_ogate += tl.dot(x_scaled, w_ogate)

        # Accumulate B (gamma * w) and C (beta * w) sums over d
        # Sum over d dimension (axis=0) -> shape (BLOCK_H,)
        # compute gamma * w_left (both fp16) -> convert to fp32 for accumulation
        B_left += tl.sum(gn[:, None] * w_left, axis=0).to(tl.float32)
        C_left += tl.sum(bn[:, None] * w_left, axis=0).to(tl.float32)

        B_right += tl.sum(gn[:, None] * w_right, axis=0).to(tl.float32)
        C_right += tl.sum(bn[:, None] * w_right, axis=0).to(tl.float32)

        B_lgate += tl.sum(gn[:, None] * w_lgate, axis=0).to(tl.float32)
        C_lgate += tl.sum(bn[:, None] * w_lgate, axis=0).to(tl.float32)

        B_rgate += tl.sum(gn[:, None] * w_rgate, axis=0).to(tl.float32)
        C_rgate += tl.sum(bn[:, None] * w_rgate, axis=0).to(tl.float32)

        B_ogate += tl.sum(gn[:, None] * w_ogate, axis=0).to(tl.float32)
        C_ogate += tl.sum(bn[:, None] * w_ogate, axis=0).to(tl.float32)

    # Compute LayerNorm stats per row
    eps = 1e-5
    mean = sum_x / dim
    var = sum_x2 / dim - mean * mean
    inv_std = tl.rsqrt(var + eps)

    # Compute final projections
    # Broadcast mean, inv_std over hidden dim
    left_proj = inv_std[:, None] * (acc_left - mean[:, None] * B_left[None, :]) + C_left[None, :]
    right_proj = inv_std[:, None] * (acc_right - mean[:, None] * B_right[None, :]) + C_right[None, :]
    lgate_proj = inv_std[:, None] * (acc_lgate - mean[:, None] * B_lgate[None, :]) + C_lgate[None, :]
    rgate_proj = inv_std[:, None] * (acc_rgate - mean[:, None] * B_rgate[None, :]) + C_rgate[None, :]
    ogate_proj = inv_std[:, None] * (acc_ogate - mean[:, None] * B_ogate[None, :]) + C_ogate[None, :]

    # Apply mask if any
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        mask_val = mask_val[:, None]                              # broadcast
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # Compute gates via sigmoid (in fp16)
    lgate = tl.sigmoid(lgate_proj.to(tl.float16))
    rgate = tl.sigmoid(rgate_proj.to(tl.float16))
    ogate = tl.sigmoid(ogate_proj.to(tl.float16))

    # Apply mask and gating to left/right
    left_final = (left_proj.to(tl.float16) * mask_val) * lgate
    right_final = (right_proj.to(tl.float16) * mask_val) * rgate
    out_gate_final = ogate

    # Store results
    dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
    tl.store(left_ptr + dst_offset, left_final,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset, right_final,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset, out_gate_final,
             mask=mask_r[:, None] & mask_h[None, :])
```

4️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final

    acc_ogate   = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

    sum_x   = tl.zeros((BLOCK_R,), dtype=tl.float32)   # for mean
    sum_x2  = tl.zeros((BLOCK_R,), dtype=tl.float32)   # for variance

    # ------------------------------------------------------------------
    # Main loop over input dimension dim
    # ------------------------------------------------------------------
    for d_start in range(0, dim, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        # ---- load input values (FP32) ---------------------------------
        x_ptr   = inp_ptr + offs_r[:, None] * dim + offs_d[None, :]
        x_fp32  = tl.load(x_ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0)

        # accumulate mean / variance
        sum_x  += tl.sum(x_fp32, axis=1)
        sum_x2 += tl.sum(x_fp32 * x_fp32, axis=1)

        # ---- LayerNorm γ (norm.weight) --------------------------------
        gn = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float16)  # γ
        # scale input for the five GEMMs:  x̃ = x * γ
        x_scaled = x_fp32.to(tl.float16) * gn                     # FP16

        # ---- Load weight tiles (transposed view: [dim, H]) ------------
        # shape of each tile: (BLOCK_D, BLOCK_H)
        w_left     = tl.load(left_w_ptr     + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_right    = tl.load(right_w_ptr    + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_lgate    = tl.load(left_gate_w_ptr+ offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_rgate    = tl.load(right_gate_w_ptr+ offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :], other=0.0)
        w_ogate    = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :], other=0.0)

        # ---- GEMMs (x̃ @ Wᵀ) -----------------------------------------
        acc_left  += tl.dot(x_scaled, w_left)    # [BLOCK_R, BLOCK_H]  FP32
        acc_right += tl.dot(x_scaled, w_right)
        acc_lgate += tl.dot(x_scaled, w_lgate)
        acc_rgate += tl.dot(x_scaled, w_rgate)
        acc_ogate += tl.dot(x_scaled, w_ogate)

    # ------------------------------------------------------------------
    # Final LayerNorm statistics (per row)
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = sum_x / dim
    var  = sum_x2 / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+ε)

    # ------------------------------------------------------------------
    # Load pre‑computed B = Σₙ γ·W and C = Σₙ β·W (per hidden plane)
    # ------------------------------------------------------------------
    B_left = tl.load(B_left_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
    C_left = tl.load(C_left_ptr + offs_h, mask=mask_h, other=0.0)

    B_right = tl.load(B_right_ptr + offs_h, mask=mask_h, other=0.0)
    C_right = tl.load(C_right_ptr + offs_h, mask=mask_h, other=0.0)

    B_lgate = tl.load(B_lgate_ptr + offs_h, mask=mask_h, other=0.0)
    C_lgate = tl.load(C_lgate_ptr + offs_h, mask=mask_h, other=0.0)

    B_rgate = tl.load(B_rgate_ptr + offs_h, mask=mask_h, other=0.0)
    C_rgate = tl.load(C_rgate_ptr + offs_h, mask=mask_h, other=0.0)

    B_ogate = tl.load(B_ogate_ptr + offs_h, mask=mask_h, other=0.0)
    C_ogate = tl.load(C_ogate_ptr + offs_h, mask=mask_h, other=0.0)

    # ------------------------------------------------------------------
    # Apply LayerNorm (μ,σ) and add bias C
    # ------------------------------------------------------------------
    # broadcast μ and rstd over hidden dimension
    mu_vec  = mean[:, None]          # [BLOCK_R, 1]
    rstd_vec = rstd[:, None]        # [BLOCK_R, 1]

    # left / right projections
    left_proj = rstd_vec * (acc_left - mu_vec * B_left[None, :]) + C_left[None, :]
    right_proj = rstd_vec * (acc_right - mu_vec * B_right[None, :]) + C_right[None, :]

    # gate projections (still in FP32)
    lgate_proj = rstd_vec * (acc_lgate - mu_vec * B_lgate[None, :]) + C_lgate[None, :]
    rgate_proj = rstd_vec * (acc_rgate - mu_vec * B_rgate[None, :]) + C_rgate[None, :]
    ogate_proj = rstd_vec * (acc_ogate - mu_vec * B_ogate[None, :]) + C_ogate[None, :]

    # ------------------------------------------------------------------
    # Optional mask (scalar per pair) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 scalar per row
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # ------------------------------------------------------------------
    # Sigmoid gates (computed in FP16 for speed)
    # ------------------------------------------------------------------
    lgate = tl.sigmoid(lgate_proj.to(tl.float16))
    rgate = tl.sigmoid(rgate_proj.to(tl.float16))
    ogate = tl.sigmoid(ogate_proj.to(tl.float16))

    # ------------------------------------------------------------------
    # Apply mask + gates and write results
    # ------------------------------------------------------------------
    left_final  = (left_proj.to(tl.float16)  * m) * lgate
    right_final = (right_proj.to(tl.float16) * m) * rgate
    out_gate_final = ogate

    # destination offset for each hidden plane inside the [B,H,N,N] buffer
    dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [M,H]

    tl.store(left_ptr     + dst_offset,
             left_final,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr    + dst_offset,
             right_final,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset,
             out_gate_final,
             mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (to_out.weightᵀ)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row LayerNorm statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden plane of out & gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)    # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)    # [K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]

        # apply out‑gate
        term = norm * gate_vals                                       # [M, K]
        term_fp16 = term.to(tl.float16)                               # [M, K] for tensor‑core dot

        # final projection weight tile  (H × dim) → (K × dim_out)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, N_out]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, N_out]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)               # [M, N_out]

    # write final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------- unpack ---------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------- prepare weight tensors ---------
    # LayerNorm parameters (FP32)
    ln_weight = w["norm.weight"]    # shape [dim]
    ln_bias   = w["norm.bias"]

    # Projection / gate weights (FP16, contiguous)
    left_w      = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w     = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).contiguous()

    # Pre‑compute B = Σ_d γ·W and C = Σ_d β·W for each matrix (FP32)
    ln_w_fp16 = ln_weight.to(torch.float16)
    ln_b_fp16 = ln_bias.to(torch.float16)

    def bc_mat(mat):
        B = (mat * ln_w_fp16).sum(dim=1).to(torch.float32)   # Σ_d γ·W_{h,d}
        C = (mat * ln_b_fp16).sum(dim=1).to(torch.float32)   # Σ_d β·W_{h,d}
        return B, C

    B_left, C_left         = bc_mat(left_w)
    B_right, C_right       = bc_mat(right_w)
    B_lgate, C_lgate       = bc_mat(left_gate_w)
    B_rgate, C_rgate       = bc_mat(right_gate_w)
    B_ogate, C_ogate       = bc_mat(out_gate_w)

    # buffers for the three tensors produced by the fused kernel
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    # --------------------------- launch fused kernel ----------
    BLOCK_R = 32   # rows per program
    BLOCK_H = 32   # hidden‑dim per program
    BLOCK_D = 32   # input‑dim tile

    grid = (triton.cdiv(R, BLOCK_R), triton.cdiv(H, BLOCK_H))
    fused_proj_gate_kernel[grid](
        x.reshape(R, dim).contiguous(),                     # inp_ptr (FP32)
        left, right, out_gate,                              # output buffers
        mask_flat,                                           # mask_ptr
        ln_weight,                                           # LN γ (FP32)
        left_w, right_w, left_gate_w, right_gate_w, out_gate_w,
        B_left, C_left,
        B_right, C_right,
        B_lgate, C_lgate,
        B_rgate, C_rgate,
        B_ogate, C_ogate,
        R, H, N, dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R, BLOCK_H=BLOCK_H, BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32   # hidden_dim ≤ 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out, mu, rstd,
        R, H, N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program (tiles over R)
    BLOCK_K_OUT = 32     # hidden‑dim chunk
    BLOCK_N_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R, H, N, dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
