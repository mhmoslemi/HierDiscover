final projection) is performed with cuBLAS (torch.nn.functional.linear /
torch.bmm) in FP16.  Two LayerNorms (input‑norm and hidden‑norm) are
executed with the highly‑optimized PyTorch kernels (float32 → float16 for
the first, float32 for the second).  The only Triton kernel required
splits the fused projection, applies the optional pair mask and the three
sigmoid gates, and writes the three tensors (`left`, `right`,
`out_gate`) in the layout required by the batched GEMM.

The algorithm (for a batch of size B, sequence length N and hidden
dimension H=hidden_dim) is

1️⃣  Input ‑> Row‑wise LayerNorm over the last dimension (float32 → half).
2️⃣  Fuse the five linear maps (left‑proj, right‑proj, left‑gate,
    right‑gate, out‑gate) into a single GEMM:   X ‑► (R,5H) .
3️⃣  Triton kernel ‑► split the fused matrix into `left`, `right`,
    `out_gate`; apply the optional mask and sigmoid gates.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]    → (B,H,N,N)  (FP16)
5️⃣  Row‑wise LayerNorm (float32) across the hidden dimension H,
    followed by the out‑gate multiplication.
6️⃣  Final linear projection (FP32) → (B,N,N,dim) and return.

Only the split+gate kernel is written in Triton; all other steps reuse
highly‑optimized PyTorch kernels.  This reduces kernel launch overhead
and improves overall throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16  (fused linear output)
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, only when HAS_MASK==1)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Load mask scalar per (b,i,k) if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]                                 # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Offsets for the hidden dimension (stride = N*N per H Plane)
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    # Iterate over the hidden dimension in BLOCK_H sized chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate   # no mask on out_gate

        # Destination offsets for left, right, out_gate (interleaved by H)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
#  Entry point – forward only
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"      (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total number of (b,i,k) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the pair representation (FP32 → FP16)
    # ------------------------------------------------------------------
    # Use PyTorch's highly‑optimised LayerNorm implementation.
    # The result is kept in float16 for the remaining kernels.
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                            # [B,N,N,dim] float32
    x_norm = x_norm_fp32.to(torch.float16).contiguous()
    # reshape to [R, dim] for the fused projection
    x_norm = x_norm.view(R, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into one GEMM (FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5*H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5*H)  FP16
    # proj is contiguous by construction
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy – never read when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    BLOCK_R = 128
    BLOCK_H = max(32, (H // 4) * 4)   # a multiple of 4 for better alignment
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left and right have shape [B, H, N, N] (FP16)
    left_mat  = left.view(B * H, N, N).contiguous()    # [B·H, N, N]
    right_mat = right.view(B * H, N, N).contiguous()   # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N]  FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm (float32) + out‑gate multiplication
    # ------------------------------------------------------------------
    # Collapse B,N,N into a single dimension R = B·N·N
    out_flat = out.view(R, H).float()          # FP32 for stable reduction

    # LayerNorm over the hidden dimension H
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                           # FP32

    # Bring back to FP16 for the gate multiplication (save memory)
    out_norm_fp16 = out_norm.half()
    gated = out_norm_fp16 * out_gate.view(R, H)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (FP32) – weight: [dim, hidden_dim]
    # ------------------------------------------------------------------
    # Use float32 matmul for the final projection (more accurate, still fast)
    final_fp32 = F.linear(gated.float(), w["to_out.weight"].float())   # [R, dim] FP32

    # reshape to [B, N, N, dim]
    final = final_fp32.view(B, N, N, dim)

    return final
```
