@triton.jit
def norm_gate_kernel(out_ptr, gate_ptr, mu_ptr, rstd_ptr, gamma_ptr, beta_ptr, gated_ptr,
                     total_rows, H, N,
                     BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode b,i,j from row index
    # row index r = b*N*N + i*N + j
    N2 = N * N
    b = offs_m // N2
    rem = offs_m - b * N2
    i = rem // N
    j = rem - i * N

    hidden_stride = N2
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # offsets for hidden dimension in out and gate
        ptr_out = out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride

        # load
        val = tl.load(ptr_out, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)   # [M, Hblk]
        gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # mean & rstd
        mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]  # [M,1]
        # gamma & beta
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1, Hblk]
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)[None, :]    # [1, Hblk]

        # compute normalized = (val - mu) * rstd
        norm = (val - mu) * rstd
        # LN2
        norm = norm * gamma + beta
        # gate
        gated = norm * gate

        # store to gated_ptr: offset = offs_m[:, None] * H + offs_h[None, :]
        # But gated_ptr is contiguous with stride H.
        ptr_out_gated = gated_ptr + offs_m[:, None] * H + offs_h[None, :]   # [M, Hblk]

        tl.store(ptr_out_gated,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    device = x.device

    # 1. LayerNorm Input
    x_flat = x.reshape(R, dim).contiguous()

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # 2. Fused projection

    fused_weight = torch.cat(
        [

            weights["right_proj.weight"],
            weights["left_gate.weight"],

Algorithm (B = batch, N = seq_len, H = hidden_dim, D = dim)

1️⃣  Row‑wise LayerNorm on the flattened input tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)

5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H

6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the

7️⃣  Final linear projection (no bias):
        out_final = gated @ W_outᵀ  where W_out ∈ ℝ^{dim×H}
    performed with a single cuBLAS GEMM (`torch.nn.functional.linear`).
    The result is reshaped to [B, N, N, dim] (float32).

All kernels target Triton 3.3.1 (H100) and exploit mixed‑precision
arithmetic.  Only a small subset of the work (steps 1, 3, 5, 6) is
implemented in Triton; the heavy matrix multiplications leverage
cuBLAS/Tensor‑cores for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise layer‑norm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, D]  FP32 input
    y_ptr,    # [R, D]  FP16 output
    w_ptr,    # [D]    LN weight γ (FP32)
    b_ptr,    # [D]    LN bias   β (FP32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized FP16 ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * D + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣  Split fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # stride between successive H slices
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                 # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣  Row‑wise mean & rsqrt‑variance over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 6️⃣  Fuse second LayerNorm, out_gate and prepare for final projection
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    gated_ptr,      # [R, H] FP16   (output)
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Load hidden slice from out and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # [M, Hblk]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [M, Hblk]

        # Row‑wise μ / rstd
        mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]   # [M,1]

        # LayerNorm γ / β
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1,Hblk]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # [1,Hblk]

        # LN2 + mask‑gate
        norm = (vals_out - mu) * rstd                # (x-μ)/σ
        norm = norm * gamma + beta                   # γ·norm + β   → shape [M, Hblk]
        gated = norm * vals_gate                     # element‑wise gate

        # store to contiguous [R, H] buffer
        dst_ptr = gated_ptr + offs_m[:, None] * H + offs_h[None, :]
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Triton‑backed TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with the following keys (all float32):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict must contain:
            * "dim"          – D
            * "hidden_dim"   – H
            * "nomask"       – bool (True ⇒ ignore mask)

    Returns
    -------
    torch.Tensor : shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # number of (b,i,k) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                        # [R, D] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, D]

    proj = F.linear(x_norm, fused_weight)               # (R, 5·H) FP16
    del x_norm, fused_weight

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise update via batched GEMM
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N]
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64                # H ≤ 128 ⇒ fits comfortably
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and prepare for final projection
    # ------------------------------------------------------------------
    gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_GATE = 128
    BLOCK_H_GATE = 64
    grid_gate = (triton.cdiv(R, BLOCK_M_GATE),)
    norm_gate_kernel[grid_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        gated,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_GATE,
        BLOCK_H=BLOCK_H_GATE,
        num_warps=8,
    )
    # free intermediate tensors we no longer need
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (hidden → dim)
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)            # [dim, H] FP16
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    out_final_flat = F.linear(gated, w_out)                # (R, dim) FP16
    out_final = out_final_flat.view(B, N, N, dim).to(torch.float32)

    return out_final
