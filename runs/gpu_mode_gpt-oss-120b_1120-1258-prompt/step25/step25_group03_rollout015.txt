    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps=1e-5

    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim

    # Flatten input for layernorm
    x_flat = x.view(R, dim).contiguous()
    # Output for normalized
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Launch kernel
    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Fuse linear for five projections
    fused_weight = torch.cat([...], dim=0).to(torch.float16)  # shape (5*H, dim)

    proj = F.linear(x_norm, fused_weight)  # [R,5*H] fp16

    # Optional mask
    if not nomask and mask is not None:
        mask_flat = mask.view(R, 1).to(dtype=torch.float16)
    else:
        mask_flat = None

    # Slice and apply gates
    # shape [R,H] each
    left = proj[:, :H]
    right = proj[:, H:2*H]
    left_gate = torch.sigmoid(proj[:, 2*H:3*H].float()).half()
    right_gate = torch.sigmoid(proj[:, 3*H:4*H].float()).half()
    out_gate = torch.sigmoid(proj[:, 4*H:5*H].float()).half()

    if mask_flat is not None:
        left = left * mask_flat
        right = right * mask_flat

    left = left * left_gate
    right = right * right_gate

    # reshape to [B,H,N,N]
    left = left.view(B, N, N, H).permute(0,3,1,2).contiguous()
    right = right.view(B, N, N, H).permute(0,3,1,2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0,3,1,2).contiguous()

    # batched gemm

    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    # right transpose last two dims
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H, N, N] fp16

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the last channel of the pairwise tensor.
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM:  (R, dim) × (5·H, dim)ᵀ.
3️⃣  Slice the fused projection, apply the optional mask, compute
    sigmoid gates (in FP32) and obtain the three tensors
        left, right, out_gate   →   shape [B, H, N, N] (FP16).
4️⃣  Batched GEMM (torch.bmm) performs the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` in FP16.
5️⃣  Standard LayerNorm (torch.nn.functional.layer_norm) over the
    hidden dimension H for each (i, j) pair.
6️⃣  Multiply by the `out_gate` (FP32), then a final linear projection
    (`to_out`) from H → dim.  The result is cast back to float‑32.

The only Triton kernel used is the custom row‑wise LayerNorm in step 1.
All remaining heavy work is delegated to highly‑optimised PyTorch
kernels (cuBLAS, flash‑attention‑style kernels inside torch.bmm, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------

    b_ptr,      # [dim]    LN bias   (β)  FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass – mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass – write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N          # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over `dim` (custom Triton kernel → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: five linear maps in one GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                         # [R, 5·H]  FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Slice, optional mask, compute sigmoid gates
    # --------------------------------------------------------------
    # split the fused projection
    left      = proj[:, 0:H]
    right     = proj[:, H:2*H]
    left_gate = torch.sigmoid(proj[:, 2*H:3*H].float()).half()
    right_gate = torch.sigmoid(proj[:, 3*H:4*H].float()).half()
    out_gate  = torch.sigmoid(proj[:, 4*H:5*H].float()).half()
    del proj

    # optional per‑pair mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.view(R, 1).to(torch.float16)   # [R,1]
        left = left * mask_flat
        right = right * mask_flat
    # apply the element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # reshape to [B, H, N, N] (required layout for the batched GEMM)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: Σₖ left·rightᵀ  (still FP16)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    # move H to the last dimension so torch.layer_norm can operate on it
    out_perm = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H] FP16
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                    # still FP16
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] FP16
    del out_perm

    # --------------------------------------------------------------
    # 6️⃣ Gate, final projection and reshape back to [B, N, N, dim]
    # --------------------------------------------------------------
    # bring everything to FP32 for the final matmul (more accurate)
    gated = out_norm.float() * out_gate.float()           # [B, H, N, N] FP32
    del out_norm, out_gate

    # reshape to (R, H) for the final linear layer
    gated = gated.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    gated_flat = gated.view(R, H)                         # [R, H]  FP32
    del gated

    # final projection: (R, H) × (dim, H)ᵀ → (R, dim)
    # weight is stored as (dim, H) in the original model
    w_out = w["to_out.weight"].to(torch.float32)          # [dim, H]
    final_flat = F.linear(gated_flat, w_out)              # [R, dim] FP32
    final = final_flat.view(B, N, N, dim)                 # [B, N, N, dim]

    return final
