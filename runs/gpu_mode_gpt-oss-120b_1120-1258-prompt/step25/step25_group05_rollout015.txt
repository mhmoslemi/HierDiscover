    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1) Initial layernorm (rowwise)
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # 2) fused projection
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)   # shape [5*H, dim]
    proj = F.linear(x_norm, fused_weight)  # shape [B,N,N,5*H]
    proj_flat = proj.view(R, 5*H).contiguous()  # flatten to 2D

    # 3) split, mask & gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # cleaning up unused tensors
    del proj, proj_flat

    # 4) batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = out.view(B, H, N, N)

    # 5) second layernorm over H and apply out_gate

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# kernels definitions as above...

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    This implementation:
    1. Row‑wise LayerNorm on the input tensor (torch.nn.functional.layer_norm).
    2. A fused linear projection that simultaneously computes left/right
       projections and their sigmoid gates.
    3. A Triton kernel (`split_fuse_gate_kernel`) that splits the fused tensor,
       applies the optional pair mask and the three sigmoid gates, writing the
       three tensors `left`, `right` and `out_gate` in layout [B,H,N,N].
    4. Batched GEMM (`torch.bmm`) to compute the core einsum
       `out[b,i,j,:] = Σₖ left[b,i,k,:]·right[b,j,k,:]`.
    5. A second LayerNorm over the hidden dimension `H`, multiplication with
       `out_gate`, and a final linear projection to `dim`.  The second LayerNorm
       and the final projection are performed with standard PyTorch operators
       (torch.nn.functional.layer_norm + torch.matmul) which use cuBLAS and are
       highly optimized for the tall‑skinny shape (R×H)·(H×dim).
    All intermediate tensors are kept in FP16, only the two LayerNorms use FP32
    statistics to retain numerical fidelity.  The only required Triton kernel
    is the splitting/fusing kernel, satisfying the “partial‑kernel” rule.
    """
    # unpack
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # torch.nn.functional.layer_norm works over the last dimension
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # 2️⃣ Fuse the five linear maps (left/right + three gates)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],

    proj_flat = proj.view(R, 5 * H).contiguous()
    del proj, x_norm  # free memory

    # 3️⃣ Split, mask and apply sigmoid gates

    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        mask_ptr = mask_flat
    else:

        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_H = 64

        proj_flat,
        left,
        right,

        R,

        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj_flat, mask_ptr

    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)
    del left, right  # we no longer need them

    # 5️⃣ Second LayerNorm + out_gate
    # Convert tensors to FP32 for the second LN (matches the reference implementation)
    out_fp32 = out.to(torch.float32)
    out_perm = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    out_gate_fp32 = out_gate.permute(0, 2, 3, 1).contiguous().to(torch.float32)
    out_norm_gate = out_norm * out_gate_fp32  # FP32
    del out, out_gate, out_gate_fp32, out_norm, out_perm, out_fp32

    # 6️⃣ Final projection (to_out)
    # weights["to_out.weight"] has shape [dim, hidden_dim]
    final = torch.matmul(
        out_norm_gate.view(R, H),  # [R, H] FP32
        weights["to_out.weight"].t(),  # [H, dim] FP32
    )  # [R, dim] FP32
    final = final.view(B, N, N, dim)  # [B, N, N, dim]

"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input pair tensor (FP32 → FP16) – uses
    torch.nn.functional.layer_norm which is highly optimized on GPUs.

2️⃣  Fuse the five linear maps (left_proj, right_proj,
    left_gate, right_gate, out_gate) into one matrix‑multiply:
    (B·N·N, dim) × (dim, 5·hidden_dim) → (B·N·N, 5·hidden_dim)  (FP16).

3️⃣  Split the fused matrix, apply the optional pair mask and the three
    sigmoid gates in a single Triton kernel (`split_fuse_gate_kernel`),
    writing the three tensors `left`, `right` and `out_gate`
    in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B,H,N,N] (FP16).

5️⃣  Second LayerNorm (over the hidden dimension H) + out‑gate multiplication
    – performed with the native `torch.nn.functional.layer_norm` (FP32
    statistics, FP32 accumulation) and a cheap element‑wise product.

6️⃣  Final linear projection to the original channel dimension (`dim`):
    a tall‑skinny matrix‑multiply (R×H)·(H×dim) using cuBLAS (FP32).

Only the splitting / gating step is written in Triton, satisfying the
“must contain a kernel” rule while the heavy linear algebra is delegated
to cuBLAS for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride when advancing hidden dim
    base = b * (H * hidden_stride) + i * N + k   # start offset inside each H‑plane

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping with the model’s parameters (all float32)
            - config_dict   : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily tuned for GPUs.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)            # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)                # [B, N, N, 5·H]  FP16
    proj_flat = proj.view(R, 5 * H).contiguous()
    del proj, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj_flat, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: core einsum (left × rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over H) + out_gate multiplication
    # ------------------------------------------------------------------
    # Cast to FP32 for the LN to keep numerical fidelity (matches reference)
    out_fp32 = out.to(torch.float32)
    out_perm = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                     # FP32

    out_gate_fp32 = out_gate.permute(0, 2, 3, 1).contiguous().to(torch.float32)
    out_norm_gate = out_norm * out_gate_fp32               # FP32
    del out, out_gate, out_fp32, out_norm, out_perm, out_gate_fp32

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection to `dim`
    # ------------------------------------------------------------------
    # (R, H) @ (H, dim)  →  (R, dim)
    final = torch.matmul(
        out_norm_gate.view(R, H),       # [R, H]  FP32
        w["to_out.weight"].t(),         # [H, dim]  FP32
    )                                   # [R, dim] FP32
    final = final.view(B, N, N, dim)    # [B, N, N, dim]

    return final
