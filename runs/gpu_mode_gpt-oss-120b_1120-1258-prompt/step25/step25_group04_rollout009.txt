"""
Outgoing TriMul (AlphaFold‑3) – mixed‐precision Triton + cuBLAS implementation
============================================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM:  (R, dim) × (5·H, dim)ᵀ → (R, 5·H)   (FP16).
3️⃣ Split the fused projection, optionally mask it, apply sigmoid gates and
   store three tensors **left**, **right**, **out_gate** in layout
   [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing **out** ∈ FP16 of shape [B, H, N, N].
5️⃣ Permute **out** and **out_gate** to [B, N, N, H] → (R, H) so that the
   hidden dimension becomes contiguous.  Compute row‑wise mean/variance
   of **out** (FP32) with a tiny Triton kernel.
6️⃣ Fuse the second LayerNorm (γ/β), the *out_gate* multiplication and
   store the gated, normalised hidden vectors **term** ∈ FP16 (R, H)
   with another Triton kernel.
7️⃣ Final projection is a single large GEMM performed by cuBLAS via
   ``torch.nn.functional.linear``:
       out_final = term @ (to_out.weight)ᵀ                 (FP16)
   Cast the result to FP32 and reshape to [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target an NVIDIA H‑100.
Mixed‑precision (FP16 bulk, FP32 for reductions) is used throughout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    γ (FP32)
    b_ptr,          # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – compute mean & variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalised values --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance (contiguous hidden dim)
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_contig_kernel(
    out_ptr,      # [R, H]  FP16 (contiguous)
    mu_ptr,       # [R]    FP32
    rstd_ptr,     # [R]    FP32
    total_rows, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate & store gated hidden vectors
# --------------------------------------------------------------
@triton.jit
def term_kernel(
    out_ptr,        # [R, H]   FP16 (contiguous)
    gate_ptr,       # [R, H]   FP16 (contiguous)
    mu_ptr,         # [R]      FP32
    rstd_ptr,       # [R]      FP32
    ln_w_ptr,       # [H]      FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]      FP32 (to_out_norm.bias)
    term_ptr,       # [R, H]   FP16 (output)
    total_rows, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load hidden‑plane values (FP16 → FP32)
        out_vals = tl.load(out_ptr  + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]
        term = norm * gate_vals

        tl.store(term_ptr + offs_m[:, None] * H + offs_h[None, :],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# 5️⃣ Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pair positions
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                    # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                       # FP16 result
    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    # (R, dim) × (5·H, dim)ᵀ → (R, 5·H)   (FP16)
    proj = F.linear(x_norm, fused_w)  # FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left, right, out_gate
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # -----------------------------------------------------------------
    # reshape to [B·H, N, N] and use torch.bmm
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    # out = left @ rightᵀ   (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Make hidden dimension contiguous: (B, N, N, H)
    # -----------------------------------------------------------------
    # After contraction we need the hidden axis as the innermost dimension
    # for fast reductions / GEMMs.
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    out_flat = out_perm.view(R, H)                          # [R, H] FP16
    gate_flat = out_gate_perm.view(R, H)                    # [R, H] FP16
    del out, out_gate, out_perm, out_gate_perm

    # -----------------------------------------------------------------
    # 6️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # -----------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_contig_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm + out_gate → gated hidden vectors `term`
    # -----------------------------------------------------------------
    term = torch.empty(R, H, dtype=torch.float16, device=device)   # [R, H]
    BLOCK_M_TERM = 128
    BLOCK_H_TERM = 32
    grid_term = (triton.cdiv(R, BLOCK_M_TERM),)
    term_kernel[grid_term](
        out_flat,
        gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        term,
        R,
        H,
        BLOCK_M=BLOCK_M_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=8,
    )
    del out_flat, gate_flat, mu, rstd

    # -----------------------------------------------------------------
    # 8️⃣ Final projection (large GEMM) via cuBLAS
    # -----------------------------------------------------------------
    # weight shape: [dim, H] (FP16)
    to_out_w = w["to_out.weight"].to(torch.float16)
    final_fp16 = F.linear(term, to_out_w)               # (R, dim) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # (B, N, N, dim) FP32
    return final
```
