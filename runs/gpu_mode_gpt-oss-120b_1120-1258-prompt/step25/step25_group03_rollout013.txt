"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm (FP16 heavy, FP32 for numerics)
-----------------------------------------
1️⃣ Input LayerNorm (torch, FP32 → FP16).
2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into one GEMM:  (R, dim) @ (5·H, dim).T
   where R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, optionally mask the pairwise
   representation and apply the three sigmoid gates.
   Output tensors: left, right, out_gate   →   shape [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) contracts over the second sequence axis:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   Producing a tensor `[B, H, N, N]` in FP16.
5️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
   (Triton kernel).  These statistics are needed for the second
   LayerNorm.
6️⃣ Fuse the second LayerNorm, multiplication by the `out_gate`,
   and the final projection into **two** Triton kernels:
   – `rowwise_stats_kernel` (step 5) and
   – `apply_ln_gate_kernel` (step 6) which writes a tensor `term`
     = LN(out) ⊙ out_gate   of shape `[B, H, N, N]` (FP16).
7️⃣ Final projection is a single large GEMM on the flattened tensors:
        term_flat  [R, H]  @  w_out_T [H, dim] → [R, dim]
   (cuBLAS, FP16 → FP32).  Reshape to `[B, N, N, dim]` and return.

All heavy compute (steps 3‑6) is performed in custom Triton kernels,
while the two matrix‑multiplications (steps 4 and 7) use cuBLAS which
is optimal on H100.  The implementation works for any
batch size ≤ 2, sequence length ∈ {128,256,512,1024},
`dim` ∈ {128,384} and `hidden_dim` = 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # broadcast mask over hidden dimension (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between two hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16 for storage
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate  # out_gate is *not* masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Apply second LayerNorm, multiply by out_gate and write “term”
# ----------------------------------------------------------------------
@triton.jit
def apply_ln_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16  (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16  (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    term_ptr,         # [B, H, N, N] FP16  (output of LN⊙gate)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load hidden slice of out and gate (FP16 → FP32)
        out_vals = tl.load(
            out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm parameters γ, β (FP32)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # (out - μ) * rstd → γ· + β → * out_gate
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_vals

        # store term back as FP16
        tl.store(
            term_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            term.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pair positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch's highly‑optimised kernel)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    x_flat = x_norm.reshape(R, dim).contiguous()   # [R, dim] FP16
    proj = F.linear(x_flat, fused_w)                # [R, 5·H] FP16
    del x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # reasonable block sizes for H100
    BLOCK_R = 128
    BLOCK_H = 32
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (hidden‑dim stats)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply second LayerNorm, multiply by out_gate → term tensor
    # ------------------------------------------------------------------
    term = torch.empty_like(out, dtype=torch.float16, device=device)

    BLOCK_M_TERM = 256
    BLOCK_H_TERM = 32
    apply_ln_gate_kernel[(triton.cdiv(R, BLOCK_M_TERM),)](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        term,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection: term_flat [R, H] @ w_out_T [H, dim]
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    term_flat = term.permute(0, 2, 3, 1).contiguous().view(R, H)      # [R, H]
    final_fp16 = torch.matmul(term_flat, w_out_T)                     # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # [B, N, N, dim]

    return final
