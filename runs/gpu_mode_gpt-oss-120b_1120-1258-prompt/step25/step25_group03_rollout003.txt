"""
Triton‐accelerated *outgoing* TriMul (AlphaFold‑3) forward pass.
----------------------------------------------------------------
The implementation follows the original PyTorch formulation but
fuses the expensive parts that cannot be expressed efficiently with
a single GEMM:

1️⃣  Layer‑Norm on the last channel of the pair‑wise tensor (FP32→FP16)
    – performed with torch.nn.functional.layer_norm (highly‑optimised).

2️⃣  A *single* fused linear projection of dimension 5·hidden_dim
    (left_proj, right_proj, left_gate, right_gate, out_gate) using a
    cuBLAS GEMM.

3️⃣  A custom Triton kernel (`split_fuse_gate_kernel`) that
    • splits the fused projection into the three tensors
      *left*, *right* and *out_gate*,
    • applies the optional per‑pair mask,
    • applies the sigmoid gates.

4️⃣  Batched matrix‑multiply (cuBLAS) for the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]

5️⃣  A second Layer‑Norm (over the hidden dimension), multiplication
    by `out_gate`, and the final projection to the original channel
    dimension – all expressed with regular torch kernels
    (layer_norm + linear) which are also cuBLAS‑accelerated.

All heavy linear algebra is left to cuBLAS/torch while the only
required Triton code (step 3) is kept deliberately simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣  Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fused projection (FP16)
    left_ptr,          # [B, H, N, N] (FP16)
    right_ptr,         # [B, H, N, N] (FP16)
    out_gate_ptr,      # [B, H, N, N] (FP16)
    mask_ptr,          # [R] optional mask (FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows processed per program
    BLOCK_H: tl.constexpr,   # hidden dimension tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcasted over hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between consecutive hidden planes in the output tensors
    hidden_stride = N * N                # distance between h and h+1
    # offset of the first hidden plane for this (b,i,k) location
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view into the fused projection:
        # [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 (stable) → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # out_gate itself is not masked

        # target address of each element in the 4‑D tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pair positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the last channel (FP32 → FP16)
    # --------------------------------------------------------------
    # torch's fused layer‑norm is faster than a hand‑written Triton one.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)           # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    # flatten to (R, dim) for the GEMM
    x_flat = x_norm.reshape(R, dim).contiguous()
    proj = F.linear(x_flat, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm, x_flat

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask & sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        # dummy tensor – never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64      # rows per program (tuneable)
    BLOCK_H = 64      # hidden‑dim tile (tuneable)

    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: the core “einsum”
    # --------------------------------------------------------------
    # left/right have shape [B, H, N, N]; we view them as a batch of
    # (B·H) matrices of shape (N, N) and multiply left @ rightᵀ.
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + gate + final projection
    # --------------------------------------------------------------
    # Bring the hidden dimension to the last axis so that torch.layer_norm
    # normalises over it.
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # Layer‑norm over the hidden channel
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                       # FP16

    # Apply the out‑gate
    out_gated = out_norm * gate_perm                        # FP16

    # Final projection back to `dim`
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    out_flat = out_gated.reshape(R, H)                      # [R, H]
    final_fp16 = F.linear(out_flat, w_out_T)                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
