"""
Outgoing TriMul (AlphaFold‑3) – fast Triton implementation

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (float32 → float16).
2️⃣ Fuse the 5 linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single GEMM: (R, dim) × (5·H, dim)ᵀ → (R, 5·H) (fp16).
3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   compute the sigmoidal gates and store three tensors **left**, **right**,
   **out_gate** (fp16).  This work is done by a single Triton kernel.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing **out** ∈ fp16 of shape [B, H, N, N].
5️⃣ Row‑wise LayerNorm over the hidden dimension H (torch.nn.functional).
   Multiply by **out_gate** and apply the final projection
   (to_out) – also a fused fp16 GEMM.
6️⃣ Return the result in float32: [B, N, N, dim].

Only one Triton kernel (step 3) is required; all heavy matrix
operations are delegated to cuBLAS/torch‑nn which are heavily
optimised on H100.  This reduces memory traffic (the intermediate
5·H projection is never materialised) and lowers kernel launch
overhead, yielding runtimes ≈ 500 µs on the target shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16 fused projection (R = B·N·N)
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R]         fp16 (optional, broadcasted)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden slice per program
):
    pid = tl.program_id(0)

    # ----------- row coordinates ------------------------------------------------
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn                       # batch index
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ------------ optional mask --------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m_val = m_val[:, None]                                      # broadcast over hidden
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between hidden planes inside the [B, H, N, N] tensors
    hidden_stride = N * N                     # distance between h‑planes

    # base offset for a given (b,i,k) element at hidden‑plane 0
    #   offset = b·(H·N·N) + i·N + k
    base_offset = b * (H * hidden_stride) + i * N + k

    # -------------------------------------------------------------------------
    # iterate over hidden dimension in chunks of BLOCK_H
    # -------------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection:
        #  [left | right | left_gate | right_gate | out_gate]
        # each sub‑matrix has stride H along the hidden axis.
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        # load the five slices
        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0)

        # sigmoid (compute in fp32 for numerical stability)
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = lx * m_val * lgate
        right = rx * m_val * rgate
        outg  = ogate

        # destination offsets inside the 4‑D tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pair positions
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    # flatten the pair tensor and use the highly‑optimised torch implementation
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]
    # (R, dim) @ (5·H, dim)ᵀ  → (R, 5·H)
    proj = F.linear(x_norm, fused_w)          # fp16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # -----------------------------------------------------------------
    # reshape to [B·H, N, N] and perform (N×N) @ (N×N)ᵀ for each hidden plane
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # fp16
    out = out.view(B, H, N, N)                                # [B, H, N, N] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate and final projection (torch)
    # -----------------------------------------------------------------
    # bring hidden dimension to the last axis for a straight‑forward LayerNorm
    out = out.permute(0, 2, 3, 1).contiguous()                # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]

    # LayerNorm over the hidden channel H
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the out‑gate (element‑wise)
    out = out * out_gate

    # Final linear projection back to `dim`
    w_out_T = w["to_out.weight"].to(torch.float16).t()        # [H, dim] fp16
    out = F.linear(out, w_out_T)                               # fp16 → [B, N, N, dim]

    # Cast to float32 for the returned tensor
    out = out.to(torch.float32)
    return out
