import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index: r = b*N*N + i*N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast across hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]  # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N  # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h=0

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        # load slices (FP16)
        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid on fp32 for numerical stability, back to fp16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

    Algorithm
    ---------
    1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
        torch.nn.functional.layer_norm.
    2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
       into a single GEMM.
    3️⃣ Split the fused projection, apply the optional pair‑mask, compute sigmoid
       gates and write three tensors *left*, *right* and *out_gate* (FP16) with
       a Triton kernel.
    4️⃣ Batched GEMM (cuBLAS) computes the core contraction
       ``out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]``.
    5️⃣ Row‑wise LayerNorm (hidden dimension) on ``out`` with learnable γ/β,
       element‑wise multiplication by ``out_gate`` and a final linear projection.
       The last three steps are implemented with `torch.nn.functional.layer_norm`,
       elementwise multiplication and a standard FP16 → FP32 GEMM.
    """
    # unpack inputs
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()
    # torch.layer_norm works on float32; later cast to half
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    proj = F.linear(x_norm, fused_w)  # (R, 5·H) FP16
    del fused_w, x_norm, x_flat

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        has_mask = 1
    else:
        # dummy tensor – will never be accessed when HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication (left @ rightᵀ)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # compute contraction: (B·H, N, N) × (B·H, N, N)ᵀ  → (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # free left/right early
    del left, right, left_mat, right_mat

    out = out.view(B, H, N, N)   # now (B, H, N, N) FP16

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm → gate → final projection
    # --------------------------------------------------------------
    # flatten to (R, H) where each row corresponds to a pair (b,i,j)
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)            # [R, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # [R, H]
    del out, out_gate

    # layer‑norm on hidden dimension (γ, β are FP32)
    out_ln = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # element‑wise gate (still FP16)
    out_gated = out_ln * out_gate_flat

    # final linear projection (FP16 × FP16 → FP32)
    final_flat = F.linear(out_gated, w["to_out.weight"])   # [R, dim] FP16
    final = final_flat.to(torch.float32).view(B, N, N, dim)

    return final

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)

4️⃣ Batched matrix multiplication (cuBLAS) computes the core contraction
   out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k].

   (layer_norm + linear).  All heavy arithmetic (step‑4) stays in FP16,

        normalized_shape=(dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                       # [R, dim] FP16

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection – creates the five intermediate tensors at once
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                           # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                               # [R, 5·H] FP16
    del fused_w, x_norm, x_flat

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid → left / right / out_gate
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        has_mask = 1
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # -----------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    del left, right, left_mat, right_mat

    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm → gate → final projection (all via torch kernels)
    # -----------------------------------------------------------------
    # reshape to (R, H) where each row corresponds to a pair (b,i,j)
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)            # [R, H] FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # [R, H] FP16
    del out, out_gate

    # LayerNorm over the hidden dimension (γ/β are FP32 → cast to FP16 for torch)
    to_out_norm_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    out_ln = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=to_out_norm_w_fp16,
        bias=to_out_norm_b_fp16,
        eps=eps,
    )                                                          # [R, H] FP16

    # Gate after the second LN (still FP16)
    gated = out_ln * out_gate_flat

    # Final linear – weight is half‑precision, bias is absent in the original model
    to_out_w_fp16 = w["to_out.weight"].to(torch.float16)        # [dim, H]
    final_flat = F.linear(gated, to_out_w_fp16)                # [R, dim] FP16

    # Convert to the required float32 output layout
    final = final_flat.to(torch.float32).view(B, N, N, dim)    # [B, N, N, dim]

    return final
```
