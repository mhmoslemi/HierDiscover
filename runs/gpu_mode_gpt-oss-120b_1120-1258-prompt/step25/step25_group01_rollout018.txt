@triton.jit
def proj_gate_fused_kernel(
    x_ptr,      # [R, dim] fp16
    weight_ptr, # [dim, 5H] fp16 (transposed fused weight)
    mask_ptr,   # optional [R] fp16
    left_ptr, right_ptr, out_gate_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
   # Program id across rows
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R) # [BLOCK_R]
   mask_r = offs_r < R

   # decode rows into b,i,k
   nn = N * N
   b = offs_r // nn
   rem = offs_r - b * nn
   i = rem // N
   k = rem - i * N

   if HAS_MASK:
       mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
       mask_val = mask_val[:, None] # [BLOCK_R, 1]
   else:
       mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

   # Compute hidden stride for storing: hidden_stride = N * N
   hidden_stride = N * N
   base_offset = b * (H * hidden_stride) + i * N + k

   # allocate accumulator
   # We will compute all 5H outputs
   acc = tl.zeros([BLOCK_R, BLOCK_H * 5], dtype=tl.float32) # Use float32 for accumulation

   for d_start in range(0, dim, BLOCK_D):
       offs_d = d_start + tl.arange(0, BLOCK_D)   # [BLOCK_D]
       mask_d = offs_d < dim

       # Load A submatrix
       a_ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]  # [BLOCK_R, BLOCK_D]
       a = tl.load(a_ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

       # Load B submatrix
       # B is weight_T: shape [dim,5H]
       b_ptr = weight_ptr + offs_d[:, None] * (5 * H) + tl.arange(0, BLOCK_H * 5)[None, :]  # [BLOCK_D, BLOCK_O]
       # However we need to apply mask for column maybe.
       b = tl.load(b_ptr, mask=mask_d[:, None] & (tl.arange(0, BLOCK_H * 5)[None, :] < (5 * H)), other=0.0).to(tl.float32)

       # dot
       acc += tl.dot(a, b)  # shape [BLOCK_R, BLOCK_O]

   # Now acc holds fused projection values (5H per row) in fp32.
   # Convert to fp16 for further processing
   fused_fp16 = acc.to(tl.float16)  # [BLOCK_R, 5H]

   # Split
   # Offsets for each segment:
   # left_raw = fused[..., 0:H]
   # right_raw = fused[..., H:2H]
   # left_gate_raw = fused[..., 2H:3H]
   # right_gate_raw = fused[..., 3H:4H]
   # out_gate_raw = fused[..., 4H:5H]

   # Compute sigmoid gates
   # We'll compute row-wise mask also.

   # For each h segment, we need to store result to appropriate tensor.

   # To process efficiently, we may loop over BLOCK_H segment stride.
   # We'll write a loop over h_start increments of BLOCK_H
   # Use offsets to store.

   # first compute left and right gating/in apply mask:
   for h_off in range(0, H, BLOCK_H):
       cur_h = h_off + tl.arange(0, BLOCK_H)
       mask_h = cur_h < H

       # slice indices
       left_raw = fused_fp16[:, h_off:h_off+BLOCK_H]  # shape [BLOCK_R, BLOCK_H]
       right_raw = fused_fp16[:, H + h_off:H + h_off+BLOCK_H]
       left_gate_raw = fused_fp16[:, 2*H + h_off:2*H + h_off+BLOCK_H]
       right_gate_raw = fused_fp16[:, 3*H + h_off:3*H + h_off+BLOCK_H]
       out_gate_raw = fused_fp16[:, 4*H + h_off:4*H + h_off+BLOCK_H]

       # Sigmoid for gates
       left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
       right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
       out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

       # Apply mask to left and right (broadcast)
       left = left_raw * mask_val * left_gate
       right = right_raw * mask_val * right_gate
       out = out_gate  # just out_gate values

       # Compute destination offsets
       dst_offset = base_offset[:, None] + cur_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

       tl.store(left_ptr + dst_offset, left, mask=mask_r[:, None] & mask_h[None, :])
       tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
       tl.store(out_gate_ptr + dst_offset, out, mask=mask_r[:, None] & mask_h[None, :])
   # end.

def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. layernorm
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat, x_norm,
        w["norm.weight"], w["norm.bias"],
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # 2. fused projection + gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # build fused weight transposed
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]

    fused_weight_T = fused_weight.t().contiguous()  # [dim, 5*H]

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_fused_kernel[grid_proj](
        x_norm,
        fused_weight_T,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # free x_norm and fused_weight to reduce memory
    del x_norm, fused_weight, fused_weight_T

    # 3. batched GEMM using torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)  # [B, H, N, N] FP16

    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,

    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

        eps,

        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final

@triton.jit
def proj_gate_fused_kernel(
    x_ptr, weight_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode row into b, i, k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask value
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute base offset for left/right/out_gate storage
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    # loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        cur_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # accumulator for 5*BLOCK_H outputs per row
        acc = tl.zeros([BLOCK_R, 5 * BLOCK_H], dtype=tl.float32)

        # loop over dim blocks
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
            mask_d = offs_d < dim

            # load x fragment

                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)                                 # fp16

                        mask=mask_d[:, None] & col_mask[None, :],
                        other=0.0)                                 # fp16

            acc += tl.dot(a, w)                                 # fp32

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣  **Fused projection + gating kernel**:
    * Compute the five linear maps (left, right, left_gate, right_gate,
      out_gate) directly from the normalized input.
    * Apply the optional mask and the sigmoid gates.
    * Store the three results (left, right, out_gate) in [B, H, N, N] FP16.
    * This replaces the heavyweight F.linear + split‑fuse‑gate step and
      removes the large intermediate (R×5H) buffer.
3️⃣  Batched GEMM (cuBLAS) over the hidden dimension:
    out = left @ rightᵀ   →   [B, H, N, N] FP16.
4️⃣  Row‑wise mean & rsqrt‑variance of `out` (triton kernel).
5️⃣  Final kernel:
    * Layer‑norm over the hidden dimension using the statistics from (4).
    * Multiply by the out‑gate.
    * Fuse the second LayerNorm, the out‑gate, and the final projection
      (to_out) into a single kernel, producing the final FP32 tensor
      `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and target an H100.  The
implementation uses mixed‑precision (FP16 for the bulk of the work and
FP32 for the reductions) and runs completely on‑device – no extra
Python‐level loops.

The code below defines the kernels and the entry‑point `custom_kernel`,
which receives `(input, mask, weights, config)` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]    LN weight (γ) FP32
    b_ptr,      # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating (replaces linear + split kernel)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    x_ptr,               # [R, dim] FP16 normalised input
    weight_ptr,          # [dim, 5*H] FP16 fused weight (transposed)
    mask_ptr,            # [R] FP16 (optional, broadcast over H)
    left_ptr,            # [B, H, N, N] FP16
    right_ptr,           # [B, H, N, N] FP16
    out_gate_ptr,        # [B, H, N, N] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden plane 0

    # iterate over hidden dimension in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        cur_h = h_off + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = cur_h < H

        # accumulator for the 5 * BLOCK_H projected values
        acc = tl.zeros([BLOCK_R, 5 * BLOCK_H], dtype=tl.float32)

        # dot‑product over the feature dimension dim
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # A = X_norm[offs_r, offs_d]   -> [BLOCK_R, BLOCK_D] (FP16)
            a = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)                                 # FP16

            # column indices for the 5 groups of weights
            col_idx  = tl.arange(0, 5 * BLOCK_H)                # [5*BLOCK_H]
            col_grp  = col_idx // BLOCK_H
            col_in   = col_idx % BLOCK_H
            col      = h_off + col_in + col_grp * H             # [5*BLOCK_H]
            col_mask = col < (5 * H)

            # B = W_T[offs_d, col]   -> [BLOCK_D, 5*BLOCK_H] (FP16)
            w = tl.load(weight_ptr + offs_d[:, None] * (5 * H) + col[None, :],
                        mask=mask_d[:, None] & col_mask[None, :],
                        other=0.0)                                 # FP16

            acc += tl.dot(a, w)                                 # acc: FP32

        # ----- split the fused projection -----
        acc_fp16 = acc.to(tl.float16)          # [BLOCK_R, 5*BLOCK_H]

        left_raw      = acc_fp16[:, 0:BLOCK_H]                     # L
        right_raw     = acc_fp16[:, BLOCK_H:2 * BLOCK_H]           # R
        left_gate_raw = acc_fp16[:, 2 * BLOCK_H:3 * BLOCK_H]       # lg
        right_gate_raw= acc_fp16[:, 3 * BLOCK_H:4 * BLOCK_H]       # rg
        out_gate_raw  = acc_fp16[:, 4 * BLOCK_H:5 * BLOCK_H]       # og

        # sigmoid gates (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # apply mask and gating
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        og    = out_gate                                            # no mask

        # destination offsets for the current hidden tile
        dst_offset = base_offset[:, None] + cur_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        store_mask = mask_r[:, None] & mask_h[None, :]
        tl.store(left_ptr      + dst_offset, left,      mask=store_mask)
        tl.store(right_ptr     + dst_offset, right,     mask=store_mask)
        tl.store(out_gate_ptr  + dst_offset, og,        mask=store_mask)


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16  (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row mean / rstd
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # values from the contracted tensor and its gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M, K]

        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm (γ,β) over the hidden dimension
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by the out‑gate
        term = norm * vals_gate                                   # [M, K]
        term_fp16 = term.to(tl.float16)                           # back to fp16 for GEMM

        # final projection weight slice
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float) or None
        - weights_dict : model weights (all float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (replaces linear + split)
    # --------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # build the fused weight matrix (5·H × dim) and transpose it
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    fused_w_T = fused_w.t().contiguous()        # [dim, 5·H]

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_fused_kernel[grid_proj](
        x_norm,
        fused_w_T,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # free temporary tensors
    del x_norm, fused_w, fused_w_T

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
