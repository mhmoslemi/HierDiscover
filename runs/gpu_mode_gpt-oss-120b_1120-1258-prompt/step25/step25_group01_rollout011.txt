def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # Step 1: Input layernorm (float32 -> float16)
    x_flat = x.reshape(R, dim)
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,),
                                            weight=w["norm.weight"],
                                            bias=w["norm.bias"],
                                            eps=eps)
    x_norm = x_norm.to(torch.float16)

    # Step 2: Fused linear projection: 5*H x dim weight
    fused_w = torch.cat([w["left_proj.weight"],
                         w["right_proj.weight"],
                         w["left_gate.weight"],
                         w["right_gate.weight"],
                         w["out_gate.weight"]], dim=0)  # [5*H, dim]
    fused_w = fused_w.to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # [R, 5*H]

    # Step 3: Split and apply mask/gates using Triton kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        # dummy pointer: shape (0,)
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free proj
    del proj, mask_ptr

    # Step 4: Batched GEMM (einsum)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    out = out.view(B, H, N, N)  # [B, H, N, N]

    # Step 5: Rowwise second LayerNorm (over hidden dim H) / out_gate multiplication / final linear
    # Flatten out and out_gate to (R, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]

    # LayerNorm over hidden dim
    out_norm = torch.nn.functional.layer_norm(out_flat, (H,),
                                              weight=w["to_out_norm.weight"],
                                              bias=w["to_out_norm.bias"],
                                              eps=eps)
    # Multiply by out_gate
    out_norm = out_norm * out_gate_flat

    # Final linear projection
    w_out = w["to_out.weight"].to(torch.float16)  # [dim, H]
    final_flat = torch.nn.functional.linear(out_norm, w_out)  # [R, dim]
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final

"""Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm of the input tensor (float32 → fp16) using
    :func:`torch.nn.functional.layer_norm`.
2️⃣  Single fused linear projection that produces the five tensors
    (left, right, left_gate, right_gate, out_gate) in one GEMM.
3️⃣  A tiny Triton kernel splits the fused projection, applies the optional
    mask and computes the three sigmoid gates, writing the three tensors
    `left`, `right` and `out_gate` in layout ``[B, H, N, N]`` (fp16).
4️⃣  Batched matrix‑multiplication (cuBLAS) implements the core einsum:
       ``out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]``.
5️⃣  Row‑wise LayerNorm over the hidden dimension ``H`` using
    :func:`torch.nn.functional.layer_norm`, then element‑wise multiplication
    with ``out_gate``.
6️⃣  Final linear projection (hidden → model dimension) with a FP16 weight
    matrix, producing the result in float32.

Only step 3 is written in Triton – the rest uses highly‑optimised PyTorch
kernels (layer‑norm, GEMM, BMM).  This drastically reduces kernel‑launch
overhead while keeping the computational pattern identical to the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣  Split fused projection + mask + sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] fused projection (fp16)
    left_ptr,       # [B, H, N, N] output left  (fp16)
    right_ptr,      # [B, H, N, N] output right (fp16)
    out_gate_ptr,   # [B, H, N, N] output out_gate (fp16)
    mask_ptr,       # [R] optional mask (fp16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k  (k is the second sequence dim)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcast over the hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for (b, h=0, i, k)
    base_offset = b * (H * hidden_stride) + i * N + k

    # Iterate over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # View of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid in fp32 for numerical stability, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets for the three output tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    # Flatten and apply LayerNorm over the last dimension (dim)
    x_flat = x.reshape(R, dim)                         # [R, dim] FP32
    x_norm = F.layer_norm(x_flat,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)                  # FP32
    x_norm = x_norm.to(torch.float16)                  # FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection producing 5·H features
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim] FP16
    proj = F.linear(x_norm, fused_w)        # [R, 5·H] FP16
    del fused_w, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates  (Triton kernel)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM implements the core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    # out = Σ_k left[i,k] * right[j,k]  →  left @ rightᵀ
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over hidden dimension, gate, final projection
    # ------------------------------------------------------------------
    # Re‑order to (B, N, N, H) so that each row corresponds to one (i,j) pair
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # [R, H] FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H] FP16

    # LayerNorm over hidden dimension H (still in FP16, params are FP32)
    out_norm = F.layer_norm(out_flat,
                            normalized_shape=(H,),
                            weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"],
                            eps=eps)                                 # FP16

    # Apply the out‑gate (already sigmoid‑ed inside the kernel)
    out_norm = out_norm * out_gate_flat                                 # FP16

    # Final linear projection  (H → dim)
    w_out = w["to_out.weight"].to(torch.float16)                         # [dim, H] FP16
    final_flat = F.linear(out_norm, w_out)                               # [R, dim] FP16

    # Reshape to the original 4‑D layout and cast to float32
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
```
