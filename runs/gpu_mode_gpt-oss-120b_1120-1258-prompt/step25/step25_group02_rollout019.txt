    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm statistics (mean & rsqrt‑var) are computed on the
   [B·N·N, dim] tensor `x` – no normalized tensor is materialised.
2️⃣ All five linear maps (left, right, left‑gate, right‑gate, out‑gate) are
   fused into a single GEMM with weights scaled by the LN weight γ.
   The GEMM yields `proj_raw` of shape (R,5·H) (R = B·N·N).
3️⃣ A Triton kernel fuses:
      – subtraction of the row‑wise mean,
      – multiplication by the LN rsqrt‑var,
      – addition of the LN bias term β·Wᵀ,
      – optional input mask,
      – sigmoid for the three gates,
      – multiplication of left/right by their respective gates,
   and writes the three tensors `left`, `right` and `out_gate`
   (all FP16, layout [B, H, N, N]).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
      out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →  FP16.
5️⃣ Row‑wise mean & rsqrt‑variance of `out` are obtained with a second
   Triton kernel (shape B·N·N rows, reduction over H).
6️⃣ The second LayerNorm, the out‑gate multiplication and the final
   projection are fused into a single Triton kernel that writes the
   final [B, N, N, dim] FP32 tensor.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision
(FP16 compute, FP32 statistics) to minimise memory traffic and kernel
launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm statistics (no output tensor)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_norm_stats_kernel(
    x_ptr,          # [R, dim] FP32 (flattened input)
    mu_ptr,         # [R]    FP32
    rstd_ptr,       # [R]    FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 2️⃣ Split, apply LN statistics, mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_scaled_kernel(
    proj_ptr,          # [R, 5·H] FP16 (raw GEMM output)
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mu_ptr,            # [R] FP32
    rstd_ptr,          # [R] FP32
    sum_left_ptr, bias_left_ptr,
    sum_right_ptr, bias_right_ptr,
    sum_lgate_ptr, bias_lgate_ptr,
    sum_rgate_ptr, bias_rgate_ptr,
    sum_out_ptr, bias_out_ptr,
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N                                    # stride of a full [N,N] plane
    base_offset = b * (H * hidden_stride) + i * N + k       # position of (h=0,i,k)

    # Load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R] FP32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R] FP32

    # Optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val.to(tl.float32)[:, None]                     # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ------- load raw projections ---------------------------------
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]         # [BR,BH]

        left_raw      = tl.load(proj_ptr + proj_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_off + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        lgate_raw     = tl.load(proj_ptr + proj_off + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        rgate_raw     = tl.load(proj_ptr + proj_off + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        outgate_raw   = tl.load(proj_ptr + proj_off + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # ------- load summed‑weights & biases (per hidden) -------------
        sum_l   = tl.load(sum_left_ptr   + offs_h, mask=mask_h, other=0.0)
        bias_l  = tl.load(bias_left_ptr  + offs_h, mask=mask_h, other=0.0)
        sum_r   = tl.load(sum_right_ptr  + offs_h, mask=mask_h, other=0.0)
        bias_r  = tl.load(bias_right_ptr + offs_h, mask=mask_h, other=0.0)
        sum_lg  = tl.load(sum_lgate_ptr  + offs_h, mask=mask_h, other=0.0)
        bias_lg = tl.load(bias_lgate_ptr + offs_h, mask=mask_h, other=0.0)
        sum_rg  = tl.load(sum_rgate_ptr  + offs_h, mask=mask_h, other=0.0)
        bias_rg = tl.load(bias_rgate_ptr + offs_h, mask=mask_h, other=0.0)
        sum_o   = tl.load(sum_out_ptr    + offs_h, mask=mask_h, other=0.0)
        bias_o  = tl.load(bias_out_ptr   + offs_h, mask=mask_h, other=0.0)

        # ------- convert to FP32 for arithmetic -----------------------
        lf  = left_raw.to(tl.float32)
        rf  = right_raw.to(tl.float32)
        lgf = lgate_raw.to(tl.float32)
        rgf = rgate_raw.to(tl.float32)
        ogf = outgate_raw.to(tl.float32)

        # ------- apply LN (mean/var) and add bias --------------------
        left      = (lf - mu[:, None] * sum_l[None, :]) * rstd[:, None] + bias_l[None, :]
        right     = (rf - mu[:, None] * sum_r[None, :]) * rstd[:, None] + bias_r[None, :]
        left_gate = (lgf - mu[:, None] * sum_lg[None, :]) * rstd[:, None] + bias_lg[None, :]
        right_gate= (rgf - mu[:, None] * sum_rg[None, :]) * rstd[:, None] + bias_rg[None, :]
        out_gate  = (ogf - mu[:, None] * sum_o[None, :]) * rstd[:, None] + bias_o[None, :]

        # ------- mask (only left/right) -------------------------------
        left  = left  * mask_val
        right = right * mask_val

        # ------- sigmoid gates (stable FP32) -------------------------
        left_gate_sig  = (1.0 / (1.0 + tl.exp(-left_gate))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate))).to(tl.float16)
        out_gate_sig   = (1.0 / (1.0 + tl.exp(-out_gate))).to(tl.float16)

        # ------- apply gates and cast back to FP16 -------------------
        left_fp  = (left * left_gate_sig).to(tl.float16)
        right_fp = (right * right_gate_sig).to(tl.float16)

        # ------- write results ----------------------------------------
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BR,BH]

        tl.store(left_ptr      + dst_off, left_fp,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right_fp,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate_sig,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate after sigmoid)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R = B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [M, K]
        term = norm * vals_gate                                         # [M, K]

        term_fp16 = term.to(tl.float16)  # cast back to FP16 for dot

        # load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N) (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of (i,k) pairs per batch
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row LayerNorm statistics (mean & rsqrt‑var)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    rowwise_norm_stats_kernel[grid_norm](
        x_flat,
        mu,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (scaled by LN weight γ) into ONE GEMM
    # ------------------------------------------------------------------
    # gather the five weight matrices
    left_w    = w["left_proj.weight"]          # [H, dim]
    right_w   = w["right_proj.weight"]
    lgate_w   = w["left_gate.weight"]
    rgate_w   = w["right_gate.weight"]
    outgate_w = w["out_gate.weight"]

    fused_w = torch.cat(
        [left_w, right_w, lgate_w, rgate_w, outgate_w],
        dim=0,
    )  # shape (5·H, dim), FP32

    # LN weight γ and bias β
    ln_gamma = w["norm.weight"]   # [dim]
    ln_beta  = w["norm.bias"]    # [dim]

    # scale weights by γ  (γ is per‑dim, broadcast over rows)
    fused_w_scaled = fused_w * ln_gamma.to(torch.float32).unsqueeze(0)   # FP32
    fused_w_scaled_fp16 = fused_w_scaled.to(torch.float16)               # for GEMM

    # pre‑compute Σ_d (W·γ) and β·Wᵀ for each of the five groups
    sum_scaled_all = fused_w_scaled.sum(dim=1)      # [5·H]  FP32
    bias_all       = (ln_beta.to(torch.float32).unsqueeze(0) * fused_w).sum(dim=1)  # [5·H] FP32

    # split per‑group tensors
    split = [H] * 5
    sum_left,    sum_right,    sum_lgate,    sum_rgate,    sum_outgate = torch.split(sum_scaled_all, split)
    bias_left,   bias_right,   bias_lgate,   bias_rgate,   bias_outgate   = torch.split(bias_all, split)

    # ------------------------------------------------------------------
    # 2️⃣   GEMM:  x (FP16) × weight_scaledᵀ  →  proj_raw [R, 5·H] (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x.reshape(R, dim).contiguous().to(torch.float16)   # [R, dim]
    proj_raw = F.linear(x_fp16, fused_w_scaled_fp16)            # (R,5·H) FP16
    del x_fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply LN statistics, mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_scaled_kernel[grid_split](
        proj_raw,
        left,
        right,
        out_gate,
        mu,
        rstd,
        sum_left,   bias_left,
        sum_right,  bias_right,
        sum_lgate,  bias_lgate,
        sum_rgate,  bias_rgate,
        sum_outgate, bias_outgate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj_raw, mu, rstd, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (core einsum):   out = left @ rightᵀ  over dim k
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                      # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    total_rows_out = B * N * N          # rows for second LayerNorm
    mu_out   = torch.empty(total_rows_out, dtype=torch.float32, device=device)
    rstd_out = torch.empty(total_rows_out, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 for all test configs
    grid_stats = (triton.cdiv(total_rows_out, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu_out,
        rstd_out,
        total_rows_out,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate (already sigmoid‑ed) and final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over B·N·N)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(total_rows_out, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_out,
        rstd_out,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        total_rows_out,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
