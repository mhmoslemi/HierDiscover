@triton.jit
def final_norm_gate_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    out_norm_ptr,
    R, H,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # broadcast across hidden tiles
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load values
        # out and gate shapes: [R, H]
        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)  # convert to FP32 for stable ops
        gate_val = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # load mu, rstd per row (r)
        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)  # FP32
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)  # FP32

        # Expand mu and rstd to match hidden dimension
        mu_vec = mu[:, None]
        rstd_vec = rstd[:, None]

        # Load gamma and beta
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # Compute normalized out
        norm = ((out_val - mu_vec) * rstd_vec) * gamma[None, :] + beta[None, :]
        out_norm = norm * gate_val   # FP32

        # Store as FP16
        tl.store(out_norm_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_norm.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the last feature dimension (dim) – FP32→FP16.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into one
   GEMM (R × 5·H) and store the result in FP16.
3️⃣ Split the fused projection, apply the optional pair‑wise mask, compute
   sigmoid gates and write three tensors: left, right, out_gate
   (layout [B, H, N, N]).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   (still FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` across the hidden dimension H
   (Triton kernel → μ, rstd per (b,i,j) pair).
6️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   linear projection (`hidden → dim`) in two steps:
   • A Triton kernel normalises `out`, applies the layer‑norm parameters
     (γ, β) and multiplies by `out_gate`, writing a flat
     `[R, H]` tensor (`R = B·N·N`).
   • A single `torch.nn.functional.linear` (FP16 GEMM) projects this
     `[R, H]` tensor to `[R, dim]`.  The result is reshaped to
     `[B, N, N, dim]` and cast to FP32.

All kernels target Triton 3.3.1 and are tuned for an NVIDIA H100.
Mixed‑precision (FP16 bulk work, FP32 reductions) gives the best
throughput while preserving numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim]   FP32 input (flattened)
    y_ptr,            # [R, dim]   FP16 output
    w_ptr,            # [dim]      γ (FP32)
    b_ptr,            # [dim]      β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values -------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)               # [BLOCK_R]
        mask_val = mask_val[:, None]                     # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    stride_hidden = N * N                     # distance between consecutive hidden planes

    # base offset for the first hidden plane
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (stable in FP32, result cast back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [R, H]

        tl.store(left_ptr     + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # start of h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate & final projection (split into two steps)
# ----------------------------------------------------------------------
@triton.jit
def final_norm_gate_kernel(
    out_ptr,          # [R, H]   FP16  (out flattened)
    gate_ptr,         # [R, H]   FP16  (out_gate flattened)
    mu_ptr,           # [R]      FP32
    rstd_ptr,         # [R]      FP32
    gamma_ptr,        # [H]      FP32  (to_out_norm.weight)
    beta_ptr,         # [H]      FP32  (to_out_norm.bias)
    out_norm_ptr,     # [R, H]   FP16  (output of LN * gate)
    R, H,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # Load μ and rstd for the whole tile (once per row)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # FP32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # FP32
    mu   = mu[:, None]      # [BLOCK_R, 1]
    rstd = rstd[:, None]    # [BLOCK_R, 1]

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out and gate (FP16 → FP32 for math stability)
        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)        # [R, H_tile]
        gate_val = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)       # [R, H_tile]

        # Load LayerNorm parameters for this hidden tile
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # [H_tile] FP32
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # [H_tile] FP32

        # (out - μ) * rstd → γ· + β → * out_gate
        norm = ((out_val - mu) * rstd) * gamma[None, :] + beta[None, :]
        out_norm = norm * gate_val

        tl.store(out_norm_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_norm.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the ten weight tensors used in the reference module
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy buffer

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate (fused) → flat [R, H]
    # --------------------------------------------------------------
    # Bring `out` and `out_gate` to layout [R, H] (contiguous)
    out_flat      = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]

    out_flat = out_flat.view(R, H)          # [R, H] FP16
    out_gate_flat = out_gate_flat.view(R, H)  # [R, H] FP16

    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_norm = (triton.cdiv(R, BLOCK_R_OUT),)
    final_norm_gate_kernel[grid_norm](
        out_flat,
        out_gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        out_norm_gate,
        R,
        H,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection (hidden → dim) – use cuBLAS FP16 GEMM
    # --------------------------------------------------------------
    # weight: [dim, hidden] → keep FP16 for maximum throughput
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]

    final_fp16 = F.linear(out_norm_gate, w_out, bias=None)   # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
