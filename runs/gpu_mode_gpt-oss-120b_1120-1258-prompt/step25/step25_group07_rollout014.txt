import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 scalar
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    hidden_stride = N * N                      # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for hidden h = 0

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H_tile]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (computed in FP32 then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,

def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.
    Steps:
    1️⃣ Layer‑norm the input tensor (float32 → float16).
    2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
       into a single half‑precision GEMM.
    3️⃣ Triton kernel splits the fused projection, applies the optional pair‑wise mask,
       and computes the sigmoid gates, writing ``left``, ``right`` and ``out_gate``.
    4️⃣ Batched half‑precision GEMM (torch.bmm) computes the core einsum:
           out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]   (FP16)
    5️⃣ Layer‑norm over the hidden dimension using PyTorch reductions (mean / var),
       followed by the out‑gate multiplication.
    6️⃣ Final linear projection (hidden → dim) via a single torch linear call.
    All heavy compute is performed in FP16 (tensor cores) while reductions are
    carried out in FP32 for numerical stability.
    """
    # Unpack inputs
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    H = hidden_dim
    R = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    # Use the native (high‑performance) LayerNorm kernel
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # Cast to FP16 for the fused projection
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H × dim)  →  FP16
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    proj = torch.nn.functional.linear(x_norm_fp16, fused_w)  # [B, N, N, 5·H] FP16
    del x_norm_fp16, fused_w

    proj_flat = proj.view(R, 5 * H).contiguous()  # [R, 5·H]
    del proj

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj_flat, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over H + out‑gate (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)       # [B, H, N, N] FP32
    mu = out_fp32.mean(dim=1, keepdim=True)                   # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                             # [B,1,N,N]

    # Apply learned LN weight / bias (γ,β) – they are 1‑D tensors of length H
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)  # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)    # [1,H,1,1]

    out_norm = (out_fp32 - mu) * rstd * gamma + beta   # [B, H, N, N] FP32

    gate_fp32 = out_gate.to(torch.float32)
    out_norm_gate = out_norm * gate_fp32                # [B, H, N, N] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (H → dim)
    # ------------------------------------------------------------------
    # reshape to [B·N·N, H]
    x_for_out = out_norm_gate.permute(0, 2, 3, 1).contiguous().view(-1, H)  # [B·N·N, H]
    final = torch.nn.functional.linear(x_for_out, w["to_out.weight"])       # [B·N·N, dim]
    final = final.view(B, N, N, dim)                                       # [B, N, N, dim]

    return final

        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj_flat, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction: batched GEMM (left @ rightᵀ) – FP16
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension + out‑gate  (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                 # [B, H, N, N] FP32
    mu = out_fp32.mean(dim=1, keepdim=True)                     # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)    # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                               # [B,1,N,N]

    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1]

    out_norm = (out_fp32 - mu) * rstd * gamma + beta           # [B, H, N, N] FP32

    gate_fp32 = out_gate.to(torch.float32)                     # [B, H, N, N] FP32
    out_norm_gate = out_norm * gate_fp32                     # FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (hidden → dim)
    # ------------------------------------------------------------------
    # bring the hidden dim to the last axis for an efficient matmul
    x_final = out_norm_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    x_final = x_final.view(-1, H)                              # [B·N·N, H]

    final = torch.nn.functional.linear(x_final, w["to_out.weight"])   # [B·N·N, dim] FP32
    final = final.view(B, N, N, dim)                                # [B, N, N, dim]

    return final
```
