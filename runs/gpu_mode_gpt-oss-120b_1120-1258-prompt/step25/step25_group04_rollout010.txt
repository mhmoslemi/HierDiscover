            x_vals = tl.load(ptr_x, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            # normalize
            x_norm = (x_vals - mean[:, None]) * rstd[:, None]

            # Load weight for each group: left_proj
            # ptr_weight = w_ptr + (group_offset + offs_h[:, None]) * dim + offs_d[None, :]
            # group_offset = g * H

            # Compute for left_proj
            w_ptr_left = w_ptr + (offs_h[:, None] + 0 * H) * dim + offs_d[None, :]
            w_left = tl.load(w_ptr_left, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            left_acc += tl.dot(x_norm, w_left)  # shape (BLOCK_R, BLOCK_H)

            # right_proj
            w_ptr_right = w_ptr + (offs_h[:, None] + 1 * H) * dim + offs_d[None, :]
            w_right = tl.load(w_ptr_right, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            right_acc += tl.dot(x_norm, w_right)

            # left_gate
            w_ptr_lgate = w_ptr + (offs_h[:, None] + 2 * H) * dim + offs_d[None, :]
            w_lgate = tl.load(w_ptr_lgate,
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            left_gate_acc += tl.dot(x_norm, w_lgate)

            # right_gate
            w_ptr_rgate = w_ptr + (offs_h[:, None] + 3 * H) * dim + offs_d[None, :]
            w_rgate = tl.load(w_ptr_rgate,
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            right_gate_acc += tl.dot(x_norm, w_rgate)

            # out_gate
            w_ptr_ogate = w_ptr + (offs_h[:, None] + 4 * H) * dim + offs_d[None, :]
            w_ogate = tl.load(w_ptr_ogate,
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            out_gate_acc += tl.dot(x_norm, w_ogate)

        # end dim loop

        # after we have accumulated for this hidden tile, we now apply sigmoid for gate outputs, mask for left/right, and store to appropriate positions.

        # Sigmoid
        left_gate_sig = (1.0 / (1.0 + tl.exp(-left_gate_acc.to(tl.float32)))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate_acc.to(tl.float32)))).to(tl.float16)
        out_gate_sig = (1.0 / (1.0 + tl.exp(-out_gate_acc.to(tl.float32)))).to(tl.float16)

        # left and right: apply mask and gate and store
        # left = left_acc (FP32) to FP16 after gating: left_val = left_acc.to(tl.float16) * left_gate_sig * mask_val
        # mask_val shape: [BLOCK_R, 1] broadcast across hidden.

        mask_val_f16 = mask_val[:, None] if HAS_MASK else tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        left_store = left_acc.to(tl.float16) * left_gate_sig * mask_val_f16
        right_store = right_acc.to(tl.float16) * right_gate_sig * mask_val_f16

        # compute global offset for left and right and out_gate storage.

        hidden_stride = N * N
        # position base: b * (H * hidden_stride) + i * N + k  (for plane h=0)
        base_offset = b * (H * hidden_stride) + i * N + k
        # final offset for each hidden within tile: base_offset + h * hidden_stride

        # compute offsets:
        dst_offsets = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # store
        tl.store(left_ptr + dst_offsets, left_store,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offsets, right_store,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offsets, out_gate_sig,
                 mask=mask_r[:, None] & mask_h[None, :])

        # Need to zero accumulators for next tile (if any)
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

Thus full kernel:

```
@triton.jit
def fused_proj_gate_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    w_ptr,
    rows, dim, H, N, eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)  # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # First pass: compute per-row mean & var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)  # [BLOCK_R]

    # Now for each hidden tile
    hidden_stride = N * N
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load raw x
            ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
            xvals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            xnorm = (xvals - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_D]

            # loading weight slices
            # group offsets: each group occupies H rows
            # left proj: offset = 0*H
            w_left_ptr = w_ptr + (offs_h[:, None] + 0 * H) * dim + offs_d[None, :]
            w_left = tl.load(w_left_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            left_acc += tl.dot(xnorm, w_left, trans_b=True)

            # right proj
            w_right_ptr = w_ptr + (offs_h[:, None] + 1 * H) * dim + offs_d[None, :]
            w_right = tl.load(w_right_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            right_acc += tl.dot(xnorm, w_right, trans_b=True)

            # left_gate
            w_lgate_ptr = w_ptr + (offs_h[:, None] + 2 * H) * dim + offs_d[None, :]
            w_lgate = tl.load(w_lgate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            left_gate_acc += tl.dot(xnorm, w_lgate, trans_b=True)

            # right_gate
            w_rgate_ptr = w_ptr + (offs_h[:, None] + 3 * H) * dim + offs_d[None, :]
            w_rgate = tl.load(w_rgate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            right_gate_acc += tl.dot(xnorm, w_rgate, trans_b=True)

            # out_gate
            w_ogate_ptr = w_ptr + (offs_h[:, None] + 4 * H) * dim + offs_d[None, :]
            w_ogate = tl.load(w_ogate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            out_gate_acc += tl.dot(xnorm, w_ogate, trans_b=True)

        # after dim blocks, compute gates and apply mask
        # sigmoid in FP32 then cast to FP16
        lgate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # left/right results

"""
Outgoing TriMul (AlphaFold‑3) – fused‑kernel Triton implementation
-----------------------------------------------------------------
The forward pass consists of three main stages:

1️⃣  **Fused LayerNorm + 5‑way Linear projection + masking & gating**  
    - Input `x` ∈ [R, C] (R = B·N·N, C = dim) is row‑wise LayerNormed.
    - The normalized rows are simultaneously multiplied by the five weight
      matrices (`left_proj`, `right_proj`, `left_gate`, `right_gate`,
      `out_gate`) packaged as a single fp16 matrix `W` ∈ [5·H, C].
    - The three results are written to `left`, `right` (both multiplied by
      their sigmoid gates and the optional pair mask) and to `out_gate`
      (sigmoid only).

2️⃣  **Batched matrix multiply** (`left @ rightᵀ`) on the hidden dimension  
    `left`/`right` have shape [B, H, N, N] → view as [B·H, N, N] and
    multiplied with cuBLAS (fp16).

3️⃣  **Row‑wise stats + final LayerNorm + out_gate + final projection**  
    - Row‑wise mean/variance of the contracted tensor are computed.
    - The second LayerNorm, the multiplication by `out_gate` and the final
      linear projection are fused in one Triton kernel.

All kernels are written for Triton 3.3.1 and target an NVIDIA H‑100.
Mixed‑precision (fp16 bulk, fp32 reductions) is used throughout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fuse LN, projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,            # [R, C]   fp32 input
    left_ptr,         # [B·H·N·N] fp16 output (flattened)
    right_ptr,        # [B·H·N·N] fp16 output (flattened)
    out_gate_ptr,     # [B·H·N·N] fp16 output (flattened)
    mask_ptr,         # [R]       fp16 mask (optional)
    w_ptr,            # [5·H, C] fp16 fused weight matrix
    rows, C, H, N, eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # decode flat index r = b·N·N + i·N + k
    # --------------------------------------------------------------
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # --------------------------------------------------------------
    # optional per‑pair mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)   # [BLOCK_R]
        mask_val = mask_val[:, None]               # [BLOCK_R, 1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # first pass – compute mean & variance of each row (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < C

        x = tl.load(x_ptr + offs_r[:, None] * C + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                 # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)                               # [BLOCK_R]

    # --------------------------------------------------------------
    # second pass – compute the five projections (fp16 → fp32 accum)
    # --------------------------------------------------------------
    hidden_stride = NN                                       # stride between hidden planes

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for this hidden tile
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(C, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < C

            # raw input tile
            x_raw = tl.load(x_ptr + offs_r[:, None] * C + offs_d[None, :],
                            mask=mask_r[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)        # [BLOCK_R, BLOCK_D]

            # normalized tile
            x_norm = (x_raw - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_D]

            # ------- weight tiles (5 groups, each H×C) -------
            # left projection
            w_left = tl.load(w_ptr + (offs_h[:, None] + 0 * H) * C + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)       # [BLOCK_H, BLOCK_D]
            left_acc      += tl.dot(x_norm, w_left, trans_b=True)

            # right projection
            w_right = tl.load(w_ptr + (offs_h[:, None] + 1 * H) * C + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            right_acc     += tl.dot(x_norm, w_right, trans_b=True)

            # left gate
            w_lgate = tl.load(w_ptr + (offs_h[:, None] + 2 * H) * C + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            left_gate_acc += tl.dot(x_norm, w_lgate, trans_b=True)

            # right gate
            w_rgate = tl.load(w_ptr + (offs_h[:, None] + 3 * H) * C + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            right_gate_acc+= tl.dot(x_norm, w_rgate, trans_b=True)

            # out gate
            w_ogate = tl.load(w_ptr + (offs_h[:, None] + 4 * H) * C + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            out_gate_acc  += tl.dot(x_norm, w_ogate, trans_b=True)

        # ----------------------------------------------------------
        # post‑process: sigmoid gates, apply mask, write results
        # ----------------------------------------------------------
        lgate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        left_val  = left_acc .to(tl.float16) * lgate * mask_val   # [BLOCK_R, BLOCK_H]
        right_val = right_acc.to(tl.float16) * rgate * mask_val
        outg_val  = ogate                                          # no mask

        # destination offsets (flattened layout)
        base_off = b * (H * hidden_stride) + i * N + k            # offset for h = 0
        dst_off  = base_off[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_off, left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off, right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, outg_val,
                 mask=mask_r[:, None] & mask_h[None, :])

# 2️⃣ Row‑wise statistics of the contracted tensor (mean & rsqrt‑var)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,               # [B, H, N, N] fp16
    mu_ptr,                # [R] fp32
    rstd_ptr,              # [R] fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base = b * (H * hidden_stride) + i * N + j          # start of plane h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mu,   mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   fp16
    gate_ptr,         # [B, H, N, N]   fp16   (out_gate)
    mu_ptr,           # [R]            fp32
    rstd_ptr,         # [R]            fp32
    ln_w_ptr,         # [H]            fp32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            fp32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       fp16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)               # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)               # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base = b * (H * hidden_stride) + i * N + j               # offset for plane h=0

    # per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)      # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)      # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted values and gates (fp16 → fp32)
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [M, K]

        # second LayerNorm parameters (γ/β) for hidden channels
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]             # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]
        term = norm * gate_vals                                      # [M, K]

        term_fp16 = term.to(tl.float16)

        # tile of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result [B, N, N, dim] (fp32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

            weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                # total pair positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  Fused LN + 5‑way projection + mask + gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # flatten input to [R, dim] (fp32)
    x_flat = x.reshape(R, dim).contiguous()

    # pack the five weight matrices (each [H, dim]) → [5·H, dim] fp16
    fused_w = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"],
    ], dim=0).to(torch.float16).contiguous()

    # optional mask (flattened fp16)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # launch parameters
    BLOCK_R = 64          # rows per program (tunable)
    BLOCK_H = 32          # hidden‑dim tile (tunable)
    BLOCK_D = 64          # column‑tile of the weight matrix (tunable)

    grid_fused = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_fused](
        x_flat,
        left.view(-1),          # flatten for pointer arithmetic
        right.view(-1),
        out_gate.view(-1),
        mask_ptr,
        fused_w,
        R,
        dim,
        H,
        N,
        eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣  Batched GEMM  (left @ rightᵀ)   –   fp16 tensor‑core gemm
    # --------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)            # [B, H, N, N] fp16

    # --------------------------------------------------------------
    # 3️⃣  Row‑wise mean & rsqrt‑variance of ``out``
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 4️⃣  Fuse second LN, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (fp32)
        w["to_out_norm.bias"],     # β (fp32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
