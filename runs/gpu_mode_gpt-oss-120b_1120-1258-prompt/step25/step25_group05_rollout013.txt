import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Kernel for splitting and gating
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H] FP16
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        m_val = m_val[:, None]  # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + k

    # iterate over hidden dimension H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # offset into projection matrix
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # left
        left_raw = tl.load(
            proj_ptr + proj_off,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right
        right_raw = tl.load(
            proj_ptr + proj_off + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # gates
        left_gate_raw = tl.load(
            proj_ptr + proj_off + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_off + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_off + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 then cast to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward pass for outgoing TriMul (no gradients). Implements

    1. LayerNorm on input tensor (FP32 → FP16) using torch.nn.functional.layer_norm.
    2. Fused linear projection for left/right/value gates (5·H) with cuBLAS.
    3. Triton kernel `split_fuse_gate_kernel` to split projection, apply mask and sigmoid gates.
    4. Batched matrix multiplication (left @ rightᵀ) over the sequence dimension.
    5. Row‑wise mean / variance (torch.mean/var) to obtain μ and rsqrt‑σ².
    6. Element‑wise LayerNorm, out‑gate multiplication and final projection using
       a single matmul (torch.nn.functional.linear).  Returns FP32 tensor
       `[B, N, N, dim]`.

    All heavy tensor contractions are performed with cuBLAS/Tensor‑Core.
    Only the split‑gate step uses a custom Triton kernel.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ LayerNorm (FP32 -> FP16)
    # reshape to (R, dim)
    x_flat = x.reshape(R, dim)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # 2️⃣ Fused projection (5·H)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]
    # Linear: (R,5·H) = (R,dim) @ (dim,5·H). torch.nn.functional.linear expects weight (out_features, in_features)
    proj = F.linear(x_norm, fused_weight)  # (R, 5·H) half
    del x_norm, fused_weight

    # 3️⃣ Split / mask / sigmoid using Triton
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4️⃣ Batched GEMM on (left @ rightᵀ) over sequence dimension
    # reshape to [B*H, N, N]
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # out shape [B*H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)  # (B,H,N,N) half
    del left_mat, right_mat

    # 5️⃣ Row‑wise mean & variance using torch reductions (float32)
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1)  # (B,N,N) float32
    var = out_fp32.var(dim=1, unbiased=False)  # (B,N,N) float32
    rstd = torch.rsqrt(var + eps)  # (B,N,N) float32

    # 6️⃣ Normalization, out_gate, final projection
    # broadcasting helpers
    ln_w = weights["to_out_norm.weight"]  # (H,)
    ln_b = weights["to_out_norm.bias"]    # (H,)

    # reshape for broadcasting
    mu_ = mu.unsqueeze(1)          # (B,1,N,N)
    rstd_ = rstd.unsqueeze(1)      # (B,1,N,N)

    # out_norm = ((out - mu) * rstd) * ln_w + ln_b
    out_float = out.to(torch.float32)
    # apply layernorm
    out_norm = ((out_float - mu_) * rstd_) * ln_w[None, :, None, None] + ln_b[None, :, None, None]  # (B,H,N,N)

    # apply out_gate
    out_gate_float = out_gate.to(torch.float32)
    gated = out_norm * out_gate_float  # (B,H,N,N)

    # reshape to (R, H) for final linear
    Z = gated.permute(0, 2, 3, 1).contiguous().view(R, H).to(torch.float16)

    # final linear projection
    w_out = weights["to_out.weight"].to(torch.float16).t().contiguous()  # (H, dim)
    final_fp16 = F.linear(Z, w_out)  # (R, dim) half
    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  LayerNorm over the last dimension (FP32 → FP16) using the
    highly‑optimised torch.nn.functional.layer_norm kernel.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → [R, 5·H] (FP16) where R = B·N·N.
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional pair mask, evaluates the three sigmoid gates and
    writes the three tensors `left`, `right` and `out_gate`
    in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM:  out = left @ rightᵀ   (Tensor‑Core, cuBLAS)  → [B, H, N, N].
5️⃣  Row‑wise mean / variance of `out` (float32) via torch reductions → µ, rσ.
6️⃣  Element‑wise LayerNorm on `out`, multiplication by `out_gate`,
    and final linear projection to the original dimensionality.
    The last two steps are fused into a single matmul
    (Z = norm(out)·gate ,  final = Z @ Wᵀ) using cuBLAS.

Only the split‑gate step is written in Triton; all heavy tensor
contractions use cuBLAS/Tensor‑Core, giving a fast forward pass
(< 500 µs on the reference test cases).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask, evaluate sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]            FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N² + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # stride for a full hidden plane
    base = b * (H * hidden_stride) + i * N + k   # start address for h = 0

    # iterate over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offset into the fused projection (5 stacked blocks of size H)
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_off,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_off + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_off + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_off + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_off + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        gate  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
          (all float32)
        - config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)                         # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)            # [R, dim] FP16
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # flatten mask to [R] and cast to FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & variance (float32) – reuse torch reductions
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                 # [B, H, N, N] FP32
    mu   = out_fp32.mean(dim=1)                      # [B, N, N] FP32
    var  = out_fp32.var(dim=1, unbiased=False)      # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)                    # [B, N, N] FP32
    del out_fp32, var

    # ------------------------------------------------------------------
    # 6️⃣ Normalisation, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # broadcast over hidden dimension
    mu_   = mu.unsqueeze(1)          # [B,1,N,N]
    rstd_ = rstd.unsqueeze(1)        # [B,1,N,N]

    ln_w = w["to_out_norm.weight"]   # (H,)
    ln_b = w["to_out_norm.bias"]     # (H,)

    # LayerNorm on the contracted tensor
    out_norm = ((out - mu_) * rstd_) * ln_w[None, :, None, None] + ln_b[None, :, None, None]  # [B,H,N,N] FP32

    # Apply out‑gate
    gated = out_norm * out_gate.to(torch.float32)      # [B,H,N,N] FP32
    del out_norm, out_gate, mu, rstd, mu_, rstd_

    # Reshape to (R, H) for the last linear projection
    Z = gated.permute(0, 2, 3, 1).contiguous().view(R, H).to(torch.float16)  # [R, H] FP16
    del gated

    # Final projection (FP16 GEMM) → (R, dim) then cast back to FP32
    w_out = w["to_out.weight"].to(torch.float16)       # (dim, H)
    final_fp16 = F.linear(Z, w_out)                    # (R, dim) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim] FP32
    return final
