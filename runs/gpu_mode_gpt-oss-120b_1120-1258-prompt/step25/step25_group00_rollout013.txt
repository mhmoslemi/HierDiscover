@triton.jit
def fused_stats_norm_proj_kernel(
    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    MAX_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # allocate shared memory for out and gate values
    shmem_out = tl.shared_memory(BLOCK_M * MAX_H, dtype=tl.float16)
    shmem_gate = tl.shared_memory(BLOCK_M * MAX_H, dtype=tl.float16)

    # accumulate sums
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass: load and store in shared mem, compute sum & sumsq
    for k_start in range(0, H, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, H - k_start)
        offs_k = k_start + tl.arange(0, BLOCK_K)  # [BLOCK_K]
        mask_k = offs_k < H

        # pointer offsets
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_block = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16
        gate_block = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # sum for stats
        out_fp32 = out_block.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq += tl.sum(out_fp32 * out_fp32, axis=1)

        # store in shared memory
        # compute index: row*MAX_H + (k_start + offset)
        row_off = tl.arange(0, BLOCK_M)[:, None] * MAX_H
        col_off = (k_start + tl.arange(0, BLOCK_K))[None, :]  # shape [1, BLOCK_K]
        # broadcast to shape [BLOCK_M, BLOCK_K]
        smem_idx = row_off + col_off
        tl.store(shmem_out + smem_idx, out_block, mask=mask_m[:, None] & mask_k[None, :])
        tl.store(shmem_gate + smem_idx, gate_block, mask=mask_m[:, None] & mask_k[None, :])

    # compute mean and rstd
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # prepare output accumulation
    offs_n = tl.arange(0, BLOCK_N)  # [BLOCK_N]
    mask_n = offs_n < dim

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # second pass: load from shared memory, apply LN, gate, dot with weight
    for k_start in range(0, H, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, H - k_start)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load from shared mem
        row_off = tl.arange(0, BLOCK_M)[:, None] * MAX_H
        col_off = (k_start + tl.arange(0, BLOCK_K))[None, :]
        smem_idx = row_off + col_off

        out_block = tl.load(shmem_out + smem_idx, mask=mask_m[:, None] & mask_k[None, :])
        gate_block = tl.load(shmem_gate + smem_idx, mask=mask_m[:, None] & mask_k[None, :])

        # layernorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # normalized out
        out_fp32 = out_block.to(tl.float32)
        gate_fp32 = gate_block.to(tl.float32)

        norm = ((out_fp32 - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_fp32  # [BLOCK_M, BLOCK_K]

        # Convert to fp16 for dot
        term_fp16 = term.to(tl.float16)

        # weight block of shape [BLOCK_K, BLOCK_N]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final out
    # compute destination offsets: final shape [B, N, N, dim]
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold3) – fused forward pass.

Algorithmic improvements over the reference implementation
---------------------------------------------------------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) – unchanged.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM.
3️⃣  Split the fused projection, apply the optional mask and the sigmoid gates.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
5️⃣  **Fused** row‑wise statistics, second LayerNorm, out_gate multiplication
    and the final projection in a single Triton kernel.
    – this removes the separate row‑wise‐stats kernel and halves the global‐
      memory traffic for the huge tensor `out` (≈0.5 GB for N=1024, H=128).
6️⃣  All kernels run in mixed‑precision (FP16 for the bulk of the work,
    FP32 for the reduction) and use tuned block sizes for H100.

The public entry point is ``custom_kernel`` taking a tuple
(input, mask, weight_dict, config_dict) and returning the
float‑32 result tensor [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  LayerNorm – row‑wise over the hidden dimension (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,     # [R, dim] FP32 input
    y_ptr,     # [R, dim] FP16 output
    w_ptr,     # [dim]      LN weight (γ) FP32
    b_ptr,     # [dim]      LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values --------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]              # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣  Split the fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]      FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)                     # [BLOCK_R]
        m_val = m_val[:, None]                         # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N       # stride to jump from one hidden slice to the next

    # base offset for (b,i,k) at hidden‑index 0
    base = b * (H * hidden_stride) + i * N + k      # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 (better accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination offset for the three tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣  Fuse row‑wise stats, second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_stats_norm_proj_kernel(
    out_ptr,        # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,       # [H]          FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H]          FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]     FP16   (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32 (final result)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (≈ 32‑64)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile (32 is a good fit for H=128)
    BLOCK_N: tl.constexpr,   # output‑dim tile (64‑128)
    MAX_H: tl.constexpr,     # compile‑time hidden‑dim ceiling (≤128)
):
    pid = tl.program_id(0)               # tile over rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # decode (b, i, j) from the flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    # ------------------------------------------------------------------
    # Shared memory to keep the whole hidden vector of each row (out & gate)
    # ------------------------------------------------------------------
    # layout: [BLOCK_M, MAX_H] flattened → 1‑D pointer arithmetic
    shmem_out  = tl.shared_memory(BLOCK_M * MAX_H, dtype=tl.float16)
    shmem_gate = tl.shared_memory(BLOCK_M * MAX_H, dtype=tl.float16)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ---- first pass: load, store to SMEM and accumulate sums ---------------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # global loads
        ptr_o = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_g = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_blk  = tl.load(ptr_o,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0)                     # FP16
        gate_blk = tl.load(ptr_g,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0)                     # FP16

        # stats in FP32
        out_fp32 = out_blk.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)        # [BLOCK_M]

        # store the raw values to shared memory for the second pass
        row_off   = tl.arange(0, BLOCK_M)[:, None] * MAX_H           # [BLOCK_M,1]
        col_off   = (k_start + tl.arange(0, BLOCK_K))[None, :]      # [1, BLOCK_K]
        smem_idx  = row_off + col_off                               # [BLOCK_M, BLOCK_K]

        tl.store(shmem_out  + smem_idx,
                 out_blk,
                 mask=mask_m[:, None] & mask_k[None, :])
        tl.store(shmem_gate + smem_idx,
                 gate_blk,
                 mask=mask_m[:, None] & mask_k[None, :])

    # ---- compute mean / rstd -------------------------------------------------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: LN + out_gate + projection ---------------------------
    offs_n = tl.arange(0, BLOCK_N)                        # [BLOCK_N]
    mask_n = offs_n < dim

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # accumulator for final output

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load from shared memory (no extra global traffic)
        row_off  = tl.arange(0, BLOCK_M)[:, None] * MAX_H
        col_off  = (k_start + tl.arange(0, BLOCK_K))[None, :]
        smem_idx = row_off + col_off

        out_blk  = tl.load(shmem_out  + smem_idx,
                           mask=mask_m[:, None] & mask_k[None, :])
        gate_blk = tl.load(shmem_gate + smem_idx,
                           mask=mask_m[:, None] & mask_k[None, :])

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32

        # LN : ((out - μ) * rstd) * γ + β
        out_fp   = out_blk.to(tl.float32)
        gate_fp  = gate_blk.to(tl.float32)

        norm = ((out_fp - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_fp                                     # [BLOCK_M, BLOCK_K]

        # dot‑product with the final projection weight (FP16·FP16 → FP32)
        term_fp16 = term.to(tl.float16)

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # FP16

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [BLOCK_M, BLOCK_N]

    # ---- write final tensor -------------------------------------------------
    dst = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor_or_None, weight_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weight_dict  : mapping with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"                (all float32)
        - config_dict must contain
              "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim       = cfg["dim"]
    hidden    = cfg["hidden_dim"]          # always 128 in the supplied tests
    nomask    = cfg.get("nomask", True)
    eps       = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise positions
    H = hidden

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                  # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split → left, right, out_gate (mask & sigmoid)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.reshape(R).contiguous().to(torch.float16)   # [R]
        mask_ptr  = mask_flat
        has_mask  = 1
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Fused row‑wise stats + LayerNorm + out_gate + final proj
    # --------------------------------------------------------------
    # transposed final projection weight (H × dim) in FP16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 32          # rows per program (tuned for H100)
    BLOCK_K_OUT = 32          # hidden‑dim tile
    BLOCK_N_OUT = 128         # output‑dim tile
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_stats_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        MAX_H=hidden,               # hidden ≤ 128 → compile‑time const
        num_warps=8,
    )
    # clean up intermediate tensors (optional)
    del out, out_gate

    return final
```
