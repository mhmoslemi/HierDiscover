@triton.jit
def rowwise_stats_kernel(
    out_ptr, # [R, H] FP16
    mu_ptr, # [R] FP32
    rstd_ptr, # [R] FP32
    rows, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < rows

   # each row loads H values in a loop with blocks
   sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
   sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

   for h_start in range(0, H, BLOCK_H):
       offs_h = h_start + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H

       ptr = out_ptr + offs_m[:, None] * H + offs_h[None, :]
       vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
       sum_val += tl.sum(vals, axis=1)
       sum_sq += tl.sum(vals * vals, axis=1)

   mean = sum_val / H
   var = sum_sq / H - mean * mean
   rstd = tl.rsqrt(var + eps)

   tl.store(mu_ptr + offs_m, mean, mask=mask_m)
   tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

def norm_gate_kernel(

    gate_ptr,  # [R, H] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    gamma_ptr, # [H] FP32 to_out_norm.weight
    beta_ptr,  # [H] FP32 to_out_norm.bias
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows?? We need to pass total_rows (R). Use "rows" param.

    # compute per-row mu and rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0) # [M]

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        # Normalize: (out - mu) * rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]
        # Apply to_out_norm affine: * gamma + beta
        norm = diff * gamma[None, :] + beta[None, :]
        # Multiply by gate
        result = norm * gate_vals

        # store back as FP16
        tl.store(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 result.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

@triton.jit
def rowwise_stats_kernel(out_ptr, mu_ptr, rstd_ptr,
                         rows, H, eps,
                         BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

@triton.jit
def norm_gate_kernel(
    out_ptr, # [R, H] FP16 (in-place)
    gate_ptr, # [R, H] FP16
    mu_ptr, # [R] FP32
    rstd_ptr, # [R] FP32
    gamma_ptr, # [H] FP32
    beta_ptr, # [H] FP32
    rows, H,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        # Normalize out
        diff = (out_vals - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]
        out_res = norm * gate_vals

        tl.store(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 out_res.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

"""
Optimized outgoing TriMul (AlphaFold3) forward‑only kernel.

Algorithm
---------
1. Row‑wise LayerNorm (FP32 → FP16) on the flattened [B·N·N, C] tensor.
2. Fused linear projection producing the five tensors (left, right,
   left_gate, right_gate, out_gate) in a single matmul.
3. Split & gate kernel:
   * applies pair‑wise mask,
   * computes sigmoid gates,
   * writes left/right/out_gate in layout [B, H, N, N] (FP16).
4. Batched GEMM (cuBLAS) computes the core contraction
   `out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]` → FP16.
5. Permute `out` and `out_gate` to [B, N, N, H] → flatten to (R, H) where
   R = B·N·N.  Hidden dimension becomes the innermost, giving fully
   coalesced loads.
6. Row‑wise statistics kernel computes μ and rstd over the hidden axis.
7. Normalisation‑gate kernel fuses:
   * LayerNorm (γ,β) on `out`,
   * multiplication by `out_gate`,
   * writes the result back as v ∈ FP16.
8. Final linear projection (FP16 → FP32) performed with a single
   cuBLAS matmul:  v (R×H) × Wᵀ (H×dim) → (R×dim) (float32).

All heavy work is in mixed‑precision (FP16) with reductions in FP32.
The implementation works for any batch size {1,2}, sequence length
{128,…,1024}, and hidden dimensions up to 128 (the test‑suite limit).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C]   FP32 input
    y_ptr,          # [R, C]   FP16 output
    w_ptr,          # [C]      γ (FP32)
    b_ptr,          # [C]      β (FP32)
    rows, cols, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < cols

        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < cols

        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * cols + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split & gate kernel (produces left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 scalar
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    hidden_stride = nn                     # distance between successive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for hidden h = 0

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H_tile]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                      # no mask for out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [R, H] FP16
    mu_ptr,       # [R]   FP32
    rstd_ptr,     # [R]   FP32
    rows, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Normalise, apply to_out_norm and out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,      # [R, H] FP16   (in‑place, will contain v = LN(out) * out_gate)
    gate_ptr,     # [R, H] FP16   (out_gate)
    mu_ptr,       # [R]    FP32
    rstd_ptr,     # [R]    FP32
    gamma_ptr,    # [H]    FP32   (to_out_norm.weight)
    beta_ptr,     # [H]    FP32   (to_out_norm.bias)
    rows, H,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate (FP16 → FP32)
        out_vals  = tl.load(out_ptr  + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

        # layer‑norm over hidden dim
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # (out - μ) / √var
        norm = diff * gamma[None, :] + beta[None, :]                  # apply γ,β

        # multiply by out_gate
        result = norm * gate_vals

        tl.store(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 result.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)           # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask & sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for cuBLAS bmm
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N]
    right_mat = right.view(B * H, N, N)              # [B·H, N, N]

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Permute to make hidden dimension innermost (coalesced)
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    out_flat = out.view(R, H)          # [R, H]  FP16
    gate_flat = out_gate.view(R, H)    # [R, H]  FP16

    # --------------------------------------------------------------
    # 6️⃣ Row‑wise mean & rsqrt‑variance over H
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_H_STATS = 128                # H ≤ 128 in all tests
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_H=BLOCK_H_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣ Fuse second LayerNorm, out_gate multiplication (in‑place)
    # --------------------------------------------------------------
    BLOCK_M_NORM = 256
    BLOCK_H_NORM = 128
    grid_norm = (triton.cdiv(R, BLOCK_M_NORM),)
    norm_gate_kernel[grid_norm](
        out_flat,               # will be overwritten with v
        gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_NORM,
        BLOCK_H=BLOCK_H_NORM,
        num_warps=8,
    )
    del gate_flat, mu, rstd

    # --------------------------------------------------------------
    # 8️⃣ Final projection: v (R×H) · Wᵀ (H×dim) → (R×dim)  (float32)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]
    final = torch.matmul(out_flat.float(), w_out_T.float())           # FP32, shape [R, dim]
    final = final.view(B, N, N, dim)                                 # [B, N, N, dim]

    return final

This implementation follows the reference TriMul logic, moves the hidden dimension to the innermost stride for fully‑coalesced row‑wise reductions, fuses the second LayerNorm with the out‑gate multiplication, and relies on cuBLAS for the two large GEMMs (the fused projection and the final projection).  
The kernels are written for Triton 3.3.1 and target mixed‑precision execution on an NVIDIA H100.
