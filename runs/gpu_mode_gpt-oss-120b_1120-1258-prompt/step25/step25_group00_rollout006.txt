"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Torch’s highly‑optimized
    `nn.functional.layer_norm`. The tensor is flattened to (R, dim) where
    R = B·N·N.

2️⃣ A *single* fused linear projection maps the normalized pair representation
    to 5·H dimensions (left, right, left‑gate, right‑gate, out‑gate).

3️⃣ A tiny Triton kernel (`split_fuse_gate_kernel`) reshapes the fused
    projection, applies the optional mask, computes the three sigmoid gates
    and stores the three tensors **left**, **right** and **out_gate** in the
    layout `[B, H, N, N]` (FP16).  This is the only required Triton kernel.

4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum
    `out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]`
    producing `[B, H, N, N]` in FP16.

5️⃣ Per‑position mean/variance over the hidden dimension H are obtained
    with Torch reductions (FP32).

6️⃣ The second LayerNorm, multiplication by `out_gate` and the final
    projection to the output dimension are performed with pure Torch
    tensor operations:
        - normalize with the previously computed mean / rstd,
        - apply `to_out_norm` (γ/β) and `out_gate`,
        - flatten to `[R, H]` and a single GEMM with `to_out.weightᵀ`
          (FP32) yields the final `[B, N, N, dim]` tensor.

All heavy linear algebra uses cuBLAS, while the custom Triton kernel
eliminates the expensive Python‑level for‑loops required to split the
fused projection.  The implementation works for both masked and
un‑masked inputs and supports the full range of test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused proj → left / right / out_gate + masking & sigmoid
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # -------- decode flat index (b,i,k) ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ---- optional mask (broadcasted over the hidden dimension) -------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between consecutive hidden planes
    # base offset for the first hidden plane (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k                # [BLOCK_R]

    # --------------- iterate over hidden‑dim tiles -----------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the five fused slices (contiguous in memory) ---------
        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ---- sigmoid gates (FP16, in‑place) ----------------------------
        left_gate  = tl.sigmoid(left_gate_raw)
        right_gate = tl.sigmoid(right_gate_raw)
        out_gate   = tl.sigmoid(out_gate_raw)

        # ---- apply mask & gates ---------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # ---- destination offsets (flattened view) ----------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,      left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,     right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,  outg,      mask=mask_r[:, None] & mask_h[None, :])

            "to_out.weight"                      (all float32)
        - config_dict : must contain
            "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------- unpack -------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of (b,i,k) token pairs
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (torch implementation – fast)
    # -----------------------------------------------------------------
    # Flatten to (R, dim) and apply fp32 LayerNorm, then cast to fp16
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                       # FP32
    x_norm = x_norm_fp32.to(torch.float16)                  # [R, dim] FP16

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection to 5·H channels (FP16 GEMM)
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]
    proj = F.linear(x_norm, fused_weight)   # [R, 5·H] FP16
    del x_norm, fused_weight

    # -----------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates (Triton)
    # -----------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128          # rows per program (tune for occupancy)
    BLOCK_H = 32           # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core O(N³) operation
    # -----------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # -----------------------------------------------------------------
    # 5️⃣ Per‑position mean / variance over the hidden dim (torch)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)            # [B, H, N, N] FP32
    mu   = out_fp32.mean(dim=1)                 # [B, N, N] FP32
    var  = out_fp32.var(dim=1, unbiased=False) # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)               # [B, N, N] FP32

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (torch)
    # -----------------------------------------------------------------
    # LayerNorm over hidden dimension using the learned γ/β
    gamma = w["to_out_norm.weight"][None, :, None, None]   # [1, H, 1, 1]
    beta  = w["to_out_norm.bias"  ][None, :, None, None]   # [1, H, 1, 1]

    # Normalisation
    out_norm = (out_fp32 - mu[:, None, :, :]) * rstd[:, None, :, :]   # [B, H, N, N]
    out_norm = out_norm * gamma + beta                               # γ/β
    out_norm = out_norm * out_gate.to(torch.float32)                  # gating

    # Final linear projection: flatten and GEMM with [H, dim] weight
    w_out_T = w["to_out.weight"].to(torch.float32).t().contiguous()   # [H, dim]
    # reshape to (R, H)
    out_flat = out_norm.permute(0, 2, 3, 1).reshape(R, H)             # [R, H]
    final_fp32 = out_flat @ w_out_T                                    # [R, dim]
    final = final_fp32.view(B, N, N, dim)                               # [B, N, N, dim]

    return final
