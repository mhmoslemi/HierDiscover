@triton.jit
def term_kernel(out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr, term_ptr, R, H, N,
               BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode (b,i,j)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N

    base_offset_out = b * (H * hidden_stride) + i * N + j
    base_offset_gate = base_offset_out

    # load mu,rstd per row
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        out_vals = tl.load(out_ptr + base_offset_out[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base_offset_gate[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # LN weight and bias
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)    # FP32

        term = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = term * gate_vals

        # store as fp16
        tl.store(term_ptr + base_offset_out[:, None] + offs_h[None,:] * hidden_stride,
                term.to(tl.float16),
                mask=mask_r[:, None] & mask_h[None, :])
    # end

@triton.jit
def layernorm_center_kernel(
    x_ptr,            # [R, dim] FP32
    y_ptr,            # [R, dim] FP16
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean and rstd
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized (x - mean) * rstd
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 norm.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    gate_ptr,       # [B, H, N, N] FP16
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32
    ln_b_ptr,       # [H] FP32
    term_ptr,       # [R, H] FP16 (output)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):

    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

            mask=mask_r[:, None] & mask_h[None, :],

            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,

        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        tl.store(term_ptr + term_offset, term.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑only forward pass.

Algorithm
---------
1️⃣  Row‑wise “center‑only” LayerNorm (mean / rsqrt‑var) on the pairwise
    tensor `x` →  `x_center = (x-μ)·rstd`  (FP16).  No γ/β are applied here.
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single GEMM:
        (R, dim) @ (dim, 5·H).T  →  (R, 5·H)  (FP16)
    where `R = B·N·N`.  The LayerNorm γ is baked into the weight matrix,
    and the LayerNorm β is baked into the bias vector.
3️⃣  Split the fused projection, apply the optional mask, compute the sigmoid
    gates and store three tensors `left`, `right` and `out_gate`
    (shape `[B, H, N, N]`, FP16).
4️⃣  Batched matrix‑multiply (cuBLAS) contracts over the sequence dimension:
        out[b, h] = left[b, h] @ right[b, h].T   →   `[B, H, N, N]` (FP16)
5️⃣  Row‑wise statistics (mean & rsqrt‑var) of `out` over the hidden dimension
    `H`.  These are needed for the second LayerNorm.
6️⃣  Apply the second LayerNorm, multiply by `out_gate`, and write the
    intermediate result into a compact buffer `term` of shape `[R, H]`.
7️⃣  Final linear projection (cuBLAS):  `final = term @ w_out.T`
    → `[R, dim]` (FP16) → cast to FP32 and reshape to `[B, N, N, dim]`.

All heavy reductions and element‑wise fusions are written in Triton;
the two large GEMMs use cuBLAS (torch.nn.functional.linear / torch.bmm /
torch.matmul) which provides optimal tensor‑core throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise mean / rsqrt‑var + (x‑mean)*rstd   (no γ / β)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_center_kernel(
    x_ptr,                # [R, dim]  FP32 input
    y_ptr,                # [R, dim]  FP16 output (center‑norm)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass – mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass – write (x‑mean)*rstd ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            norm.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16

    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                    # stride between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of hidden = 0

    # iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # out_gate is not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Apply second LayerNorm, multiply by `out_gate`, write compact term
# ----------------------------------------------------------------------
@triton.jit
def compute_term_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    term_ptr,       # [R, H] FP16   (compact buffer)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load contracted tensor and gate
        out_vals = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        gate_vals = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm parameters for the hidden channel
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_vals                                     # FP32

        # write compact term   →   term[r, h]  (row‑major)
        term_offset = offs_r[:, None] * H + offs_h[None, :]
        tl.store(term_ptr + term_offset, term.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3).

    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config       : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pair positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise centre‑only LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] float32
    x_center = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_center_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_center,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse LN γ into the linear weights and add the LN β contribution
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                          # [5·H, dim]

    # γ from the first LayerNorm
    norm_weight_fp16 = w["norm.weight"].to(torch.float16)  # [dim]
    scaled_fused_weight = fused_weight * norm_weight_fp16  # broadcast over dim → [5·H, dim]

    # β contribution (bias for the fused linear)
    bias_proj_fp32 = torch.matmul(w["norm.bias"], fused_weight.to(torch.float32).t())  # [5·H]
    bias_proj_fp16 = bias_proj_fp32.to(torch.float16)

    # fused projection: (R, dim) @ (dim, 5·H).T
    proj = F.linear(x_center, scaled_fused_weight, bias_proj_fp16)   # [R, 5·H] FP16
    del x_center, fused_weight, scaled_fused_weight, bias_proj_fp16

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply second LayerNorm, multiply by out_gate, write compact term
    # --------------------------------------------------------------
    term = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_TERM = 128
    BLOCK_H_TERM = 64
    compute_term_kernel[(triton.cdiv(R, BLOCK_R_TERM),)](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        term,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final projection (cuBLAS) → [B, N, N, dim] (float32)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final_fp16 = torch.matmul(term, w_out_T)          # [R, dim] FP16
    final = final_fp16.to(torch.float32).reshape(B, N, N, dim)   # [B, N, N, dim] FP32

    return final
