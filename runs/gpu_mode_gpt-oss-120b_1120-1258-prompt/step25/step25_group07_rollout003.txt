"""
Outgoing TriMul (AlphaFold3) – forward only.
-------------------------------------------------
Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim) is performed with the
    highly‑optimized PyTorch kernel (torch.nn.functional.layer_norm).

2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    are fused into a single GEMM (`torch.nn.functional.linear`) with a
    weight matrix of shape (5·hidden_dim, dim).  The result is reshaped to
    (R,5,hidden_dim) where R = B·N·N.

3️⃣  The mask (if any) and the three sigmoid gates are applied with
    vectorised PyTorch ops.  The three tensors `left`, `right`,
    `out_gate` are reshaped to (B,hidden_dim,N,N) (FP16).

4️⃣  Batched matrix‑multiplication (cuBLAS) computes the core contraction:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    producing `out` of shape (B,hidden_dim,N,N) in FP16.

5️⃣  Per‑position (b,i,j) mean and inverse‑std over the hidden dimension are
    obtained with a fast FP32 reduction (torch).

6️⃣  The second LayerNorm (over hidden_dim), multiplication with `out_gate`,
    and the final linear projection to `dim` are fused in a single Triton
    kernel (`final_output_kernel`).  The kernel reads `out` and `out_gate`,
    normalises using the pre‑computed statistics, applies the LN scale/bias,
    multiplies by the gate, and accumulates a matrix‑multiply with the
    transposed `to_out.weight`.  The final tensor has shape
    (B,N,N,dim) in FP32.

Only the final kernel is written in Triton – the rest relies on
well‑optimised PyTorch kernels, guaranteeing correctness while keeping
launch overhead minimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Final fused kernel: LN (hidden_dim) + out_gate + final proj.
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16 (contracted tensor)
    gate_ptr,         # [B, H, N, N]   FP16 (out_gate)
    mu_ptr,           # [R]            FP32  (mean over H)
    rstd_ptr,         # [R]            FP32  (rsqrt(var+eps))
    ln_w_ptr,         # [H]            FP32  (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32  (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16  (to_out.weight.T)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    """
    Row‑wise computation:
        out_norm = (out - μ) * rstd      # (FP32)
        out_norm = out_norm * ln_w + ln_b
        out_norm = out_norm * out_gate
        out_final[b,i,j,:] = Σ_h out_norm[h] · W_out_T[h,:]
    """
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [Nout]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b  = offs_m // nn
    rem = offs_m - b * nn
    i  = rem // N
    j  = rem - i * N

    hidden_stride = N * N                # distance between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics (μ, rstd) – one value per position
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    # accumulator for the output block (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over hidden dimension in tiles
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ----- load tensors -------------------------------------------------
        # out   : [M, K]   FP16 → FP32
        # gate  : [M, K]   FP16 → FP32
        out_vals = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # LN parameters for this tile (γ, β) – shape [K]
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # ----- LayerNorm + out_gate -----------------------------------------
        # (out - μ) * rstd  ->  normalized
        diff = (out_vals - mu[:, None]) * rstd[:, None]       # FP32
        norm = diff * gamma[None, :] + beta[None, :]          # FP32
        term = norm * gate_vals                               # FP32

        # ---- dot with final weight -----------------------------------------
        term_fp16 = term.to(tl.float16)                       # back to FP16 for WMMA
        # weight tile [K, Nout]
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )                                                     # FP16

        # accumulate (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ----- store final result ------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(
        dst_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim) – use PyTorch kernel
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim]  FP32
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right and three gates
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # (5*H, dim)

    # (R, 5*H)  FP16
    proj = F.linear(x_norm_fp16, fused_w)

    # reshape to (R,5,H)
    proj = proj.view(R, 5, H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and sigmoid gates (still in PyTorch)
    # ------------------------------------------------------------------
    left      = proj[:, 0, :]                     # (R, H)
    right     = proj[:, 1, :]                     # (R, H)
    left_gate = torch.sigmoid(proj[:, 2, :].float()).to(torch.float16)
    right_gate = torch.sigmoid(proj[:, 3, :].float()).to(torch.float16)
    out_gate  = torch.sigmoid(proj[:, 4, :].float()).to(torch.float16)

    if not nomask and mask is not None:
        # mask shape (B,N,N) -> flatten to (R,)
        mask_flat = mask.to(torch.float16).reshape(R)
        left  = left * mask_flat[:, None]
        right = right * mask_flat[:, None]

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # reshape to (B, H, N, N) – use a permutation so that hidden dim is stride‑contiguous
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ  (core TriMul contraction)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)           # (B*H, N, N)
    right_mat = right.view(B * H, N, N)          # (B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)  FP16
    out = out.view(B, H, N, N)                               # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Compute per‑position mean & rsqrt‑variance over hidden dimension (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    mu   = out_fp32.mean(dim=1)                                 # (B, N, N) FP32
    rstd = (out_fp32.var(dim=1, unbiased=False) + eps).rsqrt()  # (B, N, N) FP32

    mu_flat   = mu.reshape(R).contiguous()
    rstd_flat = rstd.reshape(R).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Final fused LN + out_gate + linear projection (Triton)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Tuning parameters – chosen to give good occupancy on H100
    BLOCK_M = 128          # rows per program (tiles over R)
    BLOCK_K = 32           # hidden‑dim tile
    BLOCK_N = 128          # output‑dim tile

    grid = (
        triton.cdiv(R, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )

    final_output_kernel[grid](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )

    return final
