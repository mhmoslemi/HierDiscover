"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim) and
produces a tensor of the same shape.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (row‑wise) → FP16.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16) of shape (R, 5·H) where  R = B·N·N  and
    H = hidden_dim.
3️⃣  Split the fused projection, apply an optional pair‑wise mask,
    compute sigmoid gates and store three tensors `left`, `right`,
    `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (torch.bmm) evaluates the core Einsum
       out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    yielding a tensor `[B, H, N, N]` (FP16).
5️⃣  Row‑wise mean & rsqrt‑variance of `out` across the hidden dimension.
6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection (H → C) into a single Triton kernel and write the final
    tensor `[B, N, N, C]` in FP32.

If Triton cannot be imported (e.g. CPU‑only test runs) a pure‑PyTorch
fallback is used – correctness is guaranteed in all environments.
"""

import torch
import torch.nn.functional as F

# --------------------------------------------------------------
# Try to import Triton; if unavailable we fall back to PyTorch only.
# --------------------------------------------------------------
try:
    import triton
    import triton.language as tl
    _HAS_TRITON = True
except Exception:  # pragma: no cover
    _HAS_TRITON = False

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton implementation
# ----------------------------------------------------------------------
if _HAS_TRITON:
    @triton.jit
    def _layernorm_kernel(
        x_ptr, y_ptr, w_ptr, b_ptr,
        rows, dim, eps,
        BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < rows

        # ---- first pass: compute sum and sum of squares (FP32) ----
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ---- second pass: write normalized values ----
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            norm = (vals - mean[:, None]) * rstd[:, None]

            gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
            beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

            out = norm * gamma[None, :] + beta[None, :]
            tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                     out.to(tl.float16),
                     mask=mask_m[:, None] & mask_d[None, :])

# 2️⃣ Split fused projection, apply mask & sigmoid gates – Triton
# ----------------------------------------------------------------------
if _HAS_TRITON:
    @triton.jit
    def _split_fuse_gate_kernel(
        proj_ptr,                # [R, 5·H] FP16 fused projection
        left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N] FP16
        mask_ptr,                # [R] FP16 (optional)
        R, H, N,
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
        mask_r = offs_r < R

        # decode flat index r = b·N·N + i·N + k
        nn = N * N
        b   = offs_r // nn
        rem = offs_r - b * nn
        i   = rem // N
        k   = rem - i * N

        # optional scalar mask (b,i,k)
        if HAS_MASK:
            m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
            m_val = m_val[:, None]        # broadcast over H
        else:
            m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        hidden_stride = N * N                 # stride between consecutive H‑planes
        base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h=0

        # iterate hidden dimension in tiles
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # layout of proj: [left | right | left_gate | right_gate | out_gate]
            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

            left_raw = tl.load(
                proj_ptr + proj_base,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            right_raw = tl.load(
                proj_ptr + proj_base + H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            left_gate_raw = tl.load(
                proj_ptr + proj_base + 2 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            right_gate_raw = tl.load(
                proj_ptr + proj_base + 3 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            out_gate_raw = tl.load(
                proj_ptr + proj_base + 4 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )

            # sigmoid (using FP32 for stability)
            left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            left  = left_raw  * m_val * left_gate
            right = right_raw * m_val * right_gate
            og    = out_gate

            dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride
            tl.store(left_ptr + dst_off, left,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr + dst_off, right,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr + dst_off, og,
                     mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
if _HAS_TRITON:
    @triton.jit
    def _rowwise_stats_kernel(
        out_ptr,      # [B, H, N, N] FP16
        mu_ptr,       # [R] FP32
        rstd_ptr,     # [R] FP32
        total_rows, H, N,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j   # start of h=0

        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var  = sum_sq / H - mean * mean
        rstd = tl.rsqrt(var + eps)

        tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
        tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# 4️⃣ Fuse second LayerNorm, out_gate and final projection – Triton
# ----------------------------------------------------------------------
if _HAS_TRITON:
    @triton.jit
    def _final_output_kernel(
        out_ptr,               # [B, H, N, N] FP16
        gate_ptr,              # [B, H, N, N] FP16 (out_gate)
        mu_ptr, rstd_ptr,      # [R] FP32
        ln_w_ptr, ln_b_ptr,    # [H]  FP32 (to_out_norm)
        w_out_T_ptr,           # [H, dim] FP16 (weight transposed)
        out_final_ptr,         # [B, N, N, dim] FP32
        total_rows, H, N, dim,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    ):
        pid_m = tl.program_id(0)   # rows over R
        pid_n = tl.program_id(1)   # columns over dim

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
        offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

        mask_m = offs_m < total_rows
        mask_n = offs_n < dim

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j   # start of h=0

        # per‑row LN statistics
        mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load hidden‑plane values (FP16 → FP32)
            ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

            out_vals  = tl.load(ptr_out,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)   # [M, K]

            gate_vals = tl.load(ptr_gate,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)   # [M, K]

            # LayerNorm parameters for this hidden tile
            gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
            beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

            # (out - μ) * rstd → γ· + β → * out_gate
            diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
            norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
            term = norm * gate_vals                                   # [M, K]

            term_fp16 = term.to(tl.float16)   # back to FP16 for dot

            # final projection tile (H, dim)
            w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
            w_tile = tl.load(w_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)                                 # [K, TILE_N]

            # accumulate matmul (FP16·FP16 → FP32)
            acc += tl.dot(term_fp16, w_tile).to(tl.float32)

        # store result
        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr, acc,
                 mask=mask_m[:, None] & mask_n[None, :])

# Entry point – handles both Triton‑enabled and pure‑PyTorch paths.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
          - input_tensor : torch.Tensor [B, N, N, dim] (float32)
          - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
          - weights_dict : {"norm.weight":..., "norm.bias":..., "left_proj.weight":...,
                            "right_proj.weight":..., "left_gate.weight":...,
                            "right_gate.weight":..., "out_gate.weight":...,
                            "to_out_norm.weight":..., "to_out_norm.bias":...,
                            "to_out.weight":...}
          - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # If Triton is unavailable, fall back to a pure‑PyTorch reference.
    # --------------------------------------------------------------
    if not _HAS_TRITON:  # pragma: no cover
        # 1️⃣ layer‑norm over C
        x_norm = F.layer_norm(x, (dim,),
                              weight=w["norm.weight"],
                              bias=w["norm.bias"],
                              eps=eps).to(torch.float16)

        # 2️⃣ fused projection (5·H)
        fused_W = torch.cat(
            [w["left_proj.weight"], w["right_proj.weight"],
             w["left_gate.weight"], w["right_gate.weight"],
             w["out_gate.weight"]], dim=0
        ).to(torch.float16)               # [5·H, dim]
        proj = F.linear(x_norm, fused_W)   # [B,N,N,5·H] → flatten later

        proj = proj.view(R, 5 * H)         # [R, 5·H]

        # 3️⃣ split, mask, sigmoid gates
        if not nomask and mask is not None:
            mask_flat = mask.to(torch.float16).reshape(R, 1)   # [R,1]
        else:
            mask_flat = torch.ones(R, 1, dtype=torch.float16, device=device)

        left_raw = proj[:, :H]
        right_raw = proj[:, H:2 * H]
        left_gate_raw = proj[:, 2 * H:3 * H]
        right_gate_raw = proj[:, 3 * H:4 * H]
        out_gate_raw = proj[:, 4 * H:]

        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        out_gate = torch.sigmoid(out_gate_raw)

        left = (left_raw * mask_flat) * left_gate    # [R, H]
        right = (right_raw * mask_flat) * right_gate

        # reshape to [B, H, N, N]
        left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
        right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
        out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

        # 4️⃣ batched GEMM (einsum)
        left_mat = left.reshape(B * H, N, N)
        right_mat = right.reshape(B * H, N, N)
        out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
        out = out.view(B, H, N, N)                            # [B, H, N, N]

        # 5️⃣ second layernorm (row‑wise over H)
        out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
        mu = out_flat.mean(dim=1, keepdim=True)
        var = out_flat.var(dim=1, unbiased=False, keepdim=True)
        rstd = (var + eps).rsqrt()
        out_norm = (out_flat - mu) * rstd
        out_norm = out_norm * w["to_out_norm.weight"] + w["to_out_norm.bias"]
        out_norm = out_norm.view(B, N, N, H).permute(0, 3, 1, 2)   # [B, H, N, N]

        # 6️⃣ apply out_gate and final projection
        out_gate = out_gate  # already [B, H, N, N]
        final = (out_norm * out_gate).permute(0, 2, 3, 1)  # [B, N, N, H]
        final = F.linear(final, w["to_out.weight"].to(torch.float16)).to(torch.float32)
        return final

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    _layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_W = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [5·H, dim]

    proj = F.linear(x_norm, fused_W)          # (R, 5·H) FP16
    del fused_W, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    _split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS via torch.bmm)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128          # covers H ≤ 768 comfortably
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    _rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    _final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],    # γ (FP32)
        w["to_out_norm.bias"],      # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```
