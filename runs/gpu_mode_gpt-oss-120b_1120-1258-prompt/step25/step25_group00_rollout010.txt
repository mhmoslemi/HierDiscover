@triton.jit
def ln_gate_kernel(out_ptr, gate_ptr, gamma_ptr, beta_ptr, post_ptr, rows, H, N, eps,
                   BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # decode row into indices: B,N,N as before (R = B * N * N)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # starting offset for hidden 0
    
    # First pass compute sum, sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        # load out values
        ptr = out_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        vals = tl.load(ptr, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute normalized and multiplied with gate
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        
        val_out = tl.load(ptr_out, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        val_gate = tl.load(ptr_gate, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

        # LayerNorm
        norm = (val_out - mean[:, None]) * rstd[:, None]  # [BLOCK_M, BLOCK_K]
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)     # [BLOCK_K]
        norm = norm * gamma[None,:] + beta[None,:]

        # Multiply with gate
        gated = norm * val_gate

        # Convert and store to post (flattened rows)
        # Destination offset: flatten as row-major: row*H + hidden
        dst_ptr = post_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr, gated.to(tl.float16), mask=mask_m[:,None] & mask_k[None,:])

def ln_gate_kernel(
    out_ptr, gate_ptr, gamma_ptr, beta_ptr, post_ptr,
    rows, H, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # First pass: sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for kk in range(0, H, BLOCK_K):
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute normalized output and multiply with gate, then store to post
    for kk in range(0, H, BLOCK_K):
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None,:] * hidden_stride
        val_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)
        val_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

        # Normalize out
        norm = (val_out - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_K]
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)  # [BLOCK_K]
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)    # [BLOCK_K]
        norm = norm * gamma[None,:] + beta[None,:]                # [BLOCK_M, BLOCK_K]

        gated = norm * val_gate   # elementwise

        # store to post (flattened rows)
        dst_ptr = post_ptr + offs_m[:, None] * H + offs_k[None,:]   # [BLOCK_M, BLOCK_K]
        tl.store(dst_ptr, gated.to(tl.float16), mask=mask_m[:, None] & mask_k[None,:])

"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input pair representation (FP32 → FP16)
    – fused mean/var and affine transform.

2️⃣  One fused linear projection produces the five hidden tensors
    (left_proj, right_proj, left_gate, right_gate, out_gate) in FP16.

3️⃣  `split_fuse_gate_kernel` extracts the five slices, applies the optional
    mask, and applies the sigmoid to the three gate tensors.
    Resulting tensors have shape      [B, H, N, N] (FP16).

4️⃣  Core pair‑wise interaction:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    – performed with a batched GEMM (cuBLAS) on the view
      [B·H, N, N] → [B·H, N, N] (FP16).

5️⃣  `ln_gate_kernel` computes a per‑row LayerNorm over the hidden dimension,
    multiplies by the already‑sigmoided `out_gate`,
    and flattens the tensor to shape [R, H] where R = B·N·N.
    This replaces the two‐kernel “stats + final‑output” implementation.

6️⃣  Final projection to the model dimension is a single cuBLAS linear:
        (R, H)  @ (H, dim)ᵀ  → (R, dim)   (FP16)
    – followed by a reshape to [B, N, N, dim] and a cast to float32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Mixed‑precision (FP16 for the heavy tensor work, FP32 for LayerNorm
statistics) yields a large speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 input
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim]   LN weight (γ) FP32
    b_ptr,        # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass : mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass : write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16 (already sigmoid‑ed)
    mask_ptr,        # [R] FP16   (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for a given (b,i,k) – points to hidden = 0
    base = b * (H * hidden_stride) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        # destination offsets – layout [B, H, N, N] flattened as B·H·N·N
        dst_offset = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  LN + out_gate + flatten (R × H) – replaces the two‑kernel version
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16
    gate_ptr,         # [B, H, N, N] FP16    (sigmoided out_gate)
    gamma_ptr,        # [H] FP32   (to_out_norm.weight)
    beta_ptr,         # [H] FP32   (to_out_norm.bias)
    post_ptr,         # [R, H] FP16   output buffer
    rows, H, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0 for this (b,i,j)

    # ---- first pass : mean & variance across H ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass : normalize, apply LN affine, multiply by out_gate, store ----
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw out and gate
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        val_out  = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)    # [M, K]
        val_gate = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)    # [M, K]

        # LayerNorm (FP32) + affine (γ,β)
        norm = (val_out - mean[:, None]) * rstd[:, None]          # [M, K]
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta  = tl.load(beta_ptr  + offs_k, mask=mask_k, other=0.0)  # [K]
        norm = norm * gamma[None, :] + beta[None, :]               # [M, K]

        # Apply out‑gate (already sigmoid‑ed)
        gated = norm * val_gate                                    # [M, K]

        # Store flattened (R, H)
        dst = post_ptr + offs_m[:, None] * H + offs_k[None, :]     # [M, K]
        tl.store(dst, gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"          (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # rows for the flattened (b,i,j) view
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat   # free the FP32 view

    # ------------------------------------------------------------------
    # 2️⃣  Fused projection producing 5·H hidden vectors
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask and apply sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Core pairwise multiplication via batched GEMM
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] and perform bmm( left , rightᵀ )
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣  LayerNorm + out_gate + flatten to (R, H)
    # ------------------------------------------------------------------
    post = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_STATS),)
    ln_gate_kernel[grid_ln_gate](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        post,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )
    del out, out_gate   # free memory

    # ------------------------------------------------------------------
    # 6️⃣  Final projection (single cuBLAS GEMM)
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H] FP16
    final_fp16 = torch.nn.functional.linear(post, w_out)   # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # -> FP32

    return final
