```
def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm (native)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps) # float32
    x_norm = x_norm.to(torch.float16)

    # 2. fused linear projection
    fused_proj = torch.nn.functional.linear(x_norm, torch.cat([...], dim=0).to(torch.float16))  # shape [R,5H] fp16

    # 3. split and apply mask + sigmoid
    # Convert mask to FP16 and reshape to [R,1] if needed.
    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16).unsqueeze(1) # [R,1]
    else:
        mask_flat = torch.ones((R,1), dtype=torch.float16, device=x.device)

    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(fused_proj, H, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left * mask_flat * left_gate
    right = right * mask_flat * right_gate

    # 4. core einsum: left @ right^T across k dimension
    left = left.view(B, H, N, N)
    right = right.view(B, H, N, N)
    out = torch.bmm(left.view(B*H, N, N), right.view(B*H, N, N).transpose(1,2))
    out = out.view(B, H, N, N)

    # 5. second layernorm stats: mu,rstd
    mu = out.mean(dim=1, keepdim=True)        # shape [B,1,N,N]
    var = out.var(dim=1, unbiased=False, keepdim=True)
    rstd = 1.0 / torch.sqrt(var + eps)

    # 6. apply second LayerNorm, out_gate, final linear
    # LN: (out - mu) * rstd * to_out_norm.weight + to_out_norm.bias
    # to_out_norm.weight shape [hidden_dim], broadcast.
    out_norm = (out - mu) * rstd
    out_norm = out_norm * w["to_out_norm.weight"][None,:,None,None] + w["to_out_norm.bias"][None,:,None,None]   # shape [B,H,N,N]

    out_norm = out_norm * out_gate   # elementwise gating

    # final linear
    # reshape to (R, H)
    out_norm_flat = out_norm.permute(0,2,3,1).reshape(R, H)  # maybe better order: B,N,N,H to flatten.
    # Actually out_norm shape [B,H,N,N], but we want row-major where each (b,i,j) -> H dimension.
    # We can transpose to [B,N,N,H] then reshape.
    out_norm_flat = out_norm.permute(0,2,3,1).contiguous().view(R, H)  # shape [R,H]
    out_final = torch.nn.functional.linear(out_norm_flat, w["to_out.weight"], bias=None) # shape [R, dim]
    out_final = out_final.view(B, N, N, dim)  # shape [B,N,N,dim]
    # ensure output dtype float32
    out_final = out_final.to(torch.float32)
    return out_final
```

```
@triton.jit
def split_fuse_gate_fp32_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode indices as before
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                   # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float32)

    hidden_stride = N * N

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # base offset for this row in fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )

        # sigmoid using float32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw)))   # or use tl.sigmoid?
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw)))
```

```python
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: split fused projection, apply optional mask and sigmoid gates
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP32 fused projection
    left_ptr,      # [B, H, N, N] FP32
    right_ptr,     # [B, H, N, N] FP32
    out_gate_ptr,  # [B, H, N, N] FP32
    mask_ptr,      # [R]       FP32 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        m_val = m_val[:, None]                     # broadcast over hidden dim
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    hidden_stride = N * N
    # base address for the first hidden slice (h=0) for this (b,i,k)
    base = b * (H * hidden_stride) + i * N + k   # shape [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)

            mask=mask_r[:, None] & mask_h[None, :],

            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
```

```python
def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold3) forward‑only implementation.

    The algorithm:
      1️⃣ Row‑wise LayerNorm on the pair tensor (float32 → float32).
      2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
         into one dense GEMM.
      3️⃣ Using a tiny Triton kernel, split the fused projection, apply an optional
         pair mask and sigmoid gates, and write the three tensors ``left``,
         ``right`` and ``out_gate`` (all float32) in layout [B, H, N, N].
      4️⃣ Core ``einsum`` is realized as a batched matrix‑multiply (cuBLAS):
         out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k].
      5️⃣ Compute per‑pair mean / variance of ``out`` with a single
         torch.mean / torch.var call (float32).
      6️⃣ Apply the second LayerNorm, the ``out_gate`` gating and the final
         linear projection (hidden_dim → dim) using torch.nn.functional.linear.
         The final result is cast to float32 and returned.

    All heavy tensor work is performed by highly‑optimized cuBLAS / torch kernels.
    Only the mask‑and‑gate split step is written in Triton.
    """

    # ---------------------------------------------------------
    # unpack input
    # ---------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ---------------------------------------------------------
    # 1️⃣ first LayerNorm (float32)
    # ---------------------------------------------------------
    # torch.layer_norm works on the last dimension; normalized_shape = (dim,)
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim], dtype float32

    # ---------------------------------------------------------
    # 2️⃣ fused projection  (float32 GEMM)
    # ---------------------------------------------------------
    # Concatenate the five weight matrices (shape [5*H, dim])
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # [5*H, dim], float32
    # Flatten x_norm to [R, dim] for a single linear call
    x_flat = x_norm.reshape(R, dim)
    proj = torch.nn.functional.linear(x_flat, fused_w)  # [R, 5*H], float32
    del x_norm, x_flat, fused_w

    # ---------------------------------------------------------
    # 3️⃣ split, mask and sigmoid gates (Triton)
    # ---------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float32, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float32, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float32, device=device)

    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float32).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float32, device=device)  # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden dimension tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ---------------------------------------------------------
    # 4️⃣ core einsum via batched GEMM (cuBLAS)
    # ---------------------------------------------------------
    # shape to [B*H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N], float32
    out = out.view(B, H, N, N)                              # [B, H, N, N]

    # ---------------------------------------------------------
    # 5️⃣ second LayerNorm statistics (mean / var) – torch implementation
    # ---------------------------------------------------------
    # mean & variance along hidden dimension (dim=1)
    mu = out.mean(dim=1, keepdim=True)        # [B,1,N,N]
    var = out.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = 1.0 / torch.sqrt(var + eps)        # [B,1,N,N]

    # ---------------------------------------------------------
    # 6️⃣ LN, gating and final projection (torch)
    # ---------------------------------------------------------
    # Apply learned scale (γ) and bias (β) of the second LayerNorm
    ln_w = weights["to_out_norm.weight"].view(1, H, 1, 1)  # broadcast over B,N,N
    ln_b = weights["to_out_norm.bias"].view(1, H, 1, 1)

    # Normalisation + learned affine transform
    out_norm = (out - mu) * rstd
    out_norm = out_norm * ln_w + ln_b

    # Apply the output gate
    out_norm = out_norm * out_gate

    # Collapse the pair dimensions so that each row corresponds to a (b,i,j)
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]

    # Final linear projection (hidden_dim -> dim)
    w_out = weights["to_out.weight"].to(torch.float32)  # [dim, hidden_dim]
    final = torch.nn.functional.linear(out_flat, w_out)  # [R, dim], float32

    # Reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)   # [B, N, N, dim]
    return final
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  fp16 fused projection
    left_ptr,      # [B, H, N, N] fp16
    right_ptr,     # [B, H, N, N] fp16
    out_gate_ptr,  # [B, H, N, N] fp16
    mask_ptr,      # [R]       fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # start address for hidden dim 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offset into the fused projection (layout: left|right|lg|rg|og)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for better accuracy, result stored FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold3) forward‑only implementation.

    Steps
    -----
    1️⃣ Apply a row‑wise LayerNorm on the pairwise tensor (float32 → FP16).
    2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
       into a single GEMM (fp16) and compute a tensor of shape [R,5·H].
    3️⃣ A small Triton kernel splits that fused projection, applies the optional
       pair mask and the sigmoidal gates, and writes three tensors
       ``left``, ``right`` and ``out_gate`` in layout [B, H, N, N] (FP16).
    4️⃣ The core ``einsum`` is performed as a batched matrix‑multiply
       (cuBLAS) producing ``out`` of shape [B, H, N, N] (FP16).
    5️⃣ Compute per‑pair mean and variance of ``out`` (float32) with a single
       torch reduction.
    6️⃣ Apply the second LayerNorm, the ``out_gate`` gating and the final
       linear projection (hidden_dim → dim) using only torch kernels.
       The final tensor is cast back to float32.

    Only the mask‑and‑gate split step is written in Triton; the heavy
    tensor work is delegated to cuBLAS / torch kernels, yielding a fast
    forward pass.
    """
    # ---------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ---------------------------------------------------------
    # 1️⃣ First layer‑norm (float32) and cast to FP16
    # ---------------------------------------------------------
    # torch.layer_norm works on the last dimension; normalized_shape = (dim,)
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)   # [B, N, N, dim] FP16

    # ---------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single FP16 GEMM
    # ---------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [5·H, dim]

    # Collapse the pair dimension for a single matmul
    x_flat = x_norm.view(R, dim)                       # [R, dim] FP16
    proj = torch.nn.functional.linear(x_flat, fused_w)  # [R, 5·H] FP16
    del x_norm, x_flat, fused_w

    # ---------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoids (Triton)
    # ---------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ---------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS, FP16)
    # ---------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)           # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                   # [B, H, N, N] FP16

    # ---------------------------------------------------------
    # 5️⃣ Per‑pair mean / variance (float32) – torch reduction
    # ---------------------------------------------------------
    mu = out.mean(dim=1, keepdim=True, dtype=torch.float32)            # [B,1,N,N] FP32
    var = out.var(dim=1, unbiased=False, keepdim=True, dtype=torch.float32)  # [B,1,N,N] FP32
    rstd = 1.0 / torch.sqrt(var + eps)                                 # [B,1,N,N] FP32

    # ---------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final projection (FP16 -> FP32)
    # ---------------------------------------------------------
    # Learned scale & bias of the second LayerNorm (float32)
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    ln_b = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1]

    # Normalise, apply affine transform and gated multiplication
    # Cast `out` to FP32 for the arithmetic, then back to FP16
    out_fp32 = out.float()
    norm = (out_fp32 - mu) * rstd                     # FP32
    norm = norm * ln_w + ln_b                         # FP32
    norm = norm * out_gate.float()                    # FP32
    norm_fp16 = norm.half()                           # FP16, shape [B,H,N,N]

    # Collapse to (R, H) for the final linear projection
    norm_flat = norm_fp16.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] FP16

    # Final linear projection (hidden_dim → dim) in FP16, then cast to FP32
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    final = torch.nn.functional.linear(norm_flat, w_out_fp16)  # [R, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)        # [B, N, N, dim] FP32

    return final
