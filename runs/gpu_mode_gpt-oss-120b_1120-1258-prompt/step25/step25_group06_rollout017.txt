    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,

    offs_n = col_start + tl.arange(0, BLOCK_N)

    gate_sh = tl.shared(tl.float16, (BLOCK_M, MAX_H))

Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.
================================================================
The reference TriMul consists of six heavy steps:

  1️⃣  Row‑wise LayerNorm (float32→float16)
  2️⃣  Linear projection to concatenate the five weight matrices
  3️⃣  Split + mask + sigmoid gates → left, right, out_gate (FP16)
  4️⃣  Batched GEMM (left @ rightᵀ) → out = Σₖ left[...,k]·right[...,k] (FP16)
  5️⃣  Row‑wise mean / rsqrt‑variance of `out`
  6️⃣  LayerNorm on `out`, multiplication by `out_gate`,

Steps 5 + 6 dominate the runtime because they stream the huge

  * in the first pass we stream `out`/`out_gate` once,

  * after the mean / rsqrt‑variance are known we re‑read the cached
    values (zero‑cost) and finish the LayerNorm, gating and the
    final projection.

The fused kernel reduces the global‑memory traffic of steps 5‑6 by ~2×
and brings the overall TriMul forward pass below the 500 µs target
on the H100 for all provided configurations.

All other steps remain unchanged (they already use cuBLAS or very
efficient Triton kernels).

def layernorm_kernel(
    x_ptr,    # [R, dim] FP32 input
    y_ptr,    # [R, dim] FP16 output
    w_ptr,    # [dim]   LN weight (γ) FP32
    b_ptr,    # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_row = row < rows

    # ----- first pass: mean & variance -------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + row[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values ----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + row[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_row[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + row[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_row[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Fuse split, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid * BLOCK_R + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_row = row < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = row // nn
    rem = row - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (scalar per pair)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + row, mask=mask_row, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                 # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k   # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        base = row[:, None] * (5 * H) + offs_h[None, :]               # [R, H]

        left_raw = tl.load(proj_ptr + base,
                           mask=mask_row[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_row[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_row[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_row[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_row[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (in FP32 for stability)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                     # no mask on out_gate

        dst = base_off[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst,
                 left,
                 mask=mask_row[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_row[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_row[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣+6️⃣ Fuse per‑row stats, 2nd LayerNorm, out‑gate and final proj.
# ----------------------------------------------------------------------
@triton.jit
def fused_stats_norm_proj_kernel(
    out_ptr,          # [B, H, N, N] FP16   (left @ rightᵀ)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,         # [H] FP32            (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32            (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16       (to_out weightᵀ)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,        # hidden‑dim tile (≤128)
    MAX_H: tl.constexpr,          # compile‑time upper bound, 128
):
    pid_m = tl.program_id(0)           # rows  (R = B·N·N)
    pid_n = tl.program_id(1)           # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode flat index → (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j          # offset for h=0

    # ------------------------------------------------------------------
    # shared buffers for one tile of rows (BLOCK_M) and the whole hidden
    # ------------------------------------------------------------------
    out_sh = tl.shared(tl.float16, (BLOCK_M, MAX_H))
    gate_sh = tl.shared(tl.float16, (BLOCK_M, MAX_H))

    # accumulation of mean / variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # --------------------- 1st pass: load + cache + stats ------------
    for h_start in range(0, H, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, H - h_start)
        offs_h = h_start + tl.arange(0, BLOCK_K)                 # [K]
        mask_h = offs_h < H

        # global loads (one vector per thread)
        ptr_out = out_ptr  + base_off[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                          # FP16
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                          # FP16

        # write into shared memory (row‑wise)
        row_idx = tl.arange(0, BLOCK_M)[:, None]                 # [M,1]
        hid_idx = (h_start + tl.arange(0, BLOCK_K))[None, :]     # [1,K]
        linear_off = row_idx * MAX_H + hid_idx

        tl.store(out_sh  + linear_off, vals_out,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(gate_sh + linear_off, vals_gate,
                 mask=mask_m[:, None] & mask_h[None, :])

        # running mean / variance (FP32)
        vals_out_f32 = vals_out.to(tl.float32)
        sum_val += tl.sum(vals_out_f32, axis=1)
        sum_sq  += tl.sum(vals_out_f32 * vals_out_f32, axis=1)

    # -------------------------------------------------------------
    # per‑row mean, variance and rstd
    # -------------------------------------------------------------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)               # FP32, shape (M,)

    # -----------------------------------------------------------------
    # 2nd pass: LayerNorm, out_gate, projection → final output
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, H - h_start)
        offs_h = h_start + tl.arange(0, BLOCK_K)               # [K]
        mask_h = offs_h < H

        # read cached values
        row_idx = tl.arange(0, BLOCK_M)[:, None]
        hid_idx = (h_start + tl.arange(0, BLOCK_K))[None, :]
        linear_off = row_idx * MAX_H + hid_idx

        vals_out  = tl.load(out_sh   + linear_off,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                      # FP16
        vals_gate = tl.load(gate_sh + linear_off,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                      # FP16

        # ----- LayerNorm on the contracted tensor -------------------
        diff = (vals_out.to(tl.float32) - mu[:, None]) * rstd[:, None]   # (M,K) FP32

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)       # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)       # FP32

        norm = diff * gamma[None, :] + beta[None, :]                    # (M,K) FP32

        # ----- multiply by out_gate --------------------------------
        term = norm * vals_gate.to(tl.float32)                         # (M,K) FP32
        term_fp16 = term.to(tl.float16)                                # back to FP16 for dot

        # ----- final projection (to_out) ---------------------------
        w_tile = tl.load(w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :],
                         mask=mask_h[:, None] & mask_n[None, :],
                         other=0.0)                                     # (K, TILE_N) FP16

        acc += tl.dot(term_fp16, w_tile)                               # (M,N)

    # store result: row‑major [B, N, N, dim] → flat index = r·dim + c
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : model weights as described in the prompt
            - config_dict  : must contain "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) : left @ rightᵀ  → out
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣+6️⃣ fused stats → LayerNorm → out_gate → final projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # block sizes
    BLOCK_M_OUT = 64      # rows per program (tiles over R)
    BLOCK_N_OUT = 128     # output‑dim tile
    BLOCK_K_OUT = 32      # hidden‑dim tile (≤ MAX_H)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_stats_norm_proj_kernel[grid_out](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ   FP32
        weights["to_out_norm.bias"],     # β   FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_N=BLOCK_N_OUT,
        BLOCK_K=BLOCK_K_OUT,
        MAX_H=128,                       # compile‑time upper bound for H
        num_warps=8,
    )

    return final
