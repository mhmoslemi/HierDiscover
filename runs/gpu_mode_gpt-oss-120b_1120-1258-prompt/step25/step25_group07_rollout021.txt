@triton.jit
def layernorm_fused_proj_gate_kernel(
    x_ptr,            # [R, dim] FP32
    left_ptr, right_ptr, out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,                         # [R] FP16 optional
    ln_w_ptr, ln_b_ptr,   # [dim] FP32
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,   # [dim, H] FP16 transposed
    B, N, H, dim, eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R)
   mask_r = offs_r < (B * N * N)   # total rows

   # Decode position
   nn = N * N
   b = offs_r // nn
   rem = offs_r - b * nn
   i = rem // N
   k = rem - i * N

   # Compute per-row mean and var across dim
   sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim

       x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

       sum_val += tl.sum(x_tile, axis=1)
       sum_sq += tl.sum(x_tile * x_tile, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # Prepare LN weight & bias tiles (same across hidden tiles)
   # We'll load gamma/beta inside dim loop to reduce duplicate loads

   # Loop over hidden tiles
   hidden_stride = N * N
   base_offset = b * (H * hidden_stride) + i * N + k

   for h_off in range(0, tl.cdiv(H, BLOCK_H)):
       offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H

       # Initialize accumulators
       left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
       right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
       left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
       right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
       out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

       # Accumulate across dim
       for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
           offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
           mask_d = offs_d < dim

           # Load x tile
           x_tile = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                            mask=mask_r[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)

           # Normalize
           norm = (x_tile - mean[:, None]) * rstd[:, None]   # [R, D]

           # Apply LN gamma & beta
           gamma_tile = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)   # [D]
           beta_tile = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)   # [D]
           norm = norm * gamma_tile[None, :] + beta_tile[None, :]

           # Cast to fp16 for dot
           norm_f16 = norm.to(tl.float16)

           # Load weight tiles (dim, H) transposed slices
           # left_w_tile shape [BLOCK_D, BLOCK_H]
           left_w_tile = tl.load(
               left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
               mask=mask_d[:, None] & mask_h[None, :],
               other=0.0
           )
           right_w_tile = tl.load(
               right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
               mask=mask_d[:, None] & mask_h[None, :],
               other=0.0
           )
           left_gate_w_tile = tl.load(
               left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
               mask=mask_d[:, None] & mask_h[None, :],
               other=0.0
           )
           right_gate_w_tile = tl.load(
               right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
               mask=mask_d[:, None] & mask_h[None, :],
               other=0.0
           )
           out_gate_w_tile = tl.load(
               out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
               mask=mask_d[:, None] & mask_h[None, :],
               other=0.0
           )

           # Accumulate dot products
           left_acc += tl.dot(norm_f16, left_w_tile).to(tl.float32)
           right_acc += tl.dot(norm_f16, right_w_tile).to(tl.float32)
           left_gate_acc += tl.dot(norm_f16, left_gate_w_tile).to(tl.float32)
           right_gate_acc += tl.dot(norm_f16, right_gate_w_tile).to(tl.float32)
           out_gate_acc += tl.dot(norm_f16, out_gate_w_tile).to(tl.float32)

       # After dim loop, apply mask (if any) and sigmoid gates
       if HAS_MASK:
           mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
           mask_vec = mask_val[:, None]   # [R,1] broadcast to [R, BLOCK_H]
       else:
           mask_vec = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

       # Sigmoid on gate accumulators (convert to fp32 for stability)
       left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
       right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
       out_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

       # Multiply
       left_out = left_acc.to(tl.float16) * mask_vec * left_gate
       right_out = right_acc.to(tl.float16) * mask_vec * right_gate
       out_gate_out = out_gate

       # Store
       dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
       tl.store(left_ptr + dst_offset, left_out,
                mask=mask_r[:, None] & mask_h[None, :])
       tl.store(right_ptr + dst_offset, right_out,
                mask=mask_r[:, None] & mask_h[None, :])
       tl.store(out_gate_ptr + dst_offset, out_gate_out,
                mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold 3) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module but moves all
memory‑intensive element‑wise work to Triton kernels:

1️⃣ Row‑wise LayerNorm (FP32 → FP16) – `layernorm_kernel`.
2️⃣ Fused linear projection (left/right + the three gates) performed by
   PyTorch’s highly‑optimized GEMM (torch.nn.functional.linear).  The
   huge (R × 5·H) intermediate is then split, masked, and the sigmoid
   gates are applied in `split_fuse_gate_kernel`.
3️⃣ Batched matrix‑multiply (torch.bmm) computes the core Einstein sum:
   `out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]`.
4️⃣ Row‑wise mean & rsqrt‑variance of `out` – `rowwise_stats_kernel`.
5️⃣ The second LayerNorm, out‑gate multiplication and the final
   projection are fused in `final_output_kernel`.

All kernels target Triton 3.3.1, run on an NVIDIA H100 and use mixed‑precision
(FP16 for bulk work, FP32 for reductions) to keep the memory traffic low while
preserving numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim] FP32 input (flattened)
    y_ptr,          # [R, dim] FP16 output
    w_ptr,          # [dim]   LayerNorm weight (γ) FP32
    b_ptr,          # [dim]   LayerNorm bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Normalize every row of a (R×dim) matrix."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Proj contains the concatenated linear maps
    [left | right | left_gate | right_gate | out_gate].
    This kernel splits the tensor, multiplies by the optional mask,
    applies the sigmoid to the three gate streams and writes the three
    tensors in layout [B, H, N, N] (FP16).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                          # distance between consecutive hidden planes
    base_offset = b * (H * stride_hidden) + i * N + k    # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]           # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for stability, back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """For every output position (b,i,j) compute mean and rsqrt‑variance across H."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h = 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """
    Implements
        out = to_out_norm(out) * out_gate
        out = to_out(out)
    where `out` is the result of the einsum.
    All ops are fused to avoid extra global memory traffic.
    """
    pid_m = tl.program_id(0)   # rows over R
    pid_n = tl.program_id(1)   # output‑dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h = 0

    # load the per‑row LN statistics (μ, rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)    # [M, K]

        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)    # [M, K]

        # LayerNorm parameters for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                        # [M, K]

        # cast back to FP16 for the matmul with the final projection weight
        term_fp16 = term.to(tl.float16)   # [M, K]

        # tile of the final weight matrix (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # [K, TILE_N]

        # Accumulate matmul (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final [B, N, N, dim] tensor (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (5·hidden_dim) – torch linear (cuBLAS)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H)   FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128               # covers all supported hidden sizes
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
