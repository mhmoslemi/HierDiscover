def ln_gate_kernel(

    out_norm_gated_ptr,
    total_rows, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Offsets for hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):

        mask_h = offs_h < H

@triton.jit
def ln_gate_fused_kernel(
    out_ptr, gate_ptr, out_norm_ptr,
    gamma_ptr, beta_ptr,
    total_rows, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # allocate shared memory for a tile of out values (FP16)
    # but we can simply load values into registers in loop across H dimension.
    # Since H relatively small, we can compute sum and sum_sq in a for loop over H blocks.
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # First pass: compute sum and sum_sq
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # [M, H_block]

        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute normalized and gated
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)  # FP16

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)  # FP16

        # load gamma, beta
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # compute
        out_f = out_vals.to(tl.float32)
        gate_f = gate_vals.to(tl.float32)
        norm = ((out_f - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated = norm * gate_f

        # write to out_norm_ptr (store as FP16)
        tl.store(out_norm_ptr + offs_m[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized Triton + PyTorch implementation

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM:  (R, dim) × (5·H, dim)ᵀ → (R, 5·H)  where R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply the optional pair mask and sigmoid gates,
   producing three tensors **left**, **right**, **out_gate** in layout
   [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, k, j]   →  [B, H, N, N] FP16.
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the
   reduction of per‑row statistics (mean / rsqrt‑variance) in a single
   Triton kernel.  The kernel writes a flat tensor **norm_gated**
   of shape [R, H] (R = B·N·N) in FP16.
6️⃣ A single dense GEMM (torch.nn.functional.linear) projects the
   normalized, gated hidden features to the output channel dimension:
       final = norm_gated @ to_out.weightᵀ   →  [R, dim] (FP16 → FP32).

All kernels target Triton 3.3.1 and make heavy use of Tensor‑Cores
(FP16 × FP16 → FP32) on an NVIDIA H‑100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                 # [R, dim]  FP32 input
    y_ptr,                 # [R, dim]  FP16 output
    w_ptr,                 # [dim]    γ (FP32)
    b_ptr,                 # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – compute mean & variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals,  axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalized values --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)                     # FP16
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and produce a flat [R, H] tensor
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_fused_kernel(
    out_ptr,               # [B, H, N, N] FP16   (raw contraction result)
    gate_ptr,              # [B, H, N, N] FP16   (out_gate)
    out_norm_ptr,          # [R, H] FP16         (output of this kernel)
    ln_w_ptr,              # [H] FP32            (to_out_norm.weight)
    ln_b_ptr,              # [H] FP32            (to_out_norm.bias)
    total_rows, hidden_dim, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [M]
    mask_m = offs_m < total_rows

    # decode (b, i, j) from the flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (hidden_dim * hidden_stride) + i * N + j   # start of plane h = 0

    # ---------- first pass – compute mean & rstd over H ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, hidden_dim, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        vals = tl.load(out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass – normalize, apply LN weight/bias & out_gate ----------
    for h_start in range(0, hidden_dim, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        out_vals = tl.load(out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)                          # FP16
        gate_vals = tl.load(gate_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                         # FP16

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        out_f  = out_vals.to(tl.float32)
        gate_f = gate_vals.to(tl.float32)

        # LayerNorm (γ/β) and out‑gate multiplication
        norm  = ((out_f - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated = norm * gate_f                                   # still FP32

        # write to a flat [R, H] tensor (row‑major)
        dst_off = offs_m[:, None] * hidden_dim + offs_h[None, :]
        tl.store(out_norm_ptr + dst_off,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# 6️⃣ Entrypoint – complete forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total number of pair positions
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] (FP32)
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                        # [R, dim] (FP16)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free the FP32 view

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    # (R, dim) × (5·H, dim)ᵀ → (R, 5·H)  – fast cuBLAS GEMM in FP16
    proj = F.linear(x_norm, fused_w)         # [R, 5·H] FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – never accessed when HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ   (FP16)
    # -----------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)               # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)              # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16
    del left, right   # free memory we no longer need

    # -----------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and flatten to [R, H]
    # -----------------------------------------------------------------
    out_norm_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128          # rows per program
    BLOCK_H_OUT = 32           # hidden‑dim tile size
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    ln_gate_fused_kernel[grid_ln_gate](
        out,
        out_gate,
        out_norm_gated,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate   # free large intermediate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection (FP16 GEMM) → float32 result
    # -----------------------------------------------------------------
    # weight: (dim, hidden_dim) → cast to FP16 for Tensor‑Core GEMM
    w_out = w["to_out.weight"].to(torch.float16)          # [dim, H]

    final_fp16 = F.linear(out_norm_gated, w_out)          # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final
