    pid_n = tl.program_id(1)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)  # rows
    offs_n = col_start + tl.arange(0, BLOCK_N)  # cols

    acc_out = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    acc_beta = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

            mask=mask_k[:, None] & mask_n[None, :],

        out_res,

    [B, H, N, N] (FP16) via a Triton kernel.
4️⃣  Batched half‑precision GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B, H, N, N] (FP16)
5️⃣  Row‑wise LayerNorm over the hidden dimension H, out‑gate multiplication,
    and the final projection are **fused** into a single Triton kernel.
    The kernel streams the tensors, computes μ/rstd, applies the second
    LayerNorm and the out‑gate, and performs the dot‑product with the
    transposed `to_out` weight → final output [B, N, N, dim] (FP32).

Only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]    LN weight (γ) FP32
    b_ptr,   # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (per pair, broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k  # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32, cast back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate                         # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out‑gate & final projection (single kernel)
# ----------------------------------------------------------------------
@triton.jit
def final_output_fused_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulators for the three dot‑products
    acc_out   = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ out·γ·gate·W
    acc_wgt   = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ γ·gate·W
    acc_beta  = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # Σ β·gate·W

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ------------------------------------------------------------------
        # Load out and gate (FP16) and cast to FP32 for statistics
        # ------------------------------------------------------------------
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_f16 = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        out_f32 = out_f16.to(tl.float32)

        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_f16 = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        gate_f32 = gate_f16.to(tl.float32)

        sum_val += tl.sum(out_f32, axis=1)
        sum_sq  += tl.sum(out_f32 * out_f32, axis=1)

        # ------------------------------------------------------------------
        # Load LN parameters (γ,β) – cast to FP16 for the GEMM
        # ------------------------------------------------------------------
        gamma_f32 = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta_f32  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        gamma_f16 = gamma_f32.to(tl.float16)
        beta_f16  = beta_f32.to(tl.float16)

        # a = γ * gate,  b = β * gate   (both FP16)
        a_f16 = gate_f16 * gamma_f16[None, :]    # [M, K]
        b_f16 = gate_f16 * beta_f16[None, :]     # [M, K]

        # ------------------------------------------------------------------
        # Load weight slice W[h, dim] (FP16)
        # ------------------------------------------------------------------
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # FP16

        # ------------------------------------------------------------------
        # Accumulate three dot‑products
        # ------------------------------------------------------------------
        # (out * a) · W
        out_a_f16 = out_f16 * a_f16                                      # [M, K]
        acc_out += tl.dot(out_a_f16, w_tile).to(tl.float32)

        # a · W
        acc_wgt += tl.dot(a_f16, w_tile).to(tl.float32)

        # b · W
        acc_beta += tl.dot(b_f16, w_tile).to(tl.float32)

    # --------------------------------------------------------------
    # Compute μ and rstd over the hidden dimension
    # --------------------------------------------------------------
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                # [M]

    mu   = mu[:, None]    # [M,1]
    rstd = rstd[:, None]  # [M,1]

    # --------------------------------------------------------------
    # Final output = rstd*Σ(out·γ·gate·W) - mu*rstd*Σ(γ·gate·W) + Σ(β·gate·W)
    # --------------------------------------------------------------
    out_res = rstd * acc_out - mu * rstd * acc_wgt + acc_beta

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(
        dst_ptr,
        out_res,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"      (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) pairs
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]
    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # shape: [B·H, N, N] where the inner matrix is (i,k) for left
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left, right   # free memory

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out‑gate & final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # out_gate no longer needed
    del out, out_gate

    return final
