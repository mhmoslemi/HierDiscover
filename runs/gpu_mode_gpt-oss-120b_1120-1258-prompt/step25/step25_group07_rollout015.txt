    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)

    ln_bias_ptr,    # [dim] FP32
    w_T_ptr,        # [dim, 5*H] FP16  (transposed weight)

    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # -------------------------------------------------
    # Compute mean and variance per row
    # -------------------------------------------------
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)

        # accumulators for five projections
        acc_l   = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_r   = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_lg  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_rg  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_og  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # loop over dim tiles to compute GEMM
        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load a slice of the input and normalize + LN affine
            x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)   # (BLOCK_R, BLOCK_D)

            # normalize
            x_centered = x_vals - mean[:, None]                # (BLOCK_R, BLOCK_D)
            x_normed = x_centered * rstd[:, None]              # (BLOCK_R, BLOCK_D)

            # apply LN weight & bias
            gamma = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            beta  = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
            norm = x_normed * gamma[None, :] + beta[None, :]                # (BLOCK_R, BLOCK_D)

            # Load the corresponding slice of the weight matrix (dim, 5*H)
            # Output columns: start at h_start, width = 5*BLOCK_H
            col_start = h_start
            col_end   = h_start + 5 * BLOCK_H
            # create range
            offs_o = col_start + tl.arange(0, 5 * BLOCK_H)
            mask_o = offs_o < (5 * H)

            w_ptrs = w_T_ptr + offs_d[:, None] * (5 * H) + offs_o[None, :]
            w_tile = tl.load(w_ptrs,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0)                # (BLOCK_D, 5*BLOCK_H) FP16
            w_tile = w_tile.to(tl.float32)

            # split weight tile
            w_l   = w_tile[:, 0:BLOCK_H]
            w_r   = w_tile[:, BLOCK_H:2*BLOCK_H]
            w_lg  = w_tile[:, 2*BLOCK_H:3*BLOCK_H]
            w_rg  = w_tile[:, 3*BLOCK_H:4*BLOCK_H]
            w_og  = w_tile[:, 4*BLOCK_H:5*BLOCK_H]

            # accumulate dot products
            acc_l  += tl.dot(norm, w_l)
            acc_r  += tl.dot(norm, w_r)
            acc_lg += tl.dot(norm, w_lg)
            acc_rg += tl.dot(norm, w_rg)
            acc_og += tl.dot(norm, w_og)

        # -------------------------------------------------
        # Apply gates and mask, store results
        # -------------------------------------------------
        # sigmoid in fp32 then cast to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)

        # apply mask and convert left/right to fp16
        mask_fp16 = mask_val[:, None].to(tl.float16)

        left  = (acc_l * left_gate).to(tl.float16) * mask_fp16
        right = (acc_r * right_gate).to(tl.float16) * mask_fp16

        # compute destination offsets for this hidden tile
        offs_h = h_start + tl.arange(0, BLOCK_H)
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # shape (BLOCK_R, BLOCK_H)

        # store

                 mask=mask_r[:, None] & (offs_h[None, :] < H))
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & (offs_h[None, :] < H))
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & (offs_h[None, :] < H))

"""
Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Fuse LayerNorm, the five linear projections (left, right,
   left_gate, right_gate, out_gate) and the optional pair‑wise mask
   into a single Triton kernel.
   – Two passes over the input dimension:
        * first pass → per‑row mean/variance (FP32)
        * second pass → compute the five projections while applying LN
          affine (γ,β) and the sigmoid gates.
   – Results (left, right, out_gate) are written directly in the
     layout required by the subsequent batched GEMM.

2️⃣ Batched half‑precision GEMM (torch.bmm) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]   (FP16)

3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
   (still FP16) → μ, ρ (FP32).

4️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the
   final projection into one Triton kernel, producing the final
   FP32 tensor of shape [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ fused LayerNorm + projection + gates (writes left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    x_ptr,          # [R, dim]      FP32 input
    left_ptr,       # [B, H, N, N]  FP16 output
    right_ptr,      # [B, H, N, N]  FP16 output
    out_gate_ptr,   # [B, H, N, N]  FP16 output
    mask_ptr,       # [R]           FP16 mask (optional)
    ln_w_ptr,       # [dim]         LN weight γ  (FP32)
    ln_b_ptr,       # [dim]         LN bias   β  (FP32)
    w_T_ptr,        # [dim, 5*H]    fused projection weights (FP16)
    R, H, N, dim,
    eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # ----- decode flat index (b,i,k) -----
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ----- 1. compute mean & variance of each row -----
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for d_start in range(0, dim, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)           # (BLOCK_R, BLOCK_D)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- optional pair‑wise mask (scalar per (b,i,k)) -----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)  # [BLOCK_R]
    else:
        mask_val = tl.full((BLOCK_R,), 1.0, tl.float32)

    mask_val = mask_val[:, None]                     # broadcast over hidden dim

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden dim 0

    # ----- 2. iterate over hidden tiles (H is small, e.g. 128) -----
    for h_start in range(0, H, BLOCK_H):
        # accumulators for the five projection groups
        acc_l  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_r  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # ----- loop over the input dimension (dim) -----
        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load raw input and apply the first LayerNorm (mean/rstd)
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # (x - mean) * rstd
            x_centered = x - mean[:, None]
            x_normed   = x_centered * rstd[:, None]

            # affine γ,β of the first LayerNorm
            gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            norm = x_normed * gamma[None, :] + beta[None, :]             # (BLOCK_R, BLOCK_D)

            # ----- load the corresponding slice of the fused weight matrix -----
            # weight layout: (dim, 5*H) row‑major, i.e. offset = d * (5*H) + out
            col_start = h_start
            col_end   = h_start + 5 * BLOCK_H
            offs_o = col_start + tl.arange(0, 5 * BLOCK_H)
            mask_o = offs_o < (5 * H)

            w_ptrs = w_T_ptr + offs_d[:, None] * (5 * H) + offs_o[None, :]
            w_tile = tl.load(w_ptrs,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0).to(tl.float32)                     # (BLOCK_D, 5*BLOCK_H)

            # split the five sub‑matrices (each BLOCK_H columns)
            w_l  = w_tile[:, 0:BLOCK_H]
            w_r  = w_tile[:, BLOCK_H:2*BLOCK_H]
            w_lg = w_tile[:, 2*BLOCK_H:3*BLOCK_H]
            w_rg = w_tile[:, 3*BLOCK_H:4*BLOCK_H]
            w_og = w_tile[:, 4*BLOCK_H:5*BLOCK_H]

            # accumulate GEMM contributions
            acc_l  += tl.dot(norm, w_l)
            acc_r  += tl.dot(norm, w_r)
            acc_lg += tl.dot(norm, w_lg)
            acc_rg += tl.dot(norm, w_rg)
            acc_og += tl.dot(norm, w_og)

        # ----- apply sigmoid gates and the optional mask -----
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)   # (BLOCK_R, BLOCK_H)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)

        left  = (acc_l * left_gate).to(tl.float16) * mask_val
        right = (acc_r * right_gate).to(tl.float16) * mask_val

        # ----- store results in the layout [B, H, N, N] -----
        offs_h   = h_start + tl.arange(0, BLOCK_H)               # hidden index within tile
        dst_off  = base_offset[:, None] + offs_h[None, :] * hidden_stride   # (BLOCK_R, BLOCK_H)

        # guards for the tail of H
        valid_h = offs_h[None, :] < H

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & valid_h)
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & valid_h)
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & valid_h)

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M, K]

        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # LayerNorm → (out - μ) * rstd → γ· + β
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]

        # Apply out‑gate
        term = norm * gate_vals                                       # [M, K]

        # FP16→FP32 dot with final projection weight
        term_fp16 = term.to(tl.float16)                               # [M, K]

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, dim_out_chunk]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, N_out]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the model weights (float32)
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ fused LayerNorm + projection + gates (writes left/right/out_gate)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # concatenate the five projection weights and transpose for easy col‑wise access
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    )                               # shape (5*H, dim)
    fused_w_T = fused_w.t().contiguous().to(torch.float16)   # (dim, 5*H)

    # optional mask handling
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # launch fused kernel
    BLOCK_R = 64   # rows per program
    BLOCK_H = 32   # hidden‑dim tile
    BLOCK_D = 32   # input‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    fused_ln_proj_gate_kernel[grid](
        x.view(R, dim).contiguous(),
        left,
        right,
        out_gate,
        mask_ptr,
        w["norm.weight"],
        w["norm.bias"],
        fused_w_T,
        R,
        H,
        N,
        dim,
        eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del fused_w, fused_w_T, mask_flat if (has_mask) else None

    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over the hidden dimension (B·H, N, N) → out (B, H, N, N)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128     # hidden_dim ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
