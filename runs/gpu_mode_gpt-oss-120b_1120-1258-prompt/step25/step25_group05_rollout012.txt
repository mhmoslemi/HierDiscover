@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    out_proj_ptr,   # destination (R, H) FP16
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    hidden_stride = N * N
    stride_b = H * hidden_stride
    # decode
    b = offs_r // (N * N)
    rem = offs_r - b * (N * N)
    i = rem // N
    j = rem - i * N

    base = b * stride_b + i * N + j

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_val = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                          mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        gate_val = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                          mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)  # [BLOCK_R]
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)  # [BLOCK_R]

        # broadcast mu/rstd to hidden dimension
        mu_tile = mu[:, None]
        rstd_tile = rstd[:, None]

        # layernorm gamma/beta for hidden
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        # compute normalized output
        val = (out_val - mu_tile) * rstd_tile   # [BLOCK_R, BLOCK_H]
        val = val * gamma[None, :] + beta[None, :]
        val = val * gate_val

        # store to proj_in shape [R,H]
        dst_ptr = out_proj_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(dst_ptr, val.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr, out_proj_ptr,
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    hidden_stride = N * N
    stride_b = H * hidden_stride

    # decode
    b = offs_r // (N * N)
    rem = offs_r - b * (N * N)
    i = rem // N
    j = rem - i * N

    base = b * stride_b + i * N + j

    # load mu and rstd per row, broadcast across hidden
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)
    mu_tile = mu[:, None]
    rstd_tile = rstd[:, None]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate
        out_val = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                          mask=mask_r[:, None] & mask_h[None, :], other=0.0)   # [M,K]
        gate_val = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # gamma, beta per hidden
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        # Normalize
        val = (out_val - mu_tile) * rstd_tile
        val = val * gamma[None, :] + beta[None, :]
        val = val * gate_val

        # store to output projection buffer (R, H)
        tl.store(out_proj_ptr + offs_r[:, None] * H + offs_h[None, :], val.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (αFold3) – forward‑only Triton implementation

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (dim) of the flattened pair tensor.
    FP32 statistics → FP16 normalized tensor.

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (cuBLAS).  The result has shape [R,5·H] where
    R = B·N·N, H = hidden_dim.

3️⃣  Triton kernel ‑ split the fused projection, apply the optional pair mask,
    compute sigmoid gates and write three tensors
        left      : [B, H, N, N] (FP16)
        right     : [B, H, N, N] (FP16)
        out_gate  : [B, H, N, N] (FP16)

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    → tensor [B, H, N, N] (FP16).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
    (another Triton kernel).  Produces µ and rstd of shape [R] (FP32).

6️⃣  Triton kernel normalises `out`, applies the second LayerNorm
    (to_out_norm), multiplies by `out_gate` and writes a compact
    buffer `proj_in` of shape [R, H] (FP16).

7️⃣  Final linear projection is a single cuBLAS GEMM:
        proj_in @ to_out.weightᵀ  →  [R, dim] (FP16)
    Cast to FP32 and reshape to [B, N, N, dim].

All kernels target Triton 3.3.1, run on H100 and use mixed precision
(FP32 for statistics, FP16 for bulk work).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]     γ (FP32)
    b_ptr,   # [dim]     β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast on H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        m = m[:, None]                                          # [BLOCK_R, 1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start of hidden slice 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # project slice (5*H) → split
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Normalise, apply to_out_norm and out_gate → compact [R, H] buffer
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,       # [B, H, N, N] FP16
    gate_ptr,      # [B, H, N, N] FP16 (out_gate)
    mu_ptr,        # [R] FP32
    rstd_ptr,      # [R] FP32
    gamma_ptr,     # [H] FP32 (to_out_norm.weight)
    beta_ptr,      # [H] FP32 (to_out_norm.bias)
    proj_ptr,      # [R, H] FP16   (output buffer)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    hidden_stride = N * N
    stride_b = H * hidden_stride

    # decode (b,i,j) from flat row index
    b = offs_r // (N * N)
    rem = offs_r - b * (N * N)
    i = rem // N
    j = rem - i * N

    base = b * stride_b + i * N + j                     # start of hidden‑dim 0

    # load µ and rstd (broadcast over H)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    mu_tile   = mu[:, None]      # [BLOCK_R,1]
    rstd_tile = rstd[:, None]    # [BLOCK_R,1]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw hidden values and gate
        out_val  = tl.load(
            out_ptr  + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        gate_val = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

        # (out - µ) * rstd → γ· + β → * out_gate
        norm = (out_val - mu_tile) * rstd_tile           # [M,K]
        norm = norm * gamma[None, :] + beta[None, :]     # [M,K]
        out  = norm * gate_val                           # [M,K]

        # store compact row‑wise representation (R, H) – contiguous
        dst_ptr = proj_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(dst_ptr, out.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()              # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are [B, H, N, N] → reshape to [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)                    # FP16
    right_mat = right.view(B * H, N, N)                   # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                 # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Normalise + to_out_norm + out_gate → compact (R, H) buffer
    # ------------------------------------------------------------------
    proj_in = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_N = 128
    BLOCK_H_N = 64
    grid_norm_gate = (triton.cdiv(R, BLOCK_R_N),)
    norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        proj_in,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_N,
        BLOCK_H=BLOCK_H_N,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS) + reshape to [B, N, N, dim]
    # ------------------------------------------------------------------
    to_out_w_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final = F.linear(proj_in, to_out_w_fp16)               # [R, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)     # cast to FP32

    return final
