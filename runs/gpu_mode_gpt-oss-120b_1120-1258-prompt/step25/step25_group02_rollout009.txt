"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32) using cuDNN (`torch.nn.functional.layer_norm`);
   result is cast to FP16.

2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM (`torch.nn.functional.linear`) producing a
   tensor of shape (R, 5·H) where R = B·N·N and H = hidden_dim.

3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection,
   applies the optional mask, computes the three sigmoid gates and writes the
   three tensors `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).

4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).

5️⃣ Row‑wise mean & variance over the hidden dimension are computed with
   `torch.mean`/`torch.var` (highly‑optimized cuDNN kernels).  The
   rsqrt‑standard‑deviation `rstd = 1/√(var+eps)` is obtained in FP32.

6️⃣ Triton kernel `final_output_kernel` fuses the second LayerNorm,
   multiplication by `out_gate` and the final projection (`to_out`) into a
   single kernel.  It reads `out`, `out_gate`, per‑row statistics (`mu`,
   `rstd`), LayerNorm parameters (`to_out_norm.weight/bias`) and the
   transposed final weight (`to_out.weight.T`), producing the final
   `[B, N, N, dim]` tensor in FP32.

All heavy data‑movement kernels are written in mixed‑precision (FP16 for
activations, FP32 for statistics) and tuned for H100 (Triton 3.3.1) with
8 warps per program.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] FP16  (fused linear projection)
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                      # stride between two hidden planes
    # offset for hidden plane h = 0 at (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    # walk over hidden dimension in one tile (BLOCK_H == H)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # slice of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # load per‑row statistics (mean & rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]            # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M,K]
        term = norm * vals_gate                                   # [M,K]

        term_fp16 = term.to(tl.float16)   # cast back to FP16 for the dot

        # Load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N) (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"      (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N            # total pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (cuDNN) → FP16
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension; we use it on the
    # full 4‑D tensor and then cast to half‑precision.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # flatten to (R, dim) for a single GEMM
    x_flat = x_norm.reshape(R, dim).contiguous()   # [R, dim] FP16
    proj = F.linear(x_flat, fused_w)                # [R, 5·H] FP16
    del x_norm, x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128                       # rows per program
    BLOCK_H = H                         # handle full hidden dim in one iteration
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (core einsum)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)             # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                     # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (cuDNN)
    # ------------------------------------------------------------------
    # Statistics are computed in FP32 for numerical stability.
    out_f32 = out.float()
    mu = out_f32.mean(dim=1)                     # [B, N, N] FP32
    var = out_f32.var(dim=1, unbiased=False)    # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)                # [B, N, N] FP32
    mu_flat = mu.reshape(R)                      # [R] FP32
    rstd_flat = rstd.reshape(R)                  # [R] FP32
    del out_f32, mu, var, rstd

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128        # rows per program
    BLOCK_K_OUT = 32         # hidden‑dim chunk
    BLOCK_N_OUT = 128        # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # gamma (FP32)
        w["to_out_norm.bias"],     # beta  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # cleanup
    del out, out_gate, mu_flat, rstd_flat, w_out_T
    return final
