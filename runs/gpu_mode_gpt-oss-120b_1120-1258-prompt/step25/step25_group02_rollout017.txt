```
@triton.jit
def term_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    term_ptr,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < B * N * N

    # decode indices
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden dimension 0
    base_offset = b * (H * nn) + i * N + j   # hidden stride = N*N

    # load mu/rstd for rows
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # compute pointer offsets for hidden slice
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * nn,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M,K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * nn,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)  # [M,K]

        # LN parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)    # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]   # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]      # [M,K]
        term = norm * gate_vals                            # [M,K]

        # store term
        tl.store(term_ptr + offs_m[:, None] * H + offs_k[None, :],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

"""
Outgoing TriMul (AlphaFold3) – fused forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair representation.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single FP16 GEMM (R × dim) × (dim × 5·H) → (R × 5·H).
3️⃣ Split the fused projection, apply the optional mask, compute sigmoid gates and
   store `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16) with a Triton kernel.
4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   →  (B·H, N, N) FP16.
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over the hidden dimension) with a
   Triton kernel → μ, rstd of shape (R = B·N·N).
6️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the storage of the
   normalized values into a single Triton kernel producing a tensor
   `mid` of shape (R, H) (FP16) where each entry is
       mid = ( (out – μ)·rstd  * γ  + β ) * out_gate .
7️⃣ Final linear projection (H → dim) is performed by a single cuBLAS GEMM
   `mid @ to_out.weight.T` (FP16·FP16 → FP32).  The result is reshaped to
   [B, N, N, dim] and returned.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision
(FP16 for the bulk of the work, FP32 for numerically‑sensitive reductions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim]  input (FP32)
    y_ptr,       # [R, dim]  output (FP16)
    w_ptr,       # [dim]    weight γ (FP32)
    b_ptr,       # [dim]    bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)        # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] (FP16) fused projection
    left_ptr,       # [B, H, N, N] (FP16)
    right_ptr,      # [B, H, N, N] (FP16)
    out_gate_ptr,   # [B, H, N, N] (FP16)
    mask_ptr,       # [R] (FP16) optional mask (broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b, i, k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                       # [BLOCK_R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability, back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] (FP16)
    mu_ptr,    # [R] (FP32)
    rstd_ptr,  # [R] (FP32)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # offset of the hidden‑0 slice for each (b,i,j)
    base = b * (H * nn) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * nn
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and write the normalized tensor
# ----------------------------------------------------------------------
@triton.jit
def term_kernel(
    out_ptr,               # [B, H, N, N] (FP16)
    gate_ptr,              # [B, H, N, N] (FP16)   ← out_gate
    mu_ptr,                # [R] (FP32)
    rstd_ptr,              # [R] (FP32)
    ln_w_ptr,              # [H] (FP32) – to_out_norm.weight
    ln_b_ptr,              # [H] (FP32) – to_out_norm.bias
    term_ptr,              # [R, H] (FP16)   output of this kernel
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < B * N * N

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden‑0 slice
    base = b * (H * nn) + i * N + j

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    # iterate over hidden dimension in one block (BLOCK_K == H for our configs)
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw out & out_gate values
        out_vals = tl.load(
            out_ptr + base[:, None] + offs_k[None, :] * nn,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        gate_vals = tl.load(
            gate_ptr + base[:, None] + offs_k[None, :] * nn,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # LN → (out – μ)·rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]           # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        term = norm * gate_vals                                    # [M, K]

        # store FP16 term
        tl.store(
            term_ptr + offs_m[:, None] * H + offs_k[None, :],
            term.to(tl.float16),
            mask=mask_m[:, None] & mask_k[None, :],
        )

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor  : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor   : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : dict with the keys used in the reference module
            - config_dict   : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM)
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for torch.bmm
    left_mat = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)               # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rstd (over hidden dim)
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64          # H = 128 → two iterations, still cheap
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Normalise with second LayerNorm, apply out_gate → (R, H) tensor
    # ------------------------------------------------------------------
    term = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_TERM = 128
    BLOCK_K_TERM = 128          # whole hidden dim in one iteration (H=128)
    grid_term = (triton.cdiv(R, BLOCK_M_TERM),)
    term_kernel[grid_term](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        term,
        B,
        N,
        H,
        BLOCK_M=BLOCK_M_TERM,
        BLOCK_K=BLOCK_K_TERM,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (H → dim) – cuBLAS GEMM
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]
    # torch.matmul promotes to FP32 accumulation and returns FP32
    final = torch.matmul(term.float(), w_out_T.float())               # [R, dim] fp32
    final = final.view(B, N, N, dim)                                 # [B, N, N, dim]

    return final
