"""
Outgoing TriMul (AlphaFold‑3) – simplified but correct implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16) using a
   Triton kernel.  The kernel computes the mean and variance per row, applies the
   learned scale (γ) and bias (β) and writes the normalized values in FP16.

2️⃣ Five independent linear projections are performed in FP16 with the model weights:
   * left_proj   : dim → hidden_dim
   * right_proj  : dim → hidden_dim
   * left_gate   : dim → hidden_dim
   * right_gate  : dim → hidden_dim
   * out_gate    : dim → hidden_dim
   The (optional) pairwise mask is applied to the left/right projections before
   the sigmoid gates.

3️⃣ The core “triangle” operation is a batched GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   (implemented with a single torch.bmm on the view (B·H, N, N)).

4️⃣ A second LayerNorm (over the hidden dimension) is applied, the out‑gate is
   multiplied in, and a final linear projection (hidden_dim → dim) produces the
   result.  These steps are done with PyTorch (FP32) to keep the implementation
   straightforward; only the first LayerNorm must be a Triton kernel.

The function `custom_kernel` follows the exact interface required by the
evaluation harness:
    data = (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
and returns a tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, D]   FP32 input (flattened pair tensor)
    y_ptr,          # [R, D]   FP16 output
    w_ptr,          # [D]      γ (FP32)
    b_ptr,          # [D]      β (FP32)
    rows, dim, eps,                    # scalar args
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Pass 1: compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2: write normalized, scaled, biased values (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# --------------------------------------------------------------
# Entry‑point: forward pass of the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
                Keys:
                    "norm.weight", "norm.bias",
                    "left_proj.weight", "right_proj.weight",
                    "left_gate.weight", "right_gate.weight", "out_gate.weight",
                    "to_out_norm.weight", "to_out_norm.bias",
                    "to_out.weight"
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dtype = x.dtype

    dim        = cfg["dim"]          # model dimension (e.g. 128 or 384)
    hidden_dim = cfg["hidden_dim"]   # hidden dimension (e.g. 128)
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise positions

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    # flatten to [R, dim] (contiguous)
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # dim tiles
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ  (FP32)
        w["norm.bias"],            # β  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory early

    # ------------------------------------------------------------------
    # 2️⃣ Five independent linear projections (all FP16)
    # ------------------------------------------------------------------
    # Cast all required weights to FP16 once
    left_proj_w   = w["left_proj.weight"].to(torch.float16)
    right_proj_w  = w["right_proj.weight"].to(torch.float16)
    left_gate_w   = w["left_gate.weight"].to(torch.float16)
    right_gate_w  = w["right_gate.weight"].to(torch.float16)
    out_gate_w    = w["out_gate.weight"].to(torch.float16)

    # Linear maps (shape: [R, hidden_dim])
    left      = F.linear(x_norm, left_proj_w)    # → [R, H]
    right     = F.linear(x_norm, right_proj_w)   # → [R, H]
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate= torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate  = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # reshape back to [B, N, N, H]
    left      = left.view(B, N, N, hidden_dim)
    right     = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate= right_gate.view(B, N, N, hidden_dim)
    out_gate  = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16
    # apply sigmoid gates
    left  = left * left_gate
    right = right * right_gate
    # out_gate stays as‑is for later use

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (triangular multiplication) via torch.bmm
    # ------------------------------------------------------------------
    # Move hidden dimension to axis 1 for a single 3‑D view
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # merge batch and hidden dims → [B·H, N, N]
    left_mat  = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    # core einsum: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    # Compute layer‑norm in FP32 for numerical stability
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],   # γ (FP32)
        bias=w["to_out_norm.bias"],       # β (FP32)
        eps=eps,
    )

    # multiply by the out‑gate (cast gate to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to the model dimensionality
    # ------------------------------------------------------------------
    to_out_w = w["to_out.weight"]          # shape [dim, hidden_dim], FP32
    final = F.linear(out_norm, to_out_w)   # → [B, N, N, dim] (FP32)

    return final
