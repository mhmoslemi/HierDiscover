@triton.jit
def apply_norm_gate_kernel(
    out_ptr,         # [B, H, N, N] FP16
    gate_ptr,        # [B, H, N, N] FP16
    mu_ptr,          # [R] FP32
    rstd_ptr,        # [R] FP32
    ln_w_ptr,        # [H] FP32
    ln_b_ptr,        # [H] FP32
    out_buf_ptr,     # [R, H] FP16
    R, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    hidden_stride = N * N  # stride between hidden layers

    # Compute b, i, j
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # Base offset for a given hidden plane
    base = b * (H * hidden_stride) + i * N + j

    # Load mu and rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)           # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)       # [M]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out and gate for this hidden slice
        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # [M, K]
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)  # [M, K]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # mu and rstd need to be broadcasted across hidden dim
        mu_broadcast = mu[:, None]
        rstd_broadcast = rstd[:, None]

        # Compute normalized value
        # ((out - mu) * rstd) * gamma + beta
        norm = ((out_vals - mu_broadcast) * rstd_broadcast)
        # apply gamma and beta
        norm = norm * gamma[None, :] + beta[None, :]

        # apply gate
        gated = norm * gate_vals

        # store into output buffer (fp16)
        dst_offset = offs_m[:, None] * H + offs_h[None, :]
        tl.store(out_buf_ptr + dst_offset,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – optimized Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM (cuBLAS) producing a tensor of shape (R, 5·H) where
   R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply the optional pair‑wise mask, compute the
   sigmoid gates and store the three tensors *left*, *right* and *out_gate*
   in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, k, j]
   producing a tensor `[B, H, N, N]` in FP16.
5️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
   (LayerNorm statistics) using a Triton kernel.
6️⃣ Triton kernel normalises `out` with the statistics, applies the second
   LayerNorm (γ,β) and the `out_gate`. The result is written to a compact
   buffer `[R, H]` (FP16).
7️⃣ Final projection to `dim` is performed by a single cuBLAS GEMM:
       out_final = (normed_gated) @ W_outᵀ
   which yields the required output shape `[B, N, N, dim]` (float32).

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Mixed‑precision (FP16 bulk, FP32 reductions) is used throughout, and the
expensive final projection is delegated to cuBLAS for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim]   FP32 input
    y_ptr,      # [R, dim]   FP16 output
    w_ptr,      # [dim]      LN weight γ (FP32)
    b_ptr,      # [dim]      LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance per row ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)   # [BLOCK_R]
        mask_val = mask_val[:, None]                     # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                         # distance between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)      # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Apply 2nd LayerNorm + out_gate, write compact [R, H] buffer
# ----------------------------------------------------------------------
@triton.jit
def apply_norm_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16 (raw result of left·rightᵀ)
    gate_ptr,         # [B, H, N, N] FP16 (out_gate)
    mu_ptr,           # [R]          FP32
    rstd_ptr,         # [R]          FP32
    ln_w_ptr,         # [H]          FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]          FP32 (to_out_norm.bias)
    out_buf_ptr,      # [R, H]       FP16 (normalized × gate)
    R, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    hidden_stride = N * N
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base = b * (H * hidden_stride) + i * N + j      # start of hidden‑plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out and gate slices (FP16 → FP32 for arithmetic)
        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)          # [M, K]

        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [K]

        # broadcast mu / rstd across hidden dim
        mu_vec   = mu[:, None]
        rstd_vec = rstd[:, None]

        # (out - μ) * rstd → γ· + β → * gate
        norm = ((out_vals - mu_vec) * rstd_vec) * gamma[None, :] + beta[None, :]
        gated = norm * gate_vals

        # write to compact buffer [R, H] (row‑major)
        dst_offset = offs_m[:, None] * H + offs_h[None, :]   # linear offset into [R,H]
        tl.store(out_buf_ptr + dst_offset,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            (all tensors are float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pair positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                     # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right have shape [B, H, N, N]; view as [B·H, N, N]
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]

    # out = left @ rightᵀ   (einsum over the third axis)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply 2nd LayerNorm + out_gate and write compact buffer
    # --------------------------------------------------------------
    out_norm_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_APPLY = 64
    BLOCK_H_APPLY = 64
    grid_apply = (triton.cdiv(R, BLOCK_M_APPLY),)
    apply_norm_gate_kernel[grid_apply](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        out_norm_gated,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_APPLY,
        BLOCK_H=BLOCK_H_APPLY,
        num_warps=8,
    )
    # Release memory that is no longer needed
    del out, out_gate, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final projection to `dim` via cuBLAS (FP16 GEMM)
    # --------------------------------------------------------------
    # Weight matrix for the final linear layer: shape [hidden_dim, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    # GEMM: (R, H) @ (H, dim) → (R, dim)
    final_fp16 = torch.matmul(out_norm_gated, w_out_T)                # FP16

    # reshape and cast back to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
