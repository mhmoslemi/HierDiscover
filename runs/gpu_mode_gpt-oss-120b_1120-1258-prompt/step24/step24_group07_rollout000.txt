"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and an optional
pair‑wise mask M ∈ {0,1}^{B×N×N}.  The computation consists of

1️⃣  Row‑wise LayerNorm over the feature dimension C.
2️⃣  A *fused* linear projection that simultaneously produces the five
    weight matrices (left, right, left‑gate, right‑gate, out‑gate) of
    shape (hidden_dim, C).  This is a single GEMM in FP16.
3️⃣  Split the fused projection, apply the optional mask, and evaluate the
    sigmoid gates (still in FP16).  The three tensors `left`, `right`
    and `out_gate` are written in layout [B, H, N, N].
4️⃣  Batched matrix‑multiply over the hidden dimension:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    realised with a single `torch.bmm` on (B·H, N, N) matrices.
5️⃣  Row‑wise mean / rsqrt‑variance of `out` (needed for the second
    LayerNorm).  This reduction is performed by a Triton kernel.
6️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    projection (hidden_dim → C) into one Triton kernel that writes the
    final result as float32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Mixed‑precision (fp16 for the bulk of the work, fp32 for reductions) is
used to keep numerical fidelity while maximising throughput.
"""

import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C] float32 input
    y_ptr,          # [R, C] float16 output
    w_ptr,          # [C]   LayerNorm γ (float32)
    b_ptr,          # [C]   LayerNorm β (float32)
    rows, dim, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # --- first pass: compute mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(dim, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_c[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --- second pass: write normalized values -------------------------------
    for c_off in range(0, tl.cdiv(dim, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_c[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_r[:, None] * dim + offs_c[None, :],
            out.to(tl.float16),
            mask=mask_r[:, None] & mask_c[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑position mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]   # broadcast over hidden dim
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # layout: [left | right | left_gate | right_gate | out_gate]
    hidden_stride = N * N                     # stride between successive hidden‑planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start address for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # [R, H] view of the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise statistics of the contracted tensor (mean & rsqrt‑variance)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] fp16
    mu_ptr,       # [R] fp32
    rstd_ptr,     # [R] fp32
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of h=0 for each (b,i,j)

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] fp16
    gate_ptr,         # [B, H, N, N] fp16   (out_gate)
    mu_ptr,           # [R] fp32
    rstd_ptr,         # [R] fp32
    ln_w_ptr,         # [H] fp32  (to_out_norm.weight)
    ln_b_ptr,         # [H] fp32  (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] fp16  (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)   # rows (R)
    pid_d = tl.program_id(1)   # output‑dim tile

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_d = dim_start + tl.arange(0, BLOCK_D)
    mask_r = offs_r < total_rows
    mask_d = offs_d < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of h=0

    # load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [R]

    # accumulator for the final dot‑product (FP32)
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # hidden‑plane values (FP16 → FP32)
        ptr_out = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(ptr_out,
                           mask=mask_r[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)    # [R, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [R, K]

        # LayerNorm parameters for this hidden‑chunk
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        norm   = (out_vals - mu[:, None]) * rstd[:, None]          # [R, K]
        norm   = norm * gamma[None, :] + beta[None, :]
        gated  = norm * gate_vals                                   # [R, K]

        # cast back to fp16 for the GEMM with the final projection weight
        gated_fp16 = gated.to(tl.float16)   # [R, K]

        # load a tile of the final weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_d[None, :],
                         other=0.0)                                   # [K, D]

        # accumulate dot product (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)           # [R, D]

    # write final tensor [B, N, N, dim] (float32)
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_r[:, None] & mask_d[None, :])

    BLOCK_C = 64

        x_norm,
        w["norm.weight"],
        w["norm.bias"],

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,

    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_H_SPLIT = 64

        right,

        R,
        H,

    right_mat = right.view(B * H, N, N)               # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))

        triton.cdiv(dim, BLOCK_D_OUT),
