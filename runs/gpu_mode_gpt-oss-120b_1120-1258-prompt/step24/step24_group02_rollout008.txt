"""
Outgoing TriMul (AlphaFold3) – highly‑fused forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ A single fused Linear (5·H outputs) applied to the normalized tensor.
3️⃣ Triton kernel `split_fuse_gate`:
   – decodes the flat (batch, i, k) index,
   – optional pair‑wise mask,
   – applies sigmoid gates,
   – writes the three tensors  left, right and out_gate  with layout [B, H, N, N] (FP16).
4️⃣ Batched matrix‑multiply (cuBLAS) computes the core einsum:
   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]   (FP16).
5️⃣ Triton kernel `rowwise_stats` computes μ and 1/√σ² per output position
   (i.e. across the hidden dimension H).
6️⃣ Triton kernel `final_output` fuses:
   – the second LayerNorm,
   – element‑wise multiplication by out_gate,
   – the final linear projection to the original channel dimension.
   The result is written as a FP32 tensor [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target NVIDIA H100.  Mixed‑precision
(FP16 for the bulk of the work, FP32 for reductions) maximises throughput while
respecting the required output dtype (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,     # [R, C]  FP32 input
    y_ptr,     # [R, C]  FP16 output
    w_ptr,     # [C]    γ (FP32)
    b_ptr,     # [C]    β (FP32)
    rows, C, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized FP16 ----------
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # FP32
        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        out = norm * gamma[None, :] + beta[None, :]               # FP32
        tl.store(
            y_ptr + offs_r[:, None] * C + offs_c[None, :],
            out.to(tl.float16),
            mask=mask_r[:, None] & mask_c[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional scalar mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16 scalar
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride within the hidden dimension (N·N per head)
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # 5 slices inside the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [R, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16 (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16 (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_r = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_r * BLOCK_R
    col_start = pid_n * BLOCK_N

    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_r = offs_r < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load μ and 1/√σ² for each row
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [R]

    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals = tl.load(
            ptr_out,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [R, K]

        gate_vals = tl.load(
            ptr_gate,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [R, K]

        # LayerNorm parameters (γ,β) for this head‑dim slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # Normalisation → γ· + β
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [R, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [R, K]

        # Apply out_gate
        term = norm * gate_vals                                      # [R, K]
        term_fp16 = term.to(tl.float16)

        # Load slice of final projection weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # [K, TILE_N]

        # Accumulate dot product (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Write final tensor
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_r[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"                     (all float32)
            - config_dict  : dict with "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_C = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # [5·H, dim]
    proj = F.linear(x_norm, fused_w)              # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    HAS_MASK = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R_S = 64
    BLOCK_H_S = 64
    grid_split = (triton.cdiv(R, BLOCK_R_S),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R_S,
        BLOCK_H=BLOCK_H_S,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 128      # hidden_dim ≤ 128 → single tile
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128
    BLOCK_K_OUT = 32      # hidden‑dim chunk (128 → 4 steps)
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
