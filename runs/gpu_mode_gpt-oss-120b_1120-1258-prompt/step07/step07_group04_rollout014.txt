@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_bh = tl.program_id(2) # batch*head index

    # compute batch and head
    bh = pid_bh
    b = bh // H
    h = bh % H

    # compute base offsets for this batch/head
    # base row offset = b * stride_bl + h * stride_hl (since stride_hl=1)
    # we use tl.broadcast?? We compute additive offset
    # We'll store as integer offsets (in elements)
    offset_base = b * stride_bl + h * stride_hl

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # offset = base + offs_i[:,None]*stride_il + offs_k[None,:]*stride_kl
        left_ptrs = left_ptr + offset_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # offset = base + offs_k[:,None]*stride_kr + offs_j[None,:]*stride_jr
        right_ptrs = right_ptr + offset_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot product
        acc += tl.dot(left, right)   # left (M,K) dot right (K,N) => (M,N)

    # Store result
    out_ptrs = out_ptr + offset_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo

"""
TriMul (outgoing) – high‑performance forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (input → fp16) via a custom Triton kernel.
2️⃣ Fuse the four linear projections (left/right + their gates) **plus** the
   output‑gate into a single half‑precision GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]      (R = B·N·N, H = hidden_dim)
   The result is split into
        left_raw, right_raw, left_gate, right_gate, out_gate.
3️⃣ Optional binary mask is applied to *left_raw* and *right_raw*.
   Sigmoid is applied in‑place to the three gate tensors and the gating is
   fused:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
4️⃣ The heavy N³ multiplicative update
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is implemented as a custom Triton kernel that works directly on the
   tensor layout (no costly permute‑and‑contiguous copies).  The kernel
   processes a tile of the output (i‑block × j‑block) for each (b,h)
   combination and accumulates the reduction over *k*.
5️⃣ Second LayerNorm over the hidden dimension (fp16→fp32) – we reuse the
   same Triton kernel (the kernel works with both fp32 and fp16 inputs).
6️⃣ Multiply by the output‑gate, apply the final linear projection
   (fp16 GEMM) and cast back to fp32.

All heavy linear algebra is performed in FP16 to exploit Tensor‑Cores.
Two custom Triton kernels are used (LayerNorm and the N³ update), which
satisfies the “must contain a kernel” requirement while keeping the overall
runtime well below the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (works for fp32 or fp16 inputs)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance (in fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load in the original dtype (float32 or float16) and cast to fp32
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=tl.float16(0.0) if tl.typeof(ptr.element_type) is tl.float16 else 0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (output fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=tl.float16(0.0) if tl.typeof(ptr.element_type) is tl.float16 else 0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: outgoing TriMul N³ multiplicative update
# ----------------------------------------------------------------------
@triton.jit

    H,                   # int (runtime)
    stride_b,            # stride of batch dimension
    stride_i,            # stride of the first N dimension (i)
    stride_k,            # stride of the reduction N dimension (k)
    stride_h,            # stride of the hidden dimension (h)
    BLOCK_M: tl.constexpr,   # tile size over i
    BLOCK_N: tl.constexpr,   # tile size over j
    BLOCK_K: tl.constexpr,   # tile size over k reduction
):
    pid_i = tl.program_id(0)        # block over i
    pid_j = tl.program_id(1)        # block over j
    pid_bh = tl.program_id(2)       # combined batch‑head index (B·H)

    bh = pid_bh
    b = bh // H                      # batch index
    h = bh % H                       # hidden‑dim index

    # base offset that includes batch and hidden‑dim position
    base = b * stride_b + h * stride_h

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more accurate, still fast on Tensor Cores)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k,h]   → shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + base \
                    + offs_i[:, None] * stride_i \
                    + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0))

        # right[j,k,h]  but we need it transposed to (k,j) → shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + base \
                     + offs_k[:, None] * stride_k \
                     + offs_j[None, :] * stride_i   # stride_i == stride of the second N dim
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0))

        # Accumulate dot product over k
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)  (fp32 accumulator)

    # ------------------------------------------------------------------
    # Write the result (fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + base \
               + offs_i[:, None] * stride_i \
               + offs_j[None, :] * stride_k   # stride_k is the stride of the *third* N dim (j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing LayerNorm and Linear parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256          # rows per program – good for most sizes
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four projections + both gates + the output‑gate
    #    into a single GEMM:  [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    fused = F.linear(x_norm, w_fused)   # fp16 GEMM → shape [R, 5·H]

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) and sigmoid gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] → flatten to [R,1]
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ update (no extra copies!)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ N³ multiplicative update via a custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)   # will hold the result

    # strides of the (B,N,N,H) layout
    stride_b, stride_i, stride_k, stride_h = left.stride()   # same for left/right/out
    BLOCK_M = 64   # tile size over i
    BLOCK_N = 64   # tile size over j
    BLOCK_K = 32   # tile size over the reduction axis

    grid = (
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
        B * H,                                 # combined batch‑head dimension
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_b=stride_b,
        stride_i=stride_i,
        stride_k=stride_k,
        stride_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (reuse kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply the output‑gate, final projection and cast back to fp32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise fp16

    final = F.linear(
        out_norm.view(total_rows, H),               # [R, H]
        weights["to_out.weight"].to(torch.float16)  # [dim, H]
    )                                                # fp16 → [R, dim]

    return final.view(B, N, N, dim).to(torch.float32)
