"""
fast_tri_mul.py
----------------
Implementation of the *outgoing* TriMul operator (AlphaFold‑3) with
major compute kernels written in Triton.

Algorithm
~~~~~~~~~
1. **First LayerNorm** – row‑wise (R = B·N·N) LayerNorm over the
   feature dimension *dim*.  Implemented in Triton, output is fp16.
2. **Fused projection + gates** – a single matmul
   ``X_norm (R,dim) @ W.T (dim,5·H)`` (H = hidden_dim) produces the
   concatenated tensors ``[left_proj, right_proj, left_gate,
   right_gate, out_gate]``.
3. **Mask & sigmoid** – a custom Triton kernel reads the concatenated
   tensor and produces three fp16 tensors:
   ``left = left_proj * sigmoid(left_gate) * mask``,
   ``right = right_proj * sigmoid(right_gate) * mask``,
   ``out_gate = sigmoid(out_gate_raw)``.
4. **Multiplicative update** – reshape ``left``/``right`` to
   ``(B, H, N, N)`` and perform a batched GEMM
   ``out = left @ right.transpose(-2,-1)`` (fp16).  This is the
   N³ kernel (B·H matmuls of size N×N) and is executed by cuBLAS.
5. **Second LayerNorm** – row‑wise LayerNorm over the hidden
   dimension H (again Triton, fp16).
6. **Output gating & projection** – ``out_norm = out_norm * out_gate``
   followed by a final linear projection back to ``dim`` (fp16 matmul
   → fp32 output).

Only the forward pass is needed; all tensors are kept in half
precision except the final result.  The kernels are deliberately kept
simple to allow the H100 Tensor‑Core units to run at peak throughput.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ================================================================
# 1. Row‑wise LayerNorm (fp32 → fp16)
# ================================================================
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const  * [rows, dim] (float32)
    y_ptr,          # mutable* [rows, dim] (float16)
    weight_ptr,     # const  * [dim]      (float32) – gamma
    bias_ptr,       # const  * [dim]      (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- 1️⃣ mean / var -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- 2️⃣ normalize & apply gamma / beta -----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """Row‑wise LayerNorm on a 2‑D tensor [rows, dim] (float32) → fp16."""
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ================================================================
# 2. Gate + mask kernel (splits fused projection)
# ================================================================
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # const  * [rows, 5*H] (fp16)
    left_ptr,          # mutable* [rows, H] (fp16)
    right_ptr,         # mutable* [rows, H] (fp16)
    out_gate_ptr,      # mutable* [rows, H] (fp16)
    mask_ptr,          # const  * [rows] (fp16)
    rows,              # int32 (runtime)
    H,                 # int32 (runtime)
    stride_proj,       # int32 (runtime) = 5*H
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden channels per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # broadcast mask (0/1) – already fp16
    mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]

    # ------------------------------------------------------------------
    # iterate over hidden dimension in blocks
    # ------------------------------------------------------------------
    for off_h in range(0, H, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)                # [BLOCK_H]
        cur_h_mask = cur_h < H
        # -------- left -------------------------------------------------
        # left_proj_raw  : offset 0*H
        left_proj = tl.load(
            proj_ptr
            + offs_r[:, None] * stride_proj
            + (cur_h[None, :] + 0).to(tl.int32),
            mask=mask_r[:, None] & cur_h_mask[None, :],
            other=0.0,
        )
        # left_gate_raw  : offset 2*H
        left_gate = tl.load(
            proj_ptr
            + offs_r[:, None] * stride_proj
            + (cur_h[None, :] + 2 * H).to(tl.int32),
            mask=mask_r[:, None] & cur_h_mask[None, :],
            other=0.0,
        )
        # sigmoid on gate (float32 for stability)
        left_gate_f = left_gate.to(tl.float32)
        left_sig = 1.0 / (1.0 + tl.exp(-left_gate_f))
        left = (left_proj.to(tl.float32) * left_sig).to(tl.float16)
        # apply mask
        left = left * mask_val[:, None]

        tl.store(
            left_ptr + offs_r[:, None] * H + cur_h[None, :],
            left,
            mask=mask_r[:, None] & cur_h_mask[None, :],
        )

        # -------- right ------------------------------------------------
        # right_proj_raw : offset 1*H
        right_proj = tl.load(
            proj_ptr
            + offs_r[:, None] * stride_proj
            + (cur_h[None, :] + H).to(tl.int32),
            mask=mask_r[:, None] & cur_h_mask[None, :],
            other=0.0,
        )
        # right_gate_raw : offset 3*H
        right_gate = tl.load(
            proj_ptr
            + offs_r[:, None] * stride_proj
            + (cur_h[None, :] + 3 * H).to(tl.int32),
            mask=mask_r[:, None] & cur_h_mask[None, :],
            other=0.0,
        )
        right_gate_f = right_gate.to(tl.float32)
        right_sig = 1.0 / (1.0 + tl.exp(-right_gate_f))
        right = (right_proj.to(tl.float32) * right_sig).to(tl.float16)
        right = right * mask_val[:, None]

        tl.store(
            right_ptr + offs_r[:, None] * H + cur_h[None, :],
            right,
            mask=mask_r[:, None] & cur_h_mask[None, :],
        )

        # -------- out_gate --------------------------------------------
        out_raw = tl.load(
            proj_ptr
            + offs_r[:, None] * stride_proj
            + (cur_h[None, :] + 4 * H).to(tl.int32),
            mask=mask_r[:, None] & cur_h_mask[None, :],
            other=0.0,
        )
        out_raw_f = out_raw.to(tl.float32)
        out_sig = 1.0 / (1.0 + tl.exp(-out_raw_f))
        tl.store(
            out_gate_ptr + offs_r[:, None] * H + cur_h[None, :],
            out_sig.to(tl.float16),
            mask=mask_r[:, None] & cur_h_mask[None, :],
        )


def _gate_mask_triton(proj: torch.Tensor,
                      mask: torch.Tensor,
                      hidden_dim: int,
                      block_r: int = 256,
                      block_h: int = 32) -> tuple:
    """
    Split the concatenated projection tensor into gated ``left``, ``right``
    and ``out_gate`` tensors, applying the optional mask.
    All tensors are fp16.
    """
    rows = proj.shape[0]
    H = hidden_dim
    left = torch.empty(rows, H, dtype=torch.float16, device=proj.device)
    right = torch.empty(rows, H, dtype=torch.float16, device=proj.device)
    out_gate = torch.empty(rows, H, dtype=torch.float16, device=proj.device)

    stride_proj = proj.stride(0)  # because proj is contiguous, this == 5*H
    grid = (triton.cdiv(rows, block_r),)
    _gate_mask_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask,
        rows,
        H,
        stride_proj,
        BLOCK_R=block_r,
        BLOCK_H=block_h,
        num_warps=4,
    )
    return left, right, out_gate


# ================================================================
# 3. Custom forward entry point
# ================================================================
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()        # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )                                                       # fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·H outputs) – single GEMM
    # --------------------------------------------------------------
    # concatenate weight matrices (all are [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    proj = F.linear(x_norm, w_fused)                       # fp16, shape [R, 5·H]

    # --------------------------------------------------------------
    # 3️⃣ Mask & sigmoids (fused Triton kernel)
    # --------------------------------------------------------------
    if nomask:
        mask_flat = torch.ones(R, dtype=torch.float16, device=device)
    else:
        mask_flat = mask_tensor.to(torch.float16).reshape(R)

    left, right, out_gate = _gate_mask_triton(proj, mask_flat, hidden_dim)

    # free the large temporary
    del proj
    torch.cuda.empty_cache()

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # bring hidden dim to axis‑1 so we can use a single batched GEMM
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (fp16)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )                                                   # [B, H, N, N]

    out = out_bh.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()              # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                                    # fp16

    out_norm = out_norm.view(B, N, N, H)                # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply out‑gate then final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                      # [B, N, N, H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
