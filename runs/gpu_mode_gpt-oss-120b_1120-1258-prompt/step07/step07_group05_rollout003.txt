"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) with a custom Triton kernel.
2️⃣ Fuse **all** linear projections/gates (left, right, left‑gate,
   right‑gate, out‑gate) into a *single* half‑precision GEMM.
3️⃣ Apply the (optional) mask, sigmoid the three gates and multiply
   the gated values.
4️⃣ Reshape → (B·H, N, N) and perform a batched fp16 matrix‑multiply
   (`torch.bmm`) which uses H100 Tensor‑Cores.
5️⃣ Second LayerNorm over the hidden dimension (fp16) – same Triton
   kernel.
6️⃣ Apply the out‑gate, a final half‑precision linear projection and
   cast the result back to fp32.

All heavy element‑wise work (layer‑norm, mask, sigmoid) is done in
Triton; the dense N³ updates are delegated to cuBLAS via `torch.bmm`,
which is the fastest path on H100.  The implementation supports
any combination of `dim ∈ {128,384}`, `hidden_dim = 128`,
batch size ∈ {1,2} and sequence length up to 1024, with or without a
mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # i32 (runtime)
    dim,            # i32 (runtime)
    eps,            # f32 (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows                # guard rows out of bounds

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over columns in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)    # fp32 load

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                       # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β (float32) – broadcast across rows
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # scale
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # bias

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    nomask = cfg.get("nomask", True)        # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                  # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # launch Triton kernel
    BLOCK_M = 256            # rows per program (tunable)
    BLOCK_D = 64             # columns per iteration (tunable)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear transforms (left, right, left‑gate,
    #    right‑gate, out‑gate) into a single GEMM.
    # ------------------------------------------------------------------
    # concatenate weight matrices: each of shape [H, dim] -> [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [5·H, dim]

    # fused linear (no bias) → [R, 5·H]
    fused_all = F.linear(ln1_out, w_fused)     # fp16

    # split the 5 blocks
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = fused_all.split(H, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gated multiplication
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # sigmoid (in‑place)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # gated values
    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring head dim to batch axis → [B, H, N, N] (contiguous)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # collapse batch & heads → [B·H, N, N]
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ   → fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # restore layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply out‑gate, final linear projection, cast to fp32
    # ------------------------------------------------------------------
    # reshape back to [B, N, N, H] (fp16)
    out_norm = ln2_out.view(B, N, N, H)

    # element‑wise gating
    out_norm = out_norm * out_gate.view(B, N, N, H)

    # final projection: [R, H] → [R, dim] (fp16)
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # fp16

    # cast back to fp32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
