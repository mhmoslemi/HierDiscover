def layernorm_gate_linear_kernel(
    x_ptr,            # const * [total_rows, hidden_dim] (float16)
    gate_ptr,         # const * [total_rows, hidden_dim] (float16)
    ln_weight_ptr,    # const * [hidden_dim] (float32)
    ln_bias_ptr,      # const * [hidden_dim] (float32)
    out_weight_ptr,   # const * [out_dim, hidden_dim] (float16)
    out_ptr,          # mutable * [total_rows, out_dim] (float32)
    total_rows, hidden_dim, out_dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    # sum over hidden dim
    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        x_ptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Prepare output accumulator
    out_acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    # loop over hidden dim blocks
    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # load x values
        x_ptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)  # fp32

        # normalize: (x - mean) * rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # apply LN weight and bias (γ, β)
        gamma = tl.load(ln_weight_ptr + offs_h,
                        mask=mask_h, other=0.0)  # fp32
        beta = tl.load(ln_bias_ptr + offs_h,
                        mask=mask_h, other=0.0)   # fp32
        norm = norm * gamma[None, :] + beta[None, :]  # fp32

        # apply gate
        gate_ptrs = gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        gate_vals = tl.load(gate_ptrs,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)
        norm = norm * gate_vals  # fp32

        # loop over output dim blocks
        for out_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
            offs_o = out_off * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_o = offs_o < out_dim
            # load weight slice: shape [BLOCK_H, BLOCK_O] (H x D)
            # pointer arithmetic for transposed weight:
            # weight[d, h] -> pointer = out_weight_ptr + d*hidden_dim + offs_h
            # we need to gather block for all h and d.
            weight_ptrs = out_weight_ptr + offs_h[:, None] + offs_o[None, :] * hidden_dim
            # shape (BLOCK_H, BLOCK_O)
            w = tl.load(weight_ptrs,
                        mask=mask_h[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)  # fp32
            # dot product
            out_acc += tl.dot(norm, w)  # (BLOCK_M, BLOCK_O)
    # store out
    out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(out_ptrs,
             out_acc,
             mask=mask_m[:, None] & mask_o[None, :])
```

def layernorm_proj_kernel(x_ptr, out_ptr, ln_weight_ptr, ln_bias_ptr, proj_weight_ptr,
    total_rows, dim, proj_out, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sum, sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over d
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # For each output block
    for out_off in range(0, tl.cdiv(proj_out, BLOCK_O)):
        offs_o = out_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < proj_out

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        # iterate over d blocks again to compute dot contributions
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x block
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_m[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)  # fp32

            # normalize x
            norm = (x_vals - mean[:, None]) * rstd[:, None]

            # apply LN weight and bias (γ,β) per input dim
            gamma = tl.load(ln_weight_ptr + offs_d,
                            mask=mask_d, other=0.0)
            beta  = tl.load(ln_bias_ptr + offs_d,
                            mask=mask_d, other=0.0)

            norm = norm * gamma[None, :] + beta[None, :]   # fp32

            # load proj weight slice: shape (BLOCK_D, BLOCK_O) (dim x proj_out)
            # transposed: we need weight_T: [dim, proj_out]
            # pointer for each (d_idx, o_idx) = proj_weight_ptr + d_idx + o_idx*dim
            weight_ptrs = proj_weight_ptr + offs_d[:, None] + offs_o[None, :] * dim
            w = tl.load(weight_ptrs,
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)   # fp32

            # dot product: acc += norm (BLOCK_M, BLOCK_D) @ w (BLOCK_D, BLOCK_O)
            acc += tl.dot(norm, w)

        # store acc to out_ptr
        out_ptrs = out_ptr + offs_m[:, None] * proj_out + offs_o[None, :]
        # convert to fp16
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])
```

    mask_m = offs_m < total_rows

    # compute mean,var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        xptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        x_vals = tl.load(xptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # iterate output blocks
    for out_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = out_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # load x
            xptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
            x_vals = tl.load(xptrs,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # normalize
            norm = (x_vals - mean[:, None]) * rstd[:, None]

            # LN weight & bias for hidden dim
            gamma = tl.load(ln_weight_ptr + offs_h,
                            mask=mask_h, other=0.0)
            beta = tl.load(ln_bias_ptr + offs_h,
                            mask=mask_h, other=0.0)

            norm = norm * gamma[None, :] + beta[None, :]   # fp32

            # gate values
            gptrs = gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
            gates = tl.load(gptrs,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

            norm = norm * gates  # fp32

            # load out_weight slice (BLOCK_H, BLOCK_O) transposed
            # weight pointer index: out_weight_ptr is [out_dim, hidden_dim]
            wptrs = out_weight_ptr + offs_h[:, None] + offs_o[None, :] * hidden_dim
            w = tl.load(wptrs,
                        mask=mask_h[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)

            acc += tl.dot(norm, w)  # (BLOCK_M, BLOCK_O)

        # store result to out_ptr (float32)
        out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_o[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim

    # 1. First fused layernorm + projections (left/right + gates + out_gate)
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    proj_weight = torch.cat(
        [weights["left_proj.weight"],
         weights["right_proj.weight"],
         weights["left_gate.weight"],
         weights["right_gate.weight"],
         weights["out_gate.weight"]],
        dim=0).to(torch.float16)   # [5*H, dim]
    proj_out = 5 * H
    proj_raw = torch.empty(total_rows, proj_out, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    BLOCK_O = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_proj_kernel[grid](
        x_flat,
        proj_raw,
        weights["norm.weight"],
        weights["norm.bias"],
        proj_weight,
        total_rows,
        dim,
        proj_out,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # slice projection results
    left_proj_raw = proj_raw[:, :H]
    right_proj_raw = proj_raw[:, H:2*H]
    left_gate_raw = proj_raw[:, 2*H:3*H]
    right_gate_raw = proj_raw[:, 3*H:4*H]
    out_gate_raw = proj_raw[:, 4*H:5*H]

    # Apply mask if present
    if not config.get("nomask", True):
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Gates sigmoid
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates to projections
    left = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate

    # Reshape to 4‑D tensors
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # 2. Cubic multiplicative update via batched gemm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).reshape(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat_T)   # [B*H, N, N] fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H] fp16

    # 3. Second fused layernorm + output gate + final linear
    out_flat = out.reshape(total_rows, H).contiguous()
    out_gate_flat = out_gate.reshape(total_rows, H).contiguous()
    final = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    BLOCK_M2 = 128
    BLOCK_H2 = 64
    BLOCK_O2 = 128
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].to(torch.float16),
        final,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H2,
        BLOCK_O=BLOCK_O2,
        num_warps=8,
    )

    # reshape to [B, N, N, dim] and return
    return final.view(B, N, N, dim)

"""
TriMul – outgoing version (AlphaFold‑3) optimized forward pass
================================================================
The implementation follows the reference PyTorch module but removes all
intermediate memory traffic by fusing the LayerNorms with the surrounding
linear layers:

*   First LayerNorm + the four projections + the out‑gate projection are
    computed in a single Triton kernel (``layernorm_proj_kernel``).  The
    kernel evaluates the row‑wise mean/variance, applies the LN scaling/
    bias and immediately performs a GEMM with the concatenated weight
    matrix (shape ``5*hidden_dim × dim``).  No temporary ``x_norm`` tensor
    is materialised.

*   The cubic multiplicative update (``out[b,i,j] = Σ_k left[b,i,k] *
    right[b,j,k]``) is performed with a batched matrix multiplication
    (``torch.bmm``) which maps each head to a separate batch element.

*   The second LayerNorm, the element‑wise output‑gate and the final
    projection back to ``dim`` are fused in ``layernorm_gate_linear_kernel``.
    This kernel reads the post‑update tensor only once, computes the
    LN statistics, multiplies by the gate, and accumulates the final
    linear projection (weights ``to_out.weight``).  The result is written
    directly as ``float32`` – the final output of the TriMul block.

The only non‑triton operation is the batch matrix multiplication, which
already uses cuBLAS Tensor‑Core kernels.  All heavy arithmetic stays in
float16 (except the small reductions needed for the statistics) and the
final output matches the reference implementation.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm + 5 linear projections (left_proj, right_proj,
#     left_gate, right_gate, out_gate) in a single kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_proj_kernel(
    x_ptr,                # const * [R, D]            (float32)
    out_ptr,              # mutable * [R, O]          (float16)
    ln_weight_ptr,        # const * [D]               (float32)
    ln_bias_ptr,          # const * [D]               (float32)
    proj_weight_ptr,      # const * [O, D]            (float16)
    total_rows, D, O,     # runtime ints
    eps,                  # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < D
        x_load_ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        x = tl.load(x_load_ptr,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                 # fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  For every output‑channel block compute dot product:
    #     out = Σ_d ((x-μ)/σ * γ + β) * W
    # --------------------------------------------------------------
    for o_off in range(0, tl.cdiv(O, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)      # [BLOCK_O]
        mask_o = offs_o < O
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < D

            # load raw input values
            x_load_ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
            x = tl.load(x_load_ptr,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)           # fp32

            # normalise
            norm = (x - mean[:, None]) * rstd[:, None]

            # apply LN γ / β (γ = ln_weight, β = ln_bias)
            gamma = tl.load(ln_weight_ptr + offs_d,
                            mask=mask_d, other=0.0)         # fp32
            beta  = tl.load(ln_bias_ptr + offs_d,
                            mask=mask_d, other=0.0)         # fp32
            norm = norm * gamma[None, :] + beta[None, :]    # fp32

            # load projection weight block Wᵀ : shape [BLOCK_D, BLOCK_O]
            w_ptr = proj_weight_ptr + offs_d[:, None] + offs_o[None, :] * D
            w = tl.load(w_ptr,
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)           # fp32

            # accumulate: acc += norm @ w
            acc += tl.dot(norm, w)                           # (BLOCK_M, BLOCK_O)
        # store result as fp16
        out_store_ptr = out_ptr + offs_m[:, None] * O + offs_o[None, :]
        tl.store(out_store_ptr,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])

# 2️⃣  LayerNorm + output‑gate + final linear (to_out) fused kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,                # const * [R, H]            (float16)
    gate_ptr,             # const * [R, H]            (float16)
    ln_weight_ptr,        # const * [H]               (float32)
    ln_bias_ptr,          # const * [H]               (float32)
    out_weight_ptr,       # const * [D, H]            (float16)
    out_ptr,              # mutable * [R, D]          (float32)
    total_rows, H, D,     # runtime ints
    eps,                  # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Mean / variance over hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H
        x_load_ptr = x_ptr + offs_m[:, None] * H + offs_h[None, :]
        x = tl.load(x_load_ptr,
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)           # fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  For each output‑dim block accumulate (LN·gate)·Wᵀ
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
            mask_h = offs_h < H

            # load x and normalise
            x_load_ptr = x_ptr + offs_m[:, None] * H + offs_h[None, :]
            x = tl.load(x_load_ptr,
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)           # fp32
            norm = (x - mean[:, None]) * rstd[:, None]

            # LN γ / β
            gamma = tl.load(ln_weight_ptr + offs_h,
                            mask=mask_h, other=0.0)          # fp32
            beta  = tl.load(ln_bias_ptr + offs_h,
                            mask=mask_h, other=0.0)          # fp32
            norm = norm * gamma[None, :] + beta[None, :]    # fp32

            # gate
            g_load_ptr = gate_ptr + offs_m[:, None] * H + offs_h[None, :]
            g = tl.load(g_load_ptr,
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)           # fp32
            norm = norm * g                                 # fp32

            # load weight slice Wᵀ : shape [BLOCK_H, BLOCK_D]
            w_ptr = out_weight_ptr + offs_h[:, None] + offs_d[None, :] * H
            w = tl.load(w_ptr,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)           # fp32

            # accumulate
            acc += tl.dot(norm, w)                          # (BLOCK_M, BLOCK_D)

        # store final output block
        out_store_ptr = out_ptr + offs_m[:, None] * D + offs_d[None, :]
        tl.store(out_store_ptr,
                 acc,
                 mask=mask_m[:, None] & mask_d[None, :])

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]        (bool/float, optional)
        - weights_dict : model weights (float32)
        - config_dict : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim
    O = 5 * H                       # left_proj, right_proj, left_gate, right_gate, out_gate

    # --------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm with the 5 linear projections
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()          # [R, D] (fp32)

    # concatenate weight matrices for the five branches
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [O, D]

    proj_raw = torch.empty(total_rows, O, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    BLOCK_O = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_proj_kernel[grid](
        x_flat,
        proj_raw,
        weights["norm.weight"],                # LN γ  (float32)
        weights["norm.bias"],                  # LN β  (float32)
        proj_weight,
        total_rows,
        dim,
        O,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # slice the five tensors
    left_proj_raw   = proj_raw[:, 0:H]
    right_proj_raw  = proj_raw[:, H:2*H]
    left_gate_raw   = proj_raw[:, 2*H:3*H]
    right_gate_raw  = proj_raw[:, 3*H:4*H]
    out_gate_raw    = proj_raw[:, 4*H:5*H]

    # optional mask (applied after projection, before gating)
    if not config.get("nomask", True):
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gates to the projected values
    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 2️⃣  Cubic multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # reshape for batched matmul: left[i,k] · right[j,k]ᵀ
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # [B*H, N, N]
    right_mat_T = right.permute(0, 3, 2, 1).reshape(B * H, N, N) # [B*H, N, N] (already transposed)

    out_mat = torch.bmm(left_mat, right_mat_T)                     # [B*H, N, N] fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)            # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 3️⃣  Fuse second LayerNorm, output‑gate and final linear
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()            # [R, H] fp16
    out_gate_flat = out_gate.reshape(total_rows, H).contiguous() # [R, H] fp16
    final = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    BLOCK_M2 = 128
    BLOCK_H2 = 64
    BLOCK_D2 = 128       # max dim (dim ≤ 384)
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],       # LN γ (float32)
        weights["to_out_norm.bias"],         # LN β (float32)
        weights["to_out.weight"].to(torch.float16),  # [dim, H] fp16
        final,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )

    return final.view(B, N, N, dim)
