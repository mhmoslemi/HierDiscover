"""
optimised outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input  (B·N·N × dim) → fp16.
2️⃣  Fuse the four linear layers (left_proj, right_proj,
    left_gate, right_gate) into a single GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N).
3️⃣  Separate linear for the output‑gate.
4️⃣  Apply optional mask, sigmoid‑gates and element‑wise gating.
5️⃣  Reshape to [B·H, N, N] and perform the N³ update with a
    batched half‑precision GEMM (torch.bmm):
        out = left @ rightᵀ.
6️⃣  Row‑wise LayerNorm over the hidden dimension (fp16).
7️⃣  Apply the output‑gate, final linear projection back to `dim`,
    and cast to float32.

Only the two LayerNorms are implemented in Triton; all dense
matrix multiplications use cuBLAS (torch.nn.functional.linear /
torch.bmm), which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]  (float32)
    y_ptr,          # mutable * [total_rows, dim]  (float16)
    w_ptr,          # const   * [dim]              (float32) scale γ
    b_ptr,          # const   * [dim]              (float32) bias β
    total_rows,     # runtime int
    dim,            # runtime int
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Output is fp16; the reduction uses fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- reduction: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # ---------- write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # scale
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # bias

        out_fp16 = (norm * gamma[None, :] + beta[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  shape [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor  shape [B, N, N]      (bool/float) – optional
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First row‑wise LayerNorm → half‑precision
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the four linear layers into a single GEMM
    # --------------------------------------------------------------
    H = hidden_dim
    # concat weights → shape [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)          # fp16 for Tensor Core GEMM

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (fp16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # bias omitted

    left_proj_raw   = proj_gate[:, :H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]            # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]          # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]          # [R, H]

    # Separate output‑gate linear (no bias)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # → [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask_tensor: [B, N, N] → broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (half‑precision)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Gating
    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]   (i, k, h)
    right = right.view(B, N, N, H)        # [B, N, N, H]   (j, k, h)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dimension to the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i, k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (j, k)

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second row‑wise LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
