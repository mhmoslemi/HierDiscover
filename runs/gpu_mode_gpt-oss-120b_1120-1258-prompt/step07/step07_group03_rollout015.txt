@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # compute block indices: b, i0, j0
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    # Flatten mapping: block id = b * num_i * num_j + i_block * num_j + j_block
    b = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j

    i_offset = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator [BLOCK_I, BLOCK_J, BLOCK_H]
    acc = tl.zeros([BLOCK_I, BLOCK_J, BLOCK_H], dtype=tl.float32)

    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left: shape [BLOCK_I, BLOCK_K, BLOCK_H]
        left_ptrs = left_ptr + b * stride_left_b + i_offset[:, None, None] * stride_left_i + k_offset[None, :, None] * stride_left_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # Load right: shape [BLOCK_J, BLOCK_K, BLOCK_H]
        right_ptrs = right_ptr + b * stride_right_b + j_offset[:, None, None] * stride_right_i + k_offset[None, :, None] * stride_right_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # Compute elementwise product over k & h, then sum over k
        # Use broadcasting: left[i,k,h] * right[j,k,h] -> shape [i, j, k, h]
        # We sum over k dimension.
        # Expand left: [i, 1, k, h], right: [1, j, k, h]
        prod = left[:, None, :, :] * right[None, :, :, :]  # shape [i, j, k, h] 
        acc += tl.sum(prod, axis=2)  # sum over k

    # Write out: shape [i, j, h]
    out_ptrs = out_ptr + b * stride_out_b + i_offset[:, None, None] * stride_out_i + j_offset[None, :, None] * stride_out_j + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None, None] & mask_j[None, :, None])
```

Implementation steps per row:

1. Compute mean = sum(out) / H
2. Compute var = sum(out^2) / H - mean^2
3. Compute rstd = rsqrt(var + eps)
4. Normalize: norm = (out - mean) * rstd
5. Scale+shift: norm = norm * to_out_norm_weight + to_out_norm_bias
6. Gated: gated = norm * out_gate (elementwise)
7. Linear projection: out_linear = gated @ to_out_weight.T (i.e., sum over H dimension of gated[h] * weight[d, h] for each d)
8. Store result: out_linear in fp16 (or fp32 later convert).

"""
TriMul (outgoing) forward pass – highly‑optimized Triton‑augmented implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor  (fp32 → fp16) – custom Triton kernel.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM:  [R, dim] @ [4·H, dim]^T → [R, 4·H] (R = B·N·N).
3️⃣ Compute the output‑gate linear layer separately.
4️⃣ Apply optional mask, sigmoid gates and the left/right gating.
5️⃣ Batched half‑precision matrix‑multiply for the N³ update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   → a single torch.bmm on tensors shaped (B·H, N, N).
6️⃣ Fuse the second LayerNorm, the output‑gate, and the final linear
   projection (hidden → dim) into ONE Triton kernel.  This removes two
   extra memory passes (LN + linear) and keeps everything in fp16
   (fp32 accumulation).  The kernel computes for each output row:
        y = ((x - μ) / √(σ²+ε)) * γ + β          (LayerNorm)
        y = y * out_gate                         (gate)
        y = y @ W_out.T                          (final projection)
7️⃣ Cast back to float32 and reshape to [B, N, N, dim].

The only explicit Triton kernels are the two row‑wise LayerNorms and the
fused LN‑gate‑linear kernel; everything else re‑uses cuBLAS/Tensor‑Core
operations (torch.bmm / torch.nn.functional.linear) for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,                # const  * [total_rows, dim] (float32)
    y_ptr,                # mutable* [total_rows, dim] (float16)
    w_ptr,                # const  * [dim]          (float32) γ
    b_ptr,                # const  * [dim]          (float32) β
    total_rows,           # runtime int
    dim,                  # runtime int
    eps,                  # compile‑time float (e.g. 1e-5)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / √(σ²+ε)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- mean & variance ------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # ----- write normalized output ----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        sum_sq  += tl.sum(x * x, axis=1)

        mask_h = offs_h < H

            inp_ptr + offs_r[:, None] * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        g = tl.load(
            gate_ptr + offs_r[:, None] * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm on the hidden dimension
        norm = (x - mean[:, None]) * rstd[:, None]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # β
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply the output‑gate
        gated = norm * g                                            # [R, H]

        # Load a slice of the output weight matrix (transposed view)
        # weight shape: [D, H] (row‑major). We need W_T[h, d] = weight[d, h].
        w_T = tl.load(
            weight_ptr + offs_c[None, :] * H + offs_h[:, None],
            mask=mask_c[None, :] & mask_h[:, None],
            other=0.0,
        ).to(tl.float32)                                            # [H, D]

        # Accumulate: gated @ W_T  →  [R, D]
        acc += tl.dot(gated, w_T)   # (R×H)·(H×D) = (R×D)
    
    # Write the final result
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + offs_r[:, None] * D + offs_c[None, :]
    tl.store(out_ptrs, out,
             mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# Custom forward entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    R = B * N * N                          # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to (R, dim)
    x_flat = input_tensor.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices: [4·H, dim]
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16).contiguous()

    proj_gate = F.linear(x_norm, proj_gate_w)          # [R, 4·H] – fp16, no bias

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16).contiguous()
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → broadcast over hidden dim
        mask_h = mask_tensor.reshape(R, 1).to(torch.float16)   # [R,1] fp16
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)             # [B, N, N, H]

    # bring hidden dim to the batch axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.reshape(B * H, N, N)       # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)      # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Fuse 2nd LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()          # [R, H] fp16
    out_gate_f = out_gate.view(R, H).contiguous()    # [R, H] fp16

    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # tuned tile sizes (tuned for H100, H up to 128, dim up to 384)
    BLOCK_R = 128
    BLOCK_D = 64
    BLOCK_H = 16
    grid2 = (triton.cdiv(R, BLOCK_R) * triton.cdiv(dim, BLOCK_D),)

    ln_gate_linear_kernel[grid2](
        out_flat,
        out_gate_f,
        weights["to_out.weight"].to(torch.float16).contiguous(),
        weights["to_out_norm.weight"],     # γ (float32)
        weights["to_out_norm.bias"],       # β (float32)
        final_half,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
