"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor  (FP32 → FP16).  
2️⃣  Fuse the four linear projections (left, right, left‑gate, right‑gate)
    into a single half‑precision GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H].  
3️⃣  Separate the four results and apply the sigmoid gates (in‑place).  
4️⃣  Optional mask multiplication on the left/right projections.  
5️⃣  Batched half‑precision GEMM  (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N).  
6️⃣  Fuse the hidden‑dim LayerNorm with the output‑gate multiplication.  
7️⃣  Final linear projection back to `dim` (FP16 → FP32).  

All heavy per‑row work (both LayerNorms) is performed in Triton kernels.
The remaining dense matrix‑multiplications use Torch's Tensor‑Core‑enabled
`torch.bmm`/`torch.nn.functional.linear`, which on an H100 is already
close to peak performance.

The implementation supports:
* `dim` ∈ {128, 384}
* `hidden_dim` = 128  
* batch size 1‑2, sequence length up to 1024  
* optional mask (passed or omitted)
* inputs in FP32 (standard normal or Cauchy)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (FP32)
    y_ptr,               # mutable * [total_rows, dim]   (FP16)
    weight_ptr,          # const   * [dim]               (FP32) γ
    bias_ptr,            # const   * [dim]               (FP32) β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # ---------- apply γ,β and write FP16 ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = norm * w[None, :] + b[None, :]   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ LayerNorm + gate multiplication (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(

    weight_ptr, bias_ptr,        # LN parameters γ,β (FP32)

# 3️⃣ Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or ``None``
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : must contain ``dim`` (int) and ``hidden_dim`` (int);
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Tune block sizes: 256 rows per program, 64 columns per inner loop
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (FP32)
        weights["norm.bias"],         # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear (four outputs in one GEMM)
    # --------------------------------------------------------------
    # Convert projection weights to FP16 once
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16) # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16) # [H, dim]

    # Concatenate along output dimension → [4·H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # (R, dim) @ (4·H, dim)^T → (R, 4·H)   (FP16)
    proj_gate = F.linear(x_norm, w_concat)          # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + sigmoid gates
    # --------------------------------------------------------------
    # If a mask tensor is supplied and ``nomask`` flag is False, apply it
    if (mask_tensor is not None) and (not cfg.get("nomask", True)):
        # mask: [B, N, N] → [R, 1] (FP16) for broadcasting
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate   # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_{b,i,j,:} = Σ_k left_{b,i,k,:} * right_{b,j,k,:}
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # Bring hidden dim to the batch dimension for a simple bmm
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]  FP16

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, Tensor‑cores

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    out_flat = out.view(total_rows, H)                                 # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + multiplication by out‑gate
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    # Cast to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
