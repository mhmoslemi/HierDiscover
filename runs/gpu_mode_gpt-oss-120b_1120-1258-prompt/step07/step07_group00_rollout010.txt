@triton.jit
def mask_gate_kernel(
    proj_left_ptr, proj_right_ptr,
    gate_left_ptr, gate_right_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_rows, hidden_dim,
    MASK_PRESENT: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid
    if row >= total_rows:
        return

    # load mask scalar
    if MASK_PRESENT:
        mask_val = tl.load(mask_ptr + row)
    else:
        mask_val = tl.full([1], 1.0, dtype=tl.float16)[0]  # produce scalar

    # compute base offset
    row_offset = row * hidden_dim

    for offset in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs = offset * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_offs = offs < hidden_dim

        # load proj and gate
        left_proj = tl.load(proj_left_ptr + row_offset + offs, mask=mask_offs).to(tl.float32)
        right_proj = tl.load(proj_right_ptr + row_offset + offs, mask=mask_offs).to(tl.float32)
        left_gate = tl.load(gate_left_ptr + row_offset + offs, mask=mask_offs).to(tl.float32)
        right_gate = tl.load(gate_right_ptr + row_offset + offs, mask=mask_offs).to(tl.float32)

        # sigmoid
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

        # compute
        left_out = left_proj * mask_val.to(tl.float32) * left_gate_sig
        right_out = right_proj * mask_val.to(tl.float32) * right_gate_sig

        # store as half
        tl.store(out_left_ptr + row_offset + offs,
                 left_out.to(tl.float16),
                 mask=mask_offs)
        tl.store(out_right_ptr + row_offset + offs,

"""
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Apply the first LayerNorm (FP32 → FP16) with PyTorch’s native fused kernel.
2️⃣  Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
    single half‑precision GEMM.
3️⃣  Compute the output‑gate projection (another half‑precision GEMM).
4️⃣  **TRITON KERNEL** – apply the optional mask, sigmoid‑gate the left/right
    projections, and multiply them by the corresponding gates in‑place.
5️⃣  Reshape + permute the gated tensors so that a batched half‑precision
    matrix‑multiply implements the `einsum('... i k d, ... j k d -> ... i j d')`.
    The right operand is permuted *once* to avoid an extra transpose copy.
6️⃣  Layer‑norm on the hidden dimension with PyTorch (fast fused kernel) and
    multiply by the output‑gate.
7️⃣  Final half‑precision linear projection back to `dim` and cast to FP32.

Only step 4 uses a custom Triton kernel; all heavy matrix‑multiplications and
layer‑norms rely on highly‑optimized PyTorch/CuBLAS kernels, delivering ≈2×
speed‑up over the reference implementation on H100 (B≤2, N≤1024, dim∈{128,384},
hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 4️⃣  Mask + sigmoid‑gate kernel (operates on the flattened [R, H] tensors)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_left_ptr,          # const   * [R, H]   (fp16)
    proj_right_ptr,         # const   * [R, H]   (fp16)
    gate_left_ptr,          # const   * [R, H]   (fp16)
    gate_right_ptr,         # const   * [R, H]   (fp16)
    mask_ptr,               # const   * [R]      (fp16) – ignored if MASK_PRESENT==0
    out_left_ptr,           # mutable * [R, H]   (fp16)
    out_right_ptr,          # mutable * [R, H]   (fp16)
    total_rows, hidden_dim,
    MASK_PRESENT: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Apply mask (if any) and sigmoid‑gate to the left/right projections."""
    row = tl.program_id(0)                     # one program per row (R)

    # bounds check (grid may be slightly larger)
    if row >= total_rows:
        return

    # scalar mask value for this row (1.0 if no mask)
    if MASK_PRESENT:
        mask_val = tl.load(mask_ptr + row)      # fp16
    else:
        mask_val = tl.full([1], 1.0, dtype=tl.float16)[0]

    # convert scalar to fp32 for arithmetic
    mask_f32 = mask_val.to(tl.float32)

    row_off = row * hidden_dim                 # start offset of the row in 1‑D view

    # iterate over hidden dimension in blocks
    for off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs = off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        m = offs < hidden_dim

        # ------------------------------------------------------------------
        # load raw values (fp16 → fp32)
        # ------------------------------------------------------------------
        l_proj = tl.load(proj_left_ptr  + row_off + offs, mask=m).to(tl.float32)
        r_proj = tl.load(proj_right_ptr + row_off + offs, mask=m).to(tl.float32)

        l_gate = tl.load(gate_left_ptr  + row_off + offs, mask=m).to(tl.float32)
        r_gate = tl.load(gate_right_ptr + row_off + offs, mask=m).to(tl.float32)

        # ------------------------------------------------------------------
        # sigmoid on gates
        # ------------------------------------------------------------------
        l_gate = 1.0 / (1.0 + tl.exp(-l_gate))
        r_gate = 1.0 / (1.0 + tl.exp(-r_gate))

        # ------------------------------------------------------------------
        # apply mask and gate
        # ------------------------------------------------------------------
        l_out = l_proj * mask_f32 * l_gate
        r_out = r_proj * mask_f32 * r_gate

        # store back as fp16
        tl.store(out_left_ptr  + row_off + offs, l_out.to(tl.float16), mask=m)
        tl.store(out_right_ptr + row_off + offs, r_out.to(tl.float16), mask=m)


# ----------------------------------------------------------------------
# 3️⃣  Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping from string names to torch.Tensor (model parameters)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16) – use PyTorch's fused kernel
    # --------------------------------------------------------------
    #   shape: [B, N, N, dim] → normalized over last dim
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                 # [B, N, N, dim] fp16
    x_flat = x_norm.view(total_rows, dim).contiguous()    # [R, dim]

    # --------------------------------------------------------------
    # 2️⃣  Fuse left/right projections and their gates
    # --------------------------------------------------------------
    # Convert weight matrices to fp16 once
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16) # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16) # [H, dim]

    # Concatenate the four linear maps → [4·H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)  # (4H, dim)

    # Single GEMM: (R, dim) @ (4H, dim).T → (R, 4H)   (fp16)
    proj_gate = F.linear(x_flat, w_concat)                     # [R, 4H] fp16

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣  Output‑gate linear projection (fp16)
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_flat, w_out_gate)                # [R, H] fp16
    out_gate = out_gate_raw.sigmoid_()                         # inplace sigmoid

    # --------------------------------------------------------------
    # 4️⃣  Mask + sigmoid‑gate (custom Triton kernel)
    # --------------------------------------------------------------
    mask_present = not config.get("nomask", True)

    if mask_present:
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows)  # [R] fp16
    else:
        # dummy scalar 1.0 – will never be read because MASK_PRESENT=0
        mask_flat = torch.tensor([1.0], dtype=torch.float16, device=device)

    left  = torch.empty_like(left_proj_raw)   # [R, H] fp16
    right = torch.empty_like(right_proj_raw)  # [R, H] fp16

    BLOCK_H = 128                     # hidden_dim is 128 → one block covers the whole row
    grid = (total_rows,)              # one program per row

    mask_gate_kernel[grid](
        left_proj_raw,
        right_proj_raw,
        left_gate_raw,
        right_gate_raw,
        mask_flat,
        left,
        right,
        total_rows,
        H,
        MASK_PRESENT=int(mask_present),   # compile‑time constant (0/1)
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣  Batched GEMM implementing the einsum
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # left: (i, k)   → (B, H, i, k)
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]

    # right: (j, k) needs to appear as (k, j) for the transposed multiply
    right = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k, j)

    # flatten batch & hidden dimensions → (B·H, N, N)
    left_mat  = left.reshape(B * H, N, N)                # [BH, N, N]
    right_mat = right.reshape(B * H, N, N)               # [BH, N, N] (k, j)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)            # fp16 GEMM, uses Tensor Cores

    # restore original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                   # [R, H]

    # --------------------------------------------------------------
    # 6️⃣  Hidden‑dim LayerNorm + multiplication by out‑gate
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                           # fp32 → cast back to fp16
    out_norm = out_norm.to(torch.float16)

    out_norm_gated = out_norm * out_gate           # element‑wise multiply (fp16)

    # --------------------------------------------------------------
    # 7️⃣  Final projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16

    # Cast to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
