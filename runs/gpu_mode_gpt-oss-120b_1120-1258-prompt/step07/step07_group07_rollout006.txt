"""
TriMul – outgoing version (AlphaFold‑3) forward pass (high‑performance)

Algorithm
---------
1. Row‑wise LayerNorm on the input tensor (float32 → float16) – Triton kernel.
2. Fuse the four linear projections (left/right + left/right‑gate) into a
   single GEMM (torch.nn.functional.linear, fp16).  The “out‑gate” projection
   is a second GEMM.
3. Apply the optional pairwise mask, sigmoid the gates and multiply the
   projected tensors.
4. Compute the cubic multiplicative update
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication (torch.bmm) – leverages Tensor‑Cores.
5. Row‑wise LayerNorm + element‑wise output‑gate (Triton kernel).  This fuses
   the second LayerNorm (`to_out_norm`) with the gating operation.
6. Final linear projection back to the model dimension (torch.nn.functional.linear).
7. Cast the result to float32 (the reference implementation’s dtype).

Both custom kernels operate on one row per thread (BLOCK_M rows per program)
and load the feature dimension in vectorised blocks (BLOCK_D = 128).  This
minimises the number of launched thread‑blocks and maximises occupancy on
H100.  All heavy arithmetic stays in fp16 with fp32 accumulation for
numerical stability.

The implementation respects the required signature:
    custom_kernel((input, mask, weights, config)) → Tensor[float32].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16) – used for the first norm
# ----------------------------------------------------------------------
@triton.jit
def row_layernorm_kernel(
    x_ptr,               # const   * [R, C]   (float32)
    y_ptr,               # mutable * [R, C]   (float16)
    weight_ptr,          # const   * [C]      (float32)
    bias_ptr,            # const   * [C]      (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program (threads)
    BLOCK_D: tl.constexpr,   # feature‑dim block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1) compute mean / variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # each thread loads a vector of length BLOCK_D
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # shape (BLOCK_M, BLOCK_D)
        sum_val += tl.sum(vals, axis=1)   # sum across the feature block
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- 2) write normalised values (fp16) ------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Row‑wise LayerNorm + output‑gate fusion (second norm)
# ----------------------------------------------------------------------
@triton.jit
def row_layernorm_gate_kernel(
    x_ptr,               # const   * [R, C]   (float32)
    y_ptr,               # mutable * [R, C]   (float16)
    weight_ptr,          # const   * [C]      (float32)
    bias_ptr,            # const   * [C]      (float32)
    gate_ptr,            # const   * [R, C]   (float16)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- mean / variance (fp32) -----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- normalise + weight/bias + gate ---------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # original x
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        x_vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = norm * gate                # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward (outgoing TriMul)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, may be None)
        - weights_dict : dict of weight tensors (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N           # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512          # rows per program (≤ 1024 for H100)
    BLOCK_D = 128          # feature‑dim block (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],                # γ (float32)
        weights["norm.bias"],                  # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=16,               # 512 threads = 16 warps
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gates (single GEMM)
    # ------------------------------------------------------------------
    # concatenate projection weight matrices: (4*H, dim)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # Linear on the normalised tensor
    proj_gate = F.linear(x_norm, w_4h)        # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out_gate projection (separate)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                  # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoids and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is [B, N, N] → flatten to [R, 1] (fp16) and broadcast
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # inplace sigmoid (still fp16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gate to the projected values
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H] fp16
    right = right.view(B, N, N, H)         # [B, N, N, H] fp16
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and perform bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B*H, N, N]

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B*H, N, N] fp16

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()          # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(total_rows, H), # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=16,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)   # fp32 output
    return final
