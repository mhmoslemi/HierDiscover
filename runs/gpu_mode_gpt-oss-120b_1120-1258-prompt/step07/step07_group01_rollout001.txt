                         batch, M, N, K,
                         stride_amat, stride_arow, stride_ak,
                         stride_bmat, stride_brow, stride_bk,

    total_blocks_per_batch = tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    batch_idx = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    pid_m = block_idx // tl.cdiv(N, BLOCK_N)
    pid_n = block_idx % tl.cdiv(N, BLOCK_N)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

"""
custom_kernel
==============

Fast forward implementation of the **outgoing TriMul** operation from AlphaFold‑3.
The algorithm follows the reference PyTorch module but moves the two LayerNorms
to custom Triton kernels (row‑wise, fp32→fp16) and replaces the N³ multiplicative
update with a batch GEMM written in Triton.  All heavy linear work stays in half‑
precision (fp16) and the final result is cast back to fp32.

Key ideas
---------
* Row‑wise LayerNorm is implemented in Triton – one kernel per 2‑D tensor.
* The four linear projections/gates are fused into a single half‑precision
  GEMM (`F.linear`).
* Optional masking and sigmoid gating are applied in‑place on the half tensors.
* The N³ update `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
  is expressed as a batched GEMM of shape (B·H, N, N) × (B·H, N, N)ᵀ and
  realised in Triton (`_trimul_kernel`).  This removes the temporary transpose
  and eliminates the overhead of `torch.matmul` for the many small batches.
* The second LayerNorm, output‑gate and final projection are performed with the
  same Triton LayerNorm kernel and a half‑precision linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,      # const   * [rows, dim] (float32)
    y_ptr,      # mutable * [rows, dim] (float16)
    w_ptr,      # const   * [dim]      (float32) – gamma
    b_ptr,      # const   * [dim]      (float32) – beta
    rows,       # int32   (runtime)
    dim,        # int32   (runtime)
    eps,        # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- 1️⃣ mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load a block of the row
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # -------- 2️⃣ write normalized output (fp16) -------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * gamma[None, :] + beta[None, :]   # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: batched N³ multiplicative update (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr,      # const * [batch, M, K] (fp16)
    b_ptr,      # const * [batch, N, K] (fp16) – will be read as transposed (K,N)
    c_ptr,      # mutable * [batch, M, N] (fp16)
    batch,      # int32 (runtime)
    M,          # int32 (runtime) – = N
    N,          # int32 (runtime) – = N
    K,          # int32 (runtime) – = N
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each batch entry:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]
    Both A and B are half‑precision; the accumulation is performed in fp32.
    """
    pid = tl.program_id(0)

    # ---------- grid layout ----------
    # one program per (batch, block_m, block_n)
    blocks_m = tl.cdiv(M, BLOCK_M)
    blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = blocks_m * blocks_n

    batch_idx = pid // blocks_per_batch
    block_id  = pid %  blocks_per_batch
    pid_m = block_id // blocks_n
    pid_n = block_id % blocks_n

    # ---------- block offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # ---------- accumulate ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_iter * BLOCK_K
        cur_k_eff = tl.minimum(BLOCK_K, K - cur_k)

        # A block: [BLOCK_M, cur_k_eff]
        a_ptrs = a_batch_ptr \
                 + offs_m[:, None] * stride_a_m \
                 + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < cur_k_eff),
                    other=0.0).to(tl.float32)

        # B block (read as transposed): [cur_k_eff, BLOCK_N]
        b_ptrs = b_batch_ptr \
                 + offs_n[None, :] * stride_b_n \
                 + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < cur_k_eff) & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BM, BK) × (BK, BN) → (BM, BN)

    # ---------- store ----------
    c_ptrs = c_batch_ptr \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _trimul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    ``left`` and ``right`` must be contiguous tensors of shape
    [B, H, N, N] and dtype torch.float16.  Returns a tensor of the same
    shape in fp16.
    """
    B, H, N, _ = left.shape
    batch = B * H
    a = left.view(batch, N, N)           # [B·H, M, K]
    b = right.view(batch, N, N)          # [B·H, N, K] (will be read transposed)

    out = torch.empty_like(a)            # [B·H, M, N] fp16

    # strides (contiguous layout)
    stride_a_batch = a.stride(0)
    stride_a_m = a.stride(1)
    stride_a_k = a.stride(2)

    stride_b_batch = b.stride(0)
    stride_b_n = b.stride(1)   # note: we will treat dim N as "row" of B
    stride_b_k = b.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_m = out.stride(1)
    stride_c_n = out.stride(2)

    blocks_m = triton.cdiv(N, block_m)
    blocks_n = triton.cdiv(N, block_n)
    total_blocks = blocks_m * blocks_n * batch

    grid = (total_blocks,)
    _trimul_kernel[grid](
        a,
        b,
        out,
        batch,
        N,          # M
        N,          # N
        N,          # K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if
                         config["nomask"] is True
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and optionally
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack and sanity‑check
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()      # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma  (fp32)
        weights["norm.bias"],     # beta   (fp32)
        eps=eps,
    )                                                          # fp16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (produces 4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # pack the four weight matrices for left/right proj and left/right gate
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                         # [4·H, dim]

    # Single half‑precision GEMM: (R, dim) @ (dim, 4·H) → (R, 4·H)
    proj_gate = F.linear(x_norm, w_proj_gate)                  # fp16

    # Split the fused result
    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]               # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    # Separate linear for the output‑gate
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gating, element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [R,1] (fp16) and broadcast over H
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (keeps fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply the element‑wise gates
    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Bring hidden dimension to axis‑1 → (B, H, N, N) and make contiguous
    left_bh = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched GEMM (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)  (fp16)
    out_bh = _trimul_triton(left_bh, right_bh)         # [B·H, N, N] fp16

    # Back to the original layout
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()           # [R, H] fp16
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        eps=eps,
    )                                                          # fp16
    out_norm = out_norm.view(B, N, N, H)                     # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection back to ``dim``, cast to fp32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                            # fp16

    # Final linear projection (fp16 weight)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),          # [dim, H]
    )                                                          # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
