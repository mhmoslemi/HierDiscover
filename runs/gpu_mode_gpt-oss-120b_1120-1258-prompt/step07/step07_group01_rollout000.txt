"""
custom_kernel
==============

Highly‑optimized forward implementation of the **outgoing** TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch module but moves
the two LayerNorms to custom Triton kernels (row‑wise, fp32 → fp16) and keeps
all large linear ops in half‑precision.  The expensive N³ multiplicative
update

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is expressed as a batched half‑precision GEMM:

    (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)

which allows the use of cuBLAS Tensor‑Core kernels.  The final result is cast
back to float32 to match the original API.

Input
-----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    * input_tensor : torch.Tensor, [B, N, N, dim] (float32)
    * mask_tensor  : torch.Tensor, [B, N, N] (bool/float) – ignored if
                     config["nomask"] is True
    * weights_dict : mapping of all Linear / LayerNorm parameters
    * config_dict  : must contain "dim", "hidden_dim" and optional
                     "nomask" (default True)

Output
------
    torch.Tensor, [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – gamma
    bias_ptr,       # const   * [dim]      (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    """
    Computes a row‑wise LayerNorm over a 2‑D tensor ``x`` (rows × dim) in
    fp32 and writes the normalized result in fp16 to ``y``.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # linear index
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized output (apply gamma/beta) in fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        norm = (vals - mean[:, None]) * rstd[:, None]       # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]                # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D float32 tensor ``x`` of shape [rows, dim].
    Returns a float16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict with all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N  # total number of (b,i,j) positions

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # Flatten to [R, dim] (row‑major) and apply Triton LN
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # [dim]  fp32
        weights["norm.bias"],     # [dim]  fp32
        eps=eps,
        block_m=256,
        block_d=32,
    )                               # fp16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4× hidden_dim outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices for the linear that produces
    # left_proj, right_proj, left_gate, right_gate.
    # Each has shape [H, dim] (fp32); we cast once to fp16.
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # [4*H, dim]

    # Single GEMM: (R, dim) @ (dim, 4*H).T → (R, 4*H)
    proj_gate = F.linear(x_norm, w_proj_gate)  # fp16

    # Slice the fused result
    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gating and element‑wise multiplication
    # --------------------------------------------------------------
    if use_mask:
        # mask: [B, N, N] → [R, 1] (fp16) and broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate   # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Move hidden dim to the batch axis → (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched half‑precision matrix multiplication:
    #   (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))  # fp16

    # Back to layout [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension) → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # [H] fp32
        weights["to_out_norm.bias"],     # [H] fp32
        eps=eps,
        block_m=256,
        block_d=32,
    )                                                # fp16, [R, H]

    out_norm = out_norm.view(B, N, N, H)            # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                  # fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )                                                # fp16, [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

**Explanation of key components**

1. **Row‑wise LayerNorm in Triton** – Computes per‑row mean/variance fully on‑GPU,
   writes the normalized result (γ·x̂ + β) in FP16.  Two separate calls handle the
   input‑norm (`dim`) and the hidden‑norm (`hidden_dim`).

2. **Fused projection & gate computation** – By concatenating the four linear
   weight matrices we replace four separate GEMMs with a single half‑precision
   GEMM (`F.linear`).  The output‑gate is a second GEMM.

3. **Mask handling** – When a mask is present we broadcast it over the hidden
   dimension after projection (as in the reference code).

4. **Batched N³ multiplicative update** – The core operation is expressed as a
   batched matrix multiplication over the sequence dimension, leveraging cuBLAS
   Tensor‑Core kernels.  No explicit Python loops are used.

5. **Second LayerNorm & final projection** – Another Triton LN brings the hidden
   representation back to a normalized space before applying the output gate
   and a final half‑precision linear projection to the original channel size.
   The result is converted back to float32 for API compatibility.

The implementation respects all constraints (no gradients, works for all
listed configurations, uses Triton for non‑trivial kernels, and returns a
float32 tensor).
