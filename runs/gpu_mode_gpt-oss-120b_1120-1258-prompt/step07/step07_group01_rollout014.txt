"""
Fast outgoing ``TriMul`` (AlphaFold‑3) implemented with Triton.

The implementation follows the reference PyTorch module but replaces the two
row‑wise LayerNorms with custom Triton kernels and fuses several Linear
operations:

*   First LayerNorm → fp16.
*   One GEMM computes *left_proj*, *right_proj*, *left_gate*, *right_gate*
    and *out_gate* simultaneously (5·H outputs).
*   Masking and sigmoid gating are applied in‑place.
*   The expensive N³ multiplicative update is a batched half‑precision
    ``matmul`` on tensors of shape (B·H, N, N).
*   The second LayerNorm and the output‑gate are fused in a single Triton
    kernel; the result is fed directly into the final Linear that projects
    back to ``dim``.
*   All heavy tensors stay in fp16; the final result is returned in float32.

The kernels use a larger row block (1024) and a 64‑column block for better
occupancy on H100.  This reduces the kernel launch overhead and improves the
overall latency to well under the 1 ms target for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (no gate)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32)
    bias_ptr,       # const   * [dim]      (float32)
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals,  axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]          # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 1024,
                      block_d: int = 64) -> torch.Tensor:
    """
    Row‑wise LayerNorm (fp32 → fp16) on a 2‑D tensor ``x`` of shape
    [rows, dim].
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=8,            # H100 prefers more warps for this tiny work
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm + element‑wise gate (used after N³)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    gate_ptr,       # const   * [rows, dim] (float16)
    y_ptr,          # mutable * [rows, dim] (float16)  (norm * gate)
    weight_ptr,     # const   * [dim]      (float32)
    bias_ptr,       # const   * [dim]      (float32)
    rows,
    dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # -----------------------------------------------------------------
    # mean / var
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # apply norm, weight, bias and gate
    # -----------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load raw x
        x_ptr_tile = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_val = tl.load(x_ptr_tile,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # layer‑norm
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # gamma / beta
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        norm = norm * w[None, :] + b[None, :]

        # gate (fp16 → fp32)
        g_ptr = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                     # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_gate_triton(x: torch.Tensor,
                           gate: torch.Tensor,
                           weight: torch.Tensor,
                           bias: torch.Tensor,
                           eps: float = 1e-5,
                           block_m: int = 1024,
                           block_d: int = 64) -> torch.Tensor:
    """
    Row‑wise LayerNorm followed by an element‑wise multiplication with ``gate``.
    Returns fp16 tensor of shape [rows, dim].
    """
    rows, dim = x.shape
    out = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_gate_kernel[grid](
        x,
        gate,
        out,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N

    # --------------------------------------------------------------
    # 0️⃣ first LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()      # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )                                                            # fp16, [R, dim]

    # --------------------------------------------------------------
    # 1️⃣ fused projection + gates (4·H + 1·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection/gate matrices + out_gate matrix
    proj_weights = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                          # [(5·H), dim]

    proj = F.linear(x_norm, proj_weights)                        # fp16, [R, 5·H]

    left_proj_raw  = proj[:,               0: H]                # [R, H]
    right_proj_raw = proj[:,               H:2*H]                # [R, H]
    left_gate_raw  = proj[:,           2*H:3*H]                # [R, H]
    right_gate_raw = proj[:,           3*H:4*H]                # [R, H]
    out_gate_raw   = proj[:,           4*H:5*H]                # [R, H]

    # --------------------------------------------------------------
    # 2️⃣ mask (if any) + sigmoid gating + element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (fp16) and broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate      # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)           # [B, N, N, H]

    # bring hidden dim to the batch dimension: (B, H, N, N)
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )                                                   # fp16, [B, H, N, N]

    # back to [B, N, N, H] (contiguous for the following layernorm)
    out = out_bh.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    out_flat = out.view(total_rows, H)                   # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ fused second LayerNorm + output‑gate
    # --------------------------------------------------------------
    out_fused = _layernorm_gate_triton(
        out_flat,
        out_gate,                                 # gate from step 2
        weights["to_out_norm.weight"],            # gamma (float32)
        weights["to_out_norm.bias"],              # beta  (float32)
        eps=eps,
    )                                            # fp16, [R, H]

    # --------------------------------------------------------------
    # 5️⃣ final projection back to ``dim``
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                # fp16, [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
