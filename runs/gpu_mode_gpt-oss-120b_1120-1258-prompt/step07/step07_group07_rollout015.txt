@triton.jit
def norm_proj_gate_fused_kernel(
    x_ptr, ln_w_ptr, ln_b_ptr,
    lp_w_ptr, rp_w_ptr, lg_w_ptr, rg_w_ptr, og_w_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, dim, hidden_dim,
    eps,
    stride_xr, stride_xd,
    stride_ln_w, stride_ln_b,
    stride_lp_out, stride_lp_in,
    stride_rp_out, stride_rp_in,
    stride_lg_out, stride_lg_in,
    stride_rg_out, stride_rg_in,
    stride_og_out, stride_og_in,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Sum pass
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * stride_xr + offs_d[None, :] * stride_xd
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    #-------------------------------------------------
    # GEMM passes for 5 projections/gates
    #-------------------------------------------------
    # allocate accumulators
    left_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    right_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    right_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    out_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    # repeated d loop
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x block
        x_ptrs = x_ptr + offs_m[:, None] * stride_xr + offs_d[None, :] * stride_xd
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize (x - mean)*rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # apply LN weight & bias
        gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)
        norm = norm * gamma[None, :] + beta[None, :]  # fp32

        # cast to fp16 for dot
        norm_fp16 = norm.to(tl.float16)

        # hidden dim mask
        offs_h = tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # left_proj
        w_lp = tl.load(lp_w_ptr + offs_h[None, :] * stride_lp_out + offs_d[:, None] * stride_lp_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        left_acc += tl.dot(norm_fp16, w_lp)

        # right_proj
        w_rp = tl.load(rp_w_ptr + offs_h[None, :] * stride_rp_out + offs_d[:, None] * stride_rp_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        right_acc += tl.dot(norm_fp16, w_rp)

        # left_gate
        w_lg = tl.load(lg_w_ptr + offs_h[None, :] * stride_lg_out + offs_d[:, None] * stride_lg_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        left_gate_acc += tl.dot(norm_fp16, w_lg)

        # right_gate
        w_rg = tl.load(rg_w_ptr + offs_h[None, :] * stride_rg_out + offs_d[:, None] * stride_rg_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        right_gate_acc += tl.dot(norm_fp16, w_rg)

        # out_gate
        w_og = tl.load(og_w_ptr + offs_h[None, :] * stride_og_out + offs_d[:, None] * stride_og_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        out_gate_acc += tl.dot(norm_fp16, w_og)

    # Apply sigmoid to gate accumulators
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    left_gate_sig = sigmoid(left_gate_acc)
    right_gate_sig = sigmoid(right_gate_acc)
    out_gate_sig = sigmoid(out_gate_acc)

    # Gated left/right
    left = left_acc * left_gate_sig
    right = right_acc * right_gate_sig

    # Cast to fp16 for storage
    left_fp16 = left.to(tl.float16)
    right_fp16 = right.to(tl.float16)
    out_gate_fp16 = out_gate_sig.to(tl.float16)

    # Store: using output strides (row stride = hidden_dim, col stride = 1)
    left_ptrs = left_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
    right_ptrs = right_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
    out_gate_ptrs = out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]

    tl.store(left_ptrs, left_fp16, mask=mask_m[:, None] & mask_h[None, :])
    tl.store(right_ptrs, right_fp16, mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptrs, out_gate_fp16, mask=mask_m[:, None] & mask_h[None, :])

@triton.jit
def final_norm_gate_linear_fused_kernel(
    out_ptr, out_gate_ptr,
    ln_w_ptr, ln_b_ptr,
    final_w_ptr,
    out_flat_ptr,
    total_rows, hidden_dim, dim,
    eps,
    stride_out_r, stride_out_h,
    stride_gate_r, stride_gate_h,
    stride_ln_w, stride_ln_b,
    stride_f_out, stride_f_in,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D_OUT: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and variance across hidden_dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        vals = tl.load(out_ptr + offs_m[:, None] * stride_out_r + offs_h[None, :] * stride_out_h,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Outer loop over output dimension blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D_OUT)):
        offs_d = d_off * BLOCK_D_OUT + tl.arange(0, BLOCK_D_OUT)
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_M, BLOCK_D_OUT], dtype=tl.float32)

        # Loop over hidden dimension blocks
        for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # Load out value
            out_vals = tl.load(out_ptr + offs_m[:, None] * stride_out_r + offs_h[None, :] * stride_out_h,
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            # Load gate
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * stride_gate_r + offs_h[None, :] * stride_gate_h,
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # Normalize out_vals
            norm = (out_vals - mean[:, None]) * rstd[:, None]
            gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
            beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]   # fp32

            # Apply gate
            v = norm * gate_vals   # fp32

            # Load final weight block (hidden_dim block x out_dim block) as fp16
            w = tl.load(final_w_ptr + offs_h[:, None] * stride_f_in + offs_d[None, :] * stride_f_out,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float16)

            v_fp16 = v.to(tl.float16)
            acc += tl.dot(v_fp16, w)

        # Store to output
        out_ptrs = out_flat_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, N_, dim_check = input_tensor.shape
    assert N == N_
    total_rows = B * N * N
    R = total_rows

    # flatten input
    x_flat = input_tensor.reshape(R, dim).contiguous()

    # allocate left/right/out_gate
    left = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # stride arguments
    stride_xr, stride_xd = x_flat.stride()
    # layernorm weights
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # project weight pointers
    lp_w = weights["left_proj.weight"].to(torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(torch.float16).contiguous()
    og_w = weights["out_gate.weight"].to(torch.float16).contiguous()

    # stride for each weight:
    stride_lp_out, stride_lp_in = lp_w.stride()
    stride_rp_out, stride_rp_in = rp_w.stride()
    stride_lg_out, stride_lg_in = lg_w.stride()
    stride_rg_out, stride_rg_in = rg_w.stride()
    stride_og_out, stride_og_in = og_w.stride()

    # output strides (left,right,out_gate)
    out_stride_r, out_stride_c = left.stride()

    BLOCK_M = 128
    BLOCK_D = 64
    BLOCK_H = 128

    grid = (triton.cdiv(R, BLOCK_M),)

    proj_fused_kernel[grid](
        x_flat,
        ln_w, ln_b,
        lp_w, rp_w, lg_w, rg_w, og_w,
        left, right, out_gate,
        R, dim, hidden_dim, eps,
        stride_xr, stride_xd,
        1, 1,  # ln weight stride (contiguous)
        stride_lp_out, stride_lp_in,
        stride_rp_out, stride_rp_in,
        stride_lg_out, stride_lg_in,
        stride_rg_out, stride_rg_in,
        stride_og_out, stride_og_in,
        out_stride_r, out_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # Apply mask if required
    if not config.get("nomask", True):
        mask_flat = mask_tensor.reshape(R).to(torch.float16)
        left = left * mask_flat[:, None]
        right = right * mask_flat[:, None]

    # Cubic multiplication
    left_4d = left.view(B, N, N, hidden_dim)
    right_4d = right.view(B, N, N, hidden_dim)

    left_mat = left_4d.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_4d.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim] fp16

    # Flatten out and out_gate for final kernel
    out_flat = out.reshape(R, hidden_dim).contiguous()
    out_gate_flat = out_gate.reshape(R, hidden_dim).contiguous()
    final_out = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Strides for out and out_gate
    out_stride_r, out_stride_c = out_flat.stride()
    gate_stride_r, gate_stride_c = out_gate_flat.stride()
    # Layernorm weight for to_out_norm
    ln2_w = weights["to_out_norm.weight"]
    ln2_b = weights["to_out_norm.bias"]
    # final linear weight
    final_w = weights["to_out.weight"].to(torch.float16).contiguous()
    stride_f_out, stride_f_in = final_w.stride()  # out stride (dim), in stride (hidden_dim)

    # final kernel grid
    grid2 = (triton.cdiv(R, BLOCK_M),)

    final_norm_gate_linear_fused_kernel[grid2](
        out_flat,
        out_gate_flat,

        gate_stride_r, gate_stride_c,

        stride_f_out, stride_f_in,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D_OUT=64,
        num_warps=8,
    )

    final = final_out.view(B, N, N, dim).to(torch.float32)
    return final

"""
TriMul (outgoing) forward pass – Triton‐accelerated implementation
-----------------------------------------------------------------

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}.  The reference
implementation is:

    X ← LayerNorm(X)
    L ← LinearL(X)       (left projection)
    R ← LinearR(X)       (right projection)
    gL ← σ(LinearGL(X))  (left gate)
    gR ← σ(LinearGR(X))  (right gate)
    gO ← σ(LinearGO(X))  (output gate)

    L ← L * gL * mask            (optional)
    R ← R * gR * mask            (optional)

    O[b,i,j,:] = Σ_k L[b,i,k,:] * R[b,j,k,:]          (cubic update)

    O ← LayerNorm(O) * gO
    Y ← LinearOut(O)                                 (back to C)

All heavy arithmetic is kept in FP16.  Two custom Triton kernels are
used:

1. `norm_proj_gate_fused_kernel`
   – row‑wise LayerNorm on X,
   – five fused linear layers (left/right projections and three gates),
   – sigmoid on gates and element‑wise gating.
   The kernel directly writes L, R and gO (all FP16).

2. `final_norm_gate_linear_fused_kernel`
   – row‑wise LayerNorm on the cubic result O,
   – element‑wise multiplication with the pre‑computed output gate,
   – a fused linear projection back to the original channel dimension.

The cubic multiplicative update is expressed as a batched matrix‑multiply
using cuBLAS (`torch.bmm`), which is already highly tuned on H100.
Only three kernel launches are required (two Triton kernels + the BMM).

The implementation works for all benchmark configurations
(B∈{1,2}, N∈{128,…,1024}, C∈{128,384}, hidden_dim=128) and both masked
and unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: LayerNorm + 5 linear projections + gating
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_gate_fused_kernel(
    # inputs
    x_ptr,                     # const   * [R, D]          (float32)
    ln_w_ptr, ln_b_ptr,        # const   * [D]            (float32)
    # projection / gate weights (all [H, D])
    lp_w_ptr, rp_w_ptr, lg_w_ptr, rg_w_ptr, og_w_ptr,
    # outputs
    left_ptr, right_ptr, out_gate_ptr,   # mutable * [R, H] (float16)
    # runtime parameters
    R, D, H, eps,
    stride_xr, stride_xd,               # strides for x
    stride_ln_w, stride_ln_b,           # usually 1
    stride_lp_out, stride_lp_in,        # strides for left_proj.weight
    stride_rp_out, stride_rp_in,
    stride_lg_out, stride_lg_in,
    stride_rg_out, stride_rg_in,
    stride_og_out, stride_og_in,
    out_stride_r, out_stride_c,         # strides for the three outputs
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Row‑wise LayerNorm → five fused linear layers → gated left/right."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # 1️⃣  Compute row‑wise mean / variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < D

        x_ptrs = x_ptr + offs_m[:, None] * stride_xr + offs_d[None, :] * stride_xd
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  GEMM for the five projections (FP16 matmul, FP32 accumulator)
    # ------------------------------------------------------------------
    # allocate accumulators
    left_acc       = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    right_acc      = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    left_gate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    right_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    out_gate_acc   = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    # hidden‑dim index (static)
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < D

        # ----- load raw input and normalise ---------------------------------
        x_ptrs = x_ptr + offs_m[:, None] * stride_xr + offs_d[None, :] * stride_xd
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # (x - μ) * rstd
        norm = (x - mean[:, None]) * rstd[:, None]

        # γ and β from LayerNorm
        gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)
        norm = norm * gamma[None, :] + beta[None, :]          # still FP32

        # cast to FP16 for the dot products
        norm_fp16 = norm.to(tl.float16)

        # ----- left projection ----------------------------------------------
        w_lp = tl.load(lp_w_ptr + offs_h[None, :] * stride_lp_out + offs_d[:, None] * stride_lp_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        left_acc      += tl.dot(norm_fp16, w_lp)

        # ----- right projection ---------------------------------------------
        w_rp = tl.load(rp_w_ptr + offs_h[None, :] * stride_rp_out + offs_d[:, None] * stride_rp_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        right_acc     += tl.dot(norm_fp16, w_rp)

        # ----- left gate ----------------------------------------------------
        w_lg = tl.load(lg_w_ptr + offs_h[None, :] * stride_lg_out + offs_d[:, None] * stride_lg_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        left_gate_acc += tl.dot(norm_fp16, w_lg)

        # ----- right gate ---------------------------------------------------
        w_rg = tl.load(rg_w_ptr + offs_h[None, :] * stride_rg_out + offs_d[:, None] * stride_rg_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        right_gate_acc+= tl.dot(norm_fp16, w_rg)

        # ----- output gate --------------------------------------------------
        w_og = tl.load(og_w_ptr + offs_h[None, :] * stride_og_out + offs_d[:, None] * stride_og_in,
                       mask=mask_h[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float16)
        out_gate_acc  += tl.dot(norm_fp16, w_og)

    # ------------------------------------------------------------------
    # 3️⃣  Sigmoid on gate tensors and final gating of left/right
    # ------------------------------------------------------------------
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    left_gate  = sigmoid(left_gate_acc)
    right_gate = sigmoid(right_gate_acc)
    out_gate   = sigmoid(out_gate_acc)

    left  = left_acc  * left_gate
    right = right_acc * right_gate

    # cast to FP16 for storage
    left_fp16      = left.to(tl.float16)
    right_fp16     = right.to(tl.float16)
    out_gate_fp16  = out_gate.to(tl.float16)

    # store results (shape [R, H])
    left_ptrs      = left_ptr + offs_m[:, None] * out_stride_r + offs_h[None, :] * out_stride_c
    right_ptrs     = right_ptr + offs_m[:, None] * out_stride_r + offs_h[None, :] * out_stride_c
    out_gate_ptrs  = out_gate_ptr + offs_m[:, None] * out_stride_r + offs_h[None, :] * out_stride_c

    tl.store(left_ptrs, left_fp16,
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(right_ptrs, right_fp16,
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptrs, out_gate_fp16,
             mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 2: LayerNorm (on cubic result) + output‑gate + final linear
# ----------------------------------------------------------------------
@triton.jit
def final_norm_gate_linear_fused_kernel(
    # inputs
    out_ptr,               # const * [R, H] (fp16)   ← cubic result
    out_gate_ptr,          # const * [R, H] (fp16)   ← output gate
    ln_w_ptr, ln_b_ptr,    # const * [H] (float32)   ← to_out_norm weight/bias
    final_w_ptr,           # const * [C, H] (fp16)   ← to_out weight
    # output
    out_flat_ptr,          # mutable * [R, C] (fp16)
    # runtime params
    R, H, C, eps,
    stride_out_r, stride_out_c,      # strides for out_ptr
    stride_gate_r, stride_gate_c,    # strides for out_gate_ptr
    stride_ln_w, stride_ln_b,        # usually 1
    stride_f_out, stride_f_in,       # strides for final weight (C×H)
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D_OUT: tl.constexpr,
):
    """Row‑wise LayerNorm → gate multiplication → linear projection."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # 1️⃣  Mean / variance across hidden dimension H
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_m[:, None] * stride_out_r + offs_h[None, :] * stride_out_c,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  For each output‑channel block: (norm * gate) @ final_weight
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(C, BLOCK_D_OUT)):
        offs_d = d_off * BLOCK_D_OUT + tl.arange(0, BLOCK_D_OUT)   # [BLOCK_D_OUT]
        mask_d = offs_d < C

        acc = tl.zeros([BLOCK_M, BLOCK_D_OUT], dtype=tl.float32)

        # loop over hidden‑dim blocks
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
            mask_h = offs_h < H

            # raw cubic output
            out_vals = tl.load(out_ptr + offs_m[:, None] * stride_out_r + offs_h[None, :] * stride_out_c,
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            # corresponding output gate
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * stride_gate_r + offs_h[None, :] * stride_gate_c,
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # layer‑norm on the cubic result
            norm = (out_vals - mean[:, None]) * rstd[:, None]
            gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
            beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]

            # gate multiplication
            v = norm * gate_vals                     # [M, H_block] FP32

            # final linear weight block (H_block × D_out_block) – transposed view
            w = tl.load(final_w_ptr + offs_h[:, None] * stride_f_in + offs_d[None, :] * stride_f_out,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float16)

            v_fp16 = v.to(tl.float16)
            acc += tl.dot(v_fp16, w)                # [M, D_out_block] FP32

        # store block of the final projection
        out_ptrs = out_flat_ptr + offs_m[:, None] * C + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, optional)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = 128 in all tests
    eps = 1e-5
    device = input_tensor.device

    B, N, N2, _ = input_tensor.shape
    assert N == N2
    R = B * N * N                                 # total rows = B·N·N

    # ------------------------------------------------------------------
    # flatten input → [R, dim] (contiguous FP32)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()

    # ------------------------------------------------------------------
    # allocate outputs of the first kernel
    # ------------------------------------------------------------------
    left      = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    # ------------------------------------------------------------------
    # stride information for all tensors
    # ------------------------------------------------------------------
    stride_xr, stride_xd = x_flat.stride()                     # (dim, 1)

    # layer‑norm parameters (float32)
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]

    # projection / gate weights (FP16, contiguous)
    lp_w = weights["left_proj.weight"].to(torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(torch.float16).contiguous()
    og_w = weights["out_gate.weight"].to(torch.float16).contiguous()

    stride_lp_out, stride_lp_in = lp_w.stride()   # (dim, 1)
    stride_rp_out, stride_rp_in = rp_w.stride()
    stride_lg_out, stride_lg_in = lg_w.stride()
    stride_rg_out, stride_rg_in = rg_w.stride()
    stride_og_out, stride_og_in = og_w.stride()

    # output strides (all three tensors share the same layout)
    out_stride_r, out_stride_c = left.stride()    # (hidden_dim, 1)

    # ------------------------------------------------------------------
    # launch kernel 1 – fused LN + 5 linear layers + gating
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_D = 64
    BLOCK_H = 128

    grid = (triton.cdiv(R, BLOCK_M),)
    norm_proj_gate_fused_kernel[grid](
        x_flat,
        ln_w, ln_b,
        lp_w, rp_w, lg_w, rg_w, og_w,
        left, right, out_gate,
        R, dim, hidden_dim, eps,
        stride_xr, stride_xd,
        1, 1,                               # LN weight/bias have stride 1
        stride_lp_out, stride_lp_in,
        stride_rp_out, stride_rp_in,
        stride_lg_out, stride_lg_in,
        stride_rg_out, stride_rg_in,
        stride_og_out, stride_og_in,
        out_stride_r, out_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.view(R).to(torch.float16)
        left  = left * mask_flat[:, None]
        right = right * mask_flat[:, None]

    # ------------------------------------------------------------------
    # cubic multiplicative update – batched GEMM (B·H, N, N)
    # ------------------------------------------------------------------
    left_4d  = left.view(B, N, N, hidden_dim)
    right_4d = right.view(B, N, N, hidden_dim)

    # reshape to (B·H, N, N) → treat each head as independent batch
    left_mat  = left_4d.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_4d.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # BMM gives the cubic result O (still FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # flatten cubic result and output‑gate for kernel 2
    # ------------------------------------------------------------------
    out_flat      = out.view(R, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(R, hidden_dim).contiguous()
    final_out     = torch.empty(R, dim, dtype=torch.float16, device=device)

    # final layer‑norm parameters (float32)
    ln2_w = weights["to_out_norm.weight"]
    ln2_b = weights["to_out_norm.bias"]

    # final linear weight (FP16)
    final_w = weights["to_out.weight"].to(torch.float16).contiguous()
    stride_f_out, stride_f_in = final_w.stride()      # (hidden_dim, 1)

    # stride info for the cubic tensor & its gate
    out_stride_r, out_stride_c = out_flat.stride()        # (hidden_dim, 1)
    gate_stride_r, gate_stride_c = out_gate_flat.stride()

    # ------------------------------------------------------------------
    # launch kernel 2 – fused LN, gate multiplication and final linear
    # ------------------------------------------------------------------
    BLOCK_D_OUT = 64
    grid2 = (triton.cdiv(R, BLOCK_M),)
    final_norm_gate_linear_fused_kernel[grid2](
        out_flat,
        out_gate_flat,
        ln2_w, ln2_b,
        final_w,
        final_out,
        R, hidden_dim, dim, eps,
        out_stride_r, out_stride_c,
        gate_stride_r, gate_stride_c,
        1, 1,                       # LN2 weight/bias have stride 1
        stride_f_out, stride_f_in,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D_OUT=BLOCK_D_OUT,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # reshape back to [B, N, N, dim] and cast to float32 (reference dtype)
    # ------------------------------------------------------------------
    result = final_out.view(B, N, N, dim).to(torch.float32)
    return result
