def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    total_rows = B*N*N

    # LayerNorm: same kernel as earlier for input

    # fused proj and gate
    w_4h = torch.cat([...], dim=0).to(torch.float16)
    projected = torch.nn.functional.linear(x_norm, w_4h) # shape (R,4*H)

    left_proj = projected[:, :hidden_dim]
    right_proj = projected[:, hidden_dim:2*hidden_dim]
    left_gate_raw = projected[:, 2*hidden_dim:3*hidden_dim]
    right_gate_raw = projected[:, 3*hidden_dim:]

    out_gate_raw = torch.nn.functional.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    # mask optionally
    if not config.get("nomask", True):
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # gating projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape to B,N,N,H (fp16)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # cubic update
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # second LayerNorm
    out_flat = out.reshape(total_rows, hidden_dim)
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # final gating + linear
    gated = out_norm * out_gate.view(B, N, N, hidden_dim)
    final = torch.nn.functional.linear(gated.view(total_rows, hidden_dim), weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sum and sum of squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, 1)
        sum_sq  += tl.sum(vals * vals, 1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward (AlphaFold‑3) – Triton‑accelerated.

    Implements the “outgoing” TriMul operation:
        1. Row‑wise LayerNorm (FP32 → FP16) via a Triton kernel.
        2. Fused linear projection + gate generation (4·H × dim) using a single
           torch.nn.functional.linear call (FP16).
        3. Optional pairwise mask, sigmoid gating and element‑wise multiplication.
        4. Cubic update
           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
           computed with a memory‑efficient torch.einsum (no extra copies).
        5. Second LayerNorm over the hidden dimension H (Triton kernel).
        6. Apply the output gate and a final linear projection back to dim.

    All heavy arithmetic is performed in float16 for Tensor‑Core throughput,
    the final result is returned in float32 to match the reference module.
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim]
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program
    BLOCK_D = 128   # columns per tile (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # gamma (FP32)
        weights["norm.bias"],     # beta  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps (4·H × dim)
    # --------------------------------------------------------------
    # Concatenate weight matrices for the four linear layers.
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape (4·H, dim)

    proj_gate = F.linear(x_norm, w_4h)          # [R, 4·H] (FP16)

    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (float16)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] (FP16)
    right = right_proj_raw * right_gate        # [R, H] (FP16)

    # --------------------------------------------------------------
    # 4️⃣ reshape to (B, N, N, H) – ready for cubic update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ cubic update via einsum (no extra copies)
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate.view(B, N, N, H)    # (B,N,N,H) FP16

    final = F.linear(
        gated.view(total_rows, H),                 # [R, H]
        weights["to_out.weight"].to(torch.float16) # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim]            (float32) – gamma
    b_ptr,          # const  * [dim]            (float32) – beta
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):

    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The mean and variance are accumulated in FP32, the
    final normalized values are written as FP16.
    """
