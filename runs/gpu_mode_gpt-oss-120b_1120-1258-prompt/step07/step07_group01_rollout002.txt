@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0) # row tile index
    pid_n = tl.program_id(1) # col tile index
    pid_b = tl.program_id(2) # batch index

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_ptrs = A + pid_b * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + pid_b * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C + pid_b * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_triton(A,B, block_m=128, block_n=128, block_k=32):
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    grid = (triton.cdiv(M, block_m), triton.cdiv(N, block_n), batch)

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(1), A.stride(2), A.stride(0),
        B.stride(1), B.stride(2), B.stride(0),
        C.stride(1), C.stride(2), C.stride(0),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return C

"""
custom_kernel – outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1.  Row‑wise LayerNorm on the flattened [B·N·N, dim] input (FP32 → FP16) via a
    custom Triton kernel.
2.  Fuse the four linear projections (left/right + left/right gates) into a
    single half‑precision GEMM:
        X_norm @ W_cat.T  →  (R,4·H)   where  R = B·N·N  and  H = hidden_dim.
3.  Apply an optional mask, sigmoid gating and the element‑wise left/right
    multiplications.
4.  Reshape to [B, H, N, N] and perform the O(N³) multiplicative update:
        out_{b,h,i,j} = Σ_k left_{b,h,i,k} * right_{b,h,j,k}
    which is a batched matrix multiplication (B·H independent GEMMs) executed
    with a custom Triton kernel that works in FP16 with FP32 accumulation.
5.  Second LayerNorm over the hidden dimension (again Triton, FP32→FP16).
6.  Apply the output‑gate, a final linear projection back to *dim* and cast
    the result to FP32.

All heavy linear work is done in FP16 (Tensor‑Core friendly) and the two
LayerNorms are implemented in Triton, satisfying the “must‑use‑kernel”
requirement while keeping the code compact and correct for all test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim] (float32) – γ
    bias_ptr,       # const   * [dim] (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute per‑row mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                    # 1 / sqrt(var + eps)

    # ---- write normalized output (FP16) ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` [rows, dim] (FP32) → FP16.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# 2️⃣ Batched half‑precision GEMM (B·H independent matmuls) – N³ update
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col
    pid_b = tl.program_id(2)            # batch index

    # offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile for this batch
    a_ptrs = A_ptr + pid_b * stride_batch_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_batch_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # mask for the current k‑slice
        cur_k_offs = tl.arange(0, BLOCK_K) + cur_k
        mask_k = cur_k_offs < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)               # (BK, BN)

        acc += tl.dot(a, b)                                 # (BM, BN)

        # advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result (FP16)
    c_ptrs = C_ptr + pid_b * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor,
                         B: torch.Tensor,
                         block_m: int = 128,
                         block_n: int = 128,
                         block_k: int = 64) -> torch.Tensor:
    """
    Batched GEMM: (batch, M, K) @ (batch, K, N) → (batch, M, N)
    All inputs/outputs are FP16, accumulation is FP32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    grid = (triton.cdiv(M, block_m),
            triton.cdiv(N, block_n),
            batch)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(1),   # stride_am
        A.stride(2),   # stride_ak
        A.stride(0),   # stride_batch_a
        B.stride(1),   # stride_bk
        B.stride(2),   # stride_bn
        B.stride(0),   # stride_batch_b
        C.stride(1),   # stride_cm
        C.stride(2),   # stride_cn
        C.stride(0),   # stride_batch_c
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# 3️⃣ Full TriMul forward (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) – ignored if config["nomask"] is True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )                                                          # FP16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear (4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate projection & gate weight matrices (all are [H, dim])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [4·H, dim]

    # single GEMM: (R, dim) @ (dim, 4·H).T  → (R, 4·H)
    proj_gate = F.linear(x_norm, w_proj_gate)                # FP16

    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]            # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]        # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]        # [R, H]

    # separate linear for the output‑gate
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                    # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise products
    # --------------------------------------------------------------
    if not nomask:
        # mask has shape [B, N, N] → broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)             # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)       # [B, N, N, H]

    # bring hidden dim to axis‑1 → (B·H, N, N) and make contiguous
    left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # [B·H, N, N]

    # transpose the second operand (the “outgoing” version)
    right_bh_T = right_bh.transpose(-2, -1).contiguous()                # [B·H, N, N]

    # batched GEMM → (B·H, N, N)
    out_bh = _batched_gemm_triton(left_bh, right_bh_T,
                                  block_m=128, block_n=128, block_k=64)

    # reshape back to [B, N, N, H]
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (FP32 → FP16)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                       # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                                             # FP16, [R, H]
    out_norm = out_norm.view(B, N, N, H)                         # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Output gate + final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                               # element‑wise gate (FP16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),            # [dim, H]
    )                                                            # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)      # FP32 output
    return final
