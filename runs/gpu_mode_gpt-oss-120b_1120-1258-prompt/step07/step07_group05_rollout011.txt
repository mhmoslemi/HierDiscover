@triton.jit
def layernorm_mul_kernel(
    x_ptr,        # [total_rows, H] fp16
    gate_ptr,     # [total_rows, H] fp16
    y_ptr,        # [total_rows, H] fp16 output
    weight_ptr,   # [H] float32
    bias_ptr,     # [H] float32
    total_rows,
    H,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute row-wise mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptr = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass to write normalized * gate
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        # load x
        ptr = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        # load gate
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_d[None, :]
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = (norm * w[None, :] + b[None, :]) * gate_vals # shape [BLOCK_M,BLOCK_D]
        out_fp16 = out.to(tl.float16)
        ptr_out = y_ptr + offs_m[:, None] * H + offs_d[None, :]
        tl.store(ptr_out, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

    x, mask, weights, config = data

    # unpack config
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5
    B,N,_,_ = x.shape; total_rows = B*N*N
    H = hidden_dim
    device = x.device

    # 1: first LN
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256; BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"], # torch tensor float32
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # 2: fused projection + gates
    # concat five weight matrices
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],   # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    fused = F.linear(ln1_out, w_proj)   # [R, 5*H] fp16

    # 3: mask handling: expand to rows
    if not nomask and mask is not None:
        # mask shape [B, N, N]
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
    else:
        mask_flat = torch.ones(total_rows, 1,
                               dtype=torch.float16,
                               device=device)

    # 4: fused gating kernel. allocate left, right, out_gate
    left = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    BLOCK_M_G = 256
    BLOCK_D_G = 64
    grid_g = (triton.cdiv(total_rows, BLOCK_M_G),)

    fuse_proj_gate_kernel[grid_g](
        fused,
        mask_flat,
        left,
        right,
        out_gate,
        total_rows,
        H,
        HAS_MASK=1,   # always have mask
        BLOCK_M=BLOCK_M_G,
        BLOCK_D=BLOCK_D_G,
        num_warps=4,
    )
    # memory: left, right, out_gate are ready.

    # 5: reshape left and right into [B,N,N,H]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H)

    # 6: batch matmul: collapse B*H
    left_mat = left.view(B*H, N, N)   # [B*H,N,N]
    right_mat = right.view(B*H, N, N) # [B*H,N,N]
    # Note: we need right transpose on last two dims: right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H,N,N] fp16

    # 7: reshape out_mat to original shape [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 8: second LN + gate multiplication fused kernel
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    out_ln_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_ln_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # 9: final linear projection: from hidden H to dim
    final = F.linear(out_ln_gate, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # Cast back to float32
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

@triton.jit
def fuse_proj_gate_kernel(
    fused_ptr,          # const * [R, 5H] fp16
    mask_ptr,           # const * [R, 1] fp16
    left_ptr,           # mutable * [R, H] fp16
    right_ptr,          # mutable * [R, H] fp16
    out_gate_ptr,       # mutable * [R, H] fp16
    total_rows,         # int
    H,                  # int
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    total_cols = 5 * H

    # if we have mask, load mask per row
    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)[:, None]   # broadcast later
    else:
        mask_vals = None

    # iterate over column tiles
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # load left_raw
        left_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # load right_raw (offset H)
        right_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # load left_gate_raw (offset 2*H)
        left_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (2*H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # right_gate_raw
        right_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (3*H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # out_gate_raw
        out_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (4*H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # Sigmoid gates (fast approx)
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # Apply mask if present to left and right raw values before gating
        if HAS_MASK:
            left_raw = left_raw * mask_vals
            right_raw = right_raw * mask_vals

        # Compute gated values
        left = left_raw * left_gate
        right = right_raw * right_gate

        # Store left, right, out_gate
        tl.store(
            left_ptr + offs_m[:, None] * H + offs_d[None, :],
            left.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )
        tl.store(
            right_ptr + offs_m[:, None] * H + offs_d[None, :],
            right.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )
        tl.store(
            out_gate_ptr + offs_m[:, None] * H + offs_d[None, :],
            out_gate_val.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

"""
Highly‑optimized Triton + PyTorch implementation of the **outgoing** TriMul
operator (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel **dim** (FP32 → FP16).
2️⃣ Fuse the four linear projections (left/right) and the three gate
   projections (left‑gate/right‑gate/out‑gate) into a single GEMM:
   [R, dim] @ [5·H, dim].T → [R, 5·H] (R = B·N·N, H = hidden_dim).
3️⃣ One Triton kernel computes
   - left = (left_proj ⊙ sigmoid(left_gate))   (mask applied)
   - right = (right_proj ⊙ sigmoid(right_gate))
   - out_gate = sigmoid(out_gate_proj)
   All tensors are FP16 and the mask is broadcasted (all‑ones if no mask).
4️⃣ Reshape left/right to [B, H, N, N] and perform a batched matrix
   multiplication:  out = left @ rightᵀ   (cuBLAS `torch.bmm`).
5️⃣ A second Triton kernel fuses the hidden‑dim LayerNorm, the
   multiplication by `out_gate`, and the affine transformation
   (γ,β) into a single pass.
6️⃣ Final linear projection from hidden → model dimension (FP16 → FP32).

All heavy element‑wise work (both LayerNorms, gating, mask, and
post‑multiplicative LayerNorm) is performed in Triton; dense matrix
multiplications use cuBLAS Tensor‑Cores.  The implementation works for
any batch size {1,2}, sequence length {128…1024}, model dim {128,384}
and hidden_dim {128,384,768}.

The entry point is `custom_kernel((x, mask, weights, config))` which
returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]  (float32)
    y_ptr,          # mutable* [total_rows, dim]  (float16)
    weight_ptr,     # const  * [dim]              (float32)
    bias_ptr,       # const  * [dim]              (float32)
    total_rows,     # int    (runtime)
    dim,            # int    (runtime)
    eps,            # float  (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- 1️⃣ mean & variance (FP32 accumulation) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- 2️⃣ write normalized + affine (FP16) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        out_fp16 = out.to(tl.float16)
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fuse projections & gates, apply mask & sigmoid
# ----------------------------------------------------------------------
@triton.jit
def fuse_proj_gate_kernel(
    fused_ptr,      # const  * [R, 5*H]   (float16)
    mask_ptr,       # const  * [R]       (float16, all‑ones if no mask)
    left_ptr,       # mutable* [R, H]    (float16)
    right_ptr,      # mutable* [R, H]    (float16)
    out_gate_ptr,   # mutable* [R, H]    (float16)
    total_rows,     # int (runtime)
    H,              # int (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # mask broadcasted per column (1.0 when mask is disabled)
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)[:, None]

    total_cols = 5 * H                      # stride in the fused matrix

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # ---- left projection -------------------------------------------------
        left_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # ---- right projection ------------------------------------------------
        right_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # ---- left gate -------------------------------------------------------
        left_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (2 * H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # ---- right gate ------------------------------------------------------
        right_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (3 * H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # ---- out gate --------------------------------------------------------
        out_gate_raw = tl.load(
            fused_ptr + offs_m[:, None] * total_cols + (4 * H + offs_d)[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # sigmoid (in‑place, fp32 → fp16 later)
        left_gate   = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate  = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate    = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # apply mask to the *projections* before gating
        left_raw  = left_raw  * mask_val
        right_raw = right_raw * mask_val

        # gated values
        left  = left_raw  * left_gate
        right = right_raw * right_gate

        # store results (FP16)
        tl.store(
            left_ptr + offs_m[:, None] * H + offs_d[None, :],
            left.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :])
        tl.store(
            right_ptr + offs_m[:, None] * H + offs_d[None, :],
            right.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :])
        tl.store(
            out_gate_ptr + offs_m[:, None] * H + offs_d[None, :],
            out_gate.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over hidden dim + multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,          # const  * [R, H] (fp16)
    gate_ptr,       # const  * [R, H] (fp16)
    y_ptr,          # mutable* [R, H] (fp16)  <-- (LN(x) * gate)
    weight_ptr,     # const  * [H]   (float32) γ
    bias_ptr,       # const  * [H]   (float32) β
    total_rows,     # int (runtime)
    H,              # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- first pass: mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptr = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalize, affine, multiply by gate ---------------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # load activations
        ptr   = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals  = tl.load(ptr,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # load gate
        gate  = tl.load(gate_ptr + offs_m[:, None] * H + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = (norm * w[None, :] + b[None, :]) * gate   # (LN(x) * gate)

        tl.store(y_ptr + offs_m[:, None] * H + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]      (bool/float) or None
        - weights_dict : dict of all Linear/LayerNorm parameters
        - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        w["norm.weight"],          # γ  (float32)
        w["norm.bias"],            # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 3️⃣ fused projection + gate GEMM
    # --------------------------------------------------------------
    # concatenate (left_proj, right_proj, left_gate, right_gate, out_gate)
    proj_weight = torch.cat(
        [
            w["left_proj.weight"],    # [H, dim]
            w["right_proj.weight"],   # [H, dim]
            w["left_gate.weight"],    # [H, dim]
            w["right_gate.weight"],   # [H, dim]
            w["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [5·H, dim]

    fused = F.linear(ln1_out, proj_weight)      # [R, 5·H] fp16

    # --------------------------------------------------------------
    # 4️⃣ mask (broadcasted) – create all‑ones if we are in “nomask” mode
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows)
    else:
        mask_flat = torch.ones(total_rows,
                               dtype=torch.float16,
                               device=device)

    # --------------------------------------------------------------
    # 5️⃣ Triton kernel: apply mask, compute sigmoids, obtain gated tensors
    # --------------------------------------------------------------
    left  = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    grid_g = (triton.cdiv(total_rows, BLOCK_M),)
    fuse_proj_gate_kernel[grid_g](
        fused,
        mask_flat,
        left,
        right,
        out_gate,
        total_rows,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ batched matrix multiplication (outgoing update)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N]

    out_mat = torch.bmm(left_mat,
                        right_mat.transpose(1, 2))      # [B·H, N, N]   fp16

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ second LayerNorm + multiplication by out_gate (fused)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    out_ln_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_mul_kernel[(grid,)](
        out_flat,
        out_gate_flat,
        out_ln_gate,
        w["to_out_norm.weight"],     # γ (float32)
        w["to_out_norm.bias"],       # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 8️⃣ final linear projection (hidden → dim) + cast back to FP32
    # --------------------------------------------------------------
    final_fp16 = F.linear(out_ln_gate,
                          w["to_out.weight"].to(device=device, dtype=torch.float16))  # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
