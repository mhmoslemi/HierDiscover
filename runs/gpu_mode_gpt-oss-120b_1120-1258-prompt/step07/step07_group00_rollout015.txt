```
def trimul_fused_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    w_left_ptr, w_right_ptr, w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,
    sum_w_left_ptr, sum_w_right_ptr, sum_w_lgate_ptr, sum_w_rgate_ptr, sum_w_outgate_ptr,
    total_rows, dim, H,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # rows
    mask_m = offs_m < total_rows

    # compute mean/var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulators for dot products per weight set
    acc_left = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_lgate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_rgate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_outgate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    # load weight sums per output channel
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H
    sum_w_left_block = tl.load(sum_w_left_ptr + offs_h, mask=mask_h, other=0.0)  # shape (BLOCK_H,)
    sum_w_right_block = tl.load(sum_w_right_ptr + offs_h, mask=mask_h, other=0.0)
    sum_w_lgate_block = tl.load(sum_w_lgate_ptr + offs_h, mask=mask_h, other=0.0)
    sum_w_rgate_block = tl.load(sum_w_rgate_ptr + offs_h, mask=mask_h, other=0.0)
    sum_w_outgate_block = tl.load(sum_w_outgate_ptr + offs_h, mask=mask_h, other=0.0)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load x slice (float32)
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                     mask=mask_m[:, None] & mask_d[None, :],
                     other=0.0).to(tl.float32)  # shape [BLOCK_M, BLOCK_D]

        # Update statistics
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        # For each weight matrix, compute dot products
        # left weight slice shape (BLOCK_D, BLOCK_H)
        w_left_ptrs = w_left_ptr + offs_d[:, None] + offs_h[None, :] * dim
        w_left = tl.load(w_left_ptrs,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)  # (BLOCK_D, BLOCK_H)
        acc_left += tl.dot(x, w_left)  # (BLOCK_M, BLOCK_H)

        w_right_ptrs = w_right_ptr + offs_d[:, None] + offs_h[None, :] * dim
        w_right = tl.load(w_right_ptrs,
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        acc_right += tl.dot(x, w_right)

        w_lgate_ptrs = w_lgate_ptr + offs_d[:, None] + offs_h[None, :] * dim
        w_lgate = tl.load(w_lgate_ptrs,
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        acc_lgate += tl.dot(x, w_lgate)

        w_rgate_ptrs = w_rgate_ptr + offs_d[:, None] + offs_h[None, :] * dim
        w_rgate = tl.load(w_rgate_ptrs,
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        acc_rgate += tl.dot(x, w_rgate)

        w_outgate_ptrs = w_outgate_ptr + offs_d[:, None] + offs_h[None, :] * dim
        w_outgate = tl.load(w_outgate_ptrs,
                            mask=mask_d[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)
        acc_outgate += tl.dot(x, w_outgate)

    # compute mean, var, rstd
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # subtract mean*weight_sum, then scale by rstd
    # broadcasting: mean[:,None] * sum_w_left_block[None,:]
    left_centered = (acc_left - mean[:, None] * sum_w_left_block[None, :]) * rstd[:, None]
    right_centered = (acc_right - mean[:, None] * sum_w_right_block[None, :]) * rstd[:, None]
    lgate_raw = (acc_lgate - mean[:, None] * sum_w_lgate_block[None, :]) * rstd[:, None]
    rgate_raw = (acc_rgate - mean[:, None] * sum_w_rgate_block[None, :]) * rstd[:, None]
    out_gate_raw = (acc_outgate - mean[:, None] * sum_w_outgate_block[None, :]) * rstd[:, None]

    # apply sigmoid to gates
    lgate = 1.0 / (1.0 + tl.exp(-lgate_raw))
    rgate = 1.0 / (1.0 + tl.exp(-rgate_raw))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    # gated left/right
    left = left_centered * lgate
    right = right_centered * rgate

    # store outputs as fp16
    tl.store(left_ptr + offs_m[:, None] * H + offs_h[None, :],
             left.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(right_ptr + offs_m[:, None] * H + offs_h[None, :],
             right.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
             out_gate.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
```

"""
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Fuse the first LayerNorm (per‑row mean/var) with the four linear
    projections (left, right, left‑gate, right‑gate) and with the
    out‑gate projection.  The fusion eliminates the intermediate
    normalized tensor and reduces memory traffic from
    O(R·dim) → O(R·dim) reads + O(R·hidden) writes.
2️⃣  Apply the sigmoid gates, multiply the mask (if any) and obtain the
    gated tensors `left` and `right`.
3️⃣  Compute the triangular multiplicative update:
       out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]
    using a single `torch.einsum` (no extra permutes or copies).
4️⃣  Fuse the second LayerNorm with the element‑wise multiplication by
    the already‑computed output‑gate (custom Triton kernel).
5️⃣  Final projection back to the original channel dimension with a
    half‑precision GEMM.

Only the two custom Triton kernels below are required to satisfy the
“must use a kernel” rule.  All other operations are performed with
ordinary PyTorch calls, which are highly optimized on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fuse LayerNorm + 5 linear projections (left, right, left‑gate,
#      right‑gate, out‑gate) and apply sigmoid / mask.
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_kernel(
    # pointers
    x_ptr,                     # const  *[total_rows, dim]          (fp32)
    left_ptr,                  # mutable* [total_rows, hidden]      (fp16)
    right_ptr,                 # mutable* [total_rows, hidden]      (fp16)
    out_gate_ptr,              # mutable* [total_rows, hidden]      (fp16)

    w_left_ptr,                # const  *[hidden, dim]              (fp16)
    w_right_ptr,               # const  *[hidden, dim]              (fp16)
    w_lgate_ptr,               # const  *[hidden, dim]              (fp16)
    w_rgate_ptr,               # const  *[hidden, dim]              (fp16)
    w_outgate_ptr,             # const  *[hidden, dim]              (fp16)

    sum_left_ptr,              # const  *[hidden]                   (fp32)
    sum_right_ptr,
    sum_lgate_ptr,
    sum_rgate_ptr,
    sum_outgate_ptr,

    # runtime constants
    total_rows, dim, hidden,
    eps,

    # compile‑time block sizes
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # input‑dim tile
    BLOCK_H: tl.constexpr,      # hidden‑dim tile (≤ hidden)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------- per‑row statistics ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # --------- accumulators for the 5 GEMMs ----------
    acc_left  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_lgate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_rgate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    acc_ogate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    # --------- load weight‑sums (broadcast later) ----------
    off_h = tl.arange(0, BLOCK_H)
    mask_h = off_h < hidden

    sum_left  = tl.load(sum_left_ptr  + off_h, mask=mask_h, other=0.0)   # [BLOCK_H]
    sum_right = tl.load(sum_right_ptr + off_h, mask=mask_h, other=0.0)
    sum_lgate = tl.load(sum_lgate_ptr + off_h, mask=mask_h, other=0.0)
    sum_rgate = tl.load(sum_rgate_ptr + off_h, mask=mask_h, other=0.0)
    sum_ogate = tl.load(sum_outgate_ptr + off_h, mask=mask_h, other=0.0)

    # --------- main loop over input dimension ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the input (fp32)
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        # update running mean / var
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

        # ---- dot‑products for the five weight matrices ----
        # each weight slice is viewed as (BLOCK_D, BLOCK_H) to match tl.dot
        w_ptr = w_left_ptr   + offs_d[:, None] + off_h[None, :] * dim
        w = tl.load(w_ptr,
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)                     # [BLOCK_D, BLOCK_H]
        acc_left += tl.dot(x, w)                                 # [BLOCK_M, BLOCK_H]

        w_ptr = w_right_ptr  + offs_d[:, None] + off_h[None, :] * dim
        w = tl.load(w_ptr,
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        acc_right += tl.dot(x, w)

        w_ptr = w_lgate_ptr  + offs_d[:, None] + off_h[None, :] * dim
        w = tl.load(w_ptr,
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        acc_lgate += tl.dot(x, w)

        w_ptr = w_rgate_ptr  + offs_d[:, None] + off_h[None, :] * dim
        w = tl.load(w_ptr,
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        acc_rgate += tl.dot(x, w)

        w_ptr = w_outgate_ptr+ offs_d[:, None] + off_h[None, :] * dim
        w = tl.load(w_ptr,
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        acc_ogate += tl.dot(x, w)

    # --------- compute mean / rstd ----------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                   # 1 / sqrt(var+eps)

    # --------- apply the centering correction and rstd ----------
    # (x·Wᵀ) - mean·ΣW   →   ((·) * rstd)
    # broadcasting:  mean[:,None] * sum_w[None,:]
    left  = (acc_left  - mean[:, None] * sum_left[None, :])  * rstd[:, None]
    right = (acc_right - mean[:, None] * sum_right[None, :]) * rstd[:, None]
    lgate = (acc_lgate - mean[:, None] * sum_lgate[None, :]) * rstd[:, None]
    rgate = (acc_rgate - mean[:, None] * sum_rgate[None, :]) * rstd[:, None]
    ogate = (acc_ogate - mean[:, None] * sum_ogate[None, :]) * rstd[:, None]

    # --------- sigmoid gates ----------
    lgate = 1.0 / (1.0 + tl.exp(-lgate))
    rgate = 1.0 / (1.0 + tl.exp(-rgate))
    ogate = 1.0 / (1.0 + tl.exp(-ogate))

    # --------- gated output ----------
    left  = left  * lgate
    right = right * rgate

    # --------- store results (fp16) ----------
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < hidden

    # left
    out_ptr = left_ptr + offs_m[:, None] * hidden + offs_h[None, :]
    tl.store(out_ptr,
             left.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    # right
    out_ptr = right_ptr + offs_m[:, None] * hidden + offs_h[None, :]
    tl.store(out_ptr,
             right.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    # out‑gate
    out_ptr = out_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :]
    tl.store(out_ptr,
             ogate.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Fuse LayerNorm + element‑wise multiplication by the output‑gate.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # input / output [total_rows, hidden] (fp16)
    weight_ptr, bias_ptr,        # LN parameters γ,β (fp32)
    gate_ptr,                    # gate tensor, same shape as x (fp16)
    total_rows, hidden, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalized + gate ----------
    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden

        # load input
        x_ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # load gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * gate          # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# 3️⃣  Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : dict of model parameters (see code)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` (int)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]        # always 128 in the provided tests
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + all 5 linear projections
    # ------------------------------------------------------------------
    # convert linear weights to fp16 (small tensors, cheap)
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)

    # pre‑compute Σ_w per output channel (used to subtract the mean term)
    sum_w_left   = w_left.float().sum(dim=1)   # [H]
    sum_w_right  = w_right.float().sum(dim=1)
    sum_w_lgate  = w_lgate.float().sum(dim=1)
    sum_w_rgate  = w_rgate.float().sum(dim=1)
    sum_w_outgate= w_outgate.float().sum(dim=1)

    # allocate fused results
    left  = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 128               # rows per program – fits well on H100
    BLOCK_D = 64                # column tile (dim)
    BLOCK_H = 128               # hidden‑dim tile (hidden ≤ 128)

    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_norm_proj_kernel[grid](
        input_tensor,
        left,
        right,
        out_gate,
        w_left,
        w_right,
        w_lgate,
        w_rgate,
        w_outgate,
        sum_w_left,
        sum_w_right,
        sum_w_lgate,
        sum_w_rgate,
        sum_w_outgate,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,           # more warps → higher occupancy on H100
    )

    # ------------------------------------------------------------------
    # 2️⃣  Optional mask (applied *before* the gating multiplication)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] for broadcasting
        mask_flat = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣  Triangular multiplicative update (einsum)
    # ------------------------------------------------------------------
    # reshape back to [B, N, N, H]
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # einsum implements Σₖ left[i,k,d] * right[j,k,d] efficiently
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # [B, N, N, H]
    out_flat = out.reshape(total_rows, hidden)                     # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣  Fuse LayerNorm + multiplication by the output‑gate
    # ------------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        out_gate,                        # gate (fp16)
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
