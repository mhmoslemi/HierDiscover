"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) with a custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate,
   out‑gate) and the optional pair‑wise mask into a single Triton kernel.
   The kernel performs a tiny GEMM for each hidden‑channel tile, applies
   the sigmoid gates on‑the‑fly and writes the three tensors
   `left`, `right` and `out_gate` directly (no large intermediate buffer).
3️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
4️⃣ Row‑wise mean / rsqrt‑variance of `out` (FP16 → FP32) in a second
   Triton kernel.
5️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   linear projection into a single Triton kernel that writes the final
   `[B, N, N, dim]` tensor in FP32.

All kernels target Triton 3.3.1, run with 8 warps on an H100 and use
mixed‑precision arithmetic (FP16 compute, FP32 reductions) for best
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim

        ptr = x_ptr + offs_m[:, None] * dim + d_off[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim

        ptr = x_ptr + offs_m[:, None] * dim + d_off[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + d_off, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + d_off, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + d_off[None, :], out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                     # [R, dim]    fp16 input (already layer‑normed)
    left_ptr,                  # [B, H, N, N] fp16 output
    right_ptr,                 # [B, H, N, N] fp16 output
    out_gate_ptr,              # [B, H, N, N] fp16 output
    mask_ptr,                  # [R] optional mask (fp16) – only read if HAS_MASK
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode (b,i,k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    hidden_stride = N * N                       # stride between successive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    # Load scalar mask (broadcast over H) if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)  # [BLOCK_R]
    else:
        mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float32)

    # Loop over hidden‑channel tiles
    for h_tile in range(0, tl.cdiv(H, BLOCK_H)):
        h_off = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = h_off < H

        # accumulators for this tile (FP32)
        left_acc       = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        right_acc      = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        left_gate_acc  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        right_gate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        out_gate_acc   = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # Loop over the input dimension
        for d_tile in range(0, tl.cdiv(dim, BLOCK_D)):
            d_off = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = d_off < dim

            # X tile: [BLOCK_R, BLOCK_D]
            x_tile = tl.load(
                x_ptr + offs_r[:, None] * dim + d_off[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # Weight tiles are loaded transposed: shape [BLOCK_D, BLOCK_H]
            # left_proj
            left_w_tile = tl.load(
                left_w_ptr + h_off[None, :] * dim + d_off[:, None],
                mask=mask_h[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)
            # right_proj
            right_w_tile = tl.load(
                right_w_ptr + h_off[None, :] * dim + d_off[:, None],
                mask=mask_h[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)
            # left_gate
            left_gate_w_tile = tl.load(
                left_gate_w_ptr + h_off[None, :] * dim + d_off[:, None],
                mask=mask_h[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)
            # right_gate
            right_gate_w_tile = tl.load(
                right_gate_w_ptr + h_off[None, :] * dim + d_off[:, None],
                mask=mask_h[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)
            # out_gate
            out_gate_w_tile = tl.load(
                out_gate_w_ptr + h_off[None, :] * dim + d_off[:, None],
                mask=mask_h[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)

            # Dot products (FP32 accumulation)
            left_acc       += tl.dot(x_tile, left_w_tile)
            right_acc      += tl.dot(x_tile, right_w_tile)
            left_gate_acc  += tl.dot(x_tile, left_gate_w_tile)
            right_gate_acc += tl.dot(x_tile, right_gate_w_tile)
            out_gate_acc   += tl.dot(x_tile, out_gate_w_tile)

        # Apply mask and sigmoid gates (still FP32)
        mask_vec = mask_val[:, None]                       # [BLOCK_R, 1]

        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_acc))

        left  = (left_acc * mask_vec) * left_gate
        right = (right_acc * mask_vec) * right_gate
        outg  = out_gate

        # Write results (FP16)
        dst_off = base_offset[:, None] + h_off[None, :] * hidden_stride
        store_mask = mask_r[:, None] & mask_h[None, :]

        tl.store(left_ptr + dst_off, left.to(tl.float16),  mask=store_mask)
        tl.store(right_ptr + dst_off, right.to(tl.float16), mask=store_mask)
        tl.store(out_gate_ptr + dst_off, outg.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim columns (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out , mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                        # [M, K]

        term_fp16 = term.to(tl.float16)

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)   # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all FP32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------- unpack -------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ----------------------- 1️⃣ LayerNorm ------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ----------------------- 2️⃣ Fused projection + gates (no large buffer) ------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – never accessed when HAS_MASK == 0
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # fp16 copies of the linear weights (H, dim)
    left_w       = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w      = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w  = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w = w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w   = w["out_gate.weight"].to(torch.float16).contiguous()

    BLOCK_R = 64
    BLOCK_H = 16
    BLOCK_D = 32

    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr, left_w, right_w, left_gate_w, right_gate_w, out_gate_w

    # ----------------------- 3️⃣ Batched GEMM (cuBLAS) ----------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ----------------------- 4️⃣ Row‑wise stats ----------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ----------------------- 5️⃣ Final LayerNorm + gate + projection --------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma (FP32)
        w["to_out_norm.bias"],     # beta  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
