@triton.jit
def apply_ln_gate_kernel(
    out_ptr,
    gate_ptr,
    mu_ptr,
    rstd_ptr,
    gamma_ptr,
    beta_ptr,
    dst_ptr,
    total_rows, H,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # row block
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # load mu and rstd
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate contiguous
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)

        # compute norm
        # diff = (out_vals - mu[:, None]) * rstd[:, None]
        diff = (out_vals - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]
        term = norm * gate_vals

        # store as fp16
        tl.store(dst_ptr + offs_m[:, None] * H + offs_k[None, :],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])
```

"""
TriMul (outgoing) – fast forward pass (no backward).

Algorithm (high‑level):
1️⃣ Row‑wise LayerNorm on the flattened pairwise tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM:  proj = x_norm @ W_fusedᵀ   → (R, 5·H).
3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   compute sigmoid gates and store three tensors `left`, `right`, `out_gate`
   in layout [B, H, N, N] (FP16, hidden stride = N·N).
4️⃣ Batched half‑precision GEMM (cuBLAS):   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k].
5️⃣ Row‑wise mean & rsqrt‑variance of `out` → μ, rstd.
6️⃣ Apply the second LayerNorm, the out_gate and the final projection in two steps:
   – A tiny Triton kernel normalises `out` with μ/rstd, multiplies by
     γ / β (to_out_norm) and by the gate, producing a contiguous matrix
     Z ∈ ℝ^{R×H} (FP16).
   – A cuBLAS GEMM (torch.nn.functional.linear) projects Z with the
     weight `to_out.weight` → final tensor of shape [B, N, N, dim] (FP32).

The implementation keeps the heavy matrix multiplications in cuBLAS (highly
optimised on H100) while the element‑wise parts are fused in Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim] input FP32
    y_ptr,           # [R, dim] output FP16
    w_ptr,           # [dim]   γ (FP32)
    b_ptr,           # [dim]   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R]        fp16 (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride of one hidden plane
    base_offset = b * (H * N * N) + i * N + k   # offset of first hidden element for (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise statistics (mean & rsqrt‑variance) of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] fp16
    mu_ptr,       # [R]          fp32
    rstd_ptr,     # [R]          fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # offset of first hidden plane (h=0)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Apply second LayerNorm, multiply by out_gate, and pack into a
#      contiguous [R, H] matrix (fp16)
# ----------------------------------------------------------------------
@triton.jit
def apply_ln_gate_kernel(
    out_ptr,          # [B, H, N, N]   fp16   (contiguous after permute → [R, H])
    gate_ptr,         # [B, H, N, N]   fp16   (same layout)
    mu_ptr,           # [R]            fp32
    rstd_ptr,         # [R]            fp32
    ln_w_ptr,         # [H]            fp32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            fp32   (to_out_norm.bias)
    dst_ptr,          # [R, H]         fp16   (output)
    total_rows, H,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # μ and rstd (per‑row statistics)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # contiguous loads because we will feed permuted tensors (shape [R, H])
        out_vals = tl.load(out_ptr  + offs_m[:, None] * H + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  fp32

        # (out - μ) * rstd → γ· + β → * gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]          # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]              # [M, K]
        term = norm * gate_vals                                   # [M, K]

        tl.store(dst_ptr + offs_m[:, None] * H + offs_k[None, :],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entrypoint – custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : keys –
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = cfg.get("eps", 1e-5)

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                   # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, compute sigmoids → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                         # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                        # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))       # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] fp16
    del left_mat, right_mat, left, right                         # free memory

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate (contiguous layout) + final projection
    # --------------------------------------------------------------
    # Permute to make hidden dimension contiguous: [B, N, N, H] → [R, H]
    out_perm = out.permute(0, 2, 3, 1).contiguous()           # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    Z = torch.empty(R, H, dtype=torch.float16, device=device)   # [R, H]

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    apply_ln_gate_kernel[grid_ln_gate](
        out_perm,
        out_gate_perm,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        Z,
        R,
        H,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=4,
    )
    del out_perm, out_gate_perm, mu, rstd   # free intermediate buffers

    # Final projection with cuBLAS (highly optimised)
    # to_out.weight has shape (dim, hidden_dim) → we can use torch.nn.functional.linear
    final_fp16 = F.linear(Z, w["to_out.weight"].to(torch.float16))   # [R, dim] fp16
    del Z

    # reshape to [B, N, N, dim] and cast to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
