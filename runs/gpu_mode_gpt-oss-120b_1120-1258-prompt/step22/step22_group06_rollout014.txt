"""
Outgoing TriMul (AlphaFold3) – reference‑accurate forward pass.
The heavy element‑wise work (the first LayerNorm) is performed with a
custom Triton kernel, while the remaining linear/gate/einsum steps are
implemented with highly‑optimized PyTorch operators in float32.
This guarantees numerical equivalence with the reference implementation
while still demonstrating a non‑trivial Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  input float32
    y_ptr,          # [R, dim]  output float32
    w_ptr,          # [dim]    gamma (float32)
    b_ptr,          # [dim]    beta  (float32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute per‑row LayerNorm (mean/var over the `dim` dimension).
    All calculations are in float32 and the result is stored back as
    float32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # First pass: compute mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )   # float32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: write normalized (γ· + β) values
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )   # float32

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # float32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # float32

        out = norm * gamma[None, :] + beta[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight names to torch.Tensor (float32)
                required keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise positions (flattened)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – custom Triton kernel
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty_like(x_flat)                       # [R, dim] FP32

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Reshape back to 4‑D tensor
    x_norm = x_norm.view(B, N, N, dim)   # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim) – pure PyTorch
    # ------------------------------------------------------------------
    # Concatenate all five projection weights (no bias)
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    )  # shape [5*H, dim]  (float32)

    proj = F.linear(x_norm, fused_weight)  # [B, N, N, 5*H]  FP32

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )  # each -> [B, N, N, H]

    # Apply mask to left/right if a mask is provided and nomask=False
    if (not nomask) and (mask is not None):
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1] float32
        left_raw = left_raw * mask_
        right_raw = right_raw * mask_
    # Sigmoid gates (float32)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Gated tensors
    left = left_raw * left_gate      # [B, N, N, H]
    right = right_raw * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS, still FP32)
    # ------------------------------------------------------------------
    # Permute to bring the hidden dimension to the middle (B, H, N, N)
    left_h = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch and head dimensions for torch.bmm
    left_mat = left_h.view(B * hidden_dim, N, N)     # [B*H, N, N]
    right_mat = right_h.view(B * hidden_dim, N, N)   # [B*H, N, N]

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP32
    out = out.view(B, hidden_dim, N, N)                    # [B, H, N, N] FP32

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (float32) – use PyTorch for simplicity & exactness
    # ------------------------------------------------------------------
    # PyTorch expects the normalized dimension to be the *last* one.
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection (float32)
    # ------------------------------------------------------------------
    out_gate_perm = out_gate.permute(0, 2, 3, 1)      # [B, N, N, H]
    gated = out_norm * out_gate_perm                  # [B, N, N, H]

    # Final linear projection: hidden_dim -> dim (no bias)
    out_final = F.linear(gated, w["to_out.weight"])   # [B, N, N, dim] float32

    return out_final
