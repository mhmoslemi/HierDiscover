@triton.jit
def final_output_fused_kernel(
    out_ptr,          # [B, H, N, N] fp16
    gate_ptr,         # [B, H, N, N] fp16
    ln_w_ptr,         # [H] fp32
    ln_b_ptr,         # [H] fp32
    w_out_T_ptr,      # [H, dim] fp16
    final_ptr,        # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    # Determine index offsets
    pid_m = tl.program_id(0)  # rows chunk
    pid_n = tl.program_id(1)  # output dim tile

    # For each row in [0, total_rows)
    row_start = pid_m * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # output dim offset
    col_start = pid_n * BLOCK_N
    offs_n = col_start + tl.arange(0, BLOCK_N)
    mask_n = offs_n < dim

    # decode b,i,j from row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start to hidden plane 0

    # compute per-row sum and sum_sq over hidden dimension
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate hidden dimension in blocks for sum
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    # compute mean and rstd
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # now compute final projection accumulation
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # second loop over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate
        out_val = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                          mask=mask_m[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

        # fetch LN parameters for this hidden index
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # fp32
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # fp32

        # normalize and apply LN weights + bias
        norm = ((out_val - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        # apply gate
        gated = norm * gate_val

        # convert to fp16 for dot
        gated_fp16 = gated.to(tl.float16)

        # w_out_T tile
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)  # [K, N_tile]

        # dot: (M,K) * (K,N) => (M,N)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # store final result
    dst_ptr = final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) using a Triton kernel.
2️⃣  Fuse the 5 linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (torch linear).
3️⃣  Split the fused projection, optionally mask the left/right tensors,
    apply sigmoid gates and store three tensors (left, right, out_gate) in
    layout [B, H, N, N] (FP16) via a Triton kernel.
4️⃣  Batched half‑precision GEMM (cuBLAS)  out[b,h,i,j] = Σₖ left[b,h,i,k] *
    right[b,h,j,k] .
5️⃣  Fuse the second LayerNorm, out‑gate multiplication and the final
    linear projection (to_out) into a single Triton kernel.  The kernel
    first computes per‑row mean/variance of `out`, applies the LayerNorm
    (γ,β), multiplies by the gate, and finally projects to `dim` using
    the transposed weight `to_out.weight`.  The result is a float‑32
    tensor [B, N, N, dim].

Only the heavy‑compute parts are written in Triton; the remaining
operations (GEMMs) use cuBLAS which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim] input FP32
    y_ptr,           # [R, dim] output FP16
    w_ptr,           # [dim]   gamma (FP32)
    b_ptr,           # [dim]   beta  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- pass 1: mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- pass 2: write normalized tensor ---------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R]        fp16 (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask broadcasted over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                 # stride of one hidden plane
    base_offset = b * (H * N * N) + i * N + k   # offset of first hidden element for (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_fused_kernel(
    out_ptr,          # [B, H, N, N]   fp16
    gate_ptr,         # [B, H, N, N]   fp16      (out_gate)
    ln_w_ptr,         # [H]            fp32      (to_out_norm.weight)
    ln_b_ptr,         # [H]            fp32      (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       fp16      (transposed to_out.weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden plane 0 for each row

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & rsqrt‑variance over hidden dim H
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Apply second LN, multiply by out_gate and project to dim
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden activation and its gate
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters (γ,β) – FP32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # apply out_gate
        gated = norm * gate_vals

        # convert to fp16 for the final GEMM
        gated_fp16 = gated.to(tl.float16)

        # load a tile of the final projection matrix Wᵀ : (K, dim)
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                               # [K, TILE_N]

        # dot product (FP16·FP16 → FP32) and accumulate
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # --------------------------------------------------------------
    # Store result: [B, N, N, dim]  (float32)
    # --------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H) – all linear maps in one GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H)  fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                         # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                        # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] fp16

    # left/right are no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ
        w["to_out_norm.bias"],     # β
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    # clean‑up temporaries
    del out, out_gate, w_out_T

    return final
```
