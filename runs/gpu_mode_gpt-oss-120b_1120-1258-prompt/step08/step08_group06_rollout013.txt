def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_l_b: tl.constexpr, stride_l_i: tl.constexpr, stride_l_k: tl.constexpr, stride_l_h: tl.constexpr,
    stride_r_b: tl.constexpr, stride_r_j: tl.constexpr, stride_r_k: tl.constexpr, stride_r_h: tl.constexpr,
    stride_o_b: tl.constexpr, stride_o_i: tl.constexpr, stride_o_j: tl.constexpr, stride_o_h: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids:
    pid_bh = tl.program_id(0)    # combined batch*H index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch index and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j within N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    # For left: shape (B,N,N,H) => index: (b, i, k, h)
    # pointer offset: b*stride_l_b + i*stride_l_i + k*stride_l_k + h*stride_l_h
    # Since we will compute loads with offsets for k tile.
    base_l = left_ptr + b * stride_l_b + h * stride_l_h

    # For right: shape (B,N,N,H) => (b, j, k, h)
    base_r = right_ptr + b * stride_r_b + h * stride_r_h

    # For out: shape (B,N,N,H) => (b, i, j, h)
    base_o = out_ptr + b * stride_o_b + h * stride_o_h

    # Initialize accumulator

"""
Optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  

The forward pass consists of:
1️⃣ Row‑wise LayerNorm on the 4‑D pair representation (FP32 → FP16).
2️⃣ Fused linear projection + four gate projections (single GEMM, FP16).
3️⃣ Optional mask, sigmoids and element‑wise gating.
4️⃣ Cubic update  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realised as a batched GEMM over the (i,k) and (j,k) planes.
   This is performed by a custom Triton kernel that works directly on the
   (B,N,N,H) layout – no extra transpose or copy is required.
5️⃣ Second row‑wise LayerNorm over the hidden dimension H.
6️⃣ Gating, final linear projection back to the original feature dimension
   and cast to FP32.

All heavy arithmetic stays in FP16 (accumulated in FP32) and the only
memory copies are the two LayerNorm passes and the final output tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,           # const   * [rows, dim]  (float32)
    y_ptr,           # mutable * [rows, dim]  (float16)
    w_ptr,           # const   * [dim]       (float32) – γ
    b_ptr,           # const   * [dim]       (float32) – β
    rows,            # int32
    dim,             # int32
    eps,             # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ---------- write normalized values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Batched GEMM for the cubic update (no extra transpose)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # const * [B, N, N, H] (fp16)
    right_ptr,  # const * [B, N, N, H] (fp16)
    out_ptr,    # mutable * [B, N, N, H] (fp16)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides (in elements) of the three tensors
    stride_l_b: tl.constexpr,
    stride_l_i: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_l_h: tl.constexpr,
    stride_r_b: tl.constexpr,
    stride_r_j: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_r_h: tl.constexpr,
    stride_o_b: tl.constexpr,
    stride_o_i: tl.constexpr,
    stride_o_j: tl.constexpr,
    stride_o_h: tl.constexpr,
    BLOCK_M: tl.constexpr,   # tile size on i‑axis
    BLOCK_N: tl.constexpr,   # tile size on j‑axis
    BLOCK_K: tl.constexpr,   # reduction tile size on k‑axis
):
    # program ids
    pid_pair = tl.program_id(0)   # combined (b, h)
    pid_i    = tl.program_id(1)   # tile row   (i)
    pid_j    = tl.program_id(2)   # tile col   (j)

    # decode batch & hidden index
    b = pid_pair // H
    h = pid_pair %  H

    # offsets of the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the fixed (b,h) pair
    base_l = left_ptr  + b * stride_l_b  + h * stride_l_h
    base_r = right_ptr + b * stride_r_b  + h * stride_r_h
    base_o = out_ptr   + b * stride_o_b  + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (M, K)
        ptr_l = base_l \
                + offs_i[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        a = tl.load(ptr_l,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right block (transposed on‑the‑fly): shape (K, N)
        # we load as (K, N) so that tl.dot can be used directly
        ptr_r = base_r \
                + offs_k[:, None] * stride_r_k \
                + offs_j[None, :] * stride_r_j
        b = tl.load(ptr_r,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    ptr_o = base_o \
            + offs_i[:, None] * stride_o_i \
            + offs_j[None, :] * stride_o_j
    tl.store(ptr_o,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N] (bool/float) – optional
        - weights_dict : contains all linear / LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_R = 256          # rows per program
    BLOCK_D = 128          # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps  (4·H × dim)
    # --------------------------------------------------------------
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4·H]  fp16
    # split
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask & sigmoid gates
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update – custom batched GEMM (no copies)
    # --------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides in elements (torch reports strides in *elements*)
    sl = left.stride()
    sr = right.stride()
    so = out.stride()
    # launch grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b = sl[0],
        stride_l_i = sl[1],
        stride_l_k = sl[2],
        stride_l_h = sl[3],
        stride_r_b = sr[0],
        stride_r_j = sr[1],
        stride_r_k = sr[2],
        stride_r_h = sr[3],
        stride_o_b = so[0],
        stride_o_i = so[1],
        stride_o_j = so[2],
        stride_o_h = so[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16
    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
