@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass store normalized * gate
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptr_tile = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_ptr_tile = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]  # gate same shape
        x_vals = tl.load(x_ptr_tile, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        g_vals = tl.load(g_ptr_tile, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]  # fp32
        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        # norm*gamma + beta
        out = norm * gamma[None, :] + beta[None, :]  # fp32
        out = out * g_vals  # apply gate
        out_fp16 = out.to(tl.float16)
        y_ptr_tile = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptr_tile, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

"""
custom_kernel – highly‑optimized outgoing TriMul (AlphaFold3)

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input channel dim (float32→float16) –
    custom Triton kernel `layernorm_kernel`.

2️⃣  Fuse the four projection / gate linear layers *and* the output‑gate
    linear into a *single* FP16 GEMM:
        X_norm @ W_fusedᵀ  →  [R, 5·H]   (R = B·N·N)
    Split the result into
        left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw.

3️⃣  Apply optional binary mask to `left_raw` / `right_raw`,
    take sigmoid of the three gate tensors and multiply:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)
        out_gate = sigmoid(out_gate_raw)

4️⃣  Multiplicative update (batched GEMM):
        out[b,h] = left[b,h] @ right[b,h]ᵀ
    (implemented with `torch.bmm` on FP16 tensors).

5️⃣  Fuse the second LayerNorm and the output‑gate multiplication into
    one Triton kernel `layernorm_gate_kernel`:
        out_norm = ((out - μ) / √(σ²+ε))·γ + β
        out_gated = out_norm * out_gate

6️⃣  Final linear projection (FP16 GEMM) with weight `to_out.weight`.

All intermediate tensors are kept in FP16; the final result is cast back
to FP32 to match the original module.  The implementation reduces the
number of kernel launches (two kernels only) and eliminates an extra
linear pass for the output‑gate, yielding a substantial speed‑up while
remaining fully‑compatible with the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)  γ
    bias_ptr,         # const   * [dim]               (float32)  β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute per‑row mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # ----- write normalized values -----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float16)
    gate_ptr,         # const   * [total_rows, dim]   (float16)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32) γ
    bias_ptr,         # const   * [dim]               (float32) β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)  # 1/√(var+eps)

    # ----- normalize, apply γ/β, then element‑wise gate --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw input and gate
        x_tile = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)   # fp32
        g_tile = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)   # fp32

        # layer‑norm
        norm = (x_tile - mean[:, None]) * rstd[:, None]       # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                  # fp32

        # element‑wise gate
        out = out * g_tile                                   # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
        - config_dict  : dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True -> no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,j) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim], fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned)
    BLOCK_D = 32    # columns per iteration (tuned)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four projections + the output‑gate into a single GEMM
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]  <-- added
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    # linear: [R, dim] @ [5·H, dim]ᵀ -> [R, 5·H]  (FP16)
    fused = F.linear(x_norm_fp16, w_fused)      # fp16

    # split
    left_raw       = fused[:, 0:H]                       # [R, H]
    right_raw      = fused[:, H:2*H]                     # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]                   # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]                   # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any), sigmoid gates and combine
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]  fp16
    right = right_raw * right_gate        # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the batch GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)       # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()              # [R, H] fp16

    out_gated_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)

        out_flat,
        out_gate_fp16,
        out_gated_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        out_gated_fp16,
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
