@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, rows, cols,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows
    for c in range(0, tl.cdiv(cols, BLOCK_C)):
        col_start = c * BLOCK_C
        offs_c = col_start + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols
        a = tl.load(a_ptr + offs_r[:, None] * cols + offs_c[None, :],
                    mask=mask_r[:, None] & mask_c[None, :], other=0.0)
        b = tl.load(b_ptr + offs_r[:, None] * cols + offs_c[None, :],
                    mask=mask_r[:, None] & mask_c[None, :], other=0.0)
        tl.store(out_ptr + offs_r[:, None] * cols + offs_c[None, :],
                 a * b,
                 mask=mask_r[:, None] & mask_c[None, :])
```

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. First LayerNorm
    # reshape to (R, dim)
    x_flat = x.reshape(R, dim)
    x_norm = torch.nn.functional.layer_norm(
            x_flat,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps
        )
    # Cast to fp16
    x_norm_f16 = x_norm.to(torch.float16)

    # 2. Concatenated weight matrix for left/right projection + gates
    # convert each weight to fp16
    left_weight = weights["left_proj.weight"].to(torch.float16)
    right_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    w_concat = torch.cat([left_weight, right_weight,
                          left_gate_weight, right_gate_weight,
                          out_gate_weight], dim=0)  # shape (5*H, dim)

    # 3. Linear projection
    proj = torch.nn.functional.linear(x_norm_f16, w_concat)  # (R, 5*H)
    del x_norm_f16, w_concat

    # 4. Split tensors
    # slice indices
    start = 0
    left_raw = proj[..., start:start+H]
    start += H
    right_raw = proj[..., start:start+H]
    start += H
    left_gate_raw = proj[..., start:start+H]
    start += H
    right_gate_raw = proj[..., start:start+H]
    start += H
    out_gate_raw = proj[..., start:start+H]
    # free proj as needed later; we will keep slices references

    # mask handling: shape (B,N,N) -> reshape (R,1)
    if cfg.get("nomask", True):
        mask_flat = None
    else:
        mask_flat = mask.reshape(R, 1).to(torch.float16)   # broadcast

    # apply mask to left and right raw
    if mask_flat is not None:
        left_masked = left_raw * mask_flat
        right_masked = right_raw * mask_flat
    else:
        left_masked = left_raw
        right_masked = right_raw

    # apply gates with sigmoid
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_masked * left_gate
    right = right_masked * right_gate
    # free raw values if needed
    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # Reshape for batch gemm
    # left: (R, H) -> (B, N, N, H) -> (B, H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    # flatten batch
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # batched matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 5. Second layernorm
    out_flat = out.view(R, H)
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    ).to(torch.float16)   # cast to fp16
    # Apply out_gate (already fp16)
    # Out_gate shape (R, H) ; cast to fp16
    out_gate_f16 = out_gate.to(torch.float16)
    out_norm = out_norm * out_gate_f16

    # 6. Final linear
    final = torch.nn.functional.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)
    )  # (R, dim) fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
Tri‑multiplicative (outgoing) update – fast Triton+PyTorch implementation.

Algorithm
---------
1️⃣  Apply row‑wise LayerNorm on the input tensor (fp16) using the highly‑optimised
    torch.nn.functional.layer_norm implementation (cuDNN).

2️⃣  Concatenate the five weight matrices (left/right projections + three gates)
    and perform a single fused GEMM (R × dim) @ (5·H × dim)ᵀ → (R × 5·H)
    in fp16 with torch.nn.functional.linear (cuBLAS / Tensor‑cores).

3️⃣  Split the fused result, apply the optional pair‑wise mask, compute the three
    sigmoid gates and write the three tensors (left, right, out_gate) with a tiny
    Triton kernel.  This kernel is the only custom‑kernel component required by the
    task and is deliberately kept simple – the heavy work is still done by cuBLAS.

4️⃣  Reshape left/right to (B·H, N, N) and compute the batched matrix‑multiplication
    left @ rightᵀ → out (B·H, N, N) with torch.bmm (Tensor‑cores).

5️⃣  Second LayerNorm on the hidden representation (fp16) using torch.nn.functional.layer_norm,
    followed by element‑wise multiplication with the output‑gate.

6️⃣  Final linear projection back to the original dimension (fp16 → fp32).

The implementation keeps the memory‑traffic to a minimum:
‑ the input is never materialised in fp32 after the first LayerNorm,
‑ the large intermediate (R×5·H) is kept in fp16,
‑ only a single lightweight Triton kernel is launched for the mask & sigmoid
  handling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]     (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are performed in fp16 (sigmoid computed in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # optional mask (broadcasted later)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [ left, right, left_gate, right_gate, out_gate ]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H] offset for segment 0
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination pointers
        left_dst_ptr   = left_ptr   + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr  = right_ptr  + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (fp16)
    # ------------------------------------------------------------------
    # convert input to fp16 once and keep it that way
    x_fp16 = x.to(torch.float16).view(R, dim)  # [R, dim]  fp16
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # fp16, shape [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim columns)
    # ------------------------------------------------------------------
    # concatenate all five weight matrices (already fp16)
    left_w      = weights["left_proj.weight"].to(torch.float16)   # [hidden_dim, dim]
    right_w     = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w= weights["right_gate.weight"].to(torch.float16)
    out_gate_w  = weights["out_gate.weight"].to(torch.float16)

    w_concat = torch.cat(
        [left_w, right_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )   # [5*H, dim] in fp16

    # (R,dim) @ (5*H,dim)ᵀ  →  (R,5*H)   (fp16)
    proj = F.linear(x_norm, w_concat)   # fp16
    del w_concat, x_norm               # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates  (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask shape [B, N, N] → [R] (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr   # not needed any more

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update  (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] then move H into the batch axis
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back‑to [B, N, N, H]  (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    ).to(torch.float16)        # fp16

    # element‑wise multiply by the output‑gate (already fp16)
    out_norm = out_norm * out_gate
    del out_gate, out_flat, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
