"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the raw 4‑D tensor (B,N,N,dim).  The kernel runs on
   fp32 input and writes fp16 output.
2️⃣ Fuse the four linear projections (left/right + two gates) **and** the
   output‑gate linear layer into a single half‑precision GEMM.
3️⃣ (Optional) apply the pair‑wise mask, compute the three σ‑gates and
   multiply left/right with their gates.
4️⃣ Compute the N³ multiplicative update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision mat‑mul: reshape to (B·H, N, N) and call
   `torch.matmul`.  This hits a cuBLAS Tensor‑Core kernel.
5️⃣ Second LayerNorm over the hidden dimension (fp16 → fp16) via a Triton
   kernel.
6️⃣ Apply the output‑gate, a final linear projection back to the original
   channel dimension and return a float32 tensor.

All heavy linear work is performed in FP16; the only FP32 work is the two
LayerNorm reductions.  The implementation works for any batch size ≤ 2,
any sequence length from the test suite and both masked / un‑masked
variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (two‑pass reduction)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output in fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D fp32 tensor ``x`` of shape [rows, dim].
    Returns an fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) –
                         ignored if config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters
                         (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
        block_m=256,
        block_d=32,
    )                                                   # fp16, [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate (including out_gate) → fp16
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (4 for proj/gates + 1 for out_gate)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                     # [5·H, dim]

    fused_out = F.linear(x_norm, fused_w)                   # [R, 5·H] fp16

    left_proj_raw  = fused_out[:, 0:H]                     # [R, H]
    right_proj_raw = fused_out[:, H:2 * H]                 # [R, H]
    left_gate_raw  = fused_out[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = fused_out[:, 3 * H:4 * H]             # [R, H]
    out_gate_raw   = fused_out[:, 4 * H:5 * H]             # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) + sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask Tensor is [B, N, N]; broadcast to [R, 1]
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()          # inplace
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched FP16 GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]

    # Convert to (B, H, N, N) layout expected by cuBLAS
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )                                                   # fp16

    # Back to original layout [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ
        weights["to_out_norm.bias"],     # β
        eps=eps,
        block_m=256,
        block_d=32,
    )                                                   # fp16 [R, H]
    out_norm = out_norm.view(B, N, N, H)               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # fp16 * fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

**Explanation of key choices**

* **LayerNorm** – implemented in Triton to avoid the costly PyTorch reduction; it reads the input in FP32, computes mean/variance per row, writes the normalized result in FP16.
* **Fused projection** – the four linear projections plus the output‑gate linear layer are concatenated into a single GEMM (`F.linear`). This reduces the number of cuBLAS launches from two to one.
* **Mask handling** – the mask is applied only to the *left* and *right* projections before gating, exactly matching the reference implementation.
* **N³ update** – the outer‑product sum is expressed as a batched matrix‑multiply (`torch.matmul`) on tensors shaped `(B·H, N, N)`. cuBLAS/Tensor‑Cores on the H100 execute this extremely fast.
* **Second LayerNorm** – identical Triton kernel as the first norm but operates on the hidden dimension.
* **Final projection** – a half‑precision linear layer followed by a cast back to FP32, yielding the same numerical type as the original PyTorch module.

The kernel works for all configurations listed in the test suite and respects the `nomask` flag. All heavy arithmetic is performed in FP16, while the two LayerNorms keep FP32 precision for numerical stability. This design achieves a good balance between speed (by minimizing kernel launches and using Tensor‑Core GEMMs) and correctness.
