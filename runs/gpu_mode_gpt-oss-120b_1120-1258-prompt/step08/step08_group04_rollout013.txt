"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (`dim`).  Implemented with a
    custom Triton kernel (`layernorm_kernel`).  The kernel works on the
    flattened tensor [R, dim] where R = B·N·N.

2️⃣ Fuse the four linear projections (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`) into a single GEMM:
        X_norm @ W_fusedᵀ → [R, 4·H]  (H = hidden_dim)
    The result is split into raw projections and raw gates.

3️⃣ Compute the output‑gate with a second GEMM.

4️⃣ (Optional) apply the binary mask to the raw projections.

5️⃣ Apply *sigmoid* to the three gate tensors and combine them:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)

6️⃣ N³ multiplicative update (batched matrix‑multiply) → a custom Triton
    kernel (`batched_gemm_kernel`).  After reshaping,
        left  ∈ [B·H, N, N]       (fp16)
        right_T = right.transpose(-2, -1).contiguous()
    the kernel computes for each batch‑head:
        out = left @ right_T   (FP16 ➜ FP32 accumulation)

7️⃣ Row‑wise LayerNorm over the hidden dimension (`H`).  Same Triton kernel as
    step 1.

8️⃣ Multiply by the output‑gate, apply the final linear projection
    (fp16 GEMM) and cast the result back to float32.

All heavy tensor‑core work is in FP16; reductions (means/variances) are done
in FP32 to preserve numerical stability.  The implementation works for any
combination of batch size (1‑2), sequence length (128‑1024) and channel sizes
listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]      (fp32)
    y_ptr,               # mutable * [total_rows, dim]      (fp16)
    weight_ptr,          # const   * [dim]                  (fp32)   γ
    bias_ptr,            # const   * [dim]                  (fp32)   β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # [BLOCK_M, BLOCK_D] tile
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        y = norm * w[None, :] + b[None, :]                       # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            y.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for the N³ update
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,                     # fp16 pointers
    stride_batch_a, stride_batch_b, stride_batch_c,
    M, N, K,                     # matrix sizes (runtime)
    stride_aim, stride_aik,      # A row/col strides (within a batch)
    stride_bkj, stride_bjn,      # B row/col strides (within a batch) – B is [K, N]
    stride_cim, stride_cjn,      # C row/col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b] = A[b] @ B[b]  (A: M×K, B: K×N)   for b = 0 … batch‑1
    All tensors are fp16; accumulation is performed in fp32.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current batch
    # ------------------------------------------------------------------
    batch_offset_a = batch_id * stride_batch_a
    batch_offset_b = batch_id * stride_batch_b
    batch_offset_c = batch_id * stride_batch_c

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx   = tl.arange(0, BLOCK_K) + k_start
        mask_k = k_idx < K

        # A: [BLOCK_M, BLOCK_K]
        a_ptr = A + batch_offset_a \
                + offs_m[:, None] * stride_aim \
                + k_idx[None, :] * stride_aik
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B: [BLOCK_K, BLOCK_N]   (B is already transposed, i.e. K×N)
        b_ptr = B + batch_offset_b \
                + k_idx[:, None] * stride_bkj \
                + offs_n[None, :] * stride_bjn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back result (fp16)
    # ------------------------------------------------------------------
    c_ptr = C + batch_offset_c \
            + offs_m[:, None] * stride_cim \
            + offs_n[None, :] * stride_cjn
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict containing all Linear / LayerNorm parameters
            - config_dict  : dict with keys
                              * "dim" (int)
                              * "hidden_dim" (int)
                              * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim                    # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 32    # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],        # γ  (float32)
        weights["norm.bias"],          # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the four projection matrices → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)     # [R, 4·H]  (fp16)

    # Split the fused result
    left_raw        = fused[:, 0:H]                 # [R, H]
    right_raw       = fused[:, H:2*H]               # [R, H]
    left_gate_raw   = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw  = fused[:, 3*H:4*H]             # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device)   # [H, dim]
    )   # [R, H]  (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and combine
    # ------------------------------------------------------------------
    nomask = config.get("nomask", True)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (saves a temporary)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]   fp16
    right = right_raw * right_gate        # [R, H]   fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update – custom batched GEMM (Triton)
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N] and make the matrices contiguous
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose right for the mat‑mul (now shape [B, H, K, N] with K = N)
    right_T = right.transpose(-2, -1).contiguous()                    # [B, H, N, N]

    # Flatten batch and head dimensions → [B·H, N, N]
    batch_heads = B * H
    left_mat   = left.view(batch_heads, N, N).contiguous()
    right_mat  = right_T.view(batch_heads, N, N).contiguous()
    out_mat    = torch.empty_like(left_mat)   # fp16

    # Strides in **elements** (not bytes)
    stride_batch_a = left_mat.stride(0)
    stride_aim     = left_mat.stride(1)      # row stride (M)
    stride_aik     = left_mat.stride(2)      # col stride (K)

    stride_batch_b = right_mat.stride(0)
    stride_bkj     = right_mat.stride(1)     # row stride (K)
    stride_bjn     = right_mat.stride(2)     # col stride (N)

    stride_batch_c = out_mat.stride(0)
    stride_cim     = out_mat.stride(1)       # row stride (M)
    stride_cjn     = out_mat.stride(2)       # col stride (N)

    # Triton grid: (batch_head, tiles_m, tiles_n)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        batch_heads,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        N,          # M
        N,          # N
        N,          # K
        stride_aim,
        stride_aik,
        stride_bkj,
        stride_bjn,
        stride_cim,
        stride_cjn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise (fp16)

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] (fp16)

    # Cast back to FP32 – model expects float32
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
