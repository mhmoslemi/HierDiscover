"""
custom_kernel – highly‑optimized "outgoing" TriMul forward pass.

Algorithm (FP16 everywhere, only final output in FP32):
1️⃣ Row‑wise LayerNorm on the input tensor  (B·N·N rows, dim columns).
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM:  X_norm @ W_fusedᵀ → [R, 4·H]   (R = B·N·N).
3️⃣ Split the result, apply the three sigmoid gates and (optionally) the binary mask.
4️⃣ Reshape left/right to [B, H, N, N] and compute the N³ multiplicative update
   with a *single* Triton kernel that performs the batched matrix multiplication
   `out = left @ rightᵀ` for all (B·H) heads at once.
5️⃣ Row‑wise LayerNorm over the hidden dimension H.
6️⃣ Multiply by the output‑gate, apply the final linear projection and cast back to FP32.
All heavy math (LayerNorms & the batched GEMM) runs in Triton, while the cheap
linear layers use cuBLAS via `torch.nn.functional.linear`.  This design
eliminates the many small cuBLAS launches of the naive implementation and
fits the H100 Tensor‑Core performance envelope.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim]   (float32)
    y_ptr,        # mutable * [total_rows, dim]   (float16)
    w_ptr,        # const   * [dim]               (float32)  γ
    b_ptr,        # const   * [dim]               (float32)  β
    total_rows,   # int (runtime)
    dim,          # int (runtime)
    eps,          # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1️⃣ mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- 2️⃣ normalize & apply γ,β -------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ multiplicative update.
# Computes   C[b] = A[b] @ B[b]ᵀ   for all b in [0, total_heads).
# A, B, C : [total_heads, N, N] (row‑major, fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    total_heads, N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)          # tile‑row index
    pid_n = tl.program_id(1)          # tile‑col index
    pid_b = tl.program_id(2)          # which head (B·H)

    # offsets for this tile
    row_off = pid_m * BLOCK_M
    col_off = pid_n * BLOCK_N
    offs_m = row_off + tl.arange(0, BLOCK_M)
    offs_n = col_off + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    batch_offset_a = pid_b * stride_a_batch
    batch_offset_b = pid_b * stride_b_batch
    batch_offset_c = pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

                + offs_m[:, None] * stride_a_row
                + offs_k[None, :] * stride_a_col,

                + offs_n[None, :] * stride_b_col,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0).to(tl.float32)

                + offs_m[:, None] * stride_c_row \
                + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all linear / LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (batch, i, j) rows
    H = hidden_dim               # hidden dimension (often = dim, but not required)

    # --------------------------------------------------------------
    # 1️⃣ input LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 32    # columns per iteration
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ fuse left/right projections + their gates into ONE GEMM
    # --------------------------------------------------------------
    # concatenate [left_proj, right_proj, left_gate, right_gate] → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4·H, dim)

    fused = F.linear(x_norm_fp16, w_fused)    # (R, 4·H), fp16

    left_raw       = fused[:, 0:H]               # (R, H)
    right_raw      = fused[:, H:2 * H]           # (R, H)
    left_gate_raw  = fused[:, 2 * H:3 * H]       # (R, H)
    right_gate_raw = fused[:, 3 * H:4 * H]       # (R, H)

    # out‑gate is a separate linear
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device)  # (H, dim)
    )   # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates and element‑wise combine
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.view(R, 1).to(dtype=torch.float16, device=device)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # (R, H) fp16
    right = right_raw * right_gate        # (R, H) fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update – a single batched GEMM kernel
    # --------------------------------------------------------------
    # reshape to [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    total_heads = B * H
    left_mat  = left.view(total_heads, N, N).contiguous()   # (B·H, N, N)
    right_mat = right.view(total_heads, N, N).contiguous()  # (B·H, N, N)
    out_mat   = torch.empty_like(left_mat)                  # (B·H, N, N), fp16

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # tile rows
        triton.cdiv(N, BLOCK_N),   # tile cols
        total_heads                # one program per head
    )

    # strides for the three tensors (contiguous layout)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total_heads,
        N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # back to shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ second LayerNorm over hidden dimension H → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],  # γ (float32)
        weights["to_out_norm.bias"],    # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 6️⃣ apply output‑gate, final linear projection, cast to FP32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # (dim, H)
    )   # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)
    return final
