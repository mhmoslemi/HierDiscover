def gate_kernel(
   left_raw_ptr, right_raw_ptr,
   left_gate_raw_ptr, right_gate_raw_ptr,
   out_gate_raw_ptr,
   mask_ptr,
   left_ptr, right_ptr, out_gate_ptr,
   total_rows, hidden_dim,
   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # row indices
    mask_m = offs_m < total_rows

    for n_off in range(0, tl.cdiv(hidden_dim, BLOCK_N)):
        offs_n = n_off * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < hidden_dim

        # load
        left_raw = tl.load(left_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                           mask=mask_m[:, None] & mask_n[None, :],
                           other=0.0)
        right_raw = tl.load(right_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                           mask=mask_m[:, None] & mask_n[None, :],
                           other=0.0)
        left_gate_raw = tl.load(left_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                                mask=mask_m[:, None] & mask_n[None, :],
                                other=0.0)
        right_gate_raw = tl.load(right_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                                 mask=mask_m[:, None] & mask_n[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(out_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                               mask=mask_m[:, None] & mask_n[None, :],
                               other=0.0)

        # compute sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        left = left_raw * left_gate
        right = right_raw * right_gate

        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_m[:, None] * 1 + 0,
                               mask=mask_m[:, None],
                               other=1.0)   # mask as 0/1 fp16
            left = left * mask_val
            right = right * mask_val

        # write to outputs
        tl.store(left_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 left,
                 mask=mask_m[:, None] & mask_n[None, :])
        tl.store(right_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 right,
                 mask=mask_m[:, None] & mask_n[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 out_gate_val,
                 mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get('nomask', True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # First LayerNorm
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # Fused Linear for projections and gates
    # build fused weight: shape [5*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16)

    fused_out = torch.nn.functional.linear(x_norm, w_fused) # [R, 5*H] fp16

    # split raw outputs
    left_raw = fused_out[:, 0:H]
    right_raw = fused_out[:, H:2*H]
    left_gate_raw = fused_out[:, 2*H:3*H]
    right_gate_raw = fused_out[:, 3*H:4*H]
    out_gate_raw = fused_out[:, 4*H:5*H]

    # optional mask flatten for per-row scaling
    if not nomask and mask is not None:
        mask_flat = mask.view(total_rows).to(torch.float16)  # shape [R]
    else:
        mask_flat = None

    # gate kernel to compute left,right,out_gate after sigmoid & mask
    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)
    out_gate = torch.empty_like(out_gate_raw)

    BLOCK_N = 64  # columns per block for hidden dimension
    grid_gate = (triton.cdiv(total_rows, BLOCK_M),)
    has_mask = 0 if nomask else 1
    gate_kernel[grid_gate](
        left_raw, right_raw,
        left_gate_raw, right_gate_raw,
        out_gate_raw,
        mask_flat,
        left, right, out_gate,
        total_rows,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=has_mask,
        num_warps=4,
    )

    # Reshape for cubic update
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    # Batched matmul (N x N) @ (N x N)^T => [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # Apply out_gate elementwise (already fp16)
    out_norm = out_norm * out_gate.view(total_rows, H)

    # Final projection
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # reshape and cast to float32
    final = final.view(B, N, N, dim).to(torch.float32)
    return final

"""
TriMul (outgoing) – high‑performance forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (fp32 → fp16) – custom Triton kernel.
2️⃣ Single cuBLAS GEMM that produces the four projections (left/right) and the
   three gate vectors (left‑gate/right‑gate/out‑gate) in one shot:  [R, 5·H].
3️⃣ Triton kernel that (a) applies sigmoid to the three gate vectors,
   (b) multiplies the projections by their gates,
   (c) (optionally) masks the two projected tensors.
   This fuses three separate sigmoid kernels and the mask‑multiply pass.
4️⃣ Batched matrix‑multiplication (N×N @ N×Nᵀ) for the cubic update
   (uses torch.bmm, i.e. the highly‑optimized Tensor‑Core GEMM).
5️⃣ Row‑wise LayerNorm on the hidden tensor (same Triton kernel).
6️⃣ Element‑wise scaling by the already‑sigmoided out‑gate.
7️⃣ Final fp16‑linear projection back to “dim”.  The output is cast to fp32.

Only two custom kernels (the LayerNorm kernel is reused for both passes,
the gate kernel fuses three element‑wise operations) are required,
all heavy linear algebra stays in cuBLAS for maximal Tensor‑Core utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# 1️⃣ Row‑wise LayerNorm (used for the input and later for the hidden tensor)
# ----------------------------------------------------------------------------- #
@triton.jit
def layernorm_kernel(
    x_ptr,           # const  * [total_rows, dim]   (any fp16/fp32)
    y_ptr,           # mutable* [total_rows, dim]   (fp16 output)
    w_ptr,           # const  * [dim]               (γ, fp32)
    b_ptr,           # const  * [dim]               (β, fp32)
    total_rows,      # int (runtime)
    dim,             # int (runtime)
    eps,             # float (static)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------------- #
    # 1️⃣ Compute mean & variance (float32 accumulator)
    # --------------------------------------------------------------------- #
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # stride = dim
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp16/fp32 → fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # --------------------------------------------------------------------- #
    # 2️⃣ Normalise, apply γ/β and write result as fp16
    # --------------------------------------------------------------------- #
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # (x‑μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------------- #
# 2️⃣ Kernel that fuses the three sigmoid gates + optional mask multiplication
# ----------------------------------------------------------------------------- #
@triton.jit
def gate_kernel(
    left_raw_ptr,          # const * [R, H]   (fp16)
    right_raw_ptr,         # const * [R, H]   (fp16)
    left_gate_raw_ptr,     # const * [R, H]   (fp16)
    right_gate_raw_ptr,    # const * [R, H]   (fp16)
    out_gate_raw_ptr,      # const * [R, H]   (fp16)
    mask_ptr,              # const * [R]      (fp16) – 0/1, unused if HAS_MASK==0
    left_ptr,              # mutable * [R, H] (fp16)
    right_ptr,             # mutable * [R, H] (fp16)
    out_gate_ptr,          # mutable * [R, H] (fp16)
    total_rows,            # int (runtime)
    hidden_dim,            # int (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_N: tl.constexpr,   # cols per program (tiled over H)
    HAS_MASK: tl.constexpr,  # 1 → use mask, 0 → ignore mask
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    for n_off in range(0, tl.cdiv(hidden_dim, BLOCK_N)):
        offs_n = n_off * BLOCK_N + tl.arange(0, BLOCK_N)  # [BLOCK_N]
        mask_n = offs_n < hidden_dim

        # ---- load raw tensors ------------------------------------------------
        LR = tl.load(left_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                     mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        RR = tl.load(right_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                     mask=mask_m[:, None] & mask_n[None, :], other=0.0)

        LGR = tl.load(left_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                      mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        RGR = tl.load(right_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                      mask=mask_m[:, None] & mask_n[None, :], other=0.0)

        OGR = tl.load(out_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                      mask=mask_m[:, None] & mask_n[None, :], other=0.0)

        # ---- sigmoid ---------------------------------------------------------
        Lg = 1.0 / (1.0 + tl.exp(-LGR))
        Rg = 1.0 / (1.0 + tl.exp(-RGR))
        Og = 1.0 / (1.0 + tl.exp(-OGR))

        # ---- apply gates ------------------------------------------------------
        L = LR * Lg
        R = RR * Rg

        # ---- optional mask (broadcasted per‑row) -----------------------------
        if HAS_MASK:
            m = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # [BLOCK_M]
            m = m[:, None]                                          # broadcast over H
            L = L * m
            R = R * m

        # ---- store ------------------------------------------------------------
        tl.store(left_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 L, mask=mask_m[:, None] & mask_n[None, :])
        tl.store(right_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 R, mask=mask_m[:, None] & mask_n[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :],
                 Og, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# 3️⃣ Main entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing the six LayerNorm / Linear weights
        - config_dict  : dict with keys "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack & constants
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256          # rows per program – tuned for H100
    BLOCK_D = 64           # channels per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 2️⃣ One cuBLAS GEMM that produces left/right projections
    #    and the three gate vectors (left‑gate, right‑gate, out‑gate)
    # -----------------------------------------------------------------
    #   [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16)                  # [5·H, dim]

    fused = F.linear(x_norm, w_fused)          # fp16, shape [R, 5·H]

    # split the 5‑head result
    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]
    out_gate_raw   = fused[:, 4 * H:5 * H]

    # -----------------------------------------------------------------
    # 3️⃣ Fuse sigmoid, mask and gate multiplication in a single kernel
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.reshape(total_rows).to(torch.float16)   # [R]
    else:
        # dummy mask (all ones) – will be ignored at compile‑time
        mask_flat = torch.ones((1,), dtype=torch.float16, device=device)

    left      = torch.empty_like(left_raw)
    right     = torch.empty_like(right_raw)
    out_gate  = torch.empty_like(out_gate_raw)

    BLOCK_N = 64                         # tile over hidden dimension
    grid_gate = (triton.cdiv(total_rows, BLOCK_M),)
    HAS_MASK = 0 if nomask else 1

    gate_kernel[grid_gate](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        mask_flat,
        left,
        right,
        out_gate,
        total_rows,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 4️⃣ Cubic multiplicative update (batched matmul)
    # -----------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)       # keep for later scaling

    # bring H into the batch dimension → [B·H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # (N×N) @ (N×N)ᵀ  -> [B·H, N, N]  (fp16, Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Apply the output‑gate (still fp16) and final linear projection
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate.view(total_rows, H)   # element‑wise scaling

    final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16 → [R, dim]

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    return final.view(B, N, N, dim).to(torch.float32)
```
