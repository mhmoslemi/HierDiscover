```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and head
    b = batch_head // H
    h = batch_head % H

    # compute offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offsets for left and right
    # left and right share same strides
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # number of K tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # load right tile: shape (BLOCK_J, BLOCK_K)
        right_ptrs = right_base + offs_j[:, None] * stride_ri + offs_k[None, :] * stride_rk
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)

        # accumulate dot product
        # left_tile: (BLOCK_I, BLOCK_K)
        # right_tile: (BLOCK_J, BLOCK_K) need transpose
        acc += tl.dot(left_tile, right_tile.t())

    # cast to fp16
    out_tile = acc.to(tl.float16)

    # store to output tensor
    out_ptrs = out_ptr + b * stride_ob + h * stride_oh + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])
```

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation follows the reference PyTorch
module but fuses the expensive N³ multiplicative update with a custom
Triton kernel and removes the costly permute‑contiguous copies that
appear in a naïve implementation.

Main ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton kernel.
2️⃣ One fused FP16 GEMM for the four linear projections (left/right
   + their gates) + in‑place sigmoid‑gating, avoiding the creation of
   intermediate gate tensors.
3️⃣ Separate FP16 GEMM for the output‑gate.
4️⃣ Optional mask applied in‑place on the gated left/right tensors.
5️⃣ Custom Triton kernel for the N³ multiplicative update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   The kernel works directly on the non‑contiguous layout produced by
   the fused projection step, thus eliminating the expensive
   `permute(...).contiguous()` copies.
6️⃣ Second row‑wise LayerNorm over the hidden dimension (H) – Triton.
7️⃣ Element‑wise out‑gate, final projection, and cast back to FP32.
All heavy compute runs in FP16 with FP32 accumulation (Tensor‑Core
friendly) while the memory‑bound parts (LayerNorms, masking) are
fused into minimal‑copy Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (fp32→fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]  (float32)
    y_ptr,          # mutable * [total_rows, dim]  (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ N³ multiplicative update kernel (left × rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs:
    #   0 → batch * head dimension (B·H)
    #   1 → tile index along i
    #   2 → tile index along j
    batch_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head indices
    b = batch_head // H
    h = batch_head - b * H

    # tile offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current (b, h)
    base_left  = left_ptr  + b * stride_lb + h * stride_lh
    base_right = right_ptr + b * stride_rb + h * stride_rh

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   → shape (BLOCK_I, BLOCK_K)
        left_ptrs = base_left + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile  → shape (BLOCK_J, BLOCK_K)
        right_ptrs = base_right + offs_j[:, None] * stride_ri + offs_k[None, :] * stride_rk
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate dot product (BLOCK_I × BLOCK_J)
        acc += tl.dot(left_tile, right_tile.t())

    # store result as FP16
    out_ptrs = out_ptr + b * stride_ob + h * stride_oh \
               + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with LayerNorm/Linear parameters (float32)
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 0️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # block sizes – dim ≤ 384, choose a moderate tile size
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 1️⃣ Fused projection + four gates  (left, right, left_gate, right_gate)
    # --------------------------------------------------------------
    # concatenate the four weight matrices (each [H, dim])
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                                   # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    # fused GEMM: (R, dim) @ (dim, 4·H)ᵀ → (R, 4·H)
    fused = F.linear(ln1_out, w_proj)   # fp16, no bias

    # split into views (no copy)
    left_raw      = fused[:, 0:H]                 # (R, H) – left projection
    right_raw     = fused[:, H:2*H]               # (R, H) – right projection
    left_gate_raw = fused[:, 2*H:3*H]             # (R, H) – left gate pre‑sigmoid
    right_gate_raw= fused[:, 3*H:4*H]             # (R, H) – right gate pre‑sigmoid

    # in‑place sigmoid on gates
    left_gate_raw = left_gate_raw.sigmoid_()
    right_gate_raw = right_gate_raw.sigmoid_()

    # apply gates in‑place (still views into fused)
    left_raw.mul_(left_gate_raw)    # now holds gated left
    right_raw.mul_(right_gate_raw)  # now holds gated right

    # --------------------------------------------------------------
    # 2️⃣ Output‑gate (separate linear)
    # --------------------------------------------------------------
    out_gate = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    ).sigmoid_()                     # (R, H), fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1]  (fp16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw.mul_(mask_flat)
        right_raw.mul_(mask_flat)

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] (non‑contiguous layout)
    # --------------------------------------------------------------
    left = left_raw.view(B, N, N, H)      # shape (B,N,N,H) – stride = (N*N*4H, N*4H, 4H, 1)
    right = right_raw.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ N³ multiplicative update via custom Triton kernel
    # --------------------------------------------------------------
    out = torch.empty(B, N, N, H,
                      dtype=torch.float16,
                      device=device)

    # tile sizes – chosen to match Tensor‑Core friendly dimensions
    TILE_I = 64
    TILE_J = 64
    TILE_K = 32

    grid = (B * H,                     # batch * head dimension
            triton.cdiv(N, TILE_I),   # i tiles
            triton.cdiv(N, TILE_J))   # j tiles

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=left.stride(0),
        stride_li=left.stride(1),
        stride_lk=left.stride(2),
        stride_lh=left.stride(3),
        stride_rb=right.stride(0),
        stride_ri=right.stride(1),
        stride_rk=right.stride(2),
        stride_rh=right.stride(3),
        stride_ob=out.stride(0),
        stride_oi=out.stride(1),
        stride_oj=out.stride(2),
        stride_oh=out.stride(3),
        BLOCK_I=TILE_I,
        BLOCK_J=TILE_J,
        BLOCK_K=TILE_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (H)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # (R, H)
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to FP32
    # --------------------------------------------------------------
    # element‑wise gating (fp16)
    ln2_out = ln2_out * out_gate

    final_fp16 = F.linear(
        ln2_out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # (R, dim), fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
