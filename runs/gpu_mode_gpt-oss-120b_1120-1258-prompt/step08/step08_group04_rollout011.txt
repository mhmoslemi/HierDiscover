"""
custom_kernel – Outgoing TriMul (AlphaFold‑3)

Implementation notes
--------------------
* The heavy N³ multiplicative update is performed with a single
  batched fp16 GEMM (`torch.bmm`).  This fully exploits Tensor‑Core
  performance on H100.
* The first layer‑norm (over the input channel **dim**) is delegated to
  the highly‑optimised PyTorch `F.layer_norm`.  This removes a costly
  Triton launch for the largest tensor.
* The second layer‑norm (over the hidden dimension **hidden_dim**) is
  fused with the element‑wise output‑gate multiplication in a custom
  Triton kernel (`layernorm_gate_kernel`).  This saves one kernel
  launch and one memory pass.
* All linear projections are executed with fp16 weights (cast once) so
  the main compute stays in half‑precision; the final result is cast
  back to fp32 as required by the model.
* The implementation supports optional binary masks and works for any
  batch size, sequence length and channel sizes listed in the task
  description.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm + optional element‑wise gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,           # const   * [total_rows, dim]   (float16 or float32)
    y_ptr,           # mutable * [total_rows, dim]   (float16)
    weight_ptr,      # const   * [dim]               (float32)   γ
    bias_ptr,        # const   * [dim]               (float32)   β
    gate_ptr,        # const   * [total_rows, dim]   (float16)   g (output‑gate)
    total_rows,      # int (runtime)
    dim,             # int (runtime)
    eps,             # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm followed by element‑wise multiplication with a
    per‑element gate (the output‑gate).  The kernel works on fp16 data
    but performs the reduction in fp32 for numerical stability."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1. Compute mean / variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # ---------- 2. Normalize, apply γ/β and gate -----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load raw input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # layer‑norm
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # fp32

        # gate (out_gate) – stored as fp16, cast to fp32 for the mul
        g_ptr = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g = tl.load(g_ptr,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        out = out * g                                            # fp32

        # store result as fp16
        out_fp16 = out.to(tl.float16)
        y_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptr,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)   # True → no binary mask
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch implementation – fast & fully‑optimised)
    # ------------------------------------------------------------------
    # F.layer_norm works on the last dimension only.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + four gates into a single GEMM
    # ------------------------------------------------------------------
    # concat   [left_proj; right_proj; left_gate; right_gate]  →  [4*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, dim)

    # GEMM: (R, dim) @ (4*H, dim)ᵀ  →  (R, 4*H)
    fused = F.linear(x_norm.view(R, dim), w_fused)   # fp16

    # split
    left_raw       = fused[:, 0:H]                 # (R, H)
    right_raw      = fused[:, H:2 * H]             # (R, H)
    left_gate_raw  = fused[:, 2 * H:3 * H]         # (R, H)
    right_gate_raw = fused[:, 3 * H:4 * H]         # (R, H)

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm.view(R, dim),
        weights["out_gate.weight"].to(dtype=torch.float16, device=device),  # (H, dim)
    )   # (R, H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise combine
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B,N,N] → [R,1]  (broadcast over H)
        mask_fp16 = mask.to(torch.float16, device=device).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()          # inplace fp16 sigmoid
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()           # (R, H) fp16

    left  = left_raw * left_gate                  # (R, H) fp16
    right = right_raw * right_gate                # (R, H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update  – batched fp16 GEMM
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten batch & heads → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # re‑shape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # (R, H) fp16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Triton launch
    BLOCK_M = 256                                     # rows per program
    BLOCK_D = 32                                      # cols per loop (tuned)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,                 # x_ptr  (fp16)
        out_norm_gate,            # y_ptr  (fp16)
        weights["to_out_norm.weight"],   # γ  (fp32)
        weights["to_out_norm.bias"],     # β  (fp32)
        out_gate,                 # gate_ptr (fp16)
        R, dim=H, eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (fp16 GEMM) → fp32 output
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate.view(R, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device),  # (dim, H)
    )   # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # required dtype
    return final
