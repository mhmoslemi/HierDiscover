"""
Optimized outgoing TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (fp32 → fp16) using a Triton kernel.
2️⃣ Fuse the three linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single half‑precision GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ → [R, 4·H]
   Split the result into left_raw, right_raw, left_gate_raw, right_gate_raw.
3️⃣ Compute the output‑gate with a second half‑precision GEMM.
4️⃣ (Optional) apply the pairwise mask to left_raw/right_raw.
5️⃣ Apply sigmoid to the gate tensors (in‑place) and multiply:
        left = left_raw * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)
6️⃣ Reshape to [B, H, N, N] and perform the cubic update as a batched
   matrix multiplication:
        out[b·H+i] = left_mat @ right_matᵀ   (fp16, Tensor‑Core)
   where left_mat/right_mat are (N×N) matrices for each head.
7️⃣ Second row‑wise LayerNorm over the hidden dimension (fp16) via the same
   Triton kernel.
8️⃣ Multiply by the output‑gate, apply the final linear projection, and cast
   back to float32.

Only the two LayerNorms are hand‑written Triton kernels; all heavy linear
algebra uses PyTorch's highly‑optimized fp16 GEMM/BMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32 or fp16)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel works for fp32 or fp16 input; the reduction is done in fp32
    and the output is written as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1. mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        # Cast to fp32 for reduction (works for both fp16/fp32 input)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ---------- 2. write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ, β are fp32 – keep compute in fp32, cast at the end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32)
        - config_dict  : dict with keys
            * "dim"         – input channel dimension
            * "hidden_dim" – hidden dimension (H)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack & constants
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 64    # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right proj + gates into a single GEMM (FP16)
    # -----------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] in fp16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # -----------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # -----------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # -----------------------------------------------------------------
    # 4️⃣ (Optional) mask & sigmoid gates
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (broadcastable to [R, H])
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update via batched matmul
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring H into batch dimension → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch&head → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul: (N×N) @ (N×N)ᵀ → [B·H, N, N] (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast to float32
    # -----------------------------------------------------------------
    # element‑wise gate (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # final linear: [R, H] @ [dim, H]ᵀ → [R, dim]
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
