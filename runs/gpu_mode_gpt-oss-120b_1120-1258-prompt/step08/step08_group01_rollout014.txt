@triton.jit
def fused_ln_linear_kernel(
    x_ptr, weight_ptr, bias_ptr, out_ptr,
    total_rows, dim, out_features,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
    # strides
    stride_x_m, stride_x_k,
    stride_w_o, stride_w_k,
    stride_out_m, stride_out_o,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean/var per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :] * stride_x_k
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute output blocks.
    # Loop over output feature blocks.
    for o in range(0, tl.cdiv(out_features, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_features

        # init accumulator
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            x_ptrs = x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :] * stride_x_k
            x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
            # x is fp32 (tl.float32)
            # Normalize
            norm = (x - mean[:, None]) * rstd[:, None]  # fp32

            w_ptrs = weight_ptr + offs_o[None, :] * stride_w_o + offs_d[:, None] * stride_w_k  # shape [BLOCK_D, BLOCK_O]
            weight = tl.load(w_ptrs, mask=mask_d[:, None] & mask_o[None, :], other=0.0)  # fp32? but weight stored in fp16; we can load as fp16 and cast.
            # weight shape (BLOCK_D, BLOCK_O)

            # weight stored in fp16; cast to fp32.
            weight_fp32 = weight.to(tl.float32)

            # acc += dot over dim block: norm (BLOCK_M x BLOCK_D) dot weight (BLOCK_D x BLOCK_O)
            # use tl.dot
            acc += tl.dot(norm, weight_fp32)

        # Add bias
        if bias_ptr != 0:
            b = tl.load(bias_ptr + offs_o, mask=mask_o, other=0.0)  # fp32
            acc += b[None, :]

        # Store result (fp16 output)
        out_ptrs = out_ptr + offs_m[:, None] * stride_out_m + offs_o[None, :] * stride_out_o
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

@triton.jit
def fused_ln_gate_linear_kernel(
    inp_ptr,           # fp16 [R, H]
    gate_ptr,          # fp16 [R, H]
    ln_weight_ptr,     # fp32 [H]   // gamma
    ln_bias_ptr,       # fp32 [H]   // beta
    out_weight_ptr,    # fp16 [dim, H] // weight of final linear
    out_bias_ptr,      # fp32 [dim] (maybe unused)
    out_ptr,
    R, H, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # Compute mean/variance per row across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        inp_ptrs = inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_k   # stride_in_m = H, stride_in_k = 1
        vals = tl.load(inp_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over output columns (dim)
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(H, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < H

            # Load inp and gate
            inp_ptrs = inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_k
            inp = tl.load(inp_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # Normalize
            norm = (inp - mean[:, None]) * rstd[:, None]  # fp32

            # Apply LayerNorm weight+bias
            gamma = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
            beta = tl.load(ln_bias_ptr + offs_d, mask=mask_d, other=0.0)    # fp32
            norm = norm * gamma[None, :] + beta[None, :]  # fp32

            # Apply out_gate
            gate_ptrs = gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_k
            gate = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            term = norm * gate

            # Load weight out subblock
            w_ptrs = out_weight_ptr + offs_d[:, None] * stride_out_k + offs_o[None, :] * stride_out_n   # careful stride orientation

            # out_weight shape [dim, H] row dimension dim, col dimension H
            # Access weight[i_out, d] -> row index i_out, col d
            # So at offset d (col index) for row o block.
            # Strides: weight row stride = stride_out_n (maybe H?), weight dim stride = stride_out_k (maybe 1).
            # In PyTorch contiguous, out_weight stride (dim, H) is (H, 1)
            # So we compute pointer: out_weight_ptr + offs_d[ :, None] * stride_out_k + offs_o[None,:] * stride_out_n

            w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_o[None, :], other=0.0)  # fp16
            w_fp32 = w.to(tl.float32)

            acc += tl.dot(term, w_fp32)

        # Add bias if needed (output bias)
        if out_bias_ptr != 0:
            b = tl.load(out_bias_ptr + offs_o, mask=mask_o, other=0.0)  # fp32
            acc += b[None, :]

        # Store output
        out_ptrs = out_ptr + offs_m[:, None] * stride_out_m + offs_o[None, :] * stride_out_n
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

    mask_o = offs_o < out_dim

        mask_d = offs_d < hidden_dim

        beta = tl.load(ln_bias_ptr + offs_d, mask=mask_d, other=0.0)    # fp32

        # load weight for final linear

"""
Highly‑optimized forward pass of the **outgoing** TriMul operator
(AlphaFold‑3 style) using fused LayerNorm + Linear kernels written in
TRITON.  The implementation removes two separate dense GEMM calls and
eliminates the intermediate LayerNorm output tensors, reducing memory
traffic and improving runtime.

Algorithm
---------
1️⃣ Fuse the first LayerNorm (norm) with the four projections
    (left/right) and their two gates plus the output‑gate linear
    projection → a single TRITON kernel producing a tensor of shape
    [B·N·N, 5·H] in fp16.

2️⃣ Split the fused tensor, apply the optional mask, compute element‑wise
    sigmoids and weight the left/right tensors.

3️⃣ Perform the N³ multiplicative update with a batched `torch.bmm`
    (left × rightᵀ) – this part is already Tensor‑Core bound.

4️⃣ Fuse the second LayerNorm (to_out_norm), the output‑gate multiplication
    and the final projection (to_out) into a second TRITON kernel that
    directly writes the fp16 result of shape [B·N·N, C]; finally cast to
    fp32.

All heavy element‑wise work (both LayerNorms, mask handling,
sigmoids, and the large linear layers) is performed in custom kernels,
while the N³ matrix multiplication leverages cuBLAS/Tensor‑Core for
maximum throughput.

The kernels work for the full range of test configurations
(B∈{1,2}, N∈{128,256,512,1024}, C∈{128,384}, H=128) and support both
masked and un‑masked inputs.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fused LayerNorm + Linear (produces left/right + gates + out_gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_linear_kernel(
    x_ptr,                     # const float*            [R, D]
    y_ptr,                     # mutable half*          [R, O]
    ln_weight_ptr,             # const float*           [D]   (γ)
    ln_bias_ptr,               # const float*           [D]   (β)
    total_rows,                # int
    eps,                       # float32 scalar
    D: tl.constexpr,           # input feature dimension
    O: tl.constexpr,           # output feature dimension (5*H)
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # input feature tile
    BLOCK_O: tl.constexpr,     # output feature tile
    stride_x_m, stride_x_k,
    stride_w_o, stride_w_k,
    stride_y_m, stride_y_o,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Row‑wise mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        x_ptrs = x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :] * stride_x_k
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+eps)

    # -----------------------------------------------------------------
    # 2) Compute O‑dim outputs block‑wise
    # -----------------------------------------------------------------
    for o_off in range(0, tl.cdiv(O, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < O

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # load a tile of X
            x_ptrs = x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :] * stride_x_k
            x = tl.load(x_ptrs,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # normalize
            norm = (x - mean[:, None]) * rstd[:, None]          # [BM, BD]

            # LayerNorm scale & shift
            gamma = tl.load(ln_weight_ptr + offs_d,
                            mask=mask_d,
                            other=0.0)                         # fp32
            beta  = tl.load(ln_bias_ptr + offs_d,
                            mask=mask_d,
                            other=0.0)                         # fp32
            norm = norm * gamma[None, :] + beta[None, :]

            # weight tile
            w_ptrs = (   # weight shape [O, D]
                weight_ptr
                + offs_d[:, None] * stride_w_k
                + offs_o[None, :] * stride_w_o
            )
            w = tl.load(w_ptrs,
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)

            # accumulate dot product
            acc += tl.dot(norm, w)                              # [BM, BO]

        # store result (fp16)
        y_ptrs = y_ptr + offs_m[:, None] * stride_y_m + offs_o[None, :] * stride_y_o
        tl.store(y_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])

# ----------------------------------------------------------------------
# Kernel 2: fused second LayerNorm, output‑gate multiplication and final Linear
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_linear_kernel(
    inp_ptr,               # const half*           [R, H]
    gate_ptr,              # const half*           [R, H]   (out_gate after sigmoid)
    ln_weight_ptr,         # const float*          [H]      (γ₂)
    ln_bias_ptr,           # const float*          [H]      (β₂)
    out_weight_ptr,        # const half*           [C, H]   (to_out weight)
    out_ptr,               # mutable half*         [R, C]
    total_rows,            # int
    eps,                   # float32 scalar
    H: tl.constexpr,       # hidden dimension (input to second LN)
    C: tl.constexpr,       # final output dimension (dim)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
    stride_in_m, stride_in_k,
    stride_gate_m, stride_gate_k,
    stride_out_weight_k, stride_out_weight_n,
    stride_out_m, stride_out_n,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Row‑wise mean & variance of the hidden dimension (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        inp_ptrs = inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_k
        vals = tl.load(inp_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2) For each output channel block, compute LN → gate → final linear
    # -----------------------------------------------------------------
    for o_off in range(0, tl.cdiv(C, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < C

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(H, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < H

            # load hidden tensor
            inp_ptrs = inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_k
            inp = tl.load(inp_ptrs,
                          mask=mask_m[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)

            # normalize
            norm = (inp - mean[:, None]) * rstd[:, None]

            # LayerNorm γ₂ & β₂
            gamma = tl.load(ln_weight_ptr + offs_d,
                            mask=mask_d,
                            other=0.0)                     # fp32
            beta  = tl.load(ln_bias_ptr + offs_d,
                            mask=mask_d,
                            other=0.0)                     # fp32
            norm = norm * gamma[None, :] + beta[None, :]

            # load gate (out_gate after sigmoid)
            gate_ptrs = gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_k
            gate = tl.load(gate_ptrs,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            gated = norm * gate                               # fp32

            # load final projection weight tile
            w_ptrs = (
                out_weight_ptr
                + offs_d[:, None] * stride_out_weight_k
                + offs_o[None, :] * stride_out_weight_n
            )
            w = tl.load(w_ptrs,
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)

            acc += tl.dot(gated, w)                           # [BM, BO]
        # store result (fp16)
        out_ptrs = out_ptr + offs_m[:, None] * stride_out_m + offs_o[None, :] * stride_out_n
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])

# ----------------------------------------------------------------------
# Entrypoint – fully fused forward TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32)
        - config_dict  : { "dim": C, "hidden_dim": H, "nomask": bool, ... }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack & basic shapes
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]                # C
    hidden = cfg["hidden_dim"]      # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                    # total number of (i,j) pairs

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + all linear projections (5·H outputs)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each [H, C])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, C]
            weights["right_proj.weight"],  # [H, C]
            weights["left_gate.weight"],   # [H, C]
            weights["right_gate.weight"],  # [H, C]
            weights["out_gate.weight"],    # [H, C]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16).contiguous()        # [5·H, C]

    out_feat = hidden * 5                                            # O = 5·H

    # flatten input to [R, C] (float32)
    x_flat = x.view(R, dim).contiguous()

    # allocate output tensor for fused projection (fp16)
    fused_out = torch.empty(R, out_feat,
                            dtype=torch.float16,
                            device=device)

    # strides (in elements)
    stride_x_m = x_flat.stride(0)
    stride_x_k = x_flat.stride(1)
    stride_w_o = w_fused.stride(0)   # = C
    stride_w_k = w_fused.stride(1)   # = 1
    stride_y_m = fused_out.stride(0) # = out_feat
    stride_y_o = fused_out.stride(1) # = 1

    # block sizes (tuned for H100, fp16)
    BLOCK_M = 128
    BLOCK_D = 32
    BLOCK_O = 64

    grid = (triton.cdiv(R, BLOCK_M),)
    fused_ln_linear_kernel[grid](
        x_flat,
        fused_out,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R, eps,
        dim, out_feat,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        stride_x_m, stride_x_k,
        stride_w_o, stride_w_k,
        stride_y_m, stride_y_o,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Split fused tensor into the individual components
    # ------------------------------------------------------------------
    left_raw      = fused_out[:, 0:hidden]
    right_raw     = fused_out[:, hidden:2*hidden]
    left_gate_raw = fused_out[:, 2*hidden:3*hidden]
    right_gate_raw= fused_out[:, 3*hidden:4*hidden]
    out_gate_raw  = fused_out[:, 4*hidden:5*hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask application (mask shape [B,N,N] → [R,1])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise weighting
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # will be used later

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)           # [B,N,N,H]
    right = right.view(B, N, N, hidden)         # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, hidden)   # [B,N,N,H]

    # move hidden dimension to batched axis and flatten batch*head
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left_perm.view(B * hidden, N, N)          # [B·H, N, N]
    right_mat = right_perm.view(B * hidden, N, N)         # [B·H, N, N]

    # batched matrix multiplication (Tensor Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate multiplication and final Linear
    # ------------------------------------------------------------------
    out_flat   = out.view(R, hidden).contiguous()
    out_gate_flat = out_gate.view(R, hidden).contiguous()
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # strides for second kernel
    stride_in_m = out_flat.stride(0)         # = hidden
    stride_in_k = out_flat.stride(1)         # = 1
    stride_gate_m = out_gate_flat.stride(0)  # = hidden
    stride_gate_k = out_gate_flat.stride(1)  # = 1
    stride_out_weight_k = weights["to_out.weight"].to(device=device, dtype=torch.float16).stride(1)  # = 1
    stride_out_weight_n = weights["to_out.weight"].to(device=device, dtype=torch.float16).stride(0)  # = hidden
    stride_out_m = final_fp16.stride(0)      # = dim
    stride_out_n = final_fp16.stride(1)      # = 1

    # final projection weight (fp16)
    w_out_fp16 = weights["to_out.weight"].to(device=device, dtype=torch.float16).contiguous()

    BLOCK_M2 = 128
    BLOCK_D2 = 32
    BLOCK_O2 = 64

    grid2 = (triton.cdiv(R, BLOCK_M2),)
    fused_ln_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ₂  (float32)
        weights["to_out_norm.bias"],     # β₂  (float32)
        w_out_fp16,
        final_fp16,
        R, eps,
        hidden, dim,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        BLOCK_O=BLOCK_O2,
        stride_in_m, stride_in_k,
        stride_gate_m, stride_gate_k,
        stride_out_weight_k, stride_out_weight_n,
        stride_out_m, stride_out_n,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 and restore original layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
