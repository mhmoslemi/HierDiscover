@triton.jit
def fused_norm_linear_kernel(
    x_ptr, y_ptr,
    weight_ptr,
    ln_weight_ptr,
    ln_bias_ptr,
    total_rows, dim, out_dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass sum
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # number of output blocks
    num_o_blocks = tl.cdiv(out_dim, BLOCK_O)

    # loop over each output block
    for o_off in range(0, out_dim, BLOCK_O):
        # allocate accumulator for this block
        cur_o = tl.minimum(BLOCK_O, out_dim - o_off)
        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        # second pass for dot product
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

"""
custom_kernel – high‑performance “outgoing” Triangle‑Multiplicative‑Update (TriMul).

Algorithm (inference only)
--------------------------
1️⃣ Row‑wise LayerNorm over the last channel dimension (dim).  The
    mean/variance are accumulated in FP32, the result is written in FP16.

2️⃣ A single fused linear layer produces **four** hidden projections
    (left/right and their two gates) **plus** the out‑gate.  The
    concatenated weight matrix has shape (5·hidden_dim, dim) and all
    four projections share the same normalised input, therefore only one
    GEMM is required.

3️⃣ Optional binary mask is applied to the two *value* projections
    (left/right) before the element‑wise gates.

4️⃣ The gated value tensors are reshaped to (B·H, N, N) and a batched
    matrix‑multiply (B·H independent GEMMs) yields the N³ multiplicative
    update:
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

5️⃣ A fused kernel performs a second LayerNorm over the hidden
    dimension **and** multiplies the result by the out‑gate (already
    passed through a sigmoid).

6️⃣ Final linear projection back to the original channel size.

All heavy‑weight memory‑bound work (the two LayerNorms) is written in
pure Triton.  The remaining dense matrix multiplications use the
highly‑optimised cuBLAS kernels that torch provides.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32) – γ
    bias_ptr,       # const   * [dim]               (float32) – β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ compute mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalised values γ·(x−μ)·rstd + β (FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # still FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel – fused LayerNorm (over hidden dim) + element‑wise gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float16)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32) – γ
    bias_ptr,       # const   * [dim]               (float32) – β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # mean / variance (no gate)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # normalise, apply γ/β and finally multiply by the gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # input activation
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # gate (same shape as x)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = (norm * w[None, :] + b[None, :]) * g_vals   # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom TriMul forward (outgoing direction)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : mapping with all Linear/LayerNorm parameters
        * config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}
    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack everything
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim
    OUT_DIM = 5 * H                    # 4·H for value/gates + H for out‑gate

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()           # [R, dim] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128      # empirically good on H100
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],        # γ (float32)
        weights["norm.bias"],          # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four hidden projections + their gates + the out‑gate
    # ------------------------------------------------------------------
    #   [left_proj, right_proj, left_gate, right_gate, out_gate]
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16).contiguous()                         # [OUT_DIM, dim]

    fused = F.linear(x_norm_fp16, fused_weight)              # FP16 → [R, OUT_DIM]

    # split the fused tensor
    left_raw       = fused[:, 0:H]                # [R, H]
    right_raw      = fused[:, H:2 * H]            # [R, H]
    left_gate_raw  = fused[:, 2 * H:3 * H]        # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]        # [R, H]
    out_gate_raw   = fused[:, 4 * H:5 * H]        # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # already FP16

    left  = left_raw  * left_gate   # [R, H]  FP16
    right = right_raw * right_gate  # [R, H]  FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)                     # [B, N, N, H]
    right = right.view(B, N, N, H)                    # [B, N, N, H]

    # bring hidden dimension to batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()           # [R, H] FP16
    out_gate_flat = out_gate                                 # [R, H] FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
