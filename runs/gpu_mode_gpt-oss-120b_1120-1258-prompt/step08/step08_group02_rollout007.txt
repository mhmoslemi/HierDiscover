"""
TriMul (outgoing) – high‑performance forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the trailing channel dimension (dim) → FP16.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM: X_norm @ W_fused.T  →  [R, 4·H]   (R = B·N·N, H = hidden_dim).
   Split the result into the four tensors.
3️⃣ Compute the output‑gate with a second linear (still FP16).
4️⃣ (Optional) Apply binary mask to the two projected tensors.
   Apply sigmoid to the three gate tensors (in‑place) and multiply:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
5️⃣ Reshape left/right → (B·H, N, N) and perform a batched GEMM:
        out_mat[b·H] = left_mat [b·H] @ right_mat[b·H].T
   Reshape back to (B, N, N, H).
6️⃣ Fuse the second LayerNorm and the element‑wise multiplication by the
   out‑gate into one Triton kernel (FP32 → FP16).
7️⃣ Final linear projection back to the original dimensionality (FP16 GEMM).

Only the two LayerNorm‑related kernels are written in Triton; the remaining
operations use highly‑optimised torch kernels (Tensor‑Core‑enabled FP16
GEMMs and batched GEMM).  This balance gives excellent portability while
still meeting the speed‑target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- mean & variance (FP32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # -------------------- normalise & apply γ,β --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                   # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise out‑gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------- mean & variance (ignoring gate) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- normalise, apply γ/β and multiply by gate --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # gate (already FP16, cast to FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g              # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # 2️⃣ Fuse left/right projections + their gates into a single GEMM
    # ------------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (FP16 matmul)
    #   we place the weight on the device and cast to FP16 once
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)  # FP16, shape [R, 4·H]

    left_raw   = fused[:, 0:H]                # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]        # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]        # [R, H]

    # separate linear for the output‑gate (still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape = [B,N,N] → flatten to [R, 1] and cast to FP16
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # inplace sigmoid (FP16 → FP16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()           # [R, H] (FP16)

    # gated projections
    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H,N,N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + element‑wise out‑gate (over hidden dim H)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()        # [R, H] FP16
    out_gate_flat = out_gate                       # already [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # reuse block sizes (they work well for both dim and hidden_dim)
    grid2 = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
