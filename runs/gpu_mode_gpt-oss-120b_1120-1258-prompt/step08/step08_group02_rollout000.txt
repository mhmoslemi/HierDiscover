Algorithm
---------

2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into one

7️⃣ Final linear projection (FP16 GEMM) → FP32 output.

Only the two LayerNorm kernels are written in Triton; the remaining

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,              # const   * [total_rows, dim] (float32)
    y_ptr,              # mutable * [total_rows, dim] (float16)
    weight_ptr,         # const   * [dim]            (float32)
    bias_ptr,           # const   * [dim]            (float32)
    total_rows,         # int32   (runtime)
    dim,                # int32   (runtime)
    eps,                # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):       # loop over channel blocks
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Write normalized values γ·(x−μ)·rstd + β  (store FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (row, col)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = norm * w[None, :] + b[None, :]           # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def layernorm_gate_kernel(

    gate_ptr,           # const   * [total_rows, dim] (float16)
    y_ptr,              # mutable * [total_rows, dim] (float16)
    weight_ptr,         # const   * [dim]            (float32)
    bias_ptr,           # const   * [dim]            (float32)
    total_rows,         # int32   (runtime)
    dim,                # int32   (runtime)
    eps,                # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of all Linear/LayerNorm parameters

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape               # x: [B, N, N, dim]
    total_rows = B * N * N              # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_D = 64       # channel block size
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],        # γ (float32)
        weights["norm.bias"],          # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,                   # H100 -> 8 warps gives good occupancy
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    # Stack the four weight matrices: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # FP16 for speed

    # Linear:   X_norm_fp16 @ w_fused.T   →  [R, 4·H]   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)   # shape [R, 4*H] FP16

    left_raw       = fused[:, 0:H]                 # [R, H]
    right_raw      = fused[:, H:2 * H]             # [R, H]
    left_gate_raw  = fused[:, 2 * H:3 * H]         # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]         # [R, H]

        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]

    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ------------------------------------------------------------------

        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)

        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] FP16

    left  = left_raw * left_gate                  # [R, H] FP16
    right = right_raw * right_gate                # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update:

    # ------------------------------------------------------------------

    right = right.view(B, N, N, H)                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out_gate multiplication over hidden dim H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()            # [R, H] FP16

        out_norm_gate_fp16,            # y_ptr (FP16)
        weights["to_out_norm.weight"], # γ (float32)
        weights["to_out_norm.bias"],   # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ``out_flat`` was cast to FP32 inside the kernel (via load) – no extra copy needed.

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
