"""
custom_kernel – Optimized outgoing TriMul (AlphaFold3)

Algorithm (mixed‑precision, fused linear layers, Triton LayerNorm):
1️⃣ Row‑wise LayerNorm over the last channel (dim) → Triton kernel.
2️⃣ Fuse *all* linear projections and gates (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM:
        X_norm (R×dim) @ W_fusedᵀ  →  R×5·H   (R = B·N·N, H = hidden_dim)
   Split the result into raw projections, raw gates and output‑gate.
3️⃣ Optional binary mask is applied to the two projection tensors.
4️⃣ In‑place `sigmoid_()` on the three gate tensors.
5️⃣ Element‑wise gating:
        left  = left_raw  * left_gate
        right = right_raw * right_gate
6️⃣ Reshape to [B, H, N, N] and compute the N³ multiplicative update
   as a batched matrix‑multiply:
        out[b,h] = left[b,h] @ right[b,h]ᵀ   (FP16 Tensor‑Core GEMM)
7️⃣ Row‑wise LayerNorm over the hidden dimension H → Triton kernel.
8️⃣ Multiply by the output‑gate, apply the final linear projection
   (FP16 GEMM) and cast the result back to FP32.

All heavy linear algebra is performed with PyTorch’s native FP16 GEMM
(Tensor‑Core) which is already optimal on H100.  The only custom
operations are the two LayerNorms, which are implemented in a single
Triton kernel that works for both FP32 and FP16 inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input can be fp32 or fp16,
# output is always fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const * [total_rows, D]  (fp32 or fp16)
    y_ptr,            # mutable * [total_rows, D] (fp16)
    weight_ptr,       # const * [D] (fp32)   – γ
    bias_ptr,         # const * [D] (fp32)   – β
    total_rows,       # int   (runtime)
    D,                # int   (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input x  : fp32 or fp16 (any dtype supported by tl.load)
    Output y : fp16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance (accumulate in fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in tiles
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D
        # address: row * D + col
        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        # Load as original dtype, then cast to fp32 for reduction
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values: γ·(x−μ)·rstd + β   (store as fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]      # fp32 intermediate
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * D + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {
              "dim"         : int,
              "hidden_dim"  : int,
              "nomask"      : bool (optional, default True)
          }

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)          # True  -> no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for per‑pair tensors

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim] (contiguous)
    x_flat = x.view(R, dim).contiguous()
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256                    # rows per program (tuned)
    BLOCK_D = 64                     # columns per iteration (tuned)
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,                # H100: many warps give high occupancy
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections + gates into a single GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    # concatenate weight matrices: [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    # GEMM: X_norm (R×dim) @ W_fusedᵀ → (R×5·H)   (fp16)
    fused = F.linear(x_norm_fp16, w_fused)      # bias = None

    # --------------------------------------------------------------
    # 3️⃣ Split fused result
    # --------------------------------------------------------------
    H = hidden
    left_raw       = fused[:, 0:H]                # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]            # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcasted over the hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # --------------------------------------------------------------
    # 5️⃣ Sigmoid gates (in‑place) and apply them
    # --------------------------------------------------------------
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()          # still [R, H]

    left  = left_raw  * left_gate      # [R, H] fp16
    right = right_raw * right_gate     # [R, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ N³ multiplicative update (batched GEMM)
    #    Convert to [B, H, N, N] → matmul → back to [B, N, N, H]
    # --------------------------------------------------------------
    # reshape & permute so that the head dimension becomes leading batch dim
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → [B·H, N, N]   (FP16 Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension H → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 8️⃣ Multiply by output‑gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise fp16

    # Final linear: hidden → dim
    w_out = weights["to_out.weight"].to(dtype=torch.float16, device=device)  # [dim, H]
    final_fp16 = F.linear(out_norm.view(R, H), w_out)   # [R, dim] fp16

    # Cast back to FP32 as required by the model
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
