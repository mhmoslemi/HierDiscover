"""
Highly‑optimized forward pass for the **outgoing** TriMul operator (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16) using a custom
    Triton kernel.  The kernel accumulates the mean/variance in fp32,
    normalises each row and applies the learned γ/β parameters.

2️⃣ Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
    into a single GEMM:
        [R,dim] @ [4·H,dim]ᵀ  →  [R,4·H]   (fp16)
   where R = B·N·N and H = hidden_dim.  The four slices are split again
   after the GEMM.

3️⃣ Optional mask (broadcast over the hidden dimension) is applied to the
    projected “left” and “right” tensors before the gates.

4️⃣ Gate tensors are obtained with an in‑place sigmoid and multiplied into the
    projected tensors.

5️⃣ The N³ multiplicative update is performed as a batched matrix‑multiply:
        out[b·H+i]_{ij} = Σ_k left[b·H+i]_{ik} * right[b·H+i]_{jk}
   i.e. `torch.bmm(left_mat, right_mat.transpose(1,2))` where the batch
   dimension is B·H.  This call uses cuBLAS Tensor‑Cores and is the dominant
   cost.

6️⃣ Second LayerNorm over the hidden dimension (again via Triton).

7️⃣ Apply the output‑gate, a final linear projection back to `dim` and cast
    the result to float32.

All heavy element‑wise work (layer‑norm, optional mask) is done in Triton,
while the dense matrix multiplications leverage the highly‑optimised cuBLAS
kernels.  Mixed‑precision (fp16) is used throughout except for the final
output type, matching the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]    (float32)
    y_ptr,          # mutable * [total_rows, dim]    (float16)
    gamma_ptr,      # const   * [dim]                (float32)
    beta_ptr,       # const   * [dim]                (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm.
    y = ((x-μ)/sqrt(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance (fp32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply scale (γ) and shift (β) and store as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * gamma[None, :] + beta[None, :]                  # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32 tensors):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : dict with at least:
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (Row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256          # rows per program – good for typical R
    BLOCK_D = 64           # columns per tile  – works for dim ≤ 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4×H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices → [4·H, dim] (fp16)
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # Linear: [R, dim] @ [4·H, dim].T → [R, 4·H]  (fp16, bias=False)
    fused = F.linear(ln1_out, w_proj)   # fp16

    # Split the concatenated results
    left_raw      = fused[:, 0:H]                   # [R, H]
    right_raw     = fused[:, H:2 * H]               # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]           # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]          # [R, H]

    # Separate output‑gate linear map
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates, and gate the projections
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid for reduced memory pressure
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] → [B, H, N, N] → (B·H, N, N)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # Batched matrix multiplication (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # (B·H, N, N)

    # Restore layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim H (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()          # [R, H]
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)           # fp16
    out_norm = out_norm * out_gate                # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
