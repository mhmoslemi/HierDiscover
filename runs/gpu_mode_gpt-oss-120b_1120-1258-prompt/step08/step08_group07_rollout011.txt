"""
TriMul “outgoing” forward pass – Triton‑accelerated first LayerNorm,
mask handling and fused *gate* linear projection.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – custom Triton
    kernel (fp32 reduction → fp16 output).

2️⃣  Concatenate the four linear weights (left‑proj, right‑proj,
    left‑gate, right‑gate) and perform a single half‑precision GEMM:
    [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   where R = B·N·N.

3️⃣  Split the GEMM result, apply the (optional) mask, sigmoid
    the gates and multiply the projections by their gates → `left`,
    `right`.  The output‑gate is computed by a second half‑precision
    GEMM.

4️⃣  Reshape `left` / `right` to [B, N, N, H] and compute the
    multiplicative update with a single ``torch.einsum``:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This avoids the costly ``view‑permute‑contiguous`` copy that a
    batched‑gemm implementation would need.

5️⃣  Apply the second LayerNorm (hidden dimension) with the
    built‑in cuDNN implementation (fast, no custom kernel required).

6️⃣  Element‑wise multiply by the output‑gate and finish with a
    half‑precision linear projection back to `dim`.

The only Triton kernel is the row‑wise LayerNorm; all heavy matrix
multiplications use cuBLAS/cuDNN which is optimal on H100.
The implementation works for all shapes listed in the statement and
produces a final float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,      # const   * [total_rows, dim] (float32)
    y_ptr,      # mutable * [total_rows, dim] (float16)
    w_ptr,      # const   * [dim]              (float32)   – gamma
    b_ptr,      # const   * [dim]              (float32)   – beta
    total_rows,  # int
    dim,         # int
    eps,         # float (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows handled per program
    BLOCK_D: tl.constexpr,   # columns handled per iteration
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ²+eps)) * γ + β
       Reduction uses fp32, output is fp16.
    """
    pid = tl.program_id(0)                     # one program → BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # Σ x
        sum_sq  += tl.sum(vals * vals, axis=1)              # Σ x²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ---------- 2️⃣ write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd   →   fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel – “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]    (bool/float, optional)
        - weights_dict : dict of torch.Tensors holding all learnable parameters
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                       # total rows for row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()        # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # block sizes tuned for H100 (128‑bit loads, ~128‑byte shared memory)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse four “proj/gate” linear layers into a single GEMM
    # --------------------------------------------------------------
    H = hidden_dim
    # [4H, dim] → concatenate all four weight matrices (already float32)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                   # cast once for Tensor‑Core

    # Linear: (R, dim) @ (4H, dim)^T → (R, 4H)   (fp16, no bias)
    proj_gate = F.linear(x_norm_fp16, w_proj_gate)   # [R, 4H]

    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]               # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]             # [R, H]

    # out‑gate (separate linear, shape [R, H])
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)  # [H, dim]
    )   # → [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [R, 1]  (fp16) and broadcast
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate      # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] for the multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)           # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)     # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ N³ “tri‑mul” – contracted Einstein sum (no extra copies)
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (cuDNN)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # still fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                            # element‑wise gating, fp16

    # Final linear: (R, H) → (R, dim)
    final_half = F.linear(
        out.view(R, hidden_dim),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
