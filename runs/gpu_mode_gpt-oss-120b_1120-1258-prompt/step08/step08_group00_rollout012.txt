"""
TriMul (outgoing) – fast Triton‑fused gating + CuBLAS batched GEMM.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – use torch.nn.functional
   which is heavily optimized (no custom kernel needed for LN).

2️⃣ Fused linear projection: (R, dim) @ (5·H, dim)ᵀ → (R, 5·H) (half‑precision).

3️⃣ Triton kernel `split_fuse_gate_kernel`  
   • Takes the fused projection, optional pair‑wise mask and computes:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)   (already transposed)
        out_gate = sigmoid(proj_out_gate)
   • Writes `left` and `right` directly in the layout required by the
     batched GEMM:  left  → [B·H, N, N]   (row = i, column = k)
                   right → [B·H, N, N]   (row = k, column = j) i.e. rightᵀ
   • Stores `out_gate` in the classic row‑wise layout (R, H) for the
     later element‑wise multiplication.

4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out_mat = torch.bmm(left, right)      # shape (B·H, N, N)

5️⃣ Second row‑wise LayerNorm (torch.nn.functional) on the hidden‑dim tensor,
   then element‑wise multiply by `out_gate`.

6️⃣ Final half‑precision linear projection back to `dim` and cast to fp32.

Only the gating+masking kernel is written in Triton – it fuses three
linear maps, the mask, and the three sigmoid gates while writing the
matrices in the exact layout required by the cuBLAS GEMM, thus removing
the costly permute+contiguous copies of the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask, sigmoid gates,
#               and write left/right in GEMM‑ready layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[B*H, N, N] (fp16)   – already transposed
    right_ptr,         # mutable *[B*H, N, N] (fp16)   – already transposed
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,    # R = B·N·N
    H: tl.int32,       # hidden dimension
    N: tl.int32,       # sequence length
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened pair (b, i, k) (row index `r`):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)   (stored transposed)
        out_gate = sigmoid(proj_out_gate)

    The left/right matrices are written directly as
        left [b*H + h, i, k]
        right[b*H + h, k, i]   # transposed, so that torch.bmm(left,right) yields the desired einsum.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Decode the (b,i,k) triple for each row -------------------------------------------------
    NN = N * N                     # N²
    b_idx = offs_r // NN           # batch index (0 .. B-1)
    pos   = offs_r % NN            # remainder = i·N + k
    i_idx = pos // N               # i coordinate (0 .. N-1)
    k_idx = pos % N                # k coordinate (0 .. N-1)

    # ------------------------------------------------------------------
    # Load mask (if present) – broadcast over hidden dim -----------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H --------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ------------------------------------------------------------------
        # Load the five raw projection slices (left, right, left_gate, right_gate, out_gate)
        # Offsets for the five segments inside the fused projection
        # layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        left_raw_ptr      = proj_ptr + base
        right_raw_ptr     = proj_ptr + base + H
        left_gate_ptr     = proj_ptr + base + 2 * H
        right_gate_ptr    = proj_ptr + base + 3 * H
        out_gate_raw_ptr  = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw     = tl.load(right_raw_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw = tl.load(left_gate_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw= tl.load(right_gate_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(out_gate_raw_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Sigmoid (computed in fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # --------------------------------------------------------------
        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # --------------------------------------------------------------
        # Destination batch index:  b*H + h
        batch_idx = b_idx[:, None] * H + offs_h[None, :]                # [R,H]

        # Linear offsets inside the (B*H, N, N) tensors
        # left  → [batch, i, k]   offset = batch*NN + i*N + k
        # right → [batch, k, i]   offset = batch*NN + k*N + i   (transposed)
        left_offset  = batch_idx * NN + i_idx[:, None] * N + k_idx[:, None]
        right_offset = batch_idx * NN + k_idx[:, None] * N + i_idx[:, None]

        # Store left / right (mask applied already)
        tl.store(left_ptr  + left_offset,  left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + right_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # Store out_gate in the original (R, H) layout
        out_gate_ptr_base = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_ptr_base, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use torch's highly‑optimised kernel)
    # ------------------------------------------------------------------
    # torch.layer_norm works on the last dimension; we keep weight/bias.
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)            # [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices once (store in fp16)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # [5*H, dim]  fp16

    # (B,N,N,dim) @ (5*H,dim)ᵀ → (B,N,N,5*H)  → flat to (R,5*H)
    proj = F.linear(x_norm, w_concat)          # fp16
    proj = proj.view(R, 5 * H)                 # (R,5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid‑gate (single Triton kernel)
    # ------------------------------------------------------------------
    # Allocate output buffers
    left_mat  = torch.empty(B * H, N, N, dtype=torch.float16, device=device)  # [B*H, N, N]
    right_mat = torch.empty(B * H, N, N, dtype=torch.float16, device=device)  # transposed version
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)         # [R, H]

    if nomask:
        # dummy mask (won’t be accessed)
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        # flatten mask to [R] and convert to half
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    # Kernel launch configuration
    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left_mat,
        right_mat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # proj and mask can be released early
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    # left_mat : [B*H, N, N]   (i,k)
    # right_mat: [B*H, N, N]   (k,j) – already transposed, so no extra .t()
    out_mat = torch.bmm(left_mat, right_mat)   # → [B*H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) for the second LN
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # Flatten for row‑wise LN (rows = B·N·N, dim = H)
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # still fp16

    # Apply the element‑wise output gate (broadcasted)
    out_norm = out_norm * out_gate
    del out_gate, out_mat, left_mat, right_mat, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    # Reshape to original 4‑D shape and convert to fp32
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
