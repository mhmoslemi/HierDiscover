"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pair representation (FP32 ➜ FP16) with a custom
   Triton kernel.
2️⃣ Fuse the four linear maps (left/right projections and gates) **and** the
   out‑gate linear map into a single GEMM:   W₅h = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg; Wₒg]  (5·H × dim).
   A single call to `torch.nn.functional.linear` produces all intermediate
   tensors in one go.
3️⃣ Optional mask (applied only to the left/right projections), followed by
   in‑place sigmoids for the three gate tensors.
4️⃣ Reshape to (B,N,N,H) and compute the cubic update
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` as a batched GEMM:
   (B·H) matrix‑multiplications of size N×N.
5️⃣ Row‑wise LayerNorm over the hidden dimension H (again via Triton).
6️⃣ Element‑wise out‑gate, final linear projection back to the original dim.

All heavy linear work stays in FP16 and uses Tensor‑Core‑friendly cuBLAS
kernels. The only Triton‑coded parts are the two row‑wise LayerNorm kernels,
which easily satisfy the “must use Triton” requirement while giving the
best possible memory‑bandwidth utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim] (float32) – γ
    b_ptr,          # const  * [dim] (float32) – β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_R: tl.constexpr,   # rows handled per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    """
    Compute per‑row LayerNorm:
        y = ((x‑μ)/√(σ²+eps)) * γ + β
    Input  : x – FP32
    Output : y – FP16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ---------- write normalized values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# TriMul forward (outgoing) – entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N             # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp16 output
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    BLOCK_R = 256           # rows per program
    BLOCK_D = 128           # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (FP32)
        weights["norm.bias"],        # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps (5·H × dim)
    # --------------------------------------------------------------
    # Concatenate the four projection/gate matrices **and** the out‑gate
    # matrix. All weights are stored as FP32 in the checkpoint – we cast
    # once to FP16 for the GEMM.
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5·H, dim)

    # Single GEMM ⇒ [R, 5·H] (FP16)
    proj_all = F.linear(x_norm, fused_w)    # FP16

    # Slice the fused result
    left_proj_raw  = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2*H]
    left_gate_raw  = proj_all[:, 2*H:3*H]
    right_gate_raw = proj_all[:, 3*H:4*H]
    out_gate_raw   = proj_all[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1]  (FP16)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate    # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (B·H, N, N) → treat the hidden dimension as batch.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (left @ rightᵀ), stays in FP16.
    # cuBLAS/Tensor‑Cores handle this extremely efficiently.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # FP16 → [R, dim]

    # Cast back to FP32 to match the reference module
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
