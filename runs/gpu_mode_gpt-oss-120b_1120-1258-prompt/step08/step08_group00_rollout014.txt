def fused_proj_gate_kernel(
    x_ptr,          # const fp16 *[R, dim]
    left_ptr,       # mutable fp16 *[R, H]
    right_ptr,      # mutable fp16 *[R, H]
    out_gate_ptr,   # mutable fp16 *[R, H]
    w_left_ptr,     # const fp16 *[H, dim]
    w_right_ptr,
    w_lgate_ptr,
    w_rg_ptr,
    w_og_ptr,

    dim: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    pid_c = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_c = col_start + tl.arange(0, BLOCK_H)

    mask_r = offs_r < R
    mask_c = offs_c < H

    # accumulators
    left_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    right_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    lgate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    rgate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    ogate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptrs,
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_D]

        # weight tiles: we load transposed tiles shaped (BLOCK_D, BLOCK_H)
        # left proj
        w_left_ptrs = w_left_ptr + offs_c[None, :] * dim + offs_d[:, None]

                         mask=mask_c[None, :] & mask_d[:, None],
                         other=0.0).to(tl.float32)   # [BLOCK_D, BLOCK_H]
        left_acc += tl.dot(x, w_left)

        # right proj
        w_right_ptrs = w_right_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_right = tl.load(w_right_ptrs,
                          mask=mask_c[None, :] & mask_d[:, None],
                          other=0.0).to(tl.float32)
        right_acc += tl.dot(x, w_right)

        # left gate
        w_lgate_ptrs = w_lgate_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_lgate = tl.load(w_lgate_ptrs,
                          mask=mask_c[None, :] & mask_d[:, None],
                          other=0.0).to(tl.float32)
        lgate_acc += tl.dot(x, w_lgate)

        # right gate
        w_rg_ptrs = w_rg_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_rg = tl.load(w_rg_ptrs,
                       mask=mask_c[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float32)

        # out gate
        w_og_ptrs = w_og_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_og = tl.load(w_og_ptrs,
                       mask=mask_c[None, :] & mask_d[:, None],
                       other=0.0).to(tl.float32)
        ogate_acc += tl.dot(x, w_og)

    right_gate = (1.0 / (1.0 + tl.exp(-rgate_acc)))

    # convert gates to fp16
    left_gate_fp16 = left_gate.to(tl.float16)
    right_gate_fp16 = right_gate.to(tl.float16)
    out_gate_fp16 = out_gate.to(tl.float16)

    # mask handling
    if HAS_MASK:

                           mask=mask_r,
                           other=1.0)[:, None]        # [BLOCK_R,1] fp16
        mask_fp32 = mask_val.to(tl.float32)

    # apply mask and gates, then cast to fp16
    left_fp = (left_acc * mask_fp32 * left_gate).to(tl.float16)
    right_fp = (right_acc * mask_fp32 * right_gate).to(tl.float16)

    # store results
    left_ptrs = left_ptr + offs_r[:, None] * H + offs_c[None, :]
    right_ptrs = right_ptr + offs_r[:, None] * H + offs_c[None, :]

    tl.store(right_ptrs, right_fp,
             mask=mask_r[:, None] & mask_c[None, :])
    tl.store(out_gate_ptrs, out_gate_fp16,
             mask=mask_r[:, None] & mask_c[None, :])

"""
High‑performance outgoing Tri‑multiplicative update (TriMul).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 → float16) using a Triton kernel.
2️⃣ Fuse the five linear maps (left/right projections & the three gates) into a
   single Triton GEMM that directly writes the three required tensors
   (left, right, out_gate).  The kernel also applies the optional pair‑wise mask
   and the sigmoid non‑linearity for the gates, staying completely in fp16
   (sigmoid computed in fp32 for stability).
3️⃣ Batched GEMM (cuBLAS) over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
4️⃣ Second row‑wise LayerNorm on the hidden tensor, followed by element‑wise
   multiplication with the output‑gate.
5️⃣ Final linear projection back to the original dimension (cuBLAS).

This design removes the intermediate “proj” tensor (size R × 5·H) and the
extra memory round‑trip, cutting bandwidth and launch overhead while keeping at
least one custom Triton kernel (the fused projection/gate kernel).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                           # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                           # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused projection + gating (+ optional mask)
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,          # const   *[R, dim]          (fp16)
    left_ptr,       # mutable *[R, H]           (fp16)
    right_ptr,      # mutable *[R, H]           (fp16)
    out_gate_ptr,   # mutable *[R, H]           (fp16)
    w_left_ptr,     # const   *[H, dim]         (fp16)
    w_right_ptr,    # const   *[H, dim]         (fp16)
    w_lgate_ptr,    # const   *[H, dim]         (fp16)
    w_rg_ptr,       # const   *[H, dim]         (fp16)
    w_og_ptr,       # const   *[H, dim]         (fp16)
    mask_ptr,       # const   *[R]              (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute the five linear maps, apply the mask and sigmoid gates.
    The result tensors are written directly as fp16."""
    pid_r = tl.program_id(0)
    pid_c = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_c = col_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]

    mask_r = offs_r < R
    mask_c = offs_c < H

    # accumulators in fp32
    left_acc  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    right_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    lgate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    rgate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
    ogate_acc = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # ------------------- load x tile ----------------------------
        x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]   # [R,D]
        x = tl.load(x_ptrs,
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0)                                      # fp16

        # ------------------- load weight tiles (transposed) ---------
        # Each weight has shape [H, dim]; we load a tile of shape (BLOCK_D, BLOCK_H)
        w_left_ptrs = w_left_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_left = tl.load(w_left_ptrs,
                         mask=mask_c[None, :] & mask_d[:, None],
                         other=0.0)                                 # fp16
        left_acc += tl.dot(x, w_left)                           # (R,H)

        w_right_ptrs = w_right_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_right = tl.load(w_right_ptrs,
                          mask=mask_c[None, :] & mask_d[:, None],
                          other=0.0)
        right_acc += tl.dot(x, w_right)

        w_lgate_ptrs = w_lgate_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_lgate = tl.load(w_lgate_ptrs,
                          mask=mask_c[None, :] & mask_d[:, None],
                          other=0.0)
        lgate_acc += tl.dot(x, w_lgate)

        w_rg_ptrs = w_rg_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_rg = tl.load(w_rg_ptrs,
                       mask=mask_c[None, :] & mask_d[:, None],
                       other=0.0)
        rgate_acc += tl.dot(x, w_rg)

        w_og_ptrs = w_og_ptr + offs_c[None, :] * dim + offs_d[:, None]
        w_og = tl.load(w_og_ptrs,
                       mask=mask_c[None, :] & mask_d[:, None],
                       other=0.0)
        ogate_acc += tl.dot(x, w_og)

    # ------------------- sigmoid (fp32) ----------------------------
    left_gate  = 1.0 / (1.0 + tl.exp(-lgate_acc))
    right_gate = 1.0 / (1.0 + tl.exp(-rgate_acc))
    out_gate   = 1.0 / (1.0 + tl.exp(-ogate_acc))

    # ------------------- mask handling ----------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)[:, None]               # fp16, shape [BLOCK_R,1]
        mask_fp32 = mask_val.to(tl.float32)
    else:
        mask_fp32 = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # ------------------- apply mask & gates (fp16 output) ----------
    left_fp  = (left_acc * mask_fp32 * left_gate).to(tl.float16)
    right_fp = (right_acc * mask_fp32 * right_gate).to(tl.float16)
    out_gate_fp = out_gate.to(tl.float16)

    # ------------------- store --------------------------------------
    left_ptrs    = left_ptr    + offs_r[:, None] * H + offs_c[None, :]
    right_ptrs   = right_ptr   + offs_r[:, None] * H + offs_c[None, :]
    out_gate_ptrs= out_gate_ptr+ offs_r[:, None] * H + offs_c[None, :]

    tl.store(left_ptrs, left_fp,
             mask=mask_r[:, None] & mask_c[None, :])
    tl.store(right_ptrs, right_fp,
             mask=mask_r[:, None] & mask_c[None, :])
    tl.store(out_gate_ptrs, out_gate_fp,
             mask=mask_r[:, None] & mask_c[None, :])

# ----------------------------------------------------------------------
# Entry point: outgoing TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (bool/float) or None
        - weights_dict : model parameters (float32, bias=False for all Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------- unpack -------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # rows for row‑wise ops
    H = hidden_dim

    # ------------------- 1️⃣ first LayerNorm (fp16) -------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x, x_flat

    # ------------------- 2️⃣ fused projection + gating -------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # convert weight matrices to fp16 once
    w_left  = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_lgate = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_rgate = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_ogate = weights["out_gate.weight"].to(torch.float16).contiguous()

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # [R]

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R), triton.cdiv(H, BLOCK_H))
    fused_proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        w_left,
        w_right,
        w_lgate,
        w_rgate,
        w_ogate,
        mask_ptr,
        R,
        dim,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_norm, w_left, w_right, w_lgate, w_rgate, w_ogate

    # ------------------- 3️⃣ triangular multiplicative update ----------
    # reshape to [B, H, N, N] for batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------- 4️⃣ second LayerNorm + output‑gate -------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out, out_flat

    out_norm = out_norm * out_gate          # apply output gate
    del out_gate

    # ------------------- 5️⃣ final projection back to `dim` ---------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
