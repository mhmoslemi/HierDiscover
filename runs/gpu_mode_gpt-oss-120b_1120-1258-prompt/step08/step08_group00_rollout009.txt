import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    rows: tl.int32, H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0) # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # base offset for the row start in proj (5*H per row)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R,H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Store
        left_ptr   = left_ptr + offs_r[:, None] * H + offs_h[None, :]
        right_ptr  = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_gate_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_ptr, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr, out_gate, mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ---------- first layernorm ----------
    x_flat = x.reshape(R, dim)  # fp32
    x_norm_fp32 = F.layer_norm(x_flat, (dim,),
                               weight=weights["norm.weight"],
                               bias=weights["norm.bias"],
                               eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # ---------- fused projection ----------
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm, w_concat)  # (R,5H) fp16

    # ---------- split & gating ----------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask may be bool or float
        mask_half = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj

    # ---------- batched GEMM ----------
    # reshape without copy
    left_ = left.view(B, N, N, H).permute(0, 3, 1, 2)   # (B,H,N,N)
    right_ = right.view(B, N, N, H).permute(0, 3, 1, 2)
    # flatten batch*H
    left_mat = left_.reshape(B * H, N, N)
    right_mat = right_.reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)

    # reshape back for layernorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ---------- second layernorm ----------
    out_flat = out.reshape(R, H)   # fp16 (may be non-contiguous)
    out_norm = F.layer_norm(out_flat, (H,),
                            weight=weights["to_out_norm.weight"].to(torch.float16),
                            bias=weights["to_out_norm.bias"].to(torch.float16),
                            eps=eps)

    # ---------- output gate ----------
    out_norm = out_norm * out_gate

    # ---------- final linear ----------
    final_half = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) using PyTorch's highly‑optimised
   `torch.nn.functional.layer_norm`.  The result is cast to FP16.
2️⃣ Fuse the five linear maps (left/right projections + the three gates) into a single
   GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)  where  R = B·N·N  and  H = hidden_dim.
3️⃣ One custom Triton kernel (`split_fuse_gate_kernel`) splits the fused matrix, applies the
   optional pair‑wise mask, evaluates the three sigmoids and writes the three tensors
   `left`, `right` and `out_gate` (all FP16).
4️⃣ Batched matrix multiply over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   (implemented with cuBLAS via `torch.bmm`).
5️⃣ Second row‑wise LayerNorm (FP16) on the hidden‑dim tensor, element‑wise multiplication
   with the output‑gate, and a final FP16 linear projection back to `dim`.
   The final tensor is cast to FP32.

Only the split‑and‑gate step is written in Triton; all large GEMMs use cuBLAS
and therefore run at near‑peak tensor‑core throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – only used if HAS_MASK=1
    rows: tl.int32,    # total rows = B·N·N
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (corresponding to a pair (b,i,k) in the original tensor) compute:
        left   = (proj_left   * mask) * sigmoid(proj_left_gate)
        right  = (proj_right  * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic is performed in FP16, the sigmoid is evaluated in FP32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16 scalar per row
        mask_val = m[:, None]                                   # shape [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over the hidden dimension in chunks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the current row‑chunk inside the fused matrix
        # layout per row: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        # ------------------------------------------------------------------
        # load raw values and gates
        # ------------------------------------------------------------------
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ------------------------------------------------------------------
        # sigmoid (computed in fp32)
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ------------------------------------------------------------------
        # apply mask + gates
        # ------------------------------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ------------------------------------------------------------------
        # write results
        # ------------------------------------------------------------------
        dst_mask = mask_r[:, None] & mask_h[None, :]

        tl.store(left_ptr   + offs_r[:, None] * H + offs_h[None, :], left,
                 mask=dst_mask)
        tl.store(right_ptr  + offs_r[:, None] * H + offs_h[None, :], right,
                 mask=dst_mask)
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :], out_gate,
                 mask=dst_mask)

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                      # (R, dim) FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                    # (R, dim) FP16

    # --------------------------------------------------------------
    # 2️⃣ fused projection (left/right + gates)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                       # (5·H, dim)

    proj = F.linear(x_norm, w_fused)                          # (R, 5·H) FP16
    del w_fused, x_norm                                        # free intermediate

    # --------------------------------------------------------------
    # 3️⃣ split, mask, sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling -------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)        # (R,)
        mask_ptr = mask_fp16

    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr                                        # free

    # --------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension:  out = left @ rightᵀ
    #    shape after GEMM: (B, N, N, H)  (still FP16)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H) and bring H to the batch axis
    left_view  = left.view(B, N, N, H).permute(0, 3, 1, 2)   # (B, H, N, N) – view only
    right_view = right.view(B, N, N, H).permute(0, 3, 1, 2)  # (B, H, N, N)

    left_mat  = left_view.reshape(B * H, N, N)   # (B·H, N, N)
    right_mat = right_view.reshape(B * H, N, N)  # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # back‑to (B, N, N, H) for the second LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ second LayerNorm (FP16) + output‑gate multiplication
    # --------------------------------------------------------------
    out_flat = out.view(R, H)                        # (R, H) – may be non‑contiguous
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # apply the element‑wise output gate
    out_norm = out_norm * out_gate
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ final linear projection back to `dim` (FP16) → cast to FP32
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
