"""
custom_kernel – Outgoing TriMul (AlphaFold‑3)

Implements the forward pass of the “outgoing” TriMul operator.
The implementation fuses most linear layers into two GEMMs,
applies three learned sigmoids, performs the N³ multiplicative update
as a batched FP16 matrix‑multiply, and uses a single Triton
row‑wise LayerNorm kernel for both normalisation steps.
All heavy arithmetic is in FP16 (Tensor‑Core) while the final
output is cast back to FP32 as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]        (float32)   γ
    bias_ptr,       # const   * [dim]        (float32)   β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Input  x : float32
    Output y : float16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- compute per‑row mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var+eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Outgoing TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (x, mask, weights, config)
        x        : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask     : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights  : dict of torch.Tensor containing all Linear/LayerNorm params
        config   : dict with keys
                   - "dim" (int)
                   - "hidden_dim" (int)
                   - "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)   # True → no mask
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned)
    BLOCK_D = 32    # columns per iteration (tuned)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates in ONE GEMM
    # ------------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]   (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)      # FP16, shape [R, 4·H]

    # split into raw tensors
    left_raw, right_raw, left_gate_raw, right_gate_raw = torch.split(
        fused, H, dim=1
    )   # each [R, H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device)  # [H, dim]
    )   # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, combine
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (FP16)
        mask_fp16 = mask.to(dtype=torch.float16, device=device).view(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]  FP16
    right = right_raw * right_gate        # [R, H]  FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & heads → [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ   →  [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise FP16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] FP16

    # cast back to FP32 as the downstream model expects
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
