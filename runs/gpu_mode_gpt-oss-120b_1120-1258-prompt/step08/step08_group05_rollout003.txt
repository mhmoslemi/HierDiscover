"""
TriMul “outgoing” forward pass – highly‑optimized Triton + PyTorch version
-----------------------------------------------------------------------

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the (B,N,N,dim) tensor → fp16.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single half‑precision GEMM and split the result.
3️⃣ (Optional) apply the pair‑wise mask, compute σ‑gates and multiply.
4️⃣ Perform the N³ multiplicative update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision mat‑mul: reshape to (B·H, N, N) and call
   ``torch.matmul`` (Tensor‑Core‐accelerated on H100).
5️⃣ Second LayerNorm over the hidden dimension → fp16.
6️⃣ Apply the output gate, project back to the original ``dim`` (half‑precision
   GEMM) and return a float32 tensor.

All heavy linear work runs in FP16; the only FP32 work is the two
LayerNorm reductions.  The implementation works for any batch size ≤ 2,
any sequence length in the test suite and both masked / un‑masked variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ (scale)
    bias_ptr,       # const   * [dim]      (float32) – β (bias)
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)                     # 1‑D grid over rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance (both fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in blocks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # linear offset assuming row‑major layout: row*dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                # fp32 load
        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized result in fp16 (γ,β are fp32)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ  fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]           # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm (fp32 → fp16) using Triton.
    Parameters
    ----------
    x : torch.Tensor
        Input of shape [rows, dim] (float32)
    weight, bias : torch.Tensor
        LayerNorm parameters γ, β of shape [dim] (float32)
    eps : float
        Numerical epsilon.
    block_m, block_d : int
        Compile‑time tile sizes.
    Returns
    -------
    torch.Tensor
        Normalized output of shape [rows, dim] (float16)
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if
                         config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
                         required keys:
                         "norm.weight", "norm.bias",
                         "left_proj.weight", "right_proj.weight",
                         "left_gate.weight", "right_gate.weight",
                         "out_gate.weight",
                         "to_out_norm.weight", "to_out_norm.bias",
                         "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # Flatten the first three dims – each (i,j) pair is a row.
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        eps=eps,
        block_m=256,
        block_d=32,
    )                               # fp16, [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (four GEMMs in one)
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices: each [H, dim]
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)                # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T → (R, 4·H)  (half‑precision)
    proj_gate = F.linear(x_norm, proj_gate_w)                # fp16

    # split the fused result
    left_proj_raw  = proj_gate[:, 0:H]                     # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]                 # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]             # [R, H]

    # out‑gate is a separate linear (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device),  # [H, dim]
    )                                                       # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) + sigmoid gating + element‑wise product
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] broadcast over hidden dim
        mask_h = mask_tensor.to(dtype=torch.float16, device=device).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # σ‑gates
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate      # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)    # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]

    # bring hidden dim to second axis for cuBLAS friendly layout
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # batched GEMM over (B·H) matrices: (N,N) @ (N,N)^T → (N,N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )   # fp16, shape [B, H, N, N]

    # restore original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
        block_m=256,
        block_d=32,
    )   # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection back to `dim`, cast to float32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)  # fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device),  # [dim, H]
    )   # fp16, [R, dim]

    final = final_half.view(B, N, N, dim).to(dtype=torch.float32)
    return final
```
