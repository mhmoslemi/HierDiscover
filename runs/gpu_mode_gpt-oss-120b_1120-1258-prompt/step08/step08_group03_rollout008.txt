"""
TriMul (outgoing) – high‑performance forward pass (no gradients).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (fp32 → fp16) via a Triton kernel.
2️⃣ Fuse the four linear projections (left/right & their gates) **and** the
   output‑gate into a **single** FP16 GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate; W_out_gate]ᵀ
   → shape [R, 5·H]  (R = B·N·N, H = hidden_dim).  Slice the result.
3️⃣ (Optional) Apply binary mask to the left/right tensors before gating.
4️⃣ Apply sigmoid to the three gate tensors (in‑place, FP16) and multiply.
5️⃣ Reshape left/right to [B·H, N, N] and perform a batched FP16 matrix
   multiplication:  out = left @ rightᵀ   (Tensor‑Core).
6️⃣ Second row‑wise LayerNorm over the hidden dimension (FP16) via the same
   Triton kernel.
7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection and
   cast back to float32.

Only the LayerNorms are hand‑written Triton kernels; all GEMMs and elementwise
ops use PyTorch (FP16) which maps to highly‑optimized cuBLAS/cutlass kernels on
the H100.  The implementation works for any B∈{1,2}, N∈{128,…,1024},
dim∈{128,384,768} and hidden_dim=128, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm: y = ((x-μ)/sqrt(σ²+eps))*γ + β
    Operates on a 2‑D view [total_rows, dim] (row‑major).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ---------- write normalized result (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32)
        - config_dict  : dict with keys
            * "dim"         – input channel dimension
            * "hidden_dim" – hidden dimension (H)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # -------------------------------------------------------------
    # 0️⃣ unpack & constants
    # -------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows = batch * N * N
    H = hidden_dim

    # -------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # -------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -------------------------------------------------------------
    # 2️⃣ Fuse left/right proj, gates & out‑gate into ONE GEMM
    # -------------------------------------------------------------
    # concatenate the five weight matrices (each [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    fused = F.linear(x_norm_fp16, w_fused)  # [R, 5*H] in fp16

    left_raw       = fused[:, 0 * H:1 * H]
    right_raw      = fused[:, 1 * H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]
    out_gate_raw   = fused[:, 4 * H:5 * H]

    # -------------------------------------------------------------
    # 3️⃣ (Optional) binary mask
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → [R, 1] broadcastable to [R, H]
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # -------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (in‑place) and multiply
    # -------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # -------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    # -------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H as a batch dimension → [B, H, N, N] (contiguous)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & heads → [B*H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul: (N×N) @ (N×N)ᵀ   →   [B*H, N, N]  (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # -------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast to float32
    # -------------------------------------------------------------
    # element‑wise gate (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)

    # final projection: [R, H] @ [dim, H]ᵀ → [R, dim]
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

**Explanation of key optimizations**

* **Linear‑fusion:** The four projection/gate matrices **plus** the output‑gate matrix are concatenated and multiplied with the normalized input in a single FP16 GEMM, cutting the number of heavy GEMMs from two to one.
* **Row‑wise LayerNorm in Triton:** A custom kernel computes mean/variance per row in FP32 and writes the normalized result directly as FP16, avoiding an extra PyTorch LayerNorm pass and extra memory traffic.
* **Batched tensor‑core matmul:** After reshaping, the cubic update becomes a batched FP16 matrix multiplication (`torch.bmm`) which maps to cuBLAS Tensor‑Core kernels on H100, delivering near‑optimal throughput.
* **In‑place sigmoids & mask:** Gate activations are applied in‑place to keep the memory footprint low; the optional binary mask is broadcast before the gating step to match the reference implementation.
* **Mixed‑precision end‑to‑end:** All heavy linear algebra runs in FP16 (Tensor‑Core), while the final output is cast back to float32 as required.

The implementation satisfies all required input variations (different `dim`, presence/absence of mask, various sequence lengths) and runs the LayerNorms via Triton while delegating large matrix multiplications to highly‑optimized cuBLAS kernels.
