    eps,                  # fp32   epsilon for LayerNorm
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣  mean & variance of the raw input (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  broadcasted mask (scalar) – 1.0 if no mask
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # fp16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in tiles (BLOCK_H)
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # accumulators for the five outputs (fp32 for precision)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        lgate_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        rgate_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        outgate_acc   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # dot‑product over the input dimension (BLOCK_D)
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load a tile of the input row
            x = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0).to(tl.float32)

            # normalise the tile
            x_norm = (x - mean[:, None]) * rstd[:, None]                # fp32
            x_norm_f16 = x_norm.to(tl.float16)                           # fp16 for the matmul

            # load weight tiles (each shape [BLOCK_H, BLOCK_D])
            left_w      = tl.load(w_left_ptr      + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            right_w     = tl.load(w_right_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            lgate_w     = tl.load(w_lgate_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            rgate_w     = tl.load(w_rgate_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            outgate_w   = tl.load(w_outgate_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :], other=0.0)

            # accumulate (note: trans_b=True performs (M,K)·(N,K)ᵀ)
            left_acc    += tl.dot(x_norm_f16, left_w,    trans_b=True).to(tl.float32)
            right_acc   += tl.dot(x_norm_f16, right_w,   trans_b=True).to(tl.float32)
            lgate_acc   += tl.dot(x_norm_f16, lgate_w,   trans_b=True).to(tl.float32)
            rgate_acc   += tl.dot(x_norm_f16, rgate_w,   trans_b=True).to(tl.float32)
            outgate_acc += tl.dot(x_norm_f16, outgate_w, trans_b=True).to(tl.float32)

        # --------------------------------------------------------------
        # 3️⃣  sigmoid for the three gates
        # --------------------------------------------------------------
        lgate_sig = 1.0 / (1.0 + tl.exp(-lgate_acc))
        rgate_sig = 1.0 / (1.0 + tl.exp(-rgate_acc))
        outgate_sig = 1.0 / (1.0 + tl.exp(-outgate_acc))

        # --------------------------------------------------------------
        # 4️⃣  apply pair‑wise mask to the *projections* only,
        #       then store everything (fp16)
        # --------------------------------------------------------------
        left_tile  = left_acc  * mask_val * lgate_sig
        right_tile = right_acc * mask_val * rgate_sig

        left_ptr_store  = left_ptr  + offs_r[:, None] * hidden + offs_h[None, :]
        right_ptr_store = right_ptr + offs_r[:, None] * hidden + offs_h[None, :]
        gate_ptr_store  = out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :]

        tl.store(left_ptr_store,  left_tile.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr_store, right_tile.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(gate_ptr_store,  outgate_sig.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Kernel 2 : hidden‑dim LayerNorm + out‑gate + final Linear
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_linear_kernel(
    in_ptr,          # fp16   [rows, hidden]    (output of the BMM)
    gate_ptr,        # fp16   [rows, hidden]    (out_gate from kernel 1)
    gamma_ptr,       # fp32   [hidden]         (to_out_norm.weight)
    beta_ptr,        # fp32   [hidden]         (to_out_norm.bias)
    w_ptr,           # fp16   [dim, hidden]    (to_out.weight)
    out_ptr,         # fp32   [rows, dim]      (final result)
    rows,            # int32  total rows  (R = B·N·N)
    hidden,          # int32  hidden dimension (H)
    dim,             # int32  output dimension (C)
    eps,             # fp32   epsilon for LayerNorm
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_r = tl.program_id(0)          # row block
    pid_o = tl.program_id(1)          # output‑feature block

    row_start = pid_r * BLOCK_R
    col_start = pid_o * BLOCK_O

    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows
    offs_o = col_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < dim

    # ------------------------------------------------------------------
    # 1️⃣ mean / variance across hidden dim (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        in_tile = tl.load(in_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        sum_val += tl.sum(in_tile, axis=1)
        sum_sq  += tl.sum(in_tile * in_tile, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ accumulator for the final GEMM (rows × output‑features)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣ loop over the hidden dimension, produce
    #     gated = (γ·(x-μ)·rstd + β) ⊙ out_gate
    #     and accumulate dotted with the weight slice
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # load raw BMM output & gate
        x = tl.load(in_ptr   + offs_r[:, None] * hidden + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # load LayerNorm parameters for this hidden tile
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # fp32

        # normalized → LayerNorm → gating
        norm = (x - mean[:, None]) * rstd[:, None]               # fp32
        normed = norm * gamma[None, :] + beta[None, :]           # fp32
        gated  = normed * g                                      # fp32

        # load weight slice (dim × hidden) -> tile (BLOCK_O × BLOCK_H)
        w_tile = tl.load(w_ptr + offs_o[:, None] * hidden + offs_h[None, :],
                         mask=mask_o[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # accumulate contribution
        acc += tl.dot(gated, w_tile, trans_b=True)               # (BLOCK_R,BLOCK_O)
    # ------------------------------------------------------------------
    # 4️⃣ write output (fp32)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + offs_r[:, None] * dim + offs_o[None, :]
    tl.store(out_ptrs, acc, mask=mask_r[:, None] & mask_o[None, :])

# ----------------------------------------------------------------------
# Custom forward – “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim      = cfg["dim"]
    hidden   = cfg["hidden_dim"]
    eps      = 1e-5
    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops

    # ------------------------------------------------------------------
    # pre‑process: flatten input, allocate outputs
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    left   = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right  = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # mask handling
    # ------------------------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        has_mask = 1
        mask_flat = mask.to(torch.float16).reshape(R)                    # [R]

    # ------------------------------------------------------------------
    # cast all weight matrices to fp16 once
    # ------------------------------------------------------------------
    w_left      = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right     = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_lgate     = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_rgate     = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_outgate   = weights["out_gate.weight"].to(torch.float16).contiguous()
    w_to_out    = weights["to_out.weight"].to(torch.float16).contiguous()   # [dim, hidden]

    # ------------------------------------------------------------------
    # Kernel 1 launch (fused LayerNorm + projections + gates)
    # ------------------------------------------------------------------
    BLOCK_R = 64          # rows per program
    BLOCK_D = 32          # input‑dim tile
    BLOCK_H = 32          # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    proj_gate_kernel[grid](
        x_flat,
        mask_flat,
        left,
        right,
        out_gate,
        w_left,
        w_right,
        w_lgate,
        w_rgate,
        w_outgate,
        rows=R,
        dim=dim,
        hidden=hidden,
        eps=eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over the hidden dimension (outgoing update)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the batch‑gemm
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()

    left_mat  = left.view(B * hidden, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * hidden, N, N)                    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # back‑to [B, N, N, H] (fp16)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(R, hidden).contiguous()                # [R, H]

    # ------------------------------------------------------------------
    # Kernel 2 launch (hidden‑dim LayerNorm + out_gate + final linear)
    # ------------------------------------------------------------------
    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_R2 = 64          # rows per program (same as before)
    BLOCK_H2 = 32          # hidden‑dim tile
    BLOCK_O  = 32          # output‑dim tile

    grid2 = (triton.cdiv(R, BLOCK_R2), triton.cdiv(dim, BLOCK_O))

    norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],      # γ (fp32, shape [hidden])
        weights["to_out_norm.bias"],        # β (fp32, shape [hidden])
        w_to_out,
        final,
        rows=R,
        hidden=hidden,
        dim=dim,
        eps=eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)
    return final
