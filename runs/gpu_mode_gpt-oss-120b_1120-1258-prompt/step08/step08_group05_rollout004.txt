"""
Optimized forward pass for the **outgoing** Triangle Multiplicative Update (TriMul)
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) – Triton kernel.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) **and**
   the output‑gate linear layer into a single half‑precision GEMM.
   The result is split into the five required tensors.
3️⃣ (Optional) apply the pair‑wise mask, compute sigmoid gates and multiply.
4️⃣ Compute the N³ update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision matrix‑multiply (`torch.matmul`) on tensors of
   shape (B, H, N, N).  This maps to a cuBLAS Tensor‑Core kernel on H100.
5️⃣ Second LayerNorm over the hidden dimension – Triton kernel.
6️⃣ Apply the output‑gate, project back to the original channel dimension and
   cast the result to FP32.

All heavy linear work is performed in FP16; the only FP32 work is the two
LayerNorm reductions.  The implementation works for any batch size ≤ 2,
any sequence length in the test suite and both masked / un‑masked variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (fp32)
    y_ptr,          # mutable * [rows, dim] (fp16)
    weight_ptr,     # const   * [dim]      (fp32) – γ
    bias_ptr,       # const   * [dim]      (fp32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]    # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D FP32 tensor ``x`` of shape [rows, dim].
    Returns an FP16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,          # H100 works well with 4‑8 warps
    )
    return y


# ----------------------------------------------------------------------
# Outgoing TriMul forward (no backward needed)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if
                         config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Flatten the 4‑D tensor to 2‑D for the first LayerNorm
    total_rows = B * N * N                     # R = B·N·N
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] FP32

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
    )                               # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four projections + output‑gate into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate weight matrices: [left, right, left_gate, right_gate, out_gate]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    # (R, dim) @ (dim, 5*H).T  -> (R, 5*H)   (FP16)
    proj = F.linear(x_norm, fused_w)        # [R, 5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Split and apply mask / sigmoid gates
    # ------------------------------------------------------------------
    left_proj_raw   = proj[:, 0:H]                     # [R, H]
    right_proj_raw  = proj[:, H:2 * H]                 # [R, H]
    left_gate_raw   = proj[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw  = proj[:, 3 * H:4 * H]             # [R, H]
    out_gate_raw    = proj[:, 4 * H:5 * H]             # [R, H]

    if not nomask:
        # mask_tensor shape: [B, N, N] → flatten → [R, 1]
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()      # [R, H]
    right_gate = right_gate_raw.sigmoid_()     # [R, H]
    out_gate   = out_gate_raw.sigmoid_()       # [R, H]

    left  = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate        # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update – batched half‑precision matmul
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)              # [B, N, N, H]

    # Rearrange to (B, H, N, N) – contiguous layout for cuBLAS
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched GEMM: (B, H, N, N) @ (B, H, N, N)^T  →  (B, H, N, N)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # FP16

    # Restore original layout: [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],    # γ (float32)
        weights["to_out_norm.bias"],      # β (float32)
        eps=eps,
    )                                     # [R, H] FP16
    out_norm = out_norm.view(B, N, N, H)            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # FP16

    final_half = F.linear(
        out_norm.view(total_rows, H),                 # [R, H]
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                 # [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)   # FP32 output
    return final
```
