"""
Outgoing Tri‑Multiplicative (TriMul) update – high‑performance Triton/PyTorch
implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (dim) of the input tensor
   (shape B·N·N × dim).  The result is cast to FP16.

2️⃣ A single fused linear projection maps the normalized tensor to
   five * hidden_dim columns:
       [left_proj , right_proj , left_gate , right_gate , out_gate]
   (all bias‑free).  The weight matrix is pre‑concatenated and the GEMM
   is performed in FP16.

3️⃣ One Triton kernel (`split_fuse_gate_kernel`) splits the fused
   projection, applies the optional pair‑wise mask, computes the three
   sigmoids (in FP32 for numerical stability) and writes out three
   tensors:
       left  = (proj_left  * mask) * sigmoid(proj_left_gate)
       right = (proj_right * mask) * sigmoid(proj_right_gate)
       out_gate = sigmoid(proj_out_gate)

4️⃣ The core “triangle‑multiplicative” operation is a batched GEMM
   over the hidden dimension:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is realised as a strided‑batched `torch.bmm` on tensors of
   shape (B·H, N, N).

5️⃣ Row‑wise LayerNorm over the hidden dimension, element‑wise multiplication
   with `out_gate`, and a final linear projection back to the original
   `dim`.  The second LayerNorm is also performed by a Triton kernel,
   while the final projection uses cuBLAS.

All heavy arithmetic is carried out in FP16; the two LayerNorms compute
statistics in FP32 to retain numerical stability.  The implementation
fuses mask handling and sigmoid gating, drastically reducing memory traffic
and kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – only accessed if HAS_MASK=1
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (0 ≤ r < rows):
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    The projection matrix layout is contiguous:
        [ left , right , left_gate , right_gate , out_gate ]
    each segment has length H.
    """
    pid = tl.program_id(0)
    r_start = pid * BLOCK_R
    offs_r = r_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load per‑row mask (if present) and broadcast to the H dimension
    # ------------------------------------------------------------------
    if HAS_MASK:
        # mask is a single FP16 scalar per row → broadcast to (BLOCK_R,1)
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- compute offsets for the five contiguous segments -----
        # stride across columns = 5*H
        row_offset = offs_r[:, None] * (5 * H)                     # [R,1]
        col_offset = offs_h[None, :]                              # [1,H]
        base = row_offset + col_offset                            # [R,H]

        # pointers to the five segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_ptr_raw= proj_ptr + base + 4 * H

        # ----- load raw values (FP16) -----
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_ptr_raw,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid in FP32 for stability -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store results -----
        tl.store(left_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (fp32)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32)   γ
    bias_ptr,       # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance (accumulate in FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output  γ·(x‑μ)·rstd + β   (cast to FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly tuned; we still need a Triton
    # kernel for the second LayerNorm (hidden dimension) to keep at least one
    # custom kernel in the solution.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim]  fp16

    x_flat = x_norm.view(R, dim).contiguous()    # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection to 5×hidden_dim columns
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)
    # (R,dim) @ (5H,dim).T → (R,5H)
    proj = F.linear(x_flat, w_fused)            # fp16
    del x_norm, x_flat, w_fused                  # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling: flatten to [R] and convert to fp16 (0.0 / 1.0)
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_fp16

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: batched GEMM over hidden dimension
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H into batch dimension
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left, right, left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden_dim) + output‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    # reuse the same Triton row‑wise LayerNorm kernel
    BLOCK_M = 256
    BLOCK_D = 64
    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out

    # element‑wise multiply by the output gate (both fp16)
    out_norm = out_norm * out_gate
    del out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
