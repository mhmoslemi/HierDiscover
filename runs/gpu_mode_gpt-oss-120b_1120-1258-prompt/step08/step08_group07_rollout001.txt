"""
TriMul “outgoing” forward pass – Triton‑accelerated LayerNorm + fused Linear

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape [B·N·N, dim]).
   Implemented in Triton (`rowwise_layernorm_kernel`).  Reduction is
   performed in FP32, result is FP16.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single half‑precision GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N)

   The result is split into the four tensors needed later.

3️⃣ Optional mask is applied (broadcasted to the hidden dimension).

4️⃣ Apply element‑wise sigmoid to the three gate tensors and perform
   the gating (left * left_gate, right * right_gate).

5️⃣ Reshape `left`/`right` to [B, H, N, N] and perform the N³ multiplicative
   update with a batched half‑precision GEMM (`torch.bmm`).

6️⃣ Second row‑wise LayerNorm over the hidden dimension (again Triton).

7️⃣ Apply the output gate, final linear projection back to `dim`, and cast
   the result to FP32.

Only the two LayerNorm passes are written in Triton; all dense matrix
multiplications use cuBLAS/Tensor‑Core via `torch.nn.functional.linear` and
`torch.bmm`, which is optimal on modern GPUs (e.g. H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,   # const   * [total_rows, dim]   (float32)
    y_ptr,   # mutable * [total_rows, dim]   (float16)
    w_ptr,   # const   * [dim]               (float32)  gamma
    b_ptr,   # const   * [dim]               (float32)  beta
    total_rows,  # int
    dim,         # int
    eps,         # float   (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Reduction uses FP32, output is FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ mean & variance per row ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_tiles = tl.cdiv(dim, BLOCK_D)
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the input (FP32)
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ---------- 2️⃣ write normalized output ----------
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        # (x - μ) * rstd   →  fp32
        norm = (x - mean[:, None]) * rstd[:, None]

        # Load γ, β (fp32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :]
        )


# ----------------------------------------------------------------------
# Custom forward kernel for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear / layer‑norm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows for row‑wise ops

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()            # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Launch Triton LN kernel
    BLOCK_M = 256            # rows per program
    BLOCK_D = 32             # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_ptr=x_flat,
        y_ptr=x_norm_fp16,
        w_ptr=weights["norm.weight"],   # γ, FP32
        b_ptr=weights["norm.bias"],     # β, FP32
        total_rows=R,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right proj + left/right gate (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices: [4*H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # FP16 for Tensor‑Core

    # Linear: [R, dim] @ [4*H, dim]ᵀ → [R, 4*H] (FP16)
    proj_gate = F.linear(x_norm_fp16, w_proj_gate)  # bias = 0

    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]              # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]          # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]          # [R, H]

    # Out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (broadcasted)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Sigmoid in‑place (FP16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gating
    left = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate       # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)             # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)       # [B, N, N, H]

    # Move hidden dim to batch axis → many independent GEMMs
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.reshape(B * H, N, N)          # [B·H, N, N]  (FP16)
    right_mat = right_perm.reshape(B * H, N, N)        # [B·H, N, N]  (FP16)

    # Batched mat‑mul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        x_ptr=out_flat,
        y_ptr=out_norm_fp16,
        w_ptr=weights["to_out_norm.weight"],   # γ, FP32
        b_ptr=weights["to_out_norm.bias"],     # β, FP32
        total_rows=R,
        dim=H,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B, N, N, H] FP16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
