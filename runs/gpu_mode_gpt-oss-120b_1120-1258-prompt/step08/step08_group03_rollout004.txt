"""
TriMul (outgoing) – high‑performance forward pass using Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (fp32 → fp16) via a Triton
    kernel (no extra sync – mean / variance computed on‑the‑fly).

2️⃣  Fuse the four linear layers (left/right projections + their gates) into a
    single FP16 GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate].T  →  [R,4·H]
    and split the result.

3️⃣  Compute the output‑gate with a second FP16 GEMM.

4️⃣  (Optional) apply the binary mask to the raw projections, then apply
    the three sigmoid gates (in‑place).

5️⃣  Reshape left/right tensors to [B·H, N, N] and perform a batched matrix
    multiplication (Tensor‑Core FP16):
        out[b·h] = left[b·h] @ right[b·h].T   →   [B·H, N, N]

6️⃣  Second LayerNorm over the hidden dimension (again Triton, fp16 output).

7️⃣  Multiply by the output‑gate, apply the final linear projection and cast
    back to fp32.

All heavy linear algebra runs in FP16 to exploit H100 Tensor‑Cores.
Only two custom Triton kernels (the two LayerNorms) are needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (fp32)
    y_ptr,          # mutable * [total_rows, dim] (fp16)
    w_ptr,          # const   * [dim]          (fp32)  – γ
    b_ptr,          # const   * [dim]          (fp32)  – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Stored as fp16 while computation stays in fp32."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------- mean & variance --------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # --------- write normalized result (fp16) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) stay fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]          # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32)
        - config_dict  : dict with keys
            * "dim"         – input channel dimension (C)
            * "hidden_dim" – hidden dimension (H)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack & static params
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, C] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per program (tuned for H100)
    BLOCK_D = 64    # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates into one GEMM
    # -----------------------------------------------------------------
    # Stack weights: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, C]
            weights["right_proj.weight"],  # [H, C]
            weights["left_gate.weight"],   # [H, C]
            weights["right_gate.weight"],  # [H, C]
        ],
        dim=0,
    ).to(torch.float16)                     # -> fp16 for Tensor‑Core

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # -----------------------------------------------------------------
    # 3️⃣ Compute output‑gate (separate GEMM)
    # -----------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, C]
    )   # [R, H] fp16

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and sigmoid gates (in‑place)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] broadcastable to [R, H]
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # sigmoid – inplace to keep memory footprint low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # gated projections
    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched matmul)
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring H into the batch dimension -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head -> [B·H, N, N]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matrix multiply (Tensor‑Core FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (fp16 output)
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast to fp32
    # -----------------------------------------------------------------
    # element‑wise output‑gate (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # final linear: [R, H] @ [dim, H].T -> [R, dim] (fp16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    # reshape to original tensor layout and cast back to fp32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
