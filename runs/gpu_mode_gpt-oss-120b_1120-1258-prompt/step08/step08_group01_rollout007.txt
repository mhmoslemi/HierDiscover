---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – custom Triton
   kernel `layernorm_kernel`.  This is the only mandatory Triton kernel.

2️⃣ Fuse the four linear projections (left/right + their gates) into a
   single half‑precision GEMM (`torch.nn.functional.linear`).  The
   output‑gate is computed by a second GEMM (still half‑precision).

3️⃣ Apply the optional pair‑wise mask, compute the three sigmoid gates
   and multiply the gates into the projected tensors.  All of this is
   done with in‑place PyTorch ops to minimise memory traffic.

4️⃣ Reshape the tensors so that the cubic multiplicative update becomes a
   batched matrix multiplication:
        out[b·h, i, j] = Σₖ left[b·h, i, k] · right[b·h, j, k]

   This is performed with `torch.bmm`, which on an H100 runs on Tensor‑Core
   FP16 units.

5️⃣ Second LayerNorm over the hidden dimension (FP16) – we use the
   highly‑optimized cuDNN implementation (`torch.nn.functional.layer_norm`).

6️⃣ Apply the output‑gate, a final half‑precision linear projection back to
   `dim`, and cast the result to `float32`.

All heavy linear algebra is performed with cuBLAS Tensor‑Cores; the only
explicit Triton work is the first LayerNorm, which is required by the
challenge statement.  The implementation works for the full range of
shapes used in the tests (B∈{1,2}, N∈{128…1024}, dim∈{128,384},
hidden_dim=128) and for both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel works on a 2‑D view of the input: [total_rows, dim].
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # linearized address = row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ  fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β  fp32
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with LayerNorm/Linear parameters (float32)
        - config_dict  : dict with keys
              "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs & configuration
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # hidden dimension (H)
    nomask = config.get("nomask", True)  # whether mask should be ignored
    eps = 1e-5

    B, N, _, _ = x.shape               # B, seq_len, seq_len, dim
    total_rows = B * N * N              # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (Row‑wise, fp32 → fp16) via Triton
    # ------------------------------------------------------------------
    # Flatten to [R, dim] for the kernel
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim, dtype=dtype_fp16, device=device)

    # Tuned block sizes (dim ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projections + gates (four GEMMs together)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices [H, dim] → [4·H, dim]
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)   # -> fp16

    # Linear: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(ln1_out, w_proj)        # fp16

    # Slice the four blocks
    left_raw      = fused[:, 0:H]
    right_raw     = fused[:, H:2 * H]
    left_gate_raw = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)  # [H, dim]
    )   # (R, H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating multiplication
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_flat = mask.to(device=device, dtype=dtype_fp16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid → saves an allocation
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate     # (R, H) fp16
    right = right_raw * right_gate   # (R, H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the cubic (N³) update and run batched GEMM
    # ------------------------------------------------------------------
    # [B, N, N, H] → permute to [B, H, N, N] → merge B·H → (B·H, N, N)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication with Tensor‑Cores (FP16)
    # out_mat[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, j, k]
    out_mat = torch.bmm(left, right.transpose(1, 2))     # (B·H, N, N), fp16

    # Restore [B, N, N, H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    # Use cuDNN implementation – fast and reliable
    out_flat = out.view(total_rows, H)  # (R, H)
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16),
        eps=eps,
    )   # fp16, shape (R, H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection & cast back to fp32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate  # element‑wise scaling, fp16

    final_fp16 = F.linear(
        out_norm,
        weights["to_out.weight"].to(device=device, dtype=dtype_fp16)   # [dim, H]
    )   # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(dtype_fp32)   # float32 output
    return final
