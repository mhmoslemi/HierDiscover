"""
custom_kernel – high‑performance forward of the **outgoing** TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (over the channel dimension “dim”) implemented in Triton.
   Input:  [B·N·N, dim]  (FP32) → Output: [B·N·N, dim] (FP16).

2️⃣ Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:
       X_norm @ W_fused.T  →  [R, 5·H]   (R = B·N·N, H = hidden_dim)
   Then slice the result into raw tensors and apply sigmoid on the gates.

3️⃣ (Optional) Apply the binary mask to the projected *left* and *right* tensors.
   Mask is broadcasted from [B,N,N] → [R,1].

4️⃣ Compute the multiplicative update (the “TriMul”):
   reshape left/right to (B, H, N, N), treat each (B·H) as an independent
   batch and perform a batched matrix multiplication:
       out[b·H] = left_mat[b·H] @ right_mat[b·H].T
   Result shape: [B, N, N, H]  (FP16).

5️⃣ Fuse the second LayerNorm (over hidden dim H) **and** element‑wise
   multiplication with the *out_gate* into one Triton kernel.
   This yields a tensor still in FP16.

6️⃣ Final linear projection back to the original dimensionality (dim) using
   a half‑precision GEMM, then cast the result to float32.

Only the two LayerNorm‑related kernels are written in Triton; the remaining
operations use highly‑optimized PyTorch kernels (FP16 GEMMs, batched
matrix‑multiply, etc.). The implementation works for any batch size B∈{1,2},
sequence length N up to 1024, channel dimensions dim∈{128,384} and hidden
dimension H=hidden_dim (commonly 128). The mask can be omitted or supplied
as a bool/float tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)   γ
    bias_ptr,       # const   * [dim]            (float32)   β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (FP16)
    # --------------------------------------------------------------
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)   γ
    bias_ptr,       # const   * [dim]            (float32)   β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean & variance (ignore gate)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalise, apply γ/β and multiply by gate (store FP16)
    # --------------------------------------------------------------
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw activation
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (same shape as x)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = (norm * w[None, :] + b[None, :]) * g_vals        # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,

    Returns:
        torch.Tensor shape [B, N, N, dim] (float32)
    """

    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for layer‑norms
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # Triton launch parameters – tuned for H100
    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # columns per iteration
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (proj + gates + out_gate) into one GEMM
    # ------------------------------------------------------------------
    # (4*H + H) = 5*H output features
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]

    # [R, dim] @ [5*H, dim].T  →  [R, 5*H]   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)           # no bias as per model

    # slice fused result
    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]            # [R, H]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # mask: [B,N,N] → [R,1] (float16)

        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()   # shape [R, H] (FP16)

    # apply gates
    left = left_raw * left_gate          # [R, H] FP16
    right = right_raw * right_gate       # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ TriMul update: batched N^2 matrix multiplication per hidden dim
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)                # [B,N,N,H]
    right = right.view(B, N, N, H)              # [B,N,N,H]

    # collapse hidden dimension into batch axis → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Re‑use the same launch config as first layernorm (R rows, H columns)
