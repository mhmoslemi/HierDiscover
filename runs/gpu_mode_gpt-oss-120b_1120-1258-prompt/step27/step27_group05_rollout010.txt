def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr, R, H, N, HAS_MASK, BLOCK_R, BLOCK_H):

    pid = tl.program_id(0)

    row_start = pid * BLOCK_R

    offs_r = row_start + tl.arange(0, BLOCK_R)

    mask_r = offs_r < R

    # compute flat index: b,i,k similarly.

    # compute coordinates:

    nn = N * N

    b = offs_r // nn

    rem = offs_r - b * nn

    i = rem // N

    k = rem - i * N

    # compute mask value for block if HAS_MASK:
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)  # shape [BLOCK_R]
        m = m[:, None]  # broadcast to hidden dimension
    else:
        m = tl.full([BLOCK_R,1], 1.0, tl.float16)

    hidden_stride = N * N

    # base offset for hidden dimension 0 for each row:
    base = b * (H * hidden_stride) + i * N + k  # shape [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # projection offset within fused proj:
        # proj has shape [R, 5*H] stored row-major: i.e., index = r*(5*H) + (proj_idx)
        # each proj_idx = h + offset*H for each of 5 groups.

        # compute base_proj = offs_r[:, None] * (5 * H) + offs_h[None, :]  # base for group0.

        # load left_raw:
        left_raw = tl.load(proj_ptr + (offs_r[:, None] * (5*H) + offs_h[None, :]), mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # similarly for other groups
        offset = H  # shift for each group
        right_raw = tl.load(proj_ptr + (offs_r[:, None] * (5*H) + offs_h[None, :] + offset), mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + (offs_r[:, None] * (5*H) + offs_h[None, :] + 2*offset), mask=mask_r[:, None] & mask_h[None, :], other=0.0)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  LayerNorm on the flattened pair tensor (FP32).  
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
    single half‑precision GEMM (R=N²·B tokens → 5·H hidden features).  
3️⃣  Triton kernel `split_fuse_gate` splits the fused projection, applies the optional
    pair mask and the three sigmoid gates, and writes the three tensors
    `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).  
4️⃣  Batched half‑precision GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B, H, N, N].
5️⃣  Row‑wise mean / variance of `out` (FP32) via two fast reductions (torch.mean/var).  
6️⃣  Second LayerNorm, multiplication with `out_gate`, and final projection
    (hidden → dim) are performed with regular torch kernels (FP32/F16 GEMM).

All heavy‑memory kernels run in half‑precision; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate(
    proj_ptr,          # [R, 5*H]  FP16 fused projection
    left_ptr,          # [B*H*N*N] FP16
    right_ptr,         # [B*H*N*N] FP16
    out_gate_ptr,      # [B*H*N*N] FP16
    mask_ptr,          # [R]    FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program (tuned)
    BLOCK_H: tl.constexpr,   # hidden‑dim tile (tuned)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---------- decode flat row index → (b, i, k) ----------
    nn = N * N                               # N²
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # broadcast
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N                     # stride between consecutive H‑slices
    # start address of the first hidden slice (h = 0) for each row
    base = b * (H * hidden_stride) + i * N + k    # [BLOCK_R]

    # strides for the fused projection (5 groups, each of size H)
    proj_group_stride = H

    # loop over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---------- load the five raw projections ----------
        # pointer to the beginning of the 5*H block for the current row
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + proj_group_stride,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * proj_group_stride,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * proj_group_stride,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * proj_group_stride,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # ---------- sigmoid (computed in FP32, stored FP16) ----------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---------- apply mask + gates ----------
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination pointer for the current hidden tile
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[ B, N, N, dim ] (float32)
        - mask_tensor  : torch.Tensor[ B, N, N ] (bool/float) or None
        - weights_dict : mapping with the keys used in the reference module
        - config_dict  : must contain ``dim`` (int), ``hidden_dim`` (int),
                         and ``nomask`` (bool, optional)

    Returns
    -------
    torch.Tensor[ B, N, N, dim ] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = bool(cfg.get("nomask", True))
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → stay in FP32)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly tuned and runs in FP32.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, dim]  FP32

    # Flatten to (R, dim) and cast to FP16 for the fused GEMM
    x_norm_flat = x_norm.view(R, dim).to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single half‑precision GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)               # (5*H, dim)

    proj = F.linear(x_norm_flat, fused_w)                # (R, 5*H)  FP16
    del fused_w, x_norm, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – will never be read because HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Tuned launch configuration
    BLOCK_R = 128   # rows per program
    BLOCK_H = 64    # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM: left @ rightᵀ  (core einsum)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)  FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm statistics (mean / rsqrt‑var) over hidden dim
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1, keepdim=True)                     # (B,1,N,N) FP32
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)     # (B,1,N,N) FP32
    rstd = torch.rsqrt(var + eps)                                 # (B,1,N,N) FP32

    # ------------------------------------------------------------------
    # 6️⃣ Apply to_out_norm, out_gate and final projection (FP32 → FP16 → FP32)
    # ------------------------------------------------------------------
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1,H,1,1)
    beta  = w["to_out_norm.bias"  ].view(1, H, 1, 1)   # (1,H,1,1)

    # LayerNorm + gate (all in FP32 for numerical stability)
    norm = (out_fp32 - mu) * rstd                     # (B,H,N,N) FP32
    norm = norm * gamma + beta                        # (B,H,N,N) FP32
    norm = norm * out_gate.float()                    # (B,H,N,N) FP32

    # Final linear projection (hidden_dim → dim)
    norm_flat = norm.view(R, H)                       # (R, H)
    final = F.linear(norm_flat, w["to_out.weight"])   # (R, dim)  FP32
    final = final.view(B, N, N, dim)                  # (B, N, N, dim)

    return final
