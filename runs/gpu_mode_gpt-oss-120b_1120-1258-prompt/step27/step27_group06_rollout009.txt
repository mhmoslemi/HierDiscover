Pseudo code:

pid = tl.program_id(0)
row_start = pid * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
mask_m = offs_m < total_rows

nn = N * N
b = offs_m // nn
rem = offs_m - b * nn
i = rem // N
j = rem - i * N

hidden_stride = nn

base_out = b * (hidden_dim * hidden_stride) + i * N + j  # start offset for h=0

# first pass: compute sum and sum_sq
sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

for k_start in range(0, hidden_dim, BLOCK_K):
    offs_k = k_start + tl.arange(0, BLOCK_K)
    mask_k = offs_k < hidden_dim
    # out pointer
    out_ptr_tile = out_ptr + base_out[:, None] + offs_k[None, :] * hidden_stride
    vals_out = tl.load(out_ptr_tile,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
    sum_val += tl.sum(vals_out, axis=1)
    sum_sq += tl.sum(vals_out * vals_out, axis=1)

mean = sum_val / hidden_dim
var = sum_sq / hidden_dim - mean * mean
rstd = tl.rsqrt(var + eps)

# second pass: compute normalized & gated and store to buf
for k_start in range(0, hidden_dim, BLOCK_K):
    offs_k = k_start + tl.arange(0, BLOCK_K)
    mask_k = offs_k < hidden_dim

    # load out values again
    out_ptr_tile = out_ptr + base_out[:, None] + offs_k[None, :] * hidden_stride
    vals_out = tl.load(out_ptr_tile,
                       mask=mask_m[:, None] & mask_k[None, :],

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm (all kernels are Triton JIT):
1️⃣  Row‑wise LayerNorm over the last dimension (float32 → float16).
2️⃣  A single fused linear projection that produces the five weight maps
    (left, right, left‑gate, right‑gate, out‑gate) in FP16.
3️⃣  Split the fused projection, apply the optional pair mask and the three
    sigmoid gates, writing `left`, `right` and `out_gate` (layout [B, H, N, N]).
4️⃣  Batched matrix‑multiply (cuBLAS) implements the core einsum:
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   → [B, H, N, N] (FP16).
5️⃣  One fused kernel computes, for every (b,i,j) pair,
       * the mean / variance of `out` over the hidden dimension H,
       * the second LayerNorm (γ,β) on `out`,
       * multiplication by `out_gate`,
       and writes the result into a temporary buffer `gated` of shape
       [R, H] (R = B·N·N).
6️⃣  A single torch.nn.functional.linear projects `gated` (FP16) with the
    final weight matrix `to_out.weight` (FP32).  The output is cast to
    float32 and reshaped to [B, N, N, dim].

Only the forward pass is required; no gradients are computed.
The kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] FP32 input (flattened pair tensor)
    y_ptr,    # [R, dim] FP16 output
    w_ptr,    # [dim] γ (FP32)
    b_ptr,    # [dim] β (FP32)
    rows,     # total number of rows R (scalar)
    dim,      # hidden dimension `dim` (scalar)
    eps,      # epsilon for numerical stability (scalar)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1st pass: compute mean and variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                    # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)             # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2nd pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                # stride when moving in hidden dimension
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden plane 0

    # Loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Slice of the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate, and write to a [R, H] buffer
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_buffer_kernel(
    out_ptr,        # [B, H, N, N] FP16 (flattened as described below)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    buf_ptr,        # [R, H] FP16   (output of LN + gating)
    total_rows,     # R = B·N·N  (scalar)
    hidden_dim,     # H (scalar)
    N,              # sequence length (scalar)
    eps,            # epsilon for LayerNorm (scalar)
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_K: tl.constexpr,      # tile over hidden dimension
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (hidden_dim * hidden_stride) + i * N + j   # offset of hidden‑plane 0

    # ---------- first pass: compute mean and variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, hidden_dim, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < hidden_dim

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: LayerNorm + out_gate, write buffer ----------
    for k_off in range(0, hidden_dim, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # load raw values
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32

        # (out - µ) * rstd → γ· + β
        norm = ((vals_out - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # Apply out_gate
        gated = norm * vals_gate

        # Store to [R, H] buffer (row‑major)
        dst = offs_m[:, None] * hidden_dim + offs_k[None, :]
        tl.store(buf_ptr + dst,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]          (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict must contain "dim", "hidden_dim" and optionally "nomask"
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack and set constants
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of (b,i,j) pair rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                         # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # shape [5·H, dim]
    proj = F.linear(x_norm, fused_weight)   # (R, 5·H) FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoids → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ   →   out (B, H, N, N) FP16
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (fused) → buffer [R, H] (FP16)
    # --------------------------------------------------------------
    buf = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    grid_norm_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_gate_buffer_kernel[grid_norm_gate](
        out,
        out_gate,
        buf,
        R,
        H,
        N,
        eps,
        weights["to_out_norm.weight"],   # γ  FP32
        weights["to_out_norm.bias"],     # β  FP32
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection (to_out)
    # --------------------------------------------------------------
    # `buf` is (R, H) FP16; torch's linear will up‑cast to FP32 because
    # the weight is FP32.  The result is directly float32.
    final = F.linear(buf, weights["to_out.weight"])          # [R, dim] FP32
    final = final.view(B, N, N, dim)                         # [B, N, N, dim]

    return final
