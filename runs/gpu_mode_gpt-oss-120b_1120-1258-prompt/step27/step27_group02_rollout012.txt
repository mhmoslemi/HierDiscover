"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (dim) using a custom Triton
   kernel (FP32 accumulation → FP16 output).

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM:
        proj = X_norm @ W_fused.T          # shape (R, 5·H)
   where R = B·N·N and H = hidden_dim.

3️⃣ Split the fused projection, apply the (optional) pair mask and compute
   sigmoid gates – all done with plain PyTorch tensor ops (cheap, fully
   vectorised).

4️⃣ Core “tri‑multiplicative” contraction:
        out[b,h,i,j] = Σ_k left[b,h,i,k] · right[b,h,j,k]
   realised as a batched GEMM (torch.bmm) on tensors of shape
   (B·H, N, N).  The hidden dimension H (=128) is treated as the batch‑
   dimension.

5️⃣ Row‑wise mean & variance of `out` over the hidden dimension (LayerNorm
   over H).  We use the native PyTorch reduction kernels (highly tuned).

6️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   linear projection into two cheap tensor ops and a final FP16 GEMM:
        out_norm = ((out - μ)·rstd)·γ + β          (γ,β = to_out_norm params)
        gated    = out_norm * out_gate
        final    = gated @ W_out.T                (W_out from to_out.weight)

The only Triton kernel required is the row‑wise LayerNorm (step 1),
satisfying the “must use a kernel” rule while keeping launch overhead
minimal.  All heavy linear algebra is delegated to cuBLAS (torch.nn.functional.)
Mixed‑precision (FP16) is used throughout, with FP32 accumulations only where
necessary (mean/variance, final LayerNorm) to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 accumulation → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] FP32 input
    y_ptr,            # [R, D] FP16 output
    w_ptr,            # [D]   LN weight (γ) FP32
    b_ptr,            # [D]   LN bias   (β) FP32
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        weights_dict : mapping containing all model parameters
        config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs / config
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened token space
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (FP32)
        weights["norm.bias"],     # β (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: 5 × hidden_dim linear maps in one GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split → mask (optional) → sigmoid gates (all in PyTorch)
    # ------------------------------------------------------------------
    # optional mask broadcasted over the hidden dimension
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R, 1)   # [R,1]
    else:
        mask_flat = torch.ones((R, 1), dtype=torch.float16, device=device)

    # slices
    left_proj = proj[:, 0:H]                 # [R, H]
    right_proj = proj[:, H:2 * H]            # [R, H]

    # sigmoid gates – compute each gate slice separately to keep memory low
    left_gate = torch.sigmoid(proj[:, 2 * H:3 * H].to(torch.float32)).to(torch.float16)
    right_gate = torch.sigmoid(proj[:, 3 * H:4 * H].to(torch.float32)).to(torch.float16)
    out_gate = torch.sigmoid(proj[:, 4 * H:5 * H].to(torch.float32)).to(torch.float16)
    del proj

    # apply mask and gates
    left = left_proj * mask_flat * left_gate
    right = right_proj * mask_flat * right_gate

    # reshape to [B, H, N, N]
    left = left.view(B, H, N, N)
    right = right.view(B, H, N, N)
    out_gate = out_gate.view(B, H, N, N)

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat = left.reshape(B * H, N, N)        # [B·H, N, N]  FP16
    right_mat = right.reshape(B * H, N, N)      # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise (over H) mean & variance → LayerNorm statistics
    # ------------------------------------------------------------------
    # Convert to FP32 for numerically stable reductions
    out_f32 = out.float()                       # [B, H, N, N] FP32

    # mean / variance over the hidden dimension
    mu = out_f32.mean(dim=1, keepdim=True)                  # [B,1,N,N]
    var = out_f32.var(dim=1, unbiased=False, keepdim=True) # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                           # [B,1,N,N]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    # to_out_norm parameters (γ, β) – reshape for broadcasting
    ln_w = weights["to_out_norm.weight"].view(1, H, 1, 1)   # FP32
    ln_b = weights["to_out_norm.bias"].view(1, H, 1, 1)     # FP32

    # LayerNorm over H (using pre‑computed μ, rstd)
    out_norm = ((out_f32 - mu) * rstd) * ln_w + ln_b       # FP32, shape [B,H,N,N]

    # Multiply by out_gate (cast gate to FP32 for the product)
    gated = out_norm * out_gate.float()                     # FP32

    # Collapse batch & spatial dimensions → (R, H) for the final linear
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)   # FP32

    # Final projection: (R, H) @ (H, dim) → (R, dim)
    w_out = weights["to_out.weight"].to(torch.float16)                # [dim, H] FP16
    # Cast to FP16 for the GEMM (Tensor Cores)
    gated_fp16 = gated_flat.to(torch.float16)
    final_fp16 = torch.nn.functional.linear(gated_fp16, w_out)        # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # FP32 output

    # ------------------------------------------------------------------
    return final
