@triton.jit
def norm_proj_gate_kernel(
    x_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, dim, H, N,
    eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # load optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # broadcast to hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute mean and variance for each row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for d_start in range(0, dim, BLOCK_K):
        offs_d = d_start + tl.arange(0, BLOCK_K)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_K] FP32
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # allocate accumulators for projections, per hidden block step
    # We'll loop over hidden blocks
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h=0

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # init accumulators for this hidden block
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # iterate over dim blocks again
        for d_start in range(0, dim, BLOCK_K):
            offs_d = d_start + tl.arange(0, BLOCK_K)
            mask_d = offs_d < dim

            x_fp32 = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            # normalize
            x_norm = ((x_fp32 - mean[:, None]) * rstd[:, None]).to(tl.float16)  # [BLOCK_R, BLOCK_K] FP16

            # load weights slices for each projection, shape [BLOCK_K, BLOCK_H]
            w_left = tl.load(left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)   # FP16
            w_right = tl.load(right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0)
            w_left_gate = tl.load(left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                  mask=mask_d[:, None] & mask_h[None, :],
                                  other=0.0)
            w_right_gate = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0)
            w_out_gate = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                 mask=mask_d[:, None] & mask_h[None, :],
                                 other=0.0)

            # dot product: (BLOCK_R, BLOCK_K) dot (BLOCK_K, BLOCK_H) -> (BLOCK_R, BLOCK_H)
            left_acc += tl.dot(x_norm, w_left)    # returns FP32
            right_acc += tl.dot(x_norm, w_right)
            left_gate_acc += tl.dot(x_norm, w_left_gate)
            right_gate_acc += tl.dot(x_norm, w_right_gate)
            out_gate_acc += tl.dot(x_norm, w_out_gate)

        # after dim loop, apply sigmoid gating and mask, store
        # sigmoid on gates (FP32) then convert to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # apply mask to left/right
        left_val = (left_acc.to(tl.float16) * left_gate * mask_val).to(tl.float16)
        right_val = (right_acc.to(tl.float16) * right_gate * mask_val).to(tl.float16)

        # store to left/right/out_gate
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])
```

@triton.jit
def final_norm_proj_kernel(
    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)   # rows
    col_pid = tl.program_id(1)   # output dim tile

    row_start = pid * BLOCK_M
    col_start = col_pid * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # first pass: compute mean and variance across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # second pass: compute final output
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # LN: (out - mu) * rstd -> apply layernorm weight & bias
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # H
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * vals_gate

        term_fp16 = term.to(tl.float16)

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – highly‑optimized Triton implementation
================================================================
The forward pass of AlphaFold‑3 “outgoing” TriMul consists of

1️⃣  Rowwise LayerNorm on the flattened ``[B,N,N,D]`` pair tensor.
2️⃣  A fused linear projection that produces the three tensors
    ``left``, ``right`` and ``out_gate`` (including the optional mask
    and the three sigmoid gates).
3️⃣  Pairwise GEMM   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    (batched half‑precision `torch.bmm` → TensorCore GEMM).
4️⃣  A single Triton kernel that
    * computes per‑row mean / variance of the contracted tensor,
    * applies the second LayerNorm (γ/β),
    * multiplies by ``out_gate``,
    * and finally projects to ``dim`` with the ``to_out`` weight.

Both the LayerNorm + projection (step 2) and the final‐stage
normalisation + projection (step 4) are fused into custom kernels,
reducing the number of kernel launches and global‑memory traffic.
The implementation works for any ``dim ∈ {128,384}``,
``hidden_dim ≤ 128`` and for the full range of sequence lengths used
in the test matrix.

The entry point is the function ``custom_kernel`` with the required
signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm + fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_gate_kernel(
    # --- inputs -------------------------------------------------------
    x_ptr,               # [R, D]  float32, flattened input
    mask_ptr,            # [R]  float16, optional (0/1). ignored if HAS_MASK==0
    # --- outputs ------------------------------------------------------
    left_ptr,            # [B, H, N, N] float16
    right_ptr,           # [B, H, N, N] float16
    out_gate_ptr,        # [B, H, N, N] float16
    # --- weights (transposed) ----------------------------------------
    left_w_ptr,          # [D, H] float32  (left_proj weight.T)
    right_w_ptr,         # [D, H] float32
    left_gate_w_ptr,     # [D, H] float32
    right_gate_w_ptr,    # [D, H] float32
    out_gate_w_ptr,      # [D, H] float32
    # --- shape / config ----------------------------------------------
    R, D, H, N,
    eps,
    # --- compile‑time params -----------------------------------------
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Decode the flat index r = b·N·N + i·N + k  →  (b,i,k)
    # ------------------------------------------------------------------
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                    # [BLOCK_R,1] FP16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean / variance of the *raw* input x
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_start in range(0, D, BLOCK_K):
        offs_d = d_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_r[:, None] * D + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_K]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1/σ

    # ------------------------------------------------------------------
    # Second pass: fused projection, gating and optional mask
    # ------------------------------------------------------------------
    hidden_stride = N * N                                   # stride of one H‑plane
    base_offset = b * (H * hidden_stride) + i * N + k      # offset for h = 0

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # per‑block accumulators for the five linear maps
        left_sum      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_sum     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_sum = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_sum= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_sum  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_start in range(0, D, BLOCK_K):
            offs_d = d_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
            mask_d = offs_d < D

            # raw input values
            x = tl.load(
                x_ptr + offs_r[:, None] * D + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                               # [BLOCK_R, BLOCK_K]

            # ---------- normalize ----------
            x_norm = ((x - mean[:, None]) * rstd[:, None]).to(tl.float16)  # FP16

            # ---------- weight slices (transposed) ----------
            # layout: [D, H] → offset  = d * H + h
            w_left      = tl.load(
                left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_right     = tl.load(
                right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_left_gate = tl.load(
                left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_right_gate= tl.load(
                right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_out_gate  = tl.load(
                out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            )

            # ---------- dot products (FP16·FP16 → FP32) ----------
            left_sum      += tl.dot(x_norm, w_left)      # [BLOCK_R, BLOCK_H] FP32
            right_sum     += tl.dot(x_norm, w_right)
            left_gate_sum += tl.dot(x_norm, w_left_gate)
            right_gate_sum+= tl.dot(x_norm, w_right_gate)
            out_gate_sum  += tl.dot(x_norm, w_out_gate)

        # --------------------------------------------------------------
        # Apply sigmoid to gate tensors, mask to left/right, store results
        # --------------------------------------------------------------
        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_sum))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_sum))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_sum))).to(tl.float16)

        # left/right before masking: cast projection to FP16
        left_proj  = left_sum.to(tl.float16)
        right_proj = right_sum.to(tl.float16)

        left_val  = left_proj * left_gate  * mask_val
        right_val = right_proj * right_gate * mask_val
        # out_gate does NOT receive the (binary) mask
        out_gate_val = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R,BLOCK_H]

        tl.store(left_ptr      + dst_offset, left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Final LayerNorm + gating + projection (merged stats + output)
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    out_ptr,            # [B, H, N, N] FP16   (result of the pairwise GEMM)
    gate_ptr,           # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,           # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,           # [H] FP32  (to_out_norm.bias)
    w_out_T_ptr,        # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,      # [B, N, N, dim] FP32   (final result)
    R, H, N, dim,
    eps,
    # compile‑time tiling
    BLOCK_M: tl.constexpr,   # rows per program (tiles over R)
    BLOCK_K: tl.constexpr,   # hidden‑dim chunk
    BLOCK_N: tl.constexpr,   # output‑dim chunk
):
    pid_m = tl.program_id(0)      # iterates over rows   (R = B·N·N)
    pid_n = tl.program_id(1)      # iterates over output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane h=0

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / rsqrt‑variance over the hidden dimension H
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Apply LN (γ,β), multiply by out_gate, and project to dim
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw tensors
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)       # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)       # [M,K]

        # LayerNorm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ)·σ⁻¹  →  γ·· + β
        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # gating
        term = norm * vals_gate                     # [M,K]  FP32

        # cast to FP16 for Tensor‑core dot with the final projection matrix
        term_fp16 = term.to(tl.float16)

        # final projection slice w_out_T[offs_k, offs_n] → [K, TILE_N]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                     # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # 3️⃣  Store result
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, already on the same device as the input)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣  Prepare transposed weight matrices (dim → hidden)
    # ------------------------------------------------------------------
    # All linear weights are stored as [out_dim, in_dim] in the dict.
    # We transpose them to [in_dim, out_dim] for easier column‑wise loads.
    left_w_T      = w["left_proj.weight"].T.to(torch.float32).contiguous()
    right_w_T     = w["right_proj.weight"].T.to(torch.float32).contiguous()
    left_gate_w_T = w["left_gate.weight"].T.to(torch.float32).contiguous()
    right_gate_w_T= w["right_gate.weight"].T.to(torch.float32).contiguous()
    out_gate_w_T  = w["out_gate.weight"].T.to(torch.float32).contiguous()

    # ------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm, projection, mask & sigmoid gates
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, D] float32
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_K = 32
    grid_norm = (triton.cdiv(R, BLOCK_R),)

    norm_proj_gate_kernel[grid_norm](
        x_flat,
        mask_ptr,
        left,
        right,
        out_gate,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R, dim, H, N,
        eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    del x_flat, mask_ptr

    # ------------------------------------------------------------------
    # 2️⃣  Batched GEMM (half‑precision TensorCore) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]  (FP16)
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]  (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣  Final LayerNorm + out_gate * projection (merged stats + output)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program (tiles over R)
    BLOCK_K_OUT = 32     # hidden‑dim chunk (≤ hidden_dim)
    BLOCK_N_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),      # over rows B·N·N
        triton.cdiv(dim, BLOCK_N_OUT),    # over output dimension
    )
    final_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
