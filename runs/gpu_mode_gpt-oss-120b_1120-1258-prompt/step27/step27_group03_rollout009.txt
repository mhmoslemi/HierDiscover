    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

"""
TriMul (outgoing) – high‑performance forward‑only implementation.

The operator works on a 4‑D pair tensor x of shape [B, N, N, dim] and
returns a tensor of the same shape.  The computation follows the
“outgoing” variant from AlphaFold‑3:

1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ One fused Linear that produces the five projections (left, right,
   left_gate, right_gate, out_gate) of shape (R, 5·H) where
   R = B·N·N and H = hidden_dim.
3️⃣ A Triton kernel splits the fused projection, applies the optional mask,
   and evaluates the three sigmoids, writing *left*, *right* and *out_gate*
   in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core contraction
   out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →   (B·H, N, N) FP16.
5️⃣ A second Triton kernel computes, for every (b,i,j):
      μ  = mean_h(out)
      σ² = var_h(out)
      rstd = rsqrt(σ²+eps)
      term[h] = ((out[h]‑μ)·rstd·γ[h] + β[h]) * out_gate[h]    (γ,β are the
      to_out_norm parameters).  The result *term* has shape [B, H, N, N]
      (FP16) and is stored in the same layout as *out*.
6️⃣ The final projection is a single large Linear (cuBLAS) that maps the
   hidden dimension H→dim:
        out_final = Linear(term, weight_to_out)   (FP16)
   The output is reshaped to [B, N, N, dim] and cast to float32.

Only the two Triton kernels (layer‑norm & split‑gate, term‑norm‑gate) are
written in‑house; all heavy matrix‑multiplication work is delegated to
cuBLAS, which on an H100 gives the best throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    LN weight (γ) FP32
    b_ptr,          # [dim]    LN bias  (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- pass‑1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass‑2: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, optional mask & three sigmoids
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Broadcasted mask over hidden dimension (if present)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mval[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden slices
    # Base offset for h = 0
    base_off = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # (R, 5·H) layout, column‑wise concat of the five weight groups
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five chunks
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 (more stable) then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offset for the current hidden slice
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Compute row‑wise μ/rstd, apply to_out_norm + out_gate
# ----------------------------------------------------------------------
@triton.jit
def term_norm_gate_kernel(
    out_ptr,        # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    gamma_ptr,      # [H]          FP32 (to_out_norm.weight)
    beta_ptr,       # [H]          FP32 (to_out_norm.bias)
    term_ptr,       # [B, H, N, N] FP16 (output of this kernel)
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each pair (b,i,j) (flattened to row ``r``) compute:
        μ  = mean_h(out[b,h,i,j])
        σ  = variance_h(out[...])
        rstd = rsqrt(σ + eps)
        term_h = ((out - μ) * rstd * γ[h] + β[h]) * gate
    The result is written as ``term_ptr`` (same layout as ``out_ptr``).
    The kernel performs two passes over the hidden dimension to avoid an
    extra global‑memory write of μ/rstd.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j     # start of hidden plane 0

    # ---------- first pass: compute μ and rstd ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                         # [BLOCK_R]

    # ---------- second pass: normalize, apply LN‑weight/bias & gate ----------
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out
        ptr_out = out_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride
        out_vals = tl.load(
            ptr_out,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_R, BLOCK_H]

        # load gate
        ptr_gate = gate_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride
        gate_vals = tl.load(
            ptr_gate,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_R, BLOCK_H]

        # load LN parameters for this hidden slice
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(beta_ptr + offs_h,  mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd  →  layer‑norm →  *γ + β
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by the previously computed out_gate
        term = norm * gate_vals                     # FP32

        # write back (FP16) in the same layout as ``out_ptr``
        ptr_term = term_ptr + base_off[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            ptr_term,
            term.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Triton‑based TriMul (outgoing) entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul implementation.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict of FP32 parameters
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # flattened pair count
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                       # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ   →   out (B, H, N, N)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16
    del left_mat, right_mat  # free memory

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise stats + to_out_norm + out_gate  →  term
    # --------------------------------------------------------------
    term = torch.empty_like(out)   # will hold the gated & normalized tensor

    BLOCK_R_TERM = 128
    BLOCK_H_TERM = 32
    grid_term = (triton.cdiv(R, BLOCK_R_TERM),)
    term_norm_gate_kernel[grid_term](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        term,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=4,
    )
    del out, out_gate  # no longer needed

    # --------------------------------------------------------------
    # 6️⃣ Final Linear (hidden_dim → dim) via cuBLAS
    # --------------------------------------------------------------
    # term layout currently: [B, H, N, N]
    term_perm = term.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    term_flat = term_perm.view(R, H)                    # [R, H]

    w_out = w["to_out.weight"].to(torch.float16)        # [dim, H]
    out_final_fp16 = F.linear(term_flat, w_out)        # [R, dim] FP16
    out_final = out_final_fp16.view(B, N, N, dim).to(torch.float32)

    return out_final
