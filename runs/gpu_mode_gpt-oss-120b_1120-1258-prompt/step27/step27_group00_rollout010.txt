    col_start = pid_n*BLOCK_N

    mask_n = offs_n < dim

    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base_offset = b * (H * hidden_stride) + i * N + j

    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    acc_S1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_S4 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_S3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        k_end = tl.minimum(k_start + BLOCK_K, H)
        cur_K = k_end - k_start
        offs_k = k_start + tl.arange(0, BLOCK_K)

                            other=0.0).to(tl.float32)   # [M,cur_K]

        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [cur_K]

        # scale weight for Gamma and Beta

        w_tile = w_tile_raw.to(tl.float32)

        # left_S4 = gate_vals * gamma[:, None]
        left_S4 = gate_vals * gamma[:, None]
        # left_S3 = gate_vals * beta[:, None]
        left_S3 = gate_vals * beta[:, None]

        acc_S4 += tl.dot(left_S4, w_gamma).to(tl.float32)
        acc_S3 += tl.dot(left_S3, w_beta).to(tl.float32)

    # Compute mean and rstd
    mu = sum_out / H

"""
Outgoing TriMul (AlphaFold‑3 style) – Triton implementation
========================================================

We fuse the two post‑GEMM kernels (row‑wise statistics + second LayerNorm
+ out‑gate + final projection) into a **single** Triton kernel.  The kernel
  * loads the contracted tensor `out` and the gate `out_gate` once,
  * accumulates per‑pair mean/variance (required for the second LayerNorm),
  * accumulates the three dot‑products needed for the final linear projection

        S1 = Σ_h  out_h·γ·gate_h·W_h
        S4 = Σ_h  γ·gate_h·W_h
        S3 = Σ_h  β·gate_h·W_h

    (where `γ,β` are the LayerNorm parameters and `W` the final weight).
  * after the hidden‑dimension loop the mean/rsqrt‑variance are obtained
    and the final output is computed as

        out_final = rstd·(S1 – μ·S4 + S3)

All tensors stay on the GPU; the final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pairwise tensor
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] FP32 input
    y_ptr,            # [R, dim] FP16 output
    w_ptr,            # [dim]    LN weight (γ)  FP32
    b_ptr,            # [dim]    LN bias   (β)  FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split the fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                   # no mask for out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fused LN + out‑gate + final projection (single pass over `out`)
# ----------------------------------------------------------------------
@triton.jit
def fused_final_kernel(
    out_ptr,          # [B, H, N, N]  FP16   (left @ rightᵀ)
    gate_ptr,         # [B, H, N, N]  FP16   (out_gate)
    ln_w_ptr,         # [H]           FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]           FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]      FP16   (transposed final weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over rows (R = B·N·N)
    pid_n = tl.program_id(1)   # tile over output dimension (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # ---------- per‑row accumulators ----------
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    acc_S1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·γ·gate·W
    acc_S4 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ γ·gate·W
    acc_S3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β·gate·W

    for k_start in range(0, H, BLOCK_K):
        k_end = tl.minimum(k_start + BLOCK_K, H)
        cur_K = k_end - k_start
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ----- load contracted tensor and gate (FP16 → FP32) -----
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)                # [M, cur_K]

        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [M, cur_K]

        # ----- update mean / variance accumulators -----
        sum_out += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

        # ----- load LN parameters (γ,β) for this hidden slice -----
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [cur_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [cur_K]

        # ----- load slice of final weight and apply γ / β scaling -----
        ptr_w = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [cur_K, BLOCK_N]
        w_tile_fp16 = tl.load(ptr_w,
                              mask=mask_k[:, None] & mask_n[None, :],
                              other=0.0)                                 # FP16
        w_tile = w_tile_fp16.to(tl.float32)                            # FP32

        w_gamma = w_tile * gamma[:, None]    # [cur_K, BLOCK_N]
        w_beta  = w_tile * beta[:, None]     # [cur_K, BLOCK_N]

        # ----- build left matrices for the three dot‑products -----
        #   (out·gate)·γ
        left_S1 = (out_vals * gate_vals) * gamma[:, None]   # [M, cur_K]
        #   γ·gate
        left_S4 = gate_vals * gamma[:, None]                # [M, cur_K]
        #   β·gate
        left_S3 = gate_vals * beta[:, None]                 # [M, cur_K]

        # ----- three independent GEMMs (M×K)·(K×N) -----
        acc_S1 += tl.dot(left_S1, w_gamma)   # [M,N]
        acc_S4 += tl.dot(left_S4, w_gamma)   # [M,N]
        acc_S3 += tl.dot(left_S3, w_beta)    # [M,N]

    # ---------- compute μ, rstd ----------
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---------- final output ----------
    # out_final = rstd·(S1 – μ·S4 + S3)
    term   = acc_S1 - mu[:, None] * acc_S4 + acc_S3
    final  = rstd[:, None] * term

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry‑point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator (AlphaFold‑3).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – cuBLAS
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                      # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                   # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fused second LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128            # rows per program (tiles over R)
    BLOCK_K_OUT = 32             # hidden‑dim chunk (≤ hidden_dim)
    BLOCK_N_OUT = 128            # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),          # over rows (B·N·N)
        triton.cdiv(dim, BLOCK_N_OUT),        # over output dimension
    )
    fused_final_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
