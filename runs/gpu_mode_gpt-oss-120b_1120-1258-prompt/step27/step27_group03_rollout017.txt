def custom_kernel(data):
    x, mask, w, cfg = data
    B,N,_,dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # 1 LayerNorm on input x (FP32)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps).to(torch.float16)

    # 2 Linear projection fused for 5*H
    fused_w = torch.cat([...], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # (R,5H) FP16

    # 3 Split and apply mask and gates
    R = B * N * N
    proj = proj.view(R,5,H)
    if not nomask:
        mask_flat = mask.view(R,1).to(torch.float16)
        proj = proj * mask_flat   # broadcast across H
    # slices
    left_raw = proj[:,0,:]   # shape (R,H)
    right_raw = proj[:,1,:]
    left_gate_raw = proj[:,2,:]
    right_gate_raw = proj[:,3,:]
    out_gate_raw = proj[:,4,:]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

"""
TriMul (outgoing) – fast forward‑only Triton+PyTorch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣  Fused linear projection: 5·hidden_dim outputs (left, right, three gates)
    with a single cuBLAS GEMM (FP16 weights).
3️⃣  Split the projection, apply the optional mask and the three sigmoid gates
    in a small Triton kernel – writes the three tensors directly in the layout
    required for the subsequent batched GEMM.
4️⃣  Batched matrix multiplication (cuBLAS) over the hidden dimension:
    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k].
5️⃣  Row‑wise LayerNorm over the hidden dimension, multiplication by the
    output‑gate and the final linear projection are fused into a single Triton
    kernel (two passes over the hidden dimension to obtain mean/var and then
    compute the final output).  The result is FP32 as required.

All kernels target Triton 3.3.1 on an H100.  Mixed‑precision (FP16/FP32)
computations are used throughout, and only the kernel in step 3‑5 is
hand‑written – the rest uses highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]    FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flat index r = b·N·N + i·N + k ----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask over hidden dim (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride inside hidden dimension
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden dim 0 for (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointer to the first of the 5 concatenated projections
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offset for the current hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr_out,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass‑2: LayerNorm, out_gate, and final projection
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        # Load hidden‑plane values and the gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # LayerNorm (γ,β) over hidden dim
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        norm = (vals_out - mean[:, None]) * rstd[:, None]          # centre & scale
        norm = norm * gamma[None, :] + beta[None, :]               # apply γ,β

        # Multiply by out_gate
        term = norm * vals_gate                                    # FP32

        # Cast to FP16 for the dot product (Tensor‑core friendly)
        term_fp16 = term.to(tl.float16)

        # Load slice of final weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # FP16

        # (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # --------------------------------------------------------------
    # Store final tensor
    # --------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation (FP16‑accelerated).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing (all FP32 unless noted):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use highly‑optimised cuDNN kernel
    # --------------------------------------------------------------
    # flatten to [R, dim] and apply LayerNorm
    x_flat = x.view(R, dim).contiguous()                                 # FP32
    x_norm = F.layer_norm(x_flat,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).to(torch.float16)               # FP16
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim outputs)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)             # (R, 5·H)  FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)                # [R]
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N).contiguous()
    right_mat = right.view(B * H, N, N).contiguous()

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate & final projection
    # --------------------------------------------------------------
    # transposed final weight (H, dim) – FP16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (tiles over R)
    BLOCK_K_OUT = 64       # hidden‑dim tile (≤ H)
    BLOCK_N_OUT = 128      # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )

    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # clean‑up
    del left, right, out_gate, out, w_out_T

    return final
