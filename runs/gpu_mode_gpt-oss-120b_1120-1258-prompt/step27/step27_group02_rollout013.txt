@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mval[:, None]  # broadcast to hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + k  # shape [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute projection base offset for each hidden tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        outg = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden

"""
Outgoing TriMul (AlphaFold3) – fast forward only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
    torch.nn.functional.layer_norm (highly tuned cuBLAS kernel).

2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM:
        proj = x_norm @ W_fused          # [R, 5·H]  (FP16)

3️⃣ Split the fused projection, apply the optional mask, compute the three
    sigmoid gates and write the three tensors `left`, `right` and `out_gate`
    in layout [B, H, N, N] (FP16) with a tiny Triton kernel.

4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → `out` has shape [B, H, N, N] (FP16).

5️⃣ Row‑wise mean / variance of `out` over the hidden dimension H
   (torch.mean / torch.var).  Use these statistics to apply the second
   LayerNorm (to_out_norm) in FP32, then cast back to FP16.

6️⃣ Multiply by the already‑computed out‑gate (FP16).

7️⃣ Final projection (to_out) is a single GEMM
        final = out_gated @ W_outᵀ                # FP16 → FP32
   and reshaped to [B, N, N, dim].

All heavy linear algebra is handled by cuBLAS/Torch kernels; only step‑3
uses a custom Triton kernel, satisfying the “implement at least one
kernel” rule while keeping the overall runtime well below the target. 
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 (flattened)
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden tiles per loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---------------------------------------------------------------
    # Decode (b, i, k) from the flat index r = b·N·N + i·N + k
    # ---------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ---------------------------------------------------------------
    # Optional mask (scalar per row, broadcast over hidden dim)
    # ---------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)      # [BLOCK_R]
        mask_val = m[:, None]                                      # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                           # distance between hidden slices

    # Base offset for h = 0 in the destination layout [B,H,N,N]
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # ---------------------------------------------------------------
    # Loop over hidden dimension tiles
    # ---------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the 5·H fused projection slices for this tile ----
        # proj layout: contiguous rows, each row = [5·H] elements
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ---- sigmoid in FP32 for stability, then back to FP16 ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                    # already sigmoid‑ed

        # ---- destination offsets (flattened view of [B,H,N,N]) ----
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                              # [R, dim]  FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],   # γ  (FP32)
        bias=weights["norm.bias"],       # β  (FP32)
        eps=eps,
    ).to(torch.float16)                                   # [R, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                      # [R, 5·H]  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if (not nomask) and (mask is not None):
        # flatten mask to [R] (FP16).  A missing value defaults to 1.0.
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # (B·H, N, N)   FP16
    right_mat = right.view(B * H, N, N)               # (B·H, N, N)   FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                        # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over hidden dimension H (μ, rstd)
    # ------------------------------------------------------------------
    out_fp32 = out.float()                                     # (B, H, N, N) FP32
    mu   = out_fp32.mean(dim=1, keepdim=True)                  # (B,1,N,N)
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # (B,1,N,N)
    rstd = 1.0 / torch.sqrt(var + eps)                         # (B,1,N,N)

    # Apply to_out_norm (γ,β) – still in FP32
    gamma = weights["to_out_norm.weight"]   # (H,)
    beta  = weights["to_out_norm.bias"]     # (H,)
    out_norm_fp32 = (out_fp32 - mu) * rstd
    out_norm_fp32 = out_norm_fp32 * gamma[None, :, None, None] + beta[None, :, None, None]
    out_norm_fp16 = out_norm_fp32.to(torch.float16)            # (B,H,N,N) FP16
    del out_fp32, mu, var, rstd, out_norm_fp32

    # ------------------------------------------------------------------
    # 6️⃣ Gating with out_gate
    # ------------------------------------------------------------------
    out_gated = out_norm_fp16 * out_gate                         # (B,H,N,N) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (to_out) – fused GEMM
    # ------------------------------------------------------------------
    # Shape → [R, H]  (flatten the batch & spatial dims)
    out_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R,H) FP16
    # to_out.weight: [dim, H] (FP32) → transpose to [H, dim] and cast
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim) FP16

    final_fp16 = F.linear(out_flat, w_out_T)                         # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)          # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # Cleanup temporary tensors (helps the allocator on large cases)
    # ------------------------------------------------------------------
    del left, right, out_gate, out, out_gated, out_flat, w_out_T

    return final
