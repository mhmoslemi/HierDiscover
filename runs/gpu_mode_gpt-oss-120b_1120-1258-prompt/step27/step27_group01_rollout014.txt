"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (custom Triton kernel).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (cuBLAS) → shape (R, 5·H).
3️⃣  Split the fused projection with PyTorch tensor ops, apply the optional mask,
    compute sigmoid gates and reshape to [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).
5️⃣  LayerNorm over the hidden dimension H, multiplication with the
    out‑gate, and the final projection to `dim` are performed with
    PyTorch’s highly‑optimized kernels (no extra Triton kernels required).
   The final result is returned as float‑32.

Only the first LayerNorm is written in Triton, satisfying the “use Triton”
requirement while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, dim] FP32 input
    y_ptr,        # [R, dim] FP16 output
    w_ptr,        # [dim]   LN weight (γ) FP32
    b_ptr,        # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                       # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, seq_len, seq_len, dim] (float32)
        - mask_tensor  : torch.Tensor[bs, seq_len, seq_len] (bool/float) or None
        - weights_dict : mapping containing the model weights
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (custom Triton)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # columns per program
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣  Fused projection → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣  Split, mask, and compute gates (pure PyTorch – vectorised)
    # --------------------------------------------------------------
    proj_view = proj.view(B, N, N, 5 * H)               # [B, N, N, 5·H]

    left_proj   = proj_view[..., 0           : H]
    right_proj  = proj_view[..., H           : 2 * H]
    left_gate_r = proj_view[..., 2 * H       : 3 * H]
    right_gate_r= proj_view[..., 3 * H       : 4 * H]
    out_gate_r  = proj_view[..., 4 * H       : 5 * H]

    if (not nomask) and (mask is not None):
        mask_exp = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left = left_proj * mask_exp
        right = right_proj * mask_exp
    else:
        left = left_proj
        right = right_proj

    left_gate  = left_gate_r.sigmoid()
    right_gate = right_gate_r.sigmoid()
    out_gate   = out_gate_r.sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # reshape to [B, H, N, N] for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()
    del proj_view, left_proj, right_proj, left_gate_r, right_gate_r, out_gate_r

    # --------------------------------------------------------------
    # 4️⃣  Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm + out‑gate + final projection (PyTorch)
    # --------------------------------------------------------------
    # 5a – bring hidden dimension to the innermost axis
    out_perm   = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]
    gate_perm  = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]

    # 5b – LayerNorm over H
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                       # FP16, shape [B,N,N,H]

    # 5c – Apply out‑gate
    out_gated = out_norm * gate_perm                         # FP16

    # 5d – Final linear projection (H → dim)
    out_flat = out_gated.view(R, H).contiguous()            # [R, H] FP16
    final_fp16 = F.linear(out_flat, w["to_out.weight"].to(torch.float16))  # [R, dim] FP16

    # --------------------------------------------------------------
    # 6️⃣  Reshape & cast to float32
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
