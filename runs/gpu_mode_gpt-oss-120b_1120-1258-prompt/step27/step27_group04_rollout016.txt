"""
Outgoing TriMul (AlphaFold‑3 style) – highly‑optimized forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
   into a single torch.nn.functional.linear call producing shape (R,5·H).
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * decodes the flat pair index (b,i,k) → (b,H,i,k) layout,
   * applies the optional mask,
   * computes sigmoid gates in FP32,
   * writes three tensors `left`, `right`, `out_gate` in layout [B,H,N,N] (FP16).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →  [B,H,N,N] (FP16)
5️⃣ Triton kernel `final_fused_kernel` combines:
   * per‑row mean / variance of the contracted tensor,
   * the second LayerNorm (γ,β on hidden dim H),
   * gating with `out_gate`,
   * the final projection (`to_out` weight) → output `[B,N,N,dim]` (FP32).

   The kernel reads `out` and `out_gate` **once**, stores them in shared memory,
   computes μ/σ, then uses the cached values for the LN‑gate‑projection step,
   halving the memory traffic of the original two‑kernel implementation.

All kernels run in mixed‑precision (FP16 compute, FP32 reductions) and are
tuned for the H100 (Triton 3.3.1).  The entry point `custom_kernel` follows the
required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, D] FP32 input
    y_ptr,               # [R, D] FP16 output
    w_ptr,               # [D]   LN weight (γ) FP32
    b_ptr,               # [D]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R,5·H] FP16 fused projection
    left_ptr,       # [B,H,N,N] FP16
    right_ptr,      # [B,H,N,N] FP16
    out_gate_ptr,   # [B,H,N,N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride of one hidden plane in [B,H,N,N]
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R,BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate, and final projection
#    (read out/out_gate only once → shared‑memory cache)
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B,H,N,N] FP16   (contracted tensor)
    gate_ptr,         # [B,H,N,N] FP16   (out_gate)
    ln_w_ptr,         # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,    # [B,N,N,dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    # each program works on a tile (BLOCK_M rows) × (BLOCK_N output‑dim)
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode flattened pair index (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # offset for hidden = 0

    # ------------------------------------------------------------------
    # Shared memory buffers for the whole hidden vector of each row
    # (max hidden dim is 128 in all test configs → set compile‑time limit)
    # ------------------------------------------------------------------
    H_MAX = 128                                            # compile‑time upper bound
    out_sh   = tl.shared(tl.float16, BLOCK_M * H_MAX)     # [M * H_MAX]
    gate_sh  = tl.shared(tl.float16, BLOCK_M * H_MAX)     # [M * H_MAX]

    # ------------------------------------------------------------------
    # First pass: accumulate μ / σ² and store (out,gate) into shared memory
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # global loads
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_tile  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)                     # [M,K]  FP16
        gate_tile = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)                     # [M,K]  FP16

        # accumulate stats (convert to FP32 first)
        out_f32 = out_tile.to(tl.float32)
        sum_val += tl.sum(out_f32, axis=1)
        sum_sq  += tl.sum(out_f32 * out_f32, axis=1)

        # write into shared memory (flattened layout)
        sh_idx = (offs_m[:, None] * H_MAX) + (k_start + offs_k)[None, :]   # [M,K]
        out_sh[sh_idx]  = out_tile
        gate_sh[sh_idx] = gate_tile

    # ------------------------------------------------------------------
    # Compute per‑row mean and rsqrt‑variance (ε added for stability)
    # ------------------------------------------------------------------
    mean = sum_val / tl.float32(H)
    var  = sum_sq / tl.float32(H) - mean * mean
    rstd = tl.rsqrt(var + eps)               # [M]

    # ------------------------------------------------------------------
    # Second pass: LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # read cached values from shared memory
        sh_idx = (offs_m[:, None] * H_MAX) + (k_start + offs_k)[None, :]   # [M,K]
        out_tile  = out_sh[sh_idx]    # FP16
        gate_tile = gate_sh[sh_idx]   # FP16

        # LayerNorm parameters (γ,β) – FP32
        gamma = tl.load(ln_w_ptr + (k_start + offs_k), mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + (k_start + offs_k), mask=mask_k, other=0.0)   # [K]

        # ----- LN + out‑gate -----
        out_f32  = out_tile.to(tl.float32)
        gate_f32 = gate_tile.to(tl.float32)

        norm = (out_f32 - mean[:, None]) * rstd[:, None]          # (x‑μ)/σ
        norm = norm * gamma[None, :] + beta[None, :]              # γ·norm + β
        term = norm * gate_f32                                    # apply out_gate
        term_fp16 = term.to(tl.float16)                           # back to FP16 for GEMM

        # ----- final linear projection (to_out) -----
        w_ptr = w_out_T_ptr + (k_start + offs_k)[:, None] * dim + offs_n[None, :]   # [K,N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                            # FP16

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Write result: final tensor has layout [B,N,N,dim] → flattened row index * dim + col
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, already on the same device as the input)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of (b,i,j) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R,5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (tiles over R)
    BLOCK_K_OUT = 32       # hidden‑dim chunk (must divide H)
    BLOCK_N_OUT = 64       # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],    # γ  FP32
        w["to_out_norm.bias"],      # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
