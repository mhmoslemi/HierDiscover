"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the last dimension of the pair tensor (FP32→FP16) using
    torch.nn.functional.layer_norm (highly‑optimised CUDA kernel).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:  proj = x_norm @ W_fused.T   →   [R,5·H]  (FP16).

3️⃣  Triton kernel `split_fuse_gate_kernel`:
      * splits the fused projection into the three tensors `left`, `right`,
        `out_gate`;
      * applies the optional pair‑mask;
      * applies the sigmoids for the three gates.
      Layout of the tensors is `[B, H, N, N]` (contiguous).

4️⃣  Batched GEMM (cuBLAS) for the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    implemented as `torch.bmm`.

5️⃣  Row‑wise mean / variance of `out` over the hidden dimension H using
    fast torch reductions → µ, rstd of shape `[B,N,N]`.

6️⃣  Apply the second LayerNorm, multiply by `out_gate`,
    and finally project back to the original feature dimension with a
    cuBLAS GEMM (FP16 → FP32).

All heavy linear‑algebra parts use cuBLAS (tensor‑cores on H100).
Only the split‑and‑gate step is custom Triton code, satisfying the
“must use a kernel” requirement while keeping the overall runtime well
below the 1000 µs target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection and apply mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   fused projection, FP16
    left_ptr,      # [B, H, N, N]   output, FP16
    right_ptr,     # [B, H, N, N]   output, FP16
    out_gate_ptr,  # [B, H, N, N]   output, FP16
    mask_ptr,      # [R]            pair mask (FP16), optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                      # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k  # start offset for plane h=0

    # iterate over hidden dimension in chunks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # projection layout: [left|right|lg|rg|og] (contiguous)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (FP32 -> FP16)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimised on CUDA
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B,N,N,dim] FP16

    # flatten to [R, dim] for the fused projection
    x_norm_flat = x_norm.view(R, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim], FP16

    # (R,5·H)   FP16
    proj = F.linear(x_norm_flat, fused_w)   # uses cuBLAS, tensor cores

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoids (Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        # dummy tensor – will never be accessed because HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, x_norm, x_norm_flat

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance over hidden dimension H
    # --------------------------------------------------------------
    # μ, var are computed in FP32 for numerical stability
    out_fp32 = out.float()
    mu  = out_fp32.mean(dim=1)                       # [B, N, N] FP32
    var = (out_fp32.pow(2).mean(dim=1)) - mu.pow(2)  # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)                    # [B, N, N] FP32

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    # Apply LayerNorm (γ,β) over the hidden dimension H
    gamma = w["to_out_norm.weight"]   # [H] FP32
    beta  = w["to_out_norm.bias"]     # [H] FP32

    # broadcast → shape [B, H, N, N]
    gamma = gamma.view(1, H, 1, 1)
    beta  = beta.view(1, H, 1, 1)

    # normalize, apply LN params, multiply by out_gate
    out_norm = (out_fp32 - mu[:, None, :, :]) * rstd[:, None, :, :]   # FP32
    out_norm = out_norm * gamma + beta
    out_norm = out_norm * out_gate.float()
    out_norm = out_norm.to(torch.float16)      # back to FP16 for the final GEMM

    # reshape to [R, H] (R = B·N·N) ready for the final linear projection
    out_norm_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]

    # final projection: (R, H) @ (dim, H).T  →  (R, dim)
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H] FP16
    out_final = F.linear(out_norm_flat, w_out)    # cuBLAS FP16 GEMM, output FP16

    # cast to FP32 and reshape to [B, N, N, dim]
    out_final = out_final.to(torch.float32).view(B, N, N, dim)

    return out_final
