"""
Optimized outgoing TriMul (AlphaFold‑3) – forward only.
----------------------------------------------------
Algorithm
---------
1️⃣  Layer‑Norm on the input tensor `x` (float32 → fp16) using
    :func:`torch.nn.functional.layer_norm`.  The weight and bias are
    cast to fp16 so the kernel can stay in the fast cuDNN LN path.

2️⃣  Fuse the five linear maps
      (left_proj, right_proj, left_gate, right_gate, out_gate)
    into a single matrix‑multiply:
        proj = x_norm @ W_fused.T          # [R, 5·H]  (fp16)

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * decodes the flat index (b,i,k) → (b,i,k) layout,
    * applies the optional mask (broadcast over the hidden dim H),
    * evaluates the three sigmoid gates in FP32 and writes three tensors
      `left`, `right`, `out_gate` in layout **[B, H, N, N]** (fp16).

4️⃣  Batched GEMM (cuBLAS) performs the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]      # fp16

5️⃣  Second LayerNorm over the hidden dimension `H` (again via cuDNN).
    The tensor is permuted to make `H` the trailing dimension,
    normalized with `to_out_norm.{weight,bias}` (cast to fp16).

6️⃣  Element‑wise gating with `out_gate` (still fp16) and a final
    linear projection:
        out_final = (out_norm * out_gate) @ W_out.T          # fp32

All kernels run on the GPU; the only Triton kernel needed is the
`split_fuse_gate_kernel`, which fuses mask‑application, gating and
layout conversion.  The rest of the work uses highly‑optimized
cuBLAS / cuDNN kernels, giving a very low launch‑overhead and excellent
throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   fp16 fused projection
    left_ptr,       # [B, H, N, N] fp16
    right_ptr,      # [B, H, N, N] fp16
    out_gate_ptr,   # [B, H, N, N] fp16
    mask_ptr,       # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between successive hidden planes in the destination layout
    hidden_stride = N * N                # (N,N) plane size
    # offset of the first hidden plane (h = 0) for this (b,i,k) triple
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                 # out_gate is not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,      mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, already on the same device as the input)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (input -> fp16)
    # ------------------------------------------------------------------
    # Cast LN parameters to fp16 to satisfy torch.nn.functional.layer_norm
    norm_w_fp16 = w["norm.weight"].to(torch.float16)
    norm_b_fp16 = w["norm.bias"].to(torch.float16)

    # cuDNN LN over the last dimension `dim`
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w_fp16, bias=norm_b_fp16, eps=eps
    ).to(torch.float16)                # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim] fp16

    # flatten to (R, dim) for a single matmul; result (R,5·H) fp16
    proj = F.linear(x_norm.view(R, dim), fused_w)   # fp16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask  = 1
        mask_ptr  = mask_flat
    else:
        has_mask  = 0
        mask_ptr  = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64          # tuneable; 64 rows per program
    BLOCK_H = 64          # tuneable; 64 hidden units per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise einsum (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] fp16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                           # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over H) + out‑gate multiplication
    # ------------------------------------------------------------------
    # Make H the trailing dimension for cuDNN LayerNorm
    out_perm      = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # same layout

    # Cast LN parameters to fp16 as required by torch.nn.functional.layer_norm
    to_out_norm_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    out_norm = torch.nn.functional.layer_norm(
        out_perm, (H,), weight=to_out_norm_w_fp16, bias=to_out_norm_b_fp16, eps=eps
    )   # still fp16

    gated = out_norm * out_gate_perm   # element‑wise fp16 product
    del out, out_gate, out_norm, out_gate_perm, out_perm

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (H → dim) – using cuBLAS linear
    # ------------------------------------------------------------------
    gated_flat = gated.view(R, H)                     # [R, H] fp16
    w_out_T    = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] fp16

    final = F.linear(gated_flat, w_out_T)            # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)   # cast back to fp32

    return final
