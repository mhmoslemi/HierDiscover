@triton.jit
def proj_gate_kernel(
    x_ptr,            # [R, dim] fp16 (normed)
    left_ptr,         # [B, H, N, N] fp16
    right_ptr,        # [B, H, N, N] fp16
    out_gate_ptr,     # [B, H, N, N] fp16
    mask_ptr,         # [R] fp16 optional
    w_left_proj_ptr, w_right_proj_ptr,
    w_left_gate_ptr, w_right_gate_ptr,
    w_out_gate_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k

    # optional mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension in blocks
    for h_start in range(0, H, BLOCK_H):
        # define hidden range for this tile
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for dot results (float32 for accuracy)
        left_proj_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_proj_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # loop over dim
        for d_start in range(0, dim, BLOCK_K):
            offs_d = d_start + tl.arange(0, BLOCK_K)
            mask_d = offs_d < dim

            # load x block
            x_block = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)   # shape [BLOCK_R, BLOCK_K]

            # load weight slices (transposed) for each projection
            # shape [BLOCK_K, BLOCK_H] after loading transposed?
            # We'll load as transposed already: weight layout transposed.

            # left_proj weight slice
            w_left = tl.load(
                w_left_proj_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)  # shape [BLOCK_K, BLOCK_H]

            w_right = tl.load(
                w_right_proj_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_left_gate = tl.load(
                w_left_gate_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_right_gate = tl.load(
                w_right_gate_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_out_gate = tl.load(
                w_out_gate_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # compute partial dot products
            # using tl.dot: tl.dot(x_block, w_left) where x [R,K], w_left [K,H] -> [R,H]
            left_proj_acc += tl.dot(x_block, w_left)  # result to float32
            right_proj_acc += tl.dot(x_block, w_right)
            left_gate_acc += tl.dot(x_block, w_left_gate)
            right_gate_acc += tl.dot(x_block, w_right_gate)
            out_gate_acc += tl.dot(x_block, w_out_gate)

        # after dim loop, we have full dot results for this hidden tile
        # apply sigmoid to gates
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        left = left_proj_acc.to(tl.float16) * m_val * left_gate
        right = right_proj_acc.to(tl.float16) * m_val * right_gate
        # out_gate = out_gate_val (no mask)
        # store results
        dst = base[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate_val, mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – forward‑only implementation for AlphaFold‑3.

The operator works on a tensor X ∈ ℝ^{B×N×N×C} and produces Y ∈ ℝ^{B×N×N×C}.
We fuse the bulk of the work into three Triton kernels:

1️⃣  Row‑wise LayerNorm (FP32 → FP16).
2️⃣  A single kernel `proj_gate_kernel` that
    * reads the normalized tensor,
    * performs the five linear maps (left_proj, right_proj,
      left_gate, right_gate, out_gate) – i.e. a GEMM with
      weight matrices of shape (H, C) – using Tensor‑cores,
    * applies the optional pair mask,
    * evaluates the three sigmoid gates and writes the three
      tensors `left`, `right` and `out_gate` in the layout
      [B, H, N, N] (FP16).  This completely removes the large
      intermediate  (R, 5·H) projection buffer used by the reference.
3️⃣  Two later kernels from the original implementation:
    * `rowwise_stats_kernel` – computes mean & rsqrt‑variance of the
      contracted tensor `out`.
    * `final_output_kernel` – fuses the second LayerNorm, the out‑gate,
      and the final projection to produce the FP32 result.

All kernels use mixed precision (FP32 for statistics, FP16 for the
majority of the work) and are tuned for the H100 (SM‑90).  No gradients
are computed – only the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, C]   FP32 input
    y_ptr,   # [R, C]   FP16 output
    w_ptr,   # [C]     γ (FP32)
    b_ptr,   # [C]     β (FP32)
    rows,    # scalar  R
    dim,     # scalar  C
    eps,     # scalar  ε
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Projection + gating (fused, no intermediate large buffer)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,              # [R, C]               FP16 – already layer‑normed
    left_ptr,           # [B, H, N, N]         FP16 – result of left_proj ⋅ mask ⋅ left_gate
    right_ptr,          # [B, H, N, N]         FP16 – result of right_proj ⋅ mask ⋅ right_gate
    out_gate_ptr,       # [B, H, N, N]         FP16 – sigmoid(out_gate)
    mask_ptr,           # [R]                  FP16 (optional, broadcast over H)
    left_proj_w_ptr,    # [H, C]               FP16
    right_proj_w_ptr,   # [H, C]               FP16
    left_gate_w_ptr,    # [H, C]               FP16
    right_gate_w_ptr,   # [H, C]               FP16
    out_gate_w_ptr,     # [H, C]               FP16
    R, H, N, C,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,   # tile size over the channel dimension C
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start of hidden plane 0

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # iterate over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for the five linear maps (FP32 for accuracy)
        left_proj_acc   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_proj_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc    = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ---- loop over the channel dimension C ----
        for d_off in range(0, tl.cdiv(C, BLOCK_K)):
            offs_d = d_off * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
            mask_d = offs_d < C

            # X block – shape [BLOCK_R, BLOCK_K]
            x_block = tl.load(
                x_ptr + offs_r[:, None] * C + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # weight slices – we load them transposed so that they have
            # shape [BLOCK_K, BLOCK_H] (i.e. [C_tile, H_tile])
            w_left_proj = tl.load(
                left_proj_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_right_proj = tl.load(
                right_proj_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_left_gate = tl.load(
                left_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_right_gate = tl.load(
                right_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_out_gate = tl.load(
                out_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # matmul: (BLOCK_R, BLOCK_K) @ (BLOCK_K, BLOCK_H) → (BLOCK_R, BLOCK_H)
            left_proj_acc   += tl.dot(x_block, w_left_proj)
            right_proj_acc  += tl.dot(x_block, w_right_proj)
            left_gate_acc   += tl.dot(x_block, w_left_gate)
            right_gate_acc  += tl.dot(x_block, w_right_gate)
            out_gate_acc    += tl.dot(x_block, w_out_gate)

        # ----- apply sigmoid gates (in FP32 → cast back) -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # ----- combine with mask and write results -----
        left  = left_proj_acc.to(tl.float16) * m_val * left_gate
        right = right_proj_acc.to(tl.float16) * m_val * right_gate
        # out_gate is already the sigmoid result
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise statistics (mean & rsqrt‑variance) of contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M,K]
        gated = norm * vals_gate                                      # [M,K]

        # cast to FP16 for the dot (fp16·fp16 → fp32)
        gated_fp16 = gated.to(tl.float16)

        # weight tile: shape [K, BLOCK_N] (column‑major view of Wᵀ)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, BLOCK_N]

        # accumulate (M,K)·(K,N) → (M,N)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys (float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim" (int), "hidden_dim" (int),
                              "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pair‑wise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (no intermediate large buffer)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # cast weights to FP16 and make them contiguous
    left_proj_w   = w["left_proj.weight"].to(torch.float16).contiguous()
    right_proj_w  = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w   = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w  = w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = w["out_gate.weight"].to(torch.float16).contiguous()

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64          # rows per program (tunable)
    BLOCK_H = 64          # hidden‑dim tiles (tunable)
    BLOCK_K = 32          # channel‑dim tiles (tunable, C ≤ 384)
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    del x_norm, left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM (cuBLAS) → core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                       # FP16
    right_mat = right.view(B * H, N, N)                      # FP16
    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64          # H ≤ 128 → at most two tiles
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
