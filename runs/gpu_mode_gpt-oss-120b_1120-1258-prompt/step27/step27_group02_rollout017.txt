"""
TriMul (outgoing) – fast forward‑only Triton+PyTorch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the pairwise tensor (FP32) – done with
    torch.nn.functional.layer_norm (high‑performance cuDNN kernel).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS).  The result has shape (R,5·H) in FP16,
    where R = B·N·N.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * decodes the flat pairwise index (b,i,k) → (b,i,k)
    * splits the fused projection,
    * applies the optional mask,
    * computes sigmoid gates,
    * writes three tensors  `left`, `right`, `out_gate`
      in layout [B, H, N, N] (FP16).

4️⃣  Core einsum  “… i k d, … j k d → … i j d”
    is a batched matrix multiplication per hidden dimension:
    `out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`.
    Implemented with a single torch.bmm over the reshaped tensors.

5️⃣  Second LayerNorm (over the hidden dimension H) – performed with
    torch.nn.functional.layer_norm (FP32 statistics, FP16 data).

6️⃣  Multiply by the pre‑computed `out_gate`.

7️⃣  Final linear projection `to_out` (H → dim) – a single GEMM
    (torch.nn.functional.linear).  The result is cast to float32.

Only the split‑&‑gate step is written in Triton; everything else
leverages highly‑optimized PyTorch kernels, reducing kernel launch
overheads and achieving the target speed on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over H
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # start of plane h = 0 for this (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    # iterate over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # linear view: [left | right | left_gate | right_gate | out_gate]
        # proj layout: rows = R, cols = 5*H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        left_g_raw   = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        right_g_raw  = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        outg_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        # destination offset for the (b, h, i, k) element
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape               # x shape = [B, N, N, dim]
    R = B * N * N                       # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32) – use torch's optimized implementation
    # ------------------------------------------------------------------
    # Normalize across the last dimension `dim`
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)                     # [B, N, N, dim] FP32

    # Cast to FP16 for the subsequent GEMM
    x_norm_fp16 = x_norm.to(torch.float16).contiguous()
    del x_norm   # free FP32 version early

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # shape [5·H, dim]

    proj = F.linear(x_norm_fp16, fused_w)       # (R, 5·H) FP16
    del fused_w, x_norm_fp16                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask will be broadcasted over H dimension
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # number of rows processed per program
    BLOCK_H = 64   # hidden dimension block size
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr  # free projection & mask

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] for a single bmm per hidden plane
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension, out‑gate multiplication
    # ------------------------------------------------------------------
    # move hidden dim to the last axis for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()         # [B, N, N, H] FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16

    # LayerNorm over the hidden dim H (FP32 statistics, FP16 data)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    # Apply the output gate (element‑wise)
    out_norm = out_norm * out_gate_perm                     # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection to `dim` (H → dim)
    # ------------------------------------------------------------------
    # collapse batch & spatial dimensions: (B·N·N, H)
    out_flat = out_norm.view(-1, H)                         # [B·N·N, H] FP16

    # Linear projection: weight shape (dim, H)
    final_fp16 = F.linear(out_flat, w["to_out.weight"])    # [B·N·N, dim] FP16

    # reshape to final tensor and cast to FP32 (as required by the spec)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
