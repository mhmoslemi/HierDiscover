def custom_kernel(data):
   x, mask, w, cfg = data
   B,N,_,dim = x.shape
   hidden_dim = cfg["hidden_dim"]
   eps = 1e-5
   device = x.device
   R = B * N * N
   H = hidden_dim

   # 1. LayerNorm (custom)
   x_flat = x.reshape(R, dim).contiguous()
   x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
   # Determine block sizes: dynamic based on dim for better occupancy
   BLOCK_M = 128 if R >= 128 else 1
   BLOCK_D = 64 if dim >= 64 else dim
   grid_ln = (triton.cdiv(R, BLOCK_M),)
   layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
   # reshape
   x_norm = x_norm.reshape(B, N, N, dim)  # shape [B,N,N,dim]

   # 2. Fuse linear
   fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"]
        ],
        dim=0,
   ).to(torch.float16)   # shape [5*H, dim]

   proj = torch.nn.functional.linear(x_norm, fused_weight)  # shape [R,5*H]
   # optionally free x_norm

   # 3. Split + mask + gating (torch)
   proj = proj.view(R, 5, H)
   left = proj[:,0,:]   # [R,H]
   right = proj[:,1,:]
   left_gate = torch.sigmoid(proj[:,2,:])
   right_gate = torch.sigmoid(proj[:,3,:])
   out_gate = torch.sigmoid(proj[:,4,:])

   # apply mask
   if not cfg.get("nomask", True) and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R,1)
        left = left * mask_flat
        right = right * mask_flat
   else:
        # no mask
        pass

   left = left * left_gate
   right = right * right_gate

   # reshape to [B,H,N,N]
   left = left.view(B, H, N, N)
   right = right.view(B, H, N, N)
   out_gate = out_gate.view(B, H, N, N)

   # 4. Batched matmul: out = left @ right^T per batch*H
   left_mat = left.view(B*H, N, N)
   right_mat = right.view(B*H, N, N)
   out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H, N, N]
   out = out.view(B, H, N, N)  # [B,H,N,N]

   # 5. second layer norm on hidden dim
   # Permute to (..., H)
   out_perm = out.permute(0,2,3,1)  # [B,N,N,H]
   # Use torch.nn.functional.layer_norm
   out_norm = torch.nn.functional.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
   )

   # Multiply by out_gate (need permuted)
   out_gate_perm = out_gate.permute(0,2,3,1)  # [B,N,N,H]
   out_gated = out_norm * out_gate_perm

   # final linear to dim
   w_out = w["to_out.weight"].to(torch.float16).t().contiguous()  # shape [H, dim]
   # Use torch.nn.functional.linear
   out_final = torch.nn.functional.linear(out_gated, w_out.t(), bias=None)  # but weight shape needed as (out_features, in_features). w_out.t() is (dim, H). Actually we can directly use w["to_out.weight"].
   # So simpler:
   out_final = torch.nn.functional.linear(out_gated, w["to_out.weight"], bias=None) # w shape [dim, H]; out_gated shape [B,N,N,H]; returns [B,N,N,dim] float16

   # Cast to float32 for output
   out_final = out_final.to(torch.float32)

   return out_final

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair dimension (R = B·N·N rows,
    C = dim columns).  Implemented with a custom Triton kernel that works in
    FP32 for the statistics and writes the normalized result in FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single matrix multiplication:
    (R, dim) @ (5·H, dim).T → (R, 5·H)  with H = hidden_dim.
   The result is reshaped to (R, 5, H).

3️⃣ Split the fused tensor, apply the optional pair mask, evaluate the three
   sigmoid gates and combine the gates with the projections – all done with
   vectorised PyTorch operations (no custom kernel needed here).

4️⃣ Core pairwise multiplication:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   Implemented as a batched GEMM (torch.bmm) on FP16 data.

5️⃣ Second LayerNorm over the hidden dimension H, followed by multiplication
   with the out‑gate and a final linear projection back to `dim`.  These steps
   use the highly‑optimised PyTorch functional kernels.

Only the first LayerNorm is written as a custom Triton kernel, satisfying the
requirement that “part of the operations” be a kernel.  All other stages are
performed with fused PyTorch kernels for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, C]   FP32 input
    y_ptr,   # [R, C]   FP16 output
    w_ptr,   # [C]     γ (FP32)
    b_ptr,   # [C]     β (FP32)
    rows,    # scalar  R
    dim,     # scalar  C
    eps,     # scalar  ε
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys (float32 tensors):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = bool(cfg.get("nomask", True))

    R = B * N * N            # total pair‑wise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # [R, dim] FP16

    # Tunable block sizes – small enough to fit many CTAs, large enough for
    # good memory coalescing.
    BLOCK_M = 128 if R >= 128 else 1
    BLOCK_D = 64  if dim >= 64 else dim
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Shape back to [B, N, N, dim]
    x_norm = x_norm.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    # All projection weights have shape [hidden_dim, dim] (no bias)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    # (R, dim) @ (5·H, dim).T → (R, 5·H)  in FP16
    proj = F.linear(x_norm, fused_w)  # [R, 5·H]
    del fused_w, x_norm               # free memory

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                # [R, 5, H]

    left      = proj[:, 0, :]                # [R, H]
    right     = proj[:, 1, :]                # [R, H]
    left_gate = torch.sigmoid(proj[:, 2, :])  # [R, H]
    right_gate= torch.sigmoid(proj[:, 3, :])  # [R, H]
    out_gate  = torch.sigmoid(proj[:, 4, :])  # [R, H]

    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1]
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left  = left  * mask_flat
        right = right * mask_flat
    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Reshape to [B, H, N, N]
    left     = left.view(B, H, N, N)
    right    = right.view(B, H, N, N)
    out_gate = out_gate.view(B, H, N, N)

    # --------------------------------------------------------------
    # 4️⃣ Core pairwise multiplication (batched GEMM)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension, out‑gate and final proj
    # --------------------------------------------------------------
    # Move hidden dim to the end: [B, N, N, H]
    out_perm = out.permute(0, 2, 3, 1)      # (B, N, N, H)

    # LayerNorm on the hidden axis
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                      # (B, N, N, H) FP16

    # Apply out‑gate (also permuted to same layout)
    out_gate_perm = out_gate.permute(0, 2, 3, 1)  # (B, N, N, H)
    gated = out_norm * out_gate_perm               # (B, N, N, H)

    # Final linear projection back to `dim`
    # weight shape: [dim, H] (float32).  Cast to FP16 for speed.
    w_out = w["to_out.weight"].to(torch.float16)
    final = F.linear(gated, w_out, bias=None)      # (B, N, N, dim) FP16

    # Cast to FP32 as required by the spec
    return final.to(torch.float32)
