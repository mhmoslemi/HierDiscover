def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm over input dim
    x_norm = F.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm_fp16 = x_norm.to(torch.float16)

    # 2. Fused linear projection
    fused_w = torch.cat([w["left_proj.weight"], w["right_proj.weight"],
                         w["left_gate.weight"], w["right_gate.weight"],
                         w["out_gate.weight"]], dim=0).to(torch.float16)
    proj = F.linear(x_norm_fp16.view(R, dim), fused_w)   # (R,5*H) fp16

    # 3. Split, mask, gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_H = 64

        out_gate,

        N,

        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)).view(B, H, N, N)

    # 5. LayerNorm across hidden dim, multiply by out_gate, final projection
    # permute to [B,N,N,H] for layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_ln = F.layer_norm(out_perm, (H,), weight=w["to_out_norm.weight"], bias=w["to_out_norm.bias"], eps=eps)  # fp16

    final = F.linear(gated_flat, w_out_fp16)  # (R, dim) fp16
    final = final.view(B, N, N, dim).to(torch.float32)

"""
TriMul (outgoing) – fast forward‑only implementation for AlphaFold‑3.

Algorithm
---------
1️⃣  LayerNorm over the last dimension of the 4‑D pair tensor (B,N,N,C).
    Implemented with the highly‑optimized CUDA kernel used by
    ``torch.nn.functional.layer_norm`` (FP32 → FP16 conversion).

2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single FP16 GEMM:
        (B·N·N, C) × (5·H, C)ᵀ → (B·N·N, 5·H)

3️⃣  Split the fused result, apply the optional mask and the three
    sigmoid gates, and store ``left``, ``right`` and ``out_gate`` in the
    layout ``[B, H, N, N]``.  This step is performed by a small Triton
    kernel (`split_fuse_gate_kernel`).  No intermediate materialisation of
    the five separate projections is required.

4️⃣  Batched matrix multiplication (FP16) over the sequence dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
    Implemented with ``torch.bmm`` on a view of shape ``[B·H, N, N]``.
    This is the dominant O(N³) work and uses cuBLAS FP16 kernels.

5️⃣  LayerNorm across the hidden dimension H for every (i,j) position,
    followed by element‑wise multiplication with ``out_gate``.
    Both the LN and the gate are kept in FP16 for speed.

6️⃣  Final projection from hidden_dim → dim:
        gated[i,j] @ W_outᵀ  (where ``W_out`` is (dim, H))
    executed as a single FP16 GEMM with ``torch.nn.functional.linear``.
    The result is cast back to float32, matching the reference model.

Only the split‑&‑gate kernel is written in Triton; all heavy matrix‑
multiplications use cuBLAS which is superior on the target H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flattened pair index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcasted mask value (if present)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
        mask_val = m[:, None]                                      # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride to step through the hidden dimension (N·N per hidden slice)
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: (R, 5*H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # load the five concatenated projection slices
        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (stable in FP32, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        # out_gate is NOT masked (as in the reference implementation)
        outg  = out_gate

        # destination offsets for the three tensors
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of model weights (all FP32 tensors)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N               # flattened pair count
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (dim) – keep FP16 for the downstream GEMMs
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is a fused CUDA kernel (fast)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                       # [B, N, N, dim] fp16

    # flatten to (R, dim) for the fused projection
    x_flat = x_norm.reshape(R, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection to (5·H) channels
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # (5·H, dim)

    proj = F.linear(x_flat, fused_w)            # (R, 5·H) fp16
    del x_norm, x_flat, fused_w                 # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16
    else:
        has_mask = 0
        # dummy tensor – will never be accessed because HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM) : out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                   # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # (B·H, N, N)
    out = out.view(B, H, N, N)                           # (B, H, N, N)

    # tensors `left` and `right` are no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension H, multiply by out_gate, final Linear
    # ------------------------------------------------------------------
    # Move hidden dimension to the last axis for layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()      # (B, N, N, H) fp16

    out_ln = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                    # (B, N, N, H) fp16

    # gate (out_gate) has the same layout after permute
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # (B, N, N, H) fp16
    gated = out_ln * gate_perm                           # (B, N, N, H) fp16

    # flatten to (R, H) for the final projection
    gated_flat = gated.reshape(R, H)

    # final FP16 linear projection (hidden_dim → dim)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)    # (dim, H)
    final_fp16 = F.linear(gated_flat, w_out_fp16)        # (R, dim) fp16

    # reshape and cast back to float32 (the reference returns FP32)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
