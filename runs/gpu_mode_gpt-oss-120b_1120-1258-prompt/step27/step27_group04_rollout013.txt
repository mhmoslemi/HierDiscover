```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]    # broadcast over hidden
    else:
        mask_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

    stride_h = N * N  # offset between hidden planes

    base_offset = b * (H * stride_h) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load five vectors for this block of hidden
        # offset in proj: each row contiguously stores 5*H values
        proj_row = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_row, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_row + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_row + 2 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_row + 3 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_row + 4 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid: compute in fp32 for better precision
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst = base_offset[:, None] + offs_h[None, :] * stride_h   # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ first LayerNorm (torch)
    # flatten input to [R, dim]
    x_flat = input_tensor.reshape(R, dim).contiguous()
    # apply layer norm (fp32) and cast to fp16
    x_norm = F.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # 2️⃣ fused projection
    # assemble fused weight
    fused_w = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = F.linear(x_norm, fused_w)  # (R, 5*H) FP16
    del fused_w, x_norm

    # allocate buffers for left, right, out_gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # handle mask
    if (not nomask) and (mask_tensor is not None):
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor for mask_ptr (size 0)
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # launch split_fuse_gate kernel
    BLOCK_R = 64
    BLOCK_H = 64

        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # 3️⃣ core einsum via batched GEMM
    # reshape for batched matmul
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # compute out = left @ right^T
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back
    out = out.view(B, H, N, N)
    del left_mat, right_mat

    # 4️⃣ second LayerNorm across hidden dim
    # permute to [B,N,N,H] (contiguous)
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    # layer norm over last dimension H
    out_norm = F.layer_norm(out_perm, (H,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate (reshape)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    out_norm_gate = out_norm * out_gate_perm

    # flatten to [R, H]
    out_flat = out_norm_gate.view(R, H)

    # final linear projection (hidden -> dim)
    w_out = weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden]
    final_flat = F.linear(out_flat, w_out)  # [R, dim] FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)
    return final

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward only
-----------------------------------------------------

Algorithm
---------
1️⃣  First LayerNorm (FP32 → FP16) on the flattened pair tensor.
    Implemented with the highly‑optimised torch.nn.functional.layer_norm.

2️⃣  A *single* fused linear projection produces the 5·H streams
    (left, right, left‑gate, right‑gate, out‑gate) in FP16.

3️⃣  Triton kernel ``split_fuse_gate_kernel``  
    • decodes the flat pair index (b,i,k) → (b,h,i,k) layout,  
    • applies the optional mask (broadcast over H),  
    • evaluates the three sigmoid gates in FP32,  
    • writes three tensors ``left``, ``right`` and ``out_gate`` in layout
      ``[B, H, N, N]`` (FP16).  
    This replaces several element‑wise PyTorch kernels.

4️⃣  Core multiplicative update (einsum) is a batched GEMM:
    ``out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]``.
    Executed with ``torch.bmm`` on the tensor shape ``[B·H, N, N]``.

5️⃣  Second LayerNorm (over the hidden dimension H) is performed with
    ``torch.nn.functional.layer_norm`` (FP16 ↔ FP32 reduction).

6️⃣  Element‑wise multiplication with ``out_gate`` (after the second LN).

7️⃣  Final linear projection (hidden → dim) using ``torch.nn.functional.linear``.
    The input has shape ``[B·N·N, H]`` and the weight ``[dim, H]`` is stored
    in FP16, so the matmul is a tall‑skinny FP16 GEMM highly optimised on H100.

All heavy element‑wise work is fused into the Triton kernel (step 3);
the remaining steps rely on cuBLAS / cuDNN kernels that are already
optimised for the target hardware.  The implementation follows the
exact mathematics of the reference PyTorch module while drastically
reducing kernel launch overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split‑fused projection, optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,               # 1 → mask present, 0 → no mask
    BLOCK_R: tl.constexpr,               # rows per program
    BLOCK_H: tl.constexpr,               # hidden‑dim tiles per program
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1️⃣  Row index handling
    # ------------------------------------------------------------------
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # 2️⃣  Mask (optional) – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if HAS_MASK:
        # mask stored as float16 (0/1)
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Destination stride for the [B,H,N,N] layout
    # ------------------------------------------------------------------
    hidden_stride = N * N                                 # stride of one hidden plane
    # base address for hidden‑plane h = 0
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # ------------------------------------------------------------------
    # 4️⃣  Loop over hidden dimension in tiles of size BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)                                 # [R,H]
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ------------------------------------------------------------------
        # 5️⃣  Sigmoid in FP32 → back to FP16 (higher accuracy)
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ------------------------------------------------------------------
        # 6️⃣  Apply mask and gates
        # ------------------------------------------------------------------
        left  = left_raw  * mask_val * left_gate   # (left_proj * mask) * gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                            # no mask on out_gate

        # ------------------------------------------------------------------
        # 7️⃣  Store results in the [B, H, N, N] layout
        # ------------------------------------------------------------------
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, already on the same device as the input)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                         # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                             # -> FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (produces 5·H streams)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                             # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                                 # (R,5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension H)
    # ------------------------------------------------------------------
    # move hidden dimension to last axis -> [B, N, N, H]
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by out_gate (broadcasted to same layout)
    # ------------------------------------------------------------------
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_norm_gate = out_norm * out_gate_perm                     # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (hidden → dim)
    # ------------------------------------------------------------------
    out_flat = out_norm_gate.view(R, H)                         # [R, H]
    w_out = w["to_out.weight"].to(torch.float16)                # [dim, H]
    final_flat = F.linear(out_flat, w_out)                      # [R, dim] FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)     # back to FP32
    return final
