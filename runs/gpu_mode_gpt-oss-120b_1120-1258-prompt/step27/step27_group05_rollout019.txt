# layernorm via torch.nn.functional.layer_norm

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # LN
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # flatten for linear
    R = B * N * N
    x_flat = x_norm.view(R, dim)

    # fused weight
    fused_w = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"],
                         weights["left_gate.weight"], weights["right_gate.weight"],
                         weights["out_gate.weight"]], dim=0).to(torch.float16)  # [5*H, dim]

    proj = F.linear(x_flat, fused_w)  # [R, 5*H] fp16

    # allocate left,right,out_gate
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    # mask handling
    if not config.get("nomask", True) and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

        left,
        right,
        out_gate,

        BLOCK_H=BLOCK_H,
        num_warps=8,

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Outgoing TriMul forward (AlphaFold‑3) implemented with Triton.
    1. LayerNorm on the input pair tensor (FP32→FP16).
    2. Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
       into a single GEMM.
    3. Split the fused projection, apply the pair mask and the three sigmoids.
    4. Batched GEMM (left × rightᵀ) – the O(N³) core of TriMul.
    5. Fuse the second LayerNorm, the out‑gate and the final projection
       (H→dim) into a single Triton kernel, avoiding an extra kernel for
       mean/variance and reducing memory traffic.
    The function returns a float‑32 tensor of shape [B, N, N, dim].
    """
    # unpack
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = bool(cfg.get("nomask", True))
    R = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (torch implementation – fast, fused, FP32 statistics)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers into a single GEMM
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # flatten batch‑pair dimension -> (R, dim)
    x_flat = x_norm.view(R, dim)

    proj = F.linear(x_flat, fused_weight)    # (R, 5·H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoid gates
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # launch split‑&‑gate kernel
    BLOCK_R = 64
    BLOCK_H = 64
    split_grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[split_grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)            # (B·H, N, N)
    right_mat = right.view(B * H, N, N)          # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                     # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # to_out weight transposed (H, dim) – keep FP32 for numerical fidelity
    w_out_T = w["to_out.weight"].t().contiguous()   # (H, dim) FP32

    # launch fused kernel
    BLOCK_M = 64          # rows per program (≈ 64)
    BLOCK_K = 32          # hidden‑dim tile (covers H≤128)
    BLOCK_N = 64          # output‑dim tile

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(dim, BLOCK_N))
    fused_final_kernel[grid](
        out,
        out_gate,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )

    return final

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask if present
    m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)
    if HAS_MASK:
        m_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_row[:, None]                                      # [BLOCK_R,1]

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    acc_B = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # gate * gamma

        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        sum_val += tl.sum(vals_out,  axis=1)               # [M]
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)    # [M]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # compute the three scaled matrices
        #   A = out * gate * γ
        #   B = gate * γ
        #   C = gate * β
        A_mat = vals_out * vals_gate * gamma[None, :]        # [M,K]
        B_mat = vals_gate * gamma[None, :]                   # [M,K]
        C_mat = vals_gate * beta[None, :]                    # [M,K]

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]    # [K, TILE_N]

        acc_C += tl.dot(C_mat, w_tile)

    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the pair tensor (FP32 statistics, FP16 data).
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM (FP16).
3️⃣  Split the fused projection, apply the optional pair mask and three
    sigmoid gates – produces tensors `left`, `right` and `out_gate`
    with layout [B, H, N, N] (H = hidden_dim).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → shape [B, H, N, N] (FP16).
5️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection (H → dim) into a single Triton kernel.  The kernel
    *first* gathers the per‑row mean/variance,
    *then* evaluates the three required matrix‑vector products:
        A = Σₕ  out·out_gate·γ·W,
        B = Σₕ  out_gate·γ·W,
        C = Σₕ  out_gate·β·W,
    finally returns  rstd·A – μ·rstd·B + C  (float‑32).

All kernels run in mixed‑precision (FP16 for the bulk of the work,
FP32 for the statistics) on an H100 with Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_row[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside fused projection: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 5️⃣ Fuse 2nd LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_final_kernel(
    out_ptr,       # [B, H, N, N] FP16
    gate_ptr,      # [B, H, N, N] FP16
    ln_w_ptr,      # [H]           FP32 (to_out_norm.weight)
    ln_b_ptr,      # [H]           FP32 (to_out_norm.bias)
    w_out_T_ptr,   # [H, dim]      FP32 (to_out.weight.T)
    out_ptr_final, # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (flattened B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # statistics for the rowwise LayerNorm (µ, σ⁻¹)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # three accumulators for the three matrix‑vector products
    acc_A = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # out·gate·γ
    acc_B = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # gate·γ
    acc_C = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # gate·β

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ------------------------------------------------------------------
        #  load out & gate (convert to FP32 for the statistics)
        # ------------------------------------------------------------------
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M,K]

        # update mean / variance accumulators
        sum_val += tl.sum(vals_out,  axis=1)               # [M]
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)    # [M]

        # ------------------------------------------------------------------
        #  load LayerNorm parameters γ, β (FP32)
        # ------------------------------------------------------------------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # three scaled matrices – all in FP32
        A_mat = vals_out * vals_gate * gamma[None, :]    # out·gate·γ
        B_mat = vals_gate * gamma[None, :]               # gate·γ
        C_mat = vals_gate * beta[None, :]                # gate·β

        # load a tile of the final linear weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, TILE_N] FP32

        # accumulate three GEMMs (each yields an [M,N] tile)
        acc_A += tl.dot(A_mat, w_tile)
        acc_B += tl.dot(B_mat, w_tile)
        acc_C += tl.dot(C_mat, w_tile)

    # ----------------------------------------------------------------------
    #  finalize LayerNorm statistics and combine the three terms
    # ----------------------------------------------------------------------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    out_tile = rstd[:, None] * acc_A - mu[:, None] * rstd[:, None] * acc_B + acc_C

    # ----------------------------------------------------------------------
    #  store the final result (row‑major [B, N, N, dim])
    # ----------------------------------------------------------------------
    dst = out_ptr_final + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst,
             out_tile,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
#  Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (torch implementation – fast, fused)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    x_flat = x_norm.reshape(R, dim)    # [R, dim]
    proj = F.linear(x_flat, fused_w)   # (R, 5·H) FP16
    del x_norm, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ 2nd LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    # to_out weight transposed (H, dim) – keep FP32 for numerical fidelity
    w_out_T = w["to_out.weight"].t().contiguous()          # [H, dim] FP32

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M = 64         # rows per program
    BLOCK_K = 32         # hidden‑dim tile (covers H ≤ 128)
    BLOCK_N = 64         # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )
    fused_final_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
