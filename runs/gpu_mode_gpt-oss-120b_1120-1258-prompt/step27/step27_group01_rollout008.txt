def custom_kernel(data):
    x, mask, w, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5

    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm via torch.nn.functional
    x_flat = x.reshape(R, dim)
    x_norm = torch.nn.functional.layer_norm(
        x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps
    ).to(torch.float16)

    # 2. Fused linear (5 * H)
    fused_weight = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"],
    ], dim=0).to(torch.float16)   # shape (5H, dim)
    proj = torch.nn.functional.linear(x_norm, fused_weight)  # (R,5H) FP16

    # 3. split + gating
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=x.device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Batched GEMM left * right^T
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) FP16
    out = out.view(B, H, N, N)

    # 5. Prepare for final linear
    # Permute to (B,N,N,H) and flatten
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_reshape = out_perm.view(R, H)   # FP16
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    gate_reshape = gate_perm.view(R, H) # FP16

    # 6. Compute per‑row mean/var (float32)
    out_fp32 = out_reshape.to(torch.float32)
    mu = out_fp32.mean(dim=1, keepdim=True)
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)

    # 7. Apply LayerNorm, gate and final projection (all in FP32 then cast)
    gamma = w["to_out_norm.weight"].to(torch.float32)  # (H,)
    beta = w["to_out_norm.bias"].to(torch.float32)    # (H,)
    gate_fp32 = gate_reshape.to(torch.float32)

    norm = (out_fp32 - mu) * rstd
    norm = norm * gamma + beta
    term = norm * gate_fp32          # (R, H)  FP32

    term_fp16 = term.to(torch.float16)

    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # (H, dim)
    final_fp16 = torch.matmul(term_fp16, w_out_T)   # (R, dim) FP16

    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
   torch.nn.functional.layer_norm (highly‑optimised on H100).

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into one GEMM producing a tensor of shape (R, 5·H) where
   R = B·N·N and H = hidden_dim.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * Splits the fused projection into `left`, `right` and `out_gate`.
   * Applies the optional mask, computes the three sigmoid gates,
     and writes the three tensors in layout [B, H, N, N] (FP16).

4️⃣ Batched GEMM (cuBLAS) over the hidden dimension:
   out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k].

5️⃣ Compute per‑row mean / variance of `out` with a single
   torch reduction (FP32 → FP16).

6️⃣ Apply the second LayerNorm, multiply by `out_gate`,
   and perform the final projection with a cuBLAS GEMM:
   term = ((out – μ)·rstd·γ + β)·gate   (shape R×H)
   output = term @ W_outᵀ               (shape R×dim)

All intermediate work is in FP16, statistics in FP32, and the final
output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑row mask (applied to left/right only)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                                 # distance between successive hidden slices

    # Base offset for (b,i,k) in the flattened [B,H,N,N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Iterate over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather the five fused slices
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,      mask=mask_r[:, None] & mask_h[None, :])

    # 7️⃣ Second LayerNorm, out_gate, and final projection
    # --------------------------------------------------------------
    gamma = w["to_out_norm.weight"].to(torch.float32)   # (H,)
    beta  = w["to_out_norm.bias"].to(torch.float32)     # (H,)
    gate_fp32 = gate_reshape.to(torch.float32)          # (R, H)

    # LayerNorm + gate
    norm = (out_fp32 - mu) * rstd                         # (R, H)
    norm = norm * gamma + beta                           # (R, H)
    term = norm * gate_fp32                               # (R, H)

    term_fp16 = term.to(torch.float16)                    # back to FP16 for the GEMM

    # Final linear projection (R, H) @ (H, dim) → (R, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    final_fp16 = torch.matmul(term_fp16, w_out_T)                     # (R, dim) FP16

    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # (B, N, N, dim) FP32

    # --------------------------------------------------------------
    return final
```
