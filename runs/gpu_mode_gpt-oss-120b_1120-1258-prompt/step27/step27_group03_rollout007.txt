    gate_ptr, # [B,H,N,N] FP16
    out_norm_gate_ptr, # [R, H] FP16 output flattened (R= B*N*N)
    ln_w_ptr, # [H] FP32
    ln_b_ptr, # [H] FP32

Pseudo:

pid = tl.program_id(0)
row_start = pid * BLOCK_R
offs_r = row_start + tl.arange(0, BLOCK_R)
mask_r = offs_r < R

# decode row indices:
nn = N * N
b = offs_r // nn
rem = offs_r - b * nn
i = rem // N
j = rem - i * N

# compute base offsets for out and gate:
hidden_stride = N * N
out_base = b * (H * hidden_stride) + i * N + j
gate_base = out_base   # same layout

# compute row_idx for output storage: row_idx = b * N * N + i * N + j
row_idx = b * nn + i * N + j

# accumulate sum and sumsq across h for each row:
# We'll need mu and var for normalization; compute in first loop

sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

for h_off in range(0, H, BLOCK_H):
    offs_h = h_off + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H
    # load out values
    ptr_out = out_ptr + out_base[:, None] + offs_h[None, :] * hidden_stride
    vals_out = tl.load(ptr_out, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

    sum_val += tl.sum(vals_out, axis=1)
    sum_sq += tl.sum(vals_out * vals_out, axis=1)

mean = sum_val / H
var = sum_sq / H - mean * mean
rstd = tl.rsqrt(var + eps)

# second pass: apply norm and gate and store result
for h_off in range(0, H, BLOCK_H):
    offs_h = h_off + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # load out and gate
    ptr_out = out_ptr + out_base[:, None] + offs_h[None, :] * hidden_stride
    vals_out = tl.load(ptr_out, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
    
    ptr_gate = gate_ptr + gate_base[:, None] + offs_h[None, :] * hidden_stride
    vals_gate = tl.load(ptr_gate,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

    # load gamma & beta
    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
    beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

    # normalized
    diff = (vals_out - mean[:, None]) * rstd[:, None]  # [BLOCK_R, BLOCK_H]
    norm = diff * gamma[None, :] + beta[None, :]       # [BLOCK_R, BLOCK_H]

    # multiply gate
    out_val = norm * vals_gate  # FP32

    # store as FP16 in output buffer
    dst_ptr = out_norm_gate_ptr + (row_idx[:, None] * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

    tl.store(dst_ptr, out_val.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3). The forward pass consists of:

1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ One fused Linear that produces the five projections (left, right,
   left_gate, right_gate, out_gate) in a single GEMM.
3️⃣ A Triton kernel that splits the fused projection, applies the optional
   mask, the sigmoid gates and writes the three tensors in the layout
   [B, H, N, N] (FP16).
4️⃣ Batched FP16 GEMM (cuBLAS) computes the core einsum
   out[b, h, i, j] = Σₖ left[b, h, i, k] • right[b, h, j, k].
5️⃣ A Triton kernel that, for each (b,i,j), computes the mean / variance of
   `out` over the hidden dimension H, applies the second LayerNorm, multiplies
   by `out_gate` and writes the result into a flat buffer
   `[R, H]` where `R = B·N·N` (the hidden dimension is now the innermost axis).
6️⃣ A large FP16‑FP32 GEMM (cuBLAS) projects the normalized gated tensor from
   hidden → model dimension (`to_out`), producing the final output
   `[B, N, N, dim]` in float32.

All kernels target Triton 3.3.1 on an H100, use mixed‑precision arithmetic
where appropriate, and avoid any gradient computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim] FP32 input
    y_ptr,          # [R, dim] FP16 output
    w_ptr,          # [dim]   LN weight (γ) FP32
    b_ptr,          # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: accumulate sum & sumsq ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcast over hidden dim
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride in the hidden dimension (distance between consecutive h)
    hidden_stride = N * N
    # Base offset for the (b, i, k) position with h = 0
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection (R, 5·H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numeric stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offset for the current hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Normalize with second LayerNorm, apply out_gate, write flat buffer
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16 (raw contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16 (out_gate)
    out_norm_gate_ptr,# [R, H] FP16  (flattened, H is innermost)
    ln_w_ptr,         # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32   (to_out_norm.bias)
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                 # distance between consecutive hidden slices
    # Base pointers for out & gate, h = 0
    out_base = b * (H * hidden_stride) + i * N + j
    gate_base = out_base

    # Compute row index in the flat [R, H] buffer
    row_idx = b * nn + i * N + j          # shape [BLOCK_R]

    # ---- First pass: compute mean & variance over H ----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + out_base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)         # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- Second pass: normalize, apply gate, write into flat buffer ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Load raw out and gate values
        ptr_out = out_ptr + out_base[:, None] + offs_h[None, :] * hidden_stride
        vals_out = tl.load(ptr_out,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        ptr_gate = gate_ptr + gate_base[:, None] + offs_h[None, :] * hidden_stride
        vals_gate = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # Load LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (x - μ) / σ → *γ + β
        diff = (vals_out - mean[:, None]) * rstd[:, None]          # [BLOCK_R, BLOCK_H]
        norm = diff * gamma[None, :] + beta[None, :]               # [BLOCK_R, BLOCK_H]

        # Multiply by out_gate
        gated = norm * vals_gate                                   # [BLOCK_R, BLOCK_H]

        # Destination offset inside flat [R, H] buffer (H is innermost)
        dst = out_norm_gate_ptr + row_idx[:, None] * H + offs_h[None, :]

        tl.store(dst,
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point: custom Triton‑accelerated TriMul (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing FP32 weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N               # number of (b,i,j) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # [R, dim] FP16

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (produces 5·H channels)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask_tensor is not None):
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right are [B, H, N, N] → view as [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LN over H, multiply by out_gate, write flat buffer
    # --------------------------------------------------------------
    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)  # flat buffer

    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_norm = (triton.cdiv(R, BLOCK_R_OUT),)
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ, FP32
        weights["to_out_norm.bias"],     # β, FP32
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    # we no longer need raw `out` or `out_gate`
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection (large GEMM, cuBLAS) → float32 output
    # --------------------------------------------------------------
    # weight for the final linear layer (dim, H) in FP16
    w_out_fp16 = weights["to_out.weight"].to(torch.float16)   # [dim, H]

    # torch.nn.functional.linear does X @ W.T + bias
    # bias is omitted (original module has bias=False)
    final_fp16 = torch.nn.functional.linear(out_norm_gate, w_out_fp16)   # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)             # [B, N, N, dim]

    return final
