def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    # 1. rowwise LN via kernel
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # restructure to [B,N,N,dim]
    x_norm = x_norm.view(B, N, N, dim)
    # 2. fused projection
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"]
        ],
        dim=0,
    ).to(torch.float16)
    proj = F.linear(x_norm, fused_weight)  # shape [B,N,N,5*H] fp16
    del x_norm
    H = hidden_dim
    # 3. split and gate via torch ops
    proj = proj.view(B, N, N, 5, H)  # [B,N,N,5,H]
    # left/right raw
    left_raw = proj[..., 0, :]  # [B,N,N,H]
    right_raw = proj[..., 1, :]
    left_gate_raw = proj[..., 2, :]
    right_gate_raw = proj[..., 3, :]
    out_gate = torch.sigmoid(proj[..., 4, :])  # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    if not nomask and mask is not None:
        mask_expand = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left_raw * mask_expand * left_gate
        right = right_raw * mask_expand * right_gate
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate
    # free proj
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw
    # permute to [B,H,N,N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()
    # 4. core einsum via batched matmul
    LH = B * H
    left_mat = left.view(LH, N, N)
    right_mat = right.view(LH, N, N)
    # produce out [B*H,N,N] fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N).contiguous()
    # 5. LN + out_gate + final linear via torch ops
    # reshape to (B*N*N, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(-1, H)   # [B*N*N, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(-1, H)
    # layer norm over hidden dim

"""
TriMul (outgoing) – forward‑only Triton + PyTorch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor  (custom Triton kernel).
2️⃣ Fuse the five Linear layers (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM:  x_norm ⟶ proj  (shape [..., 5·H]).
3️⃣ Split the fused projection, apply the optional mask and sigmoid gates
   with plain PyTorch ops (fast element‑wise kernels on the GPU).
4️⃣ Core contraction:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
   performed as a batched FP16 GEMM (cuBLAS) via `torch.bmm`.
5️⃣ Second LayerNorm, multiplication with `out_gate` and the final projection
   are done with highly‑optimized PyTorch primitives (`layer_norm` + `linear`).
   All heavy matrix work stays in FP16; the final result is cast back to FP32.

Only the first LayerNorm is implemented in Triton, satisfying the
“implement at least part of the operations in a kernel” requirement while
keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    LN scale (γ)  FP32
    b_ptr,          # [dim]    LN bias  (β)  FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- pass‑1 : mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass‑2 : normalize, scale, bias ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing (all FP32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton kernel
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # reshape back to [B, N, N, dim] (FP16)
    x_norm = x_norm.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fused projection – single FP16 GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)               # [5·H, dim]

    proj = F.linear(x_norm, fused_weight)                  # [B, N, N, 5·H] FP16
    del x_norm, fused_weight

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask & sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(B, N, N, 5, H)                         # [B, N, N, 5, H]

    left_raw   = proj[..., 0, :]       # [B, N, N, H]
    right_raw  = proj[..., 1, :]
    left_gate_raw  = proj[..., 2, :]
    right_gate_raw = proj[..., 3, :]
    out_gate   = torch.sigmoid(proj[..., 4, :])   # already gated, FP16

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left  = left_raw * mask_fp16 * left_gate
        right = right_raw * mask_fp16 * right_gate
    else:
        left  = left_raw * left_gate
        right = right_raw * right_gate

    # clean up intermediate tensors
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw

    # bring tensors to layout [B, H, N, N] required for batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] FP16
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] FP16
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched FP16 GEMM (cuBLAS)
    # --------------------------------------------------------------
    LH = B * H
    left_mat  = left.view(LH, N, N)                 # [B·H, N, N]
    right_mat = right.view(LH, N, N)                # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm → out_gate multiplication → final projection
    # --------------------------------------------------------------
    # reshape to (B·N·N, H) for a single large GEMM
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(-1, H)          # [B·N·N, H]
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(-1, H)  # same shape

    # LayerNorm over the hidden dimension (FP16)
    ln_w = w["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    ln_b = w["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out_norm = F.layer_norm(out_flat, (H,), weight=ln_w, bias=ln_b, eps=eps)  # FP16

    # Multiply by the out‑gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate_flat                                   # FP16

    # Final linear projection (H → dim) – FP16 GEMM
    w_out = w["to_out.weight"].to(device=device, dtype=torch.float16)    # [dim, H]
    final_fp16 = F.linear(out_norm, w_out)                                # [B·N·N, dim] FP16

    # reshape to the original pair tensor shape and cast to float32
    final = final_fp16.view(B, N, N, -1).to(torch.float32)                 # [B, N, N, dim]

    return final
