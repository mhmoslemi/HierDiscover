    pid = program_id(0)
    row_start = pid * BLOCK_M

    offs_m = row_start + arange(0, BLOCK_M); mask_m = offs_m < total_rows

    # decode b,i,j...

    # first pass: compute mu, sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, H, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)   # shape [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None,:] * stride_hidden

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu*mu
    rstd = tl.rsqrt(var + eps)

    # second pass: accumulate final projection
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_gate = gate_ptr + base[:, None] + offs_k[None,:] * stride_hidden

"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the flattened pair tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS).  The result has shape (R, 5·H) with R = B·N·N.
3️⃣  Split the fused projection, apply the optional mask and sigmoid gates
    in a Triton kernel → tensors `left`, `right`, `out_gate` in layout
    [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]   → [B, H, N, N] (FP16).
5️⃣  Fuse the second LayerNorm (over H), the `out_gate` multiplication and
    the final linear projection (`to_out`) into a single Triton kernel.
    This removes the intermediate “row‑wise stats” kernel and saves one launch.

All kernels target Triton 3.3.1 on an NVIDIA H100, use mixed precision (FP16
computation, FP32 accumulation) and produce the final result in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] FP32 input
    y_ptr,            # [R, D] FP16 output
    w_ptr,            # [D]   γ (FP32)
    b_ptr,            # [D]   β (FP32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- pass 1 : mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2 : write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k   (i = first axis, k = second)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask broadcasting (optional)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mval[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between hidden slices

    # base offset for (b,i,k) with hidden index = 0  (layout [B,H,N,N])
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets into the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate, & final projection (to_out)
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N] FP16  (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16  (out_gate)
    ln_w_ptr,         # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32 (final tensor)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)                     # rows (R)
    pid_n = tl.program_id(1)                     # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + j   # hidden‑slice start for h=0

    # -----------------------------------------------------------------
    # 1) Row‑wise mean & rsqrt‑variance over hidden dimension H
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2) LayerNorm → gate → final projection (H × dim → dim)
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < H

        # load raw out & gate values (FP16)
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * stride_hidden

        out_raw_fp16  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        gate_raw_fp16 = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # convert to FP32 for LN & gating
        out_fp32  = out_raw_fp16.to(tl.float32)
        gate_fp32 = gate_raw_fp16.to(tl.float32)

        # LayerNorm parameters (γ,β) for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # Normalize + LN
        norm = (out_fp32 - mu[:, None]) * rstd[:, None]           # (out-μ)*rstd
        norm = norm * gamma[None, :] + beta[None, :]              # apply γ,β

        # Apply out_gate
        term = norm * gate_fp32                                   # [M,K] FP32

        # Cast to FP16 for the dot with weight (FP16×FP16 → FP32)
        term_fp16 = term.to(tl.float16)

        # Load weight tile w_out_T[h, d]  (shape [K, BLOCK_N])
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # FP16

        # Accumulate (M,K)·(K,N) → (M,N)   (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # -----------------------------------------------------------------
    # Store final tensor (flattened view of [B, N, N, dim])
    # -----------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs & config
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right layout: [B, H, N, N] → view as (B·H, N, N)
    left_mat  = left.view(B * H, N, N)    # FP16
    right_mat = right.view(B * H, N, N)   # FP16
    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Fuse LN (to_out_norm), out_gate and final projection
    # --------------------------------------------------------------
    # to_out.weight: [dim, hidden_dim] → transpose -> [hidden_dim, dim]
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows (R) per program
    BLOCK_K_OUT = 32   # hidden‑dim tile
    BLOCK_N_OUT = 128  # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ (FP32, shape [H])
        weights["to_out_norm.bias"],     # β (FP32, shape [H])
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # --------------------------------------------------------------
    return final
