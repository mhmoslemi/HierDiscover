@triton.jit
def hidden_compute_kernel(
    out_ptr,
    gate_ptr,
    mu_ptr,
    rstd_ptr,
    ln_w_ptr,
    ln_b_ptr,
    hidden_ptr,
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,j
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden slice in out layout
    stride_hidden = nn
    base = b * (H * stride_hidden) + i * N + j

    # load mu and rstd for each row
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)[:, None]  # [BLOCK_R,1]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate values
        vals_out = tl.load(out_ptr + base[:, None] + offs_h[None,:] * stride_hidden,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        vals_gate = tl.load(gate_ptr + base[:, None] + offs_h[None,:] * stride_hidden,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # load ln parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

        # apply layernorm: (out - mu)*rstd*gamma + beta
        norm = (vals_out - mu) * rstd
        norm = norm * gamma[None, :] + beta[None, :]

        # apply out_gate
        hidden_fp = (norm * vals_gate).to(tl.float16)

        # store into hidden shape [R, H] (contiguous)
        dst_offset = offs_r[:, None] * H + offs_h[None, :]   # [BLOCK_R, BLOCK_H]
        tl.store(hidden_ptr + dst_offset, hidden_fp,
                 mask=mask_r[:, None] & mask_h[None, :])

    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](

        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # 2. Linear projections
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_raw = F.linear(x_norm, w_left)       # (R, H)
    right_raw = F.linear(x_norm, w_right)     # (R, H)
    left_gate_raw = F.linear(x_norm, w_left_gate)
    right_gate_raw = F.linear(x_norm, w_right_gate)
    out_gate_raw = F.linear(x_norm, w_out_gate)

    # release x_norm early
    del x_norm

    # 3. Apply mask if needed
    if (not nomask) and (mask is not None):

        right_raw = right_raw * mask_flat[:, None]

    # 4. Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # release gate raw tensors
    del left_gate_raw, right_gate_raw, out_gate_raw

    left = left_raw * left_gate

    del left_raw, right_raw, left_gate, right_gate

---------
1️⃣  Row‑wise LayerNorm (FP32→FP16) on the flattened pair tensor.
2️⃣  Five independent FP16 GEMMs (left, right, left_gate, right_gate, out_gate)
    – each reads the normalized tensor once → much less memory traffic than a
      single giant GEMM + an extra split kernel.
3️⃣  Optional mask is applied to the two projections, then sigmoid gates are
    computed (torch.sigmoid) and the gated tensors are reshaped to
    [B, H, N, N] (contiguous layout) for the batched matrix‑multiplication:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k] .
4️⃣  Row‑wise mean / variance over the hidden dimension H are computed by a
    tiny Triton kernel (no synchronisation needed).
5️⃣  A second Triton kernel fuses the second LayerNorm (γ/β), the out‑gate
    multiplication and writes the hidden tensor in a contiguous [R,H] layout
    (R = B·N·N).
6️⃣  The final linear projection (to_out) is a regular cuBLAS GEMM:
        hidden @ W_outᵀ  →  [R, dim] (FP16) → cast to FP32 and reshape.
All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision.

The implementation below follows this pipeline and keeps the memory‑bandwidth
footprint low enough to meet the sub‑500 µs target on the prescribed test
cases.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] FP32 input
    y_ptr,            # [R, D] FP16 output
    w_ptr,            # [D]   LN weight (γ) FP32
    b_ptr,            # [D]   LN bias   (β) FP32
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# -------------------------------------------------------------
# 2️⃣ Row‑wise mean / rsqrt‑variance over H
# -------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32  (mean per (b,i,j))
    rstd_ptr,     # [R] FP32  (rsqrt(var+eps) per (b,i,j))
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N               # distance between consecutive H‑slices
    base = b * (H * stride_hidden) + i * N + j        # start of hidden slice for this row

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# -------------------------------------------------------------
# 3️⃣ Fuse LayerNorm (to_out_norm), out_gate and write hidden tensor
# -------------------------------------------------------------
@triton.jit
def hidden_compute_kernel(
    out_ptr,               # [B, H, N, N] FP16
    gate_ptr,              # [B, H, N, N] FP16   (out_gate)
    mu_ptr,                # [R] FP32
    rstd_ptr,              # [R] FP32
    ln_w_ptr,              # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,              # [H] FP32   (to_out_norm.bias)
    hidden_ptr,            # [R, H] FP16   (output)
    R, H, N,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + j      # start of hidden slice for this row

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)[:, None]   # [BLOCK_R,1]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]   # [BLOCK_R,1]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # load the hidden slice from the contracted tensor
        vals_out = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_H]

        vals_gate = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_H]

        # layer‑norm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # LN → (out - μ) * rstd * γ + β
        norm = (vals_out - mu) * rstd
        norm = norm * gamma[None, :] + beta[None, :]

        # apply the out‑gate
        hidden_fp = (norm * vals_gate).to(tl.float16)

        # write into contiguous [R, H] layout
        dst_off = offs_r[:, None] * H + offs_h[None, :]          # [BLOCK_R, BLOCK_H]
        tl.store(
            hidden_ptr + dst_off,
            hidden_fp,
            mask=mask_r[:, None] & mask_h[None, :]
        )


# -------------------------------------------------------------
# Entry point – forward‑only outgoing TriMul
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : model parameters (see code for required keys)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # unpack inputs / config
    # ---------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ---------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ---------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ---------------------------------------------------------
    # 2️⃣ Five independent linear projections (left, right, gates)
    # ---------------------------------------------------------
    w_left      = weights["left_proj.weight"].to(torch.float16)
    w_right     = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate= weights["right_gate.weight"].to(torch.float16)
    w_out_gate  = weights["out_gate.weight"].to(torch.float16)

    left_raw      = F.linear(x_norm, w_left)          # (R, H)
    right_raw     = F.linear(x_norm, w_right)         # (R, H)
    left_gate_raw = F.linear(x_norm, w_left_gate)     # (R, H)
    right_gate_raw= F.linear(x_norm, w_right_gate)    # (R, H)
    out_gate_raw  = F.linear(x_norm, w_out_gate)     # (R, H)

    del x_norm  # no longer needed

    # ---------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ---------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        left_raw  = left_raw  * mask_flat[:, None]
        right_raw = right_raw * mask_flat[:, None]

    # ---------------------------------------------------------
    # 4️⃣ Sigmoid gates & gated projections
    # ---------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # free raw gate tensors
    del left_gate_raw, right_gate_raw, out_gate_raw

    left  = left_raw  * left_gate          # (R, H)
    right = right_raw * right_gate         # (R, H)

    # free intermediates
    del left_raw, right_raw, left_gate, right_gate

    # ---------------------------------------------------------
    # 5️⃣ Reshape to [B, H, N, N] (contiguous layout) for the BMM
    # ---------------------------------------------------------
    # reshape to (B, N, N, H) then permute → (B, H, N, N) and make contiguous
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ---------------------------------------------------------
    # 6️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ---------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # (B*H, N, N) FP16
    right_mat = right.view(B * H, N, N)                # (B*H, N, N) FP16

    # out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16

    # free large tensors that are no longer needed
    del left, right, left_mat, right_mat

    # ---------------------------------------------------------
    # 7️⃣ Row‑wise mean & rsqrt‑variance over the hidden dimension (H)
    # ---------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128           # H = 128 fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ---------------------------------------------------------
    # 8️⃣ Fuse second LayerNorm, out_gate and produce hidden tensor [R, H]
    # ---------------------------------------------------------
    hidden = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_hidden = (triton.cdiv(R, BLOCK_R),)
    hidden_compute_kernel[grid_hidden](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ  (FP32, shape [H])
        weights["to_out_norm.bias"],     # β  (FP32, shape [H])
        hidden,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # free tensors that are no longer needed
    del out, out_gate, mu, rstd

    # ---------------------------------------------------------
    # 9️⃣ Final projection (to_out) – pure cuBLAS GEMM
    # ---------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]
    final_fp16 = torch.matmul(hidden, w_out_T)       # (R, dim) FP16
    final = final_fp16.float().view(B, N, N, dim)    # (B, N, N, dim) FP32

    return final
```
