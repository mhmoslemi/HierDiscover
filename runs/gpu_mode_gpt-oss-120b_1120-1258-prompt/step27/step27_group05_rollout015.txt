"""
High‑performance Triton implementation of the outgoing TriMul operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input pair tensor (shape [B,N,N,dim]) using the
   built‑in ``torch.nn.functional.layer_norm`` (float32 → float16 conversion).
2. Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
   into a single matmul:  X @ W_fused^T  →  [R,5·H] (half).
3. Triton kernel ``split_fuse_gate_kernel`` splits the fused projection,
   applies the optional pair mask, computes the three sigmoid gates
   and writes the three tensors ``left``, ``right`` and ``out_gate``
   in layout [B,H,N,N].
4. Batched GEMM (cuBLAS) computes the core einsum:
      out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   by a single ``torch.bmm`` on the flattened batch dimension.
5. Second layer‑norm over the hidden dimension (H) is performed with
   ``torch.nn.functional.layer_norm`` on a view where H is the last axis.
6. The normalized tensor is multiplied element‑wise by ``out_gate``,
   then a final linear projection to the original ``dim`` is applied
   with another fused ``F.linear`` call.
All heavy compute stays in FP16; the small statistics are accumulated
in FP32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1 – split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection (input)
    left_ptr,      # [B, H, N, N]  FP16 output
    right_ptr,     # [B, H, N, N]  FP16 output
    out_gate_ptr,  # [B, H, N, N]  FP16 output
    mask_ptr,      # [R] (optional) FP16 mask (broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [BLOCK_R]
        # Expand to hidden dimension
        m_val = m_val[:, None]                                             # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # Underlying stride for one hidden slice in the output tensors
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # Process hidden dimension in tiles of size BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection: [R,5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]         # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16 (more accurate)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate   # [BLOCK_R, BLOCK_H]
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination pointer in the [B,H,N,N] layout
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: processed tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N           # flattened pair dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    # Reshape to [R, dim] and apply torch‑implemented LayerNorm.
    x_flat = x.reshape(R, dim).contiguous()
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    # Cast to half for the rest of the pipeline.
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (five linear maps in one GEMM)
    # ------------------------------------------------------------------
    # Build the fused weight matrix: [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5·H, dim)

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) half
    # ``proj`` is the only large temporary we keep in half‑precision.

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, mask, and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # Convert mask to half and flatten to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # Dummy 0‑size tensor – Triton will never read from it.
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Choose tile sizes that give high occupancy on H100.
    BLOCK_R = 64   # rows per program (tuned for the R dimension)
    BLOCK_H = 64   # hidden tiles per program

    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Release the large intermediate tensors early to ease memory pressure.
    del proj, x_norm, x_norm_fp32, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # Shape: (B·H, N, N)  (row‑major layout, half‑precision)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out = left @ rightᵀ   →   shape (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)       # [B, H, N, N]   half

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    # Move the hidden axis to the innermost position so that torch.layer_norm
    # can reduce over it without extra transposes inside the kernel.
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )   # still half‑precision

    # Bring the tensor back to [B, H, N, N] layout for the gate multiplication.
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Apply the multiplicative out‑gate (element‑wise).
    out_gated = out_norm * out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection to the original channel dimension (dim)
    # ------------------------------------------------------------------
    # Flatten the spatial dimensions so that we can use a single matmul:
    #   (B·N·N, H)  ×  (H, dim)  →  (B·N·N, dim)
    out_gated_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(R, H)

    w_out = w["to_out.weight"].to(torch.float16)      # (dim, H)
    final_flat = F.linear(out_gated_flat, w_out)     # (R, dim) half

    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final

"""
High‑performance Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3 / AlphaFold‑MPNN style).  The routine follows the exact mathematical
steps of the reference PyTorch module but keeps the bulk of the work in FP16
and fuses the five linear maps into a single GEMM.  Only one custom Triton
kernel is required (splitting the fused projection, applying the optional mask,
and computing the three sigmoid gates); all other heavy‑weight ops (batched
GEMM, the two LayerNorms and the final projection) are delegated to the highly
optimised PyTorch / cuBLAS kernels.

Algorithm
---------
1. **First LayerNorm** – normalises the pairwise token tensor over the last
   dimension (`dim`).  Implemented with `torch.nn.functional.layer_norm` (FP32
   statistics, FP16 output) and reshaped to `[R, dim]` where `R = B·N·N`.

2. **Fused projection** – concatenates the five weight matrices
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) into a
   single matrix `[5·H, dim]` and performs one half‑precision GEMM:
   `proj = X_norm @ W_fused.T` → shape `[R, 5·H]`.

3. **Split + mask + sigmoid gates** – a Triton kernel (`split_fuse_gate_kernel`)
   extracts the five sub‑matrices, multiplies `left`/`right` by the optional
   mask, applies the three sigmoid gates and stores the three tensors
   `left`, `right` and `out_gate` in layout `[B, H, N, N]`.

4. **Core pairwise update** – a batched GEMM (`torch.bmm`) computes
   `out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]`,
   i.e. `out = left @ rightᵀ` on the flattened batch dimension
   `(B·H, N, N)` → shape `[B, H, N, N]`.

5. **Second LayerNorm** – normalises over the hidden dimension `H`.  The
   hidden axis is moved to the innermost position (`[B,N,N,H]`) so that the
   built‑in LayerNorm can operate on contiguous memory.  The normalised tensor
   is multiplied element‑wise by the previously computed `out_gate`.

6. **Final linear projection** – the gated, normalised tensor (flattened to
   `[R, H]`) is linearly projected back to the original channel dimension
   `dim` (`[R, dim]`) with another half‑precision GEMM, and finally cast to
   `float32`.

All heavy arithmetic stays in FP16; the few reductions for the two LayerNorms
are performed in FP32 for numerical stability.  The implementation supports
both masked and un‑masked inputs and runs entirely on the device.

Runtime target: ≤ 500 µs on an H100 for the benchmark configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection (input)
    left_ptr,      # [B, H, N, N]  FP16 output
    right_ptr,     # [B, H, N, N]  FP16 output
    out_gate_ptr,  # [B, H, N, N]  FP16 output
    mask_ptr,      # [R] (optional) FP16 mask (broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # Base offset for writing into the [B,H,N,N] tensors.
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # Process hidden dimension in tiles of size BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection: [R,5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → cast back to FP16 for accuracy
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets for the three output tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) implementation.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            • input_tensor : torch.Tensor [B, N, N, dim] (float32)
            • mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            • weights_dict : mapping containing the model parameters
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            • config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: processed tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # number of pairwise “tokens”
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # Reshape to [R, dim] and normalise over the last dimension.
    x_flat = x.reshape(R, dim).contiguous()
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # [R, dim] half‑precision

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (five linear maps in one GEMM)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) half

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, mask, compute sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # Tile sizes – tuned for H100
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Release temporaries early
    del proj, x_norm, x_norm_fp32, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise multiplication (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N]
    right_mat = right.view(B * H, N, N)              # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] half
    out = out.view(B, H, N, N)                          # [B, H, N, N] half

    # No longer need the raw projections
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    # Move H to the innermost axis so the memory for each (b,i,j) is contiguous.
    out_perm = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H] half

    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )   # still half‑precision

    # Return to the original layout [B, H, N, N] for the gate multiplication.
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # Apply the element‑wise out‑gate.
    out_gated = out_norm * out_gate
    del out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # Flatten spatial dimensions → shape (R, H)
    out_gated_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(R, H)

    w_out = w["to_out.weight"].to(torch.float16)          # (dim, H)
    final_flat = F.linear(out_gated_flat, w_out)         # (R, dim) half

    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
