"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces Y ∈ ℝ^{B×N×N×C}.
The heavy work is the fused linear projection (5·H outputs) and the
pairwise contraction ∑ₖ left[i,k]·right[j,k].  All expensive custom work is
handled by a single Triton kernel that extracts the five fused
projections, applies the optional pair mask, evaluates the three sigmoid
gates and writes the three tensors `left`, `right` and `out_gate` in the
layout required by a cuBLAS batched GEMM.

All remaining steps (two LayerNorms, the final projection and the
element‑wise gating) are performed with native PyTorch kernels (cuDNN /
cuBLAS).  This drastically reduces kernel‑launch overhead while still
exploiting Tensor‑Core acceleration for the large matrix multiplies.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
#  Kernel: split the fused projection + optional mask + sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] fused projection, FP16
    left_ptr,      # [B, H, N, N] output left  , FP16
    right_ptr,     # [B, H, N, N] output right , FP16
    out_gate_ptr,  # [B, H, N, N] out‑gate   , FP16
    mask_ptr,      # [R] optional mask (FP16) – broadcast over H
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows of the flattened pair tensor
    BLOCK_H: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------- decode flat row index r = b·N·N + i·N + k ----------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for a single hidden plane (N rows × N cols)
    hidden_stride = N * N
    # base address for (b, i, k) in the flattened [B·H·N·N] layout
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # -----------------------------------------------------------------
    #  Loop over hidden dimension tiles and write three blocks
    # -----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # index into the fused projection (5·H columns)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (more accurate in FP32, then back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination address in the flat [B·H·N·N] layout
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])
        

# --------------------------------------------------------------
#  Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    torch.cuda.empty_cache()                     # sanity – free unused memory
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N           # total pair‑wise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm  (native cuDNN – fast)
    # --------------------------------------------------------------
    # Shape (B,N,N,dim) → normalize over the last dim.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                       # float32
    x_norm = x_norm.to(torch.float16)   # half for the big GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # --------------------------------------------------------------
    # All five weight matrices have shape [hidden_dim, dim] (no bias)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # (5·H, dim)
    # fused linear projection – keep the (B,N,N,5·H) layout
    proj = F.linear(x_norm, fused_w)   # (B,N,N,5·H)  FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split / mask / apply sigmoid gates  → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # launch configuration – tuned for H≤128, N up to 1024
    BLOCK_R = 64   # rows per program (tuned)
    BLOCK_H = 64   # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core pairwise contraction  (batched GEMM, Tensor‑Core)
    # --------------------------------------------------------------
    # reshape to  (B·H, N, N)  – each hidden plane is a separate batch
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # shape (B·H,N,N) → (B,N,N,H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate
    # --------------------------------------------------------------
    # LayerNorm over the hidden dimension (last axis)
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                       # FP16, shape (B,N,N,H)

    # multiply by the previously computed out‑gate (broadcast over (B,N,N))
    out_norm = out_norm * out_gate
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection  (Linear hidden_dim → dim)
    # --------------------------------------------------------------
    # weight shape (dim, H) – keep FP16 for Tensor‑Core throughput
    w_out_fp16 = w["to_out.weight"].to(torch.float16)
    final = F.linear(out_norm, w_out_fp16)   # (B,N,N,dim)  FP16
    final = final.to(torch.float32)         # required output dtype
    return final
```
