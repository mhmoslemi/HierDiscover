"""
High‑performance outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  LayerNorm over the channel dimension *dim* of the 4‑D pair tensor.
    (torch.nn.functional.layer_norm, FP32 → FP16)

2️⃣  Fuse the five linear projections
        left, right, left_gate, right_gate, out_gate
    into a single GEMM producing a tensor of shape (R, 5·H) with R = B·N·N.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * decodes the flattened index (b,i,k) → (b,i,j) layout,
    * extracts the five slices,
    * applies sigmoid to the three gate slices,
    * applies the optional pair‑wise mask,
    * stores `left`, `right` and `out_gate` in layout (B, H, N, N) (FP16).

4️⃣  Batched matrix‑multiply (cuBLAS) over the sequence dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]

5️⃣  LayerNorm over the hidden dimension H of the contracted tensor,
    followed by element‑wise multiplication with `out_gate`.

6️⃣  Final linear projection (H → dim) using a single GEMM
    (`torch.nn.functional.linear`).  The result is float‑32 and reshaped
    to [B, N, N, dim].

Only the split‑and‑gate step is written in Triton; the remaining work
leverages highly‑optimized Torch/CuBLAS kernels.

                "left_proj.weight", "right_proj.weight",

                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"

    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )

    # 2️⃣ Fused projection (5 × hidden_dim)
    # --------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                            # [5·H, dim]

    # --------------------------------------------------------------

    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

        mask_ptr = mask_flat

        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_H = 64           # hidden‑dim tile (tunable)

        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4️⃣ Batched GEMM over the sequence dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * hidden_dim, N, N)          # [B·H, N, N] (fp16)
    right_mat = right.view(B * hidden_dim, N, N)        # [B·H, N, N] (fp16)

    # out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)

    # --------------------------------------------------------------

    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, hidden_dim)        # [R, H] fp16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, hidden_dim)

    out_norm = F.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still fp16 (internally computed in fp32)

    # element‑wise gate multiplication (still fp16)
    out_norm_gate = out_norm * out_gate_flat

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection (H → dim) – cuBLAS
    # --------------------------------------------------------------
    # weight shape for torch.nn.functional.linear is (out_features, in_features) = (dim, H)
    final_fp32 = F.linear(out_norm_gate, w["to_out.weight"])   # [R, dim] (float32)

    final = final_fp32.view(B, N, N, dim)                     # [B, N, N, dim] (float32)

    return final
```
