@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_R
    offs_r = start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask value per row if needed
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        m_val = m_val[:, None]  # broadcast over hidden
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    # base offset for hidden plane 0
    base = b * (H * hidden_stride) + i * N + k  # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute address into proj: each row has 5*H values
        # offset into proj: rows*5*H + hidden_offset
        proj_row = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_row,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        right_raw = tl.load(
            proj_ptr + proj_row + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_row + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_row + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_row + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )

        # sigmoid on gate values
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg = out_gate_val

        # compute destination offsets into left/right/out_gate
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

            proj_ptr + proj_row + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_row + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets into the 4‑D tensors (flattened)
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R, H_tile]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                               # [B,N,N,dim] → FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    # (R, dim) × (dim, 5·H) → (R, 5·H)
    proj = F.linear(x_norm.reshape(R, dim), fused_w)   # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, and sigmoid‑gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr                     # free intermediate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N).contiguous()
    right_mat = right.reshape(B * H, N, N).contiguous()
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dim + out_gate
    # ------------------------------------------------------------------
    # Move hidden dim to the last axis for torch.nn.functional.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()

    out_ln = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    out_ln = out_ln * out_gate_perm                            # apply out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (H → dim) – half‑precision matmul then FP32 cast
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)               # [dim, H]
    out_flat = out_ln.reshape(R, H)                            # [R, H]
    final_flat = F.linear(out_flat, w_out)                     # [R, dim] (FP16)
    final = final_flat.view(B, N, N, dim).to(torch.float32)    # cast back to FP32

    return final
