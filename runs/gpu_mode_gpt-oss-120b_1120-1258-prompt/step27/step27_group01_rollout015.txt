Pseudo:

@triton.jit
def norm_gate_kernel(
    out_ptr, out_gate_ptr,
    ln_w_ptr, ln_b_ptr,
    out_norm_ptr,
    rows, H,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Compute per-row sum and sum_sq across H
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out element
        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None,:],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: normalize, apply LN weight/bias, multiply out_gate, store
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out & gate
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None,:],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None,:],
                            mask=mask_r[:, None] & mask_h[None,:],
                            other=0.0).to(tl.float32)

        # Ln weight and bias per hidden dimension
        ln_w = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        ln_b = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # Normalize
        norm = ((out_vals - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]

        # Apply out_gate

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.
==========================================================

The operator works on a pairwise tensor *x* of shape [B, N, N, C] and
produces a tensor of the same shape.  The forward pass consists of

1️⃣  Row‑wise LayerNorm over the channel dimension *C* (FP32 → FP16).
2️⃣  A single fused projection that creates five hidden tensors
    (left, right, left_gate, right_gate, out_gate) of shape
    [B, H, N, N] (FP16).  The projection is performed with a cuBLAS GEMM.
3️⃣  Masking (optional) and sigmoidal gating are fused inside a Triton
    kernel that writes the three tensors ``left``, ``right`` and
    ``out_gate``.
4️⃣  Batched matrix‑multiply (cuBLAS) computes the core einsum:
    ``out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]``.
5️⃣  A Triton kernel computes, for each row **r = (b,i,j)**,
    the mean/variance of ``out[r]`` over the hidden dimension *H*,
    applies the second LayerNorm (scale & bias), multiplies by
    ``out_gate`` and writes the resulting tensor ``norm`` of shape
    (R, H) (R = B·N·N) in FP16.
6️⃣  The final linear projection
    :math:`y = norm·W_{out}^T`  (cuBLAS) maps (R, H) → (R, C) and the
    result is reshaped back to `[B, N, N, C]` in FP32.

All heavy linear algebra is delegated to cuBLAS (tensor‑core
accelerated), while the element‑wise work (layer‑norms, gating,
masking) is executed in custom Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, C] FP32
    y_ptr,       # [R, C] FP16 output
    w_ptr,       # [C]   LN weight (γ) FP32
    b_ptr,       # [C]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------ first pass: mean & variance -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------ second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split projection, apply mask and sigmoidal gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices

    # Base offset for (b,i,k) in the flattened [B, H, N, N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather the 5·H fused projection slices
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate multiplication → (R, H) tensor
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,       # [R, H] FP16   (raw contracted tensor)
    out_gate_ptr,  # [R, H] FP16   (gate produced in step 3)
    ln_w_ptr,      # [H]   FP32    (to_out_norm.weight)
    ln_b_ptr,      # [H]   FP32    (to_out_norm.bias)
    out_norm_ptr,  # [R, H] FP16   (result of LN + gate)
    rows, hidden, eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- first pass: compute mean & variance over H -----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden

        vals = tl.load(
            out_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_R]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_R]

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalize, apply LN‑scale/bias, apply out_gate -----
    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden

        # load raw out and gate values (FP16 → FP32)
        out_vals = tl.load(
            out_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        gate_vals = tl.load(
            out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm scale & bias (γ, β)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd → *γ + β
        norm = ((out_vals - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # Apply out_gate
        gated = norm * gate_vals

        # Store result (FP16)
        tl.store(
            out_norm_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            gated.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : dictionary containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & constants
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N        # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the channel dimension (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()           # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 128
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=8,
    )
    del x_flat                                            # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5·H hidden features (cuBLAS)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, C]
    proj = F.linear(x_norm, fused_w)                  # (R, 5·H) FP16
    del fused_w, x_norm                                 # free

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum (batched GEMM): out = left · rightᵀ   (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left, right                                        # free

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (produces (R, H) tensor)
    # ------------------------------------------------------------------
    # reshape to (R, H) where the hidden dimension is the innermost one
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)        # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP16
    del out, out_gate

    norm_out = torch.empty(R, H, dtype=torch.float16, device=device)  # [R, H]

    BLOCK_R_NORM = 128
    BLOCK_H_NORM = 64
    grid_norm = (triton.cdiv(R, BLOCK_R_NORM),)
    norm_gate_kernel[grid_norm](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],      # γ  (FP32, shape [H])
        w["to_out_norm.bias"],        # β  (FP32, shape [H])
        norm_out,
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R_NORM,
        BLOCK_H=BLOCK_H_NORM,
        num_warps=4,
    )
    del out_flat, out_gate_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (cuBLAS) → [R, dim] → reshape to [B, N, N, dim]
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)          # shape (dim, H)
    final_fp16 = F.linear(norm_out, w_out)               # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
