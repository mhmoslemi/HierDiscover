"""
Highly‑optimized outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣  Layer‑norm over the channel dimension of the input tensor using the
    torch built‑in kernel (FP32 → FP16).

2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into one `torch.nn.functional.linear` call.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * decodes the flat pair index (b,i,k) → (b,i,k)
    * optionally applies the mask,
    * computes the three sigmoid gates,
    * writes three tensors `left`, `right` and `out_gate` in layout
      `[B, H, N, N]` (FP16).

4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    → shape `[B, H, N, N]` (FP16).

5️⃣  Second layer‑norm over the hidden dimension H, element‑wise multiplication
    with `out_gate`, then a final linear projection to the original channel
    dimension.  All of this is expressed with highly‑vectorised PyTorch ops
    (mean/var in FP32, `torch.nn.functional.linear` for the projection).

Only the split‑and‑gate step is written in Triton, satisfying the requirement
to contain a custom kernel while leveraging Torch’s optimised kernels for the
heavy linear algebra.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Split + fuse + mask + sigmoid gates (Trion part of the pipeline)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H] fused projection (FP16)
    left_ptr,          # [B, H, N, N] output (FP16)
    right_ptr,         # [B, H, N, N] output (FP16)
    out_gate_ptr,      # [B, H, N, N] output (FP16)
    mask_ptr,          # [R] optional mask (FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For every pair token (b,i,k) we read the five fused projection vectors,
    apply an optional mask, compute sigmoid gates and write the three
    FP16 tensors required by the rest of the TriMul pipeline.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N                     # stride of one hidden plane in [B,H,N,N]
    # start offset for h = 0
    base_offset = b * (H * hidden_stride) + i * N + k

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout:
        #   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # mask not applied to the out‑gate

        # linear offset into the destination tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32, already on the same device as the input)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    R = B * N * N                 # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.layer_norm is heavily tuned; we use it and cast to FP16.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                       # [B, N, N, dim] FP16
    x_flat = x_norm.view(R, dim).contiguous()  # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single FP16 matmul
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5*H, dim]
    proj = F.linear(x_flat, fused_w)            # (R, 5*H) FP16
    del fused_w, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = (not nomask) and (mask is not None)
    if has_mask:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
    else:
        # dummy tensor – never read when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # launch the triton kernel
    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=int(has_mask),
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)           # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over H, gating, final projection
    # ------------------------------------------------------------------
    # compute mean / variance in FP32 (fast torch kernels)
    out_f32 = out.to(torch.float32)
    mu = out_f32.mean(dim=1, keepdim=True)                 # [B,1,N,N]
    var = out_f32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                         # [B,1,N,N]

    # layer‑norm (γ,β) over H
    ln_weight = w["to_out_norm.weight"]
    ln_bias = w["to_out_norm.bias"]
    # apply γ,β in FP32 then cast back to FP16
    out_norm = ((out_f32 - mu) * rstd)                     # (B,H,N,N) FP32 normalized
    out_norm = out_norm * ln_weight[None, :, None, None] + ln_bias[None, :, None, None]
    out_norm = out_norm.to(torch.float16)                  # back to FP16

    # multiply by the out‑gate (already FP16)
    out_norm = out_norm * out_gate                         # [B,H,N,N] FP16

    # final linear projection: move H to the last dim and project to `dim`
    out_perm = out_norm.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    final = F.linear(out_perm, w["to_out.weight"])        # [B,N,N,dim] FP16

    # return FP32 tensor as required
    return final.to(torch.float32)
