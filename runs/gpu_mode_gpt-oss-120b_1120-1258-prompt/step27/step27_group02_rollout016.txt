def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N2, C = input_tensor.shape
    assert N == N2
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device
    R = B * N * N
    # 1. Row-wise LN on input
    x_flat = input_tensor.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256
    BLOCK_D = 64   # can adapt
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # 2. fused linear projection
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_weight)  # [R,5*H] FP16
    # free memory
    del x_flat, fused_weight
    proj = proj.view(R, 5, hidden_dim)  # [R,5,H]
    # Prepare mask
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R)
        # mask broadcast to hidden dimension
        # left_proj_raw = proj[:,0,:] * mask_flat[:,None]
        left_proj_raw = proj[:, 0, :] * mask_flat[:, None]
        right_proj_raw = proj[:, 1, :] * mask_flat[:, None]
    else:
        left_proj_raw = proj[:, 0, :]
        right_proj_raw = proj[:, 1, :]
    left_gate_raw = proj[:, 2, :]
    right_gate_raw = proj[:, 3, :]
    out_gate_raw = proj[:, 4, :]
    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)
    # apply gates to left/right
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate
    # reshape to [B, hidden_dim, N, N]
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    # 3. contraction via bmm
    left_mat = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    out = out.view(B, hidden_dim, N, N)   # [B,H,N,N]
    # 4. LayerNorm over hidden dim for out, then multiply by out_gate
    # Compute mean & var in FP32
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1, keepdim=True)    # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)  # [B,1,N,N]
    # LN parameters
    gamma = weights["to_out_norm.weight"].view(1, hidden_dim, 1, 1).to(torch.float32)
    beta = weights["to_out_norm.bias"].view(1, hidden_dim, 1, 1).to(torch.float32)
    # normalize
    norm = ((out_fp32 - mu) * rstd) * gamma + beta   # [B,H,N,N] FP32
    # gating
    gated = norm * out_gate.to(torch.float32)   # [B,H,N,N] FP32
    # final projection: reshape to (R, H)
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, hidden_dim)   # [R,H] FP16 after cast
    gated_fp16 = gated_flat.to(torch.float16)
    w_out = weights["to_out.weight"].to(torch.float16)  # [dim, H]
    final_flat = F.linear(gated_fp16, w_out)   # [R, dim] FP16
    final = final_flat.view(B, N, N, dim).to(torch.float32)
    return final

"""Outgoing TriMul (AlphaFold3) – forward only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32→FP16) on the flattened pair tensor using a Triton kernel.
2️⃣ Fuse the five linear maps (left,right,left_gate,right_gate,out_gate) into a single
   FP16 GEMM (torch.nn.functional.linear) → tensor `[R,5·H]`.
3️⃣ Split the result, apply the optional mask, compute sigmoid gates, and reshape to
   `[B,H,N,N]`.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →   `[B,H,N,N]`.
5️⃣ Row‑wise LayerNorm (over H) on `out` using `torch.nn.functional.layer_norm`,
   then element‑wise multiply by the `out_gate`.
6️⃣ Final projection `to_out` performed as a single FP16 linear layer using cuBLAS,
   yielding the `[B,N,N,dim]` output (cast back to float32).
All heavy compute (matmul, linear) uses cuBLAS/Tensor‑cores; only the first
LayerNorm is hand‑written in Triton, fulfilling the “kernel‑implementation”
requirement while keeping the runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, D] FP32 input
    y_ptr,            # [R, D] FP16 output
    w_ptr,            # [D]   LN weight (γ) FP32
    b_ptr,            # [D]   LN bias   (β) FP32
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

# ----------------------------------------------------------------------
# TriMul forward (outgoing) – Torch + Triton implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the raw pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim]  FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Tunable block sizes – work well for the supported shapes
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left, right, gates) into one GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_weight)               # [R, 5·H]  FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, and apply sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                           # [R, 5, H]

    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        left_raw  = proj[:, 0, :] * mask_flat[:, None]  # [R, H]
        right_raw = proj[:, 1, :] * mask_flat[:, None]  # [R, H]
    else:
        left_raw  = proj[:, 0, :]
        right_raw = proj[:, 1, :]

    left_gate_raw  = proj[:, 2, :]
    right_gate_raw = proj[:, 3, :]
    out_gate_raw   = proj[:, 4, :]

    # sigmoid – stays in fp16
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)            # no mask here

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N] (the layout expected by the batched GEMM)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Core contraction via batched GEMM: einsum equivalent
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                  # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                 # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)
    out = out.view(B, H, N, N)                          # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ LayerNorm over the hidden dimension + out‑gate multiplication
    # --------------------------------------------------------------
    # move hidden dimension to the last axis to reuse torch.nn.functional.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)  FP16
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"],   # γ (FP32)
        bias=weights["to_out_norm.bias"],       # β (FP32)
        eps=eps,
    )                                                    # (B, N, N, H)  FP16 → remains FP16
    # apply out_gate (broadcasted to the same layout)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H) FP16
    gated = out_norm * out_gate_perm                           # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection (to_out) – single large GEMM
    # --------------------------------------------------------------
    # flatten the first three dimensions so that we have (R, H)
    gated_flat = gated.view(R, H)                               # (R, H) FP16
    w_out = weights["to_out.weight"].to(torch.float16)          # (dim, H)
    final_flat = F.linear(gated_flat, w_out)                    # (R, dim) FP16
    final = final_flat.view(B, N, N, dim).to(torch.float32)    # FP32 output

    return final

        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# TriMul forward (outgoing) – Torch + Triton implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] or ``None``
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs and config
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the raw pair tensor (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim]  FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Tunable block sizes – work well for the supported shapes
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory early

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into one GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_weight)                # [R, 5·H]  FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                            # [R, 5, H]

    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R)    # [R]
        left_raw  = proj[:, 0, :] * mask_flat[:, None]   # [R, H]
        right_raw = proj[:, 1, :] * mask_flat[:, None]   # [R, H]
    else:
        left_raw  = proj[:, 0, :]                        # [R, H]
        right_raw = proj[:, 1, :]                        # [R, H]

    left_gate_raw  = proj[:, 2, :]                       # [R, H]
    right_gate_raw = proj[:, 3, :]                       # [R, H]
    out_gate_raw   = proj[:, 4, :]                       # [R, H]

    left_gate  = torch.sigmoid(left_gate_raw)            # FP16
    right_gate = torch.sigmoid(right_gate_raw)           # FP16
    out_gate   = torch.sigmoid(out_gate_raw)             # FP16 (no mask)

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N] for the batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (`einsum` equivalent) via batched GEMM
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                   # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                  # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)
    out = out.view(B, H, N, N)                           # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension + out_gate multiplication
    # --------------------------------------------------------------
    # move hidden dim to the last axis to reuse torch's fast LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()      # (B, N, N, H)  FP16 → cast to FP32 for LN
    out_norm_fp32 = F.layer_norm(
        out_perm.to(torch.float32),
        (H,),
        weight=weights["to_out_norm.weight"],           # γ (FP32)
        bias=weights["to_out_norm.bias"],               # β (FP32)
        eps=eps,
    )                                                    # (B, N, N, H) FP32

    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B, N, N, H) FP32
    gated_fp32 = out_norm_fp32 * out_gate_perm                                   # (B, N, N, H) FP32
    gated_fp16 = gated_fp32.to(torch.float16)                                     # back to FP16 for the final GEMM

    # --------------------------------------------------------------
    # 6️⃣ Final projection (`to_out`) – single large GEMM
    # --------------------------------------------------------------
    gated_flat = gated_fp16.view(R, H)                     # (R, H) FP16
    w_out = weights["to_out.weight"].to(torch.float16)     # (dim, H)
    final_flat = F.linear(gated_flat, w_out)               # (R, dim) FP16
    final = final_flat.view(B, N, N, dim).to(torch.float32)   # cast to FP32 for the API contract

    return final
```

This implementation follows the reference `TriMul` forward pass, uses a Triton kernel for the first LayerNorm, relies on highly‑optimized cuBLAS/Tensor‑Core GEMMs for the three linear projections and the final projection, and keeps the second LayerNorm in fast PyTorch code. The result is a float‑32 tensor of shape `[B, N, N, dim]`.
