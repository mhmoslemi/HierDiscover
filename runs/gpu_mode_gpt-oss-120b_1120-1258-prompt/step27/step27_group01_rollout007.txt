"""
Outgoing TriMul (AlphaFold‑3) – forward only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input pair tensor (torch.nn.functional.layer_norm).
    This kernel is heavily optimized in cuDNN and is faster than a pure Triton
    implementation for the very large [R, dim] tensors (R = B·N·N).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM:
        proj = Linear(x_norm, W_fused)               # (R, 5·H)
    where H = hidden_dim.

3️⃣  Triton kernel ``split_fuse_gate_kernel``:
    * Decodes the flat row index r ↦ (b,i,k).
    * Splits the fused projection into three tensors ``left``, ``right`` and
      ``out_gate`` of shape [B, H, N, N] (FP16).
    * Applies the optional mask and the sigmoid gates (computed in FP32
      then cast back to FP16).

4️⃣  Batched matrix multiplication (cuBLAS):
        out = bmm(left, right^T)                     # [B·H, N, N] → [B, H, N, N]

5️⃣  Second LayerNorm over the hidden dimension H (torch.nn.functional.layer_norm
    on a view where H is the last axis).  The normalized tensor is multiplied
    element‑wise with ``out_gate`` (converted to FP32).

6️⃣  Final linear projection from hidden_dim → dim:
        out_final = (out_gated_flat @ to_out.weight.T)   # FP32
    The result is reshaped to [B, N, N, dim] and returned.

Only the split/fuse/gate step is written in Triton; the remaining
operations use highly‑optimized PyTorch/cuBLAS kernels, which on an
NVIDIA H100 yields runtimes well below the 500 µs target for the required
test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,            # [R, 5·H]   FP16 fused projection
    left_ptr,            # [B, H, N, N] FP16
    right_ptr,           # [B, H, N, N] FP16
    out_gate_ptr,        # [B, H, N, N] FP16
    mask_ptr,            # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcasted to hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices
    # Base offset for (b,i,k) in the flattened [B,H,N,N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather the 5·H fused projection slices for this hidden tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numerical stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimized (cuDNN) and
    # outperforms a custom Triton implementation for this large tensor.
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half().contiguous()   # [B,N,N,dim] → FP16
    del x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    # Concatenate the five projection matrices (each of shape [H, dim]).
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    # Flatten pair tensor to [R, dim] and apply the fused GEMM (cuBLAS).
    x_flat = x_norm.view(R, dim)               # already contiguous
    proj = F.linear(x_flat, fused_w)           # (R, 5·H) FP16
    del x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # Flatten mask to match the (b,i,k) ordering used in the kernel.
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (torch implementation)
    # ------------------------------------------------------------------
    # Move H to the last dimension to reuse torch.layer_norm.
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] FP16
    # LayerNorm expects FP32 for highest accuracy; cast up.
    out_ln = F.layer_norm(
        out_perm.float(),
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                        # [B, N, N, H] FP32
    out_ln = out_ln.permute(0, 3, 1, 2)                      # back to [B, H, N, N] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Apply out_gate (cast to FP32) and final projection
    # ------------------------------------------------------------------
    out_gated = out_ln * out_gate.float()                   # [B, H, N, N] FP32

    # Final linear projection from hidden_dim → dim.
    # Reshape to (B·N·N, H) so we can use a single GEMM.
    out_gated_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(-1, H)   # [B·N·N, H]
    # Weight shape for torch.nn.functional.linear: (out_features, in_features)
    w_out = w["to_out.weight"].to(torch.float32)           # [dim, H]
    out_final = F.linear(out_gated_flat, w_out, None)     # [B·N·N, dim] FP32
    out_final = out_final.view(B, N, N, dim)               # [B, N, N, dim]

    return out_final
