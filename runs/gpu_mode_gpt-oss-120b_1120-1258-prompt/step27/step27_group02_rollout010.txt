def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', True)
    eps = 1e-5
    device = input_tensor.device
    B,N,_,_ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm on x (float32) using torch.nn.functional.layer_norm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x_norm_fp16 = x_norm.to(torch.float16)

    # 2. Fuse linear projection
    fused_w = torch.cat([weights['left_proj.weight'], weights['right_proj.weight'], weights['left_gate.weight'], weights['right_gate.weight'], weights['out_gate.weight']], dim=0).to(torch.float16)
    x_flat = x_norm_fp16.view(R, dim)
    proj = torch.nn.functional.linear(x_flat, fused_w)  # (R,5*H) FP16

    # 3. Split, mask, gate via Triton kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).view(R)  # flatten
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64  # tune
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_fp16

    # 4. Batched GEMM
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N) FP16
    out = out.view(B, H, N, N)  # (B,H,N,N)

    # free left,right if memory needed
    del left, right

    # 5. LayerNorm over hidden dimension per (b,i,j)
    # reshape to (R, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H).float()
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        (H,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias'],
        eps=eps

    w_out_T = weights['to_out.weight'].to(torch.float16).t().contiguous()  # (H, dim)

    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – fully fused forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D pair tensor (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (cuBLAS) → shape (R, 5·H) where R = B·N·N.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
       * decodes the flat index (b,i,k) → (b,i,k) in the pair tensor,
       * extracts the 5 · H projections,
       * applies an optional mask and the sigmoid gates,
       * writes three tensors `left`, `right`, `out_gate` in layout
         [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →  FP16.
5️⃣  Row‑wise LayerNorm on `out` over the hidden dimension H
    (torch.nn.functional.layer_norm, FP32).
6️⃣  Multiply by `out_gate` (broadcasted) and project back to the
    original channel dimension with a fused FP16 GEMM
    (`torch.nn.functional.linear`).  The final result is cast to FP32.

Only one Triton kernel is used (step 3); all heavy linear algebra
operations are delegated to cuBLAS which on an H100 gives > 1 TFLOP/s,
allowing the whole operator to run well under the 500 µs target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                               # distance between hidden slices

    # base offset for hidden‑plane 0 in layout [B, H, N, N] (flattened)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # loop over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # address of the 5·H fused projections for this tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for stability, result stored as FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the channel dimension (torch, FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                               # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: 5·hidden_dim outputs (cuBLAS, FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    x_flat = x_norm.reshape(R, dim)                     # [R, dim]
    proj = F.linear(x_flat, fused_w)                    # [R, 5·H] FP16
    del x_norm, x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # tuneable; works well for all configs
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise contraction  (batched GEMM, cuBLAS)
    #    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension H (torch, FP32)
    # ------------------------------------------------------------------
    # flatten to (R, H) and cast to FP32 for numeric stability
    out_fp32 = out.permute(0, 2, 3, 1).contiguous().view(R, H).float()
    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=weights["to_out_norm.weight"],   # γ FP32
        bias=weights["to_out_norm.bias"],       # β FP32
        eps=eps,
    )                                               # [R, H] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection (FP16 GEMM)
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # FP16
    gated = (out_norm * out_gate_flat.float()).to(torch.float16)          # FP16

    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim] FP16
    final_fp16 = F.linear(gated, w_out_T)                                 # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)                # FP32

    return final
