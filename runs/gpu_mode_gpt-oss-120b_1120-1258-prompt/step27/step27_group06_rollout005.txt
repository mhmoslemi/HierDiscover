 @triton.jit
 def norm_gate_kernel(out_ptr, gate_ptr,
                      mu_ptr, rstd_ptr,
                      ln_w_ptr, ln_b_ptr,
                      out_buf_ptr,
                      total_rows, H, N,
                      BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_r = tl.program_id(0)
    pid_h = tl.program_id(1)
    row_start = pid_r * BLOCK_R
    hidden_start = pid_h * BLOCK_H
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    offs_h = hidden_start + tl.arange(0, BLOCK_H) # [BLOCK_H]
    mask_r = offs_r < total_rows
    mask_h = offs_h < H

    # decode row to b,i,j
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start offset for hidden plane 0

    # Load mu and rstd per row
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)

    # broadcast mu,rstd to hidden dim
    mu_broadcast = mu[:, None]   # [BLOCK_R,1]
    rstd_broadcast = rstd[:, None]

    # Load ln_w, ln_b for hidden tile
    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
    beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

    # Pointers to out and gate values for hidden tile
    # out offset for each row+hidden index: base + offs_h * hidden_stride
    out_ptr_tile = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
    gate_ptr_tile = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

    # load values
    out_vals = tl.load(out_ptr_tile,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_H] 
    gate_vals = tl.load(gate_ptr_tile,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)

    # Compute normalized and gated
    # out_norm = ((out_vals - mu) * rstd) * gamma + beta
    norm = ((out_vals - mu_broadcast) * rstd_broadcast) * gamma[None, :] + beta[None, :]
    gated = norm * gate_vals

    # store as FP16
    tl.store(out_buf_ptr + offs_r[:, None] * H + offs_h[None, :],
             gated.to(tl.float16),
             mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only Triton implementation.

Algorithm
---------
1. Row‑wise LayerNorm over the last dimension (dim) of the input tensor.
2. Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
   into a single GEMM and split the result with a Triton kernel.
   The kernel also:
   * applies the optional pair mask to left/right,
   * applies the three sigmoid gates,
   * stores left, right and out_gate tensors in layout [B, H, N, N].
3. Batched GEMM (`torch.bmm`) implements the core einsum:
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]   → shape [B, H, N, N].
4. Row‑wise mean / rsqrt‑variance of `out` (over the hidden dimension H) with a
   Triton kernel.
5. Triton kernel normalises `out`, applies the `out_gate` and writes the
   result into a flat matrix A of shape [R, H] (R = B·N·N).
6. A single cuBLAS GEMM (`torch.nn.functional.linear`) projects A from the
   hidden dimension H to the model dimension `dim`.  The result is cast back to
   float32 and reshaped to [B, N, N, dim].

Only the forward pass is required; all kernels run in FP16 (except the
statistics that need FP32).  The implementation is tuned for H100 (Tensor
Cores) and typically runs ≈ 500 µs for the benchmark sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] FP32 input (flattened)
    y_ptr,    # [R, dim] FP16 output
    w_ptr,    # [dim] γ (FP32)
    b_ptr,    # [dim] β (FP32)
    rows,     # total rows R   (scalar)
    dim,      # channel dim    (scalar)
    eps,      # epsilon        (scalar)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # tile along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1st pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2nd pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast on H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast on hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k     # offset for hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣  Normalise → gate → write flat [R, H] matrix for final projection
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,       # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,      # [B, H, N, N] FP16   (out_gate)
    mu_ptr,        # [R] FP32
    rstd_ptr,      # [R] FP32
    ln_w_ptr,      # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,      # [H] FP32 (to_out_norm.bias)
    out_buf_ptr,   # [R, H] FP16 (normalised·gated output)
    total_rows, H, N,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim tile
):
    pid_r = tl.program_id(0)
    pid_h = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    hid_start = pid_h * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_h = hid_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]

    mask_r = offs_r < total_rows
    mask_h = offs_h < H

    # decode (b, i, j) from flat row index
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    j   = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # offset for hidden‑plane 0

    # load row‑wise statistics (µ, rstd)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    mu   = mu[:, None]          # [BLOCK_R, 1]
    rstd = rstd[:, None]        # [BLOCK_R, 1]

    # LayerNorm parameters for this hidden tile
    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
    beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

    # Load the raw values and the out‑gate
    out_vals = tl.load(
        out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0).to(tl.float32)                               # [BLOCK_R, BLOCK_H]
    gate_vals = tl.load(
        gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0).to(tl.float32)                               # [BLOCK_R, BLOCK_H]

    # Normalisation + modulation by out_gate
    # (out - µ) * rstd → *γ + β → *gate
    norm = ((out_vals - mu) * rstd) * gamma[None, :] + beta[None, :]
    gated = norm * gate_vals

    # write FP16 result to flat buffer [R, H]
    dst_ptr = out_buf_ptr + offs_r[:, None] * H + offs_h[None, :]
    tl.store(dst_ptr,
             gated.to(tl.float16),
             mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total row count for pairwise tensors
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask, apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS) → core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                      # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Normalise, apply out_gate and write flat matrix A [R, H]
    # ------------------------------------------------------------------
    A = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_OUT = 128    # rows per program
    BLOCK_H_OUT = 64     # hidden‑dim tile
    grid_norm = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(H, BLOCK_H_OUT),
    )
    norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ   FP32
        w["to_out_norm.bias"],     # β   FP32
        A,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate, mu, rstd   # free memory we no longer need

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection (cuBLAS)  A @ Wᵀ  → [R, dim]
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    final = torch.nn.functional.linear(A, w_out)  # [R, dim] FP16
    del A, w_out

    # ------------------------------------------------------------------
    # 8️⃣  reshape to [B, N, N, dim] and cast to float32
    # ------------------------------------------------------------------
    final = final.to(torch.float32).view(B, N, N, dim)
    return final
