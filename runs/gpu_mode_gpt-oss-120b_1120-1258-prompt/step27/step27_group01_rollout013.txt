    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)

    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps).to(torch.float16)

    # 2. fused linear to produce (5*H) projections
    fused_w = torch.cat([...], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_w) # shape (R, 5*H) half

    # 3. Split gating kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        mask_flat = mask.to(torch.float16).reshape(R).contiguous()

        proj,
        left,
        right,
        out_gate,

        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    del proj

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_R
    offs_r = start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R
    # decode indices b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]

    # offsets for left/right/out_gate in flattened [B, H, N, N]
    stride_hidden = N * N
    base = b * (H * stride_hidden) + i * N + k  # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_offset = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]
        # load slices
        left_raw = tl.load(proj_ptr + proj_offset, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_offset + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_offset + 2 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_offset + 3 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_offset + 4 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid for gates (in fp32 for accuracy)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets
        dst = base[:, None] + offs_h[None, :] * stride_hidden  # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:   (R, dim) @ (dim, 5·H) → (R, 5·H)   (R = B·N·N).

3️⃣  Triton kernel *split_fuse_gate_kernel* extracts the five slices,
    applies the optional pair‑wise mask, computes three sigmoid gates
    (in FP32, cast back to FP16) and writes the three tensors
    `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).

4️⃣  Batched half‑precision GEMM (`torch.bmm`) performs the core
    contraction
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    (equivalent to the einsum in the reference code).

5️⃣  Row‑wise LayerNorm over the hidden dimension H, multiplication with
    `out_gate` and the final projection to *dim* are expressed with
    PyTorch primitives (`layer_norm` + `linear`).  All work is in
    FP16/Tensor‑Core; the tiny statistics are computed in FP32.

All kernels target Triton 3.3.1 on an H100 and use mixed precision.
Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]    FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]         FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from the flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask – broadcasted over the hidden dimension
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                         # distance between successive hidden slices
    # base offset for hidden‑slice h = 0 in the flattened [B, H, N, N] tensor
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # iterate over the hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ------- sliced view of the fused projection ---------------------------------
        # each row r contains 5·H values; we pick the H‑wide slice for the current h_tile
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for numeric stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets (flattened view of [B, H, N, N])
        dst = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm over `dim`  (FP32 stats, FP16 output)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    # torch.layer_norm internally uses FP32 for the reduction → we keep the result in FP16
    x_norm = F.layer_norm(x_flat,
                          (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5 × hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # [5·H, dim]
    proj = F.linear(x_norm, fused_w)              # (R, 5·H) FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM over the hidden dimension (cuBLAS, FP16 Tensor‑Cores)
    # ------------------------------------------------------------------
    # left / right are [B, H, N, N] → reshape to [B·H, N, N]
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise LayerNorm over the hidden dimension H, gate & final projection
    # ------------------------------------------------------------------
    # bring the hidden dimension to the last axis for the next ops
    out_perm      = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H] FP16
    out_rows      = out_perm.view(-1, H)                      # [R2, H] where R2 = B·N·N
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H] FP16
    out_gate_rows = out_gate_perm.view(-1, H)                # [R2, H]

    # layer‑norm on the hidden axis (FP32 statistics → FP16 output)
    out_norm = F.layer_norm(out_rows.float(),
                            (H,),
                            weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"],
                            eps=eps).half()

    # apply the out‑gate
    gated = out_norm * out_gate_rows   # FP16

    # final projection to `dim` (FP16 GEMM)
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final = F.linear(gated, w_out)                # [R2, dim] FP16

    # cast back to FP32 and reshape to the original 4‑D layout
    final = final.float().view(B, N, N, dim)

    return final
```
