    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = bool(config.get("nomask", True))
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim
    # 1: Row-wise layernorm
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    # launch kernel
    BLOCK_M = 256
    BLOCK_D = 64
    triton.cdiv(R, BLOCK_M)
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        weight_dict["norm.weight"],
        weight_dict["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # 2: fused projection (linear)
    fused_w = torch.cat(
        [
            weight_dict["left_proj.weight"],
            weight_dict["right_proj.weight"],
            weight_dict["left_gate.weight"],
            weight_dict["right_gate.weight"],
            weight_dict["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    # proj: (R,5H)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # FP16
    # free x_norm
    del x_norm
    # 3: gating and mask
    proj = proj.view(R, 5, H)  # (R,5,H)
    left_proj = proj[:, 0, :]  # (R,H)
    right_proj = proj[:, 1, :]
    left_gate = proj[:, 2, :]
    right_gate = proj[:, 3, :]
    out_gate = proj[:, 4, :]  # (R,H)
    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate_sig = torch.sigmoid(out_gate)
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(R).to(torch.float16)
        left = left_proj * left_gate * mask_flat[:, None]
        right = right_proj * right_gate * mask_flat[:, None]
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    # free proj and gate raw
    del proj, left_proj, right_proj, left_gate, right_gate
    # 4: reshape and batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2)   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2) # [B,H,N,N]

    left_mat = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H,N,N] FP16
    out = out.view(B, H, N, N)  # FP16
    # free left/right
    del left, right, left_mat, right_mat
    # 5: second LayerNorm (per token across H)
    out_flat = out.view(R, H)  # FP16
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        (H,),
        weight=weight_dict["to_out_norm.weight"],
        bias=weight_dict["to_out_norm.bias"],
        eps=eps,
    )
    # free out
    del out, out_flat
    # 6: out_gate multiplication
    out_gate_sig = out_gate_sig.view(R, H)  # ensure shape
    gated = out_norm * out_gate_sig
    # free out_gate and out_norm
    del out_norm, out_gate_sig
    # 7: final projection
    w_out = weight_dict["to_out.weight"].to(torch.float16)  # (dim, H)
    final = torch.nn.functional.linear(gated, w_out)  # (R, dim) FP16
    final = final.to(torch.float32).view(B, N, N, dim)
    return final

"""
Outgoing TriMul (AlphaFold‑3) – highly optimized forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened (B·N·N) tokens.
   – Custom Triton kernel `layernorm_kernel` reads FP32 input, computes
     mean/variance in FP32 and writes the normalized tensor in FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (`torch.nn.functional.linear`).
   The result has shape [R, 5·H] (R = B·N·N).

3️⃣ Split the fused projection with cheap PyTorch ops, apply the optional
   pair mask, evaluate the three sigmoids and obtain the three tensors
   `left`, `right` and `out_gate`.  All tensors are kept in FP16.

4️⃣ Batched GEMM (cuBLAS) implements the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
   The computation is performed as a batch of B·H matrix‑multiplications
   of size N×N (torch.bmm).  Result shape = [B,H,N,N] (FP16).

5️⃣ Second LayerNorm across the hidden dimension H.
   – Per‑token LayerNorm is performed with `torch.nn.functional.layer_norm`
     on the flattened view (R,H).  We use half‑precision weights for
     `to_out_norm` (the kernel works in FP32 internally, so no loss).

6️⃣ Gating with `out_gate` (already sigmoid‑ed) → `out_norm * out_gate`.

7️⃣ Final linear projection `to_out` (hidden_dim → dim) via a single
   GEMM (`torch.nn.functional.linear`).  The output is cast back to
   FP32 and reshaped to `[B, N, N, dim]`.

Only the first LayerNorm is realised as a Triton kernel; everything else
uses highly‑optimised PyTorch/cuBLAS kernels, drastically reducing kernel
launch overhead while preserving exact semantics of the reference
implementation.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

    rows,    # scalar  R
    dim,     # scalar  C
    eps,     # scalar  ε
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pair‑wise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) via Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free fp32 copy

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, and compute sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)               # (R, 5, H)

    left_proj   = proj[:, 0, :]             # (R, H)
    right_proj  = proj[:, 1, :]             # (R, H)
    left_gate   = proj[:, 2, :]             # (R, H)
    right_gate  = proj[:, 3, :]             # (R, H)
    out_gate    = proj[:, 4, :]             # (R, H)

    # sigmoid gates
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)   # will be used later

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16)      # (R,)
        mask_flat = mask_flat[:, None]                  # (R,1) broadcast
        left  = left_proj * left_gate * mask_flat
        right = right_proj * right_gate * mask_flat
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    del proj, left_proj, right_proj, left_gate, right_gate

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM – core einsum
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for the batch matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left.reshape(B * H, N, N)    # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)   # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right, left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dimension H (per token)
    # --------------------------------------------------------------
    # bring H to the last dimension to match the reference layout
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H)                    # (R, H) FP16

    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                            # (R, H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(R, H)               # (R, H) FP16

    gated = out_norm * out_gate                  # (R, H) FP16
    del out_norm, out_gate, out, out_flat

    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final = F.linear(gated, w_out)                # (R, dim) FP16
    final = final.to(torch.float32).view(B, N, N, dim)   # [B,N,N,dim] FP32

    return final
