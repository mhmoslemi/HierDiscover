"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pair representation (FP32 → FP16) using
   `torch.nn.functional.layer_norm` (highly tuned cudnn kernel).

2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
   into a single half‑precision GEMM:  `proj = X @ W_fused`  (X is [R,dim]).

3️⃣ A small Triton kernel (`split_fuse_gate_kernel`) splits the fused
   projection, applies the optional mask and the sigmoid gates, and writes
   the three tensors `left`, `right` and `out_gate` directly in the
   layout `[B, H, N, N]`.

4️⃣ Core einsum is a batched half‑precision GEMM:
      out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
   realised with `torch.bmm` (cuBLAS).

5️⃣ Layer‑Norm over the hidden dimension H of `out` (per pair) with the
   `to_out_norm` parameters, again via `torch.nn.functional.layer_norm`.

6️⃣ Multiply by the `out_gate` (element‑wise) and fuse the final projection
   (`to_out`) with a half‑precision GEMM (`torch.nn.functional.linear`).

All heavy matrix‑multiplies stay in FP16 to exploit Tensor‑cores.  Only a
single Triton kernel is used (step 3), satisfying the “must use a kernel”
requirement while minimizing kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16   fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]      FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode (b,i,k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over the hidden dimension
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # stride between hidden slices
    # Base offset of (b,i,k) in layout [B,H,N,N] for h = 0
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Loop over hidden‑dim tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection (5·H per row)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 (more stable) → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B,H,N,N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------

    """

    Args

        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs and config
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape               # x : [B, N, N, dim]
    R = B * N * N                       # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly‑optimised on H100
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, dim]  float32
    x_fp16 = x_norm.to(torch.float16)   # cast once
    del x_norm

    # Flatten to [R, dim] for the fused projection
    x_flat = x_fp16.reshape(R, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]

    proj = F.linear(x_flat, fused_w)    # [R, 5·H]  FP16
    del x_flat, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask broadcasted to [R] (float16)
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS, FP16)
    # --------------------------------------------------------------
    # left/right: [B, H, N, N] → view as (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right                    # free memory

    # --------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension H (per (b,i,j) pair)
    # --------------------------------------------------------------
    # Rearrange to (B, N, N, H) – a contiguous view is required
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                                                      # [B, N, N, H] FP16
    out_norm = out_norm.permute(0, 3, 1, 2)                # back to [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 6️⃣ Multiply with out_gate and final projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                              # FP16
    del out_norm, out_gate

    # reshape to (R, H) for the final linear layer
    gated_flat = gated.permute(0, 2, 3, 1).reshape(R, H)    # [R, H] FP16
    del gated

    w_out = weights["to_out.weight"].to(torch.float16)      # [dim, H]
    final_fp16 = F.linear(gated_flat, w_out)                # [R, dim] FP16
    final = final_fp16.to(torch.float32).reshape(B, N, N, dim)  # FP32 output

    return final
