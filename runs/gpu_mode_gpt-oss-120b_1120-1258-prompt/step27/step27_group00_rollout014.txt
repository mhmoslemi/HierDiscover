        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)   # FP16

        # stats using float32
        vals_out_f32 = vals_out_f16.to(tl.float32)
        sum_val += tl.sum(vals_out_f32, axis=1)
        sum_sq += tl.sum(vals_out_f32 * vals_out_f32, axis=1)

        # store to shared memory (flattened layout)
        row_idx = tl.arange(0, BLOCK_M)   # [BLOCK_M]
        # compute linear indices for each element
        shmem_offs = row_idx[:, None] * H + (k_start + tl.arange(0, BLOCK_K))[None, :]
        tl.store(out_shmem + shmem_offs, vals_out_f16,
                 mask=mask_m[:, None] & mask_k[None, :])
        tl.store(gate_shmem + shmem_offs, vals_gate,
                 mask=mask_m[:, None] & mask_k[None, :])

    # compute mean and rsqrt var
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # second pass: compute final projection
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load from shared memory
        row_idx = tl.arange(0, BLOCK_M)
        shmem_offs = row_idx[:, None] * H + (k_start + tl.arange(0, BLOCK_K))[None, :]
        vals_out_f16 = tl.load(out_shmem + shmem_offs,
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0)
        vals_gate = tl.load(gate_shmem + shmem_offs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        vals_out = vals_out_f16.to(tl.float32)
        vals_gate_f = vals_gate.to(tl.float32)

        # layernorm params
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # normalized term
        diff = (vals_out - mu[:, None]) * rstd[:, None]
        norm = diff * gamma[None, :] + beta[None, :]
        term = norm * vals_gate_f   # [M, K] FP32

        # weight tile (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)   # FP16

        # accumulate dot product (M,K)·(K,N) => (M,N)
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3 style) – optimized Triton implementation
===================================================================

The forward pass consists of the following stages (all tensors stay on GPU):

1️⃣  Row‑wise LayerNorm on the flattened pairwise tensor (FP32 → FP16).
2️⃣  A single fused linear projection (cuBLAS) produces five tensors:
    left_proj, right_proj, left_gate, right_gate and out_gate
    (shape (R, 5·H) with H = hidden_dim).
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional mask, computes the sigmoid gates and writes
    `left`, `right` and `out_gate` in layout `[B, H, N, N]` (FP16).
4️⃣  Batched GEMM (cuBLAS) evaluates the core einsum
    `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]`.
5️⃣  Triton kernel `fused_output_kernel` loads the contracted tensor `out`
    **once**, computes per‑row mean/variance, applies the second
    LayerNorm (`to_out_norm`), multiplies by the `out_gate` and finally
    projects to the model dimension (`to_out`).  All of these steps are
    fused into a single kernel, eliminating an extra global‑memory scan
    over `out`.

All tensors are kept in FP16 except the statistics (`mu`, `rstd`) which
are computed in FP32 for numerical stability.  The final result is
returned as a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] FP32 input (contiguous)
    y_ptr,            # [R, dim] FP16 output
    w_ptr,            # [dim]   LN weight (γ) FP32
    b_ptr,            # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                         # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# --------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for numeric stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask for out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and final projection
# --------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,         # [H]          FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]          FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]     FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32  (result)
    total_rows, N, dim,
    eps,
    H: tl.constexpr,               # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    Computes for each pair (b,i,j) the following:
        mu, rstd ← mean & rsqrt‑var of out[b,:,i,j]
        out_norm   ← ((out - mu) * rstd) * ln_w + ln_b
        out_norm   ← out_norm * out_gate
        out_final  ← Σ_h out_norm[h] * W_out_T[h, :]
    Everything is fused: the  `out` tensor is read once, statistics are
    accumulated, values are stored to shared memory and then the final
    projection is performed without a second global‑memory scan.
    """
    pid_m = tl.program_id(0)   # tiles over rows (R = B·N·N)
    pid_n = tl.program_id(1)   # tiles over output dimension (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from the flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn                              # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + j     # start address for h == 0

    # ------------------------------------------------------------------
    # Shared memory buffers for a tile of rows (BLOCK_M) × hidden (H)
    # ------------------------------------------------------------------
    out_shmem = tl.shared([BLOCK_M * H], dtype=tl.float16)
    gate_shmem = tl.shared([BLOCK_M * H], dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: load `out` & `gate`, compute row‑wise mean/var,
    # and store the values in shared memory for reuse.
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        # global pointers with stride = hidden_stride
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out_f16 = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                                                   # FP16
        vals_gate_f16 = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                                                   # FP16

        # accumulate statistics in FP32
        vals_out_f32 = vals_out_f16.to(tl.float32)
        sum_val += tl.sum(vals_out_f32, axis=1)
        sum_sq  += tl.sum(vals_out_f32 * vals_out_f32, axis=1)

        # store the raw values to shared memory (flattened layout)
        row_idx = tl.arange(0, BLOCK_M)   # [BLOCK_M]
        shmem_offs = row_idx[:, None] * H + (k_start + tl.arange(0, BLOCK_K))[None, :]  # [M,K]
        tl.store(out_shmem + shmem_offs, vals_out_f16,
                 mask=mask_m[:, None] & mask_k[None, :])
        tl.store(gate_shmem + shmem_offs, vals_gate_f16,
                 mask=mask_m[:, None] & mask_k[None, :])

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: read from shared memory, apply the second LayerNorm,
    # multiply by out_gate and accumulate the final projection.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load the previously stored values from shared memory
        row_idx = tl.arange(0, BLOCK_M)
        shmem_offs = row_idx[:, None] * H + (k_start + tl.arange(0, BLOCK_K))[None, :]
        vals_out_f16 = tl.load(out_shmem + shmem_offs,
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0)           # FP16
        vals_gate_f16 = tl.load(gate_shmem + shmem_offs,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0)           # FP16

        vals_out_f32 = vals_out_f16.to(tl.float32)
        vals_gate_f32 = vals_gate_f16.to(tl.float32)

        # LayerNorm parameters (γ, β) for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # (out - μ) * rstd → γ· + β → *out_gate
        diff = (vals_out_f32 - mu[:, None]) * rstd[:, None]          # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]
        term = norm * vals_gate_f32                                 # [M,K] (FP32)

        # Load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # FP16

        # Accumulate (M,K)·(K,N) → (M,N) ; the dot operates in FP16→FP32
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # Write the final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – cuBLAS for speed
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64    # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        N,
        dim,
        eps,
        H=H,                       # compile‑time hidden dimension
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
```
