import triton
import triton.language as tl

@triton.jit
def final_fused_kernel(
    out_ptr, gate_ptr, ln_w_ptr, ln_b_ptr, w_out_T_ptr, out_final_ptr,
    R, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    # Determine row and column offsets
    row_start = pid * BLOCK_R
    col_start = pid_n * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_d = col_start + tl.arange(0, BLOCK_D)

    mask_r = offs_r < R
    mask_d = offs_d < dim

    nn = N * N
    hidden_stride = N * N

    # Decode batch and i,j from offs_r
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden=0
    base_offset = b * (H * hidden_stride) + i * N + j   # [BLOCK_R]

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    acc1 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)  # out_h * gate_h * w
    acc2 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)  # gate_h * gamma_h * w
    acc3 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)  # gate_h * beta_h * w

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate values
        # shape [BLOCK_R, BLOCK_H]
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)   # FP16
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)   # FP16

        # accumulate sums for mean/var
        out_fp32 = out_vals.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

        # compute out*gate
        out_gate = out_vals * gate_vals   # FP16

        # load weight slice for current hidden block, dim tile
        w = tl.load(w_out_T_ptr + offs_h[:, None] * dim + offs_d[None, :],
                    mask=mask_h[:, None] & mask_d[None, :],
                    other=0.0)   # FP16

        # acc1 contributions
        acc1 += tl.dot(out_gate, w)   # [BLOCK_R, BLOCK_D], FP32

        # load gamma and beta, cast to fp16
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float16)   # [BLOCK_H]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float16)   # [BLOCK_H]

        # compute gate * gamma and gate * beta
        gate_gamma = gate_vals * gamma[None, :]   # FP16
        gate_beta  = gate_vals * beta[None, :]    # FP16

        acc2 += tl.dot(gate_gamma, w)   # [BLOCK_R, BLOCK_D]
        acc3 += tl.dot(gate_beta, w)    # [BLOCK_R, BLOCK_D]

    # compute mean/variance
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)   # [BLOCK_R]

    # broadcast rstd and mu to shape [BLOCK_R,1]
    rstd_b = rstd[:, None]   # [BLOCK_R,1]
    mu_b = mu[:, None]

    # final output
    out_tile = rstd_b * (acc1 - mu_b * acc2) + acc3   # [BLOCK_R, BLOCK_D]

    # store
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr, out_tile,
             mask=mask_r[:, None] & mask_d[None, :])

"""
TriMul (outgoing) – high‑performance forward only Triton implementation.
--------------------------------------------------------------------------------
The operator works on a pair‑tensor `x` of shape [B, N, N, dim] and returns a
tensor of the same shape.  The algorithm follows the “outgoing” version from
AlphaFold‑3:

1️⃣  Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16).
2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into one FP16 GEMM that produces a tensor of shape (R,5⋅H) where
    R = B·N·N and H = hidden_dim.
3️⃣  Split the fused projection, apply the optional mask and the three sigmoid
    gates (`left_gate`, `right_gate`, `out_gate`) in a Triton kernel.
4️⃣  Batched FP16 GEMM (cuBLAS) computes the core contraction:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]    (H independent mat‑muls)
5️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    linear projection (`to_out`) into a single Triton kernel.  This kernel:
      * reads the contracted tensor `out` and `out_gate`,
      * computes the per‑row mean/variance of `out`,
      * performs the LayerNorm (γ,β from `to_out_norm`),
      * multiplies by `out_gate`,
      * projects to the model dimension `dim` using `to_out.weight`.
    All work is done in registers; the intermediate `[B, H, N, N]` tensor is
    streamed only once.

The implementation uses mixed‑precision (FP16 arithmetic, FP32 accumulation) and
targets Triton 3.3.1 on an NVIDIA H100.  The final result is float‑32 as required
by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      γ (FP32)
    b_ptr,          # [dim]      β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------- pass‑1: mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- pass‑2: write normalized tensor (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]               # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flattened index r -> (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside proj: (R,5·H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Gather the five concatenated projections
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offsets for the current hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,         # [H]          FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]          FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]     FP16 (transposed to_out weight)
    out_final_ptr,    # [R, dim]     FP32 output, R = B·N·N
    R, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)      # rows over R
    pid_n = tl.program_id(1)    # output‑dim tiles

    row_start = pid * BLOCK_R
    col_start = pid_n * BLOCK_D

    offs_m = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    offs_d = col_start + tl.arange(0, BLOCK_D)   # [BLOCK_D]

    mask_m = offs_m < R
    mask_d = offs_d < dim

    nn = N * N
    hidden_stride = N * N

    # decode (b,i,j) from flattened row index
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # start of the hidden dimension for this (b,i,j)
    base_offset = b * (H * hidden_stride) + i * N + j   # [BLOCK_R]

    # accumulators
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)        # Σ out_h
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)        # Σ out_h²

    acc1 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32) # Σ out_h * gate_h * W_h,d
    acc2 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32) # Σ gate_h * γ_h * W_h,d
    acc3 = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32) # Σ gate_h * β_h * W_h,d

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---- load contracted tensor and gate (FP16) ----
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)                     # [BLOCK_R, BLOCK_H] FP16
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)                    # [BLOCK_R, BLOCK_H] FP16

        # ---- statistics for LayerNorm (FP32) ----
        out_fp32 = out_vals.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)                    # [BLOCK_R]
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)         # [BLOCK_R]

        # ---- prepare tensors for the three dot‑products ----
        out_gate = out_vals * gate_vals                         # FP16

        # load weight slice W[h, d] for this hidden block
        w = tl.load(w_out_T_ptr + offs_h[:, None] * dim + offs_d[None, :],
                    mask=mask_h[:, None] & mask_d[None, :],
                    other=0.0)                                 # [BLOCK_H, BLOCK_D] FP16

        # acc1 ← Σ out_h * gate_h * W_h,d
        acc1 += tl.dot(out_gate, w)                             # FP32

        # load γ and β (FP32 → FP16)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float16)  # [BLOCK_H]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float16)  # [BLOCK_H]

        # acc2 ← Σ gate_h * γ_h * W_h,d
        gate_gamma = gate_vals * gamma[None, :]                  # FP16
        acc2 += tl.dot(gate_gamma, w)                            # FP32

        # acc3 ← Σ gate_h * β_h * W_h,d
        gate_beta = gate_vals * beta[None, :]                    # FP16
        acc3 += tl.dot(gate_beta, w)                             # FP32

    # ---- LayerNorm statistics ----
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                                  # [BLOCK_R]

    # ---- final output computation:
    # out_final = rstd * (acc1 - mu * acc2) + acc3
    mu_b   = mu[:, None]                                        # [BLOCK_R,1]
    rstd_b = rstd[:, None]                                      # [BLOCK_R,1]

    out_tile = rstd_b * (acc1 - mu_b * acc2) + acc3             # [BLOCK_R, BLOCK_D]

    # ---- store result (row‑major [R, dim]) ----
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr,
             out_tile,
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation using Triton + cuBLAS.
    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing (all FP32) model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 256
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: (5·H) columns
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the contract‑axis (cuBLAS FP16)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()               # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    # Transpose to_out weight for better memory access (H, dim) → FP16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 64     # rows per program (tiles over R)
    BLOCK_H_OUT = 32     # hidden‑dim chunk (H ≤ 128 → 4 tiles)
    BLOCK_D_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],      # γ  (FP32)
        w["to_out_norm.bias"],        # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=8,
    )
    # out and out_gate are no longer needed
    del out, out_gate, w_out_T

    # reshape final tensor to the required layout [B, N, N, dim]
    return final
