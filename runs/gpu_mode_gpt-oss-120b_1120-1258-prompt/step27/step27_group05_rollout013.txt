@triton.jit
def norm_gate_term_kernel(
    out_ptr, gate_ptr, term_ptr,
    ln_w_ptr, ln_b_ptr,
    R, H, N, eps,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start_row = pid * BLOCK_R
    offs_r = start_row + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j  # start offset for hidden dim 0

    # First pass: compute sum and sum_sq per row.
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_tile = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)   # shape [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(out_tile, axis=1)
        sum_sq += tl.sum(out_tile * out_tile, axis=1)

    # Compute mean, var, rstd
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute term and store
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate
        out_tile = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)

        gate_tile = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).  
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single cuBLAS GEMM → tensor [R, 5·H] (R = B·N·N).  
3️⃣  Split the fused projection, apply the optional pair mask and the three
    sigmoid gates via a Triton kernel → tensors `left`, `right`, `out_gate`
    with layout [B, H, N, N] (FP16).  
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → [B, H, N, N] (FP16).  
5️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection into a single Triton kernel that writes the intermediate
    tensor `term` ∈ ℝ^{R×H} (FP16).  The kernel computes per‑row
    μ,σ², then
        term = ((out - μ)·rstd·γ + β)·out_gate.
6️⃣  A single cuBLAS GEMM (torch.nn.functional.linear) projects `term`
    from H → dim, yielding the final output `[B, N, N, dim]` (float32).

All kernels use mixed precision (FP32 for statistics, FP16 for bulk data)
and run on an H100 with Triton 3.3.1.  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]  FP32
    y_ptr,    # [R, dim]  FP16 output
    w_ptr,    # [dim]    γ (FP32)
    b_ptr,    # [dim]    β (FP32)
    rows,     # scalar R
    dim,      # scalar dim
    eps,      # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off + tl.arange(0, BLOCK_H)                 # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside fused projection: [R, 5·H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate multiplication and produce a
#      term matrix [R, H] ready for a cuBLAS GEMM.
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_term_kernel(
    out_ptr,      # [B, H, N, N] FP16  (raw contracted tensor)
    gate_ptr,     # [B, H, N, N] FP16  (out_gate)
    term_ptr,     # [R, H] FP16 (output term)
    ln_w_ptr,     # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,     # [H] FP32 (to_out_norm.bias)
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    # -------------------------------------------------
    # First pass: compute row‑wise mean & variance
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # Second pass: compute (norm·γ+β)*gate  → term
    # -------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load tensors
        out_tile = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)      # [BLOCK_R, BLOCK_H]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)     # [BLOCK_R, BLOCK_H]

        # layer‑norm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32

        # ((out - μ)·rstd) → γ· + β → *gate
        normed = (out_tile - mean[:, None]) * rstd[:, None]          # [R, H]
        scaled = normed * gamma[None, :] + beta[None, :]             # [R, H]
        term_fp16 = (scaled * gate_tile).to(tl.float16)             # [R, H]

        # store term: [R, H] row‑major
        term_offset = offs_r[:, None] * H + offs_h[None, :]
        tl.store(term_ptr + term_offset,
                 term_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])

    """

    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm, fused_w)           # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right  # no longer needed

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate → term matrix [R, H]
    # ------------------------------------------------------------------
    term = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_TERM = 64
    BLOCK_H_TERM = 64
    grid_term = (triton.cdiv(R, BLOCK_R_TERM),)
    norm_gate_term_kernel[grid_term](
        out,
        out_gate,
        term,
        w["to_out_norm.weight"],   # γ   FP32
        w["to_out_norm.bias"],     # β   FP32
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=8,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    # weight shape for torch.nn.functional.linear: [out_features, in_features] = [dim, H]
    w_out = w["to_out.weight"].to(torch.float16).contiguous()   # [dim, H]

    final = torch.nn.functional.linear(term, w_out)  # [R, dim]  FP16 (accumulates in FP32)
    final = final.to(torch.float32)                  # cast back to FP32
    final = final.view(B, N, N, dim)                 # [B, N, N, dim]

    return final
```
