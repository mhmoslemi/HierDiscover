"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (cuBLAS) → shape (R, 5·H) where R = B·N·N.
3️⃣  Split the fused projection, apply the optional mask,
    compute the three sigmoid gates and write `left`, `right`,
    `out_gate` into layout [B, H, N, N] (FP16) via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B, H, N, N] (FP16)
5️⃣  A *single* Triton kernel fuses:
       – row‑wise mean / variance of `out` (LayerNorm over hidden dim H)
       – the second LayerNorm (`to_out_norm`), multiplication by `out_gate`
       – the final projection (`to_out`) → output `[B, N, N, dim]` (FP32).

All heavy linear algebra (steps 2 & 4) runs on cuBLAS, the only
memory‑intensive kernels are 1, 3 and the fused 5 which together
read the contracted tensor only once.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, D] FP32 input
    y_ptr,      # [R, D] FP16 output
    w_ptr,      # [D]   LN weight γ (FP32)
    b_ptr,      # [D]   LN bias   β (FP32)
    rows, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]
    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k   (i: first seq idx, k: second)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mval[:, None]                                   # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # stride between hidden slices

    # base offset for (b,i,k) at hidden‑dim = 0  (layout [B,H,N,N])
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, then back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate, and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N]  contracted tensor (FP16)
    gate_ptr,         # [B, H, N, N]  out_gate (FP16)
    ln_w_ptr,         # [H]           to_out_norm.weight γ (FP32)
    ln_b_ptr,         # [H]           to_out_norm.bias   β (FP32)
    w_out_T_ptr,      # [H, dim]      transposed to_out weight (FP16)
    out_final_ptr,    # [B, N, N, dim]  output (FP32)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """
    One‑pass kernel:
        * computes μ, σ² over the hidden dimension H,
        * applies LayerNorm (γ,β) and multiplies by out_gate,
        * performs the final linear projection in the same loop.
    """
    pid_m = tl.program_id(0)        # rows (R)
    pid_n = tl.program_id(1)        # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N                     # stride between hidden slices

    # base offset for the hidden‑dim slice h = 0
    base = b * (H * stride_hidden) + i * N + j   # [BLOCK_M]

    # accumulators
    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·a·W
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ a·W
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ b·W
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)         # Σ out_h
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)         # Σ out_h²

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)            # [BLOCK_K]
        mask_h = offs_h < H

        # ------------------- loads -------------------
        out_tile = tl.load(
            out_ptr + base[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )                             # FP16  [BLOCK_M, BLOCK_K]

        gate_tile = tl.load(
            gate_ptr + base[:, None] + offs_h[None, :] * stride_hidden,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )                             # FP16  [BLOCK_M, BLOCK_K]

        w_tile = tl.load(
            w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :],
            mask=mask_h[:, None] & mask_n[None, :],
            other=0.0,
        )                             # FP16  [BLOCK_K, BLOCK_N]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32 [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32 [BLOCK_K]

        # cast to fp16 for the dot products
        gamma_fp16 = gamma.to(tl.float16)          # [BLOCK_K]
        beta_fp16  = beta.to(tl.float16)           # [BLOCK_K]

        # a = gate * gamma ,  b = gate * beta   (both fp16)
        a_fp16 = gate_tile * gamma_fp16[None, :]   # [BLOCK_M, BLOCK_K]
        b_fp16 = gate_tile * beta_fp16[None, :]    # [BLOCK_M, BLOCK_K]

        # ---- statistics (need fp32) ----
        out_fp32 = out_tile.to(tl.float32)        # [BLOCK_M, BLOCK_K]
        sum_out += tl.sum(out_fp32, axis=1)       # [BLOCK_M]
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

        # ---- contributions to the final projection ----
        # (out * a)·W
        out_a = out_tile * a_fp16                 # fp16
        acc1 += tl.dot(out_a, w_tile).to(tl.float32)

        # a·W
        acc2 += tl.dot(a_fp16, w_tile).to(tl.float32)

        # b·W
        acc3 += tl.dot(b_fp16, w_tile).to(tl.float32)

    # ------------------- post‑processing -------------------
    mu   = sum_out / H                                 # [BLOCK_M]
    var  = sum_sq / H - mu * mu                        # [BLOCK_M]
    rstd = tl.rsqrt(var + eps)                         # [BLOCK_M]

    # final formula: rstd*(acc1 - mu*acc2) + acc3
    term = rstd[:, None] * (acc1 - mu[:, None] * acc2) + acc3   # [BLOCK_M, BLOCK_N]

    # store (flattened view of [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, term, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (FP32)
        weights["norm.bias"],     # β (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    #    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse LN (to_out_norm), out_gate and final projection
    # ------------------------------------------------------------------
    # transposed final weight: [H, dim]
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program (tuned for occupancy)
    BLOCK_K_OUT = 32          # hidden‑dim tile (H ≤ 128)
    BLOCK_N_OUT = 32          # output‑dim tile (dim ≤ 384)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ (FP32, [H])
        weights["to_out_norm.bias"],     # β (FP32, [H])
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # --------------------------------------------------------------
    return final
