    stride_ba, stride_bk, stride_bn, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,

"""
TriMul (outgoing) – Triton‑accelerated forward pass
---------------------------------------------------

Algorithm
~~~~~~~~~
1. Layer‑norm the input tensor on the last dimension (dim).
2. Two linear projections → *left* and *right* (shape [B,N,N,H]).
3. Optional mask is applied to the projected tensors.
4. Gating: sigmoid(linear) on the normalized input and element‑wise
   multiplication with *left*/*right*.
5. Compute the cubic reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (`left @ rightᵀ`) for every hidden slice.
   The GEMM is fused into a custom Triton kernel that works in FP16
   (half‑precision) for the inputs and accumulates in FP32.
   No layout permutation is performed – the kernel reads directly
   from the original [B,N,N,H] layout, avoiding a costly 0.5 GB copy.
6. Layer‑norm on the hidden dimension, element‑wise gating with *out_gate*,
   and a final linear projection back to the original *dim*.

The kernel tiles the multiplication in the (M,N,K) space with
BLOCK_M × BLOCK_N × BLOCK_K and uses ``tl.dot`` (tensor‑core aware)
to obtain high throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction (left @ rightᵀ)
# ----------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (FP16 for A/B, FP32 for C)
    M, N, K,                           # shapes (M=N=seq_len, K=seq_len)
    stride_am, stride_ai, stride_ak, stride_ah,   # A strides: batch, i, k, hidden
    stride_ba, stride_bk, stride_bn, stride_bh,   # B strides: batch, k, j, hidden
    stride_cm, stride_ci, stride_cj, stride_ch,   # C strides: batch, i, j, hidden
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """Compute C = A @ Bᵀ for every (batch, hidden) slice.

    A : (batch, hidden, M, K)  – stored as (B, M, K, H) in memory
    B : (batch, hidden, N, K)  – stored as (B, N, K, H) in memory
    C : (batch, hidden, M, N)  – stored as (B, M, N, H) in memory
    All tensors are flat‑pointer interpreted with the provided strides.
    """
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile col index
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # --------------------------------------------------------------------- #
    # Decode batch and hidden indices
    # --------------------------------------------------------------------- #
    batch_idx  = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # --------------------------------------------------------------------- #
    # Thread‑local offsets within the tile
    # --------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # K‑loop (tiling)
    # --------------------------------------------------------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile : (M × K)   – shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile : (K × N)   – shape (BLOCK_K, BLOCK_N)
        # Note: B is accessed as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot product (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------- #
    # Write the result back to C
    # --------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, S, S, C]   (C == dim, float32)
        - mask_tensor   : [B, S, S]     (bool / float32  or None)
        - weights_dict  : all linear / layer‑norm parameters (already on device)
        - config_dict   : {'dim': C, 'hidden_dim': H,
                           'nomask': bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, S, S, C] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # Grab model weights (all tensors are on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm on the last (dim) axis
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  (FP32)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → left / right
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B, S, S, H]  (FP32)
    right = F.linear(x, right_proj_weight)  # [B, S, S, H]  (FP32)

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the hidden axis)
    # --------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections) and element‑wise gating
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Batched GEMM (left @ rightᵀ) – Triton
    # --------------------------------------------------------------------- #
    # Cast to half for the compute‑intensive GEMM
    left_h  = left.to(torch.float16)   # [B, S, S, H]  (contiguous)
    right_h = right.to(torch.float16)  # [B, S, S, H]

    # Output buffer (FP32 accumulation)
    out_fp32 = torch.empty((B, S, S, hidden_dim),
                           dtype=torch.float32,
                           device=device)

    # Choose tile sizes that are a multiple of 16 (tensor‑core friendly)
    BLOCK_M = 128   # rows of the output tile
    BLOCK_N = 128   # cols of the output tile
    BLOCK_K = 64    # reduction tile size (must be multiple of 16)

    # Grid definition:
    #  - pid_m : ⌈S / BLOCK_M⌉  rows
    #  - pid_n : ⌈S / BLOCK_N⌉  cols
    #  - pid_z : B * hidden_dim  (flattened batch‑hidden)
    grid = (
        triton.cdiv(S, BLOCK_M),          # number of tiles along M
        triton.cdiv(S, BLOCK_N),          # number of tiles along N
        B * hidden_dim                    # flattened (batch, hidden)
    )

    # Launch the kernel --------------------------------------------------- #
    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        # Strides for A (left) – note the layout is [B, S, S, H]
        stride_am=left_h.stride(0),   # batch stride
        stride_ai=left_h.stride(1),   # i‑stride      = S * H
        stride_ak=left_h.stride(2),   # k‑stride      = H
        stride_ah=left_h.stride(3),   # hidden stride = 1
        # Strides for B (right) – we read it as (k, j)
        stride_ba=right_h.stride(0),  # batch stride
        stride_bk=right_h.stride(2),  # k‑stride      = H
        stride_bn=right_h.stride(1),  # j‑stride      = S * H
        stride_bh=right_h.stride(3),  # hidden stride = 1
        # Strides for C (output) – layout [B, S, S, H]
        stride_cm=out_fp32.stride(0),  # batch stride
        stride_ci=out_fp32.stride(1),  # i‑stride      = S * H
        stride_cj=out_fp32.stride(2),  # j‑stride      = H
        stride_ch=out_fp32.stride(3),  # hidden stride = 1
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_WARPS=8,                   # launch 8 warps per block (H100 friendly)
        hidden_dim=hidden_dim,
    )

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm on the hidden dimension
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # --------------------------------------------------------------------- #
    # 7. Apply out‑gate and final linear projection back to *dim*
    # --------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, S, S, dim]

    return out
