import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton Batched MatMul (outgoing TriMul)
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32, 'num_warps': 8}, num_stages=3),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32, 'num_warps': 4}, num_stages=3),
        triton.Config({'BLOCK_M': 256, 'BLOCK_N': 64, 'BLOCK_K': 64, 'num_warps': 8}, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128, 'BLOCK_K': 32, 'num_warps': 4}, num_stages=2),
    ],
    key=['M', 'N', 'K'],
)
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ak2, # a: batch, i, k
    stride_bm, stride_bn, stride_bk, # b: batch, j, k (j is row index after permute)
    stride_cm, stride_ci, stride_cj, # c: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A: (batch, M, K)    fp16
    B: (batch, N, K)    fp16   note: we read B as (K,N) for transpose
    C: (batch, M, N)    fp32
    """
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch*hidden

    # Compute the start offsets of the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask to avoid out‑of‑bounds.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current block for A and B.
    # A (batch, i, k)
    a_ptrs = a_ptr + pid_b * stride_am + offs_m[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * stride_ak2
    # B (batch, j, k) but we need Bᵀ → (k, j)
    b_ptrs = b_ptr + pid_b * stride_bm + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    # Accumulator in fp32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for this tile.
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile.
        a = tl.load(
            a_ptrs + cur_k[None, :] * stride_ak2,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (as transposed).
        b = tl.load(
            b_ptrs + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute partial product.
        acc += tl.dot(a, b).to(tl.float32)

    # Writeback.
    c_ptrs = c_ptr + pid_b * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Implements:
      out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    The heavy matrix multiplication is executed by a
    fused Triton kernel that treats (batch * hidden_dim) as a
    single batch dimension.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm on input (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # 2. Linear projections
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # 3. Apply mask if present
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # 4. Gates (left/right)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # 5. Output gate – keep in fp32 for later multiplication.
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    # 6. Prepare for batched GEMM.
    B, N, _, H = left.shape   # H == hidden_dim
    # Rearrange to [B*H, N, N] contiguous
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output buffer (fp32)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # 7. Launch Triton kernel.
    # Determine block sizes from the autotuned kernel (they will be supplied by the kernel)
    # We just need a grid.
    grid = (
        (N + 127) // 128,   # rows (BLOCK_M=128 assumed, fallback safe)
        (N + 127) // 128,   # cols
        B * H,
    )
    # The kernel will pick the best config based on the actual size; we only need to provide a grid.
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(0),
        stride_ak=left_flat.stride(1),
        stride_ak2=left_flat.stride(2),
        stride_bm=right_flat.stride(0),
        stride_bn=right_flat.stride(1),
        stride_bk=right_flat.stride(2),
        stride_cm=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cj=out_flat.stride(2),
        # No explicit BLOCK_* arguments – they are chosen by autotune.
        num_warps=8,
    )

    # 8. Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 9. Final LayerNorm, out‑gate, linear projection.
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])
    return output
```

"""
TriMul (outgoing) – fast Triton implementation.

The core operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication L @ Rᵀ for every hidden
channel d.  We fuse the heavy N³ work into a single Triton kernel that
treats the combined (batch * hidden_dim) dimension as the batch axis.
The rest of the TriMul pipeline (LayerNorm, linear projections,
gating, mask handling and the final projection) is performed with
standard PyTorch ops.  The kernel works on fp16 data and accumulates
in fp32 for numerical stability; an autotuned config list lets Triton
pick the best tile size (128/64‑wide with 8 warps on an H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A : (batch, M, K)  fp16
#   B : (batch, N, K)  fp16   (read as (K,N) to obtain the transpose)
#   C : (batch, M, N)  fp32
# ------------------------------------------------------------------------
@triton.autotune(
    configs=[
        # good for large N (≥ 512)
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 8},
            num_stages=3,
        ),
        # fallback for medium / small N
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 4},
            num_stages=3,
        ),
        # alternative with larger K tile
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 64, "BLOCK_K": 64, "num_warps": 8},
            num_stages=2,
        ),
        # mixed shape
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 4},
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_kernel(
    a_ptr,                     # *fp16  [B*H, M, K]
    b_ptr,                     # *fp16  [B*H, N, K]   (read as K×N)
    c_ptr,                     # *fp32  [B*H, M, N]
    M, N, K,                   # matrix dims (all equal to seq_len)
    stride_am, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bm, stride_bi, stride_bk,   # strides for B (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16×fp16→fp32 GEMM (A @ Bᵀ)."""
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_b = tl.program_id(2)          # batch*hidden

    # --------------------------------------------------------------------
    # Offsets of the current tile
    # --------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for the current row/col tiles (no K offset yet)
    a_tile_ptr = a_ptr + pid_b * stride_am + offs_i[:, None] * stride_ai
    b_tile_ptr = b_ptr + pid_b * stride_bm + offs_j[None, :] * stride_bi

    # --------------------------------------------------------------------
    # Accumulator in fp32
    # --------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K‑wide slices
    # --------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_tile_ptr + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile as (K, BLOCK_N) – this is the transposed view.
        b = tl.load(
            b_tile_ptr + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product (Tensor‑Core friendly)
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------------
    c_tile_ptr = (
        c_ptr
        + pid_b * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_tile_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ------------------------------------------------------------------------
# Python entry‑point used by the evaluator
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    Tensor
        Shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2) Linear projections → fp16 (the heavy part)
    # --------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------
    # 4) Gating (keep in fp16 to stay inside the kernel)
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"])
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"])
    ).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5) Output gate (float32 – applied after the matmul)
    # --------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM:
    #    [B, N, N, hidden_dim] → [B*hidden_dim, N, N] (contiguous)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------------
    # 7) Launch the Triton kernel.
    #    We build a grid that is safe for the smallest BLOCK size (64);
    #    kernels with larger tiles will simply see masked‑out work.
    # --------------------------------------------------------------------
    grid = (
        (N + 63) // 64,   # rows
        (N + 63) // 64,   # cols
        B * H,            # batch*hidden
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bm=right_flat.stride(0),
        stride_bi=right_flat.stride(1),
        stride_bk=right_flat.stride(2),
        stride_cm=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cj=out_flat.stride(2),
        num_warps=8,            # hint for the autotuned configs
    )

    # --------------------------------------------------------------------
    # 8) Reshape back to the original 4‑D layout
    # --------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------
    # 9) Final LayerNorm → out‑gate → linear projection to the original dim
    # --------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # bias is False in the model

    return output
