import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b] = A[b] @ B[b].T for many batches (B*H) where
    A,B have shape (BH, M, K) and (BH, N, K) respectively.
    A and B are FP16, C is accumulated in FP32 and written back as FP16.
    The kernel is optimized for H100:

    - 8 warps, 4 pipeline stages.
    - Loads use read‑only cache.
    """
    pid0 = tl.program_id(0)  # encodes batch_id * tiles_m + tile_m
    pid1 = tl.program_id(1)  # tile_n

    # ------------ decode batch and tile coordinates -------------
    tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // tiles_m
    tile_m = pid0 % tiles_m
    tile_n = pid1

    # ------------ compute offsets -------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for the first k‑block
    a_ptrs = a_ptr + batch_id * stride_az \
                       + offs_m[:, None] * stride_am \
                       + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                       + offs_k[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ k‑loop with pipelined prefetch ---------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_idx = cur_k + offs_k

        # out‑of‑bounds mask
        mask_a = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        mask_b = (k_idx[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, cache_modifier='cg')
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, cache_modifier='cg')

        # dot‑product (tiling over K)
        acc += tl.dot(a, b)  # fp16 ⋅ fp16 → fp32

        # advance pointers for the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------ write back result --------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

    The surrounding layer‑norms, linear projections and gates are executed
    in PyTorch for simplicity.  Input and all weight tensors are expected to
    be on the same CUDA device as the input."""

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm over `dim` ------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections (no bias) -----------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if present) ---------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Switch to FP16 for the N³ GEMM -------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # _ == hidden_dim

    # Reshape to (B*H, N, N) where H = hidden_dim.
    # left: (i,k) ; right: (j,k)  → need right transposed on the last two axes.
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (fp16, will be cast back later)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_N = 128

    grid_n = triton.cdiv(N, BLOCK_N)

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the TriMul operator is the three‑dimensional
contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication
C = A @ Bᵀ for every hidden dimension d (and every batch).  The
kernel below implements a fused, Tensor‑Core‑friendly GEMM for the
packed batch = B * hidden_dim.  All surrounding LayerNorms,
linear projections and gating are performed in PyTorch; only the
N³ contraction is off‑loaded to Triton.

Key optimisations
-----------------
* BLOCK_M = BLOCK_N = 128, BLOCK_K = 64 – a tile that maps well to
  the H100 Tensor‑Core MMA.
* 8 warps per CTA and 4 pipeline stages for deep pre‑fetching.
* Loads use the read‑only cache (``cache_modifier='cg'``) and
  out‑of‑bounds masking is only applied on the last partial tile.
* Accumulation is done in FP32 (the default of ``tl.dot`` for FP16
  inputs) and the result is written back as FP16 before being cast
  to FP32 in the surrounding PyTorch code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr,                     # *fp16  left  (batch*H, M, K)
    b_ptr,                     # *fp16  right (batch*H, N, K)  – we treat the K‑dimension as rows
    c_ptr,                     # *fp16  output (batch*H, M, N)
    M, N, K,                   # int32 matrix sizes (all = seq_len)
    stride_az, stride_am, stride_ak,   # int32 strides for A
    stride_bz, stride_bk, stride_bn,   # int32 strides for B
    stride_cz, stride_cm, stride_cn,   # int32 strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:  C = A @ Bᵀ   where
        A : (batch*H, M, K)   (fp16)
        B : (batch*H, N, K)   (fp16)   – we load it as (K,N) so the
                                      dot does the transpose implicitly.
    The kernel works on a 2‑D grid:
        program_id(0) → combined (batch*H) * tile_m
        program_id(1) → tile_n
    """
    pid0 = tl.program_id(0)  # encodes batch_id * tiles_m + tile_m
    pid1 = tl.program_id(1)  # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // tiles_m
    tile_m   = pid0 %  tiles_m
    tile_n   = pid1

    # -------------------------------------------------------------
    # Offsets within the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # [BLOCK_K]

    # -------------------------------------------------------------
    # Pointers to the start of the tile (no k‑offset yet)
    # -------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                   + offs_m[:, None] * stride_am \
                   + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                   + offs_k[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop – each iteration processes a BLOCK_K‑wide slice
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_idx = cur_k + offs_k

        # Masks for the trailing partial tiles (only needed on the
        # last iteration, but cheap to compute every time)
        mask_a = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        mask_b = (k_idx[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0,
                    cache_modifier='cg')   # read‑only cache
        b = tl.load(b_ptrs, mask=mask_b, other=0.0,
                    cache_modifier='cg')

        # fp16·fp16 → fp32 accumulation (Tensor‑Core MMA)
        acc += tl.dot(a, b)

        # Advance pointers for the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Store the result (back to FP16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N]  (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with at least {"dim": int,
                                             "hidden_dim": int,
                                             "nomask": bool}
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension `dim`
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid after linear)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Switch to FP16 for the N³ GEMM
    # -----------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape                 # _ == hidden_dim

    # Rearrange tensors to (B, H, N, N) and flatten the first two axes
    # -> shape (B*H, N, N).  `right` is transposed on the last two axes
    # so that the GEMM we launch computes A @ Bᵀ automatically.
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer for the GEMM (fp16, later cast back to fp32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6️⃣ Triton launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # larger K‑tile exploits H100 Tensor‑Core better

    grid_m = triton.cdiv(N, BLOCK_M)      # tiles along M per batch*H
    grid_n = triton.cdiv(N, BLOCK_N)      # tiles along N
    # pid0 = batch_id * grid_m + tile_m
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_gemm_outgoing_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps per CTA – good utilisation on H100
        num_stages=4,         # double‑buffered pipeline
    )

    # -----------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 8️⃣ Output LayerNorm, gating, final projection to `dim`
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                       # element‑wise out‑gate

    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim) → dim
    return out
