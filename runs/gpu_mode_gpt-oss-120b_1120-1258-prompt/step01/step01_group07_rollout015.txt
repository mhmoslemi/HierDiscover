def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)

    # linear projections (cast to fp16)
    left = F.linear(x, weights['left_proj.weight']).to(torch.float16)
    right = F.linear(x, weights['right_proj.weight']).to(torch.float16)

    # mask
    if use_mask:
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight'])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight'])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))  # keep fp32

    left = left * left_gate
    right = right * right_gate

    # transpose to group dimension using permute (no copy)
    left_t = left.permute(0, 3, 1, 2)   # B,H,N,N (noncontiguous)
    right_t = right.permute(0, 3, 2, 1) # B,H,N,N (noncontiguous)

    # kernel dimensions
    B = left_t.shape[0]
    H = left_t.shape[1]
    N = left_t.shape[2]  # N
    assert left_t.shape[3] == N

    # Compute strides
    stride_ag = left_t.stride(0)   # batch stride
    stride_ah = left_t.stride(1)   # head stride (likely 1)
    stride_am = left_t.stride(2)   # row stride (i)
    stride_ak = left_t.stride(3)   # col stride (k)

    stride_bg = right_t.stride(0)
    stride_bh = right_t.stride(1)
    stride_bk = right_t.stride(2)  # row stride (k)
    stride_bn = right_t.stride(3)  # col stride (j)

    # allocate output buffer as (B*H, N, N)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # Kernel block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, B * H)

    _batch_matmul_kernel[grid](
        left_t, right_t, C,
        N, N, N,
        stride_ag, stride_am, stride_ak,
        stride_bg, stride_bk, stride_bn,
        stride_cg, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HEAD_DIM=H,
        num_warps=8,
        num_stages=3,  # optional for prefetch
    )
    # reshape output
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=1e-5)

    out = out * out_gate

    output = F.linear(out, weights['to_out.weight'])

    return output
```

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The forward pass follows the AlphaFold3 “outgoing” version:

1. Layer‑norm the 4‑D input (B, N, N, C).
2. Two FP16 linear projections (left/right) from C → H.
3. Optional binary mask is applied in FP16.
4. Three sigmoid gates (left, right, out) are computed from the normalized
   input; the left/right gates are applied in‑place.
5. Core work:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h] .
   This is a batched GEMM (H × B independent N×N matrix multiplies).
   A custom Triton kernel is used that works directly on the original
   (B,N,N,H) layout, thereby avoiding the costly ``permute(...).contiguous()``
   copies required by a naïve implementation.  The kernel uses FP16 inputs,
   accumulates in FP32, and is tiled with 128×128×32 blocks (8 warps) which
   fully exploits H100 Tensor‑Cores.
6. The hidden result is Layer‑normed, multiplied by the out‑gate and finally
   projected back to the original channel dimension C.
All extra work (LayerNorm, Linear, sigmoid) is performed with highly‑optimised
PyTorch primitives; only the O(N³) contraction is hand‑written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_kernel(
    a_ptr,                     # left  tensor  (B,N,N,H)  fp16
    b_ptr,                     # right tensor  (B,N,N,H)  fp16
    c_ptr,                     # output tensor (B,N,N,H) fp32
    M, N, K,                   # all equal to seq_len
    stride_ab, stride_ah,      # A: batch‑stride, head‑stride
    stride_ai, stride_ak,      # A: row‑stride (i), col‑stride (k)
    stride_bb, stride_bh,      # B: batch‑stride, head‑stride
    stride_bk, stride_bj,      # B: row‑stride (k), col‑stride (j)
    stride_cb, stride_ch,      # C: batch‑stride, head‑stride
    stride_ci, stride_cj,      # C: row‑stride (i), col‑stride (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,   # launch‑time hint
):
    """Batched GEMM: C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,k,j,h].

    The kernel is launched with a 3‑D grid:
        (tiles over (M,N), batch, head)
    """
    pid   = tl.program_id(0)   # tile id in the (M,N) plane
    pid_b = tl.program_id(1)   # batch index  (0 ≤ pid_b < B)
    pid_h = tl.program_id(2)   # head  index  (0 ≤ pid_h < H)

    # ------------------------------------------------------------------
    # Tile coordinates in the output matrix (M × N)
    # ------------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n_tiles
    pid_n = pid %  num_n_tiles

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the current batch / head slice
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_b * stride_ab + pid_h * stride_ah
    b_base = b_ptr + pid_b * stride_bb + pid_h * stride_bh
    c_base = c_ptr + pid_b * stride_cb + pid_h * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (the summed dimension)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[i, k] tile
        a = tl.load(
            a_base
            + offs_i[:, None] * stride_ai   # i‑stride
            + offs_k[None, :] * stride_ak,  # k‑stride
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B[k, j] tile (notice swapped axes compared to A)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk   # k‑stride (row)
            + offs_j[None, :] * stride_bj,  # j‑stride (col)
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Tensor‑core dot (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result back (FP32)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask   = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (cast to FP16 for speed)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_fp16 = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))          # keep FP32

    # Apply left/right gates in‑place
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # left shape is (B, N, N, H)

    # Allocate output buffer (FP32 accumulation)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides for the kernels (as int64)
    stride_ab = left.stride(0)   # batch stride for A
    stride_ah = left.stride(3)   # head stride   for A (last dim)
    stride_ai = left.stride(1)   # i‑stride
    stride_ak = left.stride(2)   # k‑stride

    stride_bb = right.stride(0)  # batch stride for B
    stride_bh = right.stride(3)  # head stride  for B
    stride_bk = right.stride(2)  # k‑stride (row of B)
    stride_bj = right.stride(1)  # j‑stride (col of B)

    stride_cb = out.stride(0)    # batch stride for C
    stride_ch = out.stride(3)    # head stride  for C
    stride_ci = out.stride(1)    # i‑stride
    stride_cj = out.stride(2)    # j‑stride

    # ------------------------------------------------------------------
    # Tile sizes – chosen for H100 Tensor‑Core utilisation
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    # Grid: (tiles over M×N, batch, head)
    grid = (num_tiles_m * num_tiles_n, B, H)

    _batch_matmul_kernel[grid](
        left,
        right,
        out,
        N, N, N,                     # M, N, K
        stride_ab, stride_ah, stride_ai, stride_ak,
        stride_bb, stride_bh, stride_bk, stride_bj,
        stride_cb, stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # 8 warps → 128 threads per block
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
