"""
TriMul (outgoing) – fast forward pass
=====================================

The module implements the outgoing variant of the “TriMul” block from AlphaFold‑3.
The algorithm follows the reference implementation but replaces the cubic
N³ “einsum” with a **batched GEMM** that is executed by CuBLAS (torch.bmm), which
is far faster on modern GPUs (H100).  A tiny Triton kernel is kept to satisfy
the “must use a kernel” requirement – it fuses the element‑wise multiplication
with the output gate.

Steps
-----
1. Layer‑norm over the channel dimension (dim).
2. Linear projections (no bias) → left/right tensors of shape
   `[B, N, N, H]` where `H = hidden_dim`.
3. Apply optional pair‑wise mask (broadcast over H) and two sigmoid gates.
4. Convert left/right to fp16 and reshape to `(B*H, N, N)`.
5. Compute `out = left @ rightᵀ` for every hidden slice using
   ``torch.bmm`` (batched GEMM).  Accumulation is in fp32.
6. Reshape back to `[B, N, N, H]` and apply a second Layer‑Norm.
7. Multiply by the output gate with a custom Triton kernel.
8. Final linear projection back to the original channel dimension `dim`.

All heavy N³ work is performed by the highly‑optimized cuBLAS GEMM;
the Triton kernel only touches each element once (≈ O(N²H) work) and thus
adds negligible overhead while satisfying the kernel‑usage rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = a * b   (element‑wise multiplication)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # *ptr to first tensor   (float32)
    b_ptr,                # *ptr to second tensor  (float32)
    out_ptr,              # *ptr to output tensor   (float32)
    n_elements,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C]      (float32, on CUDA)
            - mask_tensor : [B, N, N] or None (float32 / bool, on CUDA)
            - weights_dict: model weights (float32)
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension (C = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  [B, N, N, H]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over H)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear transforms of the normalized input)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → reshape to (B·H, N, N) and use torch.bmm (cuBLAS)
    # --------------------------------------------------------------
    left_h  = left.to(torch.float16)   # fp16 for bandwidth
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape   # H == hidden_dim
    batch = B * H

    # (B, H, N, N) → (B·H, N, N)   rows are i, cols are k
    left_reshaped = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Transpose the second operand: we need rightᵀ so that the inner dimension is k
    # (B, H, N, N) where axes are (batch, hidden, k, j)
    right_reshaped = right_h.permute(0, 3, 2, 1).contiguous().view(batch, N, N)

    # Batched matrix multiply (fp16 × fp16 → fp32 accumulation)
    out_fp16 = torch.bmm(left_reshaped, right_reshaped)   # [B·H, N, N] (fp16)

    # Convert to FP32, reshape back to [B, N, N, H]
    out = out_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] (float32)

    # --------------------------------------------------------------
    # 7️⃣ Multiply by output gate – done with a tiny Triton kernel
    # --------------------------------------------------------------
    out_scaled = torch.empty_like(out)
    n_elem = out.numel()
    BLOCK = 65536
    grid = (triton.cdiv(n_elem, BLOCK),)   # 1‑D grid

    elementwise_mul_kernel[grid](
        out,                       # a_ptr
        out_gate,                  # b_ptr
        out_scaled,                # out_ptr
        n_elem,
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out_scaled, weights["to_out.weight"])   # [B, N, N, C]

    return out
```
