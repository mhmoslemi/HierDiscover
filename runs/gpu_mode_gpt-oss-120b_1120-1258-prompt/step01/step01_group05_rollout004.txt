"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the 4‑D tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) → L,R ∈ ℝ^{B×N×N×H}.
3. Optional pairwise mask M ∈ {0,1}^{B×N×N} is broadcasted on H.
4. Three element‑wise sigmoid gates (left, right, out) are computed from X.
   L ← L·σ(L_gate) ,   R ← R·σ(R_gate)
5. The expensive tensor contraction
       Out[b,i,j,h] = Σ_k L[b,i,k,h]·R[b,j,k,h]
   is a batch of matrix‑multiplications:
       L[b,:,:,h]  (N×N)  ×  R[b,:,:,h]ᵀ (N×N) .
   We fuse all B·H independent GEMMs in a single Triton kernel
   (no extra Python loops).  Inputs are FP16, accumulation is FP32.
6. Second Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension C.

The kernel is tuned for the H100 (Tensor‑cores) with BLOCK_M=N_BLOCK=64,
BLOCK_K=64.  It uses `tl.dot` which maps to WMMA/Tensor‑Core instructions.
All other cheap operations (layer‑norm, linear, sigmoid) are kept in
PyTorch which already uses highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_outgoing_kernel(
    A_ptr,               # *ptr left  (batch, M, K)   fp16
    B_ptr,               # *ptr right (batch, K, N)   fp16  (right is transposed)
    C_ptr,               # *ptr out   (batch, M, N)   fp32
    BATCH: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B  (k‑stride, n‑stride)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM with A @ Bᵀ for every hidden slice."""
    pid_batch = tl.program_id(2)          # batch (B·H)
    pid_m = tl.program_id(0)              # block row index
    pid_n = tl.program_id(1)              # block col index

    # -----------------------------------------------------------------
    # Block offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tile of A and B
        a_ptrs = (A_ptr + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # Load tiles (masked to avoid out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑accumulate (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write back the result
    # -----------------------------------------------------------------
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight name → torch.Tensor (float32)
            - config_dict  : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension C
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)          # [B, N, N, 1]

    left  = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections of the normalized input)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    -> left @ rightᵀ per hidden slice.
    # -----------------------------------------------------------------
    # Reduce memory traffic: cast to FP16 before the heavy multiplication.
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape                    # H == hidden_dim
    batch = B * H                                 # treat each (b,h) as an independent batch

    # Re‑order to (B, H, N, N) and fuse the first two dimensions.
    left_h  = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Output buffer (FP32 accumulator)
    out_h = torch.empty_like(left_h, dtype=torch.float32)

    # ---------------------------------------------------------------
    # Kernel launch configuration
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        batch                      # batch dimension (B·H)
    )

    # Strides for the three tensors (required by Triton)
    stride_Ab = left_h.stride(0)
    stride_Am = left_h.stride(1)
    stride_Ak = left_h.stride(2)

    # B is used as (batch, K, N) i.e. transposed version of right_h
    stride_Bb = right_h.stride(0)
    stride_Bk = right_h.stride(2)   # original column stride → K‑stride after transpose
    stride_Bn = right_h.stride(1)   # original row stride → N‑stride after transpose

    stride_Cb = out_h.stride(0)
    stride_Cm = out_h.stride(1)
    stride_Cn = out_h.stride(2)

    # Launch the Triton kernel
    _batched_gemm_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------------
    # Reshape back to the original 4‑D layout
    # ---------------------------------------------------------------
    out = out_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ---------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension H
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H]

    # ---------------------------------------------------------------
    # 7) Output gate
    # ---------------------------------------------------------------
    out = out * out_gate

    # ---------------------------------------------------------------
    # 8) Final projection back to the original channel dimension C
    # ---------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,C]

    return out
