import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward implementation with Triton‑accelerated batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias):
      L = X·W_left   (shape B×N×N×H)
      R = X·W_right  (shape B×N×N×H)
3. Optional pair‑wise mask M ∈ {0,1}^{B×N×N} is broadcasted to the H dimension.
4. Compute three sigmoid gates (also linear, no bias):
      G_L = σ(X·W_left_gate)
      G_R = σ(X·W_right_gate)
      G_O = σ(X·W_out_gate)
   Apply G_L and G_R to L and R respectively.
5. For every hidden slice h we need
      Out[...,i,j,h] = Σ_k L[...,i,k,h] * R[...,j,k,h] .
   This is a batch of matrix multiplications:
      reshape L,R → (B·H, N, N) and compute
      Out_batch = L_batch @ R_batchᵀ .
   The heavy N³ work is performed by a custom Triton kernel
   (batched GEMM with the K‑dimension summed).  Accumulation is in fp32
   while loads/stores use fp16 for bandwidth.
6. Reshape the result back to B×N×N×H and apply a second LayerNorm.
7. Multiply by the output gate G_O.
8. Final linear projection back to the original channel dimension C.

The kernel fuses the inner‑product loop over K and avoids any Python‑level
loops, giving a fast O(B·H·N²) GEMM per hidden slice.  The rest of the
operations are performed in PyTorch (LayerNorm, Linear, gating) which are
already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with transposed B.
# Computes:  C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (note the transpose on B)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,               # *ptr to left matrix  (batch, M, K)  fp16
    B_ptr,               # *ptr to right matrix (batch, K, N)  fp16 (originally (batch, N, K))
    C_ptr,               # *ptr to output matrix (batch, M, N) fp32
    BATCH, M, N, K,      # matrix sizes (int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (note: Bk is stride over K, Bn over N after transpose)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)              # block row index
    pid_n = tl.program_id(1)              # block col index

    # ----------------- block offsets -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile
        a_ptrs = (A_ptr + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # Load tiles with masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # fp16 -> will be cast to fp32 for dot
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)                # result accumulated in fp32

    # ----------------- write back --------------------
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights_dict: mapping from weight names to torch.Tensor (float32)
            - config_dict: must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. First LayerNorm on the input (over the channel dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided).  Mask broadcasted over hidden dim.
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor
    # Ensure mask dtype matches hidden tensors
    mask = mask.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Cast to half for bandwidth; accumulator stays fp32 inside kernel.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # H == hidden_dim
    batch = B * H

    # Rearrange to (batch, N, N) where batch = B*H
    left_reshaped = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # [B*H, N, N]
    right_reshaped = right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)  # [B*H, N, N]

    # Allocate output tensor (fp32 for accumulator precision)
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float32)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        batch                      # batch dimension (B*H)
    )

    # Strides (must be integers)
    stride_Ab = left_reshaped.stride(0)
    stride_Am = left_reshaped.stride(1)
    stride_Ak = left_reshaped.stride(2)

    # For B we need a view where the K dimension is the original last dim
    #   right_reshaped is (batch, N, N) where dim0=row, dim1=col, dim2=inner.
    #   The kernel expects B in layout (batch, K, N), i.e. we have to swap axes 1 and 2.
    stride_Bb = right_reshaped.stride(0)
    stride_Bk = right_reshaped.stride(2)   # stride over original K (last dim)
    stride_Bn = right_reshaped.stride(1)   # stride over original N (first after batch)

    stride_Cb = out_reshaped.stride(0)
    stride_Cm = out_reshaped.stride(1)
    stride_Cn = out_reshaped.stride(2)

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still [B,N,N,H]

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,C]

    return out
