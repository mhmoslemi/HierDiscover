"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` on its channel dimension.
2. Compute two linear projections (left/right) and three gating projections.
3. Apply the optional pairwise mask and the per‑position gates.
4. Fuse the heavy O(N³·H) “pairwise‑multiplicative” update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for every hidden slice.
   We launch a Triton kernel that works on the original `[B,N,N,H]`
   layout (no permutes, no extra copies). The kernel receives the exact
   strides of the tensors, merges the batch and hidden dimensions into the
   third program‑id, and performs FP16 matrix multiplication with FP32
   accumulation using `tl.dot`.
5. Layer‑norm the result, multiply by the output gate and project back to
   the original channel dimension.
The final output is returned as `float32`.
"""

import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict


@triton.jit
def _trimul_batched_matmul_kernel(
    # Pointers
    A, B, C,
    # Matrix sizes
    M, N, K,
    # Strides for tensor A (left projection) – shape [B, N, N, H]
    stride_Ab, stride_Ar, stride_Ac, stride_Ah,
    # Strides for tensor B (right projection) – same shape, but we treat B as K×N
    stride_Bb, stride_Br, stride_Bc, stride_Bh,
    # Strides for output tensor C – shape [B, N, N, H]
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    # Compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # Hidden dimension (required to split the merged batch‑hidden id)
    H: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ for tensors shaped (B,N,N,H).

    A has logical shape (M, K) = (N, N)   – row stride = stride_Ar, col stride = stride_Ac
    B is interpreted as (K, N) = (N, N)   – row stride = stride_Br (k‑stride), col stride = stride_Bc (j‑stride)
    C has shape (M, N) = (N, N)           – row stride = stride_Ci, col stride = stride_Cj
    The third program id encodes (batch, hidden) as pid_bh = b*H + h.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_bh = tl.program_id(2)  # merged (batch, hidden)

    # Recover batch and hidden indices
    b = pid_bh // H
    h = pid_bh - b * H

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    # A(b, i, k, h)   – row i (M‑dim), column k (K‑dim)
    a_ptrs = (
        A
        + b * stride_Ab
        + h * stride_Ah
        + offs_m[:, None] * stride_Ar
        + offs_k[None, :] * stride_Ac
    )
    # B(b, j, k, h) interpreted as B(k, j)   – row k, column j
    b_ptrs = (
        B
        + b * stride_Bb
        + h * stride_Bh
        + offs_k[:, None] * stride_Br   # move along k (row of B)
        + offs_n[None, :] * stride_Bc   # move along j (col of B)
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)                     # FP16·FP16 → FP32 accumulate
        a_ptrs += BLOCK_K * stride_Ac           # advance K‑tile in A (col stride)
        b_ptrs += BLOCK_K * stride_Br           # advance K‑tile in B (row stride)

    # ------------------------------------------------------------------
    # Write result – C(b, i, j, h)  (row i, col j)
    # ------------------------------------------------------------------
    c_ptrs = (
        C
        + b * stride_Cb
        + h * stride_Ch
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32)
            - mask: [B, N, N] (bool/float) or None
            - weights: dictionary with all linear / norm parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # shapes: [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # all: [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Heavy pairwise‑multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # Cast to half‑precision to reduce bandwidth; accumulation stays FP32.
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = left_f16.shape  # B, N, N, hidden

    # Allocate output tensor in FP16 (same layout as inputs)
    out_f16 = torch.empty_like(left_f16)  # [B, N, N, hidden]

    # Tensor strides (int64) – required by Triton kernel
    stride_Ab = left_f16.stride(0)
    stride_Ar = left_f16.stride(1)
    stride_Ac = left_f16.stride(2)
    stride_Ah = left_f16.stride(3)

    # For the right tensor we treat it as (K, N) inside the kernel:
    stride_Bb = right_f16.stride(0)
    stride_Br = right_f16.stride(2)  # stride for K (the third dimension)
    stride_Bc = right_f16.stride(1)  # stride for N (the second dimension)
    stride_Bh = right_f16.stride(3)

    stride_Cb = out_f16.stride(0)
    stride_Ci = out_f16.stride(1)
    stride_Cj = out_f16.stride(2)
    stride_Ch = out_f16.stride(3)

    # Block sizes – tuned for H100/Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,
    )

    _trimul_batched_matmul_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        N,                # M
        N,                # N
        N,                # K
        stride_Ab,
        stride_Ar,
        stride_Ac,
        stride_Ah,
        stride_Bb,
        stride_Br,
        stride_Bc,
        stride_Bh,
        stride_Cb,
        stride_Ci,
        stride_Cj,
        stride_Ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=hidden,
        num_warps=8,        # leverage tensor‑core warps
    )

    # Convert back to FP32 for the remaining layers
    out = out_f16.to(torch.float32)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + out‑gate + final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of speed‑up**

* **No layout permutations / copies** – the kernel works directly on the original `[B,N,N,H]` memory layout.  
* **Merged batch × hidden dimension** – a single Triton launch handles all `B·H` independent GEMMs, keeping the GPU fully occupied.  
* **FP16 arithmetic with FP32 accumulation** – halves memory traffic while preserving numerical stability.  
* **Tensor‑core friendly blocking (128×128×32) and `num_warps=8`** – gives high occupancy on H100.  

All other lightweight operations (layer‑norm, linear, gating) remain in PyTorch, which is already highly optimized for those shapes. The function complies with the required signature and returns a float‑32 tensor of shape `[B, seq_len, seq_len, dim]`.
