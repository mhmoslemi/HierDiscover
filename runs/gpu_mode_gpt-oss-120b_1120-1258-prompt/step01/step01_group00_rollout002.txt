    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cn,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)

"""
custom_kernel
==============

Forward pass of the *outgoing* TriMul module (AlphaFold‑3 style) using a
hand‑written Triton batched GEMM.

Algorithm
---------
1. **Layer‑norm** over the last dimension of the input (dim).
2. **Linear projections** → left / right (shape [B, N, N, H]).
3. Optional **mask** applied to the projected tensors.
4. **Gates** (sigmoid of three linear layers) and element‑wise gating of
   left / right.
5. **Tri‑multiplicative reduction**  
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
   This is a batched matrix multiply for each hidden slice:
   `C = A @ Bᵀ` with A,B ∈ ℝ^{B·H × N × N}.  The heavy N³ work is performed
   by a fused Triton kernel that tiles the GEMM (BLOCK_M×BLOCK_N×BLOCK_K)
   and accumulates in FP32 while loading A and B in FP16.
6. **Layer‑norm** on the hidden dimension, gating with `out_gate`,
   and a final linear projection back to the original `dim`.

Only the reduction step (5) is implemented in Triton; the remaining
operations use highly‑optimized PyTorch kernels.  The kernel treats the
combined batch‑hidden dimension as a single batch index, which removes an
extra hidden‑stride indirection and yields a compact, high‑throughput
implementation suitable for H100 (≈ 100 TFLOP FP16 peak).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the Tri‑multiplicative reduction.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_batch_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # matrix shapes (all = seq_len)
    stride_am, stride_ai, stride_ak,   # A strides: batch, row, col
    stride_bm, stride_bk, stride_bn,   # B strides: batch, row(K), col(N)
    stride_cm, stride_ci, stride_cn,   # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for each batch slice.
    A : (batch, M, K)  fp16
    B : (batch, N, K)  fp16    (we access it as K×N)
    C : (batch, M, N)  fp32
    """
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    pid_z = tl.program_id(2)      # batch (flattened B×H)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_base = a_ptr + pid_z * stride_am
    b_base = b_ptr + pid_z * stride_bm
    c_base = c_ptr + pid_z * stride_cm

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- tiled K loop ------------------------------------------------------ #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K × N) – note B is stored as (K,N)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)   # a,b are fp16 → acc is fp32

    # Write back the result
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


# --------------------------------------------------------------------------- #
# Entry point for the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, S, S, C]      (float32)
            - mask_tensor : [B, S, S] or None
            - weights_dict: all linear / layer‑norm parameters
            - config_dict : {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Load weights (all already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1️⃣ Layer‑norm over the channel dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  (FP32)

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias) → left / right  (shape [B,S,S,H])
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)    # [B, S, S, H]
    right = F.linear(x, right_proj_weight)   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of three linear layers)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Batched GEMM (Tri‑multiplicative reduction) using Triton
    # ------------------------------------------------------------------- #
    # Cast to half‑precision for bandwidth‑efficiency
    left_h  = left.to(torch.float16).contiguous()   # [B, S, S, H]
    right_h = right.to(torch.float16).contiguous()  # [B, S, S, H]

    # Rearrange layout to (B, H, S, S) → memory‑contiguous across the hidden dim
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, S, S]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()  # [B, H, S, S]

    # Transpose right so we can read it as (K, N) inside the kernel
    right_h = right_h.permute(0, 1, 3, 2).contiguous()  # [B, H, S, S]  (K,N)

    # Flatten the (batch, hidden) dimensions → single batch index for the kernel
    left_flat   = left_h.view(B * hidden_dim, S, S)   # [B*H, M, K]
    right_flat  = right_h.view(B * hidden_dim, S, S)  # [B*H, K, N]
    out_flat    = torch.empty_like(left_flat, dtype=torch.float32)  # [B*H, M, N]

    # ------------------------------------------------------------------- #
    # Kernel launch configuration (tuned for H100)
    # ------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows tiles
        triton.cdiv(S, BLOCK_N),          # cols tiles
        B * hidden_dim,                   # batch × hidden
    )

    _trimul_gemm_batch_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=S,
        N=S,
        K=S,
        stride_am=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bm=right_flat.stride(0),
        stride_bk=right_flat.stride(1),   # K dimension stride
        stride_bn=right_flat.stride(2),   # N dimension stride
        stride_cm=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good default for 128×128 tiles on H100
        num_stages=3,         # pipeline depth
    )

    # ------------------------------------------------------------------- #
    # 6️⃣ Restore original layout: [B, S, S, H]
    # ------------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 7️⃣ Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 8️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, S, S, dim]

    return out
