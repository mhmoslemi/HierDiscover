"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm over the channel dimension (C).
2. Two linear projections (left / right) to the hidden dimension H (no bias).
3. Optional pair‑wise mask (broadcasted over H) and two sigmoid gates
   (left_gate, right_gate).  All three tensors are cast to fp16 and fused.
4. Batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]    (k‑axis summed)
   This is a GEMM with the second operand transposed.  We launch a single
   Triton kernel that processes the whole batch B·H as independent matrix
   multiplies of size N×N.  The kernel works on fp16 inputs, accumulates in
   fp32 (using tl.dot(..., out_dtype=tl.float32)) and writes fp32 results.
   Block sizes are chosen adaptively, and the launch uses 8 warps (the
   sweet‑spot for H100 Tensor‑Cores).
5. Layer‑norm over the hidden dimension H, multiplicative output‑gate and a
   final linear projection back to the original channel dimension C.
All heavy arithmetic stays on the GPU; only light‐weight operations (norm,
gates, final linear) are delegated to PyTorch which already uses cuBLAS /
cuDNN kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with transposed B.
# Computes C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]    (B is read transposed)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                     # *ptr to A (batch, M, K)   fp16
    B_ptr,                     # *ptr to B (batch, K, N)   fp16 (originally (batch, N, K))
    C_ptr,                     # *ptr to C (batch, M, N)   fp32
    BATCH, M, N, K,            # static sizes
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (note Bk = stride over K, Bn over N after transpose)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)              # block row index
    pid_n = tl.program_id(1)              # block col index

    # ------------------------------------------------------------------
    # 1. Compute block-relative offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2. Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. Loop over K dimension (the sum axis)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tile
        a_ptrs = (A_ptr + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)

        b_ptrs = (B_ptr + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # Load tiles (with masking)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16

        # Multiply‑accumulate (accumulate directly in fp32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4. Write back the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C], dtype float32
            - mask_tensor : torch.Tensor [B, N, N] (float32, bool or None)
            - weights_dict: mapping name -> torch.Tensor (float32)
            - config_dict : contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm over channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                     # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcasted) – cast to fp16 once
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    mask = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – also cast to fp16 for the element‑wise multiplies
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    # Cast tensors that will be multiplied together to fp16
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)
    mask_h = mask.to(torch.float16)

    # Apply gates and mask in fp16 (keeps bandwidth low)
    left_h = left_h * left_gate_h * mask_h
    right_h = right_h * right_gate_h * mask_h

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM (out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h])
    #      Treat each hidden slice as an independent batch element.
    # ------------------------------------------------------------------
    B, N, _, H = left_h.shape                 # H == hidden_dim
    batch = B * H                               # total GEMM batch size

    # Rearrange to (batch, N, N) where batch = B*H
    #   left:  [B, N, N, H] -> [B*H, N, N]
    left_reshaped = left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_reshaped = right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Output buffer (fp32 accumulator)
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 5.1 Choose block sizes based on N (heuristic for H100)
    # ------------------------------------------------------------------
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # up to 1024 in the benchmarks
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Grid: (row_blocks, col_blocks, batch)
    grid = (
        triton.cdiv(N, BLOCK_M),   # block rows  (i dimension)
        triton.cdiv(N, BLOCK_N),   # block cols  (j dimension)
        batch                      # batch dimension (B*H)
    )

    # Strides (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = left_reshaped.stride()
    stride_Bb, stride_Bk, stride_Bn = (right_reshaped.stride(0),
                                      right_reshaped.stride(2),   # K‑axis (original inner N)
                                      right_reshaped.stride(1))   # N‑axis (original outer N)
    stride_Cb, stride_Cm, stride_Cn = out_reshaped.stride()

    # ------------------------------------------------------------------
    # 5.2 Launch kernel (8 warps → best occupancy on H100)
    # ------------------------------------------------------------------
    batched_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                     # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣  Apply output gate (still fp32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣  Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,C]

    return out
