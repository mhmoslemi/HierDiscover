def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    right_gate = torch.sigmoid(gate_logits[:, hidden_dim:2*hidden_dim])
    out_gate = torch.sigmoid(gate_logits[:, 2*hidden_dim:])

    # Apply elementwise gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for TriMul
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Permute for batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (j,k)
    BH = B * hidden_dim

    right_mat = right_perm.view(BH, N, N) # [BH, N, N]

    out_mat = torch.bmm(left_mat, right_mat)   # half

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # LayerNorm2 (float32)
    # Cast to float32 for LN2
    out_fp32 = out.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_fp32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)

        out_flat,
        out_norm_flat,

        hidden_dim,
        eps,

    out_proj = F.linear(out_norm_h, to_out_w_h)  # half

    final = out_proj.to(torch.float32)
    return final

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------

1️⃣ Layer‑norm the input tensor over the channel dimension (`dim`).  
   A lightweight Triton kernel does a row‑wise mean/var reduction and
   writes the normalized result (float32).

2️⃣ Cast the normalized tensor to FP16 – all subsequent work (linear
   projections, gating, the N³ triangular update and the final
   projection) runs in half‑precision and therefore exploits Tensor‑Core
   throughput.

3️⃣ Mask handling – when a mask is present we pre‑multiply the
   FP16‑normalized tensor by the mask (shape [B,N,N,1]) before the
   two linear projections.  This is mathematically equivalent to the
   PyTorch implementation which applies the mask after projection.

4️⃣ Linear projections & gates – we fuse the three gate linear layers
   (`left_gate`, `right_gate`, `out_gate`) into a single mat‑mul by
   concatenating their weight matrices.  The two projection layers
   (`left_proj`, `right_proj`) stay separate because they need the
   masked input.  All mat‑muls use `torch.nn.functional.linear` in
   FP16.

5️⃣ Apply element‑wise gates:
       left  = left_proj  * sigmoid(left_gate)
       right = right_proj * sigmoid(right_gate)

6️⃣ Triangular multiplicative update (the O(N³) step)  
   For each hidden channel we need `L @ Rᵀ`.  By permuting the tensors we
   obtain two batched matrices of shape `[B*H, N, N]` and call
   `torch.bmm` (FP16).  This runs as a highly‑optimised GEMM on an H100.

7️⃣ Second Layer‑norm over the hidden dimension (`hidden_dim`).  
   The output of the GEMM is cast back to FP32 and normalised with the
   same Triton kernel used in step 1.

8️⃣ Multiply by the output gate, project back to `dim` with a final
   FP16 linear layer and cast the result to FP32.

The kernel implements the two LayerNorms; all heavy linear algebra is
performed by fused PyTorch kernels in half‑precision, giving a large
speed‑up over the reference implementation while preserving numerical
accuracy (final result is float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                # *const float*   [total_rows, dim]
    y_ptr,                # *mut   float*   [total_rows, dim]
    weight_ptr,           # *const float*   [dim]
    bias_ptr,             # *const float*   [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (2‑D view)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # row mask

    # -----------------------------------------------------------------
    # First pass – compute mean and variance
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # FP32 accumulator
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass – write normalized values
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()                     # [B, N, N, dim]
    B, N, _, _ = x.shape
    total_rows = B * N * N

    x_flat = x.view(total_rows, dim)                  # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)            # buffer for LN output

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)            # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Cast to FP16 – everything after this runs in half‑precision
    # -----------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Mask handling (pre‑multiply the normalized tensor)
    # -----------------------------------------------------------------
    use_mask = not config.get("nomask", True) and mask_tensor is not None
    if use_mask:
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).contiguous().view(total_rows, 1)
        x_masked_h = x_norm_h * mask_h
    else:
        x_masked_h = x_norm_h

    # -----------------------------------------------------------------
    # 4) Prepare FP16 weight tensors
    # -----------------------------------------------------------------
    left_proj_w_h   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w_h  = weights["right_proj.weight"].to(torch.float16)

    # Concatenate the three gate weight matrices (out_features stacked)
    gate_w_h = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [(3H), dim]

    to_out_w_h = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 5) Linear projections (masked) – half‑precision
    # -----------------------------------------------------------------
    left_proj_h = F.linear(x_masked_h, left_proj_w_h)   # [R, H]
    right_proj_h = F.linear(x_masked_h, right_proj_w_h) # [R, H]

    # -----------------------------------------------------------------
    # 6) Gates – single mat‑mul (unmasked input)
    # -----------------------------------------------------------------
    gate_logits_h = F.linear(x_norm_h, gate_w_h)        # [R, 3H]
    left_gate_h  = torch.sigmoid(gate_logits_h[:, :hidden_dim])
    right_gate_h = torch.sigmoid(gate_logits_h[:, hidden_dim:2*hidden_dim])
    out_gate_h   = torch.sigmoid(gate_logits_h[:, 2*hidden_dim:])

    # -----------------------------------------------------------------
    # 7) Apply element‑wise gates
    # -----------------------------------------------------------------
    left_h  = left_proj_h * left_gate_h                # [R, H]
    right_h = right_proj_h * right_gate_h

    # -----------------------------------------------------------------
    # 8) Reshape for the triangular update
    # -----------------------------------------------------------------
    left_h  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_h = right_h.view(B, N, N, hidden_dim)  # [B, N, N, H]
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)

    # Permute to obtain batch×hidden leading dimension
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k, j)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)   # [B*H, N, N]

    # -----------------------------------------------------------------
    # 9) Triangular multiplicative update (batched GEMM, FP16)
    # -----------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] – half precision

    # Back to 4‑D tensor [B, N, N, H]
    out_h = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 10) Second LayerNorm (float32)
    # -----------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_fp32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm_fp32 = out_norm_flat.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 11) Final gating + projection back to `dim`
    # -----------------------------------------------------------------
    out_gate_fp32 = out_gate_h.to(torch.float32)
    out_norm_fp32 = out_norm_fp32 * out_gate_fp32   # element‑wise

    # Cast to half for the final linear layer
    out_norm_h = out_norm_fp32.to(torch.float16)
    out_flat_h = out_norm_h.view(total_rows_h, hidden_dim)

    final_h = F.linear(out_flat_h, to_out_w_h)      # [R, dim]  (half)
    final = final_h.to(torch.float32).view(B, N, N, dim)

    return final
```
