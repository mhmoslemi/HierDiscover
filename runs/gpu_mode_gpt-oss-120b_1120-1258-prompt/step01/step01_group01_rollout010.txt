```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # 1. LayerNorm with float32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # cast to half for rest
    x_h = x.to(dtype)

    # cast weight tensors to half
    left_proj_w_h = weights["left_proj.weight"].to(dtype)
    right_proj_w_h = weights["right_proj.weight"].to(dtype)
    left_gate_w_h = weights["left_gate.weight"].to(dtype)
    right_gate_w_h = weights["right_gate.weight"].to(dtype)
    out_gate_w_h = weights["out_gate.weight"].to(dtype)
    to_out_w_h = weights["to_out.weight"].to(dtype)
    to_out_norm_w_h = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b_h = weights["to_out_norm.bias"].to(dtype)

    # 2. Projections
    left = torch.nn.functional.linear(x_h, left_proj_w_h)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w_h)

    # 3. Mask
    if not nomask and mask is not None:
        mask_h = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # 4. Gates
    left_gate = torch.nn.functional.linear(x_h, left_gate_w_h).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w_h).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_w_h).sigmoid()

    # 5. Apply gates (fused)
    left = left * left_gate
    right = right * right_gate

    # 6. Heavy multiplication via torch.bmm or Triton depending on N
    B, N, _, H = left.shape   # left shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()      # (B, H, N, N) i,k
    right_perm = right.permute(0,3,2,1).contiguous()    # (B, H, K, J)
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # choose compute path
    # if N <= 512 use Triton matmul else use torch.bmm
    if N <= 512:
        # use triton kernel
        out_flat = torch.empty_like(left_flat)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        grid = ( (N + BLOCK_M - 1)//BLOCK_M,
                 (N + BLOCK_N - 1)//BLOCK_N,
                 BH )
        _batched_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N, N, N,
            left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
            right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
            out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
    else:
        out_flat = torch.bmm(left_flat, right_flat)   # both half

    # reshape
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()   # B,N,N,H half

    # 7. Output layer norm on hidden dim (half)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w_h,
        bias=to_out_norm_b_h,
        eps=1e-5,
    )
    # 8. Multiply with out_gate
    out = out * out_gate

    # 9. Linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w_h)

    # Cast to float32 for final output
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward pass
------------------------------------
Algorithm
---------
1. Layer‑norm over the channel dimension (float32, keeps the original
   statistics).
2. Cast to FP16 and apply three bias‑free linear projections:
      left  = x @ W_left
      right = x @ W_right
   and three gating projections (sigmoid) – all in FP16.
3. Optional pairwise mask is broadcast and applied to the projections.
4. Multiply each projection by its gate (still FP16).
5. Heavy “pairwise‑multiplicative” update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply C = A @ Bᵀ where each hidden slice
   (h) together with the batch (b) forms an independent matrix‑multiply.
   For small sequence lengths we use a custom Triton kernel; for larger
   lengths we delegate to cuBLAS via ``torch.bmm`` (Tensor‑Core, FP16
   accumulation in FP32).  The kernel works on FP16 inputs and accumulates
   in FP32, storing the result back in FP16.
6. Reshape back to [B, N, N, H] (still FP16) and apply a second
   Layer‑norm (FP16) with its own learned scale/bias.
7. Multiply by the output‑gate, project back to the original channel
   dimension (FP16 linear) and finally cast to float32.

The implementation keeps the heavy O(N³·H) part on the GPU, fuses the
mask/gating point‑wise work in FP16, and falls back to a highly‑optimized
cuBLAS batched GEMM when the problem size exceeds the Triton kernel’s
sweet spot.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,   # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,   # batch, row, col strides for B
    stride_Cb, stride_Cr, stride_Cc,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 matmul C = A @ B where each batch slice has shape (M, K)
    and (K, N). Accumulates in FP32 and writes out FP16."""
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # fused batch (B * H)

    # --------------------------------------------------------------
    # Offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the M and N dimensions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for A and B (k == 0)
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ar
    b_ptrs = B + pid_b * stride_Bb + offs_n[None, :] * stride_Bc

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K sized chunks
    # --------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # absolute offsets for the current K‑block
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        # masks for the K dimension
        mask_k = offs_k < K

        # Load a tile of A (shape BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptrs + k_offset * stride_Ac,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B (shape BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptrs + k_offset * stride_Br,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product (promoted to FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result back (FP16)
    # --------------------------------------------------------------
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data (Tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of weight tensors required by the module
            - config: dict containing keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. First LayerNorm (float32) – keeps numerical stability
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] (fp32)

    # Cast everything after the first LN to FP16
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast model weights to FP16 (except the first LN which stayed FP32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (FP16, no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Optional pairwise mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Gating tensors (FP16)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply gates (point‑wise FP16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Heavy pairwise‑multiplicative update:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left/ right shapes: [B, N, N, hidden]

    # Re‑order to (B, H, N, N) – left: (i,k), right: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J) where K == N

    BH = B * H                                          # merged batch dimension
    left_flat = left_perm.view(BH, N, N)                # (BH, N, N)
    right_flat = right_perm.view(BH, N, N)              # (BH, N, N)

    # ------------------------------------------------------------------
    # Choose the most efficient path for the batched GEMM
    # ------------------------------------------------------------------
    # Small N → Triton kernel (avoids cuBLAS kernel launch overhead)
    # Large N → cuBLAS via torch.bmm (Tensor‑Core, highly optimized)
    if N <= 256:
        out_flat = torch.empty_like(left_flat)          # fp16 buffer

        # Tuned block sizes for N ≤ 256 on H100
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            BH,
        )

        _batched_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,               # M
            N,               # N
            N,               # K
            left_flat.stride(0),
            left_flat.stride(1),
            left_flat.stride(2),
            right_flat.stride(0),
            right_flat.stride(1),
            right_flat.stride(2),
            out_flat.stride(0),
            out_flat.stride(1),
            out_flat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    else:
        # Use cuBLAS batched GEMM (FP16 × FP16 → FP32 accumulation)
        out_flat = torch.bmm(left_flat, right_flat)     # (BH, N, N)

    # Reshape back to original 4‑D layout (still FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8. Second LayerNorm (FP16) + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` (FP16) and cast to FP32
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)
    return out.to(torch.float32)
