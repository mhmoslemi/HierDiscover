"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm on the last dimension of the input tensor (dim = C).
2. Four linear projections (no bias): left_proj, right_proj, left_gate, right_gate,
   out_gate.  The gate projections are passed through a sigmoid.
3. Apply optional pairwise mask (broadcasted) and fuse the gate values:
      left  = (left_proj  * left_gate ) * mask
      right = (right_proj * right_gate) * mask
   The result is kept in FP16 to minimise memory traffic.
4. Compute the N³ reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply (left @ rightᵀ) for every hidden slice.
   A custom Triton GEMM kernel tiles the multiplication (BLOCK_M×BLOCK_N×BLOCK_K)
   and accumulates in FP32.
5. Layer‑norm on the hidden dimension, element‑wise gating with out_gate,
   and a final linear projection back to the original dimension.

The kernel works on H100 (SM‑90) with mixed‑precision (FP16 → FP32) and
exposes a single entry point `custom_kernel` that matches the required
signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes (all equal to seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for every (batch, hidden) pair.
    A : (batch, hidden, M, K)          – fp16
    B : (batch, hidden, N, K)          – fp16  (interpreted as (K, N) inside the kernel)
    C : (batch, hidden, M, N)          – fp32
    """
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # --------------------------------------------------------------------- #
    #  Decode batch and hidden indices.
    # --------------------------------------------------------------------- #
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # Base pointers for this slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # --------------------------------------------------------------------- #
    #  Tile offsets within the matrix.
    # --------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #  K‑loop (tiling).
    # --------------------------------------------------------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K × N) – note we treat B as (K, N)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------- #
    #  Write the result.
    # --------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor:  [B, S, S]  (bool / float) or None
            - weights_dict: all linear / layer‑norm weights (already on device)
            - config_dict: {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Extract weight tensors (all are float32 and already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  (float32)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free) and gate generations
    # ------------------------------------------------------------------- #
    # Projections
    left_proj  = F.linear(x, left_proj_weight)      # [B, S, S, H]
    right_proj = F.linear(x, right_proj_weight)     # [B, S, S, H]

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3. Fuse mask (if present) and elementwise gating, then cast to FP16
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left_proj  = left_proj  * mask
        right_proj = right_proj * mask

    # Fuse gating
    left  = (left_proj  * left_gate).to(torch.float16)   # [B, S, S, H] FP16
    right = (right_proj * right_gate).to(torch.float16) # [B, S, S, H] FP16

    # ------------------------------------------------------------------- #
    # 4. Rearrange for the GEMM kernel: [B, H, S, S] (hidden dimension first)
    # ------------------------------------------------------------------- #
    left_h  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, S, S] FP16
    right_h = right.permute(0, 3, 1, 2).contiguous()  # [B, H, S, S] FP16

    # ------------------------------------------------------------------- #
    # 5. Batched GEMM (TriMul reduction) via Triton
    # ------------------------------------------------------------------- #
    out_fp32 = torch.empty(
        (B, hidden_dim, S, S), dtype=torch.float32, device=device
    )   # accumulator buffer (FP32)

    # Tuning parameters – these work well on H100 for the given size range.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    NUM_WARPS = 4   # 4 warps ≈ 64×64 tile

    grid = (
        triton.cdiv(S, BLOCK_M),            # rows (i dimension)
        triton.cdiv(S, BLOCK_N),            # cols (j dimension)
        B * hidden_dim,                     # flattened (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),
        stride_ah=left_h.stride(1),   # hidden stride (==1 after permute)
        stride_ai=left_h.stride(2),
        stride_ak=left_h.stride(3),
        stride_ba=right_h.stride(0),
        stride_bh=right_h.stride(1),
        stride_bk=right_h.stride(3),  # k‑dimension of B (the “second” S)
        stride_bn=right_h.stride(2),  # j‑dimension of B (the “first” S)
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        NUM_WARPS=NUM_WARPS,
    )

    # ------------------------------------------------------------------- #
    # 6. Restore original layout: [B, S, S, H]
    # ------------------------------------------------------------------- #
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 7. Hidden‑dim layer‑norm, gating with out_gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H] (float32)

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, to_out_weight)       # back to original dim → [B, S, S, dim]

    return out
```
