    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    right = right * right_gate

    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    BLOCK_N = 128
    BLOCK_K = 64

        stride_rk=right_flat.stride(2),

        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        # optionally set num_stages for pipelining
    )

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) – highly‑optimized Triton kernel.

The core of the outgoing TriMul is the batched matrix multiplication
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
For every hidden channel *h* we compute a matrix product L @ Rᵀ where
L,R ∈ ℝ^{N×N}.  By reshaping the tensors to [B*H, N, N] we obtain a
simple batched GEMM that can be executed with Tensor‑Core‐accelerated
`tl.dot`.  The kernel uses 128×128 tiles (BLOCK_M/N) and a K‑tile of
64, which matches the 16‑×‑16 Tensor‑Core granularity and yields high
occupancy on H100 (8 warps / block).  The data are stored in fp16 and
accumulated in fp32 for numerical stability.  All surrounding
operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch, but the O(N³) multiplication is off‑loaded
to the Triton kernel.

The implementation works for any batch size (B ∈ {1,2}), any sequence
length N (including non‑multiple‑of‑128 lengths – a small mask is used
for the border tiles), and any channel dimensions dim and hidden_dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,        # fp16 * left  [B*H, N, N]
    right_ptr,       # fp16 * right [B*H, N, N]
    out_ptr,         # fp32 * out   [B*H, N, N]
    M, N, K,         # all equal to seq_len
    stride_fa, stride_fi, stride_fk,   # left strides (batch_h, i, k)
    stride_rb, stride_ri, stride_rk,   # right strides (batch_h, j, k)
    stride_ob, stride_oi, stride_oj,   # out strides   (batch_h, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ   (A,B are fp16, C is fp32)
    A: [B*H, M, K]   B: [B*H, N, K]   C: [B*H, M, N]
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # which (batch*hidden) matrix

    # -------------------------------------------------------------
    # Tile offsets + masks
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile for A and B
    a_ptr = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_ptr + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (N×K) – we read it as (K,N) to perform A @ Bᵀ
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b).to(tl.float32)

    # Write the result tile
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask_tensor : [B, N, N]      (bool/float)
            - weights_dict: model weights (LayerNorm, Linear, etc.)
            - config_dict : {"dim": int,
                             "hidden_dim": int,
                             "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2) Linear projections (fp16 for the heavy tensors)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on the channel dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (keep everything in fp16 to stay in the kernel)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Output gate (applied after the heavy multiplication, keep fp32)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim] fp32

    # -----------------------------------------------------------------
    # 6) Rearrange tensors for the Triton kernel
    #    Shape after permute: [B, hidden_dim, N, N] → view → [B*hidden_dim, N, N]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Triton kernel launch (128×128 tiles, K‑tile 64, 8 warps)
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tile rows
        (N + BLOCK_N - 1) // BLOCK_N,   # tile cols
        B * H,                          # batch * hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # 8 warps (=256 threads) give good utilisation on H100
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to the original layout: [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # -----------------------------------------------------------------
    # 9) Final LayerNorm, output gating and projection to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # bias is False in the reference model

    return output
