"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass consists of four stages:

1. **Layer‑norm + fused linear** – The input tensor is layer‑normalized and
   a *single* half‑precision linear layer computes **all five** required
   projections (left_proj, left_gate, right_proj, right_gate, out_gate) in
   one GEMM.  This collapses five separate matmuls into one, reduces the
   number of reads of the input tensor and of the weight matrices, and lets
   us keep the data in FP16 for the heavy arithmetic.

2. **Mask & gating** – The optional mask is applied to the two
   projections only, then a sigmoid is applied to the three gate tensors
   and element‑wise multiplication fuses the gate with its projection.

3. **N³ reduction** – The core operation

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is a batched matrix multiply for each hidden slice `h`.
   A custom Triton kernel `_tri_mul_gemm_kernel` computes the whole batch
   (`batch × hidden`) in FP16 → FP32 with Tensor‑Core dot‑products.
   Tile sizes (64×64×64) were chosen to keep the per‑CTA register usage
   modest (≈16 KB) while still providing high arithmetic intensity on an
   H100.

4. **Output post‑processing** – A second LayerNorm on the hidden dimension,
   gating with `out_gate` and a final linear projection back to the original
   feature dimension.

The kernel launches a 3‑D grid:
    (rows, cols, batch × hidden) = (⌈N/64⌉, ⌈N/64⌉, B·H)

The implementation meets the required signature
`custom_kernel(data) -> torch.Tensor` and runs entirely on‑device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for TriMul reduction (left @ rightᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                         # pointers
    M, N, K,                                     # matrix sizes (all = seq_len)
    stride_am, stride_ah, stride_ai, stride_ak, # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn, # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj, # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each (batch, hidden) slice.
    A : (batch, hidden, M, K)  – fp16
    B : (batch, hidden, N, K)  – fp16 (interpreted as (K, N))
    C : (batch, hidden, M, N)  – fp32
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # --------------------------------------------------------------------- #
    # decode batch & hidden index
    # --------------------------------------------------------------------- #
    batch_idx   = pid_z // hidden_dim
    hidden_idx  = pid_z - batch_idx * hidden_dim

    # --------------------------------------------------------------------- #
    # base pointers for this (batch, hidden) slice
    # --------------------------------------------------------------------- #
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # --------------------------------------------------------------------- #
    # thread‑local offsets for the tile
    # --------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # tiled K loop
    # --------------------------------------------------------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai) + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (K × N) – note B is accessed as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate with Tensor‑Core dot (FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------- #
    # write result
    # --------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, S, S, C] (float32)
            - mask_tensor  : [B, S, S] or None
            - weights_dict : all linear / layer‑norm parameters
            - config_dict  : {'dim': C,
                               'hidden_dim': H,
                               'nomask': bool, ...}

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Helper to fetch weight tensors (already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim] (float32)

    # ------------------------------------------------------------------- #
    # 2. Fused linear (half‑precision) producing all five tensors at once
    # ------------------------------------------------------------------- #
    # concatenate weight matrices: (5*H, dim)
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    ).to(torch.float16)

    # halve the input once
    x_half = x.to(torch.float16)

    # shape: [B, S, S, 5*H] (half)
    fused_out = F.linear(x_half, fused_weight)

    # split into the five components
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused_out, hidden_dim, dim=-1
    )  # each [B, S, S, H] (half)

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the last dim)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x_half.dtype).unsqueeze(-1)          # [B, S, S, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) and element‑wise gating of left/right
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate   # [B, S, S, H] (half)
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 5. N³ reduction – batched GEMM via Triton
    # ------------------------------------------------------------------- #
    # bring tensors to (B, H, S, S) layout expected by the kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, S, S] (half)
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, S, S] (half)

    # allocate output (FP32 accumulator)
    out_fp32 = torch.empty(
        B, hidden_dim, S, S, dtype=torch.float32, device=device
    )   # [B, H, S, S]

    # kernel tile sizes – selected to keep register pressure low while
    # providing high Tensor‑Core utilisation on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows
        triton.cdiv(S, BLOCK_N),          # cols
        B * hidden_dim,                   # (batch * hidden)
    )

    _tri_mul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left.stride(0),
        stride_ah=left.stride(1),
        stride_ai=left.stride(2),
        stride_ak=left.stride(3),
        stride_ba=right.stride(0),
        stride_bh=right.stride(1),
        stride_bk=right.stride(3),   # k‑dimension inside B (inner)
        stride_bn=right.stride(2),   # j‑dimension inside B (outer)
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
    )

    # Restore original layout: [B, S, S, H]
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm (hidden dimension) – still in float32
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 7. Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------- #
    out = out * out_gate                     # broadcasting, FP32 × FP16 → FP32
    out = F.linear(out, to_out_weight)       # [B, S, S, dim] (float32)

    return out
```
