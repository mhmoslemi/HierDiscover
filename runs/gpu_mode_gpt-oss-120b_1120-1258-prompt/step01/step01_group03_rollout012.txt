@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # *const fp16* [total_rows, dim]
    out_ptr,        # *mut fp16* [total_rows, dim]
    gate_ptr,       # *const fp16* [total_rows, dim]   # out_gate
    weight_ptr,     # *const fp16* [dim]
    bias_ptr,       # *const fp16* [dim]
    total_rows, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   off_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = off_m < total_rows

   # compute mean & var
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + off_m[:, None] * dim + offs_d[None,:]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # second pass: write normalized * gate
   for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + off_m[:, None] * dim + offs_d[None,:]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # gamma, beta
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out_val = norm * w[None,:] + b[None,:]

        # multiply by gate
        gate_vals = tl.load(gate_ptr + off_m[:, None] * dim + offs_d[None,:], mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        out_val = out_val * gate_vals

        # store
        out_ptrs = out_ptr + off_m[:, None] * dim + offs_d[None,:]
        tl.store(out_ptrs, out_val.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])
```

"""
TriMul (outgoing) – fast Triton+PyTorch implementation
-----------------------------------------------------

Algorithm
---------
1. **First LayerNorm** (row‑wise)  
   Custom Triton kernel works on the flattened tensor
   `[B·N·N, dim]`.  The reduction (mean/var) is done in FP32,
   the normalized result is written back in FP16.

2. **Fused projection + gating**  
   The five linear transforms (left_proj, right_proj, left_gate,
   right_gate, out_gate) share the same input.  We concatenate the
   five weight matrices → `W_comb` of shape
   `[5·hidden_dim, dim]` and perform a single `F.linear`
   (half‑precision GEMM).  Afterwards the result is split,
   sigmoid is applied to the gate slices and the optional mask
   is multiplied with the two projection slices.

3. **Triangular multiplicative update**  
   `left` and `right` tensors of shape `[B, N, N, H]` are
   reshaped to `[B·H, N, N]` and a batched matrix‑multiply
   (`torch.bmm`) computes `out = left @ rightᵀ` for every hidden
   channel.  All tensors are FP16, therefore Tensor‑Core GEMM is
   used.

4. **Second LayerNorm + output‑gate**  
   A second Triton kernel normalises `out` across the hidden
   dimension, applies the learned γ/β and finally multiplies the
   result element‑wise with `out_gate`.  This fuses two passes
   (norm + gate) into one.

5. **Final projection**  
   A standard `F.linear` (half‑precision) maps the gated hidden
   tensor back to the original `dim`.  The final result is cast
   to FP32.

All heavy reductions are performed in Triton; the remaining dense
linear algebra uses PyTorch’s highly‑optimised kernels.  The
implementation works for any configuration described in the prompt,
supports optional masking, and runs in FP16 except for the very
small reductions that stay in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Row‑wise LayerNorm
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *const fp16*   [total_rows, dim]
    y_ptr,            # *mut   fp16*   [total_rows, dim]
    weight_ptr,       # *const fp16*   [dim]
    bias_ptr,         # *const fp16*   [dim]
    total_rows,       # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # 1st pass – compute mean & variance
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -------------------------------------------------------------
    # 2nd pass – write normalised values
    # -------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
#  LayerNorm + element‑wise gate (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # *const fp16*   [total_rows, dim]
    out_ptr,        # *mut   fp16*   [total_rows, dim]
    gate_ptr,       # *const fp16*   [total_rows, dim]   (out_gate)
    weight_ptr,     # *const fp16*   [dim]
    bias_ptr,       # *const fp16*   [dim]
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # first pass – mean / variance
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------------------
    # second pass – normalise, apply γ/β and finally the gate
    # -------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        # multiply by the already‑sigmoided out_gate
        gate = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate

        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # make everything half‑precision (except small reductions)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous().to(torch.float16)            # [B,N,N,dim]
    B, N, _, _ = x.shape
    total_rows = B * N * N                                       # rows for first LN

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                             # [M, dim]
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 64          # chosen for good occupancy on H100
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"].to(torch.float16),
        weights["norm.bias"].to(torch.float16),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused projection + gating (single linear)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices → (5*H, dim)
    W_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()          # [5*H, dim]

    # fused linear (M, dim) @ (dim, 5H)ᵀ  -> (M, 5H)
    y = F.linear(x_norm_flat, W_comb)         # half‑precision output

    # split the concatenated result
    H = hidden_dim
    left_proj   = y[:, 0:H]                     # (M, H)
    right_proj  = y[:, H:2 * H]                # (M, H)
    left_gate_pre  = y[:, 2 * H:3 * H]          # (M, H)
    right_gate_pre = y[:, 3 * H:4 * H]          # (M, H)
    out_gate_pre   = y[:, 4 * H:5 * H]          # (M, H)

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # reshape back to [B,N,N,H]
    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # (B, H, N, N) for efficient bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)        # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)       # [BH, N, N]

    # torch.bmm uses Tensor‑Core GEMM for half‑precision → fast
    out_mat = torch.bmm(left_mat, right_mat)    # [BH, N, N] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) Second LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                 # positions
    out_flat      = out.view(total_rows_h, H).contiguous()          # [M2, H]
    out_gate_flat = out_gate.view(total_rows_h, H).contiguous()    # [M2, H]
    out_norm_gated = torch.empty_like(out_flat)                     # result buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),
        weights["to_out_norm.bias"].to(torch.float16),
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Final linear projection (hidden → dim)
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
        bias=None,
    )
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
