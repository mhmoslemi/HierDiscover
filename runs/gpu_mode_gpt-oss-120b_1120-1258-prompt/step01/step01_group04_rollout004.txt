def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract config
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    eps = 1e-5
    # unpack weights; convert to half where needed
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']
    # layernorm
    x = torch.nn.functional.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=eps,
        )
    # cast to half for subsequent linear ops
    x_h = x.to(torch.float16)
    # linear projection to hidden_dim
    left_proj_weight_h = left_proj_weight.to(torch.float16)
    right_proj_weight_h = right_proj_weight.to(torch.float16)
    left = torch.nn.functional.linear(x_h, left_proj_weight_h)   # [B,N,N,H] fp16
    right = torch.nn.functional.linear(x_h, right_proj_weight_h)
    # mask
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h
    # gating
    left_gate_weight_h = left_gate_weight.to(torch.float16)
    right_gate_weight_h = right_gate_weight.to(torch.float16)
    out_gate_weight_h = out_gate_weight.to(torch.float16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight_h))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight_h))
    left = left * left_gate
    right = right * right_gate
    # arrange for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    M_total = B * H
    left_flat = left_perm.view(M_total, N, N)   # [M, N, N] half
    right_flat = right_perm.view(M_total, N, N) # [M, N, N] half
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)
    # define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # compute strides for kernel
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)  # original j stride
    stride_bk = right_flat.stride(2)  # original k stride
    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)
    # launch
    grid = ((N + BLOCK_M - 1)//BLOCK_M,
            (N + BLOCK_N - 1)//BLOCK_N,
            M_total)
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape output to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # layernorm on hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # scale by output gate
    out = out * out_gate
    # final linear back to dim
    output = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]
    return output

"""
TriMul (outgoing) – fast Triton implementation

Algorithm
---------
1. Layer‑norm over the input channel dim (dim) (float32).
2. Cast to fp16 and compute four bias‑free linear projections:
   * left  = x @ left_proj_weight.T          (shape [B,N,N,H])
   * right = x @ right_proj_weight.T
   * left_gate  = sigmoid(x @ left_gate_weight.T)   (float32 → fp16)
   * right_gate = sigmoid(x @ right_gate_weight.T)
   * out_gate   = sigmoid(x @ out_gate_weight.T)   (float32)
   * left  = left  * left_gate   (fp16)
   * right = right * right_gate   (fp16)
   * optional elementwise mask (fp16) is applied before gating.
3. Fuse the cubic contraction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   into a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
   * Input tensors are reshaped to (M, N, N) with M = B·H.
   * The kernel tiles the i‑, j‑ and k‑dimensions (BLOCK_M×BLOCK_N×BLOCK_K)
     and uses Tensor‑Core‑enabled `tl.dot` on fp16 data while accumulating
     into fp32.
4. Reshape the result to [B,N,N,H] (float32) and apply a second
   LayerNorm over the hidden dimension (H).
5. Multiply by the output gate (float32) and project back to the original
   channel dimension with a final linear layer (float32).

The heavy O(N³·H) work is performed entirely in fp16 on the GPU, giving
large speed‑ups on an H100 while the final result stays in float32.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                # left, right, output pointers
    N,                                   # sequence length
    stride_am, stride_ai, stride_ak,    # strides of A (M,i,k)
    stride_bm, stride_bj, stride_bk,    # strides of B (M,j,k) – we read Bᵀ as (k,j)
    stride_cm, stride_ci, stride_cj,    # strides of C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile along i (rows)
    pid_n = tl.program_id(1)   # tile along j (cols)
    pid_b = tl.program_id(2)   # batch·hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # output tile pointer
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i,k]   (left)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # Bᵀ[k,j] = right[j,k] (right, transposed on‑the‑fly)
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0))

        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

    # write tile
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  `data` is a tuple (input_tensor, mask, weights, config).

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------------- #
    # Configuration & hyper‑parameters
    # --------------------------------------------------------------------- #
    dim = config["dim"]                 # input channel dim
    hidden_dim = config["hidden_dim"]   # projection dim (H)
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Unpack model weights (already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the input channel dimension (float32)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                 # [B, N, N, dim]  float32

    # --------------------------------------------------------------------- #
    # 2️⃣ Cast to fp16 for all subsequent linear ops
    # --------------------------------------------------------------------- #
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3️⃣ Linear projections (bias‑free) → fp16
    # --------------------------------------------------------------------- #
    left_proj_weight_h  = left_proj_weight.to(torch.float16)
    right_proj_weight_h = right_proj_weight.to(torch.float16)

    left  = torch.nn.functional.linear(x_h, left_proj_weight_h)   # [B, N, N, H] fp16
    right = torch.nn.functional.linear(x_h, right_proj_weight_h)

    # --------------------------------------------------------------------- #
    # 4️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------------- #
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------------- #
    # 5️⃣ Gating (sigmoid) – computed in float32, then cast to fp16
    # --------------------------------------------------------------------- #
    left_gate_fp32  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate_fp32 = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate_fp32   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left_gate_h  = left_gate_fp32.to(torch.float16)
    right_gate_h = right_gate_fp32.to(torch.float16)

    left  = left  * left_gate_h
    right = right * right_gate_h

    # --------------------------------------------------------------------- #
    # 6️⃣ Prepare tensors for the batched GEMM (A @ Bᵀ)
    # --------------------------------------------------------------------- #
    #   left/ right : [B, N, N, H] → [B, H, N, N] → (M, N, N) where M = B·H
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat  = left_perm.view(M_total, N, N)   # (M, N, N)  fp16
    right_flat = right_perm.view(M_total, N, N)  # (M, N, N)  fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # --------------------------------------------------------------------- #
    # 7️⃣ Triton kernel launch – the heavy O(N³·H) step
    # --------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                     # works well for fp16 TensorCore ops

    # strides (in elements, not bytes)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)  # original column dimension
    stride_bk = right_flat.stride(2)  # original row dimension (used as k after transpose)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        M_total                         # one program per (batch, hidden) pair
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 8️⃣ Reshape back to [B, N, N, H] (float32)
    # --------------------------------------------------------------------- #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    #  🔟 Apply the output gate and final projection back to `dim`
    # --------------------------------------------------------------------- #
    out = out * out_gate_fp32                     # element‑wise (float32)
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] float32

    return output
