"""
TriMul (outgoing) – fast Triton implementation
================================================

This file defines a single entry point ``custom_kernel`` that computes the forward
pass of the *outgoing* TriMul module used in AlphaFold‑3 and related protein
prediction models.

Algorithm (per batch ``B`` and hidden dimension ``H``):
-----------------------------------------------------------------
1️⃣ Layer‑norm the 4‑D input ``x`` (shape ``[B,N,N,C]``).
2️⃣ Linear projections to the hidden dimension ``H`` produce ``left`` and ``right``.
   Both are cast to ``float16`` for the heavy compute.
3️⃣ Optional mask (broadcast over ``H``) and three sigmoid gates
   (``left_gate``, ``right_gate``, ``out_gate``).  ``left``/``right`` are gated.
4️⃣ Core operation – batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of GEMMs for each pair *(b,h)*.  The work is carried out by a
   fused Triton kernel that processes all ``G = B·H`` groups in parallel.
5️⃣ Layer‑norm over the hidden dimension, multiplication by the ``out_gate`` and
   a final linear projection back to the original channel size ``C``.

All heavy arithmetic (the N³ contraction) is executed in FP16 with FP32
accumulation for numerical stability.  The rest of the pipeline uses high‑
performance PyTorch primitives.

The implementation supports both masked and mask‑free inputs and works for any
valid configuration of ``B ∈ {1,2}``, ``N ∈ {128,256,512,1024}``, ``C`` and
``H`` (hidden_dim) as described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_kernel(
    a_ptr,                     # pointer to left  tensor (G, M, K)
    b_ptr,                     # pointer to right tensor (G, K, N)
    c_ptr,                     # pointer to output (G, M, N)
    M, N, K,                   # matrix sizes
    stride_ag, stride_am, stride_ak,   # strides for A: group, row, col
    stride_bg, stride_bk, stride_bn,   # strides for B: group, row(K), col(N)
    stride_cg, stride_cm, stride_cn,   # strides for C: group, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        C[g, i, j] = Σ_k A[g, i, k] * B[g, k, j]
    ``g`` = batch * hidden_dim (group) dimension.
    """
    pid = tl.program_id(0)      # tile index in the (M,N) plane
    pid_g = tl.program_id(1)    # group index (0 .. G‑1)

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n_tiles
    pid_n = pid % num_n_tiles

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this group
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_g * stride_ag
    b_base = b_ptr + pid_g * stride_bg
    c_base = c_ptr + pid_g * stride_cg

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (K × N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Accumulate dot product (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result back
    # ------------------------------------------------------------------
    c = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask         : torch.Tensor, shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys:
                         * "dim"        – input channel size C
                         * "hidden_dim" – projection size H
                         * "nomask"     – True → no mask, False → mask present

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C] – FP32

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections → FP16 for the heavy work
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left = F.linear(x, left_proj_w).to(torch.float16)   # [B, N, N, H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Gating tensors
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))                     # keep FP32

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Prepare tensors for the batched GEMM
    #     We need: C[g,i,j] = Σ_k left[g,i,k] * right[g,j,k]
    #     This is a standard GEMM left @ rightᵀ per group.
    # ------------------------------------------------------------------
    #   left : (B,N,N,H) → (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    #   right: (B,N,N,H) → (B,H,N,N) with (k,i) swapped → (B,H,K,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    G = B * H                         # total number of independent GEMMs

    # Collapse batch & head dimensions
    A = left_perm.view(G, N, N)      # (G, M=N, K=N), FP16
    B_ = right_perm.view(G, N, N)    # (G, K=N, N), FP16

    # Output buffer – accumulate in FP32 for precision
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # (G,N,N)

    # ------------------------------------------------------------------
    # 7️⃣ Compute strides for the Triton kernel
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)   # stride over K dimension
    stride_bn = B_.stride(2)   # stride over N dimension

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    M = N
    K = N

    # ------------------------------------------------------------------
    # 8️⃣ Triton kernel launch
    # ------------------------------------------------------------------
    # Tile sizes – empirically good for H100 (shared‑memory / register friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)   # (tiles over M×N, groups)

    _batch_matmul_kernel[grid](
        A,
        B_,
        C,
        M,
        N,
        K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well on H100 (SM‑90)
        num_stages=5,         # pipelined shared‑memory stages
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape the result back to [B, N, N, H] (still FP32)
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10️⃣ Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # FP32

    out = out * out_gate                     # element‑wise gate (FP32)

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)         # [B, N, N, C] – final FP32 result

    return output
```
