#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
TriMul (outgoing) – fast Triton implementation
================================================
The forward pass consists of:
  1️⃣ Layer‑norm over the last dimension (dim).
  2️⃣ A *single* fused linear layer that produces:
     left_proj, right_proj, left_gate, right_gate, out_gate
  3️⃣ Optional mask (applied before the gates).
  4️⃣ Element‑wise gating of the left/right projections.
  5️⃣ Batched matrix‑multiply (left @ rightᵀ) for every hidden slice.
     This is the N³ heavy part and is executed by a custom Triton GEMM.
  6️⃣ Layer‑norm over the hidden dimension, out‑gate application and
     a final linear projection back to the original dim.

All heavy arithmetic (the GEMM) is done in FP16 with FP32 accumulation.
The surrounding linear / layer‑norm work is kept in PyTorch (they are
memory‑bandwidth bound and cheap compared to the N³ reduction).

The kernel works on tensors in the original layout
    (B, S, S, H)   # B = batch, S = sequence length, H = hidden_dim
without any costly permutes or copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton GEMM kernel (batched, per‑hidden‑slice)
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all equal to seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for each (batch, hidden) slice.
    A : (batch, hidden, M, K)   – fp16
    B : (batch, hidden, N, K)   – fp16  (interpreted as (K, N))
    C : (batch, hidden, M, N)   – fp32
    """
    pid_m = tl.program_id(0)      # row tile
    pid_n = tl.program_id(1)      # col tile
    pid_z = tl.program_id(2)      # (batch * hidden) tile

    # ----- decode batch and hidden indices -----
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # ----- base pointers for this (batch, hidden) slice -----
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # ----- offsets within the tile -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop (tiled) -----
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile : (M, K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile : (K, N)   (we read B as (k, j))
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate product
        acc += tl.dot(a, b)

    # ----- write result -----
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Perform the outgoing TriMul forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, S, S, C]  (C == dim, float32)
        - mask_tensor  : [B, S, S]    (bool/float, may be None)
        - weights_dict : all model parameters (torch tensors)
        - config_dict  : {'dim': C,
                          'hidden_dim': H,
                          'nomask': bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Extract raw weight tensors
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]    # (dim, hidden_dim)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  (float32)

    # ------------------------------------------------------------------- #
    # 2️⃣ Fused linear projection that yields all five tensors at once
    # ------------------------------------------------------------------- #
    # Cast input to fp16 for the linear ops (weights are cast on‑the‑fly)
    x_fp16 = x.to(torch.float16)

    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    # Linear produces a huge tensor [B, S, S, 5*H] in fp16
    fused = F.linear(x_fp16, fused_weight)  # no bias
    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )   # each => [B, S, S, H] (fp16)

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (applied before the gates, exactly as reference)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid) and element‑wise gating of the projections
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left  = left_proj * left_gate   # [B, S, S, H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Batched GEMM via Triton (left @ rightᵀ per hidden slice)
    # ------------------------------------------------------------------- #
    # Allocate output buffer (FP32 accumulation)
    out_fp32 = torch.empty(
        (B, S, S, hidden_dim), dtype=torch.float32, device=device
    )   # [B, S, S, H]

    # Kernel launch parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(S, BLOCK_M),           # rows
        triton.cdiv(S, BLOCK_N),           # cols
        B * hidden_dim,                    # (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left.stride(0),   # batch stride
        stride_ah=left.stride(3),   # hidden stride (==1)
        stride_ai=left.stride(1),   # i stride = S*H
        stride_ak=left.stride(2),   # k stride = H
        stride_ba=right.stride(0),
        stride_bh=right.stride(3),
        stride_bk=right.stride(2),  # k stride = H
        stride_bn=right.stride(1),  # j stride = S*H
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(3),   # hidden stride (==1)
        stride_ci=out_fp32.stride(1),   # i stride = S*H
        stride_cj=out_fp32.stride(2),   # j stride = H
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        NUM_WARPS=4,
    )

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------- #
    # Layer‑norm over hidden dimension
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H] (float32)

    out = out * out_gate.to(out.dtype)  # broadcast gate (now float32)

    # Final linear back to original dim
    out = F.linear(out, to_out_weight)   # [B, S, S, dim] (float32)

    return out
