```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # layernorm
    x = torch.nn.functional.layer_norm(...)

    # linear projections: left, right
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gating values
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # mask: if not nomask -> broadcast mask and multiply left,right
    if not nomask:

        left = left * mask_exp
        right = right * mask_exp

    # apply gates
```

```python
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i tile
    pid_n = tl.program_id(1)   # j tile
    pid_b = tl.program_id(2)   # batch*hidden

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # output pointer for tile
    c_ptrs = c_ptr + pid_b * stride_cm + i[:,None] * stride_ci + j[None,:] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        a_ptrs = a_ptr + pid_b * stride_am + i[:,None] * stride_ai + k[None,:] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:,None] & mask_k[None,:],
                    other=0.0)

        b_ptrs = b_ptr + pid_b * stride_bm + k[:,None] * stride_bk + j[None,:] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:,None] & mask_j[None,:],
                    other=0.0)

        # a,b are fp16; dot accumulates into fp32
        acc += tl.dot(a, b)

    # Cast to fp16 before store
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:,None] & mask_j[None,:])
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The operator works on a 4‑D tensor  [B, N, N, C]  (C = dim) and performs:

1. Layer‑norm over the feature axis.
2. Two bias‑free linear projections (left / right) to a hidden size H.
3. Optional pair‑wise mask (broadcasted on H).
4. Three sigmoid gates (left, right, out) computed from the same normalized input.
5. Gated left/right tensors are contracted with an O(N³·H) “tri‑mul”:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM (A @ Bᵀ) using a Triton kernel.
   The kernel works on fp16 tensors, uses Tensor‑Core `tl.dot` (fp16→fp32 accumulation)
   and dynamically adapts the tile size (64/128) to the sequence length.
6. Layer‑norm over the hidden dimension, multiplication with the out‑gate,
   and a final bias‑free linear projection back to the original dimension.

Only the heavy cubic contraction is written in Triton; the surrounding
operations stay in PyTorch. The result is returned as a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  (M, N, N)  – fp16
    b_ptr,               # right (M, N, N)  – fp16 (read as k‑j)
    c_ptr,               # output (M, N, N) – fp16
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, j, k) – we read (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ (fp16 → fp32 accumulator)."""
    pid_i = tl.program_id(0)   # tile along i (rows)
    pid_j = tl.program_id(1)   # tile along j (cols)
    pid_m = tl.program_id(2)   # batch * hidden index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # output pointer for this tile
    c_ptrs = c_ptr + pid_m * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k]  (M, i, k)
        a_ptrs = a_ptr + pid_m * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # B[k, j]  (we read B transposed: original layout (M, j, k))
        b_ptrs = b_ptr + pid_m * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0))

        # fp16 * fp16 → fp32 accumulation (Tensor‑Core)
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # store fp16 result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – ignored if config['nomask']==True
            - weights      : dict of torch.Tensors (already on CUDA)
            - config       : dict containing:
                * dim          – input feature dimension C
                * hidden_dim   – projection dimension H
                * nomask       – True → ignore mask
                * eps          – optional layer‑norm epsilon (default 1e‑5)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------ #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcasted on the hidden axis)
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → same dtype as left/right (float32)
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------ #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # used later (float32)

    # Apply gates (still in fp32)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Cast to fp16 and launch Triton kernel for the cubic contraction
    # ------------------------------------------------------------------ #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring the hidden dimension to the second axis for a clean (M, N, N) view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (fp16)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] (fp16)

    B, H, N, _ = left_perm.shape
    M_total = B * H                         # treated as batch dimension for the kernel

    left_flat = left_perm.view(M_total, N, N)   # [M, N, N] (fp16)
    right_flat = right_perm.view(M_total, N, N) # [M, N, N] (fp16)

    out_flat = torch.empty_like(left_flat)      # fp16 output

    # ------------------------------------------------------------------ #
    # 6. Kernel launch – choose tile sizes based on sequence length
    # ------------------------------------------------------------------ #
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Strides (Python ints)
    stride_am = left_flat.stride(0)   # M stride = N*N
    stride_ai = left_flat.stride(1)   # i stride = N
    stride_ak = left_flat.stride(2)   # k stride = 1

    stride_bm = right_flat.stride(0)  # same as left_flat
    stride_bj = right_flat.stride(1)  # j stride = N
    stride_bk = right_flat.stride(2)  # k stride = 1

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M_total                         # one program per (batch, hidden) pair
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 7. Reshape back to [B, N, N, H] and convert to float32 for the rest
    # ------------------------------------------------------------------ #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------ #
    # 8. Post‑contraction LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------ #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                        # out‑gate (float32)
    output = F.linear(out, to_out_weight)       # final projection, still float32

    return output
