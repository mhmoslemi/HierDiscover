"""
TriMul (outgoing) – fast Triton implementation.

The core of the TriMul operator is the 3‑tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix multiplication L @ Rᵀ for every hidden channel *h*.
The kernel below computes this contraction with fp16 inputs, fp32 accumulation,
and stores the result in fp16.  All per‑element linear projections and gating
operations are fused into a single matrix multiplication (5 × hidden_dim
outputs) to minimise memory traffic.  This reduces the number of passes over the
large [B, N, N, dim] tensor from five to one and therefore cuts the overall
runtime dramatically while preserving the exact forward behaviour of the
reference PyTorch model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,                # *fp16  [B*H, N, N]
    right_ptr,               # *fp16  [B*H, N, N]
    out_ptr,                 # *fp16  [B*H, N, N]  (final result stored as fp16)
    M, N, K,                 # all equal to seq_len
    stride_la, stride_li, stride_lk,   # left strides (batch*H, i, k)
    stride_rb, stride_ri, stride_rk,   # right strides (batch*H, j, k)
    stride_ob, stride_oi, stride_oj,   # out strides (batch*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out = left @ rightᵀ
    left  : [B*H, M, K]  (fp16)
    right : [B*H, N, K]  (fp16) – accessed as the transposed view for the matmul
    out   : [B*H, M, N]  (fp16) – stored after fp32 accumulation
    """
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch * hidden channel

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the tiles
    # ------------------------------------------------------------------
    a_ptr = left_ptr + pid_b * stride_la + offs_m[:, None] * stride_li
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # ------------------------------------------------------------------
    # Accumulator (fp32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of left:  (BLOCK_M, BLOCK_K)  – fp16
        a = tl.load(
            a_ptr + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of right (transposed view): (BLOCK_K, BLOCK_N) – fp16
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate the partial product
        acc += tl.dot(a, b).to(tl.float32)

        # Advance pointers to the next K‑tile
        a_ptr += BLOCK_K * stride_lk
        b_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # Write back (store as fp16)
    # ------------------------------------------------------------------
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]  (bool/int/float)
        - weights_dict : dict with all model parameters (float32)
        - config_dict  : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input feature dimension
    hidden_dim = config["hidden_dim"] # projection dimension (usually 128)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # 2. Fuse the five per‑position linear projections (half‑precision)
    # ------------------------------------------------------------------
    # Cast the normalized tensor to fp16 before the big matmul.
    x_h = x.half()  # [B, N, N, dim] – fp16

    # Concatenate projection & gate weight matrices (shape: [5*hidden_dim, dim])
    proj_weights = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,

    # One massive linear layer: produces all five tensors at once.
    combined = F.linear(x_h, proj_weights)          # [B, N, N, 5*hidden_dim] – fp16

    # Split into the five components.
    left_proj, right_proj, left_gate, right_gate, out_gate = combined.split(hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 3. Apply sigmoid gates (still fp16) and optional mask.
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)    # [B, N, N, hidden_dim]  fp16
    right_gate = torch.sigmoid(right_gate)  # fp16
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not config.get("nomask", False):
        # mask shape -> [B, N, N, 1] and fp16 for broadcasting
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Prepare tensors for the Triton kernel.
    #    We reshape to [B*H, N, N] where each matrix corresponds to one
    #    hidden channel of a particular batch.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Bring the hidden dimension in front of the two sequence axes.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # Output buffer (fp16, fp32 accumulation inside the kernel)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5. Launch the Triton kernel.
    # ------------------------------------------------------------------
    # Choose tile sizes according to the sequence length.
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_la=left_flat.stride(0),
        stride_li=left_flat.stride(1),
        stride_lk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps works well on H100 for fp16 matmul
        num_stages=3,              # double‑buffering for better latency hiding
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to the original layout and cast to float32 for the
    #    remaining LayerNorm / linear steps.
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()  # [B, N, N, hidden_dim] fp32

    # ------------------------------------------------------------------
    # 7. Final LayerNorm on the hidden dimension, out‑gate and projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # out_gate was computed in fp16; bring it to float32 for the multiplication.
    out_gate = torch.sigmoid(out_gate).float()  # shape [B, N, N, hidden_dim]
    out = out * out_gate

    # Final linear projection back to the original dim.
    output = F.linear(out, weights["to_out.weight"])  # bias is False in the model

    return output
```
