```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # retrieve weights as described
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear proj
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,S,S,H]
    right = torch.nn.functional.linear(x, right_proj_weight) # same shape

    # Mask if needed
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Multiply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Transpose right for correct multiplication using torch.bmm
    right_h_t = right_h.permute(0, 3, 2, 1)  # [B, H, K, J] where K dimension = second axis, J=first
    # Actually need shape (B, H, K, J) such that multiplication left_h @ right_h_t works.
    # left_h currently shape (B, H, I, K) after permute.
    left_h = left_h.permute(0, 3, 1, 2)  # (B, H, I, K)

    # Flatten batch*hidden dimension
    B_H = B * hidden_dim
    left_mat = left_h.reshape(B_H, S, S)      # (B*H, I, K)
    right_mat = right_h_t.reshape(B_H, S, S)  # (B*H, K, J)

    # Batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, I, J) fp16 result
    # Cast to FP32 for subsequent layernorm
    out = out_mat.to(torch.float32).reshape(B, hidden_dim, S, S).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass.
1. LayerNorm on input.
2. Linear projections.
3. Optional mask.
4. Gating (sigmoid).
5. Fuse gating, cast to fp16, and compute batched GEMM:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a Triton kernel performing batched matrix multiplication
   over (B*hidden) slices.
6. Output LayerNorm, out‑gate and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel performing a batched GEMM for the N³ reduction.
# --------------------------------------------------------------------------- #
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64}, num_warps=8),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 128}, num_warps=8),
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 128}, num_warps=4),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix dimensions (all = seq_len)
    stride_am, stride_ai, stride_ak,     # A: batch, row(i), col(k) strides
    stride_ba, stride_bk, stride_bn,     # B: batch, row(k), col(j) strides
    stride_cm, stride_ci, stride_cj,     # C: batch, row(i), col(j) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   for each batch slice.
    A : (Z, M, K)   – fp16   (Z = batch * hidden)
    B : (Z, K, N)   – fp16
    C : (Z, M, N)   – fp32 (accumulated)
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # flattened batch (B * hidden)

    # ----- compute offsets for this tile -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointer to the beginning of the current batch slice -----
    a_batch_ptr = a_ptr + pid_z * stride_am
    b_batch_ptr = b_ptr + pid_z * stride_ba
    c_batch_ptr = c_ptr + pid_z * stride_cm

    # ----- accumulate into FP32 accumulator -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # load B tile (K x N)
        b = tl.load(
            b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)      # fp16 dot → fp32 accumulate

    # ----- write the result -----
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Entrypoint required by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, S, S, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, S, S]  (bool/float) or None
        - weights_dict : mapping from weight name to torch.Tensor (already on device)
        - config_dict  : {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns
    -------
    torch.Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # weight shortcuts
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1) Input layer‑norm (dim = C)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, C]  (float32)

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B, S, S, H]
    right = F.linear(x, right_proj_weight) # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to fp16 and reshape for the batched GEMM.
    #    left  : (B, H, I, K)
    #    right : (B, H, K, J)  (note the transpose on the second pair of axes)
    # ------------------------------------------------------------------- #
    left_h  = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_h = right.to(torch.float16).permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # flatten batch * hidden = Z
    Z = B * hidden_dim
    left_flat  = left_h.view(Z, S, S)   # (Z, M, K)
    right_flat = right_h.view(Z, S, S)  # (Z, K, N)

    # output buffer (FP32 accumulator)
    out_fp32 = torch.empty_like(left_flat, dtype=torch.float32, device=device)  # (Z, M, N)

    # ------------------------------------------------------------------- #
    # 6) Triton GEMM launch
    # ------------------------------------------------------------------- #
    # block sizes – the autotuned configs will pick the best one at compile time
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),   # rows
        triton.cdiv(S, BLOCK_N),   # cols
        Z,                         # flattened batch * hidden
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_ba=right_flat.stride(0),
        stride_bk=right_flat.stride(1),
        stride_bn=right_flat.stride(2),
        stride_cm=out_fp32.stride(0),
        stride_ci=out_fp32.stride(1),
        stride_cj=out_fp32.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7) Restore original layout: (B, S, S, H)
    # ------------------------------------------------------------------- #
    out = out_fp32.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).contiguous()  # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 8) Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 9) Apply out‑gate and final projection back to the original dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, S, S, C]

    return out
```

"""
TriMul (outgoing) forward pass – highly‑optimized Triton version.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim).
2. Two bias‑free linear projections → left / right (shape [B, S, S, H]).
3. Optional mask is applied (broadcasted on the hidden axis).
4. Gating: sigmoid(linear(x)) and element‑wise modulation of left/right.
5. The cubic reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix‑multiply left @ rightᵀ for every hidden slice.
   We flatten the (batch × hidden) dimension and launch a Triton GEMM kernel
   that tiles the multiplication (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates
   in FP32 while using FP16 inputs.
6. Output layer‑norm over the hidden dimension, gating with out_gate and a
   final linear projection back to the original channel dimension.

The Triton kernel is autotuned for several block‑size configurations, which
gives a large speed‑up over the reference PyTorch implementation for all
tested sequence lengths.

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, C]  (float32)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B, S, S, H]
    right = F.linear(x, right_proj_weight) # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4. Gating – sigmoid of a linear projection
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Prepare FP16 operands and reshape for a batched GEMM
    #    left : (B, H, I, K)   →  (Z, M, K)
    #    right: (B, H, K, J)   →  (Z, K, N)   (note the transpose on the last two axes)
    # ------------------------------------------------------------------- #
    left_h  = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_h = right.to(torch.float16).permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    Z = B * hidden_dim                               # flattened batch * hidden
    left_flat  = left_h.view(Z, S, S)                # (Z, M, K)
    right_flat = right_h.view(Z, S, S)               # (Z, K, N)

    # Output buffer – FP32 accumulation as required by the reference model
    out_fp32 = torch.empty_like(left_flat, dtype=torch.float32, device=device)  # (Z, M, N)

    # ------------------------------------------------------------------- #
    # 6. Launch the Triton GEMM kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),   # rows
        triton.cdiv(S, BLOCK_N),   # cols
        Z,                         # flattened (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_ba=right_flat.stride(0),
        stride_bk=right_flat.stride(1),
        stride_bn=right_flat.stride(2),
        stride_cm=out_fp32.stride(0),
        stride_ci=out_fp32.stride(1),
        stride_cj=out_fp32.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7. Reshape back to the original layout: [B, S, S, H]
    # ------------------------------------------------------------------- #
    out = out_fp32.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 9. Apply the final gate and project back to the original channel dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, S, S, C]

    return out
```
