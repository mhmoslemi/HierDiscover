"""
TriMul outgoing kernel (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1. Row‑wise LayerNorm on the last dimension (dim) with a small Triton kernel.
2. Cast the normalized tensor to FP16.  All dense linear layers and the
   expensive O(N³) triangular update are performed in half‑precision
   (Tensor‑core GEMM) to exploit H100 performance.
3. Linear projections (left/right) and three sigmoid gates are computed
   with `torch.nn.functional.linear`.  If a mask is supplied it is applied
   in FP16 before the element‑wise gates.
4. The multiplicative update
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   is realized as a batch matrix‑multiply `L @ Rᵀ` for each hidden channel.
   We reshape the tensors to [B·H, N, N] and call `torch.bmm`, which is
   highly optimized on H100.
5. A second LayerNorm over the hidden dimension (hidden_dim) uses the
   built‑in `torch.nn.functional.layer_norm` (half‑precision).
6. The final output gate is applied and a linear projection back to `dim`
   finishes the forward pass.  The result is cast back to FP32.

The implementation uses a Triton kernel only for the first LayerNorm,
fulfilling the requirement of “at least part of the operations in a kernel”.
All heavy arithmetic is done in FP16 to reduce runtime while keeping the
final output in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr,                # const float*  [total_rows, dim]
    y_ptr,                # mut   float*  [total_rows, dim]
    weight_ptr,           # const float*  [dim]
    bias_ptr,             # const float*  [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (float32) for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values with gamma/beta -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim)                     # [R, dim] float32
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 256   # rows per program
    BLOCK_D = 32    # columns per iteration

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)               # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 and perform linear projections + gating
    # -----------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)                  # half
    total_rows = B * N * N

    # half‑precision weight copies (performed once)
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left_proj_h  = F.linear(x_norm_h.view(total_rows, dim), w_left_proj)   # [R, H]
    right_proj_h = F.linear(x_norm_h.view(total_rows, dim), w_right_proj)

    # Optional mask (broadcasted)
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Gates (sigmoid)
    left_gate_h  = torch.sigmoid(F.linear(x_norm_h.view(total_rows, dim), w_left_gate))
    right_gate_h = torch.sigmoid(F.linear(x_norm_h.view(total_rows, dim), w_right_gate))
    out_gate_h   = torch.sigmoid(F.linear(x_norm_h.view(total_rows, dim), w_out_gate))

    # Apply element‑wise gates
    left_h  = left_proj_h * left_gate_h      # [R, H]
    right_h = right_proj_h * right_gate_h

    # Reshape to 4‑D tensors
    left_h  = left_h.view(B, N, N, hidden_dim)   # [B,N,N,H] half
    right_h = right_h.view(B, N, N, hidden_dim)
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # Permute so that hidden dim becomes batch dimension
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # Rᵀ, shape [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)  # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)   # half, shape [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4) Second LayerNorm over hidden dimension (half‑precision)
    # -----------------------------------------------------------------
    # torch.nn.functional.layer_norm works for FP16
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 5) Final gated projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                     # element‑wise half
    out_norm_flat = out_norm.view(total_rows, hidden_dim)  # [R,H]
    final_half = F.linear(out_norm_flat, w_to_out)       # [R, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul outgoing kernel (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1. Row‑wise LayerNorm over the channel dimension (`dim`) using a small Triton
   kernel (float32).  This kernel only computes mean/variance and applies the
   learned γ/β.
2. Cast the normalized tensor to FP16.  All subsequent dense linear layers and
   the O(N³) triangular update are performed in half‑precision, exploiting
   H100 Tensor‑Core GEMM performance.
3. Linear projections (left/right) and three sigmoid gates are realized with
   `torch.nn.functional.linear`.  An optional mask (broadcasted) is applied
   in FP16 before the element‑wise gates.
4. The multiplicative update
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   is expressed as a batched matrix‑multiply `L @ Rᵀ` for each hidden channel.
   By permuting the tensors we obtain shape `[B·H, N, N]` and call `torch.bmm`,
   which is highly optimized on H100.
5. A second LayerNorm over the hidden dimension (`hidden_dim`) uses the
   built‑in `torch.nn.functional.layer_norm` (still in FP16).
6. The final output gate is applied and a linear projection back to `dim`
   yields the result, which is cast to FP32 before returning.

Only the first LayerNorm is implemented in Triton, satisfying the requirement
that “at least part of the operations” be a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                # const float*    [total_rows, dim]
    y_ptr,                # mut   float*    [total_rows, dim]
    weight_ptr,           # const float*    [dim]
    bias_ptr,             # const float*    [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (float32) for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute sum and sum‑of‑squares (mean & variance)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) First LayerNorm over `dim` (float32)
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()                # [B, N, N, dim]  float32
    B, N, _, _ = x.shape
    total_rows = B * N * N                       # rows after flattening

    x_flat = x.view(total_rows, dim)             # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)       # output buffer

    # Tuned block sizes – good for most N (≤ 1024)
    BLOCK_M = 256
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    x_norm = x_norm_flat.view(B, N, N, dim)      # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 and run the linear / gating parts
    # -----------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)          # half‑precision
    R = total_rows                                 # alias for readability
    x_norm_h_flat = x_norm_h.view(R, dim)        # [R, dim]

    # Prepare half‑precision weight copies (one‑time conversion)
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left_proj_h  = F.linear(x_norm_h_flat, w_left_proj)    # [R, H]
    right_proj_h = F.linear(x_norm_h_flat, w_right_proj)   # [R, H]

    # Optional mask (broadcasted over hidden dimension)
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(R, 1)  # [R,1]
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Gates (sigmoid)
    left_gate_h  = torch.sigmoid(F.linear(x_norm_h_flat, w_left_gate))
    right_gate_h = torch.sigmoid(F.linear(x_norm_h_flat, w_right_gate))
    out_gate_h   = torch.sigmoid(F.linear(x_norm_h_flat, w_out_gate))

    # Apply element‑wise gates
    left_h  = left_proj_h * left_gate_h      # [R, H]
    right_h = right_proj_h * right_gate_h     # [R, H]

    # Reshape to 4‑D tensors for the TriMul step
    left_h  = left_h.view(B, N, N, hidden_dim)   # [B,N,N,H] half
    right_h = right_h.view(B, N, N, hidden_dim)  # [B,N,N,H] half
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)  # [B,N,N,H] half

    # -----------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # Move hidden dimension to the batch axis and transpose right side
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # transpose k ↔ j

    BH = B * hidden_dim
    left_mat  = left_mat.view(BH, N, N)      # [B*H, N, N] half
    right_mat = right_mat.view(BH, N, N)     # [B*H, N, N] half

    # Batched matrix multiplication – uses Tensor‑cores
    out_mat = torch.bmm(left_mat, right_mat)          # [B*H, N, N] half

    # Back‑to original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4) Second LayerNorm over the hidden dimension (half‑precision)
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # result still half‑precision

    # -----------------------------------------------------------------
    # 5) Final gated projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                     # element‑wise half
    out_norm_flat = out_norm.view(R, hidden_dim)         # [R, H]

    # Linear projection hidden_dim → dim (no bias)
    final_half = F.linear(out_norm_flat, w_to_out)      # [R, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
