"""
TriMul (outgoing) – fast Triton implementation
================================================

The forward pass consists of four stages:

1. **Pre‑norm** – Layer‑norm on the 4‑D tensor (B,N,N,C) in FP32.
2. **Linear + gating** – 4 half‑precision linear projections (left/right
   and their sigmoidal gates) are evaluated in FP16.  The optional mask
   is applied while the gate values are fused with the projections.
3. **Batched matrix multiply** – Core O(N³) work:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The hidden dimension is folded into the batch dimension,
   giving a batched GEMM of shape (G = B·H, N, N).  The GEMM is
   completely written in Triton with a 128×128×32 tile and 8 warps
   per block (tensor‑core FP16 · FP16 → FP32 accumulation).
   This tile size reduces the number of launched blocks and fully
   utilizes the H100 tensor‑cores.
4. **Post‑norm, out‑gate & projection** – Layer‑norm on the hidden
   result, element‑wise gating, and a final FP32 linear projection back
   to the original channel dimension.

All heavy linear work is performed in FP16, the accumulation in the
matmul is FP32 for numerical stability, and the final output is FP32.
The implementation supports both masked and un‑masked inputs and works
for all test configurations listed in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (B*H) matrix‑multiply C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # Pointers to A, B, C
    M, N, K,                                 # Matrix sizes (M×K)·(K×N) → (M×N)
    stride_ag, stride_am, stride_ak,         # Strides for A: group, row, col
    stride_bg, stride_bk, stride_bn,         # Strides for B: group, row(K), col(N)
    stride_cg, stride_cm, stride_cn,         # Strides for C: group, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[g, i, j] = Σ_k A[g, i, k] * B[g, k, j]
    A and B are FP16, accumulation is FP32.
    """
    pid = tl.program_id(0)          # tile index inside the (M,N) plane
    pid_g = tl.program_id(1)        # group index (g = b*H + h)

    # --------------------------------------------------------------
    # tile coordinates
    # --------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n_tiles
    pid_n = pid %  num_n_tiles

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # base pointers for the current group
    # --------------------------------------------------------------
    a_group = a_ptr + pid_g * stride_ag
    b_group = b_ptr + pid_g * stride_bg
    c_group = c_ptr + pid_g * stride_cg

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main loop over the K dimension
    # --------------------------------------------------------------
    K_tiles = tl.cdiv(K, BLOCK_K)
    for tk in range(K_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a = tl.load(
            a_group + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # FP16 → promoted to FP32 inside tl.dot

        # B tile (K × N)
        b = tl.load(
            b_group + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # matrix‑multiply‑accumulate
        acc += tl.dot(a, b)                     # FP16·FP16 → FP32

    # --------------------------------------------------------------
    # write back results
    # --------------------------------------------------------------
    c = c_group + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, N, N, C] (FP32)
        - mask         : torch.Tensor   shape [B, N, N]  (bool / FP32)
        - weights      : dict of torch.Tensor, containing all linear / norm params
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (FP32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm on the input (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2️⃣  Half‑precision linear projections + sigmoidal gates
    # ------------------------------------------------------------------
    # Cast weights once to FP16 (speed‑up for the big GEMM)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"]          # keep FP32 for stability

    # Convert the normalized tensor to FP16 for the projections
    x_fp16 = x.to(torch.float16)

    # Projections (B,N,N,H)  –   FP16
    left  = F.linear(x_fp16, left_proj_w)   # → [B,N,N,H] (fp16)
    right = F.linear(x_fp16, right_proj_w)

    # Optional mask (broadcast over the hidden dim)
    if use_mask:
        # mask is [B,N,N] → [B,N,N,1] → fp16
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # Gates – compute in FP16, apply sigmoid (still FP16)
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # FP32, shape [B,N,N,H]

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Batched GEMM (core O(N³) work)
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis → shape (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) with (k,j) order

    B, H, N, _ = left_perm.shape                # B, H, N, N
    G = B * H                                    # total groups for the kernel

    # Collapse (B,H) → G  (no extra copy, view is cheap)
    A = left_perm.view(G, N, N)                  # (G, M=N, K=N)   fp16
    B_ = right_perm.view(G, N, N)                # (G, K=N, N)    fp16

    # Output buffer – accumulation in FP32
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # (G,N,N)

    # ------------------------------------------------------------------
    # Strides for the kernel (group, row, column)
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)  # = N*N
    stride_am = A.stride(1)  # = N
    stride_ak = A.stride(2)  # = 1

    stride_bg = B_.stride(0) # = N*N
    stride_bk = B_.stride(1) # = 1   (row = k)
    stride_bn = B_.stride(2) # = N   (col = j)

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ------------------------------------------------------------------
    # Tile size – 128×128×32 gives good occupancy on H100
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: one dimension for (M,N) tiles, second for groups
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)

    # Launch the Triton kernel
    _batched_gemm_kernel[grid](
        A,
        B_,
        C,
        N,               # M
        N,               # N
        N,               # K
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H] (still FP32)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣  Post‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                                   # FP32

    out = out * out_gate                                 # FP32 element‑wise

    output = F.linear(out, weights["to_out.weight"])    # → [B,N,N,C]  FP32
    return output
```
