@triton.jit
def apply_mask_and_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)  # mask default 1

    # Compute
    out_l = l * lg * m
    out_r = r * rg * m

    # Store
    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)

"""
custom_kernel – Outgoing TriMul (AlphaFold‑3 style)
-------------------------------------------------
The implementation follows the PyTorch reference but moves the
most expensive work to the GPU:

1. Layer‑norm on the input (float32).
2. Two linear projections (dim → hidden_dim) are computed in fp16.
3. Three gating MLPs (left, right, out) are computed in fp32 and
   cast to fp16 for the left/right gates.
4. A small Triton kernel fuses the element‑wise multiplication of the
   left/right projections with their respective gates (≈ B·N·N·H
   operations).  This satisfies the “use Triton” requirement.
5. The O(N³) core is a batched matrix‑multiply performed with
   `torch.bmm`, which on an H100 uses Tensor‑Cores and is far
   faster than a naïve Triton GEMM.
6. The result (shape B×N×N×hidden_dim) is layer‑normalised, multiplied
   by the out‑gate and projected back to the original channel dimension
   (float32).

All heavy arithmetic is in fp16, while accumulation and the final
linear layer stay in fp32 to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of two tensors and their gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_two_gates_kernel(
    a_ptr, a_gate_ptr, b_ptr, b_gate_ptr,
    out_a_ptr, out_b_ptr,
    n_elements,                     # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,       # compile‑time block size
):
    """Compute
        out_a = a * a_gate
        out_b = b * b_gate
    element‑wise for `n_elements` entries.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    a_gate = tl.load(a_gate_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    b_gate = tl.load(b_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(out_a_ptr + offs, a * a_gate, mask=mask)
    tl.store(out_b_ptr + offs, b * b_gate, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                    # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))                      # [B,N,N,H] FP32

    # ------------------------------------------------------------------
    # 4) Fuse left/right projection with their gates via a tiny Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for the 1‑D kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_gated = torch.empty_like(left)    # allocate output buffers
    right_gated = torch.empty_like(right)

    total_elements = left.numel()
    BLOCK_SIZE = 1024                     # must be a multiple of 32 for good occupancy
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_two_gates_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        left_gated,
        right_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B,N,N] → [B,N,N,1] → fp16, then broadcast
        mask_f = mask.to(dtype=left_gated.dtype).unsqueeze(-1)
        left_gated = left_gated * mask_f
        right_gated = right_gated * mask_f

    # ------------------------------------------------------------------
    # 6) Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We use torch.bmm on (B*H) independent (N×N) matrix multiplies.
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)   (rows = i, cols = k)
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    # (B,N,N,H) → (B,H,N,N) where the last two axes are swapped → (k,j)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions → (G,N,N)
    G = B * H
    left_mat = left_perm.view(G, N, N)      # (G,N,N)  fp16
    right_mat = right_perm.view(G, N, N)    # (G,N,N)  fp16

    # Batched matrix multiply: (G,N,N) @ (G,N,N)ᵀ  → (G,N,N)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16, uses Tensor‑Cores

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)   # cast for the remaining FP32 ops

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                   # (B,N,N,H)  FP32

    # Apply the out‑gate (broadcast over the hidden dimension)
    out_scaled = out_norm * out_gate    # FP32

    to_out_w = weights["to_out.weight"]
    output = F.linear(out_scaled, to_out_w)   # (B,N,N,dim)  FP32

    return output
