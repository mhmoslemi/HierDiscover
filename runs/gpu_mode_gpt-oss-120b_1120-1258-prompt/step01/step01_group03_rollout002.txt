"""
TriMul (outgoing) – fast FP16 implementation.

Algorithm
---------
1. Row‑wise LayerNorm over the channel dimension (`dim`) using a
   custom Triton kernel.  Reduction is performed in FP32 for numerical
   stability, while the output stays in FP16.

2. Fuse the two linear projections and their gates for the *left* and
   *right* branches:
        [proj, gate] = x_norm @ [W_proj; W_gate]ᵀ
        proj = proj * mask (if a mask is supplied)
        proj = proj * sigmoid(gate)
   All operations are carried out in FP16.

3. Triangular multiplicative update:
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   This is a batched matrix‑multiply L @ Rᵀ for every hidden channel.
   By reshaping `(B, H, N, N)` → `(B*H, N, N)` we can use a single
   `torch.bmm` call, which on H100 automatically maps to Tensor‑core
   GEMM (FP16).

4. Second LayerNorm (over the hidden dimension) – identical Triton
   kernel as step 1 but with `hidden_dim`.

5. Apply the output gate, a final FP16 linear projection back to `dim`,
   and cast the result to FP32.

All heavy reductions are done in Triton, the dense GEMM uses the
high‑performance cuBLAS kernels in FP16, and the linear layers are
fused to minimise memory traffic.  The function works for any
combination of batch‑size, sequence length, `dim` and `hidden_dim`
supported by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP16 input → FP16 output, reduction in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const *half*  [total_rows, dim]
    y_ptr,                # *mut   half*  [total_rows, dim]
    weight_ptr,           # const *half*  [dim]
    bias_ptr,             # const *half*  [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # fp32 (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor in FP16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows                               # bool mask for rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance (in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        # pointer to element (row, col)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)                                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                       # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)                # row‑wise sumsq

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1/√(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values with γ (weight) and β (bias)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)                                     # [BLOCK_M, BLOCK_D]

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]          # FP32

        # γ and β (FP16, broadcast over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=tl.float16(0.0))
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=tl.float16(0.0))

        # apply scale/shift and cast back to FP16
        out = (norm * w[None, :].to(tl.float32) + b[None, :].to(tl.float32)).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2️⃣ Prepare half‑precision weights (all linear layers are bias‑free)
    # ------------------------------------------------------------------
    w_norm_w = weights["norm.weight"].to(torch.float16)
    w_norm_b = weights["norm.bias"].to(torch.float16)

    # fuse proj & gate for left and right branches
    left_cat_w = torch.cat([weights["left_proj.weight"],
                            weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_cat_w = torch.cat([weights["right_proj.weight"],
                             weights["right_gate.weight"]], dim=0).to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # x: [B, N, N, dim]
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).to(torch.float16)   # [R, dim] FP16

    y_norm = torch.empty_like(x_flat)                       # output buffer FP16

    # launch Triton LayerNorm kernel
    BLOCK_M = 128   # rows per program
    BLOCK_D = 32    # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        y_norm,
        w_norm_w,
        w_norm_b,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape normalized tensor back to 4‑D (still FP16)
    x_norm = y_norm.view(B, N, N, dim)          # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections + gating (fused)
    # ------------------------------------------------------------------
    x_norm_flat = x_norm.view(total_rows, dim)  # [R, dim] FP16

    # left branch: [proj | gate]
    left_all = F.linear(x_norm_flat, left_cat_w)          # [R, 2*H] FP16
    left_proj = left_all[:, :hidden_dim]
    left_gate = torch.sigmoid(left_all[:, hidden_dim:])   # FP16

    # right branch: [proj | gate]
    right_all = F.linear(x_norm_flat, right_cat_w)        # [R, 2*H] FP16
    right_proj = right_all[:, :hidden_dim]
    right_gate = torch.sigmoid(right_all[:, hidden_dim:]) # FP16

    # out gate (single linear + sigmoid)
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))   # [R, H] FP16

    # optional mask (mask shape [B, N, N] → [R, 1])
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)  # broadcast
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # apply element‑wise gates
    left = left_proj * left_gate           # [R, H] FP16
    right = right_proj * right_gate        # [R, H] FP16

    # reshape for the triangular multiplication
    left  = left.view(B, N, N, hidden_dim)     # [B, N, N, H] FP16
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)   # keep for later

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # permute to [B, H, N, N] and collapse batch*H dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # [B, H, N, N] (Rᵀ)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # [B*H, N, N] FP16
    right_mat = right_perm.view(BH, N, N)     # [B*H, N, N] FP16

    # torch.bmm uses Tensor‑core GEMM for FP16 → very fast
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] FP16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)       # [R_h, H] FP16
    out_norm_flat = torch.empty_like(out_flat)          # FP16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise FP16

    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)   # [R_h, H] FP16
    final_half = F.linear(out_norm_flat, to_out_w)           # [R_h, dim] FP16
    final = final_half.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
