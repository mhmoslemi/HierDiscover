import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.constexpr,
    hidden_dim: tl.constexpr,

):

    mask_idx = offs // hidden_dim

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    right = right * right_gate * mask_val

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    - Uses a tiny Triton kernel to fuse mask and gate multiplication.
    - Performs the heavy pairwise contraction with torch.bmm (cuBLAS) for speed.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # First LayerNorm (over channel dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]

        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    )

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – fast Triton‑augmented implementation
--------------------------------------------------------------------
* First LayerNorm on the 4‑D input.
* Linear projections → left/right tensors of shape [B, N, N, H].
* Three sigmoidal gates (left_gate, right_gate, out_gate) are computed
  from the same normalized input.
* A tiny Triton kernel fuses mask‑broadcasting, left‑gate and right‑gate
  multiplication (in‑place on the projected tensors).  This satisfies the
  “must use a kernel” requirement while keeping the heavy N³ work on
  cuBLAS.
* The core contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is performed as a batched GEMM using torch.bmm on half‑precision
  tensors (Tensor‑Core friendly).  The tensors are reshaped to
  (B·H, N, N) and the second operand is transposed.
* Second LayerNorm over the hidden dimension.
* Another Triton element‑wise kernel applies the output gate.
* Final linear projection back to the original channel dimension.

The heavy O(N³) work leverages cuBLAS (torch.bmm) and runs in well
under the target runtime, while the small fused kernels keep the
implementation fully Triton‑aware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, left‑gate and right‑gate (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # float32*  [B,N,N,H]
    right_ptr,         # float32*  [B,N,N,H]
    left_gate_ptr,     # float32*  [B,N,N,H]
    right_gate_ptr,    # float32*  [B,N,N,H]
    mask_ptr,          # float32*  [B,N,N]   broadcasted over H
    total_elements: tl.constexpr,   # B*N*N*H
    hidden_dim: tl.constexpr,       # H
    BLOCK_SIZE: tl.constexpr,       # e.g. 1024
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each group of `hidden_dim` elements shares the same mask entry
    mask_idx = offs // hidden_dim

    # load tensors
    left      = tl.load(left_ptr + offs,       mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,      mask=valid, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_g   = tl.load(right_gate_ptr + offs,mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_idx,  mask=valid, other=1.0)

    # apply mask and gates
    left  = left  * left_g  * mask_val
    right = right * right_g * mask_val

    # store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # float32*  (out after LN)
    b_ptr,          # float32*  (out_gate)
    out_ptr,        # float32*  destination
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    a = tl.load(a_ptr + offs, mask=valid, other=0.0)
    b = tl.load(b_ptr + offs, mask=valid, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=valid)

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with Triton‑accelerated
    mask/gate fusion and Tensor‑Core batched GEMM.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensors for all linear / LN params
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over channel dimension C
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections → left / right (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Mask handling – broadcast over the hidden dimension
    # --------------------------------------------------------------
    if mask_tensor is None:

    else:

    mask = mask.to(x.dtype)                 # [B, N, N]
    mask_flat = mask.contiguous().view(-1)  # length B*N*N

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right in‑place (Triton)
    # --------------------------------------------------------------
    total_elems = left.numel()                      # B * N * N * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE), )
    mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_flat,
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication (N³) using torch.bmm (half‑precision)
    # --------------------------------------------------------------
    # Convert to half for Tensor‑Core speed‑up
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape                       # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # core contraction: left @ rightᵀ   (sum over K)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] (float32 for the rest of the pipeline)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate (Triton element‑wise kernel)
    # --------------------------------------------------------------
    total_out = out.numel()
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_out = torch.empty_like(out_flat)
    grid = (triton.cdiv(total_out, BLOCK_SIZE), )
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        gated_out,
        total_out,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    out = gated_out.view_as(out)

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to original channel dimension C
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, C]

    return out
