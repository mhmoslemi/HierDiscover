"""
custom_kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm over the input feature dimension (`dim`).
2. Fuse the five linear layers (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matrix multiplication:
        combined = X @ W_combined.T       # X : (B·N·N, dim)
   with   W_combined = [left_proj; right_proj;
                        left_gate; right_gate; out_gate]   (shape (5·H, dim)).
   The result is reshaped to (B, N, N, 5·H) and split into the five tensors.
3. Apply the optional mask (broadcast on the hidden dimension) and the three
   sigmoid gates (`left_gate`, `right_gate`, `out_gate`).  All tensors are kept
   in float16 to exploit Tensor‑Core arithmetic.
4. Compute the cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   for each (batch, hidden) pair as a batched GEMM C = A @ Bᵀ.
   A and B are stored as (M, N, N) with M = B·H and are processed by a
   Triton kernel that tiles the i, j and k dimensions
   (BLOCK_M × BLOCK_N × BLOCK_K).  The kernel loads tiles in fp16,
   accumulates in fp32, and writes the result back as fp32.
5. Layer‑norm over the hidden dimension, element‑wise multiplication with the
   output gate, and a final bias‑free linear projection back to `dim`.

All heavy arithmetic (the fused linear projection + the O(N³) GEMM) is executed
in half‑precision; the surrounding normalisations and the final projection are
performed in float32 to preserve numerical stability.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor  (M, N, N)  – fp16, layout (i,k)
    b_ptr,               # right tensor  (M, N, N)  – fp16, layout (j,k)
    c_ptr,               # output tensor (M, N, N) – fp32 accumulator
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,j,k) – we load Bᵀ as (k,j)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile along i
    pid_n = tl.program_id(1)            # tile along j
    pid_b = tl.program_id(2)            # batch * hidden index (M)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Pointer to the output tile C[i,j]
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A[i,k]  – (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ -> we read B[k,j] from the tensor that is stored as (j,k)
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Write the tile back
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused linear
    projection and a Triton‑accelerated batched GEMM.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # unpack weights (already on the correct CUDA device)
    # ------------------------------------------------------------------ #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1️⃣ Layer‑norm over the input features (float32)
    # ------------------------------------------------------------------ #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------ #
    # 2️⃣ Fuse the five linear projections into one matmul (half‑precision)
    # ------------------------------------------------------------------ #
    B, N, _, _ = x.shape
    # flatten spatial dims
    x_flat = x.view(-1, dim)                       # (B·N·N, dim)
    x_fp16 = x_flat.to(torch.float16)

    # stack all weights (shape (5·H, dim))
    combined_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    ).to(torch.float16)

    # single linear layer
    combined = torch.nn.functional.linear(x_fp16, combined_weight)  # (B·N·N, 5·H)
    # reshape back to (B, N, N, 5·H)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------ #
    # 3️⃣ Split, apply mask (if any) and sigmoid gates (still fp16)
    # ------------------------------------------------------------------ #
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: [B, N, N, H]

    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.to(left_proj.dtype).unsqueeze(-1)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 4️⃣ Triton GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------ #
    # bring hidden dimension to the second axis for easy tiling
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (j,k)

    B_, H_, N_, _ = left_perm.shape
    M_total = B_ * H_

    left_flat  = left_perm.view(M_total, N_, N_)   # (M, N, N)
    right_flat = right_perm.view(M_total, N_, N_)  # (M, N, N)
    out_flat   = torch.empty_like(left_flat, dtype=torch.float32)  # will hold (M,N,N)

    # -------------------- kernel launch parameters -------------------- #
    BLOCK_M = 64   # rows of the output tile
    BLOCK_N = 64   # cols of the output tile
    BLOCK_K = 32   # reduction tile

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over j dimension
    stride_bk = right_flat.stride(2)   # stride over k dimension

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N_ + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N_ + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M_total                           # one kernel per (batch, hidden)
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------ #
    # 5️⃣ Post‑processing: layer‑norm → out‑gate → final linear (float32)
    # ------------------------------------------------------------------ #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # out_gate still fp16 → broadcast casts to fp32
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
