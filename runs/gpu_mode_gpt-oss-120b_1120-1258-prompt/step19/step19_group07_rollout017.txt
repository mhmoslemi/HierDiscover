import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Outgoing TriMul forward pass implemented with Triton kernels.
    
    Steps:
    1. Row-wise LayerNorm on the input (float32 → float16).
    2. One fused linear projection produces left, right and three gate tensors.
    3. A Triton kernel splits the fused projection, applies the optional mask 
       and evaluates the three sigmoid gates, storing left, right and out_gate 
       in layout [B, H, N, N] (H = hidden_dim).
    4. Batched GEMM (torch.bmm) contracts left and right → out of shape 
       [B, H, N, N] (fp16).
    5. Row‑wise statistics (mean and rsqrt‑variance) of ``out`` are computed 
       with a Triton kernel.  They are required for the second LayerNorm.
    6. A final Triton kernel fuses the second LayerNorm, the out‑gate, and the 
       linear projection to the original ``dim`` producing a float‑32 tensor
       of shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # Import and unpack arguments
    # ------------------------------------------------------------------
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl
    
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16) -------------------------------------------------
    # ------------------------------------------------------------------
    @triton.jit
    def layernorm_kernel(
        x_ptr,    # const   *[rows, dim] (float32)
        y_ptr,    # mutable *[rows, dim] (float16)
        w_ptr,    # const   *[dim]      (float32)   γ
        b_ptr,    # const   *[dim]      (float32)   β
        rows: tl.int32,
        dim: tl.int32,
        eps: tl.float32,
        BLOCK_M: tl.constexpr,
        BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
        mask_m = offs_m < rows

        # ------------------------------------------------------------------
        # Compute mean & variance (fp32)
        # ------------------------------------------------------------------
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ------------------------------------------------------------------
        # Normalise, apply γ / β and store as FP16
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

            norm = (vals - mean[:, None]) * rstd[:, None]

            gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
            beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

            out = norm * gamma[None, :] + beta[None, :]              # fp32
            out_fp16 = out.to(tl.float16)

            out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptr, out_fp16,
                     mask=mask_m[:, None] & mask_d[None, :])

    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right and the three gates
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # shape (5*H, dim)

    proj = F.linear(x_norm, fused_w)            # (R, 5*H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    @triton.jit
    def split_fuse_gate_kernel(
        proj_ptr,       # const   *[R, 5*H]    (fp16)   fused projection
        left_ptr,       # mutable *[B, H, N, N] (fp16)
        right_ptr,      # mutable *[B, H, N, N] (fp16)
        out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
        mask_ptr,       # const   *[R]       (fp16)   (ignored if HAS_MASK=0)
        R: tl.int32,
        H: tl.int32,
        N: tl.int32,
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr,
        BLOCK_H: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
        mask_r = offs_r < R

        # decode flat index (b,i,k)
        nn = N * N
        b = offs_r // nn
        rem = offs_r - b * nn
        i = rem // N
        k = rem - i * N

        # optional mask (scalar)
        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
            mask_val = mask_val[:, None]                                   # broadcast H
        else:
            mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        hidden_stride = N * N
        base_offset = b * (H * hidden_stride) + i * N + k   # offset of hidden‑plane 0

        # process H dimension in chunks
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
            mask_h = offs_h < H

            # layout inside proj: [ left | right | left_gate | right_gate | out_gate ]
            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

            left_raw      = tl.load(proj_ptr + proj_base,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_raw     = tl.load(proj_ptr + proj_base + H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)

            # sigmoid in fp32 for stability, cast back to fp16
            left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            left   = left_raw  * mask_val * left_gate
            right  = right_raw * mask_val * right_gate
            outg   = out_gate   # not masked

            dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

            tl.store(left_ptr    + dst_offset, left,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr   + dst_offset, right,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr+ dst_offset, outg,
                     mask=mask_r[:, None] & mask_h[None, :])

    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(dtype=torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ) --------------------------------------------
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # back to (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics of `out` (mean & rsqrt‑var) -------------------------
    # ------------------------------------------------------------------
    @triton.jit
    def rowwise_stats_kernel(
        out_ptr,    # const *[B, H, N, N] (fp16)
        mu_ptr,     # mutable *[R]         (float32)
        rstd_ptr,   # mutable *[R]         (float32)
        total_rows: tl.int32,   # = B * N * N
        H: tl.int32,
        N: tl.int32,
        eps: tl.float32,
        BLOCK_M: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
        mask_m = offs_m < total_rows

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j          # offset of hidden‑plane 0

        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H
            ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M, K]
            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var  = sum_sq / H - mean * mean

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the reference PyTorch module but fuses the
expensive parts with custom Triton kernels:

1️⃣ Row‑wise LayerNorm (float32 → float16) – computes mean/var in FP32,
   writes the normalised tensor in FP16.

2️⃣ A single torch.nn.functional.linear computes a *fused* projection that
   contains left/right projections plus the three sigmoid gates.

3️⃣ `split_fuse_gate_kernel` splits the fused projection, applies the optional
   pair‑wise mask, evaluates the three sigmoid gates and stores ``left``,
   ``right`` and ``out_gate`` in layout [B, H, N, N] (H = hidden_dim).

4️⃣ Batched GEMM (`torch.bmm`) contracts ``left`` and ``right``:
   out[b, h] = left[b, h] @ right[b, h]ᵀ → shape (B·H, N, N) in FP16.

5️⃣ `rowwise_stats_kernel` computes per‑position mean and rsqrt‑variance of
   the contracted tensor (required for the second LayerNorm).

6️⃣ `final_output_kernel` fuses the second LayerNorm, the out‑gate and the
   final linear projection into a single kernel, producing a float‑32 tensor
   of shape [B, N, N, dim].

All kernels use FP16 arithmetic with FP32 accumulation for numerical
stability and are compatible with the H100 (triton‑3.3.1).

    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with LayerNorm / Linear parameters (float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    @triton.jit
    def layernorm_kernel(
        x_ptr,    # const   *[rows, dim] (float32)
        y_ptr,    # mutable *[rows, dim] (float16)
        w_ptr,    # const   *[dim]      (float32)   γ
        b_ptr,    # const   *[dim]      (float32)   β
        rows: tl.int32,
        dim: tl.int32,
        eps: tl.float32,
        BLOCK_M: tl.constexpr,
        BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
        mask_m = offs_m < rows

        # ---------- mean / var (FP32) ----------
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ---------- normalise, apply γ/β ----------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

            beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

            out = norm * gamma[None, :] + beta[None, :]              # fp32
            out_fp16 = out.to(tl.float16)

            out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptr, out_fp16,
                     mask=mask_m[:, None] & mask_d[None, :])

    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

        right_ptr,      # mutable *[B, H, N, N] (fp16)
        out_gate_ptr,   # mutable *[B, H, N, N] (fp16)

            mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
            mask_h = offs_h < H

            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

            left_raw      = tl.load(proj_ptr + proj_base,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_raw     = tl.load(proj_ptr + proj_base + H,
                                    mask=mask_r[:, None] & mask_h[None, :],

            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        eps: tl.float32,

        col_start = pid_n * BLOCK_N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j          # offset of hidden‑plane 0

        mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # Load out and gate (FP16 → FP32)
            ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

            out_vals  = tl.load(ptr_out,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)   # [M, K]
            gate_vals = tl.load(ptr_gate,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)   # [M, K]

            gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
            beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

            diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
            norm = diff * gamma[None, :] + beta[None, :]                # [M, K]

            term = norm * gate_vals                                    # [M, K]
            term_fp16 = term.to(tl.float16)                            # fp16 for dot

            # Load final weight tile (H × dim)
            w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
            w_tile = tl.load(w_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)                                 # [K, dim]

            acc += tl.dot(term_fp16, w_tile).to(tl.float32)             # [M, dim]

        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr, acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    # transpose final weight for efficient dot product
    w_out_T = w["to_out.weight"].to(dtype=torch.float16, device=device).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 64
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    out_flat = out.view(-1).contiguous()
    out_gate_flat = out_gate.view(-1).contiguous()
    final_output_kernel[grid_out](
        out_flat,
        out_gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # Return result (float32) – shape [B, N, N, dim]
    # ------------------------------------------------------------------
    return final
