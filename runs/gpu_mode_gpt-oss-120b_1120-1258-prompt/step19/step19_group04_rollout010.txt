1️⃣ Layer‑norm the flattened input (`x`) in FP32 → FP16.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM:   proj = x_norm @ W_fusedᵀ   →   (R, 5·H)  (FP16).
3️⃣ Split the fused projection, optionally apply the pair‑wise mask,
   compute sigmoid gates and store three tensors
   `left`, `right`, `out_gate`   [B, H, N, N] (FP16).  (Triton kernel)
4️⃣ Batched half‑precision GEMM (cuBLAS):
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (second LayerNorm).  (Triton kernel)
6️⃣ Fuse the second LayerNorm, `out_gate` multiplication and the final
   linear projection into a single Triton kernel, producing the final
   tensor `[B, N, N, dim]` (float32).

All heavy‑weight tensor traversals are done in Triton; the ordinary
linear layers and the batched GEMM use cuBLAS/ATen which are already
highly optimized on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b,i,k)
    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    k  = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m_val = m_val[:, None]   # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # hidden‑plane stride inside left/right/out_gate tensors
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # index of hidden‑plane 0

    # iterate over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R,H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid – stable version, operate in FP32 then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate   # already masked by sigmoid

        # destination offset for each hidden plane
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [R,H]

        tl.store(left_ptr     + dst_off,
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # reduce over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M,K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,      # [B, H, N, N] fp16   (contracted tensor)
    gate_ptr,    # [B, H, N, N] fp16   (out_gate)
    mu_ptr,      # [R] fp32
    rstd_ptr,    # [R] fp32
    ln_w_ptr,    # [H] fp32   (to_out_norm.weight)
    ln_b_ptr,    # [H] fp32   (to_out_norm.bias)
    w_out_T_ptr, # [H, dim] fp16  (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # offset of hidden‑plane 0

    # load per‑row mean / rstd (scalar per row)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    # accumulator for the final dim tile (float32)
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over hidden dimension in blocks
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # 1) contracted tensor (fp16 → fp32)
        ptr_out = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M,K]

        # 2) out‑gate (fp16 → fp32)
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # 3) second LayerNorm (γ,β are FP32)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M,K]

        term = norm * gate_vals                                    # [M,K]

        # 4) final projection Wᵀ : (K×dim)  (fp16)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, dim_tile]

        # dot: (M,K)·(K,dim_tile) → (M,dim_tile)   (fp16→fp32 accumulation)
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # write final result [B, N, N, dim] (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : {
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              }  (all float32, already on the target device)
            - config_dict  : must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (use ATen – fast and highly tuned)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                          # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for all five maps
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                             # (R,5·H) FP16
    del fused_w, x_norm                                           # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if not cfg.get("nomask", True):
        # mask shape [B, N, N] → flatten to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – never accessed because HAS_MASK=0
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128   # rows per program (tune for N≈1024)
    BLOCK_H = 32    # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) – cuBLAS works on contiguous FP16 tensors
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (each corresponds to a (b,i,j))
    BLOCK_K_OUT = 64       # hidden‑dim chunk
    BLOCK_N_OUT = 128      # output‑dim tile
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),   # rows
        triton.cdiv(dim, BLOCK_N_OUT), # output‑dim
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
