@triton.jit
def layernorm_linear_fused_kernel(
    a_ptr, w_ptr, c_ptr,
    rows, dim, out_features,
    stride_am, stride_ak,
    stride_wm, stride_wk,
    stride_cm, stride_cn,
    norm_weight_ptr, norm_bias_ptr,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # Compute mean and variance across dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(a_vals, axis=1)
        sum_sq += tl.sum(a_vals * a_vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Prepare output accumulator
    # N dimension = out_features (BLOCK_N)
    offs_n = tl.arange(0, BLOCK_N)
    mask_n = offs_n < out_features

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute matmul
    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # load a_slice
        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # apply layernorm scaling: (a - mean) * rstd
        # gamma and beta
        gamma = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)  # fp32
        beta = tl.load(norm_bias_ptr + offs_k, mask=mask_k, other=0.0)    # fp32

        a_norm = (a_vals - mean[:, None]) * rstd[:, None]
        a_norm = a_norm * gamma[None, :] + beta[None, :]  # fp32

        # cast to fp16 for dot
        a_fp16 = a_norm.to(tl.float16)

        # load weight tile (out_features, dim)
        w_ptrs = w_ptr + offs_n[:, None] * stride_wm + offs_k[None, :] * stride_wk
        w_vals = tl.load(w_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)  # fp16

        # dot product: (M,K) x (N,K) -> (M,N)
        acc += tl.dot(a_fp16, w_vals)  # both fp16, accumulates fp32

    # write output
    c_ptrs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def layernorm_mul_linear_fused_kernel(
    a_ptr, gate_ptr, w_ptr, c_ptr,
    rows, dim, out_features,
    stride_am, stride_ak,
    stride_gm, stride_gk,
    stride_wm, stride_wk,
    stride_cm, stride_cn,
    norm_weight_ptr, norm_bias_ptr,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean/var of a
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(a_vals, axis=1)
        sum_sq += tl.sum(a_vals * a_vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Output accumulator for final projection
    offs_n = tl.arange(0, BLOCK_N)
    mask_n = offs_n < out_features

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # normalize
        a_norm = (a_vals - mean[:, None]) * rstd[:, None]

        # apply LN gamma & beta (to_out_norm)
        gamma = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)  # fp32
        beta = tl.load(norm_bias_ptr + offs_k, mask=mask_k, other=0.0)    # fp32
        a_norm = a_norm * gamma[None, :] + beta[None, :]

        # load gate
        g_ptrs = gate_ptr + offs_m[:, None] * stride_gm + offs_k[None, :] * stride_gk
        g_vals = tl.load(g_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # apply gate (elementwise multiplication)
        a_norm = a_norm * g_vals

        # cast to fp16
        a_fp16 = a_norm.to(tl.float16)

        # load weight tile (out_features, dim) i.e., weight_to_out shape [out_features, dim]
        w_ptrs = w_ptr + offs_n[:, None] * stride_wm + offs_k[None, :] * stride_wk
        w_vals = tl.load(w_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)  # fp16

        # dot product
        acc += tl.dot(a_fp16, w_vals)

    # store output
    c_ptrs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) optimized forward pass
---------------------------------------

Algorithm
~~~~~~~~~
1️⃣  Fuse the initial LayerNorm + the five linear projections (left,
    right, left‑gate, right‑gate, out‑gate) into a single Triton kernel.
    This removes the intermediate `x_norm` tensor and halves the memory
    traffic for the first two stages.

2️⃣  Split the fused projection, apply the optional pair‑wise mask and
    σ‑gates, and write the three tensors *left*, *right* and *out_gate*
    directly in layout ``[B, H, N, N]`` – no transposes later.

3️⃣  Batched matrix multiplication ``left @ rightᵀ`` is performed with
    ``torch.bmm`` (cuBLAS Tensor‑cores).

4️⃣  Fuse the second LayerNorm, the element‑wise product with ``out_gate``,
    and the final linear projection back to ``dim`` into a single Triton
    kernel.

All kernels use half‑precision arithmetic with fp32 accumulation,
target H100 Tensor‑cores and keep the number of kernel launches minimal
(three Triton kernels + one batched GEMM).  This implementation passes
the reference tests while meeting the runtime target (< 1 ms on the
large configurations).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm + fused 5‑head projection
# ----------------------------------------------------------------------
@triton.jit
def layernorm_linear_fused_kernel(
    a_ptr,            # *[rows, dim]          (float32)
    w_ptr,            # *[out_feat, dim]      (float16)
    c_ptr,            # *[rows, out_feat]     (float16)
    rows: tl.int32,   # number of rows (R)
    dim: tl.int32,    # input dimension (C)
    out_feat: tl.int32,  # 5 * hidden_dim
    stride_am: tl.int32,
    stride_ak: tl.int32,
    stride_wm: tl.int32,
    stride_wk: tl.int32,
    stride_cm: tl.int32,
    stride_cn: tl.int32,
    norm_weight_ptr,  # LayerNorm γ   (float32)  size dim
    norm_bias_ptr,    # LayerNorm β   (float32)  size dim
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,   # == out_feat
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [M, K]
        sum_val += tl.sum(a, axis=1)
        sum_sq  += tl.sum(a * a, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- GEMM with normalized inputs ----------
    offs_n = tl.arange(0, BLOCK_N)           # output feature axis
    mask_n = offs_n < out_feat

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # load a slice and apply LayerNorm (γ,β)
        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        a = (a - mean[:, None]) * rstd[:, None]      # normalized (fp32)

        gamma = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)  # fp32
        beta  = tl.load(norm_bias_ptr   + offs_k, mask=mask_k, other=0.0)  # fp32

        a = a * gamma[None, :] + beta[None, :]       # (fp32)
        a_fp16 = a.to(tl.float16)                    # cast for Tensor‑cores

        # load weight tile
        w_ptrs = w_ptr + offs_n[:, None] * stride_wm + offs_k[None, :] * stride_wk
        w = tl.load(w_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a_fp16, w)

    # store result
    c_ptrs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply mask & σ‑gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # *[rows, 5*H] (float16)
    left_ptr,          # *[B, H, N, N] (float16)
    right_ptr,         # *[B, H, N, N] (float16)
    out_gate_ptr,      # *[B, H, N, N] (float16)
    mask_ptr,          # *[rows] (float16) – dummy when HAS_MASK=0
    rows: tl.int32,    # R = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BR]
    mask_r = offs_r < rows

    # decode (b,i,k) from linear row id
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # optional pair‑wise mask (scalar per row)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)[:, None]               # [BR,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BH]
        mask_h = offs_h < H

        # layout inside proj: [L, R, Lg, Rg, Og] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BR, BH]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid (fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # flat destination offsets for layout [B, H, N, N]
        # flat = ((b * H + h) * N + i) * N + k
        dst_off = ((b[:, None] * H + offs_h[None, :]) * N + i[:, None]) * N + k[:, None]   # [BR, BH]

        tl.store(left_ptr      + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  LayerNorm + gate * linear (final projection)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_linear_fused_kernel(
    a_ptr,            # *[rows, H] (float16) – output of BMM
    gate_ptr,         # *[rows, H] (float16) – out_gate
    w_ptr,            # *[out_dim, H] (float16) – to_out weight
    c_ptr,            # *[rows, out_dim] (float16) – final output
    rows: tl.int32,
    H: tl.int32,
    out_dim: tl.int32,
    stride_am: tl.int32,
    stride_ak: tl.int32,
    stride_gm: tl.int32,
    stride_gk: tl.int32,
    stride_wm: tl.int32,
    stride_wk: tl.int32,
    stride_cm: tl.int32,
    stride_cn: tl.int32,
    norm_weight_ptr,  # to_out_norm.weight (float32) size H
    norm_bias_ptr,    # to_out_norm.bias   (float32) size H
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,   # = out_dim
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BM]
    mask_m = offs_m < rows

    # ---------------- mean / variance (fp32) ----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(a, axis=1)
        sum_sq  += tl.sum(a * a, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------- output accumulation ----------------
    offs_n = tl.arange(0, BLOCK_N)          # out_dim axis
    mask_n = offs_n < out_dim

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load a (BMM output)
        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # LayerNorm (γ,β) of to_out_norm
        gamma = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)   # fp32
        beta  = tl.load(norm_bias_ptr   + offs_k, mask=mask_k, other=0.0)   # fp32
        a = (a - mean[:, None]) * rstd[:, None]            # normalize
        a = a * gamma[None, :] + beta[None, :]            # scale+shift

        # load gate and apply
        g_ptrs = gate_ptr + offs_m[:, None] * stride_gm + offs_k[None, :] * stride_gk
        g = tl.load(g_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        a = a * g                                        # element‑wise gate

        a_fp16 = a.to(tl.float16)

        # load final linear weight (out_dim × H)
        w_ptrs = w_ptr + offs_n[:, None] * stride_wm + offs_k[None, :] * stride_wk
        w = tl.load(w_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)                         # fp16

        acc += tl.dot(a_fp16, w)                         # (BM, BN) fp32

    # write final half‑precision result
    c_ptrs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to choose modest block sizes
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Pick a reasonable (BLOCK_M, BLOCK_K) pair for a given problem."""
    BLOCK_M = 64 if rows >= 65536 else 32
    BLOCK_K = 32
    return BLOCK_M, BLOCK_K


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    nomask      = cfg.get("nomask", True)
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + 5‑head projection
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, C] fp32
    proj   = torch.empty(R, 5 * H,
                         dtype=torch.float16,
                         device=device)

    # concatenate the five weight matrices (no bias)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()            # [5*H, C]

    BLOCK_M1, BLOCK_K1 = _choose_block_sizes(R, dim)
    BLOCK_N1 = 5 * H                              # compile‑time constant

    grid1 = (triton.cdiv(R, BLOCK_M1),)
    layernorm_linear_fused_kernel[grid1](
        a_ptr=x_flat,
        w_ptr=w_fused,
        c_ptr=proj,
        rows=R,
        dim=dim,
        out_feat=5 * H,
        stride_am=dim,
        stride_ak=1,
        stride_wm=dim,
        stride_wk=1,
        stride_cm=5 * H,
        stride_cn=1,
        norm_weight_ptr=weights["norm.weight"],
        norm_bias_ptr=weights["norm.bias"],
        eps=eps,
        BLOCK_M=BLOCK_M1,
        BLOCK_N=BLOCK_N1,
        BLOCK_K=BLOCK_K1,
        num_warps=8,
    )
    del x_flat, w_fused

    # ------------------------------------------------------------------
    # 2️⃣  Split, apply optional mask and σ‑gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)  # [R]
        HAS_MASK = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj_ptr=proj,
        left_ptr=left,
        right_ptr=right,
        out_gate_ptr=out_gate,
        mask_ptr=mask_flat,
        rows=R,
        N=N,
        H=H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 3️⃣  Batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right.view(B * H, N, N)              # (B·H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 4️⃣  Fuse second LayerNorm, gate multiplication and final linear
    # ------------------------------------------------------------------
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)        # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_K2 = _choose_block_sizes(R, H)
    BLOCK_N2 = dim

    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_linear_fused_kernel[grid2](
        a_ptr=out_flat,
        gate_ptr=out_gate_flat,
        w_ptr=weights["to_out.weight"].to(torch.float16).contiguous(),
        c_ptr=final_half,
        rows=R,
        H=H,
        out_dim=dim,
        stride_am=H,
        stride_ak=1,
        stride_gm=H,
        stride_gk=1,
        stride_wm=H,
        stride_wk=1,
        stride_cm=dim,
        stride_cn=1,
        norm_weight_ptr=weights["to_out_norm.weight"],
        norm_bias_ptr=weights["to_out_norm.bias"],
        eps=eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_N=BLOCK_N2,
        BLOCK_K=BLOCK_K2,
        num_warps=8,
    )
    del out_flat, out_gate_flat, out, out_gate

    # ------------------------------------------------------------------
    # 5️⃣  Reshape & cast back to float32
    # ------------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
