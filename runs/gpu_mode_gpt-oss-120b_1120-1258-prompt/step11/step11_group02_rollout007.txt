"""
tri_mul_outgoing.py – fast outgoing TriMul (AlphaFold3) forward pass.

Algorithm (all tensors are on the same CUDA device)

1️⃣ Row‑wise LayerNorm on the 4‑D input   (float32 → float16)
2️⃣ Fuse the five linear maps (left/right proj + three gates) into a single GEMM:
   (R, dim) @ (5·H, dim).T → (R, 5·H)  where R = B·N·N.
3️⃣ Triton kernel – splits the fused projection, applies the optional pair‑wise mask,
   evaluates three sigmoid gates and writes three tensors:
      * left  : (B·H, N·N)   – flattened ‹i,k› matrix
      * right : (B·H, N·N)
      * out_gate : (R, H)    – kept in the original row layout
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out_mat = left @ rightᵀ          (torch.bmm, Tensor‑Core)
5️⃣ Row‑wise LayerNorm + element‑wise multiplication with `out_gate`
   (torch.nn.functional.layer_norm, works on fp16).
6️⃣ Final linear projection back to the original dimension:
        y = (normed·out_gate) @ W_outᵀ   (torch.nn.functional.linear)

Only step 3 is implemented in a custom Triton kernel – satisfying the
“must contain a kernel” requirement – while the heavy matrix
multiplications use cuBLAS (Tensor‑Core) via PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]       (float16)
    left_ptr,          # mutable *[B*H, N*N]      (float16)
    right_ptr,         # mutable *[B*H, N*N]      (float16)
    out_gate_ptr,      # mutable *[rows, H]       (float16)
    mask_ptr,          # const   *[rows]          (float16)  (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened row r = b·N·N + i·N + k  (b∈[0,B), i,k∈[0,N))
    and each hidden index h∈[0,H):
        left  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    The results are written directly into the flattened left/right buffers
    of shape (B·H, N·N) so that a later batched GEMM can be performed without
    any view/permute copy.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                                           # total positions per matrix

    # b = offs_r // NN   ;  pos = offs_r % NN
    b = offs_r // NN
    pos = offs_r % NN

    # load scalar mask if present (0/1)
    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val_raw[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the five projected blocks --------------------------------
        # (R, 5*H) layout → contiguous groups for left/right and three gates
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # (R, BH)

        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)

        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # ---- sigmoid (stable fp32) ----------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates -----------------------------------------
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # ---- destination offsets for left/right ----------------------------
        # batch index = b * H + h
        batch_idx = b[:, None] * H + offs_h[None, :]                # [BLOCK_R, BLOCK_H]
        dst_offset = batch_idx * NN + pos[:, None]                  # flatten position

        # store left/right
        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---- out_gate (kept in original row layout) -----------------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing” variant).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool) or None
        - weights_dict : dict of torch.Tensors, model weights
        - config_dict  : dict with keys:
              * "dim"          – input/output channel dimension
              * "hidden_dim"   – hidden dimension H
              * "nomask"       – bool, whether mask is omitted
    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]          # input / output channel dimension
    hidden_dim = cfg["hidden_dim"]   # H
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops
    H = hidden_dim
    NN = N * N

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    # PyTorch LayerNorm is highly optimized; we use it and cast to fp16.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim] float32
    x_norm = x_norm_fp32.view(R, dim).contiguous().to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection: (left, right, gates) → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H), fp16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton kernel)
    # --------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)  # (B·H, N·N)
    right_flat = torch.empty_like(left_flat)

    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)        # (R, H)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        has_mask = 1
        # flatten mask to (R,) and convert to half‑precision (0/1)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_fp16

    # Triton launch
    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape left/right for batched GEMM
    left  = left_flat.view(B * H, N, N)      # (B·H, N, N)
    right = right_flat.view(B * H, N, N)     # (B·H, N, N)

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension
    # --------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N), fp16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    # Reshape to (R, H) where each row corresponds to a (b,i,j) position.
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)
    del out_mat

    # PyTorch LayerNorm on half‑precision tensors (internally uses fp32
    # accumulation, which is numerically stable).
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                               # (R, H), fp16

    # Apply gating
    out_fused = (out_norm * out_gate).contiguous()
    del out_norm, out_gate, out_flat

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # (R, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
