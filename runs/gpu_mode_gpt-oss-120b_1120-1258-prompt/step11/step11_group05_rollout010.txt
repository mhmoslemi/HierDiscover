def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden = cfg["hidden_dim"]; eps = 1e-5; device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    # Convert input to half
    x_h = x.to(torch.float16)
    # First layernorm
    x_norm = F.layer_norm(x_h, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps) # fp16
    # Flatten for linear
    x_norm_flat = x_norm.view(R, dim)
    # Fuse weights
    w_fused = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"], weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).to(torch.float16) # [5*H, dim]
    # Linear projection
    proj = F.linear(x_norm_flat, w_fused) # (R,5*H) half
    # Allocate outputs
    left = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden, dtype=torch.float16, device=device)
    # mask handling
    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)
    # launch split_fuse_gate_kernel
    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free proj
    del proj
    # Reshape left/right for batched gemm
    left4 = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right4 = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    left_mat = left4.view(B * hidden, N, N)   # [B*H, N, N]
    right_mat = right4.view(B * hidden, N, N)
    # Matmul
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))  # [B*H,N,N] half
    # reshape
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    # Second layernorm + gating
    out_norm = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps) # fp16
    out_gate4 = out_gate.view(B, N, N, hidden)  # [B,N,N,H]
    out_fused = out_norm * out_gate4
    # Final projection
    final_half = F.linear(out_fused, weights["to_out.weight"].to(torch.float16))  # [B,N,N,dim] half
    final = final_half.to(torch.float32)
    return final
```

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    rows, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Load mask per row if needed.
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        mask_val = mask_val[:, None]  # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # gather base offsets
        # each row has 5*H elements in proj. layout: left, right, left_gate, right_gate, out_gate
        # Compute pointer offsets for each type
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]
        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)
        # Apply mask and gates
        left_out = left_raw * mask_val * left_gate
        right_out = right_raw * mask_val * right_gate

        # Store
        tl.store(left_ptr + offs_r[:, None] * H + offs_h[None, :],
                 left_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm:
1. Cast input to fp16 and apply LayerNorm (torch.nn.functional.layer_norm) – fp16 output.
2. Fuse the five linear maps (left/right proj + three gates) into a single
   matmul with a concatenated weight matrix of shape (5·H, C).
3. Triton kernel `split_fuse_gate_kernel` splits the fused result into
   left, right and out_gate tensors, applies the optional pair‑wise mask
   and sigmoid gates, producing three fp16 tensors of shape (R, H)
   where R = B·N·N.
4. Reshape left/right to (B·H, N, N) and perform a batched GEMM
   (Tensor‑Core) `left @ rightᵀ` → (B·H, N, N) (fp16).
5. Apply second LayerNorm to the result and multiply element‑wise by
   out_gate (both fp16) using torch.nn.functional.layer_norm.
6. Final linear projection back to the original channel dimension.

Only the mask‑aware split‑and‑gate step is custom Triton code; all heavy
matrix multiplications are delegated to cuBLAS/Tensor‑Cores via PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Split the fused projection (shape [rows,5*H]) into three tensors:
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)
    All ops are performed in fp16; sigmoid evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- load per‑row mask (if any) ----------
    if HAS_MASK:
        # mask values are 0/1 in fp16, broadcast over the hidden dimension
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]                     # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---------- iterate over hidden dimension ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw    = tl.load(proj_ptr + base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid in fp32, back to fp16 -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store -----
        tl.store(left_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : dict containing 'dim', 'hidden_dim', 'nomask' (bool)

    Returns

    torch.Tensor

    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N  # rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Cast to fp16 and apply first LayerNorm (torch implementation)
    # ------------------------------------------------------------------
    x_half = x.to(torch.float16)  # [B, N, N, C] fp16
    x_norm = F.layer_norm(
        x_half,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps into a single matmul
    # ------------------------------------------------------------------
    # flatten for linear
    x_norm_flat = x_norm.view(R, dim)                     # [R, C] fp16
    # concatenate weights: each weight has shape [hidden, dim]
    w_fused = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, C]

    # fused projection (R, 5*H)  – uses cuBLAS, fp16
    proj = F.linear(x_norm_flat, w_fused)                 # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (custom Triton kernel)
    # ------------------------------------------------------------------
    left = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj  # free memory

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension: left @ rightᵀ
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) then to (B, H, N, N)
    left_4d = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    right_4d = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()

    left_mat = left_4d.view(B * hidden, N, N)   # [B*H, N, N]
    right_mat = right_4d.view(B * hidden, N, N) # [B*H, N, N]

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)   fp16 matmul uses TensorCores
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication by out_gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # fp16

    out_gate_4d = out_gate.view(B, N, N, hidden)
    out_fused = out_norm * out_gate_4d   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, hidden]
    )  # fp16

    # Cast back to float32 as required by the specification
    return final_half.to(torch.float32)

    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                       # fp16

    out_gate_4d = out_gate.view(B, N, N, hidden)
    out_fused = out_norm * out_gate_4d       # fp16
    del out_norm, out_gate, out_gate_4d

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, hidden]
    )                                           # fp16

    # cast to float32 for the API contract
    return final_half.to(torch.float32)
```
