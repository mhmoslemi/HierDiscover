"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM (R × dim) @ (5·H × dim)ᵀ → (R × 5·H)  (fp16).
3️⃣ Triton kernel `split_fuse_gate_fused_kernel`
   * extracts the five slices,
   * (optionally) multiplies by the pair‑wise mask,
   * evaluates sigmoid gates (fp32 → fp16),
   * writes *left* and *right* directly in (B·H, N, N) layout,
   * stores the output‑gate tensor for the second LayerNorm.
4️⃣ Batched half‑precision GEMM (cuBLAS/Tensor‑Cores):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
5️⃣ Triton kernel `layernorm_mul_strided_kernel` performs the second
   Row‑wise LayerNorm **in‑place** on the GEMM result (the hidden dimension is
   strided) and multiplies the normalized values by the previously computed
   `out_gate`.
6️⃣ Final linear projection back to ``dim`` (fp16 → fp32).

Only the two LayerNorms and the split/fuse gate logic are written in Triton;
all heavy matrix multiplications use cuBLAS (through ``torch.bmm``) which
leverages Tensor‑Cores on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values: γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates,
#       write left/right in (B·H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each (b,i,j) entry the fused projection yields five slices:
      left, right, left_gate, right_gate, out_gate.
    The kernel applies the optional mask, the sigmoid gates (computed in fp32
    for numerical stability) and writes the left/right tensors straight into
    the layout required for the subsequent batched GEMM:
        (B·H, N, N) where the first dimension combines batch and hidden channel.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # Compute (b,i,j) from flat row index
    # --------------------------------------------------------------
    seqsq = N * N                     # N²
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # --------------------------------------------------------------
    # Optional mask (broadcast across hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # broadcast => [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # Loop over hidden dimension blocks
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----------- load the five projected slices -------------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        left_raw      = tl.load(proj_ptr + base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_raw     = tl.load(proj_ptr + base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        # ----------- sigmoid gates (fp32) -----------------------
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----------- apply mask + gates -------------------------
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ----------- write left/right into (B·H, N, N) ----------
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----------- store out_gate (rows, H) -------------------
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (float16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (float16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (float16)   (final fused output)
    weight_ptr,     # const   *[H]         (float32)   γ
    bias_ptr,       # const   *[H]         (float32)   β
    rows: tl.int32,     # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each (b,i,j) the hidden dimension H is stored with stride N².
    This kernel
        1) computes mean/variance over H (fp32),
        2) normalizes, applies γ/β,
        3) multiplies by the previously computed out_gate,
        4) writes the result as a contiguous (rows, H) tensor (fp16).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # ---------- decode (b,i,j) ----------
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset for the first hidden slice (h = 0)
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ accumulate mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply with out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ, β (fp32)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # [Hblk]
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # [Hblk]

        norm = norm * w[None, :] + b[None, :]               # fp32

        # out_gate (fp16 → fp32)
        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32, [dim])
        weights["norm.bias"],            # β (float32, [dim])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat   # free early

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # --------------------------------------------------------------
    # Concatenate all five weight matrices (shape [5*H, dim]) and cast to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # [5*H, dim]

    proj = F.linear(x_norm, w_concat)                       # (R, 5*H) fp16
    del w_concat, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split fused result, apply mask & sigmoid gates,
    #    write left/right directly in (B·H, N, N) layout
    # --------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16,
                            device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0,
                                 dtype=torch.float16,
                                 device=device)  # dummy
    else:
        # flatten to [R]; cast to 0./1. fp16
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:
    #     out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # --------------------------------------------------------------
    # left/right are (B*H, N, N) fp16 → use torch.bmm which maps to cuBLAS/Tensor‑Cores
    out_mat = torch.bmm(left, right.transpose(1, 2))       # (B*H, N, N) fp16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_lnorm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_strided_kernel[grid_lnorm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_D,                 # reuse D size for hidden dim blocks
        num_warps=4,
    )
    del out_mat, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
