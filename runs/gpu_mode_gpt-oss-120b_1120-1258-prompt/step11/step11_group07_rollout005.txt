"""
Highly‑optimized Triton implementation of the **outgoing** Triangular‑Multiplicative
update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape B·N·N × C) → fp16.
2️⃣ Fuse the 5 linear projections (left/right + three gates) into a single
   GEMM:  X_norm @ W_fusedᵀ  → (R, 5·H)   where R = B·N·N, H = hidden_dim.
3️⃣ Triton kernel splits the fused result, applies the optional pairwise mask,
   computes three sigmoid gates and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
   (all arithmetic in fp16, sigmoid evaluated in fp32 for stability)
4️⃣ Batched half‑precision matrix‑multiply over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   (implemented with `torch.bmm`, i.e. cuBLAS Tensor‑Core kernels)
5️⃣ Row‑wise LayerNorm on the intermediate tensor + element‑wise
   multiplication with `out_gate` (fused Triton kernel).
6️⃣ Final linear projection back to `dim` (half‑precision GEMM) and cast to fp32.

Only the two LayerNorm passes and the split‑gate pass are written in Triton,
all heavy matrix‑multiplications use cuBLAS which gives near‑peak performance on
an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const  *[rows, dim]   (float32)
    y_ptr,      # mutable*[rows, dim]  (float16)
    w_ptr,      # const  *[dim]        (float32)   γ
    b_ptr,      # const  *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalized output  γ·(x‑μ)·rstd + β  (store fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out = norm * w[None, :] + b[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: split fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened pair (b,i,j):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic fp16, sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask value (scalar per row) if present
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Process hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection:
        # [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw_ptr      = proj_ptr + base
        right_raw_ptr     = proj_ptr + base + H
        left_gate_ptr     = proj_ptr + base + 2 * H
        right_gate_ptr    = proj_ptr + base + 3 * H
        out_gate_raw_ptr  = proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(right_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(left_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # Store results
        left_dst_ptr      = left_ptr      + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr     = right_ptr     + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr  = out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (fp32)
        norm = norm * w[None, :] + b[None, :]               # fp32

        # gate
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * g                                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper: simple heuristic for block sizes
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Return (BLOCK_M, BLOCK_D) tuned for the current problem size."""
    BLOCK_M = 128 if rows >= 128 else max(1, rows)
    BLOCK_D = 32 if dim >= 32 else max(1, dim)
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (all torch.float32 tensors)
                         keys:
                           "norm.weight", "norm.bias",
                           "left_proj.weight", "right_proj.weight",
                           "left_gate.weight", "right_gate.weight", "out_gate.weight",
                           "to_out_norm.weight", "to_out_norm.bias",
                           "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (left/right + 3 gates) → (R, 5·H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]
    proj = F.linear(x_norm, w_fused)                     # fp16, shape (R, 5·H)
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ split, apply mask, sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask → float16, flatten to [R]
        mask_half = mask.to(torch.float16).contiguous().view(R)
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to bring H into batch axis
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # batched matrix multiply: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + multiplication by out_gate (fused kernel)
    # ------------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()          # [R, H] fp16
    out_gate_f = out_gate.view(R, H).contiguous()    # [R, H] fp16
    out_fused  = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_f,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate_f, out

    # ------------------------------------------------------------------
    # 6️⃣ final linear projection back to `dim` (half‑precision GEMM) → fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # shape (R, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
