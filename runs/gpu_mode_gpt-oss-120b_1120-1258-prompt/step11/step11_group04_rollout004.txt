"""
TriMul (outgoing) – fast Triton + cuBLAS forward implementation
================================================================

Algorithm (all heavy work is on the GPU)
----------------------------------------
1️⃣  **Row‑wise LayerNorm** on the input tensor (B·N·N × dim), converting
    FP32 → FP16.  Implemented in ``layernorm_kernel``.
2️⃣  **Fused Linear** – concatenate the five linear transforms
    (left/right projections + three gates) into one weight matrix
    (5·H × dim) and apply it with a single half‑precision GEMM
    (cuBLAS).  The result has shape (R, 5·H) with ``R = B·N·N``.
3️⃣  **Split / Mask / Gate** – a Triton kernel rewrites the fused output
    directly into the layout required for the next batched GEMM:
    ``left``, ``right`` and ``out_gate`` of shape (B, H, N, N).  The mask
    (if present) and sigmoid gates are applied on‑the‑fly.
4️⃣  **Tri‑multiplicative update** – a batched half‑precision GEMM
    (``torch.bmm``) computes for every hidden slice ``h``:
        out[b,h] = left[b,h] @ right[b,h]ᵀ    (shape N×N)
5️⃣  **Second LayerNorm + out‑gate** – fused in ``layernorm_mul_kernel``.
    Operates on rows (B·N·N × H) and multiplies the normalized result by
    the previously computed ``out_gate``.
6️⃣  **Final projection** – a half‑precision linear layer maps the hidden
    dimension back to ``dim``.  The output is converted to FP32.

Only three Triton kernels are used; all large matrix multiplications are
delegated to cuBLAS Tensor‑Core kernels, giving near‑peak performance on
an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   *[rows, dim]  (float32)
    y_ptr,        # mutable *[rows, dim]  (float16)
    w_ptr,        # const   *[dim]       (float32)   γ
    b_ptr,        # const   *[dim]       (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split / mask / sigmoid gates (write directly into (B,H,N,N) layout)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R] (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flat row r = b·N·N + i·N + k (0 ≤ r < R) we produce:
        left[b,h,i,k]  = (proj_left  * mask) * sigmoid(left_gate)
        right[b,h,i,k] = (proj_right * mask) * sigmoid(right_gate)
        out_gate[b,h,i,k] = sigmoid(proj_out_gate)
    All ops are half‑precision except the sigmoid (computed in fp32 for
    stability).  The tensors are written directly in the (B,H,N,N) layout
    required by the subsequent batched GEMM, thus avoiding any transpose.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask handling (scalar per row)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                 # distance between consecutive h slices

    # base offset common to all h in the row (points to h==0)
    base_offset = b * (H * N * N) + i * N + k           # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load fused projection slices ----------------------------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ---- sigmoid in fp32 -----------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates -------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store -----------------------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)
    y_ptr,            # mutable *[rows, dim]   (fp16)
    w_ptr,            # const   *[dim]        (float32)  γ
    b_ptr,            # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (fp16) followed by element‑wise multiplication
    with ``gate_ptr``.  The result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate ------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (fp16 → fp32)
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd   (fp32)
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ, β (fp32)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        norm = norm * w[None, :] + b[None, :]

        # gate value (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                      # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    else:
        # Convert mask to 0/1 half‑precision and flatten to [R]
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256      # rows per program
    BLOCK_H = 64       # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    # Shape (B·H, N, N) – contiguous in memory
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out_mat[b·H, i, j] = Σ_k left[b·H,i,k] * right[b·H,j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to (B, H, N, N)
    out = out_mat.view(B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate (fused)
    # ------------------------------------------------------------------
    # Rearrange to (B·N·N, H) for row‑wise processing
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    out_flat = out_perm.view(R, H).contiguous()
    gate_flat = gate_perm.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        gate_flat,
        out_fused,
        w["to_out_norm.weight"],   # γ, float32
        w["to_out_norm.bias"],     # β, float32
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, gate_flat, out, out_gate, out_perm, gate_perm

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to ``dim`` (FP16 → FP32)
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # -> [R, dim], fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
