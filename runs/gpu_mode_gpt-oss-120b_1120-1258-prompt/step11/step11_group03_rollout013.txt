"""
custom_kernel
==============

Highly‑optimized forward pass for the *outgoing* TriMul operator used in
AlphaFold‑3, Chai, Protenix … (no grads required).

Algorithm
---------
1️⃣ Layer‑Norm on the input tensor (FP32 → FP16) – done with
   ``torch.nn.functional.layer_norm`` (cuDNN‑accelerated).

2️⃣ Fuse the five linear projections (left, right, three gates) into a
   single half‑precision GEMM.

3️⃣ Triton kernel ``split_fuse_gate_kernel`` parses the fused GEMM,
   applies the optional pair‑wise mask, evaluates the three σ‑gates,
   and writes the three tensors **left**, **right** and **out_gate**
   directly in layout ``[B, N, N, H]`` (H = hidden_dim).  This layout makes
   the hidden dimension (the 4‑th axis) contiguous – perfect for the
   second normalisation.

4️⃣ Convert ``left``/``right`` to ``[B, H, N, N]`` (by a cheap
   ``permute``+``contiguous``) and perform the batch matrix
   multiplication
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   with a single ``torch.bmm`` (Tensor‑core FP16).

5️⃣ Permute ``out`` back to ``[B, N, N, H]`` (contiguous H) and run a
   second Layer‑Norm + element‑wise multiplication with ``out_gate``
   (again using cuDNN ``layer_norm``).

6️⃣ Final linear projection back to the original channel dimension.

Only one Triton kernel is required (the split/fuse/gate kernel); everything
else leverages highly‑optimised PyTorch / cuDNN primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split‑fuse‑gate kernel
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[B, N, N, H] (fp16)
    right_ptr,         # mutable *[B, N, N, H] (fp16)
    out_gate_ptr,      # mutable *[B, N, N, H] (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,    # R = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk size
):
    """
    For each (b,i,k) row:
        left  = (proj_left  * mask) * σ(proj_left_gate)
        right = (proj_right * mask) * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)

    The three results are stored directly in layout [B, N, N, H] so that
    the hidden dimension is the innermost (contiguous) axis.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode (b,i,k) from the linear row index ----
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # ---- optional pair‑wise mask (scalar per row) ----
    if HAS_MASK:
        # mask is a scalar 0/1 per (b,i,k); broadcast over H
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- loop over hidden dimension in chunks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        # each slice is sized H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R, H_chunk]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw   = tl.load(right_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_g_raw  = tl.load(left_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ---- sigmoid (computed in fp32) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # ---- destination offset for layout [B, N, N, H] ----
        # flat = (((b * N) + i) * N + k) * H + h
        b_exp = b[:, None]                     # [R,1]
        i_exp = i[:, None]
        k_exp = k[:, None]
        h_exp = offs_h[None, :]                # [1, Hc]

        dst_off = ((b_exp * N + i_exp) * N + k_exp) * H + h_exp   # [R, Hc]

        # ---- store ----
        tl.store(left_ptr      + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim      = cfg["dim"]
    hidden   = cfg["hidden_dim"]          # H
    eps      = 1e-5
    nomask   = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16) – cuDNN implementation
    # ------------------------------------------------------------------
    # Normalise across the channel dimension 'dim'.
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                       # [B, N, N, dim] float32
    x_norm = x_norm_fp32.to(torch.float16)  # -> fp16 for the following GEMM

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (left + right + 3 gates)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]  (fp16)

    # Linear projects the last dimension; result shape [B, N, N, 5*H]
    proj = F.linear(x_norm, w_fused)        # fp16

    # ------------------------------------------------------------------
    # 3️⃣ split, mask, σ‑gates → left / right / out_gate in layout [B, N, N, H]
    # ------------------------------------------------------------------
    left      = torch.empty(B, N, N, H, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        # flatten mask to the row dimension (B·N·N) and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1

    BLOCK_R = 128                         # rows per program
    BLOCK_H = 32                          # hidden‑dim chunk (fits into registers)

    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free memory we no longer need
    del proj, x_norm, x_norm_fp32, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Bring H to the second axis for torch.bmm: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    del left, right

    left_mat  = left_perm.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)              # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # back to layout [B, N, N, H] (contiguous H → fast for next LN)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, out_mat, left_perm, right_perm

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + multiplication by out_gate (cuDNN)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                # (R, H) – H is innermost
    out_gate_flat = out_gate.view(R, H)

    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                         # float32

    # Element‑wise modulation with out_gate
    out_fused = (out_norm * out_gate_flat.to(torch.float32)).to(torch.float16)

    del out, out_gate, out_flat, out_gate_flat, out_norm

    # ------------------------------------------------------------------
    # 6️⃣ final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
