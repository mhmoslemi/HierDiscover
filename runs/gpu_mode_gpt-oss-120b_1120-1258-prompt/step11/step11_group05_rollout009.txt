"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × C) – use
    `torch.nn.functional.layer_norm` (fp32 → fp16).

2️⃣  Fuse the five linear maps (left/right projections + three gates)
    into a single GEMM:
        X_norm (R×C) @ W_fusedᵀ  → (R, 5·H)   (R = B·N·N).
    The fused weight `[5·H, C]` is stored in half‑precision.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
        * splits the fused result into
              left_proj, right_proj, left_gate, right_gate, out_gate
        * applies the (optional) pair‑wise mask
        * evaluates three sigmoid gates (sigmoid in fp32 for stability)
        * writes the three tensors `left`, `right`, `out_gate` (fp16).

4️⃣  Batched matrix multiplication over the hidden dimension:
        left  : (B, H, N, N)      – rows = i, cols = k
        right : (B, H, N, N)      – rows = j, cols = k
        out   = left @ rightᵀ      → (B, H, N, N)  (fp16)

5️⃣  Second LayerNorm on the hidden dimension (fp32) followed by
    element‑wise multiplication with `out_gate` (still fp16).

6️⃣  Final linear projection `out → dim` (fp16 → fp32).

Only step 3 is a custom Triton kernel; all heavy matrix‑multiplications
use cuBLAS/Tensor‑Cores via `torch.nn.functional.linear` and `torch.matmul`,
which gives near‑peak throughput on an H100.

The implementation works for all shapes listed in the specification,
supports optional masking, and returns a `float32` tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 3 – Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (float16)
    left_ptr,          # mutable *[rows, H]  (float16)
    right_ptr,         # mutable *[rows, H]  (float16)
    out_gate_ptr,      # mutable *[rows, H]  (float16)
    mask_ptr,          # const   *[rows]     (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are fp16; sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- mask value per row (broadcast over H) -----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- iterate over hidden dimension -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in fp32 for numerical stability, then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        tl.store(left_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns

    torch.Tensor

    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim       = cfg["dim"]

    device    = x.device

    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],     # γ (float32)
        bias=w["norm.bias"],         # β (float32)
        eps=eps,
    )                                 # float32

            w["left_gate.weight"],

            w["out_gate.weight"],
        ],

    out_gate = torch.empty(R, hidden, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)   # (R,)

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

        right,
        out_gate,

    left  = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.reshape(B * hidden, N, N)      # [B·H, N, N]
    right_mat = right.reshape(B * hidden, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))   # fp16

    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiply with out_gate

        out,
