@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, weight_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    rows, dim, H,
    eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_h = tl.program_id(1)
    row_start = pid_r * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    hidden_start = pid_h * BLOCK_H
    offs_h = hidden_start + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # mask for rows and hidden dims
    # load mask if has
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        m_val = m_val[:, None] # broadcast
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute mean/var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulators for 5 groups
    acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_out_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # second pass over dim
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                         mask=mask_r[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        x_norm = (x_vals - mean[:, None]) * rstd[:, None]  # fp32
        x_norm_fp16 = x_norm.to(tl.float16)  # shape (BLOCK_R, BLOCK_D)

        # for each group g
        # group offsets: base = g * H + hidden_start
        for g in range(5):
            weight_base = g * H + hidden_start  # row index start for this group
            w_ptr = weight_ptr + offs_d[:, None] * dim + (weight_base + offs_h)[None, :]  # shape (BLOCK_D, BLOCK_H)
            w = tl.load(w_ptr, mask=mask_d[:, None] & mask_h[None, :], other=0.0)  # fp16

            partial = tl.dot(x_norm_fp16, w)  # (BLOCK_R, BLOCK_H) fp32

            if g == 0:
                acc_left += partial
            elif g == 1:
                acc_right += partial
            elif g == 2:
                acc_left_gate += partial
            elif g == 3:
                acc_right_gate += partial
            else:
                acc_out_gate += partial

    # compute gates sigmoid
    left_gate = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

    # apply mask (broadcast)
    left = acc_left.to(tl.float16) * m_val * left_gate
    right = acc_right.to(tl.float16) * m_val * right_gate

    # store results
    left_ptrs = left_ptr + offs_r[:, None] * H + offs_h[None, :]
    right_ptrs = right_ptr + offs_r[:, None] * H + offs_h[None, :]

"""
Highly‑optimized Triton implementation of the *outgoing* Triangular‑Multiplicative
update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16).  The mean/variance are
   accumulated in FP32, the normalized result is stored as FP16.
2️⃣ Linear projection of the normalized tensor into five heads
   (left, right, left‑gate, right‑gate, out‑gate).  The three gate tensors
   are turned into sigmoid values (computed in FP32 for stability) and the
   mask (if any) is applied.  The projection and gating are fused in a
   single Triton kernel to avoid an extra write‑read of the normalized
   tensor.
3️⃣ Batched GEMM (FP16) computes the triangular multiplicative update:
   out[b·h+i] = left_mat[b·h+i] @ right_mat[b·h+i]ᵀ   (using cuBLAS `torch.bmm`).
4️⃣ Second row‑wise LayerNorm on the result, immediately multiplied by the
   out‑gate (again fused in Triton).
5️⃣ Final linear projection back to *dim* (FP16 GEMM) and cast to FP32.

Only the forward pass is required.  The heavy GEMM work is delegated to cuBLAS,
while the small but bandwidth‑intensive reductions (LayerNorm, masking,
sigmoids) are written in Triton for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)  γ
    bias_ptr,       # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1️⃣ mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2️⃣ write normalized values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β (FP32)

        out = norm * w[None, :] + b[None, :]                    # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split projections, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All arithmetic in FP16, sigmoid evaluated in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- mask (broadcasted) ----------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---------- loop over hidden dimension ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [left, right, left_gate, right_gate, out_gate] each H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # offsets into proj

        # ----- load the five slices -----
        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid (computed in FP32) -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask & gates -----
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # ----- store -----
        left_dst_ptr     = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)  (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)  (result)
    weight_ptr,       # const   *[dim]        (float32)  γ
    bias_ptr,         # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm on `x` (FP16) followed by element‑wise multiplication
    with the gate (`out_gate`).  All internal reductions are FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply γ/β, multiply with gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load X
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]           # FP32

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (FP32)
        norm = norm * w[None, :] + b[None, :]                     # FP32

        # gate
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                          # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper: pick reasonable block sizes
# ----------------------------------------------------------------------
def _choose_ln_blocks(rows: int, dim: int):
    # heuristic tuned for H100; keep blocks modest to stay inside registers
    BLOCK_M = 256 if rows >= 256 else max(1, rows)
    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D


def _choose_split_blocks(rows: int, hidden: int):
    BLOCK_R = 128                # rows per program
    BLOCK_H = 32                 # hidden dim per program
    return BLOCK_R, BLOCK_H

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32 tensors, bias=False
                         for linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_ln_blocks(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,                # H100 prefers more warps
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs) + mask + gates
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5·H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, w_fused)        # FP16, shape (R, 5·H)
    del w_fused, x_norm

    # allocate output buffers
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_half = mask.to(torch.float16).contiguous().view(R)   # [R]
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R, BLOCK_H = _choose_split_blocks(R, H)
    grid_split = (triton.cdiv(R, BLOCK_R),)   # 1‑D grid, hidden dim tiled inside kernel
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # shape -> [B, N, N, H] and bring H into batch axis
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # out_mat[i] = left_mat[i] @ right_mat[i]^T   (FP16, uses cuBLAS tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + multiplication by out_gate (fused kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_ln_blocks(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 5️⃣ Final projection back to `dim` (FP16 GEMM) → FP32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] FP16
    )   # (R, dim) FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
