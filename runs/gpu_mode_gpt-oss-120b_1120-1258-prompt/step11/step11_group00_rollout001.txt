"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass
=================================================================
The implementation follows the exact forward described in the reference
PyTorch module but removes all costly transposes and fuses several
operations:

1️⃣  Row‑wise LayerNorm on the 4‑D input (FP32 → FP16) – done with
    torch.nn.functional.layer_norm (highly‑optimised cuDNN).

2️⃣  A **single fused linear** that produces the five tensors
    (left, right, three gates) in FP16.

3️⃣  Triton kernel `split_fuse_gate_kernel` writes `left` and `right`
    directly into the layout ``[B, H, N, N]`` while applying the optional
    mask and the sigmoid gates.  It also stores the output gate.

4️⃣  Batched GEMM (`torch.bmm`) performs the true tri‑multiplicative
    update: ``out = left @ rightᵀ``  (shape ``[B·H, N, N]``).

5️⃣  A second LayerNorm (torch) + element‑wise multiplication with
    ``out_gate`` (still in FP16).

6️⃣  Final linear projection back to ``dim`` (FP16 GEMM) and cast to
    FP32.

Only step 3 is written in Triton – satisfying the “must‑use‑kernel”
requirement – while the heavy reductions (LayerNorm) and matrix
multiplications use the best‑available cuDNN/cuBLAS kernels on H100.
The chosen block sizes (256 rows × 128 hidden) map well to the H100
SM architecture and keep launch overhead low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates, write `left`/`right`
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]      (fp16)
    left_ptr,          # mutable *[B, H, N, N]   (fp16)
    right_ptr,         # mutable *[B, H, N, N]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]      (fp16)
    mask_ptr,          # const   *[rows]         (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,    # rows = B * N * N
    H: tl.int32,       # hidden dimension
    B: tl.int32,       # batch size
    N: tl.int32,       # sequence length
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows processed per program
    BLOCK_H: tl.constexpr,   # hidden dimension tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                     # N²
    stride_h = NN                  # stride between hidden slices
    stride_b = H * NN              # stride between batches

    # Decode (b, i, k) from flat row index
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # Load mask (if present) and broadcast over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection: each row stores (5*H) values
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R,H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # Sigmoid (computed in fp32 for numerical stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offset for [B, H, N, N] layout
        #   offset = b * stride_b + h * stride_h + i * N + k
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        # Store left/right
        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # Store out_gate (still flat [rows, H])
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    rows = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using cuDNN
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    # It returns FP32; we cast to FP16 for the following ops.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)               # [B, N, N, dim] fp16
    x_norm_flat = x_norm.view(rows, dim)                 # [rows, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (produces [rows, 5*H] in FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5*H, dim]
    proj = F.linear(x_norm_flat, fused_w)                # [rows, 5*H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & gates, store left/right/out_gate
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask is [B, N, N] -> flatten to [rows] and cast to half
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256          # rows per program
    BLOCK_H = 128          # hidden‑dim tile (covers full hidden_dim)
    grid = (triton.cdiv(rows, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B*H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out_mat = left @ rightᵀ   -> shape [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Bring back to [B, N, N, H] for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (cuDNN) + out‑gate multiplication
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still FP16 because input is FP16

    # Multiply element‑wise by out_gate (broadcasted)
    out_gate_view = out_gate.view(B, N, N, H)                # [B,N,N,H]
    out_norm = out_norm * out_gate_view

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM) + cast to FP32
    # ------------------------------------------------------------------
    # Flatten to [rows, H] for the final linear layer
    out_norm_flat = out_norm.view(rows, H)

    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # -> [rows, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
