"""
TriMul (outgoing) – fast forward‑only implementation
===================================================

The operator works on a 4‑D tensor *x* of shape [B, N, N, D] (float32) and
produces a tensor of the same shape.  The forward pass consists of:

1️⃣ Row‑wise LayerNorm on the last dimension (fp32 → fp16).  Implemented in
   `layernorm_kernel`.

2️⃣ A single fused linear projection that computes the three linear maps
   (left/right projection and three gates) at once:
       proj = x_norm @ W_fusedᵀ        # shape [R, 5·H]   (R = B·N·N)

3️⃣ A Triton kernel (`split_fuse_gate_kernel`) that
   * extracts the five sub‑vectors,
   * optionally masks the pairwise values,
   * applies the sigmoid non‑linearity to the three gates,
   * multiplies the projections by the masks and gates,
   * writes the three tensors *left*, *right* and *out_gate* (fp16).

4️⃣ Batched matrix multiplication over the hidden dimension H (cuBLAS/Tensor‑Core):
       out[b, h] = left[b, h] @ right[b, h]ᵀ
   This yields a tensor of shape [B, N, N, H] in fp16.

5️⃣ Second LayerNorm + element‑wise multiplication with *out_gate*,
   fused in `layernorm_mul_kernel`.

6️⃣ Final linear projection back to the original dimension D
   (fp16 GEMM) and a cast to float32.

All heavy arithmetic is performed in fp16 with fp32 reduction for
numerical stability.  The implementation works with or without an
optional mask and supports the dimension ranges used in AlphaFold3
(and the test suite of this challenge).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]     (fp32)
    y_ptr,          # mutable *[rows, dim]     (fp16)
    weight_ptr,     # const   *[dim]          (fp32)  γ
    bias_ptr,       # const   *[dim]          (fp32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- mean & variance (accumulate in fp32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r ∈ [0,R):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # mask per row (1.0 if absent)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)      # [BLOCK_R]
        mask_val = mask_val[:, None]                         # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout within the fused vector: [L, R, Lg, Rg, Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply optional mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3 – second LayerNorm + multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)  – out_gate
    y_ptr,            # mutable *[rows, dim]   (fp16)  – result
    weight_ptr,       # const   *[dim]        (fp32)  γ
    bias_ptr,         # const   *[dim]        (fp32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- compute mean & variance in fp32 -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalize, apply γ/β and multiply with gate -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)
        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data = (x, mask, weights, cfg)
            x        – torch.Tensor [B, N, N, D]      (float32)
            mask     – torch.Tensor [B, N, N] or None (bool/float)
            weights  – dict with all linear / layer‑norm parameters
                       (all stored as float32 tensors, bias=False for Linear)
            cfg      – dict containing:
                         "dim"         – input channel dimension D
                         "hidden_dim" – hidden dimension H
                         "nomask"      – bool, True → ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # rows for row‑wise kernels
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, D]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # channels per program (tuned for D∈{128,384})
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right & gates
    # -----------------------------------------------------------------
    # concatenate the five weight matrices (shape [5*H, D]) and cast to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, D]
            weights["right_proj.weight"],  # [H, D]
            weights["left_gate.weight"],   # [H, D]
            weights["right_gate.weight"],  # [H, D]
            weights["out_gate.weight"],    # [H, D]
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, D]

    proj = F.linear(x_norm, w_concat)                     # (R, 5*H) fp16
    del w_concat, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid – Triton
    # -----------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        # flatten mask → half precision; values are 0/1 (or any float)
        mask_half = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS/Tensor‑Core)
    # -----------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (fp16)
    out_mat = torch.matmul(left, right.transpose(-1, -2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------

    )   # fp16, shape [R, dim]
