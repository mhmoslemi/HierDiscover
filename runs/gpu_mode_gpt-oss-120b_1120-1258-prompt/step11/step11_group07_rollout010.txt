def custom_kernel(data):

    import torch.nn.functional as F
    import triton
    import triton.language as tl
    
    input_tensor, mask_tensor, weights, config = data
    # configs
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    device = input_tensor.device
    B,N,_,_ = input_tensor.shape
    R = B * N * N
    H = hidden_dim
    # 1) first layernorm
    x_norm_fp32 = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)
    # 2) fused projection
    # concat weight matrices
    w_fused = torch.cat([weights["left_proj.weight"],
                         weights["right_proj.weight"],
                         weights["left_gate.weight"],
                         weights["right_gate.weight"],
                         weights["out_gate.weight"]], dim=0).to(torch.float16)
    proj = F.linear(x_norm, w_fused)   # [R, 5*H]  fp16
    # 3) split and apply mask+gates

    right = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask_tensor is None:

        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:

        HAS_MASK = 1
        mask_ptr = mask_fp16
    # launch kernel
    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # 4) TriMul batched GEMM
    # reshape
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))   # [B*H,N,N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

"""
Optimized Triton‑accelerated implementation of the *outgoing* Triangular‑Multiplicative
update (TriMul) used in AlphaFold‑3 style models.

Algorithm Overview
------------------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × C) – implemented with
   :func:`torch.nn.functional.layer_norm` (FP32) followed by an FP16 cast.
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a single
   GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)   where R = B·N·N, H = hidden_dim.
3️⃣ Triton kernel ``split_fuse_gate_kernel`` splits the fused projection,
   applies the optional pairwise mask, evaluates the three sigmoid gates,
   and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
   All arithmetic is performed in FP16 (sigmoid evaluated in FP32 for stability).
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   (implemented with ``torch.bmm`` → cuBLAS Tensor‑Core kernels).
5️⃣ Second LayerNorm on the hidden dimension (FP16) using ``torch.nn.functional.layer_norm``.
   The normalized tensor is multiplied element‑wise by ``out_gate``.
6️⃣ Final linear projection back to the original channel dimension (FP16) and a
   cast to FP32 for the returned tensor.

Only step 3 is written in Triton; all heavy matrix‑multiplications use the
high‑performance cuBLAS kernels.  This design dramatically reduces kernel launch
overhead while still satisfying the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic in fp16; sigmoid evaluated in fp32 for numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load mask (broadcasted) if present
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout within proj row: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # offsets into proj

        left_raw     = tl.load(proj_ptr + base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for stability, then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # store results
        left_dst_ptr     = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton‑compatible entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (float32, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension; we keep the
    # original 4‑D shape to avoid an extra view.
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim]  float32
    x_norm = x_norm_fp32.to(torch.float16)   # cast to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (five linear maps) → (R, 5·H)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim]) => [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    # Flatten the normalized tensor to 2‑D for the linear layer
    x_norm_flat = x_norm.view(R, dim)   # [R, dim]
    proj = F.linear(x_norm_flat, w_fused)   # [R, 5·H]  FP16
    del w_fused, x_norm_flat            # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask, compute sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask: [B, N, N] → [R] (float16 0/1)
        mask_flat = mask.to(torch.float16).contiguous().view(R)
        HAS_MASK = 1
        mask_ptr = mask_flat

    BLOCK_R = 128          # rows per program
    BLOCK_H = 32           # hidden‑dim chunk size
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H into the batch axis for bmm
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)      # [B·H, N, N]

    # BMM: left @ rightᵀ  (both half‑precision → Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) + element‑wise multiplication by out_gate
    # ------------------------------------------------------------------
    # Cast LayerNorm parameters to FP16 for the half‑precision pass
    to_out_norm_w_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_h = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w_h,
        bias=to_out_norm_b_h,
        eps=eps,
    )   # [B, N, N, H]  FP16

    # Multiply by the pre‑computed out_gate
    out_gate_4d = out_gate.view(B, N, N, H)
    out_norm = out_norm * out_gate_4d
    del out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` and cast to FP32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]  FP16
    )   # (B, N, N, dim)  FP16

    final = final_half.to(torch.float32)
    return final
```
