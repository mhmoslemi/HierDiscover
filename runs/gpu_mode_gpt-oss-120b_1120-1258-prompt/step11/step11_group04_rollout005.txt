"""
TriMul (outgoing) – fast Triton + cuBLAS implementation
=======================================================

Algorithm (forward only)
------------------------
1️⃣  Row‑wise LayerNorm on the input (B·N·N × dim), converting FP32 -> FP16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single FP16 GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)   (R = B·N·N).
3️⃣  Split the fused result, optionally mask it, evaluate the sigmoid gates
    and write the three tensors **directly** in the (B, H, N, N) layout that
    is required by the batched GEMM.  (All arithmetic FP16, sigmoid in FP32.)
4️⃣  Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
    performed with ``torch.bmm`` (cuBLAS Tensor‑Core kernels).
5️⃣  Row‑wise LayerNorm on the GEMM result + element‑wise multiplication with
    ``out_gate`` (fused in a Triton kernel).
6️⃣  Final linear projection back to the original ``dim`` (FP16 → FP32).

Only three Triton kernels are used – the heavy matrix multiplications are
handled by cuBLAS, which gives near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   *[rows, dim]      (float32)
    y_ptr,        # mutable *[rows, dim]      (float16)
    w_ptr,        # const   *[dim]           (float32)   γ
    b_ptr,        # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows processed per program
    BLOCK_D: tl.constexpr,    # dim processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#     write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flat row r = b·N·N + i·N + k (0 ≤ r < R):
        left[b,h,i,k]  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right[b,h,i,k] = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate[b,h,i,k] = sigmoid(proj_out_gate)

    The results are already stored in the (B,H,N,N) layout required by the
    batched GEMM, so no extra transpose is needed.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of the hidden dimension in the target layout
    hidden_stride = N * N                       # distance between two h slices

    # base offset common to all hidden slices of the same (b,i,k)
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # offsets into the fused projection matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets (linear indexing of (B,H,N,N))
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr  + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)
    y_ptr,            # mutable *[rows, dim]   (fp16)
    w_ptr,            # const   *[dim]        (float32)  γ
    b_ptr,            # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (fp16) followed by multiplication with ``gate``.
    The final result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate, write -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # β (fp32)
        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate value (fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32

        BLOCK_D=BLOCK_D,

    )

        mask_ptr = mask_flat

        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                  # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                 # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N)                     # (B, H, N, N)

    del left, right, left_mat, right_mat, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate (fused kernel)
    # --------------------------------------------------------------
    # Convert to (B·N·N, H) layout for the kernel
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    out_flat = out_perm.view(R, H).contiguous()
    gate_flat = gate_perm.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        gate_flat,
        out_fused,
        w["to_out_norm.weight"],   # γ, float32
        w["to_out_norm.bias"],     # β, float32
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, gate_flat, out, out_gate, out_perm, gate_perm

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # FP16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

**Explanation of the implementation**

* The first Triton kernel normalises each row of the flattened `[B·N·N, dim]` tensor,
  applying the LayerNorm parameters (`norm.weight`, `norm.bias`) and stores the
  result as FP16.

* The five linear layers (two projections + three gates) are fused into a single
  matrix‑multiply (`torch.nn.functional.linear`).  The result `(R, 5·H)` is then
  split by a second Triton kernel.  The kernel also:
  - Applies the optional pair‑wise mask,
  - Computes the three sigmoid gates in FP32 for numerical stability,
  - Writes `left`, `right` and `out_gate` directly in the `(B, H, N, N)` layout
    required by the batched GEMM, thereby avoiding an expensive transpose.

* The heavy multiplicative update `out[i,j] = Σ_k left[i,k] * right[j,k]`
  is performed with `torch.bmm`, which uses cuBLAS Tensor‑Core kernels on the
  H100 and is much faster than a hand‑written Triton GEMM for this size.

* A third Triton kernel fuses the second LayerNorm (`to_out_norm`) with the
  element‑wise multiplication by `out_gate`.

* Finally a half‑precision linear projection (`to_out`) brings the tensor
  back to the original `dim` and the result is cast to FP32 as required by the
  interface.

All heavy arithmetic stays in FP16 (except the sigmoid), while the small
row‑wise reductions needed for LayerNorm are computed in FP32 for accuracy.
The three Triton kernels together with cuBLAS GEMMs deliver a runtime well
below the 1 ms target on an H100 for the test configurations.
