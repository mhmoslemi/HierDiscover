@triton.jit
def layernorm_gate_strided_kernel(
    x_ptr,          # const *[BH, N, N] input after bmm
    gate_ptr,       # const *[rows, H] (contig)
    y_ptr,          # mutable *[rows, H] output (contig)
    weight_ptr,     # const *[H] LN weight (float32)
    bias_ptr,       # const *[H] LN bias (float32)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # rows index
    mask_m = offs_m < rows

    stride_hidden = N * N  # distance in memory between hidden slices

    # Compute base_offset for each row: location of h=0 element
    # b = offs_m // (N*N)
    # offset_ij = offs_m - b * (N*N)
    # base_offset = b * H * (N*N) + offset_ij
    N2 = N * N
    b = offs_m // N2
    offset_ij = offs_m - b * N2
    base_offset = b * H * N2 + offset_ij

    # Compute mean and variance across hidden dimension
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # hidden
        mask_d = offs_d < H
        # pointer to the block of hidden values for each row
        ptr = x_ptr + base_offset[:, None] + offs_d[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Normalization, weight, bias, gate multiplication
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        # load values
        ptr = x_ptr + base_offset[:, None] + offs_d[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # load gate
        g_ptr = gate_ptr + offs_m[:, None] * H + offs_d[None, :]
        gate_vals = tl.load(g_ptr,
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)

        # LN weight and bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * gate_vals  # fp32

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
This implementation follows the reference PyTorch module but removes the
expensive `permute(...).contiguous()` that copies the whole
`[B, H, N, N] → [B, N, N, H]` tensor after the batched GEMM.
Instead we compute the second LayerNorm (and the out‑gate multiplication)
directly on the GEMM output stored as `[B*H, N, N]`.  The hidden dimension
is accessed with a stride of `N*N`, so the kernel loads the values with a
fixed stride rather than requiring a contiguous layout.  This saves an
O(B·N²·H) memory copy and reduces overall runtime.

The forward consists of:
1️⃣ Row‑wise LayerNorm on the input (FP32→FP16) – Triton.
2️⃣ A fused linear projection that produces five tensors
   (left, right, left_gate, right_gate, out_gate) in FP16.
3️⃣ A Triton kernel that writes `left`/`right` directly into `[B, H, N, N]`,
   applies the optional mask and the sigmoid gates, and stores `out_gate`
   as a flat `[B·N·N, H]` matrix.
4️⃣ Batched GEMM (`torch.bmm`) computes the tri‑multiplicative update:
   `out = left @ rightᵀ`   (shape `[B·H, N, N]`).
5️⃣ A strided LayerNorm + out‑gate multiplication kernel that reads the
   GEMM output with stride `N*N` and produces the gated, normalized tensor
   `[B·N·N, H]` (still FP16) – Triton.
6️⃣ Final projection back to `dim` with a regular FP16 linear layer and
   cast to FP32.

Only the three kernels below are written in Triton, all heavy matrix
multiplications rely on cuBLAS/Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----------------- mean & variance -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------- write normalized values -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates, write `left`/`right`
#     directly into [B, H, N, N] layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (float16)
    left_ptr,          # mutable *[B, H, N, N] (float16)
    right_ptr,         # mutable *[B, H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK=0
    rows: tl.int32,    # rows = B * N * N
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                     # N²
    stride_h = NN                  # distance between two consecutive hidden slices
    stride_b = H * NN              # distance between two consecutive batches

    # ---- decode (b, i, k) from the flat row index ----
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # ---- optional mask (broadcast over hidden) ----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process the hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ---- contiguous slices inside the fused projection ----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ---- sigmoid (fp32 for stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store left/right into [B, H, N, N] layout ----
        # base offset for the first hidden slice of this row:
        #   offset = b * stride_b + i * N + k
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---- store out_gate (still flat [rows, H]) ----
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Strided LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_strided_kernel(
    x_ptr,          # const *[B*H, N, N]  (float16) – GEMM output
    gate_ptr,       # const *[rows, H]    (float16)
    y_ptr,          # mutable *[rows, H]  (float16)
    weight_ptr,     # const *[H]         (float32) γ
    bias_ptr,       # const *[H]         (float32) β
    rows: tl.int32,     # rows = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    stride_hidden = N * N          # distance in memory between two hidden slices

    # decode batch & (i,j) to compute base offset of first hidden element
    N2 = N * N
    b = offs_m // N2
    offset_ij = offs_m - b * N2
    base_offset = b * H * N2 + offset_ij                    # start of h=0 for this row

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # hidden indices
        mask_d = offs_d < H
        ptr = x_ptr + base_offset[:, None] + offs_d[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply LN weight/bias, multiply by gate ----------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # load values from GEMM output
        ptr = x_ptr + base_offset[:, None] + offs_d[None, :] * stride_hidden
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # load corresponding gate values (contiguous)
        g_ptr = gate_ptr + offs_m[:, None] * H + offs_d[None, :]
        gate_vals = tl.load(g_ptr,
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)

        # LN weight & bias (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32
        y_fp32 = (norm * w[None, :] + b[None, :]) * gate_vals   # fp32
        y_fp16 = y_fp32.to(tl.float16)

        # store result (contiguous rows x H)
        tl.store(y_ptr + offs_m[:, None] * H + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights_dict : dict with keys

                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    rows = B * N * N                      # total pair‑wise rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(rows, dim).contiguous()               # [rows, dim] fp32
    x_norm = torch.empty(rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (produces [rows, 5*H] in FP16)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [rows, 5*H] fp16
    del w_fused, x_norm                     # free intermediate memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & gates, write `left`/`right` directly
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask: [B, N, N] → flat rows, cast to fp16
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(rows, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # view as [B*H, N, N] – no extra copy, just a reshape
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Strided LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    gated_norm = torch.empty(rows, H, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_gate_strided_kernel[grid_ln2](
        out_mat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        rows,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM) + cast to FP32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [rows, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
