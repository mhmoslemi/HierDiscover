"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (FP32 → FP16) – Triton.
2️⃣  Fuse the five linear projections (left, right, left_gate,
    right_gate, out_gate) into a single GEMM (FP16) using torch.nn.functional.linear.
3️⃣  Split the fused projection, reshape to [B, H, N, N],
    apply the optional mask (broadcast over H) and the sigmoid gates.
4️⃣  Batched GEMM over the hidden dimension (cuBLAS):
        out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
5️⃣  Compute per‑pair mean / rsqrt‑variance of `out` (over H) in FP32 (torch).
6️⃣  Fuse second LayerNorm, out_gate multiplication and the final
    projection (H → dim) into a single Triton kernel.
    The kernel loads the row‑wise statistics, normalises,
    applies learned γ/β, multiplies by out_gate and projects to dim.
    Output is FP32 tensor [B, N, N, dim].

All heavy element‑wise work (both LayerNorms, gating, final projection) is
performed by Triton kernels; the only non‑Triton part is the large batched
matrix multiplication (step 4) which is efficiently handled by cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim]  input FP32
    y_ptr,      # [R, dim]  output FP16
    w_ptr,      # [dim]    γ (FP32)
    b_ptr,      # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalised values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6️⃣ Final LayerNorm + out_gate * final projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out.weight)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # hidden stride (distance between successive H‑planes)
    hidden_stride = N * N
    # offset of the first hidden entry of the current pair
    base_offset = b * (H * N * N) + i * N + j   # scalar offset per row

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden tensor and gate (fp16 → fp32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ,β) – fp32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                        # [M, K]

        # Cast back to fp16 for the final matmul
        term_fp16 = term.to(tl.float16)

        # Load slice of final projection matrix Wᵀ : (K, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        # Dot product (fp16·fp16 → fp32) and accumulate
        acc += tl.dot(term_fp16, w_tile)   # fp32 accumulation

    # Write final result [B, N, N, dim]  (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping from weight name to torch.Tensor (float32)
                required keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N              # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5*H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5*H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, apply sigmoid gates
    # --------------------------------------------------------------
    # slice the five blocks
    left_raw      = proj[:, :H]
    right_raw     = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw= proj[:, 3*H:4*H]
    out_gate_raw  = proj[:, 4*H:]

    # reshaping to [B, H, N, N]
    def to_BHNN(t):
        return t.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_raw   = to_BHNN(left_raw)
    right_raw  = to_BHNN(right_raw)
    left_gate  = torch.sigmoid(to_BHNN(left_gate_raw))
    right_gate = torch.sigmoid(to_BHNN(right_gate_raw))
    out_gate   = torch.sigmoid(to_BHNN(out_gate_raw))

    del proj, left_gate_raw, right_gate_raw, out_gate_raw

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(1)  # [B,1,N,N]
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # apply gates
    left  = left_raw * left_gate
    right = right_raw * right_gate
    del left_raw, right_raw, left_gate, right_gate

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B*H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # [B*H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                       # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Per‑pair mean & rsqrt‑variance of `out` (over H)
    # --------------------------------------------------------------
    out_fp32 = out.float()
    mu    = out_fp32.mean(dim=1)                     # [B, N, N] FP32
    var   = out_fp32.var(dim=1, unbiased=False)      # [B, N, N] FP32
    rstd  = torch.rsqrt(var + eps)                   # [B, N, N] FP32
    mu_flat   = mu.reshape(R).contiguous()
    rstd_flat = rstd.reshape(R).contiguous()
    del out_fp32, mu, var, rstd

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused Triton)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (≈pairs)
    BLOCK_K_OUT = 32    # hidden‑dim tiling
    BLOCK_N_OUT = 128   # output‑dim tiling

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # Cleanup
    del out, out_gate, mu_flat, rstd_flat, w_out_T

    return final
