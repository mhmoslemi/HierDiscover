"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Apply a row‑wise LayerNorm on the last dimension (dim) of the input tensor.
    The reduction is done by torch.nn.functional.layer_norm (CUDA kernel).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single FP16 GEMM (torch.nn.functional.linear).  The result
    has shape (R, 5·H) where R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional pair‑mask, and compute the
    three sigmoid gates (left_gate, right_gate, out_gate) in a single Triton
    kernel.  The kernel writes three tensors `left`, `right` and `out_gate`
    in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    realised as `torch.bmm(left.view(B·H,N,N),
                           right.view(B·H,N,N).transpose(1,2))`.

5️⃣  Apply a second LayerNorm over the hidden dimension H (torch.nn.functional.
    layer_norm), multiply by `out_gate` and finally project back to `dim`
    with another FP16 GEMM (torch.nn.functional.linear).  The final result is
    returned as float‑32 (`[B, N, N, dim]`).

Only the split‑&‑gate kernel is written in Triton – the rest of the work
leverages highly‑tuned cuBLAS / CUDA kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ----------- decode flat index r = b·N·N + i·N + k ---------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ---------- optional mask (scalar per pair) --------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    # base offset of the first hidden plane (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k

    # --------------------------------------------------------------
    # iterate over hidden tiles
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # --------- sigmoid in FP32 (more accurate) -------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # -------------- apply mask & gates --------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate  # no mask on out_gate

        # store into the destination tensors (layout [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (CUDA kernel via torch.nn.functional)
    # --------------------------------------------------------------
    # Normalise over the last dimension `dim`.  The torch implementation
    # computes mean/var in FP32 and writes back in FP16 after scaling.
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)
    x_norm = x_norm.to(torch.float16)    # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [5·H, dim]

    # flatten batch & pair dimensions for the GEMM
    x_flat = x_norm.view(R, dim)                      # [R, dim]  FP16
    proj = F.linear(x_flat, fused_w)                  # [R, 5·H]  FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – never accessed when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    # tile sizes – empirically good for H≤384 and N up to 1024
    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free memory we no longer need
    del proj, mask_ptr, x_flat

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # view as [B·H, N, N] for each hidden plane
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    # PyTorch's layer_norm over the hidden dimension H (shape ... H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    out_norm = F.layer_norm(out_perm,
                            normalized_shape=(H,),
                            weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"],
                            eps=eps)                       # FP16
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] FP16

    # apply the output gate (element‑wise)
    out_norm = out_norm * out_gate

    # final linear projection back to `dim`
    # flatten to (R, H) and use a single GEMM
    out_flat = out_norm.permute(0, 2, 3, 1).reshape(R, H)   # [R, H]
    final = F.linear(out_flat, w["to_out.weight"])          # [R, dim] (FP16)
    final = final.to(torch.float32).view(B, N, N, dim)      # [B, N, N, dim] (FP32)

    return final
