"""
custom_kernel – fast outgoing TriMul implementation (forward only)

The operator works on a tensor x of shape   [B, N, N, C]  (C = dim)
and produces an output of the same shape.  The forward pass consists of

1️⃣ Row‑wise LayerNorm over the last dimension (fp32 → fp16)                – triton
2️⃣ Fuse the five linear maps (left/right projections + three gates)
   into a single GEMM (fp16)                                               – torch.nn.functional.linear
3️⃣ Split the fused projection, apply the optional pairwise mask and the
   three sigmoid gates, store the results in layout [B, H, N, N]            – triton
4️⃣ Batched half‑precision GEMM over the hidden dimension (cuBLAS)          – torch.bmm
   out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
5️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor (fp32)          – triton
6️⃣ Fuse the second LayerNorm, the out‑gate and the final linear
   projection (H → dim) into one Triton kernel, writing the final float32   – triton
   output tensor.

The implementation keeps the heavy N³ contraction as a regular torch
batched matmul (cuBLAS) and uses Triton for all element‑wise and reduction
steps, giving a fast end‑to‑end kernel suitable for the test configurations.

# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # input   [rows, dim]   (fp32)
    y_ptr,               # output  [rows, dim]   (fp16)
    w_ptr,               # gamma   [dim]        (fp32)
    b_ptr,               # beta    [dim]        (fp32)
    rows: tl.int32,      # total rows = B * N * N
    dim: tl.int32,       # channel dim = dim
    eps: tl.float32,     # epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # channels per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- accumulate sum & sum of squares (fp32) -----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized values (γ·(x‑μ)·rstd + β) -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β (fp32)
        out = norm * w[None, :] + b[None, :]                   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H] fused projection (fp16)
    left_ptr,          # [B, H, N, N] (fp16)
    right_ptr,         # [B, H, N, N] (fp16)
    out_gate_ptr,      # [B, H, N, N] (fp16)
    mask_ptr,          # [R] (fp16) – only read if HAS_MASK==1
    R: tl.int32,       # B * N * N
    H: tl.int32,       # hidden_dim
    N: tl.int32,       # seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from linear index r = b*N*N + i*N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask over hidden dimension if present
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for hidden dimension inside left/right/out_gate tensors
    hidden_stride = N * N
    # base offset for the first hidden slice (h=0)
    base_offset = b * (H * N * N) + i * N + k                # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [R, 5*H] (row‑major)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # read five groups (left, right, left_gate, right_gate, out_gate)
        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (fp32 for stability, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

def rowwise_stats_kernel(

    total_rows: tl.int32,   # R = B * N * N

    i = rem // N

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict of model parameters (float32, bias=False for Linear)

        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (fp16)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    proj = F.linear(x_norm, fused_w)         # [R, 5*H] fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → (B, H, N, N)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if cfg.get("nomask", True):
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # mask is expected shape [B, N, N] (bool/float). Flatten to [R].
        mask_fp = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_fp

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS, fp16)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B*H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B*H, N, N]

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N]
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) of contracted tensor
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate, and final projection
    # --------------------------------------------------------------
    # to_out.weight : [dim, hidden_dim] → transpose to [hidden_dim, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64    # rows per program (positions)
    BLOCK_K_OUT = 32    # hidden‑dim tile size
    BLOCK_N_OUT = 64    # output‑dim tile size
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```
