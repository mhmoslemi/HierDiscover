@triton.jit
def final_norm_gate_kernel(
    out_ptr,          # const *[B, H, N, N]   (fp16)
    out_gate_ptr,     # const *[B, H, N, N]   (fp16)
    mu_ptr,           # const *[R]            (fp32)
    rstd_ptr,         # const *[R]            (fp32)
    ln_w_ptr,         # const *[H]            (fp32) gamma
    ln_b_ptr,         # const *[H]            (fp32) beta
    out_pre_ptr,      # mutable *[R, H]       (fp16)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # decode (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden stride in out / out_gate (N*N)
    base_offset = b * (H * nn) + i * N + j   # each hidden slice offset start

    hidden_stride = nn  # distance between successive hidden dimension values in out memory

    # load mu and rstd to broadcast
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    # iterate hidden blocks
    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # load out and gate
        ptr_out = out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = out_gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride

        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # load layernorm params
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K] fp32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)    # [K] fp32

        # normalise: (out - mu) * rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        gated = norm * gate_vals                                      # [M, K]

        # store to out_pre: contiguous layout row-major (R,H)
        # out_pre offset = offs_m[:, None] * H + offs_h[None, :]
        out_pre_offset = offs_m[:, None] * H + offs_h[None, :]
        tl.store(out_pre_ptr + out_pre_offset,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but moves all
memory‑bound work into three Triton kernels:

1️⃣ **LayerNorm (first)** – row‑wise mean/variance in FP32, output FP16.
2️⃣ **Split + mask + gates** – decode the fused (left/right + three gates)
   projection, apply the optional pair‑wise mask and the three sigmoid
   gates, and store the tensors in the (B,H,N,N) layout required by the
   batched GEMM.
3️⃣ **Row‑wise statistics** – compute μ, σ⁻¹ for the contracted tensor
   `out = left @ rightᵀ`.
4️⃣ **Norm + gate (final)** – fuse the second LayerNorm, the element‑wise
   multiplication with `out_gate`, and write the normalised gated vector
   to a contiguous `(R, H)` buffer (`R = B·N·N`).  
   The remaining linear projection (`(R, H) @ (H, dim)`) is performed by
   cuBLAS via `torch.nn.functional.linear`, which is far faster than a
   hand‑written Triton dot‑product.

All heavy matrix‑multiplications (the two GEMMs) use Tensor‑Core FP16
cuBLAS, while the Triton kernels keep the per‑element work (layer‑norms,
masking, gating) on‑GPU and avoid any Python‑level loops.

The final output is returned in `float32` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32)   γ
    b_ptr,    # const   *[dim]      (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalised values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
#    (layout B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)   fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index (b,i,k) → destination layout (B,H,N,N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask handling (scalar per pair (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                       # distance between successive hidden slices
    base_offset = b * (H * nn) + i * N + k   # offset of hidden‑plane 0 for this (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # fused layout: [ left | right | left_gate | right_gate | out_gate ]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 → fp16 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                               # out_gate NOT masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise μ/σ⁻¹ for ``out`` (used by the second LayerNorm)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # const *[B, H, N, N] (fp16)
    mu_ptr,     # mutable *[R]         (float32)
    rstd_ptr,   # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn                      # distance between successive hidden entries
    base_offset = b * (H * nn) + i * N + j   # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm + out_gate → contiguous (R, H) buffer
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,          # const *[B, H, N, N] (fp16)
    out_gate_ptr,     # const *[B, H, N, N] (fp16)
    mu_ptr,           # const *[R]            (fp32)
    rstd_ptr,         # const *[R]            (fp32)
    ln_w_ptr,         # const *[H]            (fp32)   γ of to_out_norm
    ln_b_ptr,         # const *[H]            (fp32)   β of to_out_norm
    out_pre_ptr,      # mutable *[R, H]       (fp16)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for hidden‑slice 0 in the (B,H,N,N) tensor
    hidden_stride = nn                         # distance between successive hidden slices
    base_offset = b * (H * nn) + i * N + j      # offset of h=0

    # broadcasted μ / σ⁻¹ for the current position
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # load out and gate
        ptr_out  = out_ptr      + base_offset[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = out_gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # layer‑norm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [K] fp32

        # (out - μ) * rstd  →  normalise
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]

        # LN: γ·diff + β
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]

        # multiply by out_gate
        gated = norm * gate_vals                                      # [M, K]

        # write to contiguous buffer (R, H)
        out_pre_offset = offs_m[:, None] * H + offs_h[None, :]
        tl.store(out_pre_ptr + out_pre_offset,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (layout B, H, N, N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                            # (B, H, N, N) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm + out_gate → contiguous (R, H) buffer
    # ------------------------------------------------------------------
    out_pre = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 256
    BLOCK_K_OUT = 32
    grid_norm_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        out_pre,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=4,
    )
    # clean up tensors no longer needed
    del out, out_gate, mu, rstd, left, right

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    # weight (dim, H) → cast to fp16 for Tensor‑Core
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # shape [dim, H]
    final_fp16 = F.linear(out_pre, w_out_fp16)          # (R, dim) fp16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # back to FP32

    return final

if __name__ == "__main__":
    torch.manual_seed(0)
    cfg = {"dim": 128, "hidden_dim": 128, "nomask": True}
    B, N, dim, H = 2, 256, cfg["dim"], cfg["hidden_dim"]
    x = torch.randn(B, N, N, dim, device="cuda", dtype=torch.float32)
    mask = None

    # Dummy weights matching the reference
    w = {
        "norm.weight": torch.ones(dim, device="cuda"),
        "norm.bias": torch.zeros(dim, device="cuda"),
        "left_proj.weight": torch.randn(H, dim, device="cuda"),
        "right_proj.weight": torch.randn(H, dim, device="cuda"),
        "left_gate.weight": torch.randn(H, dim, device="cuda"),
        "right_gate.weight": torch.randn(H, dim, device="cuda"),
        "out_gate.weight": torch.randn(H, dim, device="cuda"),
        "to_out_norm.weight": torch.ones(H, device="cuda"),
        "to_out_norm.bias": torch.zeros(H, device="cuda"),
        "to_out.weight": torch.randn(dim, H, device="cuda"),
    }

    # Triton implementation
    out_triton = custom_kernel((x, mask, w, cfg))

    # Reference PyTorch implementation
    class TriMulRef(torch.nn.Module):
        def __init__(self, dim, hidden_dim):
            super().__init__()
            self.norm = torch.nn.LayerNorm(dim, eps=1e-5)
            self.left_proj = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.right_proj = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.left_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.right_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.out_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.to_out_norm = torch.nn.LayerNorm(hidden_dim, eps=1e-5)
            self.to_out = torch.nn.Linear(hidden_dim, dim, bias=False)

        def forward(self, x, mask):
            x = self.norm(x)
            left = self.left_proj(x)
            right = self.right_proj(x)
            if mask is not None:
                mask = mask.unsqueeze(-1)
                left = left * mask
                right = right * mask
            left_gate = torch.sigmoid(self.left_gate(x))
            right_gate = torch.sigmoid(self.right_gate(x))
            out_gate = torch.sigmoid(self.out_gate(x))
            left = left * left_gate
            right = right * right_gate
            out = torch.einsum("...ikd,...jkd->...ijd", left, right)
            out = self.to_out_norm(out)
            out = out * out_gate
            out = self.to_out(out)
            return out

    model = TriMulRef(dim, H).to("cuda")
    with torch.no_grad():
        model.norm.weight.copy_(w["norm.weight"])
        model.norm.bias.copy_(w["norm.bias"])
        model.left_proj.weight.copy_(w["left_proj.weight"])
        model.right_proj.weight.copy_(w["right_proj.weight"])
        model.left_gate.weight.copy_(w["left_gate.weight"])
        model.right_gate.weight.copy_(w["right_gate.weight"])
        model.out_gate.weight.copy_(w["out_gate.weight"])
        model.to_out_norm.weight.copy_(w["to_out_norm.weight"])
        model.to_out_norm.bias.copy_(w["to_out_norm.bias"])
        model.to_out.weight.copy_(w["to_out.weight"])

    out_ref = model(x, mask)
    torch.testing.assert_allclose(out_triton, out_ref, rtol=1e-3, atol=1e-3)
    print("Sanity check passed!")
