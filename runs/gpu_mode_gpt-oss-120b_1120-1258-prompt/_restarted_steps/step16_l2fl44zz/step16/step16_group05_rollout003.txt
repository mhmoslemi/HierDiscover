"""
TriMul (outgoing) – fast forward pass with a fused row‑wise LayerNorm kernel.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×D}.  The computation is:

1️⃣ Row‑wise LayerNorm over the last dimension (FP32 → FP16) – done with a
   custom Triton kernel (no extra synchronisation needed).

2️⃣ A single GEMM projects the normalized tensor to five tensors
   (left, right, left‑gate, right‑gate, out‑gate) in one shot
   (weights are concatenated).  The result stays in FP16.

3️⃣ The tensors are reshaped to [B,N,N,H] (H = hidden_dim).  An optional
   pair‑wise mask is applied, followed by sigmoid gates.  The gated left/right
   tensors are multiplied element‑wise with their gates.

4️⃣ Batched half‑precision matrix multiplication implements the
   `einsum('... i k h, ... j k h -> ... i j h')` reduction:
   for each hidden channel h we perform B·H independent GEMMs
   (left @ rightᵀ) over the sequence dimension N.

5️⃣ A second LayerNorm (over the hidden dimension) is performed with
   `torch.nn.functional.layer_norm`.  The result is multiplied by the
   out‑gate and finally projected back to the original dimension D.

All heavy‑memory‑bound work stays in FP16, only the two normalisations and the
final projection use FP32 for numerical stability.  The entry point follows
the required signature `custom_kernel(data)`.
"""

# 1️⃣ Row‑wise LayerNorm kernel (FP32→FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # float32 input   [R, D]
    y_ptr,               # float16 output  [R, D]
    w_ptr,               # float32 weight  [D]
    b_ptr,               # float32 bias    [D]
    rows: tl.int32,      # R
    dim: tl.int32,       # D
    eps: tl.float32,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per tile
):
    """
    Compute LayerNorm row‑wise (mean & variance in FP32) and write the
    normalized values in FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean / variance in FP32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the D dimension in tiles of size BLOCK_D
    for d_tile in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the row (FP32)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_tile in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # normalise
        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        # load γ and β (FP32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]     # FP32

        # store as FP16
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor : [B, N, N, D] (float32)
    """
    # unpack arguments ----------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, D = x.shape
    hidden_dim = cfg["hidden_dim"]
    dim = cfg["dim"]
    assert D == dim, "Input last dimension must match config dim"

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over D (FP32→FP16)
    # ------------------------------------------------------------------
    R = B * N * N                      # number of (b,i,j) positions
    x_flat = x.reshape(R, D).contiguous()           # [R, D] float32
    x_norm = torch.empty(R, D, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        D,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Single fused projection to (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each shape [H, D])
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, D]

    proj = F.linear(x_norm, fused_weight)                 # [R, 5*H] fp16
    del fused_weight, x_norm

    # reshape and split
    proj = proj.view(R, 5, hidden_dim)                    # [R, 5, H]
    left       = proj[:, 0, :].reshape(B, N, N, hidden_dim)  # [B,N,N,H]
    right      = proj[:, 1, :].reshape(B, N, N, hidden_dim)
    left_gate  = proj[:, 2, :].reshape(B, N, N, hidden_dim)
    right_gate = proj[:, 3, :].reshape(B, N, N, hidden_dim)
    out_gate   = proj[:, 4, :].reshape(B, N, N, hidden_dim)
    del proj

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    if mask is not None and not cfg.get("nomask", True):
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)    # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # sigmoid gates (computed in FP32 for stability, cast back to FP16)
    left_gate  = torch.sigmoid(left_gate.float()).to(left.dtype)
    right_gate = torch.sigmoid(right_gate.float()).to(right.dtype)
    out_gate   = torch.sigmoid(out_gate.float()).to(out_gate.dtype)

    # apply gates to left/right
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring hidden dimension into the batch dimension
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * hidden_dim, N, N)      # [B*H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)     # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension, apply out_gate, final proj
    # ------------------------------------------------------------------
    # LayerNorm over the hidden dim (H) – use torch.nn.functional for brevity
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )

    # multiply by out_gate (both in FP16)
    out_norm = out_norm * out_gate

    # final linear projection back to D (casting to FP32 for the result)
    # to_out.weight shape: [dim, hidden_dim]
    out_final = F.linear(out_norm.float(), w["to_out.weight"], bias=None)  # [B,N,N,dim] FP32

    return out_final
