"""
Outgoing TriMul (AlphaFold‑3) – heavily fused forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input pair representation (FP32 → FP16).
2️⃣ One Triton kernel fuses the five linear projections, the three
   sigmoid gates, the optional mask and stores the three tensors
   `left`, `right` and `out_gate` directly in the layout
   [B, H, N, N] (FP16).  No intermediate “[R,5·H]” tensor is ever
   written – this cuts global memory traffic by >1 GB.
3️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → shape [B, H, N, N] (FP16).
4️⃣ Row‑wise mean & rsqrt‑variance of `out` (Triton).
5️⃣ A second Triton kernel fuses the second LayerNorm, the `out_gate`
   multiplication and the final projection (to_out) – it yields the
   final tensor `[B, N, N, dim]` in FP32.

The implementation is completely gradient‑free and targets Triton 3.3.1
on an H100.  Mixed‑precision (FP16/FP32) is used throughout to minimise
memory bandwidth while preserving numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim] FP32 input
    y_ptr,           # [R, dim] FP16 output
    w_ptr,           # [dim] LN weight γ (FP32)
    b_ptr,           # [dim] LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)         # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fuse five linear projections, sigmoid gates and optional mask
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                # [R, dim] FP16 (layer‑normed input)
    mask_ptr,             # [R]    FP16 (optional)
    left_w_ptr,           # [dim, H] FP16 (transposed left_proj weight)
    right_w_ptr,          # [dim, H] FP16
    left_gate_w_ptr,      # [dim, H] FP16
    right_gate_w_ptr,     # [dim, H] FP16
    out_gate_w_ptr,       # [dim, H] FP16
    left_out_ptr,         # [B, H, N, N] FP16
    right_out_ptr,        # [B, H, N, N] FP16
    out_gate_out_ptr,     # [B, H, N, N] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # Loop over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Accumulators for the five outputs (FP32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # Loop over the input dimension
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < dim

            # Load a tile of the normalized input (FP16 → FP32)
            x = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)   # [BLOCK_R, BLOCK_D]

            # Load the corresponding weight tiles (transposed layout)
            w_l = tl.load(
                left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)   # [BLOCK_D, BLOCK_H]

            w_r = tl.load(
                right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_lg = tl.load(
                left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_rg = tl.load(
                right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_og = tl.load(
                out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # Matrix‑multiply the tile (blocked GEMM)
            left_acc      += tl.dot(x, w_l)   # [BLOCK_R, BLOCK_H]
            right_acc     += tl.dot(x, w_r)
            left_gate_acc += tl.dot(x, w_lg)
            right_gate_acc+= tl.dot(x, w_rg)
            out_gate_acc  += tl.dot(x, w_og)

        # Apply sigmoid to the three gates (FP32 → FP16)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate= (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate  = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # Gated & masked left/right (FP16)
        left  = left_acc.to(tl.float16)  * mask_val * left_gate
        right = right_acc.to(tl.float16) * mask_val * right_gate

        # Destination offsets for the 4‑D layout
        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_out_ptr  + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight, transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters (per hidden channel)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [M, K]
        term = norm * vals_gate                                         # [M, K]

        term_fp16 = term.to(tl.float16)

        # Load a tile of the final projection matrix (H, dim) – already transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()             # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat   # free original tensor

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (no intermediate tensor)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Prepare transposed weight matrices for the fused kernel
    left_w      = w["left_proj.weight"].to(torch.float16).t().contiguous()   # [dim, H]
    right_w     = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).t().contiguous()

    # Optional mask
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32                     # dim is modest (≤384)
    grid_fuse = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_fuse](
        x_norm,
        mask_ptr,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        left,
        right,
        out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Release resources we no longer need
    del x_norm, left_w, right_w, left_gate_w, right_gate_w, out_gate_w, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM: left @ rightᵀ  (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise statistics of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + projection to `dim`
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program (tiles over R)
    BLOCK_K_OUT = 32     # hidden‑dim chunk
    BLOCK_N_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
