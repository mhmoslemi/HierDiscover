@triton.jit
def apply_norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr,
    R, H,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    row_start = pid * BLOCK_R
    h_start = pid_h * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    mask_r = offs_r < R
    mask_h = offs_h < H

    # compute flatten index: index = offs_r[:,None] * H + offs_h[None,:]
    idx = offs_r[:, None] * H + offs_h[None, :]

    mask = mask_r[:, None] & mask_h[None, :]

    out_val = tl.load(out_ptr + idx, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + idx, mask=mask, other=0.0)

    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)[:, None]  # [BLOCK_R,1]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :]  # [1,BLOCK_H]
    beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]

    # compute norm
    # Convert to fp32
    out_f32 = out_val.to(tl.float32)
    gate_f32 = gate_val.to(tl.float32)

    mu_f32 = mu.to(tl.float32)
    rstd_f32 = rstd.to(tl.float32)

    lambda_f32 = gamma.to(tl.float32)
    beta_f32 = beta.to(tl.float32)

    norm = (out_f32 - mu_f32) * rstd_f32
    norm = norm * lambda_f32 + beta_f32
    norm = norm * gate_f32

    tl.store(out_ptr + idx, norm.to(tl.float16), mask=mask)

    # No return.

def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim
    # 1. LayerNorm
    # use our layernorm_kernel
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # 2. Fused projection: concatenate weights
    fused_w = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = F.linear(x_norm, fused_w)  # (R,5*H) FP16
    del x_norm, fused_w
    # 3. Split and apply mask and sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr
    # 4. Batched GEMM for core einsum
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H,N,N] FP16
    out = out.view(B, H, N, N)
    del left_mat, right_mat, left, right
    # 5. Rowwise stats
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)
    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )
    # 6. Apply LayerNorm, out_gate and LN weight+bias to out in-place
    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_norm_gate = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(H, BLOCK_H_OUT)
    )
    apply_norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out_gate, mu, rstd
    # 7. Final projection to dim
    out_flat = out.reshape(R, H)  # FP16
    w_out = w["to_out.weight"].to(torch.float16)  # (dim, H)
    final = F.linear(out_flat, w_out)  # (R, dim) FP16
    final = final.reshape(B, N, N, dim).to(torch.float32)
    return final

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

The forward consists of six stages:
1️⃣  Row‑wise LayerNorm over the input feature dimension (FP32 → FP16).
2️⃣  One fused Linear that simultaneously produces left/right projections
    and three gates (shape `[R,5·H]`, R = B·N·N).
3️⃣  Split the fused tensor, apply the optional pair‑wise mask and sigmoid
    gates, and store `left`, `right` and `out_gate` (layout `[B,H,N,N]`).
4️⃣  Core Einstein‑summation → batched GEMM `left·rightᵀ` (cuBLAS, FP16).
5️⃣  Row‑wise mean / rsqrt‑variance of the contracted tensor (LayerNorm stats).
6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection (`to_out`) in two kernels:
   – `apply_norm_gate_kernel` normalises, scales, biases and masks the tensor
     **in‑place**.
   – A standard FP16 linear layer completes the projection to the output
     dimension.

All heavy compute (matmuls, reductions) runs in FP16; reductions that need
high precision are performed in FP32.  The implementation works for batch
size 1‑2, sequence length up to 1024, hidden dimension 128 and model
dimensions 128/384 on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, D]  FP32 input (flattened)
    y_ptr,        # [R, D]  FP16 output
    w_ptr,        # [D]    LayerNorm weight γ (FP32)
    b_ptr,        # [D]    LayerNorm bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalised values -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16  (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b,i,k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # broadcast optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                        # [BLOCK_R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k          # start for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr      + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R]          FP32
    rstd_ptr,     # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base_off = b * (H * hidden_stride) + i * N + j      # start at h = 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    base_off = b * (H * hidden_stride) + i * N + j      # start for h = 0
    idx = base_off[:, None] + offs_h[None, :] * hidden_stride

    mask = mask_r[:, None] & mask_h[None, :]

    out_val  = tl.load(out_ptr  + idx, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + idx, mask=mask, other=0.0)

    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)[:, None]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)[:, None]

    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :]
    beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]

    # FP32 arithmetic for stability
    out_f   = out_val.to(tl.float32)
    gate_f  = gate_val.to(tl.float32)
    mu_f    = mu.to(tl.float32)
    rstd_f  = rstd.to(tl.float32)
    gamma_f = gamma.to(tl.float32)
    beta_f  = beta.to(tl.float32)

    norm = (out_f - mu_f) * rstd_f          # (out‑μ)·rstd
    norm = norm * gamma_f + beta_f          # γ·norm + β
    norm = norm * gate_f                    # * out_gate

    tl.store(out_ptr + idx, norm.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# 5️⃣ Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim]    (float32)
            - mask_tensor  : torch.Tensor [B, N, N]        (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()              # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused Linear (5·H) → produce left/right & gates
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum → batched GEMM (left·rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                 # covers H ≤ 768
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and LN parameters (in‑place)
    # --------------------------------------------------------------
    BLOCK_R_OUT = 128
    BLOCK_H_OUT = 64
    grid_norm_gate = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(H, BLOCK_H_OUT),
    )
    apply_norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ   (FP32)
        w["to_out_norm.bias"],     # β   (FP32)
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out_gate, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final projection to `dim`
    # --------------------------------------------------------------
    # Move hidden dimension to the last axis and make it contiguous.
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out_perm.view(R, H)                    # [R, H] FP16

    w_out = w["to_out.weight"].to(torch.float16)      # [dim, H]
    final = F.linear(out_flat, w_out)                # [R, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
