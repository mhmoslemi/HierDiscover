"""
Optimized outgoing TriMul forward implementation (no gradients).

Algorithm:
1. Apply a per‑pair LayerNorm on the last dimension (FP32 → FP16) using PyTorch's highly‑optimized layer_norm.
2. Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate) into a single
   half‑precision GEMM (cuBLAS). The result has shape (R,5·H) where R = B·N·N.
3. A small Triton kernel decodes the flattened rows, applies the optional pair‑mask,
   computes the sigmoid gates and stores the three tensors (left, right, out_gate) in layout
   [B, H, N, N] (FP16).
4. Batched GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
5. Transpose the tensors to shape [B, N, N, H] and apply the second LayerNorm (to_out_norm)
   using `torch.nn.functional.layer_norm`. Multiply by the out_gate and finally a linear
   projection (to_out) to obtain the output of shape [B, N, N, dim] (float32).

All heavy linear algebra is delegated to cuBLAS; the Triton kernel only handles the
element‑wise mask and gating, which reduces kernel launch overhead and improves overall
runtime compared to a fully‑Triton implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: split fused projection, apply mask and sigmoid gates
# -------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16 output
    right_ptr,      # [B, H, N, N] FP16 output
    out_gate_ptr,   # [B, H, N, N] FP16 output
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride for hidden dimension inside the flattened tensor
    hidden_stride = N * N                         # distance between two consecutive H‑planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Load the five concatenated projection slices
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numerical stability, then back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                    # no mask for out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient) – highly optimized with
    cuBLAS / torch.nn.functional.layer_norm and a small Triton kernel for
    the element‑wise mask & sigmoid gates.
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    # Use the highly‑optimized torch implementation
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B,N,N,dim]  FP32
    x_norm = x_norm.to(torch.float16)    # [B,N,N,dim]  FP16

    # -----------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·H)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # [R, 5·H]  FP16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣  Split, optional mask, sigmoid gates (Triton)
    # -----------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor; never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64          # rows per program
    BLOCK_H = 64          # hidden channels per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣  Core pairwise multiplication (batched GEMM)
    # -----------------------------------------------------------------
    # Fuse the hidden dimension into the batch dimension for cuBLAS
    left_mat  = left.view(B * H, N, N)                  # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                 # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣  Second LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    # Bring the hidden dimension to the innermost axis to make the
    # LayerNorm reduction contiguous.
    out = out.permute(0, 2, 3, 1).contiguous()             # [B, N, N, H]  FP16
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]  FP16

    # Apply LayerNorm over H (the hidden dimension)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # still FP16 (internally computed in FP32)

    # Multiply by the out‑gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim` (FP32 result)
    final = F.linear(out_norm, w["to_out.weight"])   # [B, N, N, dim]  FP32

    return final
```

"""
Optimised outgoing TriMul (forward‑only) implementation.

Algorithm
---------
1️⃣  Input LayerNorm on the last dimension (FP32 → FP16) using the highly‑optimised
    PyTorch `layer_norm` kernel.

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM (`torch.nn.functional.linear`).  The result has
    shape (R, 5·H) where R = B·N·N and H = hidden_dim.

3️⃣  A small Triton kernel (`split_fuse_gate_kernel`) decodes the flattened rows,
    optionally applies the pair‑mask, computes the sigmoid gates and writes the
    three tensors (`left`, `right`, `out_gate`) in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (`torch.bmm`) evaluates the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   →  shape [B, H, N, N] (FP16).

5️⃣  Bring the hidden dimension to the innermost axis, apply the second LayerNorm
    (`to_out_norm`) using `torch.nn.functional.layer_norm`, multiply by the
    `out_gate` and finally a linear projection (`to_out`) back to `dim`.  The
    final output is FP32 with shape [B, N, N, dim].

Only the mask‑/gate‑fusion step uses a custom Triton kernel; all heavy linear
algebra is delegated to cuBLAS / the PyTorch kernels, which gives a large speed‑up
versus a fully‑Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask and sigmoid gates
# -----------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]    FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16 output
    right_ptr,      # [B, H, N, N] FP16 output
    out_gate_ptr,   # [B, H, N, N] FP16 output
    mask_ptr,       # [R]          FP16 mask (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flat index r = b·N·N + i·N + k ----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ---- optional mask (broadcast over H) ----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                                 # distance between two H‑planes
    base_offset = b * (H * hidden_stride) + i * N + k    # offset for h = 0

    # ---- process H in tiles ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the five concatenated projection slices ----
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ---- sigmoid (computed in FP32 for stability) ----
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                        # out_gate is *not* masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# -----------------------------------------------------------------
# Entry point
# -----------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient).  The implementation relies on
    cuBLAS / PyTorch kernels for the heavy linear algebra and on a compact
    Triton kernel for the mask / sigmoid fusion.
    """
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim]  FP32
    x_norm = x_norm.to(torch.float16)  # cast to FP16 for the subsequent GEMM

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # [R, 5·H] FP16
    del x_norm, fused_w

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates (Triton)
    # -----------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64   # rows processed per program
    BLOCK_H = 64   # hidden channels processed per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Core pairwise multiplication (batched GEMM)
    # -----------------------------------------------------------------
    # Collapse B and H into the batch dimension for cuBLAS
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm, gate multiplication and final projection
    # -----------------------------------------------------------------
    # Move the hidden dimension to the innermost axis (contiguous layout)
    out = out.permute(0, 2, 3, 1).contiguous()           # [B, N, N, H]  FP16
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]  FP16

    # LayerNorm over the hidden dimension H
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                   # still FP16 (internally FP32)

    # Multiply by the pre‑computed out‑gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim` (produces FP32)
    final = F.linear(out_norm, w["to_out.weight"])      # [B, N, N, dim]  FP32

    return final
