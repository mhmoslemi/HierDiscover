        v_beta = gate_fp16 * beta_fp16[None, :]              # [M, K]

        acc_out += tl.dot(v_out, w_tile).to(tl.float32)
        acc_gamma += tl.dot(v_gamma, w_tile).to(tl.float32)
        acc_beta += tl.dot(v_beta, w_tile).to(tl.float32)

    var = sum_sq / H - mu * mu

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fused Triton implementation (forward only).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16).
2️⃣ One fused linear projection computes the five maps
   (left, right, left_gate, right_gate, out_gate) in FP16.
3️⃣ A Triton kernel splits the projection, applies an optional mask,
   evaluates the sigmoid gates and stores `left`, `right` and `out_gate`
   in layout [B, H, N, N] (H = hidden_dim).
4️⃣ Batched GEMM (`torch.bmm`) evaluates the core einsum
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] → tensor `[B, H, N, N]`.
5️⃣ The final Triton kernel fuses **row‑wise statistics**, the
   second LayerNorm, the post‑norm gate, and the projection to the output
   dimension.  For every pair‑index (b,i,j) it computes

        μ   = mean_h(out)
        σ²  = var_h(out)
        rstd = rsqrt(σ² + eps)

        out_norm = γ * (out - μ) * rstd + β          (γ,β = to_out_norm)
        out_gate = σ( out_gate_raw )                (already in fp16)

        out_final = Σ_h (out_norm * out_gate)·W_out   (W_out : H×dim)

   The kernel accumulates three dot‑products in one pass:
        Σ_h out·gate·γ·W,   Σ_h gate·γ·W,   Σ_h gate·β·W,
   then combines them with the mean / rstd to produce the final FP32
   result `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and target an H100.  Mixed
precision (FP16 compute, FP32 reduction) is used throughout while
keeping the API identical to the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim] FP32 input
    y_ptr,          # [R, dim] FP16 output
    w_ptr,          # [dim]    LN weight (γ) FP32
    b_ptr,          # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values --------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index into (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # View of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → FP16 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        # out_gate is used after the second LN, therefore only sigmoid here
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse per‑row statistics, second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,        # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # statistics accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # three dot‑product accumulators (out·gate·γ·W, gate·γ·W, gate·β·W)
    acc_out   = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_gamma = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_beta  = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        # load out and gate (FP16)
        ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_fp16  = tl.load(ptr_out,
                             mask=mask_m[:, None] & mask_k[None, :],
                             other=0.0)
        gate_fp16 = tl.load(ptr_gate,
                             mask=mask_m[:, None] & mask_k[None, :],
                             other=0.0)

        # statistics (FP32)
        out_fp32 = out_fp16.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

        # LayerNorm parameters for this hidden slice (FP32 → FP16)
        gamma_fp32 = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)
        beta_fp32  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)
        gamma_fp16 = gamma_fp32.to(tl.float16)
        beta_fp16  = beta_fp32.to(tl.float16)

        # load weight tile (FP16)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                     # [K, TILE_N]

        # ---- three fused contributions ---------------------------------
        # 1) out·gate·γ
        v_out = out_fp16 * gate_fp16 * gamma_fp16[None, :]      # [M, K]
        # 2) gate·γ
        v_gamma = gate_fp16 * gamma_fp16[None, :]               # [M, K]
        # 3) gate·β
        v_beta = gate_fp16 * beta_fp16[None, :]                 # [M, K]

        # dot products   (M,K)·(K,N) → (M,N)   -> FP32 accumulation
        acc_out   += tl.dot(v_out,   w_tile).to(tl.float32)
        acc_gamma += tl.dot(v_gamma, w_tile).to(tl.float32)
        acc_beta  += tl.dot(v_beta,  w_tile).to(tl.float32)

    # ----- per‑row mean / rsqrt‑var ---------------------------------------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----- combine the three partial results -------------------------------
    # final = rstd * ( Σ(out·gate·γ·W) - μ * Σ(gate·γ·W) ) + Σ(gate·β·W)
    term = acc_out - mu[:, None] * acc_gamma
    final = rstd[:, None] * term + acc_beta

    # store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # Shape: [B, H, N, N] → view as [B·H, N, N] for bmm
    left_mat = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    # Release memory we no longer need
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Fused row‑wise stats + second LayerNorm + out_gate + final proj
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # Cleanup
    del out, out_gate, w_out_T

    return final
