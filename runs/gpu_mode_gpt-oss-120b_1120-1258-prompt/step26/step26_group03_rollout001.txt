"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flat pair tensor (FP32 → FP16) – use the
    highly‑optimized torch.nn.functional.layer_norm.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:  (R, dim) @ (5·H, dim).T  → (R, 5·H)  (FP16).

3️⃣  Split the fused projection, apply the optional pair mask and three
    sigmoid gates in a Triton kernel → tensors *left*, *right*,
    *out_gate* with layout [B, H, N, N] (FP16).

4️⃣  Core einsum (out[i,j] = Σₖ left[i,k]·right[j,k]) is a batched GEMM:
    left.reshape(B·H,N,N) @ right.reshape(B·H,N,N)ᵀ → out (FP16).

5️⃣  Row‑wise mean/variance of *out* over the hidden dimension H,
    LayerNorm (γ/β), multiplication with *out_gate*,
    and the final projection to the original dimensionality are all
    performed with plain PyTorch (float32 statistics → float16 math).

All tensors are kept on the device; the only Triton kernels are the
gate‑splitting kernel (step 3) and a tiny utility kernel for the first
LayerNorm (optional – we use torch‑layer‑norm for speed).  The final
output is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened pair index r = b·N·N + i·N + k,
    read the 5·H fused projection values, split them into
    left, right, left_gate, right_gate, out_gate,
    apply an optional pair mask and the three sigmoid gates,
    and write the results into three tensors of shape
    (B, H, N, N).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for one hidden plane inside the output tensors
    hidden_stride = N * N
    # base offset for hidden‑plane 0 (b, h=0, i, k)
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # iterate over hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # address of the fused projection for this block of rows and hidden
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        # left / right / gates (raw FP16)
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination offset for hidden plane `h = offs_h`
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing the nine model weights:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all stored as float32)
        - config_dict must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using PyTorch’s fast kernel
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous().float()
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # [R, dim] FP16
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into one GEMM (FP16)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    # (R, dim) @ (5·H, dim).T  → (R, 5·H)
    proj = F.linear(x_norm, fused_weight)   # FP16
    del fused_weight, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask is broadcast over the hidden dimension, stored as FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – Triton still expects a pointer
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Triton configuration
    BLOCK_R = 64   # rows per program (tunable)
    BLOCK_H = 64   # hidden tiles per program (tunable)

    grid = (triton.cdiv(R, BLOCK_R),)      # only one dimension of programs
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) → treat hidden planes as independent batches
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm over hidden dimension, out_gate multiplication,
    #    and final projection to `dim`
    # ------------------------------------------------------------------
    # mean / variance in FP32 for numerical stability
    mu   = out.mean(dim=1, keepdim=True, dtype=torch.float32)      # [B,1,N,N]
    var  = out.var(dim=1, unbiased=False, keepdim=True, dtype=torch.float32)
    rstd = torch.rsqrt(var + eps)                                 # [B,1,N,N]

    # normalize (still FP32)
    out_fp32 = (out.to(torch.float32) - mu) * rstd               # [B,H,N,N] FP32

    # LayerNorm parameters (γ,β) are FP32 – broadcast over (B,N,N)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)               # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)                # [1,H,1,1]

    out_fp32 = out_fp32 * gamma + beta                          # [B,H,N,N] FP32

    # multiply by out_gate (FP16 → back to FP16)
    out_fp16 = out_fp32.to(torch.float16) * out_gate            # [B,H,N,N] FP16

    # flatten (B,N,N,H) → (R, H) for final linear projection
    out_flat = out_fp16.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R,H) FP16

    # final projection weight: shape (dim, H) → convert to FP16 once
    to_out_w = w["to_out.weight"].to(torch.float16)                     # (dim, H)

    final_fp16 = F.linear(out_flat, to_out_w)                           # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)              # (B,N,N,dim) FP32

    return final
