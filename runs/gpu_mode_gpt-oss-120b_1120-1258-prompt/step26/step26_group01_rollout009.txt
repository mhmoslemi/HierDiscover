"""
TriMul outgoing operation implementation with fused final LayerNorm, gating and
projection. The first LayerNorm, linear projections, and the core pairwise
multiplication are performed with torch (cuBLAS / torch kernels). A single
Triton kernel fuses the second LayerNorm, out‑gate multiplication and the final
linear projection, reducing memory traffic and kernel launch overhead.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def final_output_kernel(
    out_ptr,          # [R, H]      FP16
    gate_ptr,         # [R, H]      FP16
    mu_ptr,           # [R]         FP32
    rstd_ptr,         # [R]         FP32
    ln_w_ptr,         # [H]         FP32
    ln_b_ptr,         # [H]         FP32
    w_out_ptr,        # [H, D]      FP16   (weight transposed)
    out_ptr_final,    # [R, D]      FP32
    total_rows, H, D,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # 2‑D grid: rows (R) × output‑dim (D)
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_d = col_start + tl.arange(0, BLOCK_D)

    mask_r = offs_r < total_rows
    mask_d = offs_d < D

    # Load per‑row mean / rstd (scalar per row)
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R] FP32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R] FP32

    # Accumulator for the final matmul (FP32)
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # Loop over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate tiles
        # out_ptr layout: row stride = H
        out_tile = tl.load(
            out_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_K]

        gate_tile = tl.load(
            gate_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_K]

        # ----- second LayerNorm (γ,β) -----
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32

        # (out - mu) * rstd  ->  normalized
        # mu and rstd are (BLOCK_R, 1) after broadcasting
        norm = (out_tile - mu[:, None]) * rstd[:, None]

        # Apply γ,β
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply out‑gate
        gated = norm * gate_tile          # still FP32

        # Cast to FP16 for the dot product (cuBLAS style)
        gated_fp16 = gated.to(tl.float16)

        # Load weight tile: [H, D] where stride0 = D
        w_tile = tl.load(
            w_out_ptr + offs_k[:, None] * D + offs_d[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0,
        )   # [BLOCK_K, BLOCK_D] FP16

        # Accumulate: (BLOCK_R, BLOCK_K)·(BLOCK_K, BLOCK_D) → (BLOCK_R, BLOCK_D)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # Store result
    tl.store(
        out_ptr_final + offs_r[:, None] * D + offs_d[None, :],
        acc,
        mask=mask_r[:, None] & mask_d[None, :]
    )

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Steps
    -----
    1️⃣ LayerNorm on the last dimension (float32→float16)
    2️⃣ Fused linear projection of five heads (5·hidden_dim)
    3️⃣ Split, apply optional mask and sigmoid gates (torch ops)
    4️⃣ Batched GEMM (left @ rightᵀ) for the O(N³) contraction (cuBLAS)
    5️⃣ Row‑wise mean / variance across the hidden dimension (torch reductions)
    6️⃣ A single Triton kernel fuses the second LayerNorm, the out‑gate
       multiplication and the final linear projection to the model dimension.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    nomask = cfg.get("nomask", True)

    B, N, _, C = x.shape   # C == dim
    R = B * N * N
    H = hidden_dim

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -------------------------------------------------
    x_norm = F.layer_norm(x, (C,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_norm = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Fused projection (5 · hidden_dim)
    # -------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, C]
    proj = F.linear(x_norm, fused_w)    # (B,N,N,5·H)  FP16
    proj = proj.reshape(R, 5, H)       # [R,5,H]

    # -------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates (all torch)
    # -------------------------------------------------
    left = proj[:, 0, :]               # [R, H]
    right = proj[:, 1, :]              # [R, H]
    left_gate = torch.sigmoid(proj[:, 2, :])
    right_gate = torch.sigmoid(proj[:, 3, :])
    out_gate = torch.sigmoid(proj[:, 4, :])

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).unsqueeze(-1)   # [R,1]
        left = left * mask_flat * left_gate
        right = right * mask_flat * right_gate
    else:
        left = left * left_gate
        right = right * right_gate
    # out_gate stays unchanged (already sigmoid)

    # reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # -------------------------------------------------
    # 4️⃣ Core O(N³) contraction (batched matmul)
    # -------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16

    # -------------------------------------------------
    # 5️⃣ Row‑wise mean / variance across H (torch)
    # -------------------------------------------------
    out_f32 = out.to(torch.float32)
    mu = out_f32.mean(dim=1)                  # [B, N, N] FP32
    var = out_f32.var(dim=1, unbiased=False) # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)             # [B, N, N] FP32

    mu_flat = mu.reshape(-1).contiguous()    # [R] FP32
    rstd_flat = rstd.reshape(-1).contiguous()# [R] FP32

    # -------------------------------------------------
    # 6️⃣ Fuse 2nd LayerNorm, out_gate and final projection
    # -------------------------------------------------
    # flatten tensors for the kernel
    out_flat = out.reshape(R, H).contiguous()
    gate_flat = out_gate.reshape(R, H).contiguous()

    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_R = 128
    BLOCK_K = 32
    BLOCK_D = 128   # tile over output dimension

    grid = (
        triton.cdiv(R, BLOCK_R),
        triton.cdiv(dim, BLOCK_D),
    )
    final_output_kernel[grid](
        out_flat,
        gate_flat,
        mu_flat,

"""
TriMul (outgoing) – forward‑only implementation.

Algorithm
---------
1️⃣  First LayerNorm (float32) on the last dimension → FP16.
2️⃣  Fuse the 5 linear maps (left, right, left_gate, right_gate, out_gate) into
    a single GEMM: (R, dim) × (5·H, dim)ᵀ  where  R = B·N·N.
3️⃣  Split the fused projection, optionally apply the pairwise mask,
    compute sigmoid gates and reshape to [B, H, N, N] (all FP16).
4️⃣  Batched matrix‑multiply:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
    implemented with a single cuBLAS batched GEMM (torch.bmm).
5️⃣  Compute the per‑pair mean / variance of ``out`` over the hidden
    dimension H with torch reductions (FP32 → FP16 loss‑less).
6️⃣  One Triton kernel fuses the second LayerNorm, the out‑gate, and the final
    projection (H → dim).  The kernel reads:
        • out    – [R, H]  FP16
        • gate   – [R, H]  FP16          (out_gate)
        • μ,σ⁻¹ – [R]    FP32
        • γ,β    – [H]    FP32          (to_out_norm)
        • W      – [H, D] FP16          (to_out weight, D = dim)
    and directly writes the final tensor [R, D] (FP32), which is reshaped
    to [B, N, N, dim].

Only step 6 uses a custom kernel; all other heavy‑weight operations
( LayerNorm, GEMM, reductions ) are delegated to highly‑optimised PyTorch
/cuBLAS kernels, giving a fast overall runtime on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣  Final fused LayerNorm + gating + projection (Triton)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [R, H]      FP16   (contracted tensor)
    gate_ptr,         # [R, H]      FP16   (out_gate)
    mu_ptr,           # [R]         FP32   (row‑wise mean)
    rstd_ptr,         # [R]         FP32   (row‑wise rsqrt‑var)
    ln_w_ptr,         # [H]         FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]         FP32   (to_out_norm.bias)
    w_out_ptr,        # [H, D]      FP16   (to_out weight transposed)
    out_final_ptr,    # [R, D]      FP32   (output)
    total_rows, H, D,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # 2‑D grid: rows (R) × output‑dim (D)
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    offs_c = col_start + tl.arange(0, BLOCK_D)      # [BLOCK_D]

    mask_r = offs_r < total_rows
    mask_c = offs_c < D

    # ------------------------------------------------------------------
    # Load per‑row statistics (mean & rsqrt‑variance) – broadcast later
    # ------------------------------------------------------------------
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R] FP32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R] FP32

    # Accumulator for the final matmul (FP32)
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over hidden dimension H in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)            # [BLOCK_K]
        mask_k = offs_k < H

        # Load out and gate tiles (row‑major layout: stride = H)
        out_tile = tl.load(
            out_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_K]

        gate_tile = tl.load(
            gate_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_K]

        # ---------- second LayerNorm (γ,β) ----------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K] FP32

        # (out - μ) * rstd  →  normalized
        norm = (out_tile - mu[:, None]) * rstd[:, None]

        # Apply γ, β
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply out‑gate
        gated = norm * gate_tile                               # FP32

        # Cast to FP16 for the dot product (cuBLAS‑style)
        gated_fp16 = gated.to(tl.float16)

        # Load weight tile: [H, D] with stride0 = D
        w_tile = tl.load(
            w_out_ptr + offs_k[:, None] * D + offs_c[None, :],
            mask=mask_k[:, None] & mask_c[None, :],
            other=0.0,
        )                                                       # [BLOCK_K, BLOCK_D] FP16

        # Accumulate (BLOCK_R, BLOCK_K)·(BLOCK_K, BLOCK_D) → (BLOCK_R, BLOCK_D)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the final output
    # ------------------------------------------------------------------
    tl.store(
        out_final_ptr + offs_r[:, None] * D + offs_c[None, :],
        acc,
        mask=mask_r[:, None] & mask_c[None, :]
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict must contain
            "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                # input/channel dimension C
    hidden_dim = cfg["hidden_dim"]  # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, C = x.shape
    R = B * N * N            # number of pairwise positions
    H = hidden_dim
    D = dim                  # final output dimension

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimised on GPUs.
    x_norm = F.layer_norm(
        x,
        (C,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )
    x_norm = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣  Fused linear projection – 5 heads (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, C]
    proj = F.linear(x_norm, fused_w)                    # (B,N,N,5·H) FP16
    proj = proj.reshape(R, 5, H)                       # [R,5,H]

    # --------------------------------------------------------------
    # 3️⃣  Split, optional mask, sigmoid gates (pure torch)
    # --------------------------------------------------------------
    left      = proj[:, 0, :]               # [R, H]
    right     = proj[:, 1, :]               # [R, H]
    left_gate = torch.sigmoid(proj[:, 2, :])
    right_gate= torch.sigmoid(proj[:, 3, :])
    out_gate  = torch.sigmoid(proj[:, 4, :])

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).unsqueeze(-1)   # [R,1]
        left  = left  * mask_flat * left_gate
        right = right * mask_flat * right_gate
    else:
        left  = left  * left_gate
        right = right * right_gate

    # reshape to [B, H, N, N] (contiguous layout)
    left      = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right     = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate  = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣  Core O(N³) contraction via a single batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣  Row‑wise mean / variance over H (torch reductions, FP32)
    # --------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    mu   = out_f32.mean(dim=1)                # [B, N, N] FP32
    var  = out_f32.var(dim=1, unbiased=False) # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)             # [B, N, N] FP32

    mu_flat   = mu.reshape(-1).contiguous()   # [R] FP32
    rstd_flat = rstd.reshape(-1).contiguous() # [R] FP32

    # ------------------------------------------------------------------
    # 6️⃣  Fuse second LayerNorm, out‑gate, and final projection (Triton)
    # ------------------------------------------------------------------
    # flatten tensors to shape [R, H] for the kernel
    out_flat = out.reshape(R, H).contiguous()
    gate_flat = out_gate.reshape(R, H).contiguous()

    # to_out weight transposed → shape (H, D)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, D]

    final = torch.empty(R, D, dtype=torch.float32, device=device)

    BLOCK_R = 128          # rows per program
    BLOCK_K = 32           # hidden‑dim tile
    BLOCK_D = 128          # output‑dim tile

    grid = (
        triton.cdiv(R, BLOCK_R),
        triton.cdiv(D, BLOCK_D),
    )
    final_output_kernel[grid](
        out_flat,
        gate_flat,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        D,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    out_final = final.view(B, N, N, D)      # [B, N, N, dim] (float32)
    return out_final
