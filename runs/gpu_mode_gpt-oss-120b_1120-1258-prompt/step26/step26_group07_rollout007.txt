@triton.jit
def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr, R, H, N, HAS_MASK: tl.constexpr, BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_R
    offs_r = start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute base offset for writing into left/right/out_gate: base = b*(H*nn) + i*N + k
    # each hidden plane h offset step: hidden_stride = nn
    hidden_stride = nn

    # The fused projection layout: [left|right|left_gate|right_gate|out_gate] contiguous in hidden dimension.
    # So for each row r, index base h offset = r * (5*H)
    # We'll load blocks of H across hidden dimension.

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0) # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast to hidden dim
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load fused projection for left
        # note: offset within proj: proj_ptr + (offs_r * 5*H + offs_h) for left
        # We'll compute base pointer per row: base_proj = offs_r * (5*H)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Input LayerNorm (torch.nn.functional.layer_norm, FP32 → FP16).
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → tensor (R, 5·H) with R = B·N·N.
3️⃣  Triton kernel **split_fuse_gate_kernel**:
        * splits the fused projection,
        * applies optional mask,
        * evaluates the sigmoid gates,
        * writes the tensors *left*, *right* and *out_gate* in layout [B, H, N, N]
          (FP16) – all operations are fused, saving bandwidth.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
        → tensor [B, H, N, N] (FP16).
5️⃣  Second LayerNorm + out‑gate multiplication + final linear projection are
    performed with PyTorch kernels (layer_norm + linear) which internally use
    highly‑optimized cuBLAS tensor‑core kernels.
6️⃣  Result is reshaped to [B, N, N, dim] and returned as FP32.

Only the split‑and‑gate step is written in Triton; all heavy matrix‑multiplications
are delegated to cuBLAS for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, apply mask & sigmoid gates (FP16)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fused projection (FP16)
    left_ptr,          # [B, H, N, N] left  (FP16)
    right_ptr,         # [B, H, N, N] right (FP16)
    out_gate_ptr,      # [B, H, N, N] out_gate (FP16)
    mask_ptr,          # [R] optional mask (FP16)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    #  Decode flat index (b, i, k) from r = b·N·N + i·N + k
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    #  Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride in the [B, H, N, N] layout
    hidden_stride = nn                     # stride for a single hidden plane
    # Base offset for the first hidden plane of (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k          # [BLOCK_R]

    # ------------------------------------------------------------------
    #  Loop over the hidden dimension in blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H

        # ---- fused projection layout  (left | right | left_gate | right_gate | out_gate)
        # --------------------------------------------------------------
        #  proj[r, h]  -> left
        #  proj[r, h+H] -> right
        #  proj[r, h+2H] -> left_gate
        #  proj[r, h+3H] -> right_gate
        #  proj[r, h+4H] -> out_gate
        # --------------------------------------------------------------
        # base address for the current hidden block
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for numerical stability, then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets into the 4‑D tensors (flattened)
        dst = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32, linear maps have *no* bias)
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of (b,i,k) tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to FP16 for the following GEMMs
    x_norm_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM (FP16)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = F.linear(x_norm_fp16, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm_fp16

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # flatten mask to [R] and cast to FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Triton launch parameters
    BLOCK_R = 64      # rows (tokens) per program
    BLOCK_H = 64      # hidden‑plane chunk size
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM: out = left @ rightᵀ
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for cuBLAS batch GEMM
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication & final projection
    # --------------------------------------------------------------
    # flatten to (R, H) – need a contiguous layout
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP16

    # LayerNorm over the hidden dimension (per‑row)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    # Apply the out‑gate
    out_norm_gate = out_norm * out_gate_flat   # FP16

    # Final linear projection (cuBLAS GEMM)
    w_out_fp16 = weights["to_out.weight"].to(torch.float16)   # shape [dim, H]
    final_flat = F.linear(out_norm_gate, w_out_fp16)          # (R, dim) FP16

    # reshape to the expected output layout and cast to FP32
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
