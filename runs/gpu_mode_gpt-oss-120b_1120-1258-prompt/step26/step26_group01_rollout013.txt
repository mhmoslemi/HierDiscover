@triton.jit
def apply_norm_gate_kernel(
    out_ptr,  # [B, H, N, N] FP16
    gate_ptr, # [B, H, N, N] FP16
    mu_ptr,   # [R] FP32
    rstd_ptr, # [R] FP32
    ln_w_ptr, # [H] FP32
    ln_b_ptr, # [H] FP32
    out_buf_ptr, # [R, H] FP16
    R, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    start_m = pid * BLOCK_M
    offs_m = start_m + tl.arange(0, BLOCK_M)  # row indices
    mask_m = offs_m < R

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j

    # Load mu and rstd for each row
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0) # [BLOCK_M] FP32

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)  # hidden idx
        mask_k = offs_k < H

        # Load out and gate on hidden tile
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # Normalize
        norm_tile = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        gated_tile = norm_tile * gate_tile

        # Cast to fp16 and store
        tl.store(out_buf_ptr + offs_m[:, None] + offs_k[None, :] * R,
                 gated_tile.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])
```

"""
TriMul (outgoing) – forward‑only Triton implementation.

The operator works on a 4‑D tensor X∈ℝ^{B×N×N×C}.
Steps
------
1️⃣  Row‑wise LayerNorm over the last dimension (C).  (torch.nn.functional.layer_norm)
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into one
    GEMM:  (B·N·N, C) @ (5·H, C)ᵀ  → (B·N·N, 5·H).  H = hidden_dim.
3️⃣  Split the fused projection, apply the optional mask, compute sigmoids for the three
    gates and store   left, right ∈ ℝ^{B×H×N×N}  (FP16) and raw out_gate.
4️⃣  Batched GEMM (cuBLAS) computes the core contraction
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   →  (B, H, N, N) (FP16).
5️⃣  Row‑wise mean & variance of `out` over the hidden dimension H (Triton kernel).
6️⃣  Fuse the second LayerNorm (γ,β), the out_gate sigmoid, and the element‑wise
    gating into a single Triton kernel which writes a dense buffer Y∈ℝ^{R×H}
    (R = B·N·N).  Y = LN(out) ⊙ σ(out_gate)   (FP16).
7️⃣  Final projection is a regular GEMM performed by cuBLAS:
        Y @ W_outᵀ  →  (R, C)  (FP16).  Cast to float32 and reshape
    to (B, N, N, C).

All kernels target Triton 3.3.1 and the H100 GPU.  Mixed‑precision (FP16) is used
except where FP32 is required for numerical stability (layer‑norm statistics,
sigmoid).  The final result matches the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16  (raw, not sigmoid yet)
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (scalar broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride from (b,i,k) to the start of hidden dim 0
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointer to concatenated projection: [R, 5·H] → (r, h)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # ------------------------------------------------------------------
        # 5 fused linear outputs
        # ------------------------------------------------------------------
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # sigmoid gates (computed in FP32 for stability)
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate_raw                          # raw; sigmoid applied later

        # store results – layout (B, H, N, N)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (out)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate sigmoid and gating → dense buffer Y
# ----------------------------------------------------------------------
@triton.jit
def apply_norm_gate_kernel(
    out_ptr,        # [B, H, N, N] FP16  (pre‑norm)
    gate_ptr,       # [B, H, N, N] FP16  (raw out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    y_ptr,          # [R, H] FP16  (output buffer)
    R, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # hidden‑dim 0 offset

    # load per‑row mean / rstd (scalar per row)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        # load raw contracted values and raw gate values
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)     # [BLOCK_M, BLOCK_K]

        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)     # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # LN: (out - μ) * rstd  →  *γ + β
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # sigmoid on gate (σ)
        gate_sig = (1.0 / (1.0 + tl.exp(-gate_tile))).to(tl.float32)

        gated = norm * gate_sig            # [BLOCK_M, BLOCK_K]

        # write to dense buffer Y (row‑major: [R, H])
        tl.store(y_ptr + offs_m[:, None] * H + offs_k[None, :],
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient) using Triton + cuBLAS.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    R = B * N * N  # number of (b,i,j) positions

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the channel dimension (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] fp32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                      # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – uses cuBLAS GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                               # (R, 5·H) fp16
    del fused_w, x_norm, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates → left, right, out_gate (raw)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.reshape(R).to(torch.float16).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] fp16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → full tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm + out_gate sigmoid + gating → dense buffer Y [R, H]
    # ------------------------------------------------------------------
    y = torch.empty(R, H, dtype=torch.float16, device=device)   # pre‑projection buffer

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 64                     # tile over hidden dim
    grid_norm_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    apply_norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        y,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del out, out_gate, mu, rstd   # free intermediate memory

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS) and reshape
    # ------------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16)          # [dim, H]
    final_fp16 = torch.nn.functional.linear(y, w_out_fp16)    # (R, dim) fp16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)    # (B, N, N, dim) fp32
    return final
