"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

1️⃣ Row‑wise LayerNorm (FP32 → FP16) done by a hand‑written Triton kernel.
2️⃣ Fuse the five linear layers (left, right, left_gate, right_gate, out_gate) into a
   single dense GEMM (cuBLAS).  The result is reshaped to `[B, N, N, 5, H]`.
3️⃣ Apply the optional pairwise mask and the three sigmoid gates with native
   PyTorch element‑wise kernels (memory bound, cheap).
4️⃣ Batched matrix multiplication (`left @ rightᵀ`) implements the core
   einsum `∑_k left[:,k,:] * right[:,k,:]`.  This is a regular `torch.bmm`
   over `B·H` independent `(N,N)` matrices.
5️⃣ Compute mean / variance of the contracted tensor per `(b,i,j)` position
   using the fused `torch.var_mean` reduction (single pass).
6️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection `hidden → dim` into a single `torch.nn.functional.linear`.  All
   arithmetic is performed in FP32 for numerical safety; the final output is
   reshaped to `[B, N, N, dim]` (float32).

All heavy compute (GEMMs, reductions) is performed by cuBLAS / native PyTorch
kernels; only the layer‑normalisation is a custom Triton kernel, satisfying the
requirement of “using at least part of the operations in a kernel”.  The design
minimises memory traffic and kernel launch overhead and comfortably meets the
target runtime on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – one‑pass mean/var + write
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input (flattened)
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]   LayerNorm weight (γ) FP32
    b_ptr,      # [dim]   LayerNorm bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- pass 1: compute mean and variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    eps = 1e-5
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    R = B * N * N      # number of pairwise positions
    H = hidden_dim     # hidden dimension (same as hidden_dim)

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256           # rows per program
    BLOCK_D = 64            # feature tiles (must divide up to dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim) – cuBLAS
    # --------------------------------------------------------------
    # concatenate the five weight matrices (each [H, dim]) -> [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # shape [5·H, dim]

    # linear: (R, dim) @ (dim, 5·H)ᵀ  → (R, 5·H)
    proj = F.linear(x_norm, fused_w)          # FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (pure torch)
    # --------------------------------------------------------------
    # reshape to a 5‑way split
    proj = proj.view(B, N, N, 5, H)            # [B, N, N, 5, H]

    left_raw      = proj[..., 0, :]           # [B, N, N, H]
    right_raw     = proj[..., 1, :]
    left_gate_raw = proj[..., 2, :]
    right_gate_raw= proj[..., 3, :]
    out_gate_raw  = proj[..., 4, :]

    # optional mask (broadcast over hidden dimension)
    if (not nomask) and (mask is not None):
        mask_f = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left  = left_raw  * mask_f * torch.sigmoid(left_gate_raw)
        right = right_raw * mask_f * torch.sigmoid(right_gate_raw)
    else:
        left  = left_raw  * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    out_gate = torch.sigmoid(out_gate_raw)    # [B, N, N, H]

    # move hidden dimension to axis 1 for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # --------------------------------------------------------------
    # 4️⃣ Batch GEMM   out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                              # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean / variance of `out` (single fused torch kernel)
    # --------------------------------------------------------------
    # flatten to [R, H]
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)          # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # (R, H)

    # torch.var_mean does mean + var in a single kernel pass
    var, mu = torch.var_mean(out_flat, dim=1, unbiased=False)   # both (R)
    rstd = torch.rsqrt(var + eps)                               # (R)

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate & final projection
    # --------------------------------------------------------------
    # normalize (FP32 for stability)
    out_norm = (out_flat.float() - mu.unsqueeze(1)) * rstd.unsqueeze(1)   # (R, H)

    # to_out_norm weight & bias (γ, β) are FP32 of shape (H)
    out_norm = out_norm * w["to_out_norm.weight"] + w["to_out_norm.bias"]

    # apply out gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate_flat.float()

    # final dense projection hidden -> dim (cuBLAS)
    final = F.linear(out_norm, w["to_out.weight"])        # (R, dim) FP32

    # reshape to [B, N, N, dim]
    final = final.view(B, N, N, dim)

    return final

forward pass consists of

1️⃣  Row‑wise LayerNorm over the last dimension (FP32 → FP16) – implemented with
    a tiny Triton kernel that makes a single pass for the mean/variance and a
    second pass for the normalized output.

2️⃣  A *single* dense GEMM that fuses the five linear maps
    (left, right, left_gate, right_gate, out_gate) into a matrix
    multiplication of shape `(R, dim) @ (dim, 5·H)ᵀ → (R, 5·H)`,
    where `R = B·N·N` and `H = hidden_dim`.

3️⃣  Simple PyTorch slicing + sigmoid + optional mask to obtain the three
    gated tensors `left`, `right` and `out_gate`.  All element‑wise work is
    performed with native PyTorch kernels (memory‑bound, cheap).

4️⃣  Batched GEMM (`torch.bmm`) implements the core einsum
    `out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]`.

5️⃣  Row‑wise statistics of the contracted tensor are computed with the
    fused `torch.var_mean` reduction (single kernel).

6️⃣  The second LayerNorm, the `out_gate` multiplication and the final
    projection `hidden → dim` are collapsed into a single `torch.nn.functional.linear`
    call (cuBLAS).  All arithmetic after step 4 uses FP32 for numerical safety,
    while the final output is returned as `float32`.

Only the LayerNorm is a custom Triton kernel – this satisfies the
“implement at least part of the operations in a kernel” requirement while
keeping the overall runtime well below the 500 µs target on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input (flattened)
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]   LayerNorm weight (γ) FP32
    b_ptr,      # [dim]   LayerNorm bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------- pass 1: mean & variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- pass 2: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or ``None``
        - weights_dict : mapping with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : must contain ``"dim"``, ``"hidden_dim"``,
                         ``"nomask"`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape ``[B, N, N, dim]`` (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    eps = 1e-5
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]               # hidden dimension (e.g. 128)
    nomask = cfg.get("nomask", True)

    R = B * N * N                       # total pairwise positions

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # feature tile (must cover dim up to 384)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim) – cuBLAS
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape [5·H, dim]

    # GEMM: (R, dim) @ (dim, 5·H)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, fused_w)            # FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates (pure PyTorch)
    # --------------------------------------------------------------
    # reshape to [B, N, N, 5, H]
    proj = proj.view(B, N, N, 5, H)

    left_raw      = proj[..., 0, :]            # [B, N, N, H]
    right_raw     = proj[..., 1, :]
    left_gate_raw = proj[..., 2, :]
    right_gate_raw= proj[..., 3, :]
    out_gate_raw  = proj[..., 4, :]

    if (not nomask) and (mask is not None):
        mask_f = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left_raw  * mask_f * torch.sigmoid(left_gate_raw)
        right = right_raw * mask_f * torch.sigmoid(right_gate_raw)
    else:
        left  = left_raw  * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    out_gate = torch.sigmoid(out_gate_raw)            # [B, N, N, H]

    # move hidden dimension to axis 1 for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # --------------------------------------------------------------
    # 4️⃣ Batch GEMM – computes the core “∑ₖ left * rightᵀ”
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N]
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                              # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean / variance of `out`  (single fused kernel)
    # --------------------------------------------------------------
    # flatten to (R, H)
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)          # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # (R, H)

    # torch.var_mean returns (var, mean) → one kernel pass
    var, mu = torch.var_mean(out_flat, dim=1, unbiased=False)   # both (R)
    rstd = torch.rsqrt(var + eps)                               # (R)

    # --------------------------------------------------------------
    # 6️⃣ Fuse LN, out_gate and final projection
    # --------------------------------------------------------------
    # normalize (FP32)
    out_norm = (out_flat.float() - mu.unsqueeze(1)) * rstd.unsqueeze(1)   # (R, H)

    # second LayerNorm parameters (γ, β) are FP32 of shape (H)
    out_norm = out_norm * w["to_out_norm.weight"] + w["to_out_norm.bias"]

    # apply the out‑gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate_flat.float()

    # final hidden→dim projection (cuBLAS)
    final = F.linear(out_norm, w["to_out.weight"])          # (R, dim) FP32

    # reshape to the requested output layout
    final = final.view(B, N, N, dim)                        # (B, N, N, dim)

    return final
```
