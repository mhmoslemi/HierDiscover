def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. First layer norm
    x_flat = x.reshape(R, dim)  # shape
    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)   # cast to fp16 for projection
    # 2. fused projection (5*H)
    fused_weight = torch.cat([ w["left_proj.weight"], w["right_proj.weight"], w["left_gate.weight"], w["right_gate.weight"], w["out_gate.weight"] ], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_weight)   # shape [R, 5*H] fp16
    del x_norm, fused_weight
    # 3. split + mask + gating // produce left,right,out_gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. BMM: left and right contracted over k dimension
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out.view(B, H, N, N)
    del left, right

    # 5. second LayerNorm (to_out_norm) over hidden_dim
    out_perm = out.permute(0, 2, 3, 1)  # B, N, N, H
    out_norm = F.layer_norm(out_perm, (H,), weight=w["to_out_norm.weight"], bias=w["to_out_norm.bias"], eps=eps)
    # apply out_gate
    out_gate_perm = out_gate.permute(0, 2, 3, 1)
    out_scaled = out_norm * out_gate_perm
    del out, out_gate, out_norm, out_gate_perm

    # 6. final linear projection to dim
    w_out = w["to_out.weight"].to(torch.float16)
    final = F.linear(out_scaled, w_out)   # [B,N,N,dim] fp16
    final = final.to(torch.float32)
    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (`x`) → FP16.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (cuBLAS) on the FP16 tensor.
3️⃣  Split the fused projection, apply the optional pair mask and three sigmoid gates
    in a Triton kernel → tensors `left`, `right`, `out_gate` in layout
    [B, H, N, N] (FP16).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → FP16.
5️⃣  Row‑wise LayerNorm over the hidden dimension H (torch‑builtin,
    much faster than the hand‑written two‑pass kernel).
6️⃣  Multiply by the out‑gate, then a final linear projection back to `dim`
    (FP16 matmul, FP32 output).

Only the mask handling / gate application and the first LayerNorm are
implemented in Triton – the rest uses highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional, broadcasted over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index   r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # stride to jump to next hidden slice
    base = b * (H * hidden_stride) + i * N + k   # start of hidden slice 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: [left|right|lg|rg|og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] and use torch's fast layer_norm
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                               # cast to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)        # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)                # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (torch implementation)
    # ------------------------------------------------------------------
    # move hidden dim to last position to match torch.layer_norm signature
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                     # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by out‑gate and final projection
    # ------------------------------------------------------------------
    out_gate_perm = out_gate.permute(0, 2, 3, 1)            # [B, N, N, H]
    out_scaled = out_norm * out_gate_perm                 # FP16
    del out, out_norm, out_gate, out_gate_perm

    w_out = w["to_out.weight"].to(torch.float16).contiguous()   # [dim, H]
    final = F.linear(out_scaled, w_out)                     # [B, N, N, dim] FP16
    final = final.to(torch.float32)                         # cast to required output dtype
    return final
