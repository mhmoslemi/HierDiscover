"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened [R, dim] tensor using
   torch.nn.functional.layer_norm (CUDA kernel, FP32 → FP16).

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into one GEMM (`F.linear`).  The result
   (R,5·H) is split into the three tensors needed for the
   multiplicative update; optional pair‑mask is applied only to the
   left/right projections; sigmoid gates are computed and applied
   (still in FP16).

3️⃣ Batched GEMM (`torch.bmm`) computes the core einsum

        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]

   → `[B, H, N, N]` (FP16).

4️⃣ Row‑wise mean / rsqrt‑variance of `out` are obtained with a small
   Triton kernel (`rowwise_stats_kernel`).  This yields μ and rstd
   per (b,i,j) row (FP32).

5️⃣ Normalisation, the second LayerNorm, and the out‑gate are performed
   with plain torch kernels (very cheap element‑wise ops).

6️⃣ The final linear projection (to the original dimension) is a single
   large matrix‑multiplication

        (R, H) · (H, dim) → (R, dim)

   executed by cuBLAS (`torch.matmul`), which fully exploits the
   tensor‑core performance of the H100.  The result is cast back to
   FP32 and reshaped to `[B, N, N, dim]`.

Only the forward pass is implemented; all heavy work is delegated to
high‑performance CUDA kernels (cuBLAS) while the few reduction /
statistics steps are kept in Triton for low‑overhead launch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # H ≤ 128 → a single iteration (or at most two) is enough.
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    # torch.nn.functional.layer_norm is a fast CUDA kernel (cuDNN)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                      # FP32
    x_norm = x_norm.to(torch.float16)                      # FP16 for the rest

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    # Assemble fused weight: shape [5·H, dim] (FP16)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    # Slice the fused projection
    left      = proj[..., :H]                      # (R, H)
    right     = proj[..., H:2 * H]                 # (R, H)
    left_gate = proj[..., 2 * H:3 * H].sigmoid()   # (R, H)
    right_gate= proj[..., 3 * H:4 * H].sigmoid()   # (R, H)
    out_gate  = proj[..., 4 * H:].sigmoid()        # (R, H)

    if not nomask and mask is not None:
        # mask is broadcasted over the hidden dimension
        mask_flat = mask.reshape(R, 1).to(torch.float16)   # (R,1)
        left  = left * mask_flat
        right = right * mask_flat

    # Apply gates (still FP16)
    left  = left * left_gate
    right = right * right_gate

    # Reshape to 4‑D tensors expected by the batched GEMM
    left  = left.view(B, H, N, N)
    right = right.view(B, H, N, N)
    out_gate = out_gate.view(B, H, N, N)   # will be used later

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)          # (B·H, N, N) FP16
    right_mat = right.reshape(B * H, N, N)         # (B·H, N, N) FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                     # (B, H, N, N) FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → single iteration
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (cuBLAS)
    # --------------------------------------------------------------
    # Flatten tensors for the element‑wise part
    out_flat = out.reshape(R, H).float()                # FP32
    out_gate_flat = out_gate.reshape(R, H).float()      # FP32

    # LayerNorm (to_out_norm) and gating
    # γ, β are FP32 vectors of length H
    gamma = w["to_out_norm.weight"]                     # (H,)
    beta  = w["to_out_norm.bias"]                       # (H,)

    # (out - μ) * rstd
    normed = (out_flat - mu[:, None]) * rstd[:, None]   # (R, H) FP32
    # Apply the second LayerNorm parameters
    normed = normed * gamma[None, :] + beta[None, :]    # (R, H) FP32
    # Multiply by the out‑gate
    normed = normed * out_gate_flat                     # (R, H) FP32

    # Cast back to FP16 for the large matrix‑multiply
    normed_fp16 = normed.to(torch.float16)              # (R, H) FP16

    # Final linear projection: (R, H) @ (H, dim) → (R, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    final_fp16 = torch.matmul(normed_fp16, w_out_T)                    # (R, dim) FP16

    # Cast to FP32 and reshape to the expected output layout
    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # (B, N, N, dim)

    return final
```
