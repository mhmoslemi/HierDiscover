   term₁ = (O * G)      @ Wγ      →  (R, dim)   FP16
   term₂ =  G           @ Wβ      →  (R, dim)   FP16
   term₃ =  G           @ Wγ      →  (R, dim)   FP16

   Final output (FP32):
       out = term₁ * rstd   +   term₂   –   (mu * rstd) * term₃

   `R = B·N·N` is the number of pairwise tokens.  All three GEMMs
   are executed with cuBLAS‑Lt (FP16 → FP32 accumulation) which is
   far faster than a custom reduction kernel.

The whole pipeline uses only one custom Triton kernel (step 3) and
remains fully compatible with the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride for consecutive H‑planes
    base_offset = b * (H * hidden_stride) + i * N + k          # offset for h = 0

    # Split the fused projection: [L | R | Lg | Rg | Og]
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left   = left_raw   * m_val * left_gate
        right  = right_raw  * m_val * right_gate
        outg   = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the *outgoing* TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"          (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use the highly‑optimized
    #    torch implementation (internally a fused kernel).
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)                       # [R, C]   FP32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                               # [R, C]   FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, C]
    proj = F.linear(x_norm, fused_w)                  # (R, 5·H)  FP16
    del fused_w, x_norm                               # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum – batched GEMM (cuBLAS / cublasLt)
    #    out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                        # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                      # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N]  FP16
    del left_mat, right_mat                                 # free

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (FP32) – torch kernels are fast
    # ------------------------------------------------------------------
    # reshape to (B, H, N²) → permute to (B, N², H) → (R, H)
    out_flat = out.view(B, H, N * N).permute(0, 2, 1).contiguous()   # [B, N², H]
    out_2d = out_flat.view(R, H)                                      # [R, H]

    # μ and σ⁻¹ per row (float32)
    mu = out_2d.mean(dim=1, keepdim=True, dtype=torch.float32)                # [R,1]
    var = out_2d.var(dim=1, unbiased=False, keepdim=True, dtype=torch.float32)  # [R,1]
    rstd = torch.rsqrt(var + eps)                                            # [R,1]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate, and final projection.
    #    Rewrite:  out = ((out * rstd)·γ + β)·out_gate  @ W
    #    → three GEMMs (cuBLAS‑Lt) as described in the docstring.
    # ------------------------------------------------------------------
    # Prepare tensors
    out_gate_flat = out_gate.view(B, H, N * N).permute(0, 2, 1).contiguous()  # [B, N², H]
    out_gate_2d = out_gate_flat.view(R, H)                                     # [R, H]

    # Weight transformations (FP16)
    w_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    gamma = w["to_out_norm.weight"].to(torch.float16)              # (H,)
    beta  = w["to_out_norm.bias"].to(torch.float16)                # (H,)

    W_gamma = w_T * gamma[:, None]                                 # (H, dim)
    W_beta  = w_T * beta[:, None]                                  # (H, dim)

    # -------- GEMM 1 : (out * out_gate) @ W_gamma -----------------
    prod1 = torch.matmul((out_2d * out_gate_2d).to(torch.float16), W_gamma)   # (R, dim) FP16
    term1 = prod1.to(torch.float32) * rstd                                   # (R, dim)

    # -------- GEMM 2 : out_gate @ W_beta ---------------------------
    prod2 = torch.matmul(out_gate_2d, W_beta)                                 # (R, dim) FP16
    term2 = prod2.to(torch.float32)                                            # (R, dim)

    # -------- GEMM 3 : out_gate @ W_gamma --------------------------
    prod3 = torch.matmul(out_gate_2d, W_gamma)                                 # (R, dim) FP16
    term3 = (mu * rstd).to(torch.float32) * prod3.to(torch.float32)           # (R, dim)

    # ------- Assemble final output ---------------------------------
    final_2d = term1 + term2 - term3                                            # (R, dim)
    final = final_2d.view(B, N, N, dim)                                         # (B, N, N, dim)

    return final
