@triton.jit
def norm_proj_gate_combined_kernel(
    x_ptr,                 # [R, dim] FP32
    mask_ptr,              # [R] FP16 mask (if HAS_MASK)
    left_wg_ptr,           # [H, dim] FP16 weight_gamma (left_proj * gamma)
    left_wb_ptr,           # [H] FP32 weight_beta_sum
    left_wg_sum_ptr,       # [H] FP32 sum_weight_gamma
    right_wg_ptr, right_wb_ptr, right_wg_sum_ptr,
    left_gate_wg_ptr, left_gate_wb_ptr, left_gate_wg_sum_ptr,
    right_gate_wg_ptr, right_gate_wb_ptr, right_gate_wg_sum_ptr,
    out_gate_wg_ptr, out_gate_wb_ptr, out_gate_wg_sum_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    R, H, N, dim,
    eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode indices as before
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        m_val = m_val[:, None] # broadcast to hidden
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # initialize accumulators for sum_x and sum_sq
    sum_x = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # Hidden stride
    hidden_stride = N * N

    # Compute base offset for output writes
    base_offset = b * (H * hidden_stride) + i * N + k  # start offset for h=0

    # Loop over hidden tiles
    for h_off in range(0, H, BLOCK_H):
        # tile size might be smaller at end
        cur_H = tl.minimum(BLOCK_H, H - h_off)
        # Load weight sums for this hidden tile
        # left weight sum_gamma for current hidden indices
        left_wg_sum_tile = tl.load(left_wg_sum_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)  # [cur_H] FP32
        left_wb_sum_tile = tl.load(left_wb_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        # similarly load for other matrices
        right_wg_sum_tile = tl.load(right_wg_sum_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        right_wb_sum_tile = tl.load(right_wb_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        left_gate_wg_sum_tile = tl.load(left_gate_wg_sum_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        left_gate_wb_sum_tile = tl.load(left_gate_wb_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        right_gate_wg_sum_tile = tl.load(right_gate_wg_sum_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        right_gate_wb_sum_tile = tl.load(right_gate_wb_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        out_gate_wg_sum_tile = tl.load(out_gate_wg_sum_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)
        out_gate_wb_sum_tile = tl.load(out_gate_wb_ptr + h_off + tl.arange(0, BLOCK_H), mask=tl.arange(0, BLOCK_H) < cur_H, other=0.0)

        # Initialize accumulators for dot products
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # Loop over dim blocks
        for d_off in range(0, dim, BLOCK_D):
            cur_D = tl.minimum(BLOCK_D, dim - d_off)
            # Load x tile
            x_ptrs = x_ptr + (offs_r[:, None] * dim) + (d_off + tl.arange(0, BLOCK_D))[None, :]
            x_tile = tl.load(x_ptrs, mask=mask_r[:, None] & (tl.arange(0, BLOCK_D)[None, :] < cur_D), other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_D]

            # Update sums
            sum_x += tl.sum(x_tile, axis=1)
            sum_sq += tl.sum(x_tile * x_tile, axis=1)

            # Load weight gamma tiles for each matrix (shape [BLOCK_D, cur_H])
            # Helper lambda to load weight tile transposed
            # We'll define load_weight_tile(weight_ptr):
            #    offs_h = h_off + tl.arange(0, BLOCK_H)
            #    offs_d = d_off + tl.arange(0, BLOCK_D)
            #    ptrs = weight_ptr + offs_h[None, :] * dim + offs_d[:, None]
            #    return tl.load(ptrs, mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H), other=0.0)

            # left
            left_w_tile = tl.load(left_wg_ptr + (h_off + tl.arange(0, BLOCK_H))[None, :] * dim + (d_off + tl.arange(0, BLOCK_D))[:, None],
                                  mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H),
                                  other=0.0).to(tl.float32)  # [cur_D, cur_H]
            left_acc += tl.dot(x_tile, left_w_tile)  # result [BLOCK_R, cur_H]
            # similarly for other matrices
            right_w_tile = tl.load(right_wg_ptr + (h_off + tl.arange(0, BLOCK_H))[None, :] * dim + (d_off + tl.arange(0, BLOCK_D))[:, None],
                                   mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H),
                                   other=0.0).to(tl.float32)
            right_acc += tl.dot(x_tile, right_w_tile)

            left_gate_w_tile = tl.load(left_gate_wg_ptr + (h_off + tl.arange(0, BLOCK_H))[None, :] * dim + (d_off + tl.arange(0, BLOCK_D))[:, None],
                                       mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H),
                                       other=0.0).to(tl.float32)
            left_gate_acc += tl.dot(x_tile, left_gate_w_tile)

            right_gate_w_tile = tl.load(right_gate_wg_ptr + (h_off + tl.arange(0, BLOCK_H))[None, :] * dim + (d_off + tl.arange(0, BLOCK_D))[:, None],
                                         mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H),
                                         other=0.0).to(tl.float32)
            right_gate_acc += tl.dot(x_tile, right_gate_w_tile)

            out_gate_w_tile = tl.load(out_gate_wg_ptr + (h_off + tl.arange(0, BLOCK_H))[None, :] * dim + (d_off + tl.arange(0, BLOCK_D))[:, None],
                                       mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & (tl.arange(0, BLOCK_H)[None, :] < cur_H),
                                       other=0.0).to(tl.float32)
            out_gate_acc += tl.dot(x_tile, out_gate_w_tile)

        # End dim loop

        # compute mu and rstd
        mu = sum_x / dim
        var = sum_sq / dim - mu * mu
        rstd = tl.rsqrt(var + eps)  # [BLOCK_R]

        # broadcast rstd, mu to shape (BLOCK_R, cur_H)
        rstd_tile = rstd[:, None]
        mu_tile = mu[:, None]

        # Now compute raw left etc:
        left_raw = rstd_tile * (left_acc - mu_tile * left_wg_sum_tile[None, :]) + left_wb_sum_tile[None, :]
        right_raw = rstd_tile * (right_acc - mu_tile * right_wg_sum_tile[None, :]) + right_wb_sum_tile[None, :]
        left_gate_raw = rstd_tile * (left_gate_acc - mu_tile * left_gate_wg_sum_tile[None, :]) + left_gate_wb_sum_tile[None, :]
        right_gate_raw = rstd_tile * (right_gate_acc - mu_tile * right_gate_wg_sum_tile[None, :]) + right_gate_wb_sum_tile[None, :]
        out_gate_raw = rstd_tile * (out_gate_acc - mu_tile * out_gate_wg_sum_tile[None, :]) + out_gate_wb_sum_tile[None, :]

        # Sigmoid gates
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw))).to(tl.float16)

        # Apply mask and gates
        left = (left_raw.to(tl.float16) * m_val) * left_gate
        right = (right_raw.to(tl.float16) * m_val) * right_gate
        # out_gate no mask
        out_gate = out_gate

        # Store results
        dst_offset = base_offset[:, None] + (h_off + tl.arange(0, BLOCK_H))[None, :] * hidden_stride

        mask_h = (h_off + tl.arange(0, BLOCK_H)) < H
        mask_store = mask_r[:, None] & mask_h[None, :]

        tl.store(left_out_ptr + dst_offset, left, mask=mask_store)
        tl.store(right_out_ptr + dst_offset, right, mask=mask_store)
        tl.store(out_gate_out_ptr + dst_offset, out_gate, mask=mask_store)

```
@triton.jit
def layernorm_proj_gate_combined_kernel(
    x_ptr,        # [R, dim] FP32
    mask_ptr,     # [R] FP16 optional
    left_wg_ptr, left_wb_ptr,
    right_wg_ptr, right_wb_ptr,
    left_gate_wg_ptr, left_gate_wb_ptr,
    right_gate_wg_ptr, right_gate_wb_ptr,
    out_gate_wg_ptr, out_gate_wb_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    R, H, N, dim,
    eps,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H_MAX: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]   # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Pass 1: compute sum and sumsq
    sum_x = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for d_off in range(0, dim, BLOCK_D):
        cur_D = tl.minimum(BLOCK_D, dim - d_off)
        d_idx = d_off + tl.arange(0, BLOCK_D)
        # load x
        x = tl.load(x_ptr + offs_r[:, None] * dim + d_idx[None, :],
                    mask=mask_r[:, None] & (tl.arange(0, BLOCK_D)[None, :] < cur_D),
                    other=0.0).to(tl.float32)
        sum_x += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mu = sum_x / dim
    var = sum_sq / dim - mu * mu
    rstd = tl.rsqrt(var + eps)

    # Allocate accumulators for each projection:
    MAX_H = BLOCK_H_MAX  # compile-time constant, assumed >= H

    proj_left = tl.zeros([BLOCK_R, MAX_H], dtype=tl.float32)
    proj_right = tl.zeros([BLOCK_R, MAX_H], dtype=tl.float32)
    proj_left_gate = tl.zeros([BLOCK_R, MAX_H], dtype=tl.float32)
    proj_right_gate = tl.zeros([BLOCK_R, MAX_H], dtype=tl.float32)
    proj_out_gate = tl.zeros([BLOCK_R, MAX_H], dtype=tl.float32)

    # Pass 2: accumulate dot products
    for d_off in range(0, dim, BLOCK_D):
        cur_D = tl.minimum(BLOCK_D, dim - d_off)
        d_idx = d_off + tl.arange(0, BLOCK_D)

        # Load x and compute normalized tile
        x = tl.load(x_ptr + offs_r[:, None] * dim + d_idx[None, :],
                    mask=mask_r[:, None] & (tl.arange(0, BLOCK_D)[None, :] < cur_D),
                    other=0.0).to(tl.float32)
        x_centered = (x - mu[:, None]) * rstd[:, None]  # FP32

        # for each matrix: compute dot with weight_gamma tile
        # define helper for loading weight_gamma tile and accumulating
        # We'll loop over hidden tiles
        for h_off in range(0, H, BLOCK_H_MAX):
            cur_H = tl.minimum(BLOCK_H_MAX, H - h_off)
            # column indices for hidden
            h_idx = h_off + tl.arange(0, BLOCK_H_MAX)
            mask_h = h_idx < H

            # Load weight_gamma tiles for each matrix (size [cur_D, cur_H])
            # left
            left_w_tile = tl.load(left_wg_ptr + (h_off + tl.arange(0, BLOCK_H_MAX))[None, :] * dim + d_idx[:, None],
                                  mask=(tl.arange(0, BLOCK_D)[:, None] < cur_D) & mask_h[None, :],
                                  other=0.0).to(tl.float32)
            # result dot
            dot = tl.dot(x_centered, left_w_tile)  # shape [BLOCK_R, cur_H]
            # accumulate

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Key optimisation
-----------------
* Fuse the first LayerNorm and the five linear projections (left, right,
  left_gate, right_gate, out_gate) into a *single* Triton kernel.
  This removes the temporary normalised tensor (R×dim) and the huge
  intermediate projection (R×5·H) that would otherwise be written to and
  read from global memory.
* All three outputs (left, right, out_gate) are written directly after the
  fused operation, so the subsequent batched GEMM and the final
  LayerNorm‑+‑Projection kernel can reuse the original implementation.
* The kernel works for any batch size B≤2, sequence length N≤1024,
  feature dimension dim∈{128,384} and hidden dimension H≤128 (the
  implementation pads H to the compile‑time constant H_MAX=128).

The rest of the data‑flow (batched GEMM, row‑wise statistics,
second LayerNorm + final projection) stays unchanged.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Fused LayerNorm + 5×Linear (produces left, right, out_gate) ----------
# ----------------------------------------------------------------------
H_MAX = 128               # compile‑time maximum hidden dimension


@triton.jit
def fused_norm_proj_gate_kernel(
    # input tensor (R, dim) – float32
    x_ptr,                                 # [R, dim]  fp32
    # optional mask (R) – float16
    mask_ptr,                              # [R]       fp16
    # weight*γ (H_MAX, dim) – fp16
    left_wg_ptr,   right_wg_ptr,
    left_gate_wg_ptr, right_gate_wg_ptr,
    out_gate_wg_ptr,
    # ∑d (w·γ) per hidden – fp32
    left_wg_sum_ptr,   right_wg_sum_ptr,
    left_gate_wg_sum_ptr, right_gate_wg_sum_ptr,
    out_gate_wg_sum_ptr,
    # ∑d (w·β) per hidden – fp32
    left_wb_ptr,   right_wb_ptr,
    left_gate_wb_ptr, right_gate_wb_ptr,
    out_gate_wb_ptr,
    # output buffers (B, H, N, N) – fp16
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # problem size
    R, N, H, dim,
    eps,
    # compile‑time constants
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    # decode flat index: r = b·N·N + i·N + k
    # --------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # optional mask (scalar per (b,i,k))
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m = m[:, None]                                          # [BLOCK_R,1] broadcast over H
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Pass‑1: mean & variance of the raw input (x)
    # ------------------------------------------------------------------
    sum_x = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):               # static loop over dim‑tiles
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_R, BLOCK_D]

        sum_x += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mu = sum_x / dim
    var = sum_sq / dim - mu * mu
    rstd = tl.rsqrt(var + eps)                                 # [BLOCK_R]

    # ------------------------------------------------------------------
    # 2️⃣ Pass‑2: compute the five projections on the *normalised* input
    # ------------------------------------------------------------------
    # allocate per‑row accumulators for each projection
    left_acc      = tl.zeros([BLOCK_R, H_MAX], dtype=tl.float32)
    right_acc     = tl.zeros([BLOCK_R, H_MAX], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, H_MAX], dtype=tl.float32)
    right_gate_acc= tl.zeros([BLOCK_R, H_MAX], dtype=tl.float32)
    out_gate_acc  = tl.zeros([BLOCK_R, H_MAX], dtype=tl.float32)

    # pre‑compute a vector of hidden‑indices (used for all weight loads)
    h_idx = tl.arange(0, H_MAX)                               # [H_MAX]

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        # load raw input tile
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_R, BLOCK_D]

        # centred & normalised tile: (x‑μ)*rstd
        x_norm = (x - mu[:, None]) * rstd[:, None]            # [BLOCK_R, BLOCK_D]

        # --------------------------------------------------------------
        # helper for loading a weight‑γ tile (shape [BLOCK_D, H_MAX])
        # --------------------------------------------------------------
        def load_wg(ptr):
            # ptr points to a [H_MAX, dim] fp16 matrix (γ‑scaled)
            # we need the transposed view [BLOCK_D, H_MAX]
            wg_tile = tl.load(
                ptr
                + h_idx[None, :] * dim               # stride over hidden
                + offs_d[:, None],                   # stride over dim
                mask=mask_d[:, None] & (h_idx[None, :] < H),
                other=0.0,
            ).to(tl.float32)                         # [BLOCK_D, H_MAX]
            return wg_tile

        # left projection
        left_wg_tile = load_wg(left_wg_ptr)
        left_acc += tl.dot(x_norm, left_wg_tile)               # [BLOCK_R, H_MAX]

        # right projection
        right_wg_tile = load_wg(right_wg_ptr)
        right_acc += tl.dot(x_norm, right_wg_tile)

        # left gate
        left_gate_wg_tile = load_wg(left_gate_wg_ptr)
        left_gate_acc += tl.dot(x_norm, left_gate_wg_tile)

        # right gate
        right_gate_wg_tile = load_wg(right_gate_wg_ptr)
        right_gate_acc += tl.dot(x_norm, right_gate_wg_tile)

        # out gate
        out_gate_wg_tile = load_wg(out_gate_wg_ptr)
        out_gate_acc += tl.dot(x_norm, out_gate_wg_tile)

    # ------------------------------------------------------------------
    # 3️⃣ Add bias terms (∑d w·β) – these are pre‑computed per hidden unit
    # ------------------------------------------------------------------
    left_bias      = tl.load(left_wb_ptr      + h_idx, mask=h_idx < H, other=0.0)
    right_bias     = tl.load(right_wb_ptr     + h_idx, mask=h_idx < H, other=0.0)
    left_gate_bias = tl.load(left_gate_wb_ptr + h_idx, mask=h_idx < H, other=0.0)
    right_gate_bias= tl.load(right_gate_wb_ptr+ h_idx, mask=h_idx < H, other=0.0)
    out_gate_bias  = tl.load(out_gate_wb_ptr  + h_idx, mask=h_idx < H, other=0.0)

    left_acc      += left_bias[None, :]
    right_acc     += right_bias[None, :]
    left_gate_acc += left_gate_bias[None, :]
    right_gate_acc+= right_gate_bias[None, :]
    out_gate_acc  += out_gate_bias[None, :]

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (FP32 → FP16) and the pair‑wise mask
    # ------------------------------------------------------------------
    sigmoid = lambda x: (1.0 / (1.0 + tl.exp(-x)))

    left_gate_fp = sigmoid(left_gate_acc).to(tl.float16)          # [BLOCK_R, H_MAX]
    right_gate_fp= sigmoid(right_gate_acc).to(tl.float16)
    out_gate_fp  = sigmoid(out_gate_acc).to(tl.float16)

    left_fp  = left_acc.to(tl.float16)   * m * left_gate_fp
    right_fp = right_acc.to(tl.float16)  * m * right_gate_fp
    out_gate_fp = out_gate_fp                                            # no mask

    # ------------------------------------------------------------------
    # 5️⃣ Store results in layout [B, H, N, N] (row‑major)
    # ------------------------------------------------------------------
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k               # start offset for h=0

    # column index for hidden dimension
    h_offset = h_idx[None, :] * hidden_stride                         # [1, H_MAX]

    dst = base_offset[:, None] + h_offset                             # [BLOCK_R, H_MAX]

    # masking for rows and valid hidden channels
    mask_h = h_idx < H
    mask_store = mask_r[:, None] & mask_h[None, :]

    tl.store(left_out_ptr      + dst, left_fp,      mask=mask_store)
    tl.store(right_out_ptr     + dst, right_fp,     mask=mask_store)
    tl.store(out_gate_out_ptr  + dst, out_gate_fp,  mask=mask_store)


# ----------------------------------------------------------------------
# 2️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def final_output_kernel(

    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows over R
    pid_n = tl.program_id(1)   # columns over dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0

    # Load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # Load LN parameters for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                      # [M, K]

        term_fp16 = term.to(tl.float16)   # back to FP16 for the dot

        # Load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # [K, TILE_N]

        # Accumulate matmul (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final [B, N, N, dim] tensor (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # Pre‑compute γ‑scaled weights and β‑bias sums (padding to H_MAX=128)
    # ------------------------------------------------------------------
    def prep_weight(weight):
        # weight : [H, dim] FP32
        wg = (weight * w["norm.weight"]).to(torch.float16)               # γ·weight
        wb = (weight * w["norm.bias"]).sum(dim=1).to(torch.float32)      # Σd weight·β
        if H < H_MAX:
            pad = (0, 0, 0, H_MAX - H)   # pad hidden dimension (rows)
            wg = F.pad(wg, pad, "constant", 0.0)
            wb = F.pad(wb, (0, H_MAX - H), "constant", 0.0)
        return wg, wb

    left_wg, left_wb           = prep_weight(w["left_proj.weight"])
    right_wg, right_wb         = prep_weight(w["right_proj.weight"])
    left_gate_wg, left_gate_wb = prep_weight(w["left_gate.weight"])
    right_gate_wg, right_gate_wb = prep_weight(w["right_gate.weight"])
    out_gate_wg, out_gate_wb   = prep_weight(w["out_gate.weight"])

    # pre‑compute sums Σd (γ·weight) per hidden unit (used in the kernel)
    left_wg_sum       = left_wg.sum(dim=1).to(torch.float32)
    right_wg_sum      = right_wg.sum(dim=1).to(torch.float32)
    left_gate_wg_sum  = left_gate_wg.sum(dim=1).to(torch.float32)
    right_gate_wg_sum = right_gate_wg.sum(dim=1).to(torch.float32)
    out_gate_wg_sum   = out_gate_wg.sum(dim=1).to(torch.float32)

    # ------------------------------------------------------------------
    # Flatten input (R, dim) – keep FP32
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32

    # ------------------------------------------------------------------
    # 1️⃣ Fused LayerNorm + 5×Linear → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 8               # rows per program (tuned for H100)
    BLOCK_D = 64              # dim‑tile size
    grid = (triton.cdiv(R, BLOCK_R),)

    fused_norm_proj_gate_kernel[grid](
        x_flat,
        mask_ptr,
        left_wg, right_wg,
        left_gate_wg, right_gate_wg,
        out_gate_wg,
        left_wg_sum, right_wg_sum,
        left_gate_wg_sum, right_gate_wg_sum,
        out_gate_wg_sum,
        left_wb, right_wb,
        left_gate_wb, right_gate_wb,
        out_gate_wb,
        left,
        right,
        out_gate,
        R, N, H, dim,
        eps,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128               # covers H ≤ 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
