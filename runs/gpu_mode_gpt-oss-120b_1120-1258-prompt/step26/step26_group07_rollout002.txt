"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened (B·N·N, dim) input.
    → fp32 → fp16 (to keep memory low).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a single
    half‑precision GEMM (`torch.nn.functional.linear`).  The result has shape (R, 5·H).

3️⃣  Split the fused projection, optionally apply the pairwise mask, and compute the three
    sigmoids – all in a single Triton kernel that writes the three tensors
    `left`, `right` and `out_gate` in layout [B, H, N, N] (fp16).

4️⃣  Batched half‑precision GEMM (`torch.bmm`) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]    →   (B·H, N, N) fp16.

5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
    (a second LayerNorm) in a Triton kernel.

6️⃣  Fuse the second LayerNorm, the per‑plane gate (`out_gate`) and the final
    linear projection (`to_out`) into a Triton kernel that directly produces the
    final output tensor of shape [B, N, N, dim] (fp32).

All kernels are written for Triton 3.3.1, target NVIDIA H100 and use mixed
precision (fp16 compute, fp32 accumulation).  The code works for any
combination of batch‑size (1‑2), sequence length (128‑1024) and hidden
dimension (≤ 384) used in the test‑suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,         # [R, D]  fp32 input
    y_ptr,         # [R, D]  fp16 output
    w_ptr,         # [D]    LN weight (γ) fp32
    b_ptr,         # [D]    LN bias   (β) fp32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: compute mean & var -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] fp16 fused projection
    left_ptr,        # [B, H, N, N] fp16
    right_ptr,       # [B, H, N, N] fp16
    out_gate_ptr,    # [B, H, N, N] fp16
    mask_ptr,        # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pairwise mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]      # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride inside the H‑plane (N×N) for a given batch & i,k coordinate
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view into the fused projection:
        #   layout = [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # [B, H, N, N] fp16
    mu_ptr,     # [R] fp32
    rstd_ptr,   # [R] fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse 2nd LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,       # [B, H, N, N] fp16
    gate_ptr,      # [B, H, N, N] fp16 (out_gate)
    mu_ptr,        # [R] fp32
    rstd_ptr,      # [R] fp32
    ln_w_ptr,      # [H] fp32   (to_out_norm.weight)
    ln_b_ptr,      # [H] fp32   (to_out_norm.bias)
    w_out_T_ptr,   # [H, dim] fp16 (transposed to_out weight)
    out_final_ptr, # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):

    pid_n = tl.program_id(1)          # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    # accumulator for the final (M,N) tile (float32)
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden‑plane values (fp16 → fp32)
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)           # [M, K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)           # [M, K]

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  γ· + β   →   * out_gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [M, K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [M, K]
        gated  = norm * vals_gate                                   # [M, K]

        # cast to fp16 for the tensor‑core dot
        gated_fp16 = gated.to(tl.float16)

        # load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # write out final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point ---------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # flattened tokens (b,i,k)
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M1 = 256
    BLOCK_D1 = 128
    grid_ln = (triton.cdiv(R, BLOCK_M1),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M1,
        BLOCK_D=BLOCK_D1,
        num_warps=8,
    )
    del x_flat

    # -----------------------------------------------------------------
    # 2️⃣ Fuse 5 linear maps into a single GEMM (FP16)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) fp16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor (never accessed when HAS_MASK == 0)
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 128
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) → contraction: left @ rightᵀ
    # -----------------------------------------------------------------
    # Reshape to (B·H, N, N) and use torch.bmm
    left_mat  = left.view(B * H, N, N)                       # fp16
    right_mat = right.view(B * H, N, N)                      # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (LayerNorm over H)
    # -----------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M2 = 256
    BLOCK_K2 = 128                     # H ≤ 128 fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M2),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_K=BLOCK_K2,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gate + projection (fused)
    # -----------------------------------------------------------------
    # transposed final weight: [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32          # chunk size over hidden dim
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma fp32
        w["to_out_norm.bias"],     # beta  fp32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
