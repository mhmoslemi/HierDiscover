"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation.

Algorithm
---------
1️⃣  Input LayerNorm  (torch.nn.functional.layer_norm) on the flattened
    [B·N·N, C] tensor, result cast to FP16.

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM (`F.linear`).  The weight
    matrix has shape (5·H, C) and the output has shape (R, 5·H) where
    R = B·N·N and H = hidden_dim.

3️⃣  `split_fuse_gate_kernel` extracts the five blocks, applies the
    optional pair‑mask, evaluates the three sigmoid gates and writes the
    three tensors `left`, `right`, `out_gate` in layout [B, H, N, N]
    (FP16).

4️⃣  Core pair‑wise operation is a batched GEMM:
         out[b, h] = left[b, h] @ right[b, h].T
    implemented with `torch.bmm`.  Result shape: [B, H, N, N] (FP16).

5️⃣  Second LayerNorm (over the hidden dimension H) is performed with
    torch.nn.functional.layer_norm on a tensor reshaped to
    [B, N, N, H] (the layout required by LayerNorm).  Weight and bias
    are taken from `to_out_norm`.

6️⃣  Element‑wise multiplication with `out_gate` (already in the same
    layout) and a final linear projection (`to_out.weight`) are done
    with `torch.nn.functional.linear`.  The final tensor is reshaped
    back to [B, N, N, C] and returned as FP32.

Only the split‑gate operation is written in Triton; the rest relies on
highly‑optimized cuBLAS/torch kernels, dramatically reducing kernel
launch overhead while preserving full numerical correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]    FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride for a single H slice
    base = b * (H * hidden_stride) + i * N + k   # start address for this (b,i,k)

    # Iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout: each row has (5·H) columns
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])
