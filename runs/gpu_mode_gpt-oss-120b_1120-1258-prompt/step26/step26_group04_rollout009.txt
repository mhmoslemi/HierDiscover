    total_rows, H,

    eps: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # rows
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

"""
Outgoing TriMul (AlphaFold‑3) – highly‑fused forward pass.

Algorithm (mixed‑precision, fp16 for all internal tensors, fp32 for statistics):
1️⃣  Row‑wise LayerNorm on the flattened pair representation (fp32 → fp16).
2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
    into a single GEMM → (R, 5·H) where R = B·N·N.
3️⃣  Split the fused projection, apply the optional mask,
    evaluate sigmoid‑gates and write the three tensors `left`, `right`, `out_gate`
    in layout [B, H, N, N] (fp16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (fp16).
5️⃣  One Triton kernel computes for every row r = (b,i,j) the mean & rsqrt‑variance
    of `out`, applies the second LayerNorm (γ,β), multiplies by `out_gate`,
    and writes the result into a flat tensor `[R, H]` (fp16).  This replaces the
    previous two‑kernel implementation (row‑wise stats + final output) and halves
    the memory traffic on `out`.
6️⃣  The final linear projection `to_out` (H → dim) is performed with a
    highly‑optimized cuBLAS GEMM (`torch.nn.functional.linear`).  The result is
    cast to fp32 and reshaped to `[B, N, N, dim]`.

All kernels target Triton 3.3.1 on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim] fp32 input
    y_ptr,       # [R, dim] fp16 output
    w_ptr,       # [dim]   γ (fp32)
    b_ptr,       # [dim]   β (fp32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] fp16 fused projection
    left_ptr,       # [B, H, N, N] fp16
    right_ptr,      # [B, H, N, N] fp16
    out_gate_ptr,   # [B, H, N, N] fp16
    mask_ptr,       # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 for stability, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Combined LayerNorm (γ,β) + out_gate multiplication
#      → produces flat tensor [R, H] ready for the final linear projection.
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_kernel(
    out_ptr,        # [B, H, N, N] fp16   (result of the einsum)
    gate_ptr,       # [B, H, N, N] fp16   (out_gate)
    outgated_ptr,   # [R, H] fp16 (output)
    ln_w_ptr,       # [H] fp32 (to_out_norm.weight)
    ln_b_ptr,       # [H] fp32 (to_out_norm.bias)
    rows, H, N,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < rows

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: apply LN, gate and write ----------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw out & gate values
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # LayerNorm: (x‑μ) * rstd
        norm = (vals_out - mean[:, None]) * rstd[:, None]          # [M, K]
        norm = norm * gamma[None, :] + beta[None, :]               # γ· + β

        gated = norm * vals_gate                                   # apply out_gate

        # store to flat layout [R, H] (row‑major)
        dst_ptr = outgated_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (AlphaFold‑3) operator – no gradients.

    Args:
        data (tuple): (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N] (bool/float) or None
            weights: dict with keys
                     "norm.weight", "norm.bias",
                     "left_proj.weight", "right_proj.weight",
                     "left_gate.weight", "right_gate.weight", "out_gate.weight",
                     "to_out_norm.weight", "to_out_norm.bias",
                     "to_out.weight"
                     (all float32)
            config : dict with entries "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise “tokens”
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left × rightᵀ) → core TriMul contraction
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for cuBLAS batched GEMM
    left_mat  = left.view(B * H, N, N)                      # fp16
    right_mat = right.view(B * H, N, N)                     # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LN (γ,β) + out_gate multiplication  → [R, H] fp16
    # ------------------------------------------------------------------
    out_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128          # rows per program
    BLOCK_K_OUT = 32           # hidden‑dim chunk (fits comfortably in registers)

    grid_ln_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    ln_gate_kernel[grid_ln_gate](
        out,
        out_gate,
        out_gated,
        w["to_out_norm.weight"],   # γ (fp32)
        w["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (H → dim) via cuBLAS
    # ------------------------------------------------------------------
    # weight shape [dim, H]; torch.nn.functional.linear does X @ Wᵀ
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = F.linear(out_gated, w_out_fp16)       # [R, dim] fp16
    final = final_fp16.view(B, N, N, dim).float()      # fp32 output

    return final
