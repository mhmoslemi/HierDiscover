def rowwise_layernorm_and_gate(out_ptr, gate_ptr, gamma_ptr, beta_ptr, eps, R, H, out_norm_gate_ptr, BLOCK_R, BLOCK_H):
    pid = tl.program_id(0)
    start_r = pid * BLOCK_R
    offs_r = start_r + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute mean & var per row for each group of H (maybe block across H)
    # We'll iterate across hidden dim in steps BLOCK_H

    # For each row, accumulate sum and sum_sq across hidden.
    sum = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # out values
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass for normalized and gating:
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

        norm = (out_vals - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]
        norm_gate = norm * gate_vals

        tl.store(out_norm_gate_ptr + offs_r[:, None] * H + offs_h[None, :], norm_gate.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

def ln_gate_fused_kernel(
    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    eps,
    rows, H, N,
    out_lnorm_gate_ptr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # compute indices: b,i,j
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    # i = rem // N   (but used only for base offset)
    # Move to compute base offset
    base = b * H * nn + rem  # offsets to out_ptr for h=0 plane

    # hidden stride = nn

    # First pass: compute mean & var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load
        ptr = out_ptr + base[:, None] + offs_h[None, :] * nn
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute normalized*gate*ln_w+ln_b
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

                            mask=mask_r[:, None] & mask_h[None, :],

        norm = (out_vals - mean[:, None]) * rstd[:, None]

        gated = norm * gate_vals

        tl.store(out_lnorm_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single FP16 GEMM (cuBLAS) producing a tensor of shape (R, 5·H), where
   R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply an optional mask, compute sigmoid gates and
   store ``left``, ``right`` and ``out_gate`` (layout [B, H, N, N], FP16) via
   a Triton kernel.
4️⃣ Batched GEMM (cuBLAS) performs the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   (shape [B, H, N, N], FP16).
5️⃣ One Triton kernel computes the second LayerNorm (mean & rsqrt‑variance)
   together with the ``out_gate`` multiplication and writes a contiguous
   buffer ``out_lnorm_gate`` of shape (R, H) (FP16).
6️⃣ A single cuBLAS mat‑mul projects the normalized values to the output
   dimension:  out_lnorm_gate @ to_out.T  →  (R, dim) (FP16).  The result is
   reshaped to (B, N, N, dim) and cast to FP32.

All kernels target Triton 3.3.1 (H100).  Mixed‑precision is used throughout;
the final tensor is float‑32 as required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] FP32 input
    y_ptr,            # [R, dim] FP16 output
    w_ptr,            # [dim]    LN weight (γ) FP32
    b_ptr,            # [dim]    LN bias    (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # offset of the first hidden slice (h = 0) for this (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and produce a contiguous (R,H) buffer
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_fused_kernel(
    out_ptr,          # [B, H, N, N] FP16
    gate_ptr,         # [B, H, N, N] FP16
    ln_w_ptr,         # [H] FP32 (γ)
    ln_b_ptr,         # [H] FP32 (β)
    eps,
    rows, H, N,
    out_lnorm_gate_ptr,   # [R, H] FP16 output
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    nn = N * N                      # N²
    hidden_stride = nn

    # Compute batch index b and the intra‑batch offset (i·N + j)
    b = offs_r // nn
    rem = offs_r - b * nn           # rem = i·N + j
    # Base pointer for h = 0 (same for out and gate)
    base = b * (H * nn) + rem       # offset in the flat buffer of out/gate

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a block of hidden values (stride = nn)
        ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize, apply LN weight/bias, multiply out_gate ----------
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

        # LayerNorm over hidden dim
        norm = (out_vals - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]

        # Multiply by out_gate
        gated = norm * gate_vals

        # Store to contiguous (R, H) buffer
        tl.store(out_lnorm_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient) using Triton + cuBLAS.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                       # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16
    # left/right are no longer needed
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm + out_gate → contiguous (R, H) buffer
    # --------------------------------------------------------------
    out_lnorm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_LN = 128
    BLOCK_H_LN = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_R_LN),)
    ln_gate_fused_kernel[grid_ln_gate](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        eps,
        R,
        H,
        N,
        out_lnorm_gate,
        BLOCK_R=BLOCK_R_LN,
        BLOCK_H=BLOCK_H_LN,
        num_warps=8,
    )
    # out and out_gate are no longer needed
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection (cuBLAS) → (B, N, N, dim) float32
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    final_fp16 = torch.matmul(out_lnorm_gate, w_out_T)               # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)          # cast to FP32

    return final
