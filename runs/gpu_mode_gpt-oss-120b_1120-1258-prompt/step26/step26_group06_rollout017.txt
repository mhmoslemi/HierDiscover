```python
@triton.jit
def split_fuse_gate_scaled_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mu_ptr, rstd_ptr,
    wsum_ptr, bias_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode batch, i, k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # load mu, rstd, compute mu_rstd
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)  # [BLOCK_R] fp32
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)  # [BLOCK_R] fp32
    mu_rstd = mu * rstd  # fp32

    # compute offsets for base indices in projection and destinations
    hidden_stride = N * N  # stride between hidden planes in B,H,N,N layout
    base_offset = b * (H * hidden_stride) + i * N + k  # start for hidden=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute base pointer offsets for projection (R rows)
        # layout in proj: left|right|left_gate|right_gate|out_gate each size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        # load raw projection values
        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        # load wsum and bias for each group
        # offsets inside wsum_ptr: for left group start 0, right start H, left_gate start 2H, right_gate start 3H, out_gate start 4H
        wsum_base = offs_h[None, :] + tl.arange(0, 0)  # not needed; we compute separate

        # direct loads:
        wsum_left = tl.load(wsum_ptr + offs_h,
                            mask=mask_h,
                            other=0.0)
        wsum_right = tl.load(wsum_ptr + H + offs_h,
                             mask=mask_h,
                             other=0.0)
        wsum_left_gate = tl.load(wsum_ptr + 2 * H + offs_h,
                                 mask=mask_h,
                                 other=0.0)
        wsum_right_gate = tl.load(wsum_ptr + 3 * H + offs_h,
                                  mask=mask_h,
                                  other=0.0)
        wsum_out_gate = tl.load(wsum_ptr + 4 * H + offs_h,
                                mask=mask_h,
                                other=0.0)

        bias_left = tl.load(bias_ptr + offs_h,
                            mask=mask_h,
                            other=0.0)
        bias_right = tl.load(bias_ptr + H + offs_h,
                             mask=mask_h,
                             other=0.0)
        bias_left_gate = tl.load(bias_ptr + 2 * H + offs_h,
                                 mask=mask_h,
                                 other=0.0)
        bias_right_gate = tl.load(bias_ptr + 3 * H + offs_h,
                                  mask=mask_h,
                                  other=0.0)
        bias_out_gate = tl.load(bias_ptr + 4 * H + offs_h,
                                mask=mask_h,
                                other=0.0)

        # compute corrected values (using fp32 for mu_rstd * wsum)
        # Cast raw proj values to fp32 for compute to maintain precision
        left_raw_f = left_raw.to(tl.float32)
        right_raw_f = right_raw.to(tl.float32)
        left_gate_raw_f = left_gate_raw.to(tl.float32)
        right_gate_raw_f = right_gate_raw.to(tl.float32)
        out_gate_raw_f = out_gate_raw.to(tl.float32)

        # apply per-row scaling
        # Note rstd and mu_rstd are fp32 scalars per row; broadcast across hidden dim
        # Using cu expressions: (raw * rstd) - (mu_rstd * wsum) + bias
        left_corr = left_raw_f * rstd[:, None] - mu_rstd[:, None] * wsum_left[None, :] + bias_left[None, :].to(tl.float32)
        right_corr = right_raw_f * rstd[:, None] - mu_rstd[:, None] * wsum_right[None, :] + bias_right[None, :].to(tl.float32)
        left_gate_corr = left_gate_raw_f * rstd[:, None] - mu_rstd[:, None] * wsum_left_gate[None, :] + bias_left_gate[None, :].to(tl.float32)
        right_gate_corr = right_gate_raw_f * rstd[:, None] - mu_rstd[:, None] * wsum_right_gate[None, :] + bias_right_gate[None, :].to(tl.float32)
        out_gate_corr = out_gate_raw_f * rstd[:, None] - mu_rstd[:, None] * wsum_out_gate[None, :] + bias_out_gate[None, :].to(tl.float32)

        # apply sigmoid to gates (cast to fp32 for stability)
        left_gate_sig = (1.0 / (1.0 + tl.exp(-left_gate_corr))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate_corr))).to(tl.float16)
        out_gate_sig = (1.0 / (1.0 + tl.exp(-out_gate_corr))).to(tl.float16)

        # apply mask and gating to left/right
        left = left_corr.to(tl.float16) * mask_val * left_gate_sig
        right = right_corr.to(tl.float16) * mask_val * right_gate_sig
        outg = out_gate_sig

        # store results
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])
```

```python
@triton.jit
def norm_stats_kernel(
    x_ptr,   # [R, dim] FP32
    mu_ptr,  # [R] FP32
    rstd_ptr, # [R] FP32
    rows, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.
----------------------------------------------------------------
Algorithm (re‑ordered for memory‑efficiency)

1️⃣  Compute per‑row LayerNorm statistics (mean & rsqrt‑variance) on the
    input tensor X ∈ ℝ^{B·N·N·C}.  Only scalars μ,ρ are written, the
    normalized tensor is NOT materialised (≈ 8 MiB vs ~768 MiB).

2️⃣  Fuse the LayerNorm scale γ and bias β into the five linear
    projections (left, right, left_gate, right_gate, out_gate):
        Ŵ = W ○ γ                     (element‑wise)
        s = Σ_d Ŵ_{h,d}               (row‑wise sum, FP32)
        b = W_{h,:}·β                  (bias, FP32)
    The fused weight matrix Ŵ ∈ ℝ^{5·H×C} (FP16) is used in a single
    GEMM:
        P = X_fp16 @ Ŵᵀ                # shape [R,5·H] (FP16)

3️⃣  Split the projection, apply the per‑row LN scaling,
    the optional pairwise mask, and the sigmoid gates.
    This is a single Triton kernel that reads:
        * P (raw projection)
        * μ,ρ (row statistics)
        * s and b for each of the five groups
        * mask (optional)
    and writes the gated tensors
        left , right ∈ ℝ^{B·H·N·N} (FP16)
        out_gate          ∈ ℝ^{B·H·N·N} (FP16)

4️⃣  Batched GEMM over the hidden dimension:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    Implemented with torch.bmm (cuBLAS).

5️⃣  Row‑wise mean / rsqrt‑variance of `out` (second LayerNorm).
    The kernel produces μ_out , ρ_out per (b,i,j).

6️⃣  Fuse the second LayerNorm, the out‑gate multiplication,
    and the final projection (H → C) into one Triton kernel that
    writes the final [B,N,N,C] tensor (FP32).

All kernels target Triton 3.3.1 on an NVIDIA H100.  Mixed‑precision
(FP16 bulk work, FP32 reductions) guarantees numerical fidelity while
greatly reducing memory traffic compared with the naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise mean / rsqrt‑variance (no write‑back of normalized data)
# --------------------------------------------------------------
@triton.jit
def norm_stats_kernel(
    x_ptr,          # [R, C] FP32 input (flattened)
    mu_ptr,         # [R]   FP32 output
    rstd_ptr,       # [R]   FP32 output
    rows, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# 2️⃣ Split / mask / sigmoid / per‑row LN scaling (fused)
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_scaled_kernel(

    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mu_ptr,            # [R] FP32   (μ from step 1)
    rstd_ptr,          # [R] FP32   (ρ from step 1)
    wsum_ptr,          # [5·H] FP32 (row‑wise sums of scaled weights)
    bias_ptr,          # [5·H] FP32 (β·W for each projection)
    mask_ptr,          # [R] FP16 (optional mask, 1⇒keep)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Load per‑row LN statistics

    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R] fp32
    mu_rstd = mu * rstd                                          # [BLOCK_R] fp32

    hidden_stride = N * N               # stride between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k      # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):

                                 mask=mask_r[:, None] & mask_h[None, :],

                                 other=0.0)

                                 other=0.0)

                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Offsets inside the concatenated buffers:
        #   [ left | right | left_gate | right_gate | out_gate ]
        wsum_left      = tl.load(wsum_ptr + offs_h,
                                 mask=mask_h, other=0.0)
        wsum_right     = tl.load(wsum_ptr + H + offs_h,
                                 mask=mask_h, other=0.0)
        wsum_lgate     = tl.load(wsum_ptr + 2 * H + offs_h,
                                 mask=mask_h, other=0.0)
        wsum_rgate     = tl.load(wsum_ptr + 3 * H + offs_h,
                                 mask=mask_h, other=0.0)
        wsum_outgate   = tl.load(wsum_ptr + 4 * H + offs_h,
                                 mask=mask_h, other=0.0)

        bias_left      = tl.load(bias_ptr + offs_h,
                                 mask=mask_h, other=0.0)
        bias_right     = tl.load(bias_ptr + H + offs_h,
                                 mask=mask_h, other=0.0)
        bias_lgate     = tl.load(bias_ptr + 2 * H + offs_h,
                                 mask=mask_h, other=0.0)
        bias_rgate     = tl.load(bias_ptr + 3 * H + offs_h,
                                 mask=mask_h, other=0.0)
        bias_outgate   = tl.load(bias_ptr + 4 * H + offs_h,
                                 mask=mask_h, other=0.0)

        # ---------- apply per‑row LN scaling ----------
        # raw_* are FP16 -> cast to FP32 for the arithmetic
        l = left_raw.to(tl.float32)
        r = right_raw.to(tl.float32)
        lg = left_gate_raw.to(tl.float32)
        rg = right_gate_raw.to(tl.float32)
        og = out_gate_raw.to(tl.float32)

        # proj_corrected = raw * rstd - mu_rstd * wsum + bias
        l_corr = l * rstd[:, None] - mu_rstd[:, None] * wsum_left[None, :] + bias_left[None, :]
        r_corr = r * rstd[:, None] - mu_rstd[:, None] * wsum_right[None, :] + bias_right[None, :]
        lg_corr = lg * rstd[:, None] - mu_rstd[:, None] * wsum_lgate[None, :] + bias_lgate[None, :]
        rg_corr = rg * rstd[:, None] - mu_rstd[:, None] * wsum_rgate[None, :] + bias_rgate[None, :]
        og_corr = og * rstd[:, None] - mu_rstd[:, None] * wsum_outgate[None, :] + bias_outgate[None, :]

        # ---------- sigmoid gates ----------
        left_gate  = (1.0 / (1.0 + tl.exp(-lg_corr))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_corr))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-og_corr))).to(tl.float16)

        # ---------- mask & final write ----------
        left  = l_corr.to(tl.float16) * mask_val * left_gate
        right = r_corr.to(tl.float16) * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32   (to_out_norm.mean)
    rstd_ptr,         # [R]            FP32   (to_out_norm.rstd)
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows over R
    pid_n = tl.program_id(1)   # columns over dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0

    # Load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M] fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M] fp32

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # Load LN parameters for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                      # [M, K]

        term_fp16 = term.to(tl.float16)   # back to FP16 for the dot

        # Load tile of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # [K, TILE_N]

        # Accumulate matmul (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final [B, N, N, dim] tensor (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator (no backward)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict  : containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row LayerNorm statistics (μ, ρ) on the raw input
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    mu_in = torch.empty(R, dtype=torch.float32, device=device)
    rstd_in = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_D_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    norm_stats_kernel[grid_stats](
        x_flat,
        mu_in,
        rstd_in,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_D=BLOCK_D_STATS,
        num_warps=8,
    )

    # Cast the input to FP16 for the bulk GEMM (same as the reference)
    x_fp16 = x_flat.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the first LayerNorm into the five linear projections
    # --------------------------------------------------------------
    norm_weight = w["norm.weight"]               # [dim] FP32
    norm_bias   = w["norm.bias"]                 # [dim] FP32

    def preprocess(proj_weight):
        # Ŵ = W ∘ γ
        w_scaled = (proj_weight * norm_weight).to(torch.float16)          # [H, dim] FP16
        w_sum    = (proj_weight * norm_weight).sum(dim=1).to(torch.float32)  # [H]  FP32
        bias     = (proj_weight @ norm_bias).to(torch.float32)               # [H]  FP32
        return w_scaled, w_sum, bias

    left_w,    left_sum,    left_bias    = preprocess(w["left_proj.weight"])
    right_w,   right_sum,   right_bias   = preprocess(w["right_proj.weight"])
    lgate_w,   lgate_sum,   lgate_bias   = preprocess(w["left_gate.weight"])
    rgate_w,   rgate_sum,   rgate_bias   = preprocess(w["right_gate.weight"])
    outgate_w, outgate_sum, outgate_bias = preprocess(w["out_gate.weight"])

    # Concatenate the five groups (order matches the split kernel)
    fused_weight = torch.cat([left_w, right_w, lgate_w, rgate_w, outgate_w], dim=0)  # [5·H, dim] FP16
    fused_wsum   = torch.cat([left_sum, right_sum, lgate_sum, rgate_sum, outgate_sum], dim=0)  # [5·H] FP32
    fused_bias   = torch.cat([left_bias, right_bias, lgate_bias, rgate_bias, outgate_bias], dim=0)  # [5·H] FP32

    # Linear projection (no bias – bias already incorporated via fused_bias)
    proj_raw = F.linear(x_fp16, fused_weight)    # [R, 5·H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates and per‑row LN scaling
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_scaled_kernel[grid_split](
        proj_raw,
        left,
        right,
        out_gate,
        mu_in,
        rstd_in,
        fused_wsum,
        fused_bias,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # Release intermediate tensors we no longer need
    # ------------------------------------------------------------------
    del proj_raw, fused_weight, fused_wsum, fused_bias, x_flat, x_fp16, mu_in, rstd_in

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]   FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]   FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics of the contracted tensor (second LN)
    # ------------------------------------------------------------------
    mu_out   = torch.empty(R, dtype=torch.float32, device=device)
    rstd_out = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_OUTSTAT = 256
    BLOCK_K_OUTSTAT = 128               # covers H ≤ 768
    grid_outstat = (triton.cdiv(R, BLOCK_M_OUTSTAT),)
    rowwise_stats_kernel[grid_outstat](
        out,
        mu_out,
        rstd_out,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUTSTAT,
        BLOCK_K=BLOCK_K_OUTSTAT,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out_gate multiplication and projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_FINAL = 128   # rows per program (tiles over R)
    BLOCK_K_FINAL = 64    # hidden‑dim tile (increased from 32)
    BLOCK_N_FINAL = 128   # output‑dim tile

    grid_final = (
        triton.cdiv(R, BLOCK_M_FINAL),
        triton.cdiv(dim, BLOCK_N_FINAL),
    )
    final_output_kernel[grid_final](
        out,
        out_gate,
        mu_out,
        rstd_out,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_FINAL,
        BLOCK_K=BLOCK_K_FINAL,
        BLOCK_N=BLOCK_N_FINAL,
        num_warps=8,
    )

    return final
