"""
TriMul (outgoing) – fast forward‑only Triton+torch implementation.

Algorithm (high‑level)
----------------------
1️⃣  Row‑wise LayerNorm on the flattened pair representation (FP32→FP16)
    – a small Triton kernel (the only required kernel).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → tensor (R, 5·H)   with R = B·N·N.

3️⃣  Slice the fused tensor, apply the optional mask, compute sigmoid
    gates and reshape to [B, H, N, N] (all FP16).  All of this is done
    with regular PyTorch ops – far cheaper than a custom kernel.

4️⃣  Batched GEMM (cuBLAS):
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    i.e. `out = torch.bmm(left.view(B*H,N,N),
                         right.view(B*H,N,N).transpose(1,2))`.

5️⃣  Layer‑norm over the hidden dimension H (torch.nn.functional.layer_norm),
    element‑wise multiplication with `out_gate`.

6️⃣  Final linear projection (FP16 weight) → float32 output.

Only the first LayerNorm is implemented in Triton; the rest uses highly
optimised cuBLAS / cuDNN kernels, giving a large speed‑up over the
reference implementation while preserving numerical correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton implementation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, D]  FP32 input (flattened tensor)
    y_ptr,          # [R, D]  FP16 output
    w_ptr,          # [D]    LN weight (γ) FP32
    b_ptr,          # [D]    LN bias  (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implemented with Triton + torch.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    # flatten to (R, dim)
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free original FP32 tensor

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right + gates (cuBLAS)
    # ------------------------------------------------------------------
    # stack the five weight matrices → (5*H, dim) in FP16
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj = F.linear(x_norm, fused_w)          # (R, 5*H)  FP16
    del fused_w, x_norm

    # reshape to (R, 5, H)
    proj = proj.view(R, 5, H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply mask, sigmoid gates, reshape to [B, H, N, N]
    # ------------------------------------------------------------------
    # optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_flat = mask.reshape(R).to(torch.float16)  # [R]
    else:
        mask_flat = None

    # extract the five components
    left = proj[:, 0, :]                     # (R, H)
    right = proj[:, 1, :]                    # (R, H)
    left_gate = torch.sigmoid(proj[:, 2, :]) # (R, H)  FP16 (sigmoid works in FP16)
    right_gate = torch.sigmoid(proj[:, 3, :])
    out_gate = torch.sigmoid(proj[:, 4, :])

    if mask_flat is not None:
        mask_f = mask_flat[:, None]           # (R, 1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # reshape to (B, H, N, N) – note the original flatten order is (b,i,k)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    del proj, left_gate, right_gate, out_gate  # keep only needed tensors

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right.view(B * H, N, N)  # (B*H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)
    out = out.view(B, H, N, N)                             # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden dim) + out_gate multiplicaton
    # ------------------------------------------------------------------
    # move hidden dim to last axis for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()

    # convert LayerNorm parameters to FP16 for speed
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )  # (B, N, N, H) FP16

    out_norm = out_norm * out_gate_perm                     # element‑wise gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (hidden_dim → dim)  (cuBLAS)
    # ------------------------------------------------------------------
    # flatten to (R, H)
    out_flat = out_norm.view(R, H)
    w_out = w["to_out.weight"].to(torch.float16)            # (dim, H)

    final_fp16 = F.linear(out_flat, w_out)                  # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
