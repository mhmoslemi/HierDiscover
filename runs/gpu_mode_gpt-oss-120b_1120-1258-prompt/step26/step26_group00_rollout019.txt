@triton.jit
def tri_mul_compute_kernel(
    proj_ptr,     # [R, 5*H] FP16
    out_ptr,      # [B, H, N, N] FP16
    out_gate_ptr, # [B, H, N, N] FP16
    mask_ptr,     # [R] FP16 (optional)
    B, N, H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids:
    # pid0: batch_head index (B*H)
    # pid1: tile row (i)
    # pid2: tile col (j)
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head
    b = pid0 // H
    h = pid0 % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute out_gate for this tile (independent of k)
    if HAS_MASK:
        # mask not used for out_gate
        pass
    # out_gate_raw location: base offset = (b*N*N + i*N + j) * (5*H) + 4*H + h
    # compute flatten index for i,j:
    NN = N * N
    row_base = b * NN
    # compute offset in projection: idx = row_base + i*N + j
    idx_ij = row_base + offs_i[:, None] * N + offs_j[None, :]  # shape [M,N] broadcast
    proj_stride = 5 * H
    out_gate_offset = idx_ij * proj_stride + 4 * H + h  # shape [M,N]
    # load raw out_gate values
    out_gate_raw = tl.load(proj_ptr + out_gate_offset,
                           mask=mask_i[:, None] & mask_j[None, :],
                           other=0.0)
    out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # store out_gate
    # compute out tensor offset:
    out_gate_out_offset = b * (H * NN) + h * NN + offs_i[:, None] * N + offs_j[None, :]
    tl.store(out_gate_ptr + out_gate_out_offset,
             out_gate_val,
             mask=mask_i[:, None] & mask_j[None, :])

    # Now compute out = sum_k left[i,k] * right[j,k]
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # iterate over k dimension in chunks
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left raw for (i,k)
        # row index for left = b*N*N + i*N + k, offset to left = h
        idx_left = row_base + offs_i[:, None] * N + offs_k[None, :]  # shape [M,K]
        left_offset = idx_left * proj_stride + h  # offset for left_proj
        left_raw = tl.load(proj_ptr + left_offset,
                           mask=mask_i[:, None] & mask_k[None, :],
                           other=0.0)
        # left_gate
        left_gate_offset = idx_left * proj_stride + 2 * H + h
        left_gate_raw = tl.load(proj_ptr + left_gate_offset,
                                mask=mask_i[:, None] & mask_k[None, :],
                                other=0.0)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)

        # mask for left
        if HAS_MASK:
            mask_val = tl.load(mask_ptr + (row_base + offs_i[:, None] * N + offs_k[None, :]),
                               mask=mask_i[:, None] & mask_k[None, :],
                               other=1.0)
            mask_val = mask_val[..., None] # broadcast later? We'll incorporate in left
        else:
            mask_val = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float16)

        left = left_raw * mask_val * left_gate

        # right raw for (j,k) (note we need shape [K,N] for dot)
        # compute idx_right = b*N*N + j*N + k
        # shape for dot: (K,N) (k rows, j columns)
        # So we need to compute offsets with shape [K,N] where first dim is k, second dim is j
        idx_right = row_base + offs_k[:, None] * N + offs_j[None, :]  # shape [K,N]
        right_offset = idx_right * proj_stride + H + h  # offset for right_proj
        right_raw = tl.load(proj_ptr + right_offset,
                            mask=mask_k[:, None] & mask_j[None, :],
                            other=0.0)
        # right_gate
        right_gate_offset = idx_right * proj_stride + 3 * H + h
        right_gate_raw = tl.load(proj_ptr + right_gate_offset,
                                 mask=mask_k[:, None] & mask_j[None, :],
                                 other=0.0)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        # mask for right side
        if HAS_MASK:
            mask_val_r = tl.load(mask_ptr + (row_base + offs_k[:, None] * N + offs_j[None, :]),
                                 mask=mask_k[:, None] & mask_j[None, :],
                                 other=1.0)
            mask_val_r = mask_val_r[..., None] # shape [K,N,1] broadcast
        else:
            mask_val_r = tl.full([BLOCK_K, 1], 1.0, dtype=tl.float16)

        right = right_raw * mask_val_r * right_gate

        # Compute dot: left (M,K) dot right.T? Actually right shape is (K,N) already, got transposed arrangement.
        # left : shape [M,K], right : shape [K,N] -> dot yields [M,N]
        acc += tl.dot(left, right).to(tl.float32)
    # store out
    out_offset = b * (H * NN) + h * NN + offs_i[:, None] * N + offs_j[None, :]
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
```

@triton.jit
def tri_mul_compute_kernel(
    proj_ptr,  # [R, 5*H] FP16
    out_ptr,   # [B, H, N, N] FP16
    out_gate_ptr, # [B, H, N, N] FP16
    mask_ptr,  # [R] FP16 (may be dummy)
    B, N, H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_hb = tl.program_id(0)  # batch * hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden
    total_heads = B * H
    b = pid_hb // H
    h = pid_hb % H

    # compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute flat base offset for this batch
    NN = N * N
    row_base = b * NN  # scalar

    # compute out_gate per (i,j)
    # idx_ij shape [M,N]
    idx_ij = row_base + offs_i[:, None] * N + offs_j[None, :]  # shape (M,N)
    proj_stride = 5 * H

    out_gate_offset = idx_ij * proj_stride + 4 * H + h
    out_gate_raw = tl.load(
        proj_ptr + out_gate_offset,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0
    )
    # sigmoid
    out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # store out_gate
    out_gate_out_offset = b * (H * NN) + h * NN + offs_i[:, None] * N + offs_j[None, :]
    tl.store(out_gate_ptr + out_gate_out_offset,
             out_gate,
             mask=mask_i[:, None] & mask_j[None, :])

    # accumulator for out
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left raw for shape (M,K)
        idx_left = row_base + offs_i[:, None] * N + offs_k[None, :]
        left_offset = idx_left * proj_stride + h
        left_raw = tl.load(
            proj_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        left_gate_offset = idx_left * proj_stride + 2 * H + h
        left_gate_raw = tl.load(
            proj_ptr + left_gate_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)

        if HAS_MASK:
            mask_val = tl.load(
                mask_ptr + idx_left,
                mask=mask_i[:, None] & mask_k[None, :],
                other=1.0
            )
        else:
            mask_val = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float16)
        # expand mask to (M,K) (the broadcast will happen automatically)
        left = left_raw * mask_val * left_gate

        # right raw for shape (K,N). Note we need to orient as (K,N) for dot.
        idx_right = row_base + offs_k[:, None] * N + offs_j[None, :]
        # right projects offset:
        right_offset = idx_right * proj_stride + H + h
        right_raw = tl.load(
            proj_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        right_gate_offset = idx_right * proj_stride + 3 * H + h
        right_gate_raw = tl.load(
            proj_ptr + right_gate_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        if HAS_MASK:
            mask_val_r = tl.load(
                mask_ptr + idx_right,
                mask=mask_k[:, None] & mask_j[None, :],
                other=1.0
            )
        else:
            mask_val_r = tl.full([BLOCK_K, 1], 1.0, dtype=tl.float16)

        right = right_raw * mask_val_r * right_gate

        # compute dot: left (M,K) dot right (K,N) -> (M,N)
        # tl.dot expects shape (M,K) x (K,N)
        acc += tl.dot(left, right).to(tl.float32)
    # store output
    out_offset = b * (H * NN) + h * NN + offs_i[:, None] * N + offs_j[None, :]
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3 style) – highly‑optimized forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣  One fused linear projection (5 × hidden_dim) on the normalized tensor.
3️⃣  Fused Triton kernel:
    • Decode the flattened (b,i,k) / (b,j,k) indices.
    • Apply mask (if present) and the 3 sigmoid gates.
    • Accumulate the batched matrix‑multiply
          out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
      while also writing the out‑gate tensor
          out_gate[b,h,i,j] = sigmoid(projected_out_gate).
    This kernel replaces the separate split‑gate + torch.bmm steps,
    saving one write‑read of the huge left/right tensors.
4️⃣  Row‑wise mean / rsqrt‑variance of the contracted tensor (LayerNorm)
    – a small Triton kernel.
5️⃣  Fuse the second LayerNorm, out‑gate multiplication and the final
    projection into a single Triton kernel that writes the result
    `[B, N, N, dim]` in float32.

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision
(FP16 for the bulk of the work, FP32 for reductions) while preserving
the numerical quality of the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, dim] FP32 input (contiguous)
    y_ptr,               # [R, dim] FP16 output
    w_ptr,               # [dim]   LN weight (γ) FP32
    b_ptr,               # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean and variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Fused kernel: compute left/right gating, out_gate and the core einsum
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_compute_kernel(
    proj_ptr,          # [R, 5*H]   FP16 fused projection
    out_ptr,           # [B, H, N, N] FP16 (contracted tensor)
    out_gate_ptr,      # [B, H, N, N] FP16 (out‑gate)
    mask_ptr,          # [R] FP16 (optional, 0/1)
    B, N, H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes:
        left  = left_proj * mask * sigmoid(left_gate)
        right = right_proj * mask * sigmoid(right_gate)
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
        out_gate[b,h,i,j] = sigmoid(out_gate_proj[b,i,j,h])
    No intermediate left/right tensors are materialised.
    """
    pid_head = tl.program_id(0)   # flattened (b, h) index
    pid_i    = tl.program_id(1)   # tile over i  (first N dim)
    pid_j    = tl.program_id(2)   # tile over j  (second N dim)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    NN = N * N
    row_base = b * NN                     # base of the flattened (i,k) / (j,k) rows
    proj_stride = 5 * H                   # stride in proj (5*H values per row)
    hidden_stride = NN                    # stride for one hidden plane in out/out_gate

    # ------------------------------------------------------------------
    # 3a) write out_gate  (sigmoid of the 5th projection block)
    # ------------------------------------------------------------------
    idx_ij = row_base + offs_i[:, None] * N + offs_j[None, :]          # [M, N]
    out_gate_offset = idx_ij * proj_stride + 4 * H + h                # [M, N]
    out_gate_raw = tl.load(
        proj_ptr + out_gate_offset,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    # sigmoid in FP32 for stability
    out_gate_fp32 = 1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))
    out_gate = out_gate_fp32.to(tl.float16)

    out_gate_base = b * (H * hidden_stride) + h * hidden_stride
    out_gate_store_off = out_gate_base + offs_i[:, None] * N + offs_j[None, :]
    tl.store(
        out_gate_ptr + out_gate_store_off,
        out_gate,
        mask=mask_i[:, None] & mask_j[None, :],
    )

    # ------------------------------------------------------------------
    # 3b) compute the batched GEMM (left @ rightᵀ) with gating & mask
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left side : shape (M, K) ------------------------------------
        idx_left = row_base + offs_i[:, None] * N + offs_k[None, :]           # [M, K]
        left_offset = idx_left * proj_stride + h                              # left_proj
        left_raw = tl.load(
            proj_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        left_gate_offset = idx_left * proj_stride + 2 * H + h                 # left_gate
        left_gate_raw = tl.load(
            proj_ptr + left_gate_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)

        if HAS_MASK:
            mask_val = tl.load(
                mask_ptr + idx_left,
                mask=mask_i[:, None] & mask_k[None, :],
                other=1.0,
            )
            left = left_raw * mask_val * left_gate
        else:
            left = left_raw * left_gate

        # ---- right side : shape (K, N) -----------------------------------
        idx_right = row_base + offs_k[:, None] * N + offs_j[None, :]          # [K, N]
        right_offset = idx_right * proj_stride + H + h                       # right_proj
        right_raw = tl.load(
            proj_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        right_gate_offset = idx_right * proj_stride + 3 * H + h               # right_gate
        right_gate_raw = tl.load(
            proj_ptr + right_gate_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        if HAS_MASK:
            mask_val_r = tl.load(
                mask_ptr + idx_right,
                mask=mask_k[:, None] & mask_j[None, :],
                other=1.0,
            )
            right = right_raw * mask_val_r * right_gate
        else:
            right = right_raw * right_gate

        # block‑wise dot product: (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right).to(tl.float32)

    # write out tensor
    out_base = b * (H * hidden_stride) + h * hidden_stride
    out_store_off = out_base + offs_i[:, None] * N + offs_j[None, :]
    tl.store(
        out_ptr + out_store_off,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over rows   (R)
    pid_n = tl.program_id(1)   # tile over output dim (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm params (γ, β) for current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  γ· + β   →   * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [M, K]
        term = norm * vals_gate                                         # [M, K]

        term_fp16 = term.to(tl.float16)  # cast back to fp16 for the matmul

        # load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------

    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # ------------------------------------------------------------------

        [

            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],

        dim=0,

    # ------------------------------------------------------------------

    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_M_OUT = 64   # rows per program tile
    BLOCK_N_OUT = 64   # cols per program tile
    BLOCK_K_OUT = 32   # reduction over the sequence dimension

    grid_compute = (

        out_gate,
        mask_ptr,
        B,
        N,
        H,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_N=BLOCK_N_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128    # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + gating + final projection
    # ------------------------------------------------------------------

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_FINAL = 128   # rows per program (tiles over R)
    BLOCK_K_FINAL = 32    # hidden‑dim chunk
    BLOCK_N_FINAL = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_FINAL),
        triton.cdiv(dim, BLOCK_N_FINAL),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_FINAL,
        BLOCK_K=BLOCK_K_FINAL,
        BLOCK_N=BLOCK_N_FINAL,
        num_warps=8,
    )

    return final
