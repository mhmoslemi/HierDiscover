"""
Outgoing TriMul (AlphaFold‑3) – forward only, highly‑optimized.
================================================================
Algorithm
---------
1️⃣ **Pairwise LayerNorm** – Use torch’s fused cuDNN LayerNorm on the
   flattened pair tensor ([R, dim]) because it is far faster than a
   custom Triton kernel for the modest channel size (dim = 128/384).

2️⃣ **Fused projection** – Concatenate the five weight matrices
   (left, right, left‑gate, right‑gate, out‑gate) into a single
   [5·H, dim] matrix and apply one fp16 GEMM:  (R,dim)·(dim,5·H) → (R,5·H).

3️⃣ **Split + mask + sigmoid** – A small Triton kernel reads the fused
   projection, applies the optional pair‑mask, evaluates the three
   sigmoids and writes the three tensors `left`, `right` and
   `out_gate` in layout [B, H, N, N] (fp16).

4️⃣ **Core contraction** – Batched GEMM (cuBLAS) over the hidden
   dimension:
        out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k] .
   Implemented with a single `torch.bmm`.

5️⃣ **Row‑wise statistics** – Compute μ and rstd over the hidden axis
   (H) with a lightweight Triton kernel; the kernel works on fp16
   data but accumulates in fp32.

6️⃣ **Second LayerNorm + gate + final projection** – All remaining
   operations (LayerNorm over H, multiplication by `out_gate`,
   and the final linear projection back to `dim`) are fused into one
   Triton kernel.  The kernel streams the hidden dimension in tiles,
   applies the learned LN parameters (γ,β), multiplies by the gate,
   and performs the final fp16·fp16 → fp32 dot‑product with the weight
   matrix (H×dim).

All heavy‑compute steps are handled by cuBLAS (steps 2 & 4) while the
memory‑bound per‑row work is done in compact Triton kernels (steps 3,
5, 6).  The overall launch overhead is minimal (only three kernels),
which yields a substantial speed‑up over the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   fp16 fused projection
    left_ptr,      # [B, H, N, N] fp16
    right_ptr,     # [B, H, N, N] fp16
    out_gate_ptr,  # [B, H, N, N] fp16
    mask_ptr,      # [R]        fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode (b,i,k) from the flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Offset inside the fused projection: [left | right | lg | rg | og] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        og    = out_gate               # no mask

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R,BLOCK_H]

        tl.store(left_ptr     + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] fp16
    mu_ptr,    # [R]           fp32
    rstd_ptr,  # [R]           fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] fp16
    gate_ptr,       # [B, H, N, N] fp16 (out_gate)
    mu_ptr,         # [R]           fp32
    rstd_ptr,       # [R]           fp32
    ln_w_ptr,       # [H]           fp32 (to_out_norm.weight)
    ln_b_ptr,       # [H]           fp32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]      fp16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # over rows (R)
    pid_n = tl.program_id(1)   # over output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # row‑wise statistics (µ, rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # Load the hidden‑plane slice of the contracted tensor and the gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters (γ, β) – one per hidden channel
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out – µ) * rstd → γ· + β → * out_gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term   = norm * vals_gate                                  # [M,K]
        term_fp16 = term.to(tl.float16)                            # back to fp16 for dot

        # Load a tile of the final projection weight (H×dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))

    # ------------------------------------------------------------------
    # 1️⃣ Pairwise LayerNorm (torch – fast cuDNN implementation)
    # ------------------------------------------------------------------
    # Input layout is [B, N, N, dim]; normalized over the last dimension.
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.to(torch.float16)   # FP16 for the heavy GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (cuBLAS)
    # ------------------------------------------------------------------
    R = B * N * N  # total number of pairwise tokens
    H = hidden_dim

    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = torch.nn.functional.linear(x_norm_fp16, fused_w)   # [R, 5·H] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64      # rows per program
    BLOCK_H = 32      # hidden‑dim tile
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # fp16
    right_mat = right.view(B * H, N, N)              # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                       # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (fast Triton kernel)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256   # rows per program
    BLOCK_K_STATS = 32    # hidden‑dim tile (covers H ≤ 128 in 4 steps)
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    # Weight for the final projection: to_out.weight is [dim, H]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program for the final kernel
    BLOCK_K_OUT = 32    # hidden‑dim tile size
    BLOCK_N_OUT = 64    # tile size along the output channel dimension (dim)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (fp32)
        w["to_out_norm.bias"],     # β (fp32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
