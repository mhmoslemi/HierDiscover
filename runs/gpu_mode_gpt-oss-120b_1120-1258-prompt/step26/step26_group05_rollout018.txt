@triton.jit
def tri_mul_outgoing_kernel(
    x_ptr,  # [R, dim] FP32
    left_ptr, right_ptr, out_gate_ptr,   # output [B, H, N, N] FP16
    mask_ptr,   # optional [R] FP16
    norm_gamma_ptr, norm_beta_ptr,
    left_w_ptr, right_w_ptr, left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, dim, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute sum and sum_sq for each row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + 1e-5)

    # pre compute mask per row if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1] (fp16)
        mask_val_f32 = mask_val.to(tl.float32)
    else:
        mask_val_f32 = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # second pass for projection and gating
    # We'll initialize accumulators for each output group for hidden blocks later; but we need to iterate over hidden tiles.
    # We'll loop over hidden tiles, compute dot products and write results directly.
    # For each hidden tile we need to accumulate contributions across dim.
    # We'll allocate accumulators for each group within hidden tile dimension.

    # We'll compute number of hidden tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        # compute offset
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Initialize accumulators for each group
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_outgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # loop over dim
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # Normalize:
            gamma = tl.load(norm_gamma_ptr + offs_d, mask=mask_d, other=0.0)  # [BLOCK_D]
            beta  = tl.load(norm_beta_ptr + offs_d, mask=mask_d, other=0.0)

            norm = (x - mean[:, None]) * rstd[:, None] * gamma[None, :] + beta[None, :]  # FP32
            # cast to FP16 for dot
            norm_fp16 = norm.to(tl.float16)

            # Load weight tiles for each group
            # Left_proj weight tile
            left_w = tl.load(left_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0)   # FP16
            # Right_proj weight tile
            right_w = tl.load(right_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            # Left_gate weight tile
            lgate_w = tl.load(left_gate_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            # Right_gate weight tile
            rgate_w = tl.load(right_gate_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)
            # Out_gate weight tile
            outgate_w = tl.load(out_gate_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                mask=mask_h[:, None] & mask_d[None, :],
                                other=0.0)

            # Compute dot products: each results shape [BLOCK_R, BLOCK_H]
            acc_left += tl.dot(norm_fp16, left_w)   # Accepts [BLOCK_R,BLOCK_D] x [BLOCK_H,BLOCK_D] -> [BLOCK_R,BLOCK_H]
            acc_right += tl.dot(norm_fp16, right_w)
            acc_lgate += tl.dot(norm_fp16, lgate_w)
            acc_rgate += tl.dot(norm_fp16, rgate_w)
            acc_outgate += tl.dot(norm_fp16, outgate_w)

        # After dim loop, compute gates and final left/right/out_gate values

        # sigmoid gates (in FP32)
        left_gate = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-acc_outgate))).to(tl.float16)

        # Apply mask to left/right before gating
        left = acc_left.to(tl.float16) * mask_val_f32 * left_gate
        right = acc_right.to(tl.float16) * mask_val_f32 * right_gate
        out_gate = out_gate_val

        # Write output to global memory layout
        # Need to compute base address for each (b,h,i,k) row.

        # Decode row indices similar to split_fuse_gate:
        nn = N * N
        b = offs_r // nn
        rem = offs_r - b * nn
        i = rem // N
        k = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

        # For each hidden offset within tile
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        # Store
        tl.store(left_ptr + dst, left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣  Compute row‑wise LayerNorm statistics (mean, rsqrt‑var) **and**
    the five projected tensors (left, right, left‑gate, right‑gate,
    out‑gate) in a single kernel.  The projection is fused with the
    normalization, the mask is applied, the sigmoid gates are evaluated
    and the three tensors *left*, *right* and *out_gate* are written
    directly in the layout expected by the downstream batched GEMM.

2️⃣  Batched GEMM (`torch.bmm`) contracts the pairwise dimension:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    (shape → `[B·H, N, N]`).

3️⃣  Row‑wise mean & rsqrt‑variance of `out` (required for the second
    LayerNorm) – a tiny Triton kernel that only reads `out`.

4️⃣  Fuse the second LayerNorm, the output‑gate, and the final
    projection (`to_out`) in another Triton kernel.  The kernel reads
    `out`, the pre‑computed statistics, the output‑gate, and the weight
    matrix `to_out` and writes the final FP32 tensor `[B, N, N, dim]`.

All kernels run on the H100 (Triton 3.3.1).  No gradients are computed.  
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Fuse LayerNorm (statistics) + projection + gating + mask
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_kernel(
    x_ptr,                     # [R, dim]                 FP32 (raw input)
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N]          FP16 (outputs)
    mask_ptr,                  # [R] (optional)           FP16
    ln_gamma_ptr, ln_beta_ptr, # [dim]                    FP32 (norm weight/bias)
    left_w_ptr, right_w_ptr,                     # [H, dim]          FP16
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,  # [H, dim]  FP16
    R, dim, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Row‑wise LayerNorm + five linear maps + mask + sigmoid gates."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # -------------------------------------------------
    # first pass: compute mean & rsqrt‑variance per row
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + 1e-5)                           # [BLOCK_R]

    # -------------------------------------------------
    # optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)  # [BLOCK_R]
        m = m[:, None]                                    # [BLOCK_R, 1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # decode flat row index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k        # [BLOCK_R]

    # -------------------------------------------------
    # second pass: projection & gating
    # -------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for the five tensors (FP32)
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_outgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # load raw input tile
            x = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

            # ----- normalize -----
            gamma = tl.load(ln_gamma_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(ln_beta_ptr  + offs_d, mask=mask_d, other=0.0)   # FP32
            norm = (x - mean[:, None]) * rstd[:, None] * gamma[None, :] + beta[None, :]   # FP32
            norm_fp16 = norm.to(tl.float16)                                 # [BLOCK_R, BLOCK_D]

            # ----- load weight tiles (each [BLOCK_H, BLOCK_D]) -----
            left_w   = tl.load(left_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            right_w  = tl.load(right_w_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            lgate_w  = tl.load(left_gate_w_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            rgate_w  = tl.load(right_gate_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            outg_w   = tl.load(out_gate_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :], other=0.0)

            # ----- dot products (FP16·FP16→FP32) -----
            acc_left   += tl.dot(norm_fp16, left_w)
            acc_right  += tl.dot(norm_fp16, right_w)
            acc_lgate  += tl.dot(norm_fp16, lgate_w)
            acc_rgate  += tl.dot(norm_fp16, rgate_w)
            acc_outgate+= tl.dot(norm_fp16, outg_w)

        # ----- sigmoid gates (computed in FP32, cast back to FP16) -----
        left_gate   = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-acc_outgate))).to(tl.float16)

        # apply mask (broadcast) **before** the gate multiplication,
        # exactly as in the reference implementation.
        left  = (acc_left.to(tl.float16)  * m) * left_gate
        right = (acc_right.to(tl.float16) * m) * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, outg,      mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Row‑wise statistics of the contracted tensor (out)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]          FP32
    rstd_ptr,  # [R]          FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 3️⃣ Final LayerNorm + out‑gate + projection (out → final)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16  (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16  (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values and gate
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out - µ) * rstd  →  γ· + β  →  * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M,K]
        term = norm * vals_gate                                       # [M,K]
        term_fp16 = term.to(tl.float16)                               # back to FP16 for dot

        # load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # Cast and pack weights (FP16) – keep them contiguous
    # ------------------------------------------------------------------
    left_w      = w["left_proj.weight"].to(torch.float16).contiguous()      # [H, dim]
    right_w     = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Allocate output tensors
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Optional mask (flattened)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # ------------------------------------------------------------------
    # 1️⃣ Fuse LayerNorm + five linear maps + mask + sigmoid gates
    # ------------------------------------------------------------------
    BLOCK_R = 64      # rows per program
    BLOCK_D = 32      # dimension tile (dim is ≤128)
    BLOCK_H = 32      # hidden‑dim tile (hidden_dim ≤ 128)

    grid = (triton.cdiv(R, BLOCK_R),)
    fused_proj_kernel[grid](
        x.view(R, dim).contiguous(),
        left,
        right,
        out_gate,
        mask_ptr,
        w["norm.weight"],
        w["norm.bias"],
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        dim,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM (contract over the sequence dimension)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32          # H ≤ 128 → at most 4 tiles
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final linear projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows of R per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk (dim)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
