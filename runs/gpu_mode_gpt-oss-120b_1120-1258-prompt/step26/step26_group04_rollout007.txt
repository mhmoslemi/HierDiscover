"""
custom_kernel
==============

Forward‑only implementation of the **outgoing** TriMul operator (AlphaFold‑3 style)
using Triton kernels.

Algorithm
---------

1️⃣ **Row‑wise LayerNorm** on the flattened pair tensor `[B·N·N, dim]`.
   The kernel computes mean/variance in FP32 and writes the normalized values
   in FP16.

2️⃣ **Fused linear projection**: a single GEMM (`torch.nn.functional.linear`)
   produces the concatenation of the five linear maps
   `(left, right, left_gate, right_gate, out_gate)`, shape `[R, 5·H]`.

3️⃣ **Split + mask + sigmoid** in a Triton kernel:
   - Decode flat index `(b,i,k)` → destination `(b, h, i, k)`.
   - Apply optional pairwise mask (broadcast over `h`).
   - Apply sigmoid to the three gate tensors.
   - Store `left`, `right` and `out_gate` as `[B, H, N, N]` FP16.

4️⃣ **Batched GEMM** `out = left @ rightᵀ` (cuBLAS `torch.bmm`), still in FP16.
   Shape after reshaping: `[B, H, N, N]`.

5️⃣ **Row‑wise statistics** of `out` (mean & r‑std over the hidden dimension `H`)
   – another Triton kernel writes `mu` and `rstd` for each `(b,i,j)` pair.

6️⃣ **Final LayerNorm + gating + projection** (single Triton kernel):
   For every output row `(b,i,j)` we compute
   ```
   norm = ((out - mu) * rstd) * γ + β            # LN over H (γ,β are to_out_norm params)
   gated = norm * out_gate
   out_fp32 = Σ_h gated[h] · W_outᵀ[h, d]        # W_outᵀ ∈ ℝ^{H×dim}
   ```
   Results are written as FP32 `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and target NVIDIA H100
(Tensor‑Core friendly shapes, mixed‑precision, 8‑warp launch config).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] FP32 input
    y_ptr,    # [R, dim] FP16 output
    w_ptr,    # [dim]   LN weight (γ) FP32
    b_ptr,    # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]   # FP32 → FP16 store
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k   (the “k” dimension of the einsum)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H).  Constant‑folded at compile time.
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start of h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Projection layout: [left | right | left_gate | right_gate | out_gate] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw     = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numerical stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate, final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < H

        # Load hidden‑plane values (out) and gate (out_gate)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]
        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        # LayerNorm over hidden dim (γ, β are per‑hidden‑dim)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out - μ) / σ  →  normalized → γ·norm + β
        diff = (vals_out - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        # Apply out_gate
        term = norm * vals_gate                                    # [M, K]
        term_fp16 = term.to(tl.float16)                            # back to half for matmul

        # Load a slice of the final projection weight: [K, TILE_N]
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                          # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N) with fp16·fp16 → fp32 accumulation
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Write final FP32 result: layout [B, N, N, dim]
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Input tuple:
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float) or None
        - weights_dict : mapping with required weight tensors (all float32)
        - config_dict  : must contain "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack everything
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)          # True → ignore mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                 # [R, 5·H] FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # Convert mask to 0/1 float16 and flatten to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # Dummy tensor – never dereferenced because HAS_MASK is compile‑time 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right shaped as [B·H, N, N] → gemm with transposed right
    left_mat = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rstd of the contracted tensor
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 fits in a single tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
