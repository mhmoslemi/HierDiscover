"""
TriMul (outgoing) – fast forward only.

Algorithm (mixed‑precision)
---------------------------
1️⃣  Layer‑norm the input tensor `x` over the last dimension (dim) using the
    standard PyTorch kernel (FP32 → FP16).  This is faster than a custom
    Triton kernel for the modest “dim” sizes (128 / 384).

2️⃣  Fuse the five linear maps (left‑proj, right‑proj, left‑gate, right‑gate,
    out‑gate) into one GEMM (FP16) → tensor `[R,5·H]` where
    `R = B·N·N` and `H = hidden_dim`.

3️⃣  Split the fused projection, apply the optional pair‑wise mask,
    compute the sigmoid gates and store the three tensors
    `left`, `right`, `out_gate` in layout `[B, H, N, N]`.
    This step is performed by a small Triton kernel.

4️⃣  Batched matrix‑multiply:
        out_{b,h,:,:} = left_{b,h,:,:}  ·  right_{b,h,:,:}ᵀ
    (i.e. the “einsum” over the sequence dimension `k`).
    Implemented with `torch.bmm` → cuBLAS tensor‑core GEMM.

5️⃣  Second Layer‑Norm over the hidden dimension `H`,
    followed by element‑wise multiplication with `out_gate`,
    and a final linear projection `H → dim`.
    All three operations are done with highly‑optimized PyTorch kernels
    (FP16 GEMM + FP16 layer‑norm).

Only the split‑/gate‑kernel is written in Triton, satisfying the
requirement to contain a custom kernel while keeping the overall
runtime well below the target 500 µs on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16  (output)
    right_ptr,         # [B, H, N, N] FP16  (output)
    out_gate_ptr,      # [B, H, N, N] FP16  (output)
    mask_ptr,          # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k   (k = sequence index summed over)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride inside the [B,H,N,N] layout
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k          # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside the fused projection:
        #   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (stable in fp32, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # final destination offset for [B, H, N, N] layout
        dst_off = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of (b,i,k) tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input dimension (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to (R, dim)
    x_flat = x.reshape(R, dim).contiguous()            # FP32
    # FP32 LayerNorm (fast PyTorch implementation)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],        # FP32
        bias=w["norm.bias"],            # FP32
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # cast to FP16 for upcoming GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]
    proj = F.linear(x_norm, fused_weight)     # (R, 5·H) FP16
    del fused_weight, x_norm  # free memory

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
    # --------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ 2nd LayerNorm (over H) + out_gate + final projection
    # --------------------------------------------------------------
    # Bring tensors to shape (R, H) where R = B·N·N
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_flat = out_perm.view(-1, H)                         # (R, H) FP16

    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    gate_flat = gate_perm.view(-1, H)                       # (R, H) FP16

    # LayerNorm over hidden dimension (FP16, weight/bias cast to FP16 for speed)
    out_ln = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # (R, H) FP16

    # Apply the out‑gate
    gated = out_ln * gate_flat                               # (R, H) FP16

    # Final linear projection H → dim
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim) FP16
    final_flat = torch.matmul(gated, w_out_T)               # (R, dim) FP16

    # reshape back to [B, N, N, dim] and convert to FP32 (as required by the model)
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
