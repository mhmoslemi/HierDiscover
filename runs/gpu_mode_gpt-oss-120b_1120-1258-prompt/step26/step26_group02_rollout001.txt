"""
Outgoing TriMul (AlphaFold‑3) – fast forward‐only implementation.

Algorithm
---------
1️⃣ **Input LayerNorm** – use PyTorch’s fused LayerNorm (FP32 → FP16)
   – this is far faster than a hand‑written Triton version.
2️⃣ **Fused projection** – a single FP16 GEMM produces the concatenation of
   * left, right, left_gate, right_gate, out_gate* (shape [R,5·H]).
3️⃣ **Split + mask + sigmoid** – a Triton kernel extracts the five tensors,
   multiplies by the (optional) mask, applies the sigmoid gates and writes
   `left`, `right` and `out_gate` in layout `[B,H,N,N]` (FP16).
4️⃣ **Core contraction** – batched GEMM (`torch.bmm`) computes

        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]

   giving a tensor `[B,H,N,N]` (FP16).
5️⃣ **Second LayerNorm + out_gate + final projection** – a second Triton kernel
   normalises `out` over the hidden dimension, multiplies by `out_gate`,
   and directly multiplies with the transposed final weight `to_out`.  The
   result is written as `[B,N,N,dim]` (FP32).

All heavy‑memory kernels are written in mixed‑precision (FP16 compute, FP32
statistics) and are tuned for the NVIDIA H100 (Triton 3.3.1).  The implementation
requires no autograd support – only the forward pass is provided.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   FP16   – fused linear output
    left_ptr,       # [B, H, N, N] FP16 – left   matrix
    right_ptr,      # [B, H, N, N] FP16 – right  matrix
    out_gate_ptr,   # [B, H, N, N] FP16 – out gate
    mask_ptr,       # [R]             FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices

    # Base offset for (b,i,k) in the flattened [B, H, N, N] layout (h = 0)
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # Loop over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather 5·H fused projection slices
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # left (0·H … 1·H‑1)
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right (1·H … 2·H‑1)
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # left_gate (2·H … 3·H‑1)
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right_gate (3·H … 4·H‑1)
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # out_gate (4·H … 5·H‑1)
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offset for each hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,      mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16  – contracted tensor
    gate_ptr,       # [B, H, N, N] FP16  – out_gate
    ln_w_ptr,       # [H]          FP32  – LN weight (γ)
    ln_b_ptr,       # [H]          FP32  – LN bias   (β)
    w_out_T_ptr,    # [H, dim]     FP16  – transposed to_out weight
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """Row‑wise LN over `out`, multiply by `out_gate`,
       and compute the final linear projection in one pass."""
    pid_m = tl.program_id(0)   # rows over R = B·N·N
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # start of first hidden slice

    # ------------------------------------------------------------------
    # First pass: compute mean & variance of `out` (per row)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)      # [BLOCK_M]

    # ------------------------------------------------------------------
    # Second pass: LN‑normalized out * out_gate  →  dot with to_out weight
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = offs_k < H

        # Load raw tensors
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden

        out_vals  = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # Normalize `out` → (out - μ) * rstd
        norm = (out_vals - mean[:, None]) * rstd[:, None]
        # Apply LN affine transform
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply out_gate
        term = norm * gate_vals                         # [M, K]

        # Cast to FP16 for the final GEMM (tensor cores)
        term_fp16 = term.to(tl.float16)

        # Load slice of final weight [K, BLOCK_N] (row‑major)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # FP16

        # Accumulate (M,K)·(K,N) → (M,N) (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final output (flattened view of [B, N, N, dim])
    out_ptr_final = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(out_ptr_final, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use PyTorch fused kernel – much faster)
    # ------------------------------------------------------------------
    # reshape to [R, dim] (FP32) and apply LayerNorm, then cast to FP16
    x_flat = x.reshape(R, dim).contiguous()
    # fused LayerNorm: returns FP32 → we cast immediately to FP16
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5·H columns (FP16 GEMM)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy tensor

    BLOCK_R = 64     # rows per program
    BLOCK_H = 64     # hidden‑dim tile
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] → gemm on each hidden slice
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (single Triton kernel)
    # ------------------------------------------------------------------
    # `to_out.weight` has shape [dim, hidden_dim]; we need its transpose.
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows of the (B·N·N) dimension per block
    BLOCK_K_OUT = 32    # hidden‑dim tile (fits in shared memory)
    BLOCK_N_OUT = 128   # output‑feature tile
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ, FP32
        w["to_out_norm.bias"],     # β, FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
