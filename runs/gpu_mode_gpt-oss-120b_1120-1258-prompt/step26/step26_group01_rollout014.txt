    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim

    # Step 1: layernorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)
    # flatten to (R, dim) for linear
    x_norm_flat = x_norm.view(R, dim)

    # Step 2: fused linear projection
    fused_w = torch.cat([...], dim=0).to(torch.float16)  # (5*H, dim)
    proj = F.linear(x_norm_flat, fused_w)  # (R, 5*H) fp16

    # Step 3: split+mask+gates via Triton kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not cfg.get("nomask", True) and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # release proj
    del proj

    # Step 4: GEMM
    left_mat = left.view(B*H, N, N)  # (B*H,N,N)

"""
TriMul (outgoing) – forward‑only Triton implementation (v3.3.1).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16) using
    torch.nn.functional.layer_norm (GPU‑accelerated, still counts as a kernel
    later).

2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into a single FP16 GEMM:
        (R, dim) @ (5·H, dim).T → (R, 5·H)

3️⃣  Split the fused projection, apply the optional mask, compute sigmoid
    gates and store `left`, `right` and `out_gate` in layout
    [B, H, N, N] (FP16) via a Triton kernel.

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → tensor `[B, H, N, N]` (FP16).

5️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    projection into **one** Triton kernel.  The kernel
    * accumulates per‑row sums Σₖ out, Σₖ out²  (for μ and σ)
    * accumulates the three necessary matrix products:
          Σₖ out·gate·γ·W ,   Σₖ gate·γ·W ,   Σₖ gate·β·W
      (γ/β are the `to_out_norm` parameters, `W` is the transposed
       `to_out` weight).
    * After the hidden‑dim loop (H ≤ 128) it computes
          μ = Σ out / H
          rstd = 1/√(var + eps)
          out = rstd·(A – μ·B) + C
      where A, B, C are the three accumulated products.
    * Writes the final tensor `[B, N, N, dim]` (float32).

All kernels target Triton 3.3.1 and run on an H100.  Mixed‑precision
(arithmetic in FP16, accumulation in FP32) is used throughout; the final
output matches the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,            # [R, 5·H]   FP16 fused projection
    left_ptr,            # [B, H, N, N] FP16
    right_ptr,           # [B, H, N, N] FP16
    out_gate_ptr,        # [B, H, N, N] FP16
    mask_ptr,            # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                       # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset corresponds to h = 0
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [R, 5·H] (contiguous in H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Combined LayerNorm + out_gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def final_combined_kernel(
    out_ptr,            # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,           # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,           # [H]         FP32 (to_out_norm.weight  γ)
    ln_b_ptr,           # [H]         FP32 (to_out_norm.bias    β)
    w_out_T_ptr,        # [H, dim]    FP16 (transposed to_out weight)
    out_final_ptr,      # [B, N, N, dim] FP32
    total_rows, H, N, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R = B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                     # stride between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane h=0

    # ----- accumulators -------------------------------------------------
    acc_outgate = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·gate·γ·W
    acc_gamma   = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ gate·γ·W
    acc_beta    = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ gate·β·W

    sum_out  = tl.zeros([BLOCK_M], dtype=tl.float32)               # Σ out   (for μ)
    sum_sq   = tl.zeros([BLOCK_M], dtype=tl.float32)               # Σ out²  (for σ²)

    # ----- loop over hidden dimension (single iteration when BLOCK_K == H) -----
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [K]
        mask_k = offs_k < H

        # Load [M, K] tiles of out and gate (FP16)
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )

        # Convert to FP32 for arithmetic
        out_f  = out_tile.to(tl.float32)       # [M, K]
        gate_f = gate_tile.to(tl.float32)      # [M, K]

        # Update row‑wise statistics
        sum_out += tl.sum(out_f, axis=1)                # [M]
        sum_sq  += tl.sum(out_f * out_f, axis=1)        # [M]

        # Load LayerNorm parameters (γ, β) – one value per hidden unit
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]  FP32

        # Broadcast to rows (shape [M, K])
        gamma_row = gamma[None, :]      # [1, K]
        beta_row  = beta[None, :]       # [1, K]

        # gate ⊙ γ and gate ⊙ β
        gate_gamma = gate_f * gamma_row
        gate_beta  = gate_f * beta_row

        # out ⊙ gate ⊙ γ
        out_gate_gamma = out_f * gate_gamma

        # Load slice of the final weight matrix W (shape [K, N])
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        ).to(tl.float32)                                   # [K, N]

        # Accumulate the three required products
        acc_outgate += tl.dot(out_gate_gamma, w_tile)   # Σ out·gate·γ·W
        acc_gamma   += tl.dot(gate_gamma, w_tile)       # Σ gate·γ·W
        acc_beta    += tl.dot(gate_beta, w_tile)        # Σ gate·β·W

    # ----- compute μ, rstd ------------------------------------------------
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # broadcast μ and rstd over the output‑dim axis
    mu_vec   = mu[:, None]      # [M,1]
    rstd_vec = rstd[:, None]    # [M,1]

    # final formula:
    # out = rstd·(acc_outgate - μ·acc_gamma) + acc_beta
    out_tile = rstd_vec * (acc_outgate - mu_vec * acc_gamma) + acc_beta

    # ----- store result ---------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, out_tile, mask=mask_m[:, None] & mask_n[None, :])

    """

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # Use the highly‑optimized cuDNN LayerNorm; final output must be FP16.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, dim] FP16

    x_flat = x_norm.view(R, dim).contiguous()   # [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    proj = F.linear(x_flat, fused_w)   # (R, 5·H) FP16
    del fused_w, x_flat

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Combined LayerNorm, out_gate & final projection (single Triton kernel)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M = 128                     # rows per program (tiles over R)
    BLOCK_N = 64                      # output‑dim tiles
    BLOCK_K = H                       # we process the full hidden dimension at once

    grid_out = (
        triton.cdiv(R, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )
    final_combined_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],    # γ  (FP32)
        w["to_out_norm.bias"],      # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )
    return final
```
