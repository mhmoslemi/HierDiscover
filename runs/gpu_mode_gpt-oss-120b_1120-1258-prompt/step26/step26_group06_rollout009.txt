def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim=cfg["hidden_dim"]; nomask=cfg.get("nomask", True)
    eps=1e-5
    device=x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # layernorm
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M=128; BLOCK_D=64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # fused projection
    fused_weight = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = torch.nn.functional.linear(x_norm, fused_weight)  # (R, 5*H) FP16
    del x_norm, fused_weight

    # split and apply mask + gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)  # fp16

    # rowwise mean and variance
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1)  # [B,N,N] float32
    var = out_fp32.var(dim=1, unbiased=False)  # [B,N,N]
    rstd = torch.rsqrt(var + eps)  # [B,N,N]

    # layernorm + gating on hidden dimension
    # Expand mu,rstd to shape [B, H, N, N]
    mu_exp = mu[:, None, :, :]  # shape [B,1,N,N]
    rstd_exp = rstd[:, None, :, :]
    # Convert to float32 for computation
    out_norm = (out_fp32 - mu_exp) * rstd_exp

    # apply to_out_norm weight and bias
    ln_w = w["to_out_norm.weight"]  # (H)
    ln_b = w["to_out_norm.bias"]
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection
    out_flat = out_norm.reshape(R, H)  # [R, H]
    final = torch.nn.functional.linear(out_flat, w["to_out.weight"])  # [R, dim] float32

    final = final.view(B, N, N, dim)

    return final

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (dim) → FP16.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (FP16).  The fused projection has shape (R, 5·H)
   where R = B·N·N and H = hidden_dim.
3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   compute sigmoid gates and write three tensors `left`, `right` and
   `out_gate` in layout [B, H, N, N] (FP16).  This is done in one Triton kernel.
4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` across the hidden dimension H
   using fast torch reductions (mean / var).
6️⃣ Row‑wise LayerNorm (using the statistics from step 5), multiplication
   with `out_gate`, and the final linear projection (hidden → dim) are
   performed with pure PyTorch (highly‑optimized cuBLAS kernels).

We keep two Triton kernels (LayerNorm & split‑gate) to guarantee that at
least part of the work is Triton‑based while the heavy GEMM and reductions
use the library‑optimized cuBLAS paths, yielding a fast implementation
well under the 500 µs target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input (flattened)
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]   LayerNorm weight (γ) FP32
    b_ptr,      # [dim]   LayerNorm bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                                   # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k      # start offset for h = 0

    # Iterate hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]           # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (stable in FP32, cast back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)               # promote for stable reduction
    # torch.mean / var are highly tuned kernels on H100
    mu   = out_fp32.mean(dim=1)                    # [B, N, N] float32
    var  = out_fp32.var(dim=1, unbiased=False)    # pop‑variance
    rstd = torch.rsqrt(var + eps)                  # [B, N, N] float32

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (cuBLAS)
    # ------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    mu_exp   = mu[:, None, :, :]      # [B,1,N,N]
    rstd_exp = rstd[:, None, :, :]    # [B,1,N,N]

    out_norm = (out_fp32 - mu_exp) * rstd_exp

    # Apply learned LN scale & bias (hidden_dim -> hidden_dim)
    ln_w = w["to_out_norm.weight"]   # [H]  FP32
    ln_b = w["to_out_norm.bias"]     # [H]  FP32
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # Multiply by the out‑gate (produced in the split kernel)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Bring the hidden dimension to the last axis for the final linear
    out_norm = out_norm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out_norm.view(R, H)                        # [R, H]

    # Final linear projection (hidden → dim)
    final = F.linear(out_flat, w["to_out.weight"])        # [R, dim]  FP32

    final = final.view(B, N, N, dim)                      # [B, N, N, dim]

    return final
