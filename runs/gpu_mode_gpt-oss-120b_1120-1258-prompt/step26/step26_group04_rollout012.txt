def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg['dim']
    hidden_dim = cfg['hidden_dim']
    nomask = cfg.get('nomask', True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B*N*N
    # Step 1: LN
    x_flat = x.reshape(R, dim)
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,), weight=w['norm.weight'], bias=w['norm.bias'], eps=eps)
    x_norm = x_norm.to(torch.float16)
    # Step 2: fused projection
    fused_w = torch.cat([w['left_proj.weight'], w['right_proj.weight'],
                         w['left_gate.weight'], w['right_gate.weight'],
                         w['out_gate.weight']], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w) # shape (R,5*H)
    # Step 3: split and mask/gates
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Step 4: einsum via BMM
    left_mat = left.view(B*hidden_dim, N, N)
    right_mat = right.view(B*hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out.view(B, hidden_dim, N, N)
    # Step 5: second LN, gate, final projection using torch ops
    out_fp32 = out.float()
    # compute stats across hidden_dim
    mean = out_fp32.mean(dim=1, keepdim=True)
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)
    # apply LN parameters
    gamma = w['to_out_norm.weight'].view(1, hidden_dim, 1, 1)  # shape broadcast
    beta = w['to_out_norm.bias'].view(1, hidden_dim, 1, 1)
    out_norm = (out_fp32 - mean) * rstd
    out_norm = out_norm * gamma + beta
    out_norm = out_norm * out_gate.float()
    # final projection
    # reshape to (R, hidden_dim)
    out_flat = out_norm.permute(0,2,3,1).contiguous().view(R, hidden_dim)  # B,N,N,H -> (R,H)
    final = torch.nn.functional.linear(out_flat, w['to_out.weight'])
    final = final.view(B, N, N, dim)
    return final

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
    torch.nn.functional.layer_norm (highly‑optimized native kernel).

2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into a single GEMM.  The result has shape (R, 5·H) where
    R = B·N·N and H = hidden_dim.

3️⃣ A Triton kernel (split_fuse_gate_kernel) splits the fused projection,
    applies the optional pair‑wise mask, computes the three sigmoid gates,
    and writes three tensors (left, right, out_gate) in layout
    [B, H, N, N] (FP16).

4️⃣ Batched GEMM (torch.bmm) computes the core einsum
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    yielding `[B, H, N, N]` (FP16).

5️⃣ The second LayerNorm, out_gate multiplication and final linear projection
    are expressed with batched torch kernels (mean/var, rsqrt, element‑wise
    ops, and a final GEMM).  This avoids extra Triton launches and benefits
    from cuBLAS/Tensor‑Core speed.

The overall forward pass uses only one custom Triton kernel (the split &
gate kernel) and relies on torch's highly‑optimized primitives for the
remaining work, giving a very fast implementation on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# -------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        # mask is stored as 0/1 in FP16
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    # Base offset for hidden‑plane 0 in destination tensors
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # Loop over hidden dimension in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid in FP32 for stability, then back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                # out_gate is not masked

        # Destination offsets (broadcast over H)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    # torch native LayerNorm (fast, uses vectorized kernels)
    x_norm = F.layer_norm(x_flat, (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)                         # FP32
    x_norm = x_norm.to(torch.float16)                            # FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask is broadcasted over hidden dimension; convert to 0/1 float16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        # dummy tensor – never accessed when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS/Tensor‑Core)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication and final projection
    #    – performed with torch kernels (fast BLAS + vector ops)
    # --------------------------------------------------------------
    out_fp32 = out.float()                                 # FP32 for statistics

    # mean / variance over the hidden dimension H
    mean = out_fp32.mean(dim=1, keepdim=True)              # [B,1,N,N]
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)                          # [B,1,N,N]

    # apply learned LN parameters (γ,β) – broadcast over H
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)       # FP32
    beta  = w["to_out_norm.bias"  ].view(1, H, 1, 1)       # FP32
    out_norm = (out_fp32 - mean) * rstd
    out_norm = out_norm * gamma + beta                     # still FP32

    # apply out_gate (stored in FP16) – cast to FP32
    out_norm = out_norm * out_gate.float()                 # FP32

    # final linear projection: (R, H) @ (dim, H).T  → (R, dim)
    # reshape to (R, H)
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    final = F.linear(out_flat, w["to_out.weight"])                     # (R, dim) FP32
    final = final.view(B, N, N, dim)                                   # [B,N,N,dim]

    return final
