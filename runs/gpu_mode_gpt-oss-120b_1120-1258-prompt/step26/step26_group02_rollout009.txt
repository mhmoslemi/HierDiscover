#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.
---------------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣ Input LayerNorm (FP32) on the pairwise tensor ``x`` → ``x_ln`` (float32).
2️⃣ Cast to FP16 and apply a *single* fused linear layer that produces the
   five projected tensors (left, right, left_gate, right_gate, out_gate) in one
   GEMM.  The fused weight has shape ``[5·H, dim]``.
3️⃣ A tiny Triton kernel ``split_fuse_gate_kernel``:
   * decodes the flat index ``r = b·N·N + i·N + k``,
   * applies the optional mask,
   * evaluates the three sigmoids,
   * writes the three tensors ``left``, ``right`` and ``out_gate`` in the
     layout ``[B, H, N, N]`` (half‑precision).
4️⃣ Batched matrix‑multiply (cuBLAS) computes the core contraction

        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]

   i.e. ``out = torch.bmm(left.view(B·H, N, N),
                         right.view(B·H, N, N).transpose(1, 2))``.
5️⃣ The hidden‑dimension output is normalised with a second LayerNorm
   (FP16 → FP32 accumulator) using the learned ``to_out_norm`` parameters.
   The resulting tensor is multiplied element‑wise with ``out_gate``.
6️⃣ A final linear projection (still FP16) maps the normalised gated
   representation from ``H`` to ``dim``.  The result is cast back to FP32.

All heavy tensor algebra (steps 2–4,6) uses Tensor‑Core‑optimised cuBLAS
operations; the only Triton kernel handles the split/gate logic and the
optional masking.  This balances maximal throughput on an H100 while keeping
the implementation simple and numerically stable.

The function ``custom_kernel`` expects a tuple
``(input, mask, weights, config)`` and returns the processed tensor
``[B, N, N, dim]`` (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask, compute sigmoids,
# and write the three 4‑D tensors (left, right, out_gate).
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]  FP16 fused projection (flattened)
    left_ptr,        # [B, H, N, N]  FP16 output
    right_ptr,       # [B, H, N, N]  FP16 output
    out_gate_ptr,    # [B, H, N, N]  FP16 output
    mask_ptr,        # [R] FP16 mask (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    # Decode flat index r → (b, i, k)  where  r = b·N·N + i·N + k
    # --------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        # broadcast mask over hidden dimension later
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride (in elements) between successive hidden slices in the
    # destination layout [B, H, N, N] (contiguous in H after flattening i,k)
    stride_hidden = N * N

    # base offset for the slice (b,i,k) at hidden index h = 0
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    # --------------------------------------------------------------
    # Iterate over hidden dimension tiles
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Compute the starting index inside the fused projection matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five projected slices (always contiguous)
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for a bit more accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask (if any) and the two gate sigmoids
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate            # no mask on out_gate itself

        # Destination offsets within the flattened view of [B, H, N, N]
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point: forward pass of outgoing TriMul (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → cast to half
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension; we keep FP32
    # for maximal numeric fidelity, then down‑cast.
    x_ln = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim]  FP32
    x_ln = x_ln.to(torch.float16)   # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection for left/right + the three gates
    # --------------------------------------------------------------
    # Stack the five weight matrices (each shape [H, dim]) vertically.
    fused_W = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]  FP16

    # Flatten pairwise dimension → (R, dim) and run a single GEMM.
    x_flat = x_ln.reshape(R, dim).contiguous()      # [R, dim]
    proj = F.linear(x_flat, fused_W)                 # [R, 5·H]  FP16
    del x_ln, x_flat, fused_W

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (Triton)
    # --------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_f16
    else:
        has_mask = 0
        # dummy tensor – never dereferenced when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64   # tune these values for the target HW
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for the batched matmul
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N]  FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension,
    #    followed by out‑gate multiplication.
    # --------------------------------------------------------------
    # Bring tensors to (B, N, N, H) layout for a standard LayerNorm.
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]  FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]  FP16

    # Convert LN parameters to half‑precision (the kernel accepts FP16).
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16)

    # LayerNorm (FP16 → FP32 accumulator internally)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                           # [B, N, N, H]  FP16

    # Apply the learned gating value
    gated = out_norm * out_gate_perm                # [B, N, N, H]  FP16
    del out_norm, out_gate_perm, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection (H → dim) and cast back to FP32
    # --------------------------------------------------------------
    gated_flat = gated.view(-1, H)                       # [R, H]  FP16
    w_out = w["to_out.weight"].to(torch.float16)         # [dim, H]  FP16
    final_fp16 = F.linear(gated_flat, w_out, bias=None) # [R, dim]  FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # FP32 output

    # --------------------------------------------------------------
    return final
