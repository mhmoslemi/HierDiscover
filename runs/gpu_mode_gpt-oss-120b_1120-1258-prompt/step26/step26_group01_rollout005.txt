    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. First LayerNorm
    # Use torch.nn.functional.layer_norm with weight and bias
    x_norm = F.layer_norm(x,
                          normalized_shape=dim,
                          weight=w["norm.weight"],  # Should be shape [dim]
                          bias=w["norm.bias"],
                          eps=eps)  # FP32
    x_norm = x_norm.to(torch.float16).contiguous()  # Cast to fp16 for projection

    # 2. Fused linear projection (5*H)
    fused_weight = torch.cat(
        [w["left_proj.weight"], w["right_proj.weight"],
         w["left_gate.weight"], w["right_gate.weight"], w["out_gate.weight"]],
        dim=0).to(torch.float16)   # shape [5*H, dim]
    # Use linear: out = x_norm @ fused_weight.T if needed.
    proj = F.linear(x_norm.view(R, dim), fused_weight)   # [R, 5*H] half
    # Release x_norm
    del x_norm

    # 3. Split, mask and sigmoid gates
    left = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    right = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, H, N, N), dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64  # At most hidden dim 128 ; 2 iter
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Compute core einsum via batched GEMM
    # left, right shape [B,H,N,N] -> view as (B*H,N,N)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)
    # Free left,right
    del left, right

    # 5. Second LayerNorm over hidden dimension H
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H).to(torch.float16)
    # Use torch.nn.functional.layer_norm; weight and bias are FP32
    out_norm_flat = F.layer_norm(out_flat,
                                 normalized_shape=(H,),
                                 weight=w["to_out_norm.weight"],
                                 bias=w["to_out_norm.bias"],
                                 eps=eps)
    # Reshape back
    out_norm = out_norm_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # 6. Apply out_gate multiplicative gating
    # out_gate shape [B,H,N,N], out_norm same shape
    gated = out_norm * out_gate
    del out_norm, out_gate

    # 7. Final linear projection to dim
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    # Convert weight to half
    to_out_weight = w["to_out.weight"].to(torch.float16)  # [dim, H]
    final_half = F.linear(gated_flat, to_out_weight)  # [R, dim] half
    final = final_half.to(torch.float32).view(B, N, N, dim)  # return float32

    return final

"""
TriMul (outgoing) – forward‑only Triton implementation.

Algorithm
---------
1️⃣  First LayerNorm (pair representation) – performed with torch.nn.functional.layer_norm
    (FP32 → FP16 conversion).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    into a single FP16 GEMM:
        proj = x_norm @ W_fused.T    # shape (R, 5·H)

3️⃣  Split the fused projection, apply the optional mask, and compute the sigmoid
    gates inside a small Triton kernel → tensors `left`, `right`, `out_gate`
    (layout [B, H, N, N], dtype fp16).

4️⃣  Core einsum  “out[i,j] = Σₖ left[i,k]·right[j,k]”  is realized as a batched
    matrix multiplication:
        out = (left.view(B·H,N,N) @ right.view(B·H,N,N).transpose(1,2))
    → shape (B, H, N, N), FP16.

5️⃣  Second LayerNorm over the hidden dimension H using torch.nn.functional.layer_norm.

6️⃣  Element‑wise multiplication with `out_gate`.

7️⃣  Final linear projection (H → dim) via a single cuBLAS GEMM:
        final = (gated @ to_out.weight.T)

All heavy GEMMs leverage cuBLAS/Tensor‑cores, while the only custom Triton
kernel handles the split, mask, and sigmoid‑gate logic.  The result is a
float‑32 tensor `[B, N, N, dim]` matching the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel: split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  (fp16) fused projection
    left_ptr,          # [B, H, N, N] (fp16)
    right_ptr,         # [B, H, N, N] (fp16)
    out_gate_ptr,      # [B, H, N, N] (fp16)
    mask_ptr,          # [R] (fp16) – optional, broadcast over H
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flattened index (b, i, k) from r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # each hidden plane occupies a contiguous slab of size N·N inside the
    # destination tensors
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k               # offset for h = 0

    # loop over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # --------------------------------------------------------------
        # Gather the five concatenated projections for the current block
        # proj layout: [R, 5·H] → stride = 5·H
        # --------------------------------------------------------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]     # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Sigmoid (computed in FP32 for stability) and mask+gate fusion
        # --------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # --------------------------------------------------------------
        # Store into the destination tensors
        # --------------------------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient) using Triton + cuBLAS.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    ####################################################################
    # 1️⃣ First LayerNorm  (FP32 → FP16)
    ####################################################################
    # torch.nn.functional.layer_norm works on the last dimension;
    # we keep the tensor in FP32 for numerical stability and cast to fp16
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16).contiguous()          # [B, N, N, dim] (fp16)

    ####################################################################
    # 2️⃣ Fuse five linear maps into a single GEMM
    ####################################################################
    # Concatenate the five weight matrices (each of shape [H, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), fused_w)           # [R, 5·H] (fp16)
    del x_norm  # free memory early

    ####################################################################
    # 3️⃣ Split projection, apply mask, compute sigmoid gates
    ####################################################################
    left = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    right = torch.empty((B, H, N, N), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, H, N, N), dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed because HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128                # rows per program
    BLOCK_H = 64                 # hidden‑dim slice per program (H ≤ 128)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    ####################################################################
    # 4️⃣ Core einsum via batched GEMM (left · rightᵀ)
    ####################################################################
    # shape: [B·H, N, N] × [B·H, N, N]ᵀ  →  [B·H, N, N]
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                # [B, H, N, N] (fp16)
    del left, right

    ####################################################################
    # 5️⃣ Second LayerNorm over the hidden dimension H
    ####################################################################
    # flatten to (R, H) where each row corresponds to (b, i, j)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] (fp16)
    out_norm_flat = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                             # [R, H] (fp16)

    # reshape back to [B, H, N, N]
    out_norm = (
        out_norm_flat.view(B, N, N, H)
        .permute(0, 3, 1, 2)
        .contiguous()
    )                                                             # [B, H, N, N] (fp16)

    ####################################################################
    # 6️⃣ Apply the output gate (element‑wise)
    ####################################################################
    gated = out_norm * out_gate
    del out_norm, out_gate

    ####################################################################
    # 7️⃣ Final projection to `dim` (FP16 GEMM) → cast to FP32
    ####################################################################
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] (fp16)
    to_out_w = w["to_out.weight"].to(torch.float16)                 # [dim, H] (fp16)
    final_fp16 = F.linear(gated_flat, to_out_w)                     # [R, dim] (fp16)
    final = final_fp16.to(torch.float32).view(B, N, N, dim)         # [B, N, N, dim]

    return final
```
