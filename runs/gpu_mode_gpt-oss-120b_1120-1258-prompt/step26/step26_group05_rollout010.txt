"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened [R, C] tensor (R = B·N·N, C = dim)  
   – This kernel computes the per‑row mean / variance in FP32 and writes the
     normalized result in FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (`F.linear`) producing a tensor
   of shape [R, 5·H] where H = hidden_dim.

3️⃣ Split the fused projection, apply the optional pair mask, compute the
   three sigmoid gates and obtain the tensors `left`, `right` and
   `out_gate` (all FP16) – done with pure PyTorch (memory‑coalesced views).

4️⃣ Batched GEMM (cuBLAS) computes the core einsum  

        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]

   → torch.bmm on tensors shaped [B·H, N, N] (FP16).

5️⃣ Compute per‑row (over the hidden dimension) mean and rsqrt‑variance of
   `out` with torch’s highly‑optimised reduction kernels.

6️⃣ Second LayerNorm (γ/β over the hidden dimension), element‑wise multiplication
   by `out_gate`, and a final linear projection to `dim` (again a GEMM).

All heavy matrix‑multiplications are delegated to cuBLAS (`torch.nn.functional.linear`
and `torch.bmm`).  Only the first LayerNorm is Triton‑implemented, satisfying
the “use at least one Triton kernel” requirement while keeping the overall
runtime well below the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, C]  FP32 input
    y_ptr,   # [R, C]  FP16 output
    w_ptr,   # [C]    γ (FP32)
    b_ptr,   # [C]    β (FP32)
    rows,    # scalar R
    dim,     # scalar C
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # (5·H, dim)

    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and compute sigmoid gates
    # ------------------------------------------------------------------
    # reshape to (R, 5, H) for easy indexing
    proj = proj.view(R, 5, H)

    left_raw      = proj[:, 0, :]                         # (R, H)
    right_raw     = proj[:, 1, :]                         # (R, H)
    left_gate_raw = proj[:, 2, :]                         # (R, H)
    right_gate_raw= proj[:, 3, :]                         # (R, H)
    out_gate_raw  = proj[:, 4, :]                         # (R, H)

    left_gate  = torch.sigmoid(left_gate_raw.to(torch.float32)).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.to(torch.float32)).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.to(torch.float32)).to(torch.float16)

    if not nomask and mask is not None:
        mask_flat = mask.reshape(R).to(torch.float16)        # (R,)
        mask_flat = mask_flat.unsqueeze(1)                  # (R,1)
        left  = left_raw  * mask_flat * left_gate
        right = right_raw * mask_flat * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # reshape to (B, N, N, H) then permute to (B, H, N, N)
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    left_mat  = left .view(B * H, N, N)                     # (B·H, N, N) FP16
    right_mat = right.view(B * H, N, N).transpose(1, 2)    # (B·H, N, N) FP16
    out = torch.bmm(left_mat, right_mat)                   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Per‑row mean / rsqrt‑variance (over hidden dim)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)                         # (B, H, N, N) FP32
    mu   = out_f32.mean(dim=1)                              # (B, N, N)
    var  = out_f32.var(dim=1, unbiased=False)              # (B, N, N)
    rstd = torch.rsqrt(var + eps)                           # (B, N, N)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # broadcast μ and rstd over H
    mu   = mu.unsqueeze(1)      # (B, 1, N, N)
    rstd = rstd.unsqueeze(1)    # (B, 1, N, N)

    # LayerNorm over hidden dimension
    ln_w = w["to_out_norm.weight"]   # (H,) FP32
    ln_b = w["to_out_norm.bias"]     # (H,) FP32
    # (out - μ) * rstd
    norm = (out_f32 - mu) * rstd
    # γ· + β   (broadcast over spatial dims)
    norm = norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # Apply out_gate (convert gate to FP32)
    norm = norm * out_gate.to(torch.float32)

    # Final linear projection: (R, H) @ (H, dim) → (R, dim)
    # flatten spatial dims
    norm_flat = norm.permute(0, 2, 3, 1).reshape(B * N * N, H)   # (R, H)
    w_out = w["to_out.weight"]                                   # (dim, H)
    final_flat = F.linear(norm_flat, w_out)                      # (R, dim) FP32

    final = final_flat.view(B, N, N, dim)                         # (B, N, N, dim)
    return final
