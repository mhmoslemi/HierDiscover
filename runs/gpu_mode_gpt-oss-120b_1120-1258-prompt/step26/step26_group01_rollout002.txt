"""
TriMul (outgoing) – forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ One fused linear projection that produces the concatenated tensors
   (left, right, left_gate, right_gate, out_gate) of shape (R,5·H).
3️⃣ Split the projection, apply the optional mask, compute sigmoid gates
   and store left/right/out_gate in the layout required for the batched GEMM.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
5️⃣ Second LayerNorm (cuBLAS + torch kernels), multiplication by out_gate,
   and the final linear projection to `dim`.  All heavy matrix‑multiplications
   are delegated to cuBLAS (via `torch.bmm` / `torch.nn.functional.linear`).

Only the first LayerNorm is written in Triton (the rest uses highly‑optimised
PyTorch kernels).  This reduces kernel launch overhead while keeping
the most memory‑intensive step fused.

The final output is a `torch.float32` tensor of shape [B, N, N, dim],
identical to the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] input FP32
    y_ptr,    # [R, dim] output FP16
    w_ptr,    # [dim]    LN weight γ (FP32)
    b_ptr,    # [dim]    LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradients) using Triton + cuBLAS/torch kernels.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left, right, gates) – torch linear
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    # (R, 5·H) FP16
    proj = F.linear(x_norm, fused_w)
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    # reshape to (R, 5, H) for easy slicing
    proj = proj.view(R, 5, H)

    left_raw      = proj[:, 0, :]   # (R, H)
    right_raw     = proj[:, 1, :]   # (R, H)
    left_gate_raw = proj[:, 2, :]   # (R, H)
    right_gate_raw = proj[:, 3, :]  # (R, H)
    out_gate_raw  = proj[:, 4, :]   # (R, H)

    # sigmoid gates (computed in FP32 for stability)
    left_gate  = torch.sigmoid(left_gate_raw.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.float()).to(torch.float16)

    # optional mask (broadcast over hidden dimension)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R)          # (R,)
        mask_flat = mask_flat[:, None]                         # (R,1)
    else:
        mask_flat = torch.ones((R, 1), dtype=torch.float16, device=device)

    # apply mask and gates
    left  = left_raw  * mask_flat * left_gate
    right = right_raw * mask_flat * right_gate
    # out_gate stays unchanged (already sigmoid‑ed)

    # reshape to (B, H, N, N) – layout required for batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)   # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    # Permute to (B, N, N, H) because torch.layer_norm normalises the last dim
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()

    # LayerNorm over the hidden dimension H
    out_ln = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],   # FP32
        bias=w["to_out_norm.bias"],       # FP32
        eps=eps,
    )

    # Apply multiplicative gate
    out_ln_gate = out_ln * out_gate_perm   # FP16

    # Final linear projection: (R, H) -> (R, dim)
    out_ln_gate_flat = out_ln_gate.reshape(R, H)
    to_out_w_fp16 = w["to_out.weight"].to(torch.float16)      # [dim, H]
    final = F.linear(out_ln_gate_flat, to_out_w_fp16)        # (R, dim) FP16

    final = final.view(B, N, N, dim).to(torch.float32)       # cast back to FP32

    return final
