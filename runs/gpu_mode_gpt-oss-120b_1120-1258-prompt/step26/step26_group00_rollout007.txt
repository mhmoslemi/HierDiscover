@triton.jit
def norm_gate_kernel(
    out_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,    # same shape, FP16
    out_norm_gate_ptr, # [R, H] FP16 output buffer
    ln_w_ptr,        # [H] FP32
    ln_b_ptr,        # [H] FP32
    B, H, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):

Implementation:

pid = tl.program_id(0)
row_start = pid * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)  # row indices
mask_m = offs_m < total_rows   where total_rows = B * N * N

nn = N * N
b = offs_m // nn
rem = offs_m - b * nn
i = rem // N
j = rem - i * N

hidden_stride = N * N
base_offset = b * (H * hidden_stride) + i * N + j

# First pass: compute sum and sum_sq
sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

for k in range(0, H, BLOCK_K):
    offs_k = k + tl.arange(0, BLOCK_K)
    mask_k = offs_k < H

    # load out values
    ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
    vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)  # shape [M,K]

    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals*vals, axis=1)

mean = sum_val / H
var = sum_sq / H - mean * mean
rstd = tl.rsqrt(var + eps)

# Second pass: compute normalized gated and store
for k in range(0, H, BLOCK_K):
    offs_k = k + tl.arange(0, BLOCK_K)
    mask_k = offs_k < H

    # load out and gate values
    ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
    out_vals = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

    ptr_gate = out_gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
    gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

    # load gamma, beta
    gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # FP32
    beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

    # compute norm
    # ((out - mu) * rstd) * gamma + beta
    # out - mu: need mu broadcast across K, shape [M,1]
    diff = (out_vals - mean[:, None]) * rstd[:, None]  # FP32
    norm = diff * gamma[None, :] + beta[None, :]       # FP32

    # gating
    gated = norm * gate_vals
    # Cast to fp16 for storage
    gated_fp16 = gated.to(tl.float16)

    # destination pointer: out_norm_gate_ptr is shape [R, H]
    # offset = offs_m[:, None] * H + offs_k[None, :]
    dst_ptr = out_norm_gate_ptr + offs_m[:, None] * H + offs_k[None, :]
    tl.store(dst_ptr, gated_fp16, mask=mask_m[:, None] & mask_k[None, :])

"""
Highly‑optimized Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension of the input pair tensor
    (FP32 → FP16).  Implemented in `layernorm_kernel`.

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single GEMM:  out = x_norm @ W_fused.T   (FP16).

3️⃣  Split the fused projection, apply the optional mask, compute the three
    sigmoid gates and write `left`, `right` and `out_gate` in layout
    [B, H, N, N] (FP16).  Implemented in `split_fuse_gate_kernel`.

4️⃣  Core “einsum” as a batched matrix multiplication:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    performed with `torch.bmm` (cuBLAS).

5️⃣  Fuse the second LayerNorm, the element‑wise `out_gate` multiplication and
    the final projection (`to_out`) into a single Triton kernel
    (`norm_gate_kernel`).  It computes, per pair (b,i,j), the mean/variance
    over the hidden dimension H, applies the learned LN parameters,
    multiplies by `out_gate` and writes the result to a temporary buffer
    `norm_gate` of shape [R, H] (R = B·N·N).

6️⃣  Perform the final projection with a regular cuBLAS GEMM
    (FP16 → FP16) via `torch.nn.functional.linear` and cast the result back
    to FP32.

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision
(FP16 for the bulk of the work, FP32 for reductions) to obtain high
throughput while preserving numerical quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # [R, dim] FP32 input (contiguous)
    y_ptr,                # [R, dim] FP16 output
    w_ptr,                # [dim]   LN weight (γ) FP32
    b_ptr,                # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # FP32

        sum_val += tl.sum(vals, axis=1)      # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values -------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

# --------------------------------------------------------------
# 2️⃣ Split fused projection & apply mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]   FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # Base offset for the (b,i,k) location with hidden‑dim = 0
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# --------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate multiplication and write a
#     [R, H] buffer ready for the final projection.
# --------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16 (contracted tensor)
    out_gate_ptr,     # [B, H, N, N] FP16 (out_gate)
    out_norm_gate_ptr,# [R, H]      FP16 (output of this kernel)
    ln_w_ptr,         # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32   (to_out_norm.bias)
    B, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < (B * N * N)

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                         # stride for hidden dimension
    base_offset = b * (H * hidden_stride) + i * N + j   # offset for h = 0

    # -------- first pass: mean / variance over H ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)               # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: LN + gating ----------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        # load raw out and gate values
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        ptr_gate = out_gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # LN parameters (γ, β) for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # FP32

        # (out - μ) * rstd → *γ + β → * out_gate
        diff = (out_vals - mean[:, None]) * rstd[:, None]           # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]
        gated = norm * gate_vals                                    # [M, K]

        # store as FP16 to the flat [R, H] buffer
        dst_ptr = out_norm_gate_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(
            dst_ptr,
            gated.to(tl.float16),
            mask=mask_m[:, None] & mask_k[None, :],
        )

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following float32 keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------
    # Unpack
    # --------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise positions
    H = hidden_dim

    # --------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)      # [R, dim] FP16

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # --------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # [5·H, dim] FP16
    proj = F.linear(x_norm, fused_w)              # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------
    # 3️⃣ Split, mask and sigmoid → left/right/out_gate
    # --------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------
    # left/right: [B, H, N, N] → view as [B*H, N, N]
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate multiplication
    # --------------------------------------------------
    # Buffer that will hold (LN(out) * out_gate) for every (b,i,j) pair:
    # shape [R, H]  (R = B·N·N)
    norm_gate_buf = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_NG = 128   # rows per program
    BLOCK_K_NG = 32    # hidden‑dim chunk (H ≤ 128)
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    norm_gate_kernel[grid_ng](
        out,
        out_gate,
        norm_gate_buf,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        B, H, N,
        eps,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_K=BLOCK_K_NG,
        num_warps=8,
    )
    del out, out_gate

    # --------------------------------------------------
    # 6️⃣ Final projection (FP16 matmul → FP32 output)
    # --------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H] FP16
    # (R, H) @ (H, dim)ᵀ  → (R, dim)
    final_flat = torch.nn.functional.linear(
        norm_gate_buf,            # (R, H) FP16
        w_out_fp16,               # (dim, H) FP16
        bias=None,
    )                                 # (R, dim) FP16
    final = final_flat.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim]

    return final
```

This implementation:

* Uses three Triton kernels (layer‑norm, split‑gate, fused LN + gate) instead of four,
  reducing kernel launch overhead.
* Leverages cuBLAS for both the heavyweight contraction (`torch.bmm`) **and**
  the final projection (`torch.nn.functional.linear`), giving the best possible
  throughput on an H100.
* Keeps all heavy‑compute loops in FP16 while performing reductions in FP32,
  preserving the numerical quality of the reference PyTorch version.
