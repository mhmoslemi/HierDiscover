"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Input LayerNorm (torch.nn.functional.layer_norm).  
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single FP16 GEMM (`torch.nn.functional.linear`).  
3️⃣  A tiny Triton kernel (`split_fuse_gate_kernel`) splits the fused projection,
    applies the optional pair mask, and evaluates the three sigmoid gates.
4️⃣  Batched GEMM (`torch.bmm`) implements the core einsum  
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k] .
5️⃣  Second LayerNorm, out‑gate multiplication and the final projection are
    performed with native PyTorch ops (all in FP16, final result cast to FP32).

Only the split‑&‑gate step is written in Triton – the rest uses highly‑optimized
cuBLAS / Torch kernels, which on an H100 brings the forward pass well below the
target latency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base points to the start of the (b,i,k) element of the H‑plane at h=0
    base = b * (H * hidden_stride) + i * N + k                       # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # each block loads the five contiguous slices belonging to the same hidden idx
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    eps = 1e-5
    device = x.device

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))

    R = B * N * N                       # number of pairwise tokens

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5·H, dim]

    x_flat = x_norm.reshape(R, dim)     # [R, dim] FP16
    proj = F.linear(x_flat, fused_w)    # [R, 5·H] FP16
    del x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if not nomask and mask is not None:
        # mask broadcasted over hidden dim, stored as FP16
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core TriMul contraction (batched GEMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * hidden_dim, N, N)    # [B·H, N, N] FP16
    right_mat = right.view(B * hidden_dim, N, N)   # [B·H, N, N] FP16

    # out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, hidden_dim, N, N)                    # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (all FP16)
    # ------------------------------------------------------------------
    # bring hidden dim to the last axis so that torch.layer_norm can act on it
    out_perm = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    out_norm = F.layer_norm(
        out_perm,
        (hidden_dim,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                                                            # FP16

    # gating
    out_gated = out_norm * gate_perm                               # FP16

    # final linear projection to original dimension
    out_flat = out_gated.view(R, hidden_dim)                       # [R, H] FP16
    w_out = w["to_out.weight"].to(torch.float16)                   # [dim, H]
    final_flat = F.linear(out_flat, w_out)                         # [R, dim] FP16

    final = final_flat.view(B, N, N, dim).float()                  # back to FP32
    return final
