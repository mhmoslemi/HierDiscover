"""
Highly‑optimized forward‑only “outgoing” TriMul implemented with Triton + PyTorch.

Algorithm
---------
1️⃣  Input LayerNorm (float32 → float16) – we use the native cuDNN layer‑norm which
    is far faster than a custom Triton version for the large number of rows
    (R = B·N·N).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single matrix multiplication (FP16 GEMM).  The result has shape
    (R, 5·H) where H = hidden_dim.

3️⃣  Split the fused projection, apply the optional mask, compute the three
    sigmoids and store the three tensors
        left  [B, H, N, N] (FP16)
        right [B, H, N, N] (FP16)
        out_gate [B, H, N, N] (FP16)
    with a single Triton kernel (≈400 µs for the biggest configs).

4️⃣  Batched GEMM over the hidden dimension:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    performed with `torch.bmm` (Tensor‑core accelerated).

5️⃣  Second LayerNorm over the hidden dimension (to_out_norm) – again cuDNN
    `F.layer_norm`.

6️⃣  Gate the normalized tensor with `out_gate` and apply the final linear
    projection `to_out` (FP16 GEMM).  The result is cast back to float‑32.

Only step 3 is written in Triton, all heavy linear algebra is delegated to
cuBLAS/cuDNN which gives us the target < 500 µs runtime on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]    FP16  (fused linear output)
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat row index → (b, i, j)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Strides for the destination tensors
    hidden_stride = N * N                         # distance between two consecutive H‑planes
    # start offset for h = 0
    base_off = b * (H * hidden_stride) + i * N + j          # [BLOCK_R]

    # Loop over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # linear offsets inside the fused projection (row major)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        # Load the 5 concatenated projections
        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates (out_gate is stored unchanged)
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradients) using Triton for the split &
    gate step and cuBLAS/cuDNN for all linear algebra.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 -> float16)
    # --------------------------------------------------------------
    # cuDNN LayerNorm (fast) – keep everything float32, cast back to half
    norm_weight_fp32 = w["norm.weight"]
    norm_bias_fp32   = w["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight_fp32,
                          bias=norm_bias_fp32, eps=eps)
    x_norm = x_norm.to(torch.float16).contiguous()      # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim] FP16

    x_flat = x_norm.view(R, dim)               # [R, dim] FP16
    proj = F.linear(x_flat, fused_w)           # [R, 5·H] FP16
    del x_flat, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden_dim H)
    # --------------------------------------------------------------
    # Move hidden dim to the last axis to match LayerNorm API
    out_perm = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H] FP16
    to_out_norm_w = w["to_out_norm.weight"]
    to_out_norm_b = w["to_out_norm.bias"]
    out_norm = F.layer_norm(out_perm, (H,), weight=to_out_norm_w,
                            bias=to_out_norm_b, eps=eps)  # FP16

    # --------------------------------------------------------------
    # 6️⃣ Gate with out_gate and final projection
    # --------------------------------------------------------------
    # Align out_gate to same layout as out_norm
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    gated = out_norm * out_gate_perm                           # FP16

    # Final linear projection to original dim
    gated_flat = gated.view(-1, H)                              # [R, H] FP16
    to_out_w = w["to_out.weight"].to(torch.float16)            # [dim, H] FP16
    final = F.linear(gated_flat, to_out_w)                     # [R, dim] FP16
    final = final.to(torch.float32).view(B, N, N, dim)         # back to float32

    return final
