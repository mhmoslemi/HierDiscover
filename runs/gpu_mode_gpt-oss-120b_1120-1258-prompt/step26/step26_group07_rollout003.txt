Outgoing TriMul (AlphaFold3) forward pass – highly‑optimized implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair representation using torch.nn.functional
   (cuDNN implementation).  Output is cast to FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single FP16 GEMM with a concatenated weight matrix.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * splits the fused projection into the three tensors `left`, `right`,
     `out_gate` (layout [B, H, N, N]),
   * optionally applies the pair mask,
   * computes sigmoid gates in FP32,
   * writes the results in FP16.

4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
      out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]

5️⃣ Row‑wise mean and rsqrt‑variance of `out` are obtained with fast Torch
   reductions (mean/var over the hidden dimension H).

6️⃣ Second LayerNorm, multiplication with `out_gate` and the final linear
   projection (`to_out`) are performed with fused PyTorch operations,
   producing the final `[B, N, N, dim]` tensor in float‑32.

):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for a 2‑D view (H, N, N) within the 4‑D tensor
    hidden_stride = N * N
    # base offset of the first hidden plane for (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of fused projection:
        #   [left | right | left_gate | right_gate | out_gate] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Store into the 4‑D tensors. Each hidden plane is spaced by `hidden_stride`.
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point: outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, 5 linear maps have no bias)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (use cuDNN via torch.nn.functional)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim)          # [R, dim] float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                   # FP32
    x_norm = x_norm_fp32.to(torch.float16)  # cast to FP16 for the next GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM (cuBLAS, FP16)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim] FP16
    proj = F.linear(x_norm, fused_weight)  # [R, 5·H] FP16
    del fused_weight, x_norm           # free memory

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, and sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128      # rows per program (tuned for N up to 1024)
    BLOCK_H = 128      # hidden‑dim tiles
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: out = left @ rightᵀ   (core einsum)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) and use torch.bmm
    left_mat  = left.view(B * H, N, N)                  # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                 # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (torch reductions)
    # --------------------------------------------------------------
    # reshape to [R, H] for per‑row statistics
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # FP16
    mu   = out_flat.mean(dim=1)                                   # (R,) FP32
    var  = out_flat.var(dim=1, unbiased=False)                    # (R,) FP32
    rstd = torch.rsqrt(var + eps)                                 # (R,) FP32

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (all in torch)
    # --------------------------------------------------------------
    # Flatten out_gate for broadcasting
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP16

    # Cast `out` to FP32 for the LN computation
    out_fp32 = out_flat.to(torch.float32)

    # (out - μ) * rstd   →   γ· + β   →   * out_gate
    diff = (out_fp32 - mu.unsqueeze(1)) * rstd.unsqueeze(1)   # (R, H) FP32
    ln_w = w["to_out_norm.weight"]                            # (H,) FP32
    ln_b = w["to_out_norm.bias"]                              # (H,) FP32
    normed = diff * ln_w + ln_b                               # (R, H) FP32
    term = normed * out_gate_flat.to(torch.float32)           # (R, H) FP32

    # Final linear projection: weight shape [dim, hidden_dim] (FP32)
    final_flat = F.linear(term, w["to_out.weight"])          # (R, dim) FP32

    # Reshape to the original 4‑D layout
    final = final_flat.view(B, N, N, dim)                     # [B, N, N, dim] FP32

    return final
