"""
Outgoing TriMul (AlphaFold3) – fused implementation.

Algorithm
---------
0️⃣  First LayerNorm (FP32) – use native torch.layer_norm (cuDNN) for speed.
1️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → tensor of shape (R, 5·H) in FP16.
2️⃣  Split the fused projection, apply the optional mask, compute the three
    sigmoid gates and store the tensors `left`, `right`, `out_gate`
    in layout [B, H, N, N] (FP16) via a Triton kernel.
3️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).
4️⃣  A single Triton kernel performs **both** the second LayerNorm
    (row‑wise over H) and the final projection in one pass:
       * accumulate Σ out    and Σ out²   → μ, rstd
       * accumulate three dot‑products per hidden‑tile:
           s1 = Σ (out·γ·gate)·W
           s2 = Σ (γ·gate)·W
           s3 = Σ (β·gate)·W
       * combine them:  out = rstd·s1 – μ·rstd·s2 + s3
    The kernel writes the final `[B, N, N, dim]` tensor (FP32).

All heavy work is done in FP16 (tensor cores) while statistics are kept in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # offset of the first hidden plane for (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view into the fused projection:
        #   layout = [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # store into [B, H, N, N]
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_out_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulators for the three dot‑products
    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·γ·gate·W
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ γ·gate·W
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β·gate·W

    inv_H = 1.0 / H

    # loop over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # ----- load out and gate -----
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_tile = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [M, K]

        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_tile = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [M, K]

        # statistics for mu / rstd
        sum_out += tl.sum(out_tile, axis=1)                 # [M]
        sum_out_sq += tl.sum(out_tile * out_tile, axis=1)   # [M]

        # ----- load LayerNorm parameters -----
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # a = γ·gate ,  b = β·gate   (broadcast gamma/beta over rows)
        a_tile = gate_tile * gamma[None, :]   # [M, K]
        b_tile = gate_tile * beta[None, :]    # [M, K]

        # ----- load final‑projection weight -----
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, N] FP16

        # three dot‑products (FP16·FP16 → FP32)
        term1 = tl.dot((out_tile * a_tile).to(tl.float16), w_tile)   # Σ out·γ·gate·W
        term2 = tl.dot(a_tile.to(tl.float16), w_tile)                # Σ γ·gate·W
        term3 = tl.dot(b_tile.to(tl.float16), w_tile)                # Σ β·gate·W

        acc1 += term1
        acc2 += term2
        acc3 += term3

    # ----- compute mu, rstd -----
    mu   = sum_out * inv_H                           # [M]
    var  = sum_out_sq * inv_H - mu * mu               # [M]
    rstd = tl.rsqrt(var + eps)                        # [M]

    # ----- combine the three accumulated terms -----
    mu_ex   = mu[:, None]          # [M,1]
    rstd_ex = rstd[:, None]        # [M,1]

    out = rstd_ex * acc1 - mu_ex * rstd_ex * acc2 + acc3   # [M,N] FP32

    # ----- write final tensor -----
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, out, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point ---------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all tensors are float32; the five linear maps have no bias)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total number of (b,i,j) positions

    # --------------------------------------------------------------
    # 0️⃣ First LayerNorm – use cuDNN implementation (fast)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim] FP32

    # flatten and cast to FP16 for the big GEMM
    x_flat = x_norm.reshape(R, dim).contiguous().to(torch.float16)   # [R, dim] FP16

    # --------------------------------------------------------------
    # 1️⃣ Fuse the five linear maps into a single GEMM (cuBLAS)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_flat, fused_w)            # (R, 5·H) FP16
    del fused_w, x_flat, x_norm                  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Split, mask & sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 128                     # hidden dim ≤ 128 → one iteration per row
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM (cublas) for the core einsum
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) and perform bmm left @ rightᵀ
    left_mat  = left.reshape(B * hidden_dim, N, N)                # [B·H, N, N] FP16
    right_mat = right.reshape(B * hidden_dim, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))         # [B·H, N, N] FP16
    out = out.view(B, hidden_dim, N, N).contiguous()             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Final LayerNorm + out_gate + projection (single kernel)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows per program
    BLOCK_K_OUT = 32   # hidden‑dim tile
    BLOCK_N_OUT = 128  # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        hidden_dim,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # clean‑up
    del left, right, out_gate, out, w_out_T

    return final
