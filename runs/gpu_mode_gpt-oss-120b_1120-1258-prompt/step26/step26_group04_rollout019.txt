def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Compute base offsets for left and right storage
    hidden_stride = nn
    base_offset_left = b * (H * hidden_stride) + i * N + k
    base_offset_right = b * (H * hidden_stride) + k * N + i

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid (in fp32)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask if present
        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
            mask_val = mask_val[:, None]  # broadcast over H
        else:
            mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        # out_gate does not depend on mask

        # Store
        dst_offset_left = base_offset_left[:, None] + offs_h[None, :] * hidden_stride
        dst_offset_right = base_offset_right[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset_left,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset_right,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + base_offset_left[:, None] + offs_h[None, :] * hidden_stride,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the flattened pair tensor (FP32 → FP16) using
    torch.nn.functional.layer_norm (cuDNN).
2️⃣ Fuse the 5 linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16) → tensor `[R, 5·H]`.
3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional mask, computes the 3 sigmoid gates and stores
    `left`, `right` (already transposed for the later batched GEMM) and
    `out_gate` in layout `[B, H, N, N]` (FP16).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum
       out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,k,j,h]
    yielding `[B, H, N, N]` (FP16) – no transpose is needed because
    `right` was stored as `[k,i]`.
5️⃣ Permute to bring the hidden dimension to the innermost position
    → shape `(B, N, N, H)` (contiguous).  Run a second LayerNorm
    (`to_out_norm`), multiply by the output gate and apply the final
    linear projection (`to_out`) – all with native torch kernels.
6️⃣ Return the result in FP32.

Only the split/fuse/gate step needs a custom kernel; everything else
uses highly‑optimized torch/cuBLAS/cuDNN kernels.  The implementation
covers both masked / un‑masked inputs and works for the full range of
test configurations from the statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16 (row‑major layout)
    right_ptr,      # [B, H, N, N] FP16 (stored transposed → [k,i])
    out_gate_ptr,   # [B, H, N, N] FP16 (same layout as left)
    mask_ptr,       # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = nn                   # stride between hidden planes
    # Offsets for the three output tensors
    # left  : (b, h, i, k)   → base = b·(H·nn) + i·N + k
    # right : (b, h, k, i)   → base = b·(H·nn) + k·N + i   (transposed)
    base_left  = b * (H * hidden_stride) + i * N + k
    base_right = b * (H * hidden_stride) + k * N + i
    base_gate  = base_left                               # out_gate has same layout as left

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # linear view: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32, cast back to FP16 for storage
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # optional mask (broadcast over H)
        if HAS_MASK:
            m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
            mask_val = m[:, None]                                    # [BLOCK_R, 1]
        else:
            mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets (including hidden‑plane stride)
        dst_left  = base_left[:, None]  + offs_h[None, :] * hidden_stride
        dst_right = base_right[:, None] + offs_h[None, :] * hidden_stride
        dst_gate  = base_gate[:, None]  + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_left,   left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_right, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_gate, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (FP32 → FP16)
    # --------------------------------------------------------------
    # reshape to [R, dim] and apply cuDNN layer‑norm
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim) in FP16
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid → left, right (transposed) & out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask → FP16 (0/1) and flatten to [R]
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)     # [B·H, N, N]   (i,k)
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]   (k,i) – already transposed
    out = torch.bmm(left_mat, right_mat)  # [B·H, N, N]   = Σₖ left[i,k]·right[k,j]
    out = out.view(B, H, N, N)            # [B, H, N, N] (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (all torch)
    # --------------------------------------------------------------
    # bring hidden dimension to the innermost axis for a fast cuDNN LN
    out_perm      = out.permute(0, 2, 3, 1).contiguous()      # (B, N, N, H)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # (B, N, N, H)

    # hidden‑dim LayerNorm
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                       # FP16

    # apply output gate
    out_gated = out_norm * out_gate_perm                     # FP16

    # final linear projection: hidden → dim
    to_out_w_fp16 = w["to_out.weight"].to(torch.float16)     # [dim, H]
    final_fp16 = F.linear(out_gated, to_out_w_fp16)          # (B, N, N, dim) FP16

    # cast to FP32 as required by the interface
    final = final_fp16.to(torch.float32)

    return final
```
