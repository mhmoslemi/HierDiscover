def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1 RowNorm
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[grid_ln] ...

    # 2 Fused projection and gating
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Convert weights to FP16 contiguous
    left_w = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w = w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w = w["out_gate.weight"].to(torch.float16).contiguous()

    # mask handling
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask_val = 1
        mask_ptr = mask_flat
    else:
        has_mask_val = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask_val,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm

    # 3 Batched GEMM (cuBLAS)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # 4 Final fused output kernel (statistics + projection)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,

Inside:

    offs_h = h_start + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Initialize accumulators
    left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc = tl.zeros([...], dtype=tl.float32)
    left_gate_acc = tl.zeros([...], dtype=tl.float32)
    right_gate_acc = tl.zeros([...], dtype=tl.float32)
    out_gate_acc = tl.zeros([...], dtype=tl.float32)

        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load x slice
        x_tile = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # shape [BLOCK_R, BLOCK_D]

        # Load weight slices: for each weight matrix
        # left weight:
        left_w_tile = tl.load(
            left_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        # similarly for right_w_tile, left_gate_w_tile, right_gate_w_tile, out_gate_w_tile

        # Perform dot for each
        left_acc += tl.dot(x_tile, left_w_tile)
        right_acc += tl.dot(x_tile, right_w_tile)
        left_gate_acc += tl.dot(x_tile, left_gate_w_tile)
        right_gate_acc += tl.dot(x_tile, right_gate_w_tile)
        out_gate_acc += tl.dot(x_tile, out_gate_w_tile)

    # After dim loop, we have raw outputs for this hidden block.

    # Compute sigmoid gates for gate (left_gate, right_gate, out_gate)
    # Use FP32 for stability:
    left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)

    # Apply mask to left/right after gating:
    if HAS_MASK:

        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # fp16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    left = left_acc.to(tl.float16) * mask_val * left_gate
    right = right_acc.to(tl.float16) * mask_val * right_gate
    out_gate = out_gate_val

    # Store to output layout
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R] start index for hidden 0

    dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

    tl.store(left_ptr + dst_offset, left, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset, out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – forward only, fused kernels.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the pair representation.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   with the mask and sigmoid gates in a single Triton kernel.  No intermediate
   [R,5·H] tensor is materialised – the result is written directly into
   `left`, `right` and `out_gate` tensors of shape [B, H, N, N] (FP16).
3️⃣ Batched GEMM (cuBLAS) computes the core contraction
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).
4️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection (to_out) into a single Triton kernel.  While iterating over the
   hidden dimension the kernel accumulates the required statistics
   (mean/variance) and the three necessary dot‑products, producing the final
   `[B, N, N, dim]` tensor (FP32) without any extra passes over the data.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim] FP32 input
    y_ptr,       # [R, dim] FP16 output
    w_ptr,       # [dim]    LN weight (γ) FP32
    b_ptr,       # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)         # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
#    (inputs: x_norm, weights transposed to [dim, hidden])
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                 # [R, dim]            FP16
    left_w_ptr,            # [dim, H]            FP16
    right_w_ptr,           # [dim, H]            FP16
    left_gate_w_ptr,       # [dim, H]            FP16
    right_gate_w_ptr,      # [dim, H]            FP16
    out_gate_w_ptr,        # [dim, H]            FP16
    mask_ptr,              # [R]                 FP16  (optional)
    left_ptr,              # [B, H, N, N]        FP16
    right_ptr,             # [B, H, N, N]        FP16
    out_gate_ptr,          # [B, H, N, N]        FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden=0 for each row

    # ------------------------------------------------------------------
    # Iterate over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # initialise accumulators for this tile
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the feature dimension dim
        # ------------------------------------------------------------------
        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim

            # Load a tile of the normalized input
            x_tile = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                               # [BLOCK_R, BLOCK_D]

            # Load corresponding weight tiles (transposed layout: [dim, H])
            left_w_tile = tl.load(
                left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                               # [BLOCK_D, BLOCK_H]

            right_w_tile = tl.load(
                right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            left_gate_w_tile = tl.load(
                left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            right_gate_w_tile = tl.load(
                right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            out_gate_w_tile = tl.load(
                out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # Accumulate dot‑products for each projection / gate
            left_acc      += tl.dot(x_tile,      left_w_tile)        # [BLOCK_R, BLOCK_H]
            right_acc     += tl.dot(x_tile,      right_w_tile)
            left_gate_acc += tl.dot(x_tile,      left_gate_w_tile)
            right_gate_acc+= tl.dot(x_tile,      right_gate_w_tile)
            out_gate_acc  += tl.dot(x_tile,      out_gate_w_tile)

        # --------------------------------------------------------------
        # Apply sigmoid gates (FP32 → FP16) and mask
        # --------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        left   = left_acc.to(tl.float16) * mask_val * left_gate
        right  = right_acc.to(tl.float16) * mask_val * right_gate
        # out_gate already contains the sigmoid output (no mask)

        # --------------------------------------------------------------
        # Write results in layout [B, H, N, N] (row‑major, last dim stride 1)
        # --------------------------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate & final projection (incl. stats)
# ----------------------------------------------------------------------
@triton.jit
def final_output_fused_kernel(
    out_ptr,            # [B, H, N, N] FP16
    gate_ptr,           # [B, H, N, N] FP16   (out_gate)
    ln_w_ptr,           # [H] FP32             (to_out_norm.weight)
    ln_b_ptr,           # [H] FP32             (to_out_norm.bias)
    w_out_T_ptr,        # [H, dim] FP16 (to_out.weight transposed)
    out_final_ptr,      # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulators for the three dot‑products described in the derivation
    acc_SA = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out_h * (γ_h·g_h·W_h)
    acc_SB = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (γ_h·g_h·W_h)
    acc_SC = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (β_h·g_h·W_h)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [K]
        mask_k = offs_k < H

        # ----------- load out & gate ---------------------------------
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)      # [M, K]
        vals_gate = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)      # [M, K]

        # stats
        sum_val += tl.sum(vals_out, axis=1)                 # [M]
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)      # [M]

        # ----------- load LayerNorm params ----------------------------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # scale = γ_h * g_h   (row‑wise)
        scale      = vals_gate * gamma[None, :]           # [M, K]
        scale_beta = vals_gate * beta[None, :]            # [M, K]

        # ----------- load projection weight slice ----------------------
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)                     # [K, N]

        # ---- accumulate the three required dot‑products -------------
        # Σ out_h * (γ_h·g_h·W_h)
        acc_SA += tl.dot(vals_out * scale, w_tile)      # (M,K)·(K,N) → (M,N)
        # Σ (γ_h·g_h·W_h)
        acc_SB += tl.dot(scale, w_tile)
        # Σ (β_h·g_h·W_h)
        acc_SC += tl.dot(scale_beta, w_tile)

    # ---------- compute mean / rsqrt‑variance ----------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ---------- final expression ----------
    # out = rstd * (SA - mu * SB) + SC
    out = rstd[:, None] * (acc_SA - mu[:, None] * acc_SB) + acc_SC

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, out, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + mask + sigmoid gates (no large intermediate)
    # ------------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # transpose weight matrices to [dim, H] for better memory access pattern
    left_w_T      = w["left_proj.weight"].to(torch.float16).t().contiguous()
    right_w_T     = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w_T = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w_T= w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w_T  = w["out_gate.weight"].to(torch.float16).t().contiguous()

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_w_T, right_w_T, left_gate_w_T, right_gate_w_T, out_gate_w_T

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 32   # rows per program (tiles over R)
    BLOCK_K_OUT = 32   # hidden‑dim chunk
    BLOCK_N_OUT = 64   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # gamma FP32
        w["to_out_norm.bias"],     # beta  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
