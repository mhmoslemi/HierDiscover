    shared_gate = tl.shared([BLOCK_M * H], dtype=tl.float16)

    # Accumulate sum and sumsq for mu/rstd
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # First pass: load out & gate into shared memory, compute sum/sum_sq
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # global memory offsets: each row has stride H
        ptr_out = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_k[None, :]

        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # compute sum & sumsq in fp32
        vals_out_f32 = vals_out.to(tl.float32)
        sum_val += tl.sum(vals_out_f32, axis=1)
        sum_sq += tl.sum(vals_out_f32 * vals_out_f32, axis=1)

        # store into shared memory
        # compute linear offset for shared memory: row_idx * H + col_idx
        shared_idx = offs_m[:, None] * H + (k_start + offs_k)[None, :]
        tl.store(shared_out + shared_idx, vals_out, mask=mask_m[:, None] & mask_k[None, :])
        tl.store(shared_gate + shared_idx, vals_gate, mask=mask_m[:, None] & mask_k[None, :])

    # Compute per-row mean and rsqrt variance
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute normalized term and dot with projection weight
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load from shared memory
        shared_idx = offs_m[:, None] * H + (k_start + offs_k)[None, :]
        vals_out = tl.load(shared_out + shared_idx, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        vals_gate = tl.load(shared_gate + shared_idx, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load LN parameters for hidden slice
        gamma = tl.load(ln_w_ptr + (k_start + offs_k), mask=mask_k, other=0.0)   # [BLOCK_K]
        beta = tl.load(ln_b_ptr + (k_start + offs_k), mask=mask_k, other=0.0)    # [BLOCK_K]

        # normalized term: ((out - mu) * rstd) * gamma + beta
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M,K]
        term = norm * vals_gate                                        # [M,K]

        # Cast to fp16 for tensor core matmul
        term_fp16 = term.to(tl.float16)

        # Load weight slice
        w_ptr = w_out_T_ptr + (k_start + offs_k)[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # [K, BLOCK_N]

        # Accumulate dot product
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final output: shape [R, dim]
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Layer‑Norm over the last dimension (FP32 → FP16) using torch.nn.functional.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → tensor (R, 5·H),  R = B·N·N.
3️⃣  Split the fused projection, apply optional mask, compute sigmoid gates,
    and store *left*, *right* and *out_gate* in layout [B, H, N, N] (FP16)
    via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core contraction:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   (FP16)
5️⃣  Fuse the second LayerNorm, the *out_gate* multiplication and the final
    projection into a *single* Triton kernel.  The kernel:
        • loads `out` and `out_gate` (shape [R, H]) once into shared memory,
        • computes per‑row mean / rsqrt‑variance,
        • normalises, applies LN parameters and *out_gate*,
        • performs a fused FP16⊙FP16 → FP32 dot with the final weight
          (H × dim) to produce the result `[B, N, N, dim]` (FP32).
    This eliminates a separate stats kernel and halves global‑memory traffic.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]      FP16 fused projection
    left_ptr,          # [B, H, N, N]  FP16
    right_ptr,         # [B, H, N, N]  FP16
    out_gate_ptr,      # [B, H, N, N]  FP16
    mask_ptr,          # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # offset of the first hidden plane for (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout = [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Store into [B, H, N, N] layout.
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    out_ptr,            # [R, H]  FP16   (R = B·N·N)
    gate_ptr,           # [R, H]  FP16
    ln_w_ptr,           # [H]    FP32   (to_out_norm.weight)
    ln_b_ptr,           # [H]    FP32   (to_out_norm.bias)
    w_out_T_ptr,        # [H, dim]  FP16 (transposed to_out weight)
    out_final_ptr,      # [B, N, N, dim]  FP32
    total_rows, H, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (e.g., 64)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile (e.g., 32)
    BLOCK_N: tl.constexpr,   # output‑dim tile (e.g., 128)
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    row_idx   = tl.arange(0, BLOCK_M)          # [BLOCK_M] — local inside the block
    offs_m    = row_start + row_idx            # absolute row ids
    offs_n    = col_start + tl.arange(0, BLOCK_N)   # output‑dim ids

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # ------------------------------------------------------------------
    # Shared memory to hold a whole hidden vector for each row in the tile
    # ------------------------------------------------------------------
    # H is a compile‑time constant (passed as `tl.constexpr`) → static size.
    shared_out  = tl.shared([BLOCK_M * H], dtype=tl.float16)
    shared_gate = tl.shared([BLOCK_M * H], dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: load out & gate, compute mean / variance, store to shared
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Global loads (stride = H because we flattened [R, H])
        ptr_out  = out_ptr  + offs_m[:, None] * H + offs_k[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_k[None, :]

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float16)
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float16)

        # Accumulate sum & sum‑of‑squares (FP32)
        vals_out_f32 = vals_out.to(tl.float32)
        sum_val += tl.sum(vals_out_f32, axis=1)
        sum_sq  += tl.sum(vals_out_f32 * vals_out_f32, axis=1)

        # Store to shared memory (linear offset = row_idx * H + hidden_idx)
        shared_idx = row_idx[:, None] * H + (k_start + offs_k)[None, :]
        tl.store(shared_out + shared_idx,  vals_out,
                 mask=mask_m[:, None] & mask_k[None, :])
        tl.store(shared_gate + shared_idx, vals_gate,
                 mask=mask_m[:, None] & mask_k[None, :])

    # Compute per‑row mean and rsqrt‑variance
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise, apply out_gate, LayerNorm params and
    #            dot‑product with the final weight matrix.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load from shared memory (now cheap)
        shared_idx = row_idx[:, None] * H + (k_start + offs_k)[None, :]
        vals_out  = tl.load(shared_out + shared_idx,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(shared_gate + shared_idx,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + (k_start + offs_k), mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + (k_start + offs_k), mask=mask_k, other=0.0)   # [K] FP32

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                 # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                    # [M, K]
        term = norm * vals_gate                                          # [M, K]

        # Cast to FP16 for Tensor‑Core GEMM
        term_fp16 = term.to(tl.float16)

        # Load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + (k_start + offs_k)[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final output tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, no bias in the 5 linear maps)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16) using Torch (highly‑optimized)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM (cuBLAS)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    # free unused tensors early
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden planes per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) and perform bmm left @ rightᵀ
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    # No longer need left/right
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate & final projection
    # --------------------------------------------------------------
    # Permute to [B, N, N, H] so that the hidden dimension becomes
    # the innermost (contiguous) axis → easier for Triton.
    out      = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    out_flat      = out.view(R, H)        # [R, H]  FP16
    out_gate_flat = out_gate.view(R, H)   # [R, H]  FP16

    # to_out.weight is [dim, hidden_dim]; we need its transpose (H, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M = 64      # rows per program (tune for occupancy)
    BLOCK_K = 32      # hidden‑dim tile (must divide H ≤ 128)
    BLOCK_N = 128     # output‑dim tile
    grid_out = (
        triton.cdiv(R, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )
    final_norm_proj_kernel[grid_out](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],   # gamma (FP32)
        w["to_out_norm.bias"],     # beta  (FP32)
        w_out_T,
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )
    # free temporary buffers
    del out, out_gate, out_flat, out_gate_flat, w_out_T

    return final
```
