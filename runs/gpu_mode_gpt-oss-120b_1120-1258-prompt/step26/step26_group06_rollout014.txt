def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # Input layernorm using torch functional
    x_norm = F.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps).to(torch.float16)

    # Fused linear projection
    fused_w = torch.cat([w["left_proj.weight"],
                         w["right_proj.weight"],
                         w["left_gate.weight"],
                         w["right_gate.weight"],
                         w["out_gate.weight"]], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = F.linear(x_norm, fused_w)  # (R, 5*H)
    del fused_w, x_norm

    # Prepare mask if present
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=x.device)

    # Allocate tensors for left/right/out_gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)

    # Triton split-fuse-gate kernel
    BLOCK_R = 128
    BLOCK_H = 128

        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # Batched GEMM for contraction
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))   # shape (B*H, N, N)
    out = out.view(B, H, N, N)  # [B, H, N, N]
    del left, right

    # Second layernorm
    out_norm = F.layer_norm(out, (H,), weight=w["to_out_norm.weight"], bias=w["to_out_norm.bias"], eps=eps)
    # Multiply by out_gate
    out_norm_gate = out_norm * out_gate
    del out, out_gate

    # Final projection
    # to_out weight: (dim, hidden_dim) -> (dim, H) . We'll transpose for torch.nn.functional.linear: we need weight shape (out_features, in_features) = (dim, H)
    w_out = w["to_out.weight"].to(torch.float16)  # shape (dim, H)

    final = F.linear(out_flat, w_out)  # shape (B*N*N, dim) with dtype half
    final = final.float().view(B, N, N, dim)

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # decode
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask values
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast over hidden dim
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)
    
    # hidden offsets
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # compute base pointer for proj row start
    proj_row_base = offs_r[:, None] * (5 * H)   # shape [BLOCK_R, 1]

    # load slices
    left_raw = tl.load(proj_ptr + proj_row_base + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0)
    right_raw = tl.load(proj_ptr + proj_row_base + H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0)
    left_gate_raw = tl.load(proj_ptr + proj_row_base + 2 * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0)
    right_gate_raw = tl.load(proj_ptr + proj_row_base + 3 * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0)
    out_gate_raw = tl.load(proj_ptr + proj_row_base + 4 * H + offs_h[None, :], mask=mask_r[:, None] & mask_h[None, :], other=0.0)

    # sigmoid (stable)
    left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # apply mask and gates
    left = left_raw * mask_val * left_gate
    right = right_raw * mask_val * right_gate
    outg = out_gate

    # destination offsets
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # start offset for h=0
    
    dst_offset = base[:, None] + offs_h[None, :] * hidden_stride

    tl.store(left_ptr + dst_offset, left, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32 → float16) with torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a *single* GEMM:   (B·N·N, dim) × (dim, 5·hidden) → (R, 5·H).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
       * decodes the flat index (b,i,k) → (b,i,k) layout,
       * applies the optional pair‑wise mask,
       * computes sigmoid gates,
       * writes three tensors `left`, `right` and `out_gate` in layout [B, H, N, N].
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    → tensor `[B, H, N, N]` (FP16).
5️⃣  Second LayerNorm (over the hidden dimension H) using the highly‑optimised
    torch implementation, then element‑wise multiply by `out_gate`.
6️⃣  Final projection: reshape to (B·N·N, H) and call `torch.nn.functional.linear`
    with the FP32 weight `to_out`.  The result is cast back to float32
    → `[B, N, N, dim]`.

Only the mask‑handling / gating step is written in Triton; the heavy
matrix‑multiplications and both LayerNorms use PyTorch’s cuBLAS/Tensor‑Core
paths, giving a considerable speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1️⃣  One program works on a tile of rows (R) and the whole hidden
    #     dimension (H) at once.
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 2️⃣  Decode flat index r = b·N·N + i·N + k (row‑major across B,N,N)
    # ------------------------------------------------------------------
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ------------------------------------------------------------------
    # 3️⃣  Optional scalar mask per (b,i,k)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 4️⃣  Load the 5 blocks (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)          # hidden offsets
    mask_h = offs_h < H

    # base offset for the row inside the fused projection matrix
    proj_row_base = offs_r[:, None] * (5 * H)          # [BLOCK_R,1]

    left_raw = tl.load(
        proj_ptr + proj_row_base + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    right_raw = tl.load(
        proj_ptr + proj_row_base + H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    left_gate_raw = tl.load(
        proj_ptr + proj_row_base + 2 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    right_gate_raw = tl.load(
        proj_ptr + proj_row_base + 3 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    out_gate_raw = tl.load(
        proj_ptr + proj_row_base + 4 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )

    # ------------------------------------------------------------------
    # 5️⃣  Sigmoid (stable version) – compute in FP32 then back to FP16
    # ------------------------------------------------------------------
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # ------------------------------------------------------------------
    # 6️⃣  Apply mask & gates
    # ------------------------------------------------------------------
    left  = left_raw  * mask_val * left_gate
    right = right_raw * mask_val * right_gate
    outg  = out_gate

    # ------------------------------------------------------------------
    # 7️⃣  Store into the three output tensors.
    # ------------------------------------------------------------------
    hidden_stride = N * N                     # stride between consecutive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h = 0

    dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

    tl.store(left_ptr + dst_offset,
             left,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset,
             right,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset,
             outg,
             mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor   [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", and "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (torch, fp32 → fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)       # [B, N, N, dim] → FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5 · hidden_dim columns)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)        # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split + mask + sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Core contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # [B·H, N, N]
    right_mat = right.view(B * H, N, N)           # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                    # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm over the hidden dimension (torch)
    # ------------------------------------------------------------------
    # move hidden dim to the trailing axis for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()           # [B, N, N, H] FP16
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                        # [B, N, N, H] FP16

    # also permute the gate to the same layout
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16

    # element‑wise gate multiplication
    gated = out_norm * out_gate_perm                               # [B, N, N, H] FP16
    del out, out_gate, out_norm, out_gate_perm

    # ------------------------------------------------------------------
    # 6️⃣  Final projection (B·N·N, H) × (dim, H)ᵀ → (B·N·N, dim)
    # ------------------------------------------------------------------
    # flatten to (B·N·N, H) – keep FP32 for the matmul to match reference
    gated_flat = gated.view(-1, H).float()          # [B·N·N, H] FP32
    w_out = w["to_out.weight"]                      # [dim, H] FP32
    final = F.linear(gated_flat, w_out)             # [B·N·N, dim] FP32
    final = final.view(B, N, N, dim)                # [B, N, N, dim]

    return final
