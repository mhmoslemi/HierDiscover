    left_out_ptr, right_out_ptr, out_gate_out_ptr,

    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

@triton.jit
def fused_proj_gate_kernel(
    x_ptr,  # [R, dim] FP16
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    mask_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode to b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_out = b * (H * hidden_stride) + i * N + k

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)
        mask_val = mask_val[:, None]  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float32)

    # loop over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for dot products
        left_proj_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_proj_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # iterate over dim
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_D]

            # load weight tiles (transposed, [dim, H])
            left_w = tl.load(left_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)  # [BLOCK_D, BLOCK_H]
            right_w = tl.load(right_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                                  mask=mask_d[:, None] & mask_h[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            out_gate_w = tl.load(out_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                                 mask=mask_d[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)

            left_proj_acc += tl.dot(x_vals, left_w)         # (BLOCK_R, BLOCK_H)
            right_proj_acc += tl.dot(x_vals, right_w)
            left_gate_acc += tl.dot(x_vals, left_gate_w)
            right_gate_acc += tl.dot(x_vals, right_gate_w)
            out_gate_acc += tl.dot(x_vals, out_gate_w)

        # after dim loop: compute gating and apply mask
        left_gate = tl.sigmoid(left_gate_acc).to(tl.float16)
        right_gate = tl.sigmoid(right_gate_acc).to(tl.float16)
        out_gate = tl.sigmoid(out_gate_acc).to(tl.float16)

        # mask broadcast multiply left and right proj
        left_proj = left_proj_acc.to(tl.float16) * mask_val * left_gate
        right_proj = right_proj_acc.to(tl.float16) * mask_val * right_gate

        # compute destination pointer offsets
        dst_off = base_out[:, None] + offs_h[None, :] * hidden_stride

        # store
        tl.store(left_out_ptr + dst_off,
                 left_proj,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + dst_off,
                 right_proj,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – one Triton kernel.
2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single kernel:
   * Input  : x_norm   [R, dim]   (R = B·N·N)
   * Weights: all 5·[dim, H] matrices are stored transposed (dim‑major) in
             FP16, where H = hidden_dim.
   * The kernel computes for every (b,i,k) and every hidden index h:
        left   = (x·W_left)   * mask * sigmoid(x·W_left_gate)
        right  = (x·W_right)  * mask * sigmoid(x·W_right_gate)
        out_g  = sigmoid(x·W_out_gate)
     and writes the three tensors directly in layout [B, H, N, N].
   This removes the huge (R,5·H) intermediate used by the reference
   implementation and cuts the memory traffic of the input tensor by ~5×.

3️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   (torch.bmm on tensors of shape (B·H, N, N)).

4️⃣ Two‑pass row‑wise mean/variance over the hidden dimension (H) – a
   Triton kernel writes μ and rstd for each (b,i,j).

5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   projection (hidden → dim) into a single kernel that streams over H in
   tiles and performs an FP16·FP16 → FP32 dot‑product with the output weight
   matrix.

All kernels target Triton 3.3.1 and are tuned for an NVIDIA H100.
The implementation works for B∈{1,2}, N≤1024, dim∈{128,384},
hidden_dim=128 and optional pair‑wise mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim] FP32 input (flattened)
    y_ptr,      # [R, dim] FP16 output
    w_ptr,      # [dim]   LayerNorm weight (γ) FP32
    b_ptr,      # [dim]   LayerNorm bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating (mask applied to left/right)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                     # [R, dim]               FP16
    left_w_ptr,                # [dim, H]               FP16 (left_proj^T)
    right_w_ptr,               # [dim, H]               FP16 (right_proj^T)
    left_gate_w_ptr,           # [dim, H]               FP16
    right_gate_w_ptr,          # [dim, H]               FP16
    out_gate_w_ptr,            # [dim, H]               FP16
    left_out_ptr,              # [B, H, N, N]           FP16 (flattened)
    right_out_ptr,             # [B, H, N, N]           FP16
    out_gate_out_ptr,          # [B, H, N, N]           FP16
    mask_ptr,                  # [R]                    FP16 (optional)
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]                     # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                                 # stride = N²
    base_off = b * (H * hidden_stride) + i * N + k       # start offset for h = 0

    # ------------------------------------------------------------------
    # Loop over hidden dimension tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for the five dot‑products (FP32 for precision)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # loop over the feature dimension
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load a tile of the normalized input
            x_tile = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                     # [BLOCK_R, BLOCK_D]

            # load corresponding weight tiles (transposed: [dim, H])
            left_w = tl.load(
                left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                     # [BLOCK_D, BLOCK_H]
            right_w = tl.load(
                right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            left_gate_w = tl.load(
                left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            right_gate_w = tl.load(
                right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            out_gate_w = tl.load(
                out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # accumulate dot‑products
            left_acc       += tl.dot(x_tile, left_w)       # (R, H)
            right_acc      += tl.dot(x_tile, right_w)
            left_gate_acc  += tl.dot(x_tile, left_gate_w)
            right_gate_acc += tl.dot(x_tile, right_gate_w)
            out_gate_acc   += tl.dot(x_tile, out_gate_w)

        # ------------------------------------------------------------------
        # Apply sigmoid gates and mask, then write the three outputs
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        left_tile  = left_acc.to(tl.float16)  * mask_val * left_gate
        right_tile = right_acc.to(tl.float16) * mask_val * right_gate

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_out_ptr + dst_off,
                 left_tile,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + dst_off,
                 right_tile,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N] FP16
    mu_ptr,       # [R] FP32
    rstd_ptr,     # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows over R
    pid_n = tl.program_id(1)   # columns over dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0

    # load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # load LN parameters for this hidden tile
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                      # [M, K]

        term_fp16 = term.to(tl.float16)   # back to FP16 for the dot

        # Load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # [K, TILE_N]

        # Accumulate matmul (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final [B, N, N, dim] tensor (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating
    # ------------------------------------------------------------------
    left_proj_T = w["left_proj.weight"].to(torch.float16).t().contiguous()   # [dim, H]
    right_proj_T = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_T = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_T = w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_T = w["out_gate.weight"].to(torch.float16).t().contiguous()

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left_proj_T,
        right_proj_T,
        left_gate_T,
        right_gate_T,
        out_gate_T,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_proj_T, right_proj_T, left_gate_T, right_gate_T, out_gate_T

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # covers H ≤ 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
