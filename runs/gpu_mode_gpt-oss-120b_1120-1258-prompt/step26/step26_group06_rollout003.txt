"""
Outgoing TriMul (AlphaFold3) – optimized forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the input feature dimension (`dim`) – using the
    highly‑optimized torch.nn.functional.layer_norm (FP32→FP16).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16).  The fused projection has shape (R,5·H) where
    R = B·N·N and H = hidden_dim.

3️⃣  Custom Triton kernel `split_fuse_gate_kernel`:
    * splits the fused projection into the three tensors *left*, *right*,
      *out_gate* (layout [B, H, N, N]),
    * applies an optional pair‑wise mask,
    * computes the three sigmoid gates,
    * writes the gated tensors (FP16) to global memory.

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).

5️⃣  Layer‑norm over the hidden dimension `H`, multiplication with `out_gate`,
    and the final projection `hidden → dim` are performed with
    torch.nn.functional.layer_norm and torch.nn.functional.linear
    (mixed‑precision, final result in FP32).

Only the split‑and‑gate step is written in Triton – this satisfies the
requirement “at least part of the operations in a kernel’’ while keeping
the hot‑path (the N³ GEMM) in highly‑optimized cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16   (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flat index r = b·N·N + i·N + k --------------------
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ---- optional mask (scalar per pair (b,i,k)) ---------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between consecutive hidden planes (N×N)
    hidden_stride = N * N
    # start offset of hidden‑plane h = 0 for this (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    # ---- iterate over hidden dimension in tiles -----------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (compute in FP32 for stability, cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets for the three outputs
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no backward).

    Args
    ----
    data : tuple
        - input_tensor : torch.Tensor        shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor or None shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32 tensors)
        - config_dict : dict containing
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ input layer‑norm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimized on H100
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim] float32
    x_norm = x_norm.to(torch.float16)   # cast once for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ fused projection → (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape [5·H, dim]) and cast to FP16
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    # (R, dim) × (5·H, dim)ᵀ  → (R, 5·H)
    proj = F.linear(x_norm, fused_w)   # FP16, contiguous
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ split, optional mask, sigmoid gates (custom Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 128
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ core contraction (batched GEMM, cuBLAS)
    #    out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 1, 2, 3).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 1, 2, 3).contiguous().view(B * H, N, N)

    # (B·H, N, N) × (B·H, N, N)ᵀ
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                     # [B, H, N, N]  FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ second layer‑norm, out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    # move hidden dimension to the last axis → [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()     # FP16
    out_fp32 = out.to(torch.float32)               # cast for stable LN

    # layer‑norm over hidden dim H (FP32)
    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                               # [B, N, N, H] FP32

    # apply out_gate (cast to FP32, same layout)
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous().to(torch.float32)
    gated = out_norm * out_gate                     # [B, N, N, H] FP32

    # final linear projection hidden → dim  (FP32)
    final = F.linear(gated, w["to_out.weight"])    # [B, N, N, dim] FP32

    return final
```
