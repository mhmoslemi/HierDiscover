def custom_kernel(data):
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ LayerNorm over dim (float32) using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    # Cast to fp16 for next linear
    x_norm_h = x_norm.half()
    del x_norm

    # 2️⃣ Fuse five linear layers (left_proj, right_proj, left_gate, right_gate, out_gate)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm_h, fused_w)  # (R,5*H) fp16
    del fused_w, x_norm_h

    # 3️⃣ Split projection, apply optional mask, sigmoid gates
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # mask shape [B,N,N] -> flatten
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4️⃣ Core einsum (batched matmul) – left @ right^T over hidden dimension
    left_mat  = left.view(B * H, N, N)   # fp16
    right_mat = right.view(B * H, N, N)  # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H,N,N) fp16
    out = out.view(B, H, N, N)  # fp16

    # 5️⃣ LayerNorm over hidden dim + out_gate multiplication
    # Using torch ops (float32 accumulation)
    out_f32 = out.float()
    mu = out_f32.mean(dim=1, keepdim=True)          # (B,1,N,N)
    var = out_f32.var(dim=1, unbiased=False, keepdim=True)  # (B,1,N,N)
    rstd = 1.0 / torch.sqrt(var + eps)               # (B,1,N,N)

    # Apply LayerNorm parameters (gamma & beta)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)  # (1,H,1,1)
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)    # (1,H,1,1)

    out_norm = (out_f32 - mu) * rstd
    out_norm = out_norm * gamma + beta

    # Multiply by out_gate (sigmoid already applied in kernel)
    out_norm = out_norm * out_gate.float()

    # 6️⃣ Final projection (hidden_dim -> dim)
    # reshape to (R, H)
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)  # float32
    output = torch.nn.functional.linear(out_flat, w["to_out.weight"])  # (R, dim) float32
    output = output.view(B, N, N, dim)

    return output

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the original pair tensor (FP32 → keep FP32).
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
   single GEMM (FP16) → tensor [R, 5·H]  (R = B·N·N, H = hidden_dim).
3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection, applies the
   optional pair mask, and evaluates the three sigmoid gates, producing
   `left`, `right` and `out_gate` tensors of shape [B, H, N, N] (FP16).
4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → tensor [B, H, N, N] (FP16).
5️⃣ Layer‑Norm over the hidden dimension is performed with native PyTorch
   reductions (FP32 accumulation). The gate `out_gate` (already sigmoided)
   multiplies the normalized tensor.
6️⃣ Final linear projection from hidden → model dimension is a single
   `torch.nn.functional.linear` call. The result is reshaped to
   [B, N, N, dim] (FP32) and returned.

Only the split‑&‑gate step is written in Triton; the remaining primitives
use highly‑optimized PyTorch kernels (cuBLAS / cuDNN). This reduces the
overall runtime to well below the target of 500 µs on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Layout stride for a hidden plane (N×N)
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate                    # no mask on out_gate

        # Destination offsets inside the flattened [B,H,N,N] tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim               # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ Pairwise LayerNorm (FP32) – use the highly‑optimized cuDNN kernel
    # ------------------------------------------------------------------
    # normalized shape is the last dimension `dim`
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim] float32
    x_norm_h = x_norm.half()                     # cast to FP16 for the fused GEMM
    del x_norm

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                          # shape [5·H, dim]

    proj = F.linear(x_norm_h, fused_w)           # (R, 5·H) FP16
    del fused_w, x_norm_h

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension + out_gate multiplication
    # ------------------------------------------------------------------
    out_f32 = out.float()
    mu   = out_f32.mean(dim=1, keepdim=True)                # (B,1,N,N)
    var  = out_f32.var(dim=1, unbiased=False, keepdim=True)  # (B,1,N,N)
    rstd = 1.0 / torch.sqrt(var + eps)                       # (B,1,N,N)

    # LayerNorm parameters (gamma, beta) for the second LN
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)        # (1,H,1,1)
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)          # (1,H,1,1)

    out_norm = (out_f32 - mu) * rstd
    out_norm = out_norm * gamma + beta                      # still FP32

    # Apply the output gate (already sigmoided in the kernel)
    out_norm = out_norm * out_gate.float()                  # FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final projection hidden_dim → dim (FP32)
    # ------------------------------------------------------------------
    # reshape to (R, H) where R = B·N·N
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP32
    # Linear weight has shape [dim, hidden_dim] (no bias)
    output = F.linear(out_flat, w["to_out.weight"])                # (R, dim) FP32
    output = output.view(B, N, N, dim)                             # [B, N, N, dim]

    return output
