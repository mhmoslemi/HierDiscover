Now we need to implement fused projection kernel in Triton. Let's outline.

**Fused projection kernel design details**:

Goal: given input x_norm (R, C) FP16, weights for 5 linear transformations (each of shape (H, C) FP16), and mask optional (R) FP16, compute left, right, out_gate in FP16 after gating and masking.

We also need left_gate and right_gate intermediate gating for left/right. Actually left_gate and right_gate will be used to multiply left and right after gating. So left_gate not needed later beyond left_mul. So we compute left_gate and right_gate via sigmoid of pre-activation (gates) and just apply them to left/right.

Thus output: left (B,H,N,N), right (B,H,N,N), out_gate (B,H,N,N). All FP16.

Implementation details:

- Use block sizes: BLOCK_R maybe 64 (like before). Block_H maybe 32.

- For each block of rows and H-block, we compute accumulators. Since we need separate accumulations for left, right, left_gate_raw, right_gate_raw, out_gate_raw (pre-sigmoid). To avoid many accumulators, we can compute left, right accumulate directly after gating? But gating depends on sigmoid, which needs raw values. So we need raw gate outputs for left_gate and right_gate, but we can compute sigmoid after full accumulation (i.e., after summing over all dim). That's fine, because we compute per row/h block: left_raw (dot product), right_raw, left_gate_raw, right_gate_raw, out_gate_raw. Then after loops over dim we compute sigmoid on left_gate_raw, etc. So we need 5 accumulator arrays of shape [BLOCK_R, BLOCK_H] (FP32). Then after finishing loops, compute gate_sigmoid, mask, multiplication, and store.

Thus kernel skeleton:

```
@triton.jit
def fused_proj_gate_kernel(
    x_ptr, # [R, C] FP16
    left_ptr, right_ptr, out_gate_ptr, # output
    mask_ptr, # optional [R] FP16
    # weight ptrs
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, C, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute b,i,k for each row (like before)
    nn = N*N
    b = offs_r // nn
    rem = offs_r - b*nn
    i = rem // N
    k = rem - i*N

    hidden_stride = N*N
    base_offset = b * (H * hidden_stride) + i * N + k

    # Load mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # For each hidden block
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Initialize accumulators
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # Loop over feature dimension
        for d_start in range(0, C, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < C

            # Load x slice
            x = tl.load(
                x_ptr + offs_r[:, None] * C + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)  # [BLOCK_R, BLOCK_D]

            # Load weight slices transposed: weight_t shape [BLOCK_D, BLOCK_H]
            # left
            left_w = tl.load(
                left_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            right_w = tl.load(
                right_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            left_gate_w = tl.load(
                left_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            right_gate_w = tl.load(
                right_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            out_gate_w = tl.load(
                out_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # dot: (BLOCK_R, BLOCK_D) @ (BLOCK_D, BLOCK_H) -> (BLOCK_R,BLOCK_H)
            left_acc += tl.dot(x, left_w)      # [BLOCK_R, BLOCK_H]
            right_acc += tl.dot(x, right_w)
            left_gate_acc += tl.dot(x, left_gate_w)
            right_gate_acc += tl.dot(x, right_gate_w)
            out_gate_acc += tl.dot(x, out_gate_w)

        # Apply sigmoid to gate accumulators
        left_gate_sig = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate_sig = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # Convert left/right acc to fp16 and apply mask and gates
        left_res = left_acc.to(tl.float16) * mask_val * left_gate_sig
        right_res = right_acc.to(tl.float16) * mask_val * right_gate_sig
        out_gate_res = out_gate_acc.to(tl.float16) * out_gate_sig  # no mask

        # Compute global offsets for stores
        off_h = h_start + offs_h                # [BLOCK_H]
        store_offset = base_offset[:, None] + off_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + store_offset,
                 left_res,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + store_offset,
                 right_res,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + store_offset,
                 out_gate_res,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The operator works on a 4‑D tensor `x` of shape [B, N, N, C] and computes

    out = LN(x)                                 # row‑wise LayerNorm
    (L, R, Lg, Rg, Og) = Linear(out)           # 5 fused linear projections
    L = L * mask * sigmoid(Lg)                  # mask + gate
    R = R * mask * sigmoid(Rg)
    O = sigmoid(Og)                             # out‑gate (no mask)

    M = BMM(L, Rᵀ)                              # (B·H, N, N)   H = hidden_dim
    µ, σ⁻¹ = row‑wise mean / rsqrt‑variance of M
    Z = (M‑µ) * σ⁻¹
    Z = Z * LN_weight + LN_bias                 # second LayerNorm
    Z = Z * O                                   # multiply by out‑gate
    Y = Z @ W_out                               # final linear projection
    reshape to [B, N, N, C]

All heavy per‑element work (LayerNorm, gating, bias, mask) is fused into
custom Triton kernels, while the dense matrix multiplications (both
batched GEMMs and the final projection) are delegated to cuBLAS
(PyTorch `torch.bmm` / `torch.nn.functional.linear`).  The implementation
uses mixed‑precision (FP16 for storage & GEMM, FP32 for accumulations) and
avoids any temporary large tensors (the 5‑projection result is never
materialised in global memory).  The final output is float‑32, identical
to the reference PyTorch module up to rounding errors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C] FP32 input
    y_ptr,          # [R, C] FP16 output
    w_ptr,          # [C] LN weight γ (FP32)
    b_ptr,          # [C] LN bias   β (FP32)
    rows,           # R = B·N·N
    dim,            # C
    eps,            # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # first pass: compute mean / variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)         # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # second pass: write normalized values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                     # [R, C]  FP16 (layer‑normed input)
    left_ptr, right_ptr, out_gate_ptr,   # outputs  [B, H, N, N] FP16
    mask_ptr,                  # [R]    FP16 (optional)
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,   # [H, C] FP16 each
    R, H, C, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) for this row
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + k   # start of h=0 plane

    # optional mask (broadcast across H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)[:, None]    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in blocks
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators in FP32
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # inner product over the feature dimension C
        # ------------------------------------------------------------------
        for d_start in range(0, C, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)         # [BLOCK_D]
            mask_d = offs_d < C

            # load a tile of the normalized input
            x = tl.load(x_ptr + offs_r[:, None] * C + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)            # [BLOCK_R, BLOCK_D]

            # weight tiles are accessed as transposed (C, H) for dot()
            wL = tl.load(left_w_ptr       + offs_d[:, None] + offs_h[None, :] * C,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)            # [BLOCK_D, BLOCK_H]
            wR = tl.load(right_w_ptr      + offs_d[:, None] + offs_h[None, :] * C,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

            wLg= tl.load(left_gate_w_ptr  + offs_d[:, None] + offs_h[None, :] * C,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wRg= tl.load(right_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * C,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wOg= tl.load(out_gate_w_ptr   + offs_d[:, None] + offs_h[None, :] * C,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

            # dot products (BLOCK_R, BLOCK_H)
            left_acc      += tl.dot(x, wL)
            right_acc     += tl.dot(x, wR)
            left_gate_acc += tl.dot(x, wLg)
            right_gate_acc+= tl.dot(x, wRg)
            out_gate_acc  += tl.dot(x, wOg)

        # sigmoid on gate pre‑activations (still FP32 → cast to FP16)
        left_gate_sig  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate_sig   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # apply mask + gates, cast back to FP16
        left_out  = left_acc.to(tl.float16)  * mask_val * left_gate_sig
        right_out = right_acc.to(tl.float16) * mask_val * right_gate_sig
        outg_out  = out_gate_acc.to(tl.float16) * out_gate_sig   # no mask

        # write results – layout: [B, H, N, N] (contiguous)
        off_h = h_start + offs_h                                 # global hidden indices
        dst_off = base_offset[:, None] + off_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst_off,
                 left_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg_out,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and preparation for final linear
# ----------------------------------------------------------------------
@triton.jit
def rowwise_ln_gate_kernel(
    out_ptr,               # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,              # [B, H, N, N] FP16   (out_gate after sigmoid)
    mu_ptr, rstd_ptr,      # [R] FP32
    ln_w_ptr, ln_b_ptr,    # [H] FP32  (to_out_norm weight / bias)
    final_ptr,             # [R, H] FP16  (output of LN+gate, ready for final linear)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics once
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0).to(tl.float32)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)   # [BLOCK_M]
    mu   = mu[:, None]          # broadcast over H
    rstd = rstd[:, None]

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)                # [BLOCK_H]
        mask_h = offs_h < H

        # load hidden slice from out and out_gate
        out_vals   = tl.load(out_ptr   + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]
        gate_vals  = tl.load(gate_ptr  + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]

        # LayerNorm parameters (γ, β) for this hidden block
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] FP32
        gamma = gamma[None, :]   # [1, BLOCK_H]
        beta  = beta[None, :]    # [1, BLOCK_H]

        # (out - μ) * rstd → γ· + β → * out_gate
        norm   = (out_vals - mu) * rstd
        scaled = norm * gamma + beta
        final  = scaled * gate_vals          # gate already contains sigmoid(out_gate)

        # store as FP16 into [R, H] matrix
        dst_off = offs_m[:, None] * H + (h_start + offs_h)[None, :]   # linear index in final_ptr
        tl.store(final_ptr + dst_off,
                 final.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : mapping with the following FP32 keys (already on device)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict must contain "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N          # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection + mask + gates
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # weight tensors (FP16, contiguous)
    left_w      = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w     = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).contiguous()

    if (not nomask) and (mask is not None):
        mask_fp = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    fused_proj_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        x_norm,
        left, right, out_gate,
        mask_ptr,
        left_w, right_w,
        left_gate_w, right_gate_w, out_gate_w,
        R, H, dim, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_w, right_w, left_gate_w, right_gate_w, out_gate_w, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ batched GEMM (left @ rightᵀ) – uses cuBLAS behind torch.bmm
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                         # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ row‑wise mean / rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 in all test configs
    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + out_gate multiplication → prepare for final linear
    # ------------------------------------------------------------------
    final_input = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_LN = 128
    BLOCK_H_LN = 32
    rowwise_ln_gate_kernel[(triton.cdiv(R, BLOCK_M_LN),)](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        final_input,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_H=BLOCK_H_LN,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 6️⃣ final projection (FP16 matmul) – cuBLAS
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16).contiguous()   # [C_out, H] = [dim, hidden_dim]
    final_fp16 = F.linear(final_input, w_out)                   # [R, dim] FP16
    final = final_fp16.to(torch.float32).reshape(B, N, N, dim)   # back to FP32

    return final
