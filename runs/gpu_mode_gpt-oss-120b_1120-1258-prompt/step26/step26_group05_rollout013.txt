        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Clean up
    del proj

    # Batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] half
    out = out.view(B, H, N, N)   # half

    # LayerNorm on out + gating + final projection
    # Flatten to (R, H) where R = B * N * N
    out_permuted = out.permute(0, 2, 3, 1).contiguous()  # shape [B, N, N, H]
    out_flat = out_permuted.view(R, H)  # half

    out_gate_permuted = out_gate.permute(0, 2, 3, 1).contiguous()
    out_gate_flat = out_gate_permuted.view(R, H)

    # LN on out_flat
    out_norm_flat = torch.nn.functional.layer_norm(out_flat, (H,), weight=weight_dict["to_out_norm.weight"], bias=weight_dict["to_out_norm.bias"], eps=eps)
    # Multiply by gate
    out_gated_flat = out_norm_flat * out_gate_flat

    # Final linear projection
    w_out = weight_dict["to_out.weight"].to(torch.float16)  # [dim, H]
    final_flat = F.linear(out_gated_flat, w_out)    # [R, dim] half

    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final

"""
Outgoing Triangle‑Multiplicative‑Update (TriMul) – forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) using
    torch.nn.functional.layer_norm (highly tuned on H100).

2️⃣  Fuse the five linear maps (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single GEMM:
    proj = 𝑥_norm @ W_fusedᵀ   →   [R, 5·H]   (R = B·N·N).

3️⃣  Triton kernel ``split_fuse_gate_kernel`` extracts the five
    blocks, applies the optional pair mask, evaluates the three
    sigmoid gates and writes the three tensors ``left``,
    ``right`` and ``out_gate`` in layout ``[B, H, N, N]`` (FP16).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    i.e. a batch of B·H matrix‑multiplications of size N×N.

5️⃣  Layer‑norm, gating and the final linear projection are fused
    using PyTorch ops:
        * reshape ``out`` to (R, H) → layer_norm over H,
        * multiply by ``out_gate`` (element‑wise),
        * project to the original channel dimension with F.linear.
    This removes the separate statistics kernel and halves the
    memory traffic on ``out`` and ``out_gate``.

The whole forward pass is gradient‑free and returns a
float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Split fused projection and apply mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16  fused linear output
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat pair index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                       # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride of one hidden plane in a [B,H,N,N] tensor
    hidden_stride = N * N
    # Base address for the first hidden slice (h = 0)
    base = b * (H * hidden_stride) + i * N + k                     # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection (row major)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid – compute in FP32 for better precision
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets: stride across hidden dimension = N·N
        dst = base[:, None] + offs_h[None, :] * hidden_stride      # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]     (bool/float) or None
        - weights_dict : dict of model parameters (all float32)
        - config_dict  : contains "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack
    # ---------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ---------------------------------------------------------------
    # Flatten to [R, dim] (float32) and apply LayerNorm.
    x_flat = x.view(R, dim)
    x_norm = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [R, dim] FP16

    # ---------------------------------------------------------------
    # 2️⃣ Fuse five linear projections
    # ---------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)         # [R, 5·H]  (FP16)

    # ---------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and compute sigmoid gates
    # ---------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # Dummy tensor – Triton still expects a pointer.
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64          # rows per program
    BLOCK_H = 64          # hidden channels per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ---------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ---------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # ---------------------------------------------------------------
    # 5️⃣ LayerNorm over H, gating and final projection (all fused in PyTorch)
    # ---------------------------------------------------------------
    # Rearrange to (B, N, N, H) → (R, H) so that H is the last dimension.
    out_perm   = out.permute(0, 2, 3, 1)                # [B, N, N, H]
    out_flat   = out_perm.reshape(R, H)                # [R, H]  FP16

    out_gate_perm = out_gate.permute(0, 2, 3, 1)       # [B, N, N, H]
    out_gate_flat = out_gate_perm.reshape(R, H)        # [R, H]  FP16

    # Layer‑norm over the hidden dimension for every (b,i,j) pair.
    out_norm_flat = torch.nn.functional.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                 # FP16

    # Apply the output gate.
    out_gated_flat = out_norm_flat * out_gate_flat    # FP16

    # Final linear projection from hidden → model dimension.
    w_out = w["to_out.weight"].to(torch.float16)       # [dim, H]
    final_flat = F.linear(out_gated_flat, w_out)       # [R, dim]  FP16

    # Reshape to the expected output layout.
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
