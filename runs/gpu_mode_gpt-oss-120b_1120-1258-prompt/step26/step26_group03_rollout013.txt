        mask_h = offs_h < H

        ptr_out = out_ptr + base[:, None] + offs_h[None, :] * NN
        vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

        norm = (vals_out - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]

        gated = norm * vals_gate

        # store to gated buffer (R, H) row-major
        dst = offs_m[:, None] * H + offs_h[None, :]
        tl.store(gated_ptr + dst, gated.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

        BLOCK_H=BLOCK_H,

    )

Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 ⇒ FP16).
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    into one GEMM (FP16) → tensor [R, 5·H] with R = B·N·N.
3️⃣  Split the fused projection, apply the optional pair mask and the three
    sigmoid gates in a Triton kernel → tensors `left`, `right`, `out_gate`
    of shape [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → shape [B, H, N, N] (FP16).
5️⃣  One Triton kernel computes the second LayerNorm over the hidden dimension,
    multiplies by `out_gate` and writes the gated tensor to a contiguous buffer
    `[R, H]` (FP16).  Statistics are computed in FP32, the rest stays in FP16.
6️⃣  Final linear projection `([R, H] @ [H, dim])` is performed with a single
    cuBLAS matmul (FP16 → FP32) and reshaped to `[B, N, N, dim]`.

All heavy work is done in FP16 (tensor‑cores) while statistics are accumulated
in FP32 for numerical stability.  The implementation uses three Triton kernels
(`layernorm_kernel`, `split_fuse_gate_kernel`, `norm_gate_kernel`) and the rest
delegates to highly‑optimized PyTorch/cuBLAS kernels.

def layernorm_kernel(
    x_ptr, y_ptr,          # [R, dim]  input FP32, output FP16
    w_ptr, b_ptr,          # [dim]     γ, β  (FP32)
    rows, dim, eps,        # scalars
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]
    # statistics
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,               # [R, 5·H]  fused projection (FP16)
    left_ptr, right_ptr,    # [B, H, N, N] (FP16)
    out_gate_ptr,           # [B, H, N, N] (FP16)
    mask_ptr,               # [R] (FP16) – only read when HAS_MASK==1
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b, i, k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over H
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for the hidden dimension inside the [B, H, N, N] tensors
    hidden_stride = N * N
    # base offset for hidden‑plane = 0
    base = b * (H * hidden_stride) + i * N + k            # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # left
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # left gate
        lgate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right gate
        rgate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # out gate (no mask)
        ogate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate_raw.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate_raw.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * lgate
        right = right_raw * m_val * rgate
        outg  = ogate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise LayerNorm + out‑gate on the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,               # [B, H, N, N]   (FP16)
    gate_ptr,              # [B, H, N, N]   (FP16)
    ln_w_ptr, ln_b_ptr,    # [H]           (FP32) – γ, β of the second LayerNorm
    gated_out_ptr,         # [R, H]         (FP16) – final gated tensor
    R, H, N, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    # decode (b,i,j) from flat index m = b·N·N + i·N + j
    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    # base offset for hidden‑plane = 0
    hidden_stride = NN
    base = b * (H * hidden_stride) + i * N + j          # [BLOCK_M]

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_h = k_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                       # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize, apply gate, write ----------
    for k_start in range(0, H, BLOCK_K):
        offs_h = k_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # load out values
        ptr_out = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # load gate values
        ptr_gate = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # (out - µ) * rstd → γ· + β → * out_gate
        norm = (vals_out - mean[:, None]) * rstd[:, None]          # [M, K]
        norm = norm * gamma[None, :] + beta[None, :]               # [M, K]
        gated = norm * vals_gate                                   # [M, K]

        # write to contiguous [R, H] buffer (row‑major)
        dst = offs_m[:, None] * H + offs_h[None, :]                # linear index
        tl.store(gated_out_ptr + dst, gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weight_dict  : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict must contain "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 ⇒ FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ  (core einsum)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate → gated tensor [R, H] (FP16)
    # ------------------------------------------------------------------
    gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_NG = 128
    BLOCK_K_NG = 32
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    norm_gate_kernel[grid_ng](
        out,
        out_gate,
        w["to_out_norm.weight"],      # γ  (FP32)
        w["to_out_norm.bias"],        # β  (FP32)
        gated,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_K=BLOCK_K_NG,
        num_warps=8,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (FP16 matmul → FP32 output)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    proj_out_fp16 = torch.matmul(gated, w_out_T)                     # [R, dim] FP16
    final = proj_out_fp16.to(torch.float32).view(B, N, N, dim)       # FP32

    return final
