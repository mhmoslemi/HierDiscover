"""
Outgoing TriMul kernel (AlphaFold‑3) – forward only, no gradients.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into
   a single cuBLAS GEMM (FP16) → tensor [R, 5·H] (R = B·N·N, H = hidden_dim).
3️⃣ Split the fused projection, apply the optional pair‑mask and three sigmoid
   gates in a Triton kernel → tensors `left`, `right`, `out_gate`
   with layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) implements the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → [B, H, N, N] (FP16).
5️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection into a single Triton kernel.  For each output position (b,i,j)
   the kernel
   * computes the mean / variance of `out` over the hidden dimension,
   * applies LayerNorm (γ,β) and the gating,
   * performs a tiny GEMM with the weight matrix `to_out` (H×dim) and writes
     the final FP32 result `[B, N, N, dim]`.

All kernels are written for Triton 3.3.1 and the H100.  The implementation
uses mixed‑precision (FP32 for statistics, FP16 for the bulk work) and
requires only the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                        # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                 # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values ---------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :]
        )

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,               # [R, 5·H] FP16 fused projection
    left_ptr,               # [B, H, N, N] FP16
    right_ptr,              # [B, H, N, N] FP16
    out_gate_ptr,           # [B, H, N, N] FP16
    mask_ptr,               # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [left|right|lg|rg|og] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (compute in FP32 then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,               # [B, H, N, N]   FP16 (contracted tensor)
    gate_ptr,              # [B, H, N, N]   FP16 (out_gate)
    ln_w_ptr,              # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,              # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,           # [H, dim]       FP16 (to_out weight transposed)
    out_final_ptr,         # [B, N, N, dim] FP32
    total_rows, H, N, DIM,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    For each (b,i,j) (flattened into row index) compute:
        mu, rstd over hidden dim H,
        apply LN (γ,β), multiply by out_gate,
        and finish with the final linear projection.
    """
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < DIM

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # ----- hidden index tile (covers whole hidden dim) -----
    k_ind = tl.arange(0, BLOCK_K)
    mask_k = k_ind < H

    # ----- load out values (FP16 -> FP32) -----
    out_vals = tl.load(
        out_ptr + base[:, None] + k_ind[None, :] * hidden_stride,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0
    ).to(tl.float32)                                   # [M, K]

    # ----- row‑wise mean / variance -----
    sum_val = tl.sum(out_vals, axis=1)                 # [M]
    sum_sq  = tl.sum(out_vals * out_vals, axis=1)      # [M]
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----- load gate values (FP16 -> FP32) -----
    gate_vals = tl.load(
        gate_ptr + base[:, None] + k_ind[None, :] * hidden_stride,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0
    ).to(tl.float32)                                   # [M, K]

    # ----- load LayerNorm parameters (γ,β) -----
    ln_w = tl.load(ln_w_ptr + k_ind, mask=mask_k, other=0.0)   # [K] FP32
    ln_b = tl.load(ln_b_ptr + k_ind, mask=mask_k, other=0.0)   # [K] FP32

    # ----- normalize, apply LN and gate -----
    norm = (out_vals - mu[:, None]) * rstd[:, None]           # [M, K]
    term = (norm * ln_w[None, :] + ln_b[None, :]) * gate_vals # [M, K]

    # ----- cast to fp16 for matmul with weight -----
    term_fp16 = term.to(tl.float16)                           # [M, K]

    # ----- load final projection slice -----
    w_tile = tl.load(
        w_out_T_ptr + k_ind[:, None] * DIM + offs_n[None, :],
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0
    )                                                          # [K, TILE_N] fp16

    # ----- compute dot : (M,K)·(K,N) → (M,N) -----
    acc = tl.dot(term_fp16, w_tile)   # fp32 => [BLOCK_M, BLOCK_N]

    # ----- store result -----
    dst_ptr = out_final_ptr + offs_m[:, None] * DIM + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens (flattened)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256     # rows per program
    BLOCK_D = 128     # dim tile
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
            w["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    # Linear projection (cuBLAS) – no bias
    proj = F.linear(x_norm, fused_w)   # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()   # [R]
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)       # FP16
    right_mat = right.view(B * H, N, N)      # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)               # [B, H, N, N] FP16
    # (left/right can be freed now)
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows (flattened B·N·N) per program
    BLOCK_K_OUT = 128  # we process the whole hidden dimension at once
    BLOCK_N_OUT = 128  # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ   FP32
        w["to_out_norm.bias"],     # β   FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # Cleanup intermediate tensors
    del out, out_gate, w_out_T

    return final
