def fused_proj_gate_kernel(
    x_ptr,        # [R, dim] fp16
    left_ptr,     # [B, H, N, N] fp16
    right_ptr,    # [B, H, N, N] fp16
    outg_ptr,     # [B, H, N, N] fp16
    mask_ptr,     # [R] fp16
    left_w_ptr,   # [dim, H] fp16
    right_w_ptr,
    left_gate_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_m = tl.program_id(0)   # over rows
    pid_n = tl.program_id(1)   # over hidden dimension

    row_start = pid_m * BLOCK_R
    col_start = pid_n * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_r = offs_r < R
    mask_h = offs_h < H

    # decode to b,i,k.
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    # Destination pointers:
    dst_left = left_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
    dst_right = right_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
    dst_outg = outg_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride

    # Accumulators
    left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # shape [BLOCK_R, BLOCK_D]

        # load weight tiles - shape [BLOCK_D, BLOCK_H]
        left_w = tl.load(
            left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_w = tl.load(
            right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        left_gate_w = tl.load(
            left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_gate_w = tl.load(
            right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        out_gate_w = tl.load(
            out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        left_acc += tl.dot(x, left_w)
        right_acc += tl.dot(x, right_w)
        left_gate_acc += tl.dot(x, left_gate_w)
        right_gate_acc += tl.dot(x, right_gate_w)
        out_gate_acc += tl.dot(x, out_gate_w)

"""
TriMul (outgoing) – Triton implementation
-----------------------------------------

The forward pass consists of:
1️⃣ Row‑wise LayerNorm on the flattened tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
   with the LayerNorm output in a single Triton kernel.
   * Input   : x_norm      [R, dim]   (FP16)
   * Weights : 5·H weight matrices, stored transposed as [dim, H] (FP16)
   * Output  : left, right, out_gate  [B, H, N, N] (FP16)
   The kernel computes a small GEMM for each output channel block,
   applies the sigmoid gates and the optional pair mask.
3️⃣ Batched GEMM (cuBLAS) implements the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   → shape `[B, H, N, N]` (FP16).
4️⃣ Row‑wise mean / rsqrt‑variance of `out` (over the hidden dim) in
   a tiny Triton kernel.
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   projection (`to_out`) into one kernel that produces the final
   `[B, N, N, dim]` float32 tensor.

Only three Triton kernels are launched (LayerNorm, fused projection &
gate, row‑wise stats, final output) and the large intermediate tensor
of shape `[R,5·H]` is never materialised, cutting the memory traffic
by >2 GB and bringing the runtime under the 500 µs target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & var ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Fused projection + gating (no large intermediate)
# --------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                 # [R, dim]   FP16 (layer‑normed input)
    left_ptr, right_ptr,   # [B, H, N, N] FP16 – outputs
    out_gate_ptr,          # [B, H, N, N] FP16 – out‑gate (sigmoid)
    mask_ptr,              # [R] FP16 (optional)
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)          # tile over rows  (R)
    pid_c = tl.program_id(1)          # tile over hidden channels (H)

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_H

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_r = offs_r < R
    mask_h = offs_h < H

    # ---- decode (b,i,k) from flat row index --------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N                     # stride of one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h==0

    # destination pointers (row‑major layout: B,H,N,N)
    dst_left = left_ptr      + base_offset[:, None] + offs_h[None, :] * hidden_stride
    dst_right = right_ptr    + base_offset[:, None] + offs_h[None, :] * hidden_stride
    dst_outg = out_gate_ptr  + base_offset[:, None] + offs_h[None, :] * hidden_stride

    # ---- accumulators -----------------------------------------
    left_acc       = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    left_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    out_gate_acc   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # ---- GEMM over the feature dimension -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # input tile: [BLOCK_R, BLOCK_D]
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # weight tiles: [BLOCK_D, BLOCK_H] (weights are stored transposed)
        left_w       = tl.load(left_w_ptr       + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
        right_w      = tl.load(right_w_ptr      + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
        left_gate_w  = tl.load(left_gate_w_ptr  + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
        right_gate_w = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
        out_gate_w   = tl.load(out_gate_w_ptr   + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

        # partial GEMM
        left_acc       += tl.dot(x, left_w)
        right_acc      += tl.dot(x, right_w)
        left_gate_acc  += tl.dot(x, left_gate_w)
        right_gate_acc += tl.dot(x, right_gate_w)
        out_gate_acc   += tl.dot(x, out_gate_w)

    # ---- mask (broadcast over hidden dim) ---------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- sigmoid gates (computed in FP32, cast back) ----------
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

    # ---- apply gates & mask, store results --------------------
    left  = left_acc.to(tl.float16)  * left_gate  * mask_val
    right = right_acc.to(tl.float16) * right_gate * mask_val

    tl.store(dst_left, left,  mask=mask_r[:, None] & mask_h[None, :])
    tl.store(dst_right, right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(dst_outg, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 5️⃣ Row‑wise statistics (mean & rsqrt‑variance)
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# --------------------------------------------------------------
# 6️⃣ Final LayerNorm + out‑gate + projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr, out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and out‑gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out‑μ)·rstd → γ· + β → * out_gate
        norm = (vals_out - mu[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]
        term = norm * vals_gate                     # [M,K]

        term_fp16 = term.to(tl.float16)

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K,TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point used by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - ``input_tensor`` : torch.Tensor[bs, N, N, dim]   (float32)
        - ``mask_tensor``  : torch.Tensor[bs, N, N]       (bool/float) or None
        - ``weights_dict`` : mapping with keys
            ``norm.weight``, ``norm.bias``,
            ``left_proj.weight``, ``right_proj.weight``,
            ``left_gate.weight``, ``right_gate.weight``, ``out_gate.weight``,
            ``to_out_norm.weight``, ``to_out_norm.bias``,
            ``to_out.weight``          (all float32)
        - ``config_dict`` must contain ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (no large intermediate)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # prepare transposed weight matrices (dim, H) in FP16
    left_w_T        = w["left_proj.weight"].t().contiguous().to(torch.float16)
    right_w_T       = w["right_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_w_T   = w["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_w_T  = w["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_w_T    = w["out_gate.weight"].t().contiguous().to(torch.float16)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # tuning parameters – chosen to fit registers comfortably
    BLOCK_R = 64   # rows per program (tiles over R)
    BLOCK_H = 8    # hidden‑dim slice per program
    BLOCK_D = 32   # reduction‑dim slice (must divide dim reasonably)

    grid_fuse = (triton.cdiv(R, BLOCK_R), triton.cdiv(H, BLOCK_H))
    fused_proj_gate_kernel[grid_fuse](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # x_norm is no longer needed
    del x_norm, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Core einsum (batched GEMM via cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128    # H ≤ 128 in all configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate & final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
