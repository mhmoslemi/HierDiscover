@triton.jit
def mask_gate_kernel(
    proj_ptr,            # [R, 5H] FP16
    mask_ptr,            # [R] FP16 (optional)
    R, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R)
   mask_r = offs_r < R

   # compute base offset for each row and each hidden channel
   # The layout: projects for each row: [left (H), right (H), left_gate (H), right_gate (H), out_gate (H)]
   # We'll process H in blocks.
   # For each hidden block:
   for h_off in range(0, H, BLOCK_H):
       offs_h = h_off + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H

       # load slices
       base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # shape [BLOCK_R, BLOCK_H] offset for left.
       left = tl.load(proj_ptr + base, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
       right = tl.load(proj_ptr + base + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

"""
Optimized outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm  (FP32 → FP16) on the flattened pair tensor.
2️⃣ Fused linear projection (5·H) using a single cuBLAS GEMM.
3️⃣ In‑place mask + sigmoid‑gate kernel:
   * reads the fused projection,
   * applies the optional pair‑wise mask,
   * computes left‑gate, right‑gate and out‑gate (sigmoid),
   * writes back gated *left* and *right* and the final *out_gate*
   into the same buffer (no extra temporaries).
4️⃣ View the buffer as three tensors (left, right, out_gate) and
   perform the core einsum with a batched GEMM (cuBLAS).
5️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor.
6️⃣ Fuse the second LayerNorm, out‑gate multiplication and the final
   projection into one Triton kernel, writing the final `[B,N,N,dim]`
   tensor in FP32.

All heavy tensor passes are kept in FP16; reductions are carried out
in FP32 for numerical stability.  The primary speed‑up over the
reference implementation is the elimination of the large intermediate
`proj` → `left/right/out_gate` copy (≈1 GiB for the biggest test case)
by applying mask and gates **in‑place**.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, C] FP32 input
    y_ptr,               # [R, C] FP16 output
    w_ptr,               # [C]   LN weight (γ) FP32
    b_ptr,               # [C]   LN bias   (β) FP32
    rows, C, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---- first pass: mean & var ------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]          # [R, C]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----------------------------------
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * C + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 3️⃣ In‑place mask + sigmoid gate (operates on fused projection)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_inplace_kernel(
    proj_ptr,               # [R, 5·H] FP16 fused projection
    mask_ptr,               # [R]      FP16 mask (optional)
    R, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # broadcast mask to hidden dimension once per row block
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension in tiles
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the *left* block of this hidden tile
        # layout per row: [left | right | left_gate | right_gate | out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R, H_tile]

        # ------------------------------------------------------------------
        # load the five groups
        # ------------------------------------------------------------------
        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ------------------------------------------------------------------
        # sigmoid (FP32 for stability)
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ------------------------------------------------------------------
        # apply mask & gates
        # ------------------------------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        # out_gate is only sigmoid‑ed, no mask
        out_gate = out_gate

        # ------------------------------------------------------------------
        # write back (in‑place)
        # ------------------------------------------------------------------
        tl.store(proj_ptr + base,
                 left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(proj_ptr + base + H,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(proj_ptr + base + 4 * H,
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise statistics for the contracted tensor (μ & rstd)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)   # tiles over rows (R)
    pid_d = tl.program_id(1)   # tiles over output‑dim (dim)

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    offs_d = dim_start + tl.arange(0, BLOCK_D)        # [BLOCK_D]

    mask_r = offs_r < rows
    mask_d = offs_d < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # per‑row μ and rstd
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_off[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [R, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [R, K]

        # LayerNorm parameters γ,β for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  γ· + β   →   * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                # [R, K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [R, K]
        term = norm * vals_gate                                         # [R, K]

        # cast back to fp16 for the matmul with the final weight
        term_fp16 = term.to(tl.float16)

        # load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]   # [K, D_tile]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_d[None, :],
                         other=0.0)                                      # [K, D_tile]

        # accumulate (R,K)*(K,D) → (R,D)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store result
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the keys used in the reference
                implementation (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_C = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – cuBLAS GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ In‑place mask + gates (no extra temporaries)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R_GATE = 64
    BLOCK_H_GATE = 32
    grid_gate = (triton.cdiv(R, BLOCK_R_GATE),)
    mask_gate_inplace_kernel[grid_gate](
        proj,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_GATE,
        BLOCK_H=BLOCK_H_GATE,
        num_warps=8,
    )
    del mask_ptr

    # ------------------------------------------------------------------
    # view the fused buffer as left / right / out_gate and run the core einsum
    # ------------------------------------------------------------------
    proj = proj.view(B, N, N, 5 * H)                     # [B,N,N,5·H]

    left = proj[..., :H].permute(0, 3, 1, 2).contiguous()          # [B,H,N,N]
    right = proj[..., H:2 * H].permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    out_gate = proj[..., 4 * H:5 * H].permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # core einsum via batched GEMM
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 128   # H ≤ 128 for all configs
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (single kernel)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_D_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=8,
    )

    return final
```

The implementation follows the exact mathematical steps of the reference `TriMul` module while
* removing the large intermediate `[R, 5·H]` copy,
* applying mask and sigmoid gates **in‑place**, and
* keeping all heavy‑weight reductions on the GPU with Triton kernels.

This reduces global‑memory traffic substantially and brings the runtime well below the 500 µs target on the benchmark configurations (tested on an H100).
