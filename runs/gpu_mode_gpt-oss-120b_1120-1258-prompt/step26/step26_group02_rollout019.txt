    device = x.device

        dim=0,

    # shape: (R, H) each

        right = right * mask_flat

    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    # apply gates
    left = left * left_gate
    right = right * right_gate
    
    # reshape to (B, H, N, N)

import triton
import triton.language as tl

def split_fuse_gate_kernel(...):

def custom_kernel(data):
    """Outgoing TriMul kernel using fused projection and
    torch native layer‑norm / linear ops.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1) Input layernorm (row‑wise)
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2) fused linear: left, right, left_gate, right_gate, out_gate
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5*H, dim]
    proj = F.linear(x_norm, fused_w)   # (R,5*H) FP16

    # 3) split, mask and sigmoid gates (Triton kernel)
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # release proj
    del proj

    # 4) batched GEMM (left * right^T)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)   # (B, H, N, N)

    # 5) second LayerNorm, out_gate multiplication and final projection
    out_perm = out.permute(0, 2, 3, 1)                 # (B, N, N, H)
    out_norm = F.layer_norm(out_perm, (H,), weight=weights["to_out_norm.weight"],

    gated = out_norm * out_gate_perm                  # (B, N, N, H) FP16

    gated_flat = gated.view(B * N * N, H)
    final = F.linear(gated_flat, weights["to_out.weight"].to(torch.float16))   # (B*N*N, dim)

    final = final.view(B, N, N, dim).to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
    torch.nn.functional.layer_norm (highly‑optimized cuDNN).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16).  The result has shape [R, 5·H] where
    R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional pair mask,
    compute the three sigmoid gates and store the three tensors
    ``left``, ``right`` and ``out_gate`` in layout [B, H, N, N] (FP16)
    with a small Triton kernel (split_fuse_gate_kernel).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → shape [B, H, N, N] (FP16).

5️⃣  Row‑wise LayerNorm on the hidden dimension H, element‑wise multiplication
    with ``out_gate`` and final projection to the model dimension ``dim``.
    All three steps are performed with torch.nn.functional.layer_norm,
    torch.nn.functional.linear and Tensor‑Core GEMMs.

Only the split+gate step is written in Triton; the remaining work is
delegated to cuDNN/cuBLAS which yields a faster overall runtime on an
NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Base offset for a hidden‑plane slice (stride = N·N)
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k                     # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: layout [left|right|lg|rg|og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (FP32 → back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"          # all float32
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                               # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                   # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R).contiguous()   # [R]
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128                       # rows processed per program
    BLOCK_H = 64                        # hidden‑dim tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left × rightᵀ) → core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                               # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, gate multiplication and final projection
    # ------------------------------------------------------------------
    # bring hidden dimension to the last axis for LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                # FP16

    # apply out‑gate (both tensors are (B, N, N, H))
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    gated = out_norm * out_gate_perm                # FP16

    # final linear projection (H → dim)
    gated_flat = gated.view(B * N * N, H)            # (R, H)
    final = F.linear(gated_flat, w["to_out.weight"].to(torch.float16))  # (R, dim)
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
