"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16)
   – a small Triton kernel does the mean/variance reduction and applies γ/β.

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single half‑precision GEMM (cuBLAS) producing a tensor of shape (R, 5·H),
   where R = B·N·N and H = hidden_dim.

3️⃣ Split the fused projection, apply the optional mask, evaluate the three
   sigmoid gates and reshape to [B, H, N, N] (still FP16).  All of this is done
   with plain PyTorch tensor operations (vectorised and therefore efficient).

4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   using torch.bmm on the view (B·H, N, N).

5️⃣ Layer‑norm over the hidden dimension H on the result, again in FP32
   (torch.nn.functional.layer_norm) and cast back to half‑precision.

6️⃣ Multiply with the out‑gate (still FP16) and a final half‑precision linear
   projection (to_out) to obtain the float‑32 output tensor
   `[B, N, N, dim]`.

Only the first LayerNorm is written in Triton; the remaining work leverages
high‑performance cuBLAS kernels and vectorised PyTorch kernels, drastically
reducing launch overhead while keeping the same numerical behaviour as the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16), custom Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # [R, dim] FP32 input
    y_ptr,                # [R, dim] FP16 output
    w_ptr,                # [dim]   LN weight (γ) FP32
    b_ptr,                # [dim]   LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (no gradient) using Triton for the first
    LayerNorm and cuBLAS / PyTorch for the remaining heavy work.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) via Triton
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # columns per program
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free FP32 copy early

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    proj = F.linear(x_norm_fp16, fused_w)                 # (R, 5·H) FP16
    del fused_w, x_norm_fp16

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates
    # --------------------------------------------------------------
    proj_view = proj.view(R, 5, H)                         # [R, 5, H]

    left      = proj_view[:, 0, :]                         # [R, H]
    right     = proj_view[:, 1, :]                         # [R, H]
    left_gate = proj_view[:, 2, :]                         # [R, H]
    right_gate= proj_view[:, 3, :]                         # [R, H]
    out_gate  = proj_view[:, 4, :]                         # [R, H]

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R)      # [R]
        mask_factor = mask_flat.unsqueeze(-1)              # [R, 1]
        left = left * mask_factor
        right = right * mask_factor
    # else: no mask → keep the values as‑is

    # Apply sigmoid gates (half‑precision is sufficient for the forward pass)
    left  = left  * torch.sigmoid(left_gate)
    right = right * torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)                     # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, H, N, N] and run the core batched GEMM
    # --------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch and hidden dimensions for cuBLAS batched GEMM
    left_mat  = left.view(B * H, N, N)                             # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                            # [B·H, N, N] FP16

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))          # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                     # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension H (FP32) + gating
    # --------------------------------------------------------------
    # Bring H to the last dimension for torch.nn.functional.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()                # [B, N, N, H]
    out_flat_fp32 = out_perm.view(R, H).to(torch.float32)          # [R, H] FP32

    out_norm_fp32 = F.layer_norm(
        out_flat_fp32,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                               # [R, H] FP32
    out_norm_fp16 = out_norm_fp32.to(torch.float16)                # back to FP16

    # Align out_gate with the same layout
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]
    out_gate_flat = out_gate_perm.view(R, H)                       # [R, H] FP16

    gated = out_norm_fp16 * out_gate_flat                          # [R, H] FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection (H → dim) – half‑precision GEMM
    # --------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)                   # [dim, H]
    final_flat = F.linear(gated, w_out)                            # [R, dim] FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)        # [B, N, N, dim] FP32

    return final
