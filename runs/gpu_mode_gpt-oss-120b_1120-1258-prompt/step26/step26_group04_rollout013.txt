"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm (mixed‑precision, FP16 for main tensors, FP32 for reductions):
1️⃣ Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   with the sigmoid‑gate and optional mask in a single Triton kernel.
   This eliminates the large temporary (R,5·H) tensor of the reference code.
3️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).
4️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H via Triton.
5️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final projection
   (`to_out`) into a single Triton kernel, writing the final `[B, N, N, dim]` tensor
   in float32.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim] FP32 input
    y_ptr,       # [R, dim] FP16 output
    w_ptr,       # [dim]    LN weight (γ) FP32
    b_ptr,       # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)         # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fuse the five linear projections, mask and sigmoids
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fuse_kernel(
    x_ptr,           # [R, dim] FP16 (layer‑normed input)
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    left_w_ptr,      # [dim, H] FP16
    right_w_ptr,     # [dim, H] FP16
    left_gate_w_ptr, # [dim, H] FP16
    right_gate_w_ptr,# [dim, H] FP16
    out_gate_w_ptr,  # [dim, H] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast across H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [BLOCK_R]
        mask_val = m_val[:, None]                                          # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for hidden channel h = 0  (flattened layout)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # ------------------------------------------------------------------
    # Loop over hidden channels in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for the five linear outputs (FP32 for stability)
        acc_left      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_outgate   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the input dimension dim in tiles
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # x slice (FP16 → FP32)
            x = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_R, BLOCK_D]

            # weight tiles (each transposed to [dim, H])
            w_l = tl.load(
                left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_D, BLOCK_H]

            w_r = tl.load(
                right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_lg = tl.load(
                left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_rg = tl.load(
                right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_og = tl.load(
                out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # ----- matrix‑multiply tile (R×D)·(D×H) → (R×H) -----
            acc_left    += tl.dot(x, w_l)      # [BLOCK_R, BLOCK_H]
            acc_right   += tl.dot(x, w_r)
            acc_lgate   += tl.dot(x, w_lg)
            acc_rgate   += tl.dot(x, w_rg)
            acc_outgate += tl.dot(x, w_og)

        # ----- apply sigmoid gates -----
        gate_l = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)   # [BLOCK_R, BLOCK_H]
        gate_r = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
        gate_o = (1.0 / (1.0 + tl.exp(-acc_outgate))).to(tl.float16)

        # ----- apply mask and write results -----
        # cast accumulators to FP16 before storing
        left_tile   = (acc_left.to(tl.float16)   * gate_l * mask_val).to(tl.float16)
        right_tile  = (acc_right.to(tl.float16)  * gate_r * mask_val).to(tl.float16)
        outg_tile   = gate_o                                            # already FP16

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(
            left_ptr + dst_offset,
            left_tile,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right_tile,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg_tile,
            mask=mask_r[:, None] & mask_h[None, :],
        )

@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]

            mask=mask_m[:, None] & mask_k[None, :],

        ).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H

    # Load per‑row statistics (mean & rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # Load hidden‑plane values
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]                # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                   # [M, K]
        term = norm * vals_gate                                         # [M, K]

        # Cast to FP16 for the dot (saves bandwidth)
        term_fp16 = term.to(tl.float16)

        # Load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N) (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + mask + sigmoid gates
    # ------------------------------------------------------------------
    left   = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Transpose weight matrices to [dim, H] for efficient Triton loading
    left_w_T      = w["left_proj.weight"].t().contiguous().half()
    right_w_T     = w["right_proj.weight"].t().contiguous().half()
    left_gate_w_T = w["left_gate.weight"].t().contiguous().half()
    right_gate_w_T= w["right_gate.weight"].t().contiguous().half()
    out_gate_w_T  = w["out_gate.weight"].t().contiguous().half()

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 32
    BLOCK_H = 32
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_fuse_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_w_T, right_w_T, left_gate_w_T, right_gate_w_T, out_gate_w_T

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are [B, H, N, N] → view as [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].t().contiguous().half()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk (fits in registers)
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
