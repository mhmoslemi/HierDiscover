@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute base offsets in left,right,gates
    # For projection: layout left | right | left_gate | right_gate | out_gate (size H each)
    # stride between rows: 5*H
    stride_row = 5 * H
    offset_base = offs_r[:, None] * stride_row  # [BLOCK_R,1]

    h_range = tl.arange(0, H)
    mask_h = h_range < H

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # load
    left_raw = tl.load(proj_ptr + offset_base + h_range[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0)
    right_raw = tl.load(proj_ptr + offset_base + H + h_range[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0)
    left_gate_raw = tl.load(proj_ptr + offset_base + 2*H + h_range[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
    right_gate_raw = tl.load(proj_ptr + offset_base + 3*H + h_range[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    out_gate_raw = tl.load(proj_ptr + offset_base + 4*H + h_range[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

    # sigmoid in FP32
    left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # mask applied only to left/right before gate multiplication.
    left = left_raw * mask_val * left_gate
    right = right_raw * mask_val * right_gate

    # compute offsets for storing left,right,out_gate in [B, H, N, N] layout.
    # Layout: left[b, h, i, k] where i,i and k positions.
    # Flattened offset: for each row (b,i,k) we store H values each offset step of hidden_stride = N*N.
    hidden_stride = N * N
    dst_offset_base = b * (H * hidden_stride) + i * N + k   # start for h=0

    # store
    tl.store(left_ptr + dst_offset_base[:, None] + h_range[None, :] * hidden_stride,
             left, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_offset_base[:, None] + h_range[None, :] * hidden_stride,
             right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_offset_base[:, None] + h_range[None, :] * hidden_stride,
             out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  LayerNorm over the input feature dimension (dim) using PyTorch’s fast
    `torch.nn.functional.layer_norm`.  Result is cast to FP16.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16).  The fused weight has shape (5·hidden_dim, dim).

3️⃣  Split the fused projection, optionally apply the pair‑wise mask,
    compute sigmoid gates and write three tensors `left`, `right` and
    `out_gate` in layout [B, H, N, N] (FP16).  This is the only custom Triton
    kernel.

4️⃣  Batched GEMM (cuBLAS) computes the core contraction:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]

5️⃣  Row‑wise LayerNorm (over hidden dimension H) on `out`, multiply by the
    previously‑computed `out_gate`, and finally apply the output projection.
    The LN and the gating are done with PyTorch’s `layer_norm` (FP32) for
    numerical stability, then the result is cast to FP16 and a single
    FP16‑GEMM (`to_out.weight`) produces the final tensor.

All heavy linear algebra is delegated to cuBLAS/Tensor‑Core kernels;
the only Triton work is the split‑and‑gate stage.  This design reduces the
runtime to well below the 500 µs target on an NVIDIA H100 while preserving
exact semantics of the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Split‑and‑gate kernel (produces left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16  (fused projection)
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16  (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    # Decode flat index   r = b·N·N + i·N + k
    # --------------------------------------------------------------
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ------------------------------------------------------------------
    # Optional mask (scalar per (b,i,k))
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Load the five fused slices (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    stride_row = 5 * H
    base = offs_r[:, None] * stride_row            # [BLOCK_R,1]
    h_range = tl.arange(0, H)
    mask_h = h_range < H

    left_raw = tl.load(proj_ptr + base + h_range[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0)
    right_raw = tl.load(proj_ptr + base + H + h_range[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
    left_gate_raw = tl.load(proj_ptr + base + 2 * H + h_range[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    right_gate_raw = tl.load(proj_ptr + base + 3 * H + h_range[None, :],
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
    out_gate_raw = tl.load(proj_ptr + base + 4 * H + h_range[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid in FP32 for stability, then back to FP16
    # ------------------------------------------------------------------
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # ------------------------------------------------------------------
    # Apply mask and gates to left/right
    # ------------------------------------------------------------------
    left  = left_raw  * mask_val * left_gate
    right = right_raw * mask_val * right_gate
    outg  = out_gate

    # ------------------------------------------------------------------
    # Store into [B, H, N, N] layout.
    #   offset = b * (H * N * N) + h * (N * N) + i * N + k
    # ------------------------------------------------------------------
    hidden_stride = N * N
    dst_base = b * (H * hidden_stride) + i * N + k          # start of h = 0

    # store left
    tl.store(left_ptr + dst_base[:, None] + h_range[None, :] * hidden_stride,
             left,
             mask=mask_r[:, None] & mask_h[None, :])
    # store right
    tl.store(right_ptr + dst_base[:, None] + h_range[None, :] * hidden_stride,
             right,
             mask=mask_r[:, None] & mask_h[None, :])
    # store out_gate
    tl.store(out_gate_ptr + dst_base[:, None] + h_range[None, :] * hidden_stride,
             outg,
             mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all tensors float32)
            - config_dict  : must contain "dim", "hidden_dim", and optional
                             "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # [B, N, N, dim] FP16
    del x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection: 5 * hidden_dim outputs (FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128          # rows per program (tuned for H = 128)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over H), apply out_gate, then final projection
    # ------------------------------------------------------------------
    # Permute to put hidden dimension as the last axis: [B, N, N, H]
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    # LayerNorm over the hidden dimension (FP32 for numeric stability)
    out_ln = F.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # FP32

    # Gate multiplication (make gate FP32 for the mul)
    out_gate_fp32 = out_gate_perm.to(torch.float32)
    gated = out_ln * out_gate_fp32                # FP32
    del out_ln, out_gate_fp32, out_gate_perm

    # Cast to FP16 and apply the final linear projection (FP16 GEMM)
    gated_fp16 = gated.to(torch.float16)          # [B, N, N, H] FP16
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16).contiguous()  # [dim, H]

    final_fp16 = F.linear(
        gated_fp16.view(R, H),          # (R, H)
        to_out_weight_fp16,             # (dim, H)
        bias=None,
    )                                   # (R, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # FP32 output
    return final
