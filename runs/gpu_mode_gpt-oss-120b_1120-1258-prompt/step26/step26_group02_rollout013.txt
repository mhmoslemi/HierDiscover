def fused_layernorm_linear_kernel(
    x_ptr,  # [R, dim] fp32
    w_ptr,  # [dim, O] fp16

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Fuse the input LayerNorm (row‑wise) and the five Linear projections
    (left, right, left_gate, right_gate, out_gate) into a single
    Triton kernel.  For each row we compute:
        mean  = sum(x) / dim
        rstd  = 1 / sqrt(var + eps)
        out   = rstd * (x·W - mean * sum_W)              # W : (dim, 5·H)
    where the multiplication is done in fp16 with tensor‑core
    acceleration and the result is stored as fp16.

2️⃣  Split the fused projection, apply the optional mask, and
    evaluate the sigmoid gates – also done in a Triton kernel.

3️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]

4️⃣  Row‑wise mean / rsqrt‑variance of the contracted tensor (LayerNorm
    over the hidden dimension).

5️⃣  Fuse the second LayerNorm, the elementwise out‑gate multiplication,
    and the final projection into a single Triton kernel that writes the
    final `[B, N, N, dim]` tensor in float‑32.

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision
(fp16 for the heavy tensor‑core work, fp32 for statistics).  No
gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5×Linear (produces [R, 5·H] fp16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_linear_kernel(
    x_ptr,          # [R, dim]  fp32
    w_ptr,          # [dim, O] fp16   where O = 5·H
    sum_w_ptr,      # [O]      fp32   (Σ_dim w)
    out_ptr,        # [R, O]   fp16
    rows, dim, O, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows
    pid_o = tl.program_id(1)          # tile over output‑features

    row_start = pid_m * BLOCK_M
    col_start = pid_o * BLOCK_O
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_o = col_start + tl.arange(0, BLOCK_O)          # [BLOCK_O]

    mask_m = offs_m < rows
    mask_o = offs_o < O

    # ------------------------------------------------------------------
    # 1️⃣  Compute row‑wise mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2️⃣  Accumulate dot‑products  x·W   (FP16·FP16 → FP32)
    # ------------------------------------------------------------------
    out_acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim

        # ----- load a tile of the input row -----
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)                    # [BLOCK_M]
        sum_sq  += tl.sum(x * x, axis=1)                # [BLOCK_M]

        # cast to fp16 for the tensor‑core GEMM
        x_fp16 = x.to(tl.float16)

        # ----- load the corresponding weight tile -----
        w = tl.load(
            w_ptr + offs_d[:, None] * O + offs_o[None, :],
            mask=mask_d[:, None] & mask_o[None, :],
            other=0.0,
        )                                                # [BLOCK_D, BLOCK_O]  fp16

        # dot‑product over the D‑dimension
        out_acc += tl.dot(x_fp16, w)                    # [BLOCK_M, BLOCK_O]  fp32

    # ------------------------------------------------------------------
    # 3️⃣  Finish LayerNorm: (x·W - mean·ΣW) * rstd
    # ------------------------------------------------------------------
    mean = sum_val / dim                                 # [BLOCK_M]
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # [BLOCK_M]

    # load ΣW for the current output slice (already reduced on the host)
    sum_w_tile = tl.load(sum_w_ptr + offs_o, mask=mask_o, other=0.0)  # [BLOCK_O] fp32

    # broadcast mean / rstd
    mean_tile = mean[:, None]                            # [BLOCK_M, 1]
    rstd_tile = rstd[:, None]                            # [BLOCK_M, 1]

    # apply the normalization
    out_norm = (out_acc - mean_tile * sum_w_tile[None, :]) * rstd_tile  # fp32

    # store result
    tl.store(
        out_ptr + offs_m[:, None] * O + offs_o[None, :],
        out_norm.to(tl.float16),
        mask=mask_m[:, None] & mask_o[None, :],
    )


# ----------------------------------------------------------------------
# 2️⃣  Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]   fp16
    left_ptr,       # [B, H, N, N] fp16
    right_ptr,      # [B, H, N, N] fp16
    out_gate_ptr,   # [B, H, N, N] fp16
    mask_ptr,       # [R]       fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    stride_hidden = N * N                     # distance between successive hidden slices

    # base offset for the first hidden slice (h = 0) in the flattened [B,H,N,N] layout
    base_offset = b * (H * stride_hidden) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: 5 blocks of size H placed consecutively
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 for a bit more accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offsets (flattened view of [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * stride_hidden   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] fp16
    mu_ptr,    # [R]          fp32
    rstd_ptr,  # [R]          fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    stride_hidden = N * N
    base_offset = b * (H * stride_hidden) + i * N + j   # start of first hidden slice

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]

        vals = tl.load(

        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H

        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * stride_hidden

        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # LN -> (out - μ) * rstd → *γ + β
        diff = (vals_out - mu[:, None]) * rstd[:, None]              # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M, K]

        # Apply out_gate
        term = norm * vals_gate                                      # [M, K]

        # Convert to fp16 for the GEMM with the final projection weight
        term_fp16 = term.to(tl.float16)

        # Load weight tile [K, TILE_N] from the transposed weight matrix
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)  (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor (flattened view of [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point – forward only TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total number of pairwise tokens
    H = hidden_dim
    O = 5 * H                        # fused output dimension

    # ------------------------------------------------------------------
    # 1️⃣  Fused LayerNorm + 5×Linear
    # ------------------------------------------------------------------
    # flatten input to [R, dim] (float32)
    x_flat = x.reshape(R, dim).contiguous()

    # concatenate the five weight matrices (shape [5·H, dim]) and transpose
    fused_w = torch.cat(
        [
            w["left_proj.weight"],      # (H, dim)
            w["right_proj.weight"],     # (H, dim)
            w["left_gate.weight"],      # (H, dim)
            w["right_gate.weight"],     # (H, dim)
            w["out_gate.weight"],       # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]
    fused_w_T = fused_w.t().contiguous()                  # [dim, 5·H]

    # pre‑compute Σ_dim w for the normalization step (fp32)
    sum_w = fused_w_T.sum(dim=0).to(torch.float32).contiguous()   # [5·H]

    # allocate projection buffer
    proj = torch.empty(R, O, dtype=torch.float16, device=device)

    # launch kernel
    BLOCK_M = 128    # rows per program
    BLOCK_D = 64     # input‑feature tile
    BLOCK_O = 64     # output‑feature tile
    grid_fused = (triton.cdiv(R, BLOCK_M), triton.cdiv(O, BLOCK_O))
    fused_layernorm_linear_kernel[grid_fused](
        x_flat,
        fused_w_T,
        sum_w,
        proj,
        R,
        dim,
        O,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )
    del x_flat, fused_w, fused_w_T, sum_w

    # ------------------------------------------------------------------
    # 2️⃣  Split, apply mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣  Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.contiguous().view(B * H, N, N)          # [B·H, N, N] fp16
    right_mat = right.contiguous().view(B * H, N, N)         # [B·H, N, N] fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                              # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 4️⃣  Row‑wise mean & rsqrt‑variance of `out` (LayerNorm over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 fits in a single tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (fp32)
        w["to_out_norm.bias"],     # β (fp32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
