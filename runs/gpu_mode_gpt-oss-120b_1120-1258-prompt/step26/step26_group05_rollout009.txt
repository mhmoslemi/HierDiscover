"""
Outgoing TriMul – forward‑only implementation (AlphaFold3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input tensor (torch.nn.functional.layer_norm,
   FP32 → FP16 for the following GEMM).

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (`F.linear`) producing a tensor
   of shape [R, 5·H] where R = B·N·N.

3️⃣ Triton kernel `split_fuse_gate_kernel` extracts the five blocks,
   applies the optional pair‑wise mask, evaluates the three sigmoid
   gates and writes the three tensors `left`, `right` and `out_gate`
   in layout [B, H, N, N] (FP16).

4️⃣ Batched GEMM (`torch.bmm`) computes the core einsum

        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]

   → `[B, H, N, N]` (FP16).

5️⃣ Row‑wise mean / variance of `out` are obtained with two vectorised
   torch reductions (FP32 accumulator).

6️⃣ In‑place LayerNorm over the hidden dimension H, multiplication by the
   `out_gate` and a final linear projection (`F.linear`) are performed
   using the highly‑optimised cuBLAS kernels.  The final result is
   returned as FP32 with shape `[B, N, N, dim]`.

Only the split‑gate step is written in Triton – all heavy GEMM / reduction
operations use PyTorch’s fast CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                       # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                         # stride for one hidden plane
    # base offset for (b, i, k) inside a hidden plane
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # loop over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointer to the start of the 5·H block for this row
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination offsets inside the output tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm  (FP32 → FP16)
    # ------------------------------------------------------------------
    # Use torch's highly‑optimised layer‑norm kernel.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"].to(torch.float16),
        bias=w["norm.bias"].to(torch.float16),
        eps=eps,
    ).to(torch.float16)                              # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left_proj, right_proj, left_gate,
    #    right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5·H, dim)

    proj = F.linear(x_norm.view(R, dim), fused_w)     # (R, 5·H) FP16
    del x_norm  # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and compute sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128           # rows per program (tuned for N up to 1024)
    BLOCK_H = 128           # hidden dimension (max hidden_dim = 128)
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM:  out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # (B·H, N, N) FP16
    right_mat = right.view(B * H, N, N)               # (B·H, N, N) FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                        # (B, H, N, N) FP16
    del left, right   # free memory

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean / variance (FP32 accumulator)
    # ------------------------------------------------------------------
    # reshape to (R, H) where each row corresponds to a (b,i,j) pair
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_flat = out_perm.view(R, H)                    # (R, H) FP16

    mu_fp32  = out_flat.mean(dim=1, dtype=torch.float32)                     # (R,)
    var_fp32 = out_flat.var(dim=1, unbiased=False, dtype=torch.float32)     # (R,)
    rstd_fp32 = torch.rsqrt(var_fp32 + eps)                                 # (R,)

    # FP16 versions for the in‑place normalization step
    mu_fp16  = mu_fp32.to(torch.float16)
    rstd_fp16 = rstd_fp32.to(torch.float16)

    # broadcast to shape (B, 1, N, N)
    mu_b = mu_fp16.view(B, N, N).unsqueeze(1)       # (B,1,N,N)
    rstd_b = rstd_fp16.view(B, N, N).unsqueeze(1)   # (B,1,N,N)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # LayerNorm parameters for the hidden dimension H (converted to FP16)
    gamma_fp16 = w["to_out_norm.weight"].to(torch.float16).view(1, H, 1, 1)
    beta_fp16  = w["to_out_norm.bias"].to(torch.float16).view(1, H, 1, 1)

    # In‑place normalization, affine transform and gating
    out = (out - mu_b) * rstd_b                # normalize (FP16)
    out = out * gamma_fp16 + beta_fp16         # affine (FP16)
    out = out * out_gate                       # apply out‑gate (FP16)

    # Final linear projection:  (R, H) @ (H, dim) → (R, dim)
    term_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16
    w_out_fp16 = w["to_out.weight"].to(torch.float16)              # (dim, H)
    final = F.linear(term_flat, w_out_fp16)                        # (R, dim) FP16

    final = final.view(B, N, N, dim).to(torch.float32)             # FP32 output
    return final
