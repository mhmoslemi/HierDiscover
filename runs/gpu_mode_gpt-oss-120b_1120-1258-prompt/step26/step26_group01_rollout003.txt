@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean and variance per row using two passes
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])
```

"""
TriMul (outgoing) – fast forward‑only implementation.

The operator works on a tensor x ∈ ℝ^{B×N×N×C} (float32) and produces a tensor
y ∈ ℝ^{B×N×N×C} (float32).  The steps are

1️⃣  Row‑wise LayerNorm over the last dimension C (FP32 → FP16).  Implemented
    in a single Triton kernel (kernel launch ≈ R/‑BLOCK_M programs).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into one GEMM:
        proj = x_norm @ W_fused.T          # shape (R, 5·H), FP16
    where H = hidden_dim.

3️⃣  Split the fused projection, apply an optional pair mask, and compute the
    sigmoid gates.  All of this is done with plain PyTorch broadcasting
    (no extra kernel needed).

4️⃣  Core “einsum’’  out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]  is a
    batched matrix‑multiply:
        left  ∈ ℝ^{B·H×N×N},  right ∈ ℝ^{B·H×N×N}
        out   = bmm(left, rightᵀ)          # FP16

5️⃣  LayerNorm across the hidden dimension H, followed by the out_gate
    multiplication:
        μ, σ²  →  (out−μ)·rsqrt(σ²+ε)·γ + β
        out ← out * out_gate

    The statistics are computed with torch.mean/var (float32) – fast cuBLAS
    kernels – and the normalization is performed in FP16.

6️⃣  Final linear projection   y = out @ W_out.T   (FP16 GEMM)  and cast to
    float32.

Only the first layer‑norm is a custom Triton kernel; the remaining work is
handled by highly‑optimized cuBLAS/GEMM kernels, giving a substantial speed‑up
over the naïve implementation while staying fully differentiable‑free (forward
only).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, C]  FP32 input
    y_ptr,        # [R, C]  FP16 output (normalized)
    w_ptr,        # [C]    LN weight γ (FP32)
    b_ptr,        # [C]    LN bias   β (FP32)
    rows, cols, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < cols

        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < cols

        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * cols + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    # compute mean / var in FP32 for numerical stability
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1, keepdim=True)               # (B,1,N,N)
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)                         # (B,1,N,N)

    # LN affine parameters (γ, β) – broadcast over (B,N,N)
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1).to(torch.float16)   # (1,H,1,1)
    ln_b = w["to_out_norm.bias"  ].view(1, H, 1, 1).to(torch.float16)

    # normalize, apply γ/β and finally the out_gate
    norm = ((out_fp32 - mu) * rstd).to(torch.float16)        # back to FP16
    norm = norm * ln_w + ln_b                               # LN
    norm = norm * out_gate                                   # multiply gate
    del out_fp32, mu, var, rstd, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection (H → C) – cuBLAS GEMM
    # --------------------------------------------------------------
    # reshape to (R, H) where R = B·N·N
    out_flat = norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16
    del norm

    w_out = w["to_out.weight"].to(torch.float16)               # (C, H)
    final = F.linear(out_flat, w_out)                          # (R, C) FP16
    final = final.to(torch.float32).view(B, N, N, dim)          # (B, N, N, C)

    return final
```
