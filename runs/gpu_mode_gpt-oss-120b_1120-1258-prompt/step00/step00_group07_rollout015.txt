@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left tile (M, K, D)
        left_ptrs = left_ptr + i_offset[:, None, None] * stride_left_i \
                                 + k_offset[None, :, None] * stride_left_k \
                                 + d_offset[None, None, :] * stride_left_d
        left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # load right tile (K, N, D) for transposed multiplication
        right_ptrs = right_ptr + k_offset[:, None, None] * stride_right_k \
                                 + j_offset[None, :, None] * stride_right_j \
                                 + d_offset[None, None, :] * stride_right_d
        right_mask = k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # accumulate dot per hidden channel
        for dd in range(BLOCK_D):
            a = left_tile[:, :, dd]  # (M, K)
            b = right_tile[:, :, dd]  # (K, N)
            acc[:, :, dd] += tl.dot(a, b)

    # write back
    out_ptrs = out_ptr + i_offset[:, None, None] * stride_out_i \
                         + j_offset[None, :, None] * stride_out_j \
                         + d_offset[None, None, :] * stride_out_d
    out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # layernorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )
    
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    if not config.get("nomask", False):
        # mask shaped (bs, N, N, 1)
        mask_unsq = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq
    
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Trimul core
    out = torch.empty_like(left)  # shape (bs, N, N, hidden_dim)
    # Convert to int64 strides
    for b in range(bs):
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b].contiguous()
        # get strides
        stride_left_i = left_b.stride(0)
        stride_left_k = left_b.stride(1)
        stride_left_d = left_b.stride(2)
        stride_right_j = right_b.stride(0)
        stride_right_k = right_b.stride(1)
        stride_right_d = right_b.stride(2)
        stride_out_i = out_b.stride(0)
        stride_out_j = out_b.stride(1)
        stride_out_d = out_b.stride(2)
        # launch kernel
        BLOCK_M = 16
        BLOCK_N = 16
        BLOCK_K = 32
        BLOCK_D = 8
        grid = ( (N + BLOCK_M - 1)//BLOCK_M,
                 (N + BLOCK_N - 1)//BLOCK_N,
                 (hidden_dim + BLOCK_D - 1)//BLOCK_D )
        trimul_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            hidden_dim,
            stride_left_i, stride_left_k, stride_left_d,
            stride_right_j, stride_right_k, stride_right_d,
            stride_out_i, stride_out_j, stride_out_d,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
            num_warps=4,  # optional tuning
        )
    # Now out contains result (bs,N,N,hidden_dim)
    
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    
    out = out * out_gate  # out_gate shape (bs,N,N,hidden_dim)
    
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out

"""
TriMul (outgoing) – Triton implementation
================================================
The TriMul operator receives a 4‑D tensor ``x`` of shape [B, N, N, D] and a
pairwise mask ``mask`` of shape [B, N, N].  It performs the following
operations (identical to the PyTorch reference):

1. Layer‑Norm over the last dimension D.
2. Linear projections (no bias) from D → H  (``left_proj`` and ``right_proj``)
   and three gating projections (``left_gate``, ``right_gate``, ``out_gate``)
   followed by a sigmoid.
3. Optional element‑wise masking of ``left`` and ``right``.
4. Element‑wise gating of ``left`` and ``right``.
5. Core “outgoing” multiplicative update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiplication for every hidden channel h.
   The kernel fuses the three‑dimensional reduction using a block‑wise GEMM
   with an inner loop over the hidden dimension.
6. Layer‑Norm over the hidden dimension H, gating with ``out_gate`` and a
   final linear projection back to D.

Only the heavy N³·H computation (step 5) is executed in Triton; the remaining
operations are performed with native PyTorch kernels (highly optimized
cuBLAS/CuDNN).  The Triton kernel tiles the i‑ and j‑axes (M‑ and N‑tiles)
and processes a sub‑block of the hidden dimension (D‑tile) at a time.
Each program (kernel instance) computes a tile of shape
[M_tile, N_tile, D_tile] = [16, 16, 8] using a standard GEMM style
accumulation with ``tl.dot`` for every hidden slice.  The algorithm
re‑uses the ``left`` tile across all j‑tiles and the ``right`` tile across all
i‑tiles, giving the same arithmetic intensity as a normal matrix‑multiply
but extended over the hidden channels.

The implementation works for batch sizes 1‑2, sequence lengths up to 1024,
and hidden dimension 128 (the only value used in the tests).  It returns a
float‑32 tensor of shape [B, N, N, D] and does **not** compute gradients.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched outer‑product reduction over K for every hidden h
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *float32   [N, N, H]  (i, k, h)
    right_ptr,         # *float32   [N, N, H]  (j, k, h)
    out_ptr,           # *float32   [N, N, H]  (i, j, h)
    N,                 # i32  sequence length
    H,                 # i32  hidden dimension
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)        # tile index over i (rows)
    pid_j = tl.program_id(1)        # tile index over j (cols)
    pid_d = tl.program_id(2)        # tile index over hidden channels

    # ------------------------------------------------------------------
    # Tile offsets (global indices)
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_base = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_base < N
    j_mask = j_base < N
    d_mask = d_base < H

    # accumulator for the (M,N,D) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # ------------------- load left[i,k,h] -------------------------
        # shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_offset = (
            i_base[:, None, None] * stride_left_i
            + k_base[None, :, None] * stride_left_k
            + d_base[None, None, :] * stride_left_h
        )
        left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
        left_tile = tl.load(left_ptr + left_offset, mask=left_mask, other=0.0)

        # ------------------- load right[j,k,h] (as K×N) -------------
        # shape (BLOCK_K, BLOCK_N, BLOCK_D) – note K is the first dim
        right_offset = (
            k_base[:, None, None] * stride_right_k
            + j_base[None, :, None] * stride_right_j
            + d_base[None, None, :] * stride_right_h
        )
        right_mask = k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        right_tile = tl.load(right_ptr + right_offset, mask=right_mask, other=0.0)

        # ------------------- accumulate per hidden slice -------------
        for d in range(BLOCK_D):
            a = left_tile[:, :, d]          # (M, K)
            b = right_tile[:, :, d]         # (K, N)
            acc[:, :, d] += tl.dot(a, b)    # (M, N)

    # ------------------- write back the result --------------------
    out_offset = (
        i_base[:, None, None] * stride_out_i
        + j_base[None, :, None] * stride_out_j
        + d_base[None, None, :] * stride_out_h
    )
    out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
    tl.store(out_ptr + out_offset, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Python entry point – builds the whole TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        (input, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the input channels (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – applied before gating (as in the reference)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B, N, N]  →  [B, N, N, 1] for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core multiplicative update – Triton kernel
    # ------------------------------------------------------------------
    # Allocate output buffer for the hidden dimension
    out_hidden = torch.empty_like(left)          # shape [B, N, N, hidden_dim]

    # Kernel launch parameters (tuned for H100 FP32)
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_D = 8

    for b in range(bs):
        # take the per‑batch slices (contiguous for good address arithmetic)
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out_hidden[b].contiguous()

        # strides in element units (not bytes)
        s_li = left_b.stride(0)  # i‑stride
        s_lk = left_b.stride(1)  # k‑stride
        s_lh = left_b.stride(2)  # hidden‑stride

        s_rj = right_b.stride(0)  # j‑stride
        s_rk = right_b.stride(1)  # k‑stride
        s_rh = right_b.stride(2)  # hidden‑stride

        s_oi = out_b.stride(0)    # i‑stride
        s_oj = out_b.stride(1)    # j‑stride
        s_oh = out_b.stride(2)    # hidden‑stride

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            (hidden_dim + BLOCK_D - 1) // BLOCK_D,
        )

        trimul_outgoing_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            hidden_dim,
            s_li, s_lk, s_lh,
            s_rj, s_rk, s_rh,
            s_oi, s_oj, s_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
            # heuristics for H100 – 4 warps gives good occupancy
            num_warps=4,
        )
        # No need to synchronize; later PyTorch ops will wait on the GPU stream.

    # ------------------------------------------------------------------
    # 7) LayerNorm over the hidden dimension, gated by out_gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
