```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_j, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_j, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hd = tl.program_id(2)

    b = pid_hd // H
    h = pid_hd % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets
        left_ptrs = left_ptr + \
            b * stride_l_batch + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_j
        # right: treating i as k
        right_ptrs = right_ptr + \
            b * stride_r_batch + h * stride_r_h + \
            offs_k[:, None] * stride_r_i + \
            offs_j[None, :] * stride_r_j

        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b_tile)

    # Store result
    out_ptrs = out_ptr + \
        b * stride_o_batch + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The heavy work in TriMul is the pairwise update
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched Gram‑product of the rows of `left` and `right`.
It can be expressed as a batched matrix‑multiplication
    out = left @ rightᵀ
where `rightᵀ` is the transpose of the last two axes of `right`.
The kernel below computes this GEMM for every batch and hidden dimension
while the surrounding LayerNorms, linear projections and gates are handled
with regular PyTorch calls (highly optimised on CUDA).

Only the forward pass is required – gradients are not computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,             # const float*  (B,N,N,H)
    right_ptr,            # const float*  (B,N,N,H)  (already transposed: row‑k, col‑j)
    out_ptr,              # float*        (B,N,N,H)
    B, N, H,              # int32 scalars
    stride_l_batch, stride_l_i, stride_l_j, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_j, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,k,j,h]
    with a classic tiled GEMM.  `right` must be provided already
    transposed (the k‑dimension is the first matrix axis).
    """
    pid_i = tl.program_id(0)          # row block (i)
    pid_j = tl.program_id(1)          # col block (j)
    pid_hd = tl.program_id(2)         # combined batch * hidden index

    # decode batch and hidden index
    b = pid_hd // H
    h = pid_hd % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (high‑precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (i, k)
        left_ptrs = left_ptr + \
            b * stride_l_batch + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_j

        # right tile (k, j) – note that `right` is already transposed
        right_ptrs = right_ptr + \
            b * stride_r_batch + h * stride_r_h + \
            offs_k[:, None] * stride_r_i + \
            offs_j[None, :] * stride_r_j

        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # (M,K) dot (K,N) → (M,N)
        acc += tl.dot(a, b_tile)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_o_batch + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] – the forward result.
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input tensor (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])       # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of linear maps on the normalized x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: out = left @ rightᵀ  (batched GEMM across H)
    #    right is transposed on‑the‑fly via a view to avoid copies.
    # ------------------------------------------------------------------
    #   rightᵀ  (row = k, col = j)  -> view with swapped seq axes.
    right_t = right.transpose(1, 2)   # shape still (B,N,N,H) but strides swapped

    B, N, _, _ = input_tensor.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Tuning parameters – good defaults for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # strides (int64) for the three tensors
    stride_l_batch, stride_l_i, stride_l_j, stride_l_h = left.stride()
    stride_r_batch, stride_r_i, stride_r_j, stride_r_h = right_t.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right_t,
        out,
        B,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_i,
        stride_l_j,
        stride_l_h,
        stride_r_batch,
        stride_r_i,
        stride_r_j,
        stride_r_h,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output normalization, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
