@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # block indices
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # pack batch * d into program id dimension? Could also use multi-dimensional program ids.
    # Use grid shape: (num_pid_m, num_pid_n, B*D). Pass B and D as separate dims? Maybe flatten B*D.

    # compute bid (batch and d)
    pid_bd = tl.program_id(1)  # size B*D
    b = pid_bd // D
    d = pid_bd % D

    # compute tile coordinates
    i0 = (pid % num_pid_m) * BLOCK_M
    j0 = (pid // num_pid_m) * BLOCK_N

    # load pointers
    # left: shape [B,N,N,D] we need left[i,k,d] where i is index, k is column dimension
    # For fixed i block (i_start..i_end), we load a block of left across i dimension but we need all k positions for reduction.
    # We'll accumulate across k blocks.

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k blocks
    for k_offset in range(0, N, BLOCK_K):
        k_end = k_offset + BLOCK_K
        # Bound masks
        i_offs = i0 + tl.arange(0, BLOCK_M)
        j_offs = j0 + tl.arange(0, BLOCK_N)
        k_offs = k_offset + tl.arange(0, BLOCK_K)

        # Load left: shape [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + b*stride_left_b + (i_offs[:,None])*stride_left_i + (k_offs[None,:])*stride_left_k + d*stride_left_d
        left = tl.load(left_ptrs, mask=(i_offs[:,None] < N) & (k_offs[None,:] < N), other=0.0)

        # Load right: shape [BLOCK_N, BLOCK_K] (note right[j,k,d] so j index first)
        right_ptrs = right_ptr + b*stride_right_b + (j_offs[:,None])*stride_right_j + (k_offs[None,:])*stride_right_k + d*stride_right_d
        right = tl.load(right_ptrs, mask=(j_offs[:,None] < N) & (k_offs[None,:] < N), other=0.0)

        # Multiply and sum across k: compute dot product accumulate
        # left (M x K) times right (N x K) elementwise across K, sum = sum_k left[i,k] * right[j,k].
        # This can be done via tl.dot(left, right) after transposition? Actually tl.dot expects (M, K) and (K, N)? But we have both (M, K) and (N, K). We can compute as (M,K) @ (N,K).T? Actually we need sum over K of left[i,k]*right[j,k] which is a matrix multiplication of left and right^T. However right shape is (N,K), we need (K,N). So we can compute right_T = tl.trans(right) produce shape (K,N). Then tl.dot(left, right_T) gives (M,N). So inside the kernel:
        right_T = tl.trans(right)  # shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right_T) # result (BLOCK_M, BLOCK_N)

    # After reduction, store out
    out_ptrs = out_ptr + b*stride_out_b + (i0[:,None])*stride_out_i + (j0[None,:])*stride_out_j + d*stride_out_d
    tl.store(out_ptrs, acc, mask=(i0[:,None] < N) & (j0[None,:] < N))

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_bd = tl.program_id(1)
    # decode batch and hidden idx
    b = pid_bd // D
    d = pid_bd % D

    # compute tile indices
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    i0 = (pid % num_tiles_m) * BLOCK_M
    j0 = (pid // num_tiles_m) * BLOCK_N

    # offsets
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # left: shape (M, K)
        left_ptrs = left_ptr + b*stride_left_b + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k + d*stride_left_d
        left = tl.load(left_ptrs, mask=(i_range[:, None] < N) & (k_range[None, :] < N), other=0.0)

        # right: shape (K, N) note order for reduction
        right_ptrs = right_ptr + b*stride_right_b + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_j + d*stride_right_d
        right = tl.load(right_ptrs, mask=(k_range[:, None] < N) & (j_range[None, :] < N), other=0.0)

        # dot product: left(M,K) * right(K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store
    out_ptrs = out_ptr + b*stride_out_b + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j + d*stride_out_d
    tl.store(out_ptrs, acc, mask=(i_range[:, None] < N) & (j_range[None, :] < N))

   device = input_tensor.device
   dtype = input_tensor.dtype

   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)

   # Load weights
   norm_weight = weights["norm.weight"]
   norm_bias = weights["norm.bias"]
   left_proj_weight = weights["left_proj.weight"]
   right_proj_weight = weights["right_proj.weight"]
   left_gate_weight = weights["left_gate.weight"]
   right_gate_weight = weights["right_gate.weight"]
   out_gate_weight = weights["out_gate.weight"]
   to_out_norm_weight = weights["to_out_norm.weight"]
   to_out_norm_bias = weights["to_out_norm.bias"]
   to_out_weight = weights["to_out.weight"]

   # LayerNorm
   # Compute mean and var across last dimension
   x = input_tensor
   mean = x.mean(-1, keepdim=True)
   var = x.var(-1, unbiased=False, keepdim=True)
   x_norm = (x - mean) * torch.rsqrt(var + eps)
   # apply weight & bias
   x_norm = x_norm * norm_weight.view(1,1,1,dim) + norm_bias.view(1,1,1,dim)

   # Linear projections
   left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N, hidden_dim)
   right = torch.nn.functional.linear(x_norm, right_proj_weight)

   # Apply mask if present and not nomask
   if not nomask:
       mask = mask_tensor.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
       left = left * mask
       right = right * mask

   # Compute gates
   left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

   left = left * left_gate
   right = right * right_gate

   # Compute heavy multiplication using Triton kernel
   B, N, _, _ = left.shape
   # ensure contiguous
   left = left.contiguous()
   right = right.contiguous()

   out = torch.empty_like(left, dtype=torch.float32) # maybe left already float32

   # strides
   stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
   stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
   stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

   # Block sizes
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 32

   import triton

   trimul_kernel[grid](

       right.data_ptr(),
       out.data_ptr(),

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (batch, N, N, dim).
2. Project the normalized tensor to a hidden dimension (left/right) with two
   bias‑free linear layers.
3. Optionally mask the projected tensors, then apply three element‑wise
   sigmoid‑gates (left, right, out) obtained from additional linear layers.
4. Compute the core O(N³) update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply L·Rᵀ for every hidden channel.
   This reduction is performed by a custom Triton kernel that tiles the i‑j
   plane (BLOCK_M×BLOCK_N) and iterates over k in blocks of size BLOCK_K.
5. Apply a second Layer‑norm over the hidden dimension, multiply by the
   out‑gate, and finally project back to the original dimension.
6. Return the tensor of shape (batch, N, N, dim).

Only the heavy O(N³) reduction is off‑loaded to Triton; all other
operations (layer‑norms, linear layers, sigmoids, masking) are done with
PyTorch for simplicity and clarity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) reduction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const *float32
    right_ptr,         # const *float32
    out_ptr,           # *float32
    B, N, D,           # int32: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The kernel processes one (batch, hidden_channel) pair per program
    instance and tiles the output i‑j plane.
    """
    pid = tl.program_id(0)          # i‑j tile index (flattened)
    pid_bd = tl.program_id(1)       # batch * hidden_dim index

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel
    # ------------------------------------------------------------------
    b = pid_bd // D
    d = pid_bd % D

    # ------------------------------------------------------------------
    # Decode tile coordinates in the i‑j plane
    # ------------------------------------------------------------------
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    i0 = (pid % num_tiles_i) * BLOCK_M
    j0 = (pid // num_tiles_i) * BLOCK_N

    # Range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator for the result tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # ------------------------------
        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        # ------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )

        # ------------------------------
        # Load a tile of right: shape (BLOCK_K, BLOCK_N)
        # Note the order (k, j) so that we can use tl.dot directly.
        # ------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + d * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
            other=0.0,
        )

        # ------------------------------
        # Accumulate dot product over the k‑dimension
        # left:  (BLOCK_M, BLOCK_K)
        # right: (BLOCK_K, BLOCK_N)
        # ------------------------------
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store the computed tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )


# ----------------------------------------------------------------------
# Entrypoint – PyTorch‑compatible custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (batch, seq_len, seq_len, dim)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # Basic configuration
    # --------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Load model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    eps = 1e-5
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight.view(1, 1, 1, dim) + norm_bias.view(1, 1, 1, dim)

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the *normalized* tensor)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy O(N³) reduction via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32)  # (B, N, N, hidden_dim)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Blocking parameters – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one program for each (i‑tile, j‑tile) pair, and one extra dimension
    # for batch * hidden_dim.
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid = (grid_i * grid_j, B * hidden_dim)

    _trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Second LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final projection back to dim
    # --------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)  # (B, N, N, dim)

    return output
