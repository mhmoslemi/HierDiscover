def custom_kernel(data):
    import torch, triton, triton.language as tl
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if not nomask:
        mask = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Prepare for Triton matmul
    # Permute to bring hidden_dim to second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Merge batch and heads
    bn = bs * hidden_dim
    left_mat = left_perm.view(bn, N, N)
    right_mat = right_perm.view(bn, N, N)

    # Cast to fp16 for compute (optional)
    left_mat_fp16 = left_mat.to(torch.float16)

    import triton
    import triton.language as tl
    torch.set_grad_enabled(False)

    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections

    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

        left = left * mask_exp
        right = right * mask_exp

    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    right_perm = right.permute(0, 3, 1, 2).contiguous()
    bn = bs * hidden_dim

    left_mat = left_perm.view(bn, N, N)
    right_mat = right_perm.view(bn, N, N)

    right_half = right_mat.to(torch.float16)

    # Output buffer

    stride_ak = left_half.stride(2)  # 1
    batch_stride_a = left_half.stride(0)  # N*N

    stride_bk = right_half.stride(2)  # 1 (original col stride)
    stride_bn = right_half.stride(1)  # N (original row stride)
    batch_stride_b = right_half.stride(0)  # N*N

    stride_cm = out_half.stride(1)  # N
    stride_cn = out_half.stride(2)  # 1
    batch_stride_c = out_half.stride(0)  # N*N

    BLOCK_N = 128
    BLOCK_K = 64

    def gemm_outgoing_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,

        b_ptrs = b_ptr + pid_batch * stride_batch_b + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

            b_ptrs += BLOCK_K * stride_bk

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

        bias=weights["to_out_norm.bias"],

def custom_kernel(data):
    """
    Outgoing Triangular Multiplicative Update (TriMul) kernel.

    This implementation follows the reference PyTorch TriMul module but accelerates
    the cubic‑complexity operation

        out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]

    with a fused Triton batched GEMM.  The steps are:

    1. Layer‑norm the input.
    2. Linear projections to hidden_dim (left/right) and gating linear layers.
    3. Apply optional mask and gate the projected tensors.
    4. Rearrange tensors to (batch*head, N, N) and launch a Triton kernel that
       computes, for each head,   left @ rightᵀ   (N×N matrix‑multiply).
    5. Fold the batch*head dimension back to (bs, N, N, hidden_dim).
    6. Layer‑norm, apply the output gate and a final linear projection back to dim.

    The heavy N³ work is performed in fp16 inside the kernel while accumulation
    and the final result are kept in fp32 for numerical stability.
    """
    import torch
    import triton
    import triton.language as tl

    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    torch.set_grad_enabled(False)          # no autograd needed
    x, mask, weights, config = data

    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3) Optional mask (broadcast on last dim)
    # ----------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(x.dtype)            # [bs, N, N, 1]
        left = left * m
        right = right * m

    # ----------------------------------------------------------------------
    # 4) Gating (sigmoid of linear maps) and apply the gates
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Rearrange for batched GEMM: (bs, hidden_dim, N, N) → (B*H, N, N)
    # ----------------------------------------------------------------------
    # (b, i, j, d) → (b, d, i, j)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B_H = bs * hidden_dim                                      # merged batch dimension
    left_mat = left.view(B_H, N, N)        # [B*H, N, N]   fp32
    right_mat = right.view(B_H, N, N)      # [B*H, N, N]   fp32

    # Cast to half for compute inside the kernel
    left_half = left_mat.to(torch.float16)
    right_half = right_mat.to(torch.float16)

    # Output buffer (fp32, because final result must be fp32)
    out_half = torch.empty_like(left_half, dtype=torch.float32)

    # ----------------------------------------------------------------------
    # 6) Strides for the kernel (contiguous tensors ⇒ row‑major layout)
    # ----------------------------------------------------------------------
    stride_am = left_half.stride(1)          # = N
    stride_ak = left_half.stride(2)          # = 1
    batch_stride_a = left_half.stride(0)     # = N*N

    # right is transposed inside the kernel → swap its row/col strides
    stride_bk = right_half.stride(2)         # = 1   (original col stride)
    stride_bn = right_half.stride(1)         # = N   (original row stride)
    batch_stride_b = right_half.stride(0)    # = N*N

    stride_cm = out_half.stride(1)           # = N
    stride_cn = out_half.stride(2)           # = 1
    batch_stride_c = out_half.stride(0)      # = N*N

    # ----------------------------------------------------------------------
    # 7) Triton kernel – batched GEMM: out = left @ rightᵀ
    # ----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # blocks along M (rows)
        (N + BLOCK_N - 1) // BLOCK_N,        # blocks along N (cols)
        B_H,                                 # merged batch*head dimension
    )

    @triton.jit
    def gemm_outgoing_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        """Compute C = A @ Bᵀ for many independent (M,N,K) mats."""
        pid_m = tl.program_id(0)          # block row
        pid_n = tl.program_id(1)          # block column
        pid_b = tl.program_id(2)          # batch*head id

        # Offsets within the tile
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_m = offs_m < M
        mask_n = offs_n < N

        # Base pointers with batch offset
        a_ptrs = a_ptr + pid_b * batch_stride_a + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
        b_ptrs = b_ptr + pid_b * batch_stride_b + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Accumulator in FP32 for stability
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K
            k_valid = (tl.arange(0, BLOCK_K) + cur_k) < K
            a = tl.load(a_ptrs,
                        mask=mask_m[:, None] & k_valid[None, :],
                        other=0.0)
            b = tl.load(b_ptrs,
                        mask=k_valid[:, None] & mask_n[None, :],
                        other=0.0)
            # tl.dot works on FP16; accumulate in FP32
            acc += tl.dot(a, b).to(tl.float32)

            # advance to next K‑tile
            a_ptrs += BLOCK_K * stride_ak
            b_ptrs += BLOCK_K * stride_bk

        # Write result
        c_ptrs = c_ptr + pid_b * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
        tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Launch the kernel
    gemm_outgoing_kernel[grid](
        left_half,
        right_half,
        out_half,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 8) Reshape back to (bs, N, N, hidden_dim)
    # ----------------------------------------------------------------------
    out = out_half.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 9) Output LayerNorm, output gate and final projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcasted element‑wise

    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
```
