@triton.jit
def triangular_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # batch index (M dimension)
    pid_i = tl.program_id(1)  # block row index
    pid_j = tl.program_id(2)  # block col index

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointer offsets
        A_ptrs = A_ptr + pid_m * stride_am + i_offset[:, None] * stride_ai + k_offset[None, :] * stride_ak
        B_ptrs = B_ptr + pid_m * stride_bm + j_offset[:, None] * stride_bi + k_offset[None, :] * stride_bk

        # load
        a = tl.load(A_ptrs, mask=(i_offset[:, None] < M) & (k_offset[None, :] < K), other=0.0)
        b = tl.load(B_ptrs, mask=(j_offset[:, None] < N) & (k_offset[None, :] < K), other=0.0)

        # compute dot product (A[i,k] * B[j,k] -> sum over k)
        # Use tl.dot product across K axis: each reduces vector.
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

        # increment pointers not needed because offsets recompute each loop

    # write out
    C_ptrs = C_ptr + pid_m * stride_cm + i_offset[:, None] * stride_ci + j_offset[None, :] * stride_cj
    tl.store(C_ptrs, acc.to(tl.float16) if tl.is_fp16(C_ptr) else acc, mask=(i_offset[:, None] < M) & (j_offset[None, :] < N))
```

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                         M, N, K,
                         stride_am, stride_ai, stride_ak,
                         stride_bm, stride_bk, stride_bj,
                         stride_cm, stride_ci, stride_cj,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # early exit if out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A sub-block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_m * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B_T sub-block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid_m * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # The rest of config may have 'nomask' but ignore.

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask if provided and not nominal (use config["nomask"]?).
    if mask is not None and mask.numel() > 0:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute pairwise multiplication sum via Triton kernel
    # reshape to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M = B * D

    left_flat = left_perm.view(M, N, N)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_am = left_flat.stride(0)

    stride_ak = left_flat.stride(2)

    stride_bk = right_flat.stride(2)  # K dimension

    stride_ci = out_flat.stride(1)

    # Reshape back to original shape
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,D]

def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
TriMul (outgoing) forward kernel.

The operation works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}
and consists of four stages:

1. Layer‑norm on the last channel (C).
2. Linear projections (left/right) + element‑wise mask,
   gated by sigmoid‑activated linear layers.
3. A batch‑wise GEMM:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply of
   left[b,d,:,:] (N×N) with the transpose of right[b,d,:,:] (N×N).
   The heavy N³ part is implemented in Triton (fused loading,
   blocking and accumulation in FP32).
4. Layer‑norm on the hidden dimension, gated by an output‑gate,
   and a final linear projection back to the original channel size.

No backward pass or gradient handling is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr,          # left  tensor  (M, N, K)   –   K == N
    B_ptr,          # right tensor  (M, N, K)   –   we read it transposed as (M, K, N)
    C_ptr,          # output tensor (M, N, N)
    M, N, K,
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bk, stride_bj,   # strides of B (note: bj = stride on N dim)
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM:   C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]   (B is read transposed)."""
    pid_m = tl.program_id(0)      # merged batch × hidden dimension
    pid_i = tl.program_id(1)      # tile over output rows    (i)
    pid_j = tl.program_id(2)      # tile over output columns (j)

    # Row / column offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds checking masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the inner reduction)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- Load a block of A (shape BLOCK_M × BLOCK_K) -----
        a_ptrs = (A_ptr
                  + pid_m * stride_am
                  + offs_i[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- Load a block of B transposed (shape BLOCK_K × BLOCK_N) -----
        # B is stored as (M, N, K); we want (M, K, N) for tl.dot.
        b_ptrs = (B_ptr
                  + pid_m * stride_bm
                  + offs_k[:, None] * stride_bk   # K‑major
                  + offs_j[None, :] * stride_bj)  # N‑minor (columns)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ----- Accumulate -----
        acc += tl.dot(a, b)   # a: (M, K), b: (K, N) → (M, N)

    # ----- Write the result -----
    c_ptrs = (C_ptr
              + pid_m * stride_cm
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    B, H, N, _ = left_perm.shape
    M = B * H                                     # merged batch × hidden dim

    left_flat  = left_perm.view(M, N, N)   # (M, N, K) with K==N
    right_flat = right_perm.view(M, N, N)

    # Output buffer for the GEMM result (still in FP32)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton launch parameters (tuned for H100, fp32 accumulation)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute strides for the flattened tensors
    stride_am, stride_ai, stride_ak = left_flat.stride()
    stride_bm, stride_bj, stride_bk = right_flat.stride()   # (M, N, K)
    # We need stride_bk (K) and stride_bj (N) in the order required by the kernel
    #   B is stored as (M, N, K) → stride_bk = stride of the last axis, stride_bj = stride of the second axis
    stride_cm, stride_ci, stride_cj = out_flat.stride()

    grid = (
        M,                                   # pid 0  → merged batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,        # pid 1  → row tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # pid 2  → column tiles
    )

    _tri_mul_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate                      # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_weight)   # back to C

    return out
```
