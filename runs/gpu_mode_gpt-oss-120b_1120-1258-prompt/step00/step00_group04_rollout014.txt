    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # x normalized
    # x shape (bs, N, N, dim)
    # layernorm over dim
    # compute mean,var across dim last
    # shape maintain
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # broadcast
    
    # linear proj
    # left = x_norm @ left_proj_weight.T
    left = torch.einsum('b i j d, h d -> b i j h', x_norm, left_proj_weight)
    right = torch.einsum('b i j d, h d -> b i j h', x_norm, right_proj_weight)

    # mask
    if mask is None:
        mask = torch.ones(x.shape[:3], device=device, dtype=x.dtype)
    mask_unsq = mask.unsqueeze(-1)  # (b, N, N, 1)
    left = left * mask_unsq
    right = right * mask_unsq

    # gates
    left_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # heavy kernel: left and right shape (bs, N, N, hidden_dim)
    bs, N, _, _ = left.shape
    
    # Allocate output placeholder
    out = torch.empty((bs, N, N, hidden_dim), dtype=x.dtype, device=device)

    # strides in elements
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()
    
    # Triton block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Ensure tile sizes do not exceed N
    # to cap: if N < BLOCK_M: adjust? but we can compute anyway with mask.

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_hd = bs * hidden_dim

    # Launch kernel
    trilu_kernel[grid_i, grid_j, grid_hd](
        left, right, out,
        bs, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # out now [bs, N, N, hidden_dim]

    # to_out_norm
    # layernorm across hidden_dim
    o_mean = out.mean(-1, keepdim=True)
    o_var = out.var(-1, unbiased=False, keepdim=True)
    out_normed = (out - o_mean) / torch.sqrt(o_var + eps)
    out_normed = out_normed * to_out_norm_weight + to_out_norm_bias

    outg = out_normed * out_gate

    # final linear to dim
    output = torch.einsum('b i j h, d h -> b i j d', outg, to_out_weight)  # to_out_weight shape (dim, hidden_dim)
    # to_out is linear without bias

    return output

def trilu_kernel(
    L_ptr, R_ptr, O_ptr,

    stride_R_b, stride_R_i, stride_R_k, stride_R_d,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)
    pid_hd = tl.program_id(2)

    b = pid_hd // D
    d = pid_hd % D

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Apply three linear projections (left/right) and three gates (left/right/out)
   using the supplied weight tensors (no bias).  The gates are passed through
   ``sigmoid``.
3. Apply the mask (if present) and the left/right gates element‑wise.
4. Compute the tensor‑wise multiplication
   ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]``.
   This is exactly a batched matrix multiplication for every channel ``d``:
   ``out_d = left_d @ right_d.T``.  The heavy O(N³·hidden_dim) part is
   performed by a Triton kernel that tiles the i/j dimensions (BLOCK_M×BLOCK_N)
   and loops over the k‑dimension in blocks (BLOCK_K).  Each program instance
   computes the output for a single batch index ``b`` and a single hidden channel
   ``d``.
5. Layer‑norm the result over the hidden channel dimension, apply the out‑gate,
   and project back to the original feature dimension with a final linear layer.

The kernel loads the projected tensors in half‑precision to reduce memory traffic,
accumulates in FP32, and writes the result back as FP16 which is cast to FP32
before the final layers.  All other operations are performed with PyTorch
(which already provides highly‑optimized kernels for the linear projections and
layer‑norms).
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------------------- #
@triton.jit
def trilu_kernel(
    L_ptr,                     # const * left  (fp16)
    R_ptr,                     # const * right (fp16)
    O_ptr,                     # * out (fp16)
    B, N, D,                   # int: batch, seq_len, hidden_dim
    stride_L_b, stride_L_i, stride_L_k, stride_L_d,
    stride_R_b, stride_R_i, stride_R_k, stride_R_d,
    stride_O_b, stride_O_i, stride_O_j, stride_O_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Tile‑wise multiplication with a reduction over the k‑axis."""
    pid_i = tl.program_id(0)            # tile row   (i)
    pid_j = tl.program_id(1)            # tile col   (j)
    pid_hd = tl.program_id(2)           # flattened (batch, hidden)

    # decode batch index and hidden channel
    b = pid_hd // D                     # batch
    d = pid_hd % D                      # channel

    # start offsets for the i/j tiles
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------ #
        # Load left tile  L[i, k, d]  (shape: BLOCK_M x BLOCK_K)
        # ------------------------------------------------------------------ #
        L_ptrs = (
            L_ptr
            + b * stride_L_b
            + offs_i[:, None] * stride_L_i
            + offs_k[None, :] * stride_L_k
            + d * stride_L_d
        )
        L = tl.load(
            L_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)                 # up‑cast to FP32 for the dot

        # ------------------------------------------------------------------ #
        # Load right tile R[k, j, d] = right^[T] (shape: BLOCK_K x BLOCK_N)
        # ------------------------------------------------------------------ #
        R_ptrs = (
            R_ptr
            + b * stride_R_b
            + offs_k[:, None] * stride_R_i   # k goes on the original i‑axis
            + offs_j[None, :] * stride_R_k   # j goes on the original k‑axis
            + d * stride_R_d
        )
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # matrix‑multiply the tiles and accumulate
        acc += tl.dot(L, R)                # (M,K)·(K,N) → (M,N)

    # ---------------------------------------------------------------------- #
    # Store the result (convert back to fp16)
    # ---------------------------------------------------------------------- #
    O_ptrs = (
        O_ptr
        + b * stride_O_b
        + offs_i[:, None] * stride_O_i
        + offs_j[None, :] * stride_O_j
        + d * stride_O_d
    )
    tl.store(
        O_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point used by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ---------------------------------------------------------------------- #
    # unpack arguments
    # ---------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------------------------------------------------------------- #
    # LayerNorm on the input (over the last dimension)
    # ---------------------------------------------------------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b  # broadcast over [bs, N, N, dim]

    # ---------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ---------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"]        # (hidden_dim, dim)

    # use einsum for the last‑dim projection (very efficient)
    left = torch.einsum("b i j d, h d -> b i j h", x, left_proj_w)
    right = torch.einsum("b i j d, h d -> b i j h", x, right_proj_w)

    # ---------------------------------------------------------------------- #
    # Optional mask (broadcast on the hidden dimension)
    # ---------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones(x.shape[:3], device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)            # [b, N, N, 1]
    left = left * mask
    right = right * mask

    # ---------------------------------------------------------------------- #
    # Gates (sigmoid)
    # ---------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------------- #
    # Cast to fp16 for the heavy kernel (halves memory traffic)
    # ---------------------------------------------------------------------- #
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    bs, N, _, _ = left_fp16.shape

    # output buffer (fp16)
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ---------------------------------------------------------------------- #
    # Strides in *elements* (Triton works with element strides)
    # ---------------------------------------------------------------------- #
    stride_L_b, stride_L_i, stride_L_k, stride_L_d = left_fp16.stride()
    stride_R_b, stride_R_i, stride_R_k, stride_R_d = right_fp16.stride()
    stride_O_b, stride_O_i, stride_O_j, stride_O_d = out_fp16.stride()

    # ---------------------------------------------------------------------- #
    # Triton launch configuration
    # ---------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_hd = bs * hidden_dim                     # batch * channel

    trilu_kernel[
        (grid_i, grid_j, grid_hd),
        # shared memory is not needed; all arguments after the grid are passed as scalars
    ](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        stride_L_b,
        stride_L_i,
        stride_L_k,
        stride_L_d,
        stride_R_b,
        stride_R_i,
        stride_R_k,
        stride_R_d,
        stride_O_b,
        stride_O_i,
        stride_O_j,
        stride_O_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ---------------------------------------------------------------------- #
    # Post‑multiply by out‑gate, layer‑norm over hidden_dim, final projection
    # ---------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)

    # LayerNorm on hidden dimension
    o_mean = out.mean(-1, keepdim=True)
    o_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - o_mean) / torch.sqrt(o_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply out‑gate
    out_gate = out_gate  # already in FP32
    gated = out_norm * out_gate

    # Final linear projection back to `dim` (no bias)
    output = torch.einsum("b i j h, d h -> b i j d", gated, to_out_w)

    return output
