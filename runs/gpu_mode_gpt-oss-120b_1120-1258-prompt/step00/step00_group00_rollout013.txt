                 B, N, hidden_dim,
                 stride_b, stride_i, stride_k, stride_d,
                 BLOCK_M: tl.constexpr,
                 BLOCK_N: tl.constexpr,
                 BLOCK_K: tl.constexpr):

    pid_n = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # combined batch/hid

    # compute batch and hidden index
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # compute start indices
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # create range offsets
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

"""
TriMul (outgoing) implementation.

The module computes:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of a LayerNorm’ed input,
element‑wise gated (sigmoid) and optionally masked.  The cubic reduction over
the sequence dimension `k` dominates the runtime.  It is implemented in a
custom Triton kernel that fuses the reduction into a blocked matrix‑multiply
(left @ right^T) per hidden dimension using fp16 tensor‑core arithmetic with
fp32 accumulation.  All remaining operations (LayerNorms, linear layers,
gates) are performed in PyTorch.

The kernel processes one hidden slice (`d`) of one batch (`b`) per program
instance, iterating over `k` in blocks.  This yields a TensorCore‑accelerated
implementation that runs comfortably within the target micro‑second budget on
an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core contraction:   out[b,i,j,d] = Σ_k L[b,i,k,d] * R[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    left_ptr,            # *fp16   left tensor  [B,N,N,HD]
    right_ptr,           # *fp16   right tensor [B,N,N,HD]
    out_ptr,             # *fp32   output tensor[B,N,N,HD]
    B, N, HD,            # runtime ints
    stride_b, stride_i, stride_k, stride_d,   # strides of the 4‑D tensors
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched block‑matrix multiply across the `k` dimension.

    Each program instance handles a tile (BLOCK_M×BLOCK_N) of the output for
    a single hidden slice `d` of a specific batch `b`.
    """
    pid_i = tl.program_id(0)          # tile index on the i‑axis
    pid_j = tl.program_id(1)          # tile index on the j‑axis
    pid_z = tl.program_id(2)          # combined batch‑hidden index

    # ------------------------------------------------------------------
    # decode combined batch / hidden index
    # ------------------------------------------------------------------
    b = pid_z // HD                   # batch id
    d = pid_z % HD                    # hidden slice id

    # ------------------------------------------------------------------
    # tile start offsets
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left block: shape (BLOCK_M, BLOCK_K)
        #   offset = b*stride_b + i*stride_i + k*stride_k + d
        # ------------------------------------------------------------------
        left_ptrs = left_ptr \
            + b * stride_b \
            + offs_i[:, None] * stride_i \
            + offs_k[None, :] * stride_k \
            + d * stride_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # ------------------------------------------------------------------
        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        #   we want right_T[k, j] = right[b, j, k, d]
        #   offset = b*stride_b + k*stride_k + j*stride_i + d
        # ------------------------------------------------------------------
        right_ptrs = right_ptr \
            + b * stride_b \
            + offs_k[:, None] * stride_k \
            + offs_j[None, :] * stride_i \
            + d * stride_d
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0,
                          dtype=tl.float16)

        # ------------------------------------------------------------------
        # Block matrix multiply with TensorCore (fp16 -> fp32 accumulation)
        # ------------------------------------------------------------------
        acc += tl.dot(left, right_T)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_b \
        + offs_i[:, None] * stride_i \
        + offs_j[None, :] * stride_k \
        + d * stride_d
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :],
             dtype=tl.float32)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def trmul_contract(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left_fp16 / right_fp16 : [B, N, N, HD]  (dtype torch.float16)

    Returns:
        out : [B, N, N, HD]  (dtype torch.float32)
    """
    B, N, N2, HD = left_fp16.shape
    assert N == N2, "Input must be square in the second two dimensions"

    # output tensor (fp32)
    out = torch.empty((B, N, N, HD), dtype=torch.float32, device=left_fp16.device)

    # strides in element units (not bytes)
    stride_b = left_fp16.stride(0)      # N * N * HD
    stride_i = left_fp16.stride(1)      # N * HD
    stride_k = left_fp16.stride(2)      # HD
    stride_d = left_fp16.stride(3)      # 1

    # block sizes – chosen to keep register pressure moderate and hit Tensor Cores
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # j tiles
        B * HD                           # combined batch‑hidden dimension
    )

    trmul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, HD,
        stride_b, stride_i, stride_k, stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

            - "dim" (int)

    torch.Tensor

    # Unpack arguments
    # ------------------------------------------------------------------

    dim = config["dim"]

    # ------------------------------------------------------------------
    if (mask_tensor is not None) and (not nomask):
        # mask shape -> [B,N,N,1] matching the spatial dims
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # broadcast to hidden dim
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7) Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    out = trmul_contract(left_fp16, right_fp16)   # [B,N,N,HD], dtype float32

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 9) Multiply by the output gate (broadcast)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_out = F.linear(out_norm, weights["to_out.weight"])

    return final_out
```
