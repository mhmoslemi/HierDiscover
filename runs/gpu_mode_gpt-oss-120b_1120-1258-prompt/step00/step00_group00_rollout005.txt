def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_ak, stride_abk,
    stride_bbk, stride_bn, stride_bnk,
    stride_ccm, stride_ccn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute batch id + block ids.
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # load a and b pointers start
    a_ptrs = a_ptr + pid_batch * stride_cbatch + offs_m[:, None] * stride_aba + tl.arange(0, BLOCK_K)[None, :] * stride_ak

        b = tl.load(b_ptrs, mask=..., other=0.0)  # shape (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b) # shape (BLOCK_M, BLOCK_N)

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    BATCH, M, N, K,
    stride_aba, stride_ak2, stride_abk,  # Actually stride for A: (batch, M, K)
    stride_bbk, stride_bkn, stride_bn2,  # B: (batch, K, N)
    stride_ccm, stride_ccn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute number of blocks of K.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for k
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask for out-of-bounds
        a_valid = (offs_m < M) & (offs_k < K)
        b_valid = (offs_k < K) & (offs_n < N)

        # Load A and B
        # A index: [bid, offs_m, offs_k] using strides stride_aba (batch), stride_am (row?), stride_ak (col)
        a_ptrs = A + bid * stride_cbatch + offs_m[:, None] * stride_aba + offs_k[None, :] * stride_ak2
        b_ptrs = B + bid * stride_cbatch + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bn2

        a = tl.load(a_ptrs, mask=a_valid[:, None], other=0.0)
        b = tl.load(b_ptrs, mask=b_valid[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C + bid * stride_cbatch + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn
    # Mask for write
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get('eps', 1e-5)
    nomask = config.get('nomask', True)

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if not nomask:
        mask_fp = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _hidden_dim = left_h.shape  # left_h: [B,N,N,hidden_dim]
    # reshape to batch*hidden_dim, N,N
    left_reshape = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # A
    right_reshape = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # original right
    right_T = right_reshape.transpose(1, 2).contiguous()  # B

    # allocate output
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=input_tensor.device)

    # get strides in number of elements
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batched_matmul_kernel[grid](
        left_reshape,
        right_T,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out_gate (shape [B,N,N,hidden_dim])
    out = out * out_gate

    # final projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel – Optimized forward of the TriMul “outgoing” operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input (normalize over the last dimension `dim`).
2. Two linear projections (`left_proj`, `right_proj`) produce tensors of shape
   [B, N, N, H] where H = hidden_dim.
3. (Optional) Apply the pair‑wise mask.
4. Compute three sigmoid gates from the normalized input:
   – `left_gate`, `right_gate` (modulate the two projections)
   – `out_gate`   (applied after the multiplicative update)
5. Fuse the gates with the projections (`left *= left_gate`, `right *= right_gate`).
6. Compute the core “TriMul” operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication for each (b,h):
      (N×N) @ (N×N)^T.
   We reshape the tensors to a single batch dimension B·H and launch a
   Triton kernel that performs a blocked FP16 mat‑mul with FP32 accumulation.
7. Reshape the result back to [B, N, N, H] and apply a second LayerNorm.
8. Multiply by `out_gate` and project back to `dim` with a linear layer.
9. Return the final tensor of shape [B, N, N, dim].

Only the heavy N³‑compute (step 6) is hand‑written in Triton; all
other ops use highly‑optimized PyTorch kernels.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B   where
        A : [batch, M, K]  (FP16)
        B : [batch, K, N]  (FP16, already transposed)
        C : [batch, M, N]  (FP32 accumulator)
    The kernel works over three program‑id dimensions:
        0 – batch id,
        1 – block‑row id,
        2 – block‑col id.
    """
    bid = tl.program_id(0)      # batch dimension
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block column

    # Compute absolute coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tiles of A and B
        a_ptrs = (A
                  + bid * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        b_ptrs = (B
                  + bid * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)

        # Load with bounds checking
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # Write the output tile
    c_ptrs = (C
              + bid * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of model parameters (all tensors on the same device)
            - config: dict containing at least:
                * "dim"          – input feature dimension
                * "hidden_dim"   – intermediate dimension H
                * "nomask" (bool, optional, default True)
                * "eps" (float, optional, default 1e-5)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣  Compute sigmoid gates from the normalized input x
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5️⃣  Apply gates to the projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Core Triplet multiplication (batched matmul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half‑precision for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Merge batch and hidden_dim so that each (b, h) pair becomes an
    # independent matrix multiplication.
    # Shape after view: [B*H, N, N]
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # We need the transpose of the right operand: (K, N) where K == N.
    right_mat_T = right_mat.transpose(1, 2).contiguous()

    # Output tensor (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Prepare strides (in element counts, not bytes) for the kernel.
    # --------------------------------------------------------------
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Blocking parameters – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch, #blocks_M, #blocks_N)
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Launch the Triton kernel
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

The `custom_kernel` function follows the reference PyTorch implementation exactly,
but the N³ triangular multiplication is executed by the `_batched_matmul_kernel`
written in Triton. This hybrid approach satisfies the requirement of using
Triton while still leveraging PyTorch’s highly‑optimized primitives for the
remaining operations.
