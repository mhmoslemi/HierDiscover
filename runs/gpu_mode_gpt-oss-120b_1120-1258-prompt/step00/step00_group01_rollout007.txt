Implementation skeleton:

```
@triton.jit
def layernorm_fwd_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_in0, stride_in1,
    stride_out0, stride_out1,
    stride_weight, stride_bias,
    N: tl.constexpr,  # total number of rows
    D: tl.constexpr,  # dim size
    EPS: tl.constexpr,
    BLOCK_D: tl.constexpr):
    row = tl.program_id(0)
    # compute offset for the row
    inp_offset = row * stride_in0
    out_offset = row * stride_out0

    # loop across dim
    sum = tl.zeros([1], dtype=tl.float32)
    sumsq = tl.zeros([1], dtype=tl.float32)
    for d in range(0, D, BLOCK_D):
        cur_block = tl.minimum(D - d, BLOCK_D)
        offs = d + tl.arange(0, BLOCK_D)[:cur_block]
        idx = inp_offset + offs * stride_in1
        x = tl.load(inp_ptr + idx, mask = offs < D, other=0.0).to(tl.float32)
        sum += tl.sum(x)
        sumsq += tl.sum(x * x)
    mean = sum / D
    var = sumsq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)
    # now normalize and apply weight & bias
    for d in range(0, D, BLOCK_D):
        cur_block = tl.minimum(D - d, BLOCK_D)
        offs = d + tl.arange(0, BLOCK_D)[:cur_block]
        idx = inp_offset + offs * stride_in1
        out_idx = out_offset + offs * stride_out1
        x = tl.load(inp_ptr + idx, mask = offs < D, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + offs * stride_weight, mask = offs < D, other=0.0)
        b = tl.load(bias_ptr + offs * stride_bias, mask = offs < D, other=0.0)
        y = (x - mean) * inv_std * w + b
        tl.store(out_ptr + out_idx, y, mask = offs < D)
```

```
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_in0, stride_in1,
    stride_out0, stride_out1,
    stride_weight, stride_bias,
    D: tl.constexpr, EPS: tl.constexpr, BLOCK_D: tl.constexpr):
    row = tl.program_id(0)
    # Compute base pointer offsets
    inp_offset = row * stride_in0
    out_offset = row * stride_out0

    sum_val = tl.zeros([1], dtype=tl.float32)
    sumsq_val = tl.zeros([1], dtype=tl.float32)

    for d in range(0, D, BLOCK_D):
        # load block
        cur = tl.minimum(D - d, BLOCK_D)
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(inp_ptr + inp_offset + offs * stride_in1, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var = sumsq_val / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    for d in range(0, D, BLOCK_D):
        cur = tl.minimum(D - d, BLOCK_D)
        offs = d + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(inp_ptr + inp_offset + offs * stride_in1, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + offs * stride_weight, mask=mask, other=0.0)
        b = tl.load(bias_ptr + offs * stride_bias, mask=mask, other=0.0)
        y = (x - mean) * inv_std * w + b
        tl.store(out_ptr + out_offset + offs * stride_out1, y, mask=mask)
```

import triton
import triton.language as tl

# Triton layernorm kernel
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_in0, stride_in1,
    stride_out0, stride_out1,
    stride_weight, stride_bias,
    D: tl.constexpr, EPS: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    # base offset for this row
    row_offset = pid * stride_in0

    sumsq_val = tl.zeros([1], dtype=tl.float32)

"""
TriMul (outgoing) forward pass – optimized implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension (dim) using a
   custom Triton kernel (no need for a reduction across batches because we
   flatten the [B,N,N] axes and treat each row independently).

2. Compute linear projections and gates with `torch.nn.functional.linear`
   (bias‑free) and a sigmoid activation for the gates.

3. Apply the (optional) pairwise mask and the left/right gates.

4. Perform the heavy “triangle‑multiplicative” contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched matrix multiplication for every hidden channel `d`.
   We reshape the tensors to ``[B*H, N, N]`` and call `torch.bmm`
   (FP16) which uses the H100 Tensor‑cores.

5. Layer‑norm the resulting tensor over the hidden dimension using the same
   Triton kernel, multiply by the output gate and linearly project back to
   `dim`.

Only the LayerNorm operations are written in Triton; all other arithmetic
remains in PyTorch (the most efficient for the large batched matmuls).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton LayerNorm (no bias reduction across batches, operates on a 2‑D view)
# --------------------------------------------------------------------------- #
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,                # pointers
    weight_ptr, bias_ptr,            # layer‑norm parameters
    stride_in0, stride_in1,          # input strides (row, col)
    stride_out0, stride_out1,        # output strides (row, col)
    stride_w, stride_b,              # weight / bias strides (always 1)
    D: tl.constexpr,                 # feature dimension (last axis)
    EPS: tl.constexpr,               # epsilon
    BLOCK_D: tl.constexpr,           # tile size along D
):
    row = tl.program_id(0)                     # which [B,N,N] row we process
    row_offset_in = row * stride_in0            # base address for this row (input)
    row_offset_out = row * stride_out0          # base address for this row (output)

    # ------------------------------------------------------------------- #
    # 1) Compute mean and variance (float32 accumulation)
    # ------------------------------------------------------------------- #
    sum_val = tl.zeros([1], dtype=tl.float32)
    sumsq_val = tl.zeros([1], dtype=tl.float32)

    for off in range(0, D, BLOCK_D):
        cur = tl.minimum(D - off, BLOCK_D)
        idx = off + tl.arange(0, BLOCK_D)
        mask = idx < D
        # load a block of D elements, cast to FP32 for reduction
        x = tl.load(inp_ptr + row_offset_in + idx * stride_in1,
                    mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)

    mean = sum_val / D
    var = sumsq_val / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------- #
    # 2) Normalize, apply weight & bias
    # ------------------------------------------------------------------- #
    for off in range(0, D, BLOCK_D):
        cur = tl.minimum(D - off, BLOCK_D)
        idx = off + tl.arange(0, BLOCK_D)
        mask = idx < D
        x = tl.load(inp_ptr + row_offset_in + idx * stride_in1,
                    mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + idx * stride_w, mask=mask, other=0.0)
        b = tl.load(bias_ptr + idx * stride_b,   mask=mask, other=0.0)
        y = (x - mean) * inv_std * w + b
        tl.store(out_ptr + row_offset_out + idx * stride_out1,
                 y, mask=mask)


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_size: int = 128) -> torch.Tensor:
    """
    Apply LayerNorm on the last dimension of ``x`` using the Triton kernel.
    ``x`` must have shape [..., D] and be contiguous in memory.
    """
    assert x.is_contiguous()
    B, N, N2, D = x.shape
    total_rows = B * N * N2        # number of independent rows to normalize

    x_flat = x.reshape(total_rows, D).contiguous()
    out_flat = torch.empty_like(x_flat)

    grid = (total_rows,)
    # strides for a 2‑D contiguous tensor: (row_stride = D, col_stride = 1)
    layernorm_kernel[grid](
        x_flat, out_flat,
        weight, bias,
        D, 1,                     # stride_in0, stride_in1
        D, 1,                     # stride_out0, stride_out1
        1, 1,                     # stride_w, stride_b
        D, eps, block_size,
        num_warps=4,
    )
    return out_flat.view(B, N, N2, D)


# --------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] – the forward pass of the
        “outgoing” TriMul operator.
    """
    # ------------------------------------------------------------------- #
    # 0) unpack arguments
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (custom Triton kernel)
    # ------------------------------------------------------------------- #
    x_norm = triton_layernorm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_size=128,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections & gates (PyTorch – highly optimized)
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])        # [B,N,N,hidden]

    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------- #
    # 3) Optional mask handling (mask shape [B,N,N])
    # ------------------------------------------------------------------- #
    if config.get("nomask", False):
        mask_val = torch.ones_like(mask, dtype=x.dtype, device=device)
    else:
        mask_val = mask.to(x.dtype)
    mask_val = mask_val.unsqueeze(-1)               # [B,N,N,1]

    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------- #
    # 4) Apply left/right gates
    # ------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Triangle‑multiplicative contraction
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We rewrite it as a batched matrix multiplication per hidden channel.
    # ------------------------------------------------------------------- #
    # Use FP16 for the heavy matmul to exploit Tensor‑cores.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape                      # H == hidden_dim
    # reshape to [B*H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    # right must be transposed on the two N axes before reshaping
    right_t = right.transpose(1, 2).contiguous()
    right_mat = right_t.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Batched matmul – leverages cuBLAS/Tensor‑cores
    out_mat = torch.bmm(left_mat, right_mat)          # [B*H, N, N]  (FP16)

    # Back to original layout and FP32 for the rest of the pipeline
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)
    # out shape: [B, N, N, hidden]

    # ------------------------------------------------------------------- #
    # 6) Output LayerNorm (again via Triton)
    # ------------------------------------------------------------------- #
    out = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_size=128,
    )   # [B, N, N, hidden]

    # ------------------------------------------------------------------- #
    # 7) Apply the output gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------- #
    out = out * out_gate                               # [B,N,N,hidden]
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim] (no bias)

    return out
