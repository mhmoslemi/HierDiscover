```
def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # x shape: (B,N,N,dim)
    # Compute mean/var over last dim
    # Using torch.mean and var
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))

    # If mask is None, create mask of ones.
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2], dtype=x.dtype, device=x.device)
    # Ensure mask shape (B,N,N)
    # Cast mask to float32
    mask = mask.to(x.dtype)

    # Apply mask and gates using Triton kernel: produce left_out,right_out
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Compute strides
    left_stride = left.stride()
    right_stride = right.stride()
    mask_stride = mask.stride()
    left_gate_stride = left_gate.stride()
    right_gate_stride = right_gate.stride()

    BLOCK_I = 64
    BLOCK_K = 64
    BLOCK_H = 32

    grid = (left.shape[0],  # batch
            (left.shape[1] + BLOCK_I - 1) // BLOCK_I,
            (left.shape[2] + BLOCK_K - 1) // BLOCK_K)

    apply_gate_mask_left_kernel[grid](
        left_ptr=left,
        mask_ptr=mask,
        gate_ptr=left_gate,
        out_ptr=left_out,
        B=left.shape[0], N=left.shape[1], H=hidden_dim,
        stride_left_b=left_stride[0], stride_left_i=left_stride[1], stride_left_k=left_stride[2], stride_left_h=left_stride[3],
        stride_mask_b=mask_stride[0], stride_mask_i=mask_stride[1], stride_mask_k=mask_stride[2],
        stride_gate_b=left_gate_stride[0], stride_gate_i=left_gate_stride[1], stride_gate_k=left_gate_stride[2], stride_gate_h=left_gate_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # Similarly for right
    apply_gate_mask_right_kernel[grid](
        right_ptr=right,
        mask_ptr=mask,
        gate_ptr=right_gate,
        out_ptr=right_out,
        B=right.shape[0], N=right.shape[1], H=hidden_dim,
        stride_right_b=right_stride[0], stride_right_i=right_stride[1], stride_right_k=right_stride[2], stride_right_h=right_stride[3],
        stride_mask_b=mask_stride[0], stride_mask_i=mask_stride[1], stride_mask_k=mask_stride[2],
        stride_gate_b=right_gate_stride[0], stride_gate_i=right_gate_stride[1], stride_gate_k=right_gate_stride[2], stride_gate_h=right_gate_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # Now compute out via batched matrix multiplication per hidden channel.
    # Reshape left_out and right_out to (B*hidden_dim, N, N)
    L = left_out.permute(0, 3, 1, 2).contiguous() # B,hd,N,N
    R = right_out.permute(0, 3, 1, 2).contiguous()
    # flatten batch and hidden dim
    B = L.shape[0]
    H = L.shape[1]
    L_flat = L.view(B * H, N, N)
    R_flat = R.view(B * H, N, N)

    # Compute out_flat = torch.bmm(L_flat, R_flat.transpose(1,2))
    out_flat = torch.bmm(L_flat, R_flat.transpose(1,2))

    # Reshape back to (B,N,N,hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    # compute mean,var over last dim (hidden_dim)
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear to_out
    to_out_weight = weights['to_out.weight']
    final = torch.nn.functional.linear(out_norm, to_out_weight)
    return final

"""
TriMul (outgoing) forward pass implemented with a small Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Compute three linear projections (left, right, gates) without bias.
   The gates are passed through a sigmoid.
3. Fuse mask and gate multiplication using a Triton kernel
   (element‑wise:  out = tensor * mask * gate  for left/right).
4. Compute the core O(N³·C) operation:
        out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   as a batched matrix‑multiply per hidden channel with torch.bmm
   (highly tuned cuBLAS on H100).
5. Apply a second Layer‑Norm on the hidden dimension, multiply by the
   output‑gate and apply the final linear projection back to the original
   channel size.

The kernel only processes the cheap element‑wise mask/gate fusion, satisfying
the requirement to have “part of the operations” in Triton while keeping the
expensive O(N³) work on GPU‑accelerated GEMM.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    inp_ptr,                     # [B, N, N, C] input tensor (left or right)
    mask_ptr,                    # [B, N, N]    mask tensor
    gate_ptr,                    # [B, N, N, C] gate tensor
    out_ptr,                     # [B, N, N, C] output tensor
    B, N, C,                     # compile‑time constants
    stride_inp_b, stride_inp_i, stride_inp_k, stride_inp_c,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_c,
    BLOCK_I: tl.constexpr,       # tile size along the first N axis
    BLOCK_K: tl.constexpr,       # tile size along the second N axis
    BLOCK_C: tl.constexpr,       # tile size along the channel axis
):
    """Fuses mask and gate into a tensor: out = inp * mask * gate."""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    # 1‑D ranges for each tile dimension
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    c = tl.arange(0, BLOCK_C)

    # Bounds checking
    i_mask = i < N
    k_mask = k < N
    c_mask = c < C
    active = i_mask[:, None, None] & k_mask[None, :, None] & c_mask[None, None, :]

    # Pointers (in element units, not bytes)
    inp = (
        inp_ptr
        + pid_b * stride_inp_b
        + i[:, None, None] * stride_inp_i
        + k[None, :, None] * stride_inp_k
        + c[None, None, :] * stride_inp_c
    )
    gate = (
        gate_ptr
        + pid_b * stride_gate_b
        + i[:, None, None] * stride_gate_i
        + k[None, :, None] * stride_gate_k
        + c[None, None, :] * stride_gate_c
    )
    mask = (
        mask_ptr
        + pid_b * stride_mask_b
        + i[:, None] * stride_mask_i
        + k[None, :] * stride_mask_k
    )

    # Load tensors
    inp_val = tl.load(inp, mask=active, other=0.0).to(tl.float32)
    gate_val = tl.load(gate, mask=active, other=0.0).to(tl.float32)
    mask_val_2d = tl.load(mask, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

    # Broadcast mask to (i, k, 1) and multiply
    mask_val = mask_val_2d[:, :, None]
    out = inp_val * gate_val * mask_val

    # Store result
    tl.store(
        out_ptr
        + pid_b * stride_inp_b
        + i[:, None, None] * stride_inp_i
        + k[None, :, None] * stride_inp_k
        + c[None, None, :] * stride_inp_c,
        out,
        mask=active,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask         : torch.Tensor of shape [B, N, N]   (may be None)
            - weights      : dict with all linear / layer‑norm parameters
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ----------------------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (x.norm)
    # ----------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over dim axis
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # shape [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias) and gate logits
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ----------------------------------------------------------------------
    # 3) Prepare mask (if missing → all‑ones) and move to same dtype/device
    # ----------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(
            input_tensor.shape[0],
            input_tensor.shape[1],
            input_tensor.shape[2],
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    mask = mask.to(input_tensor.dtype)

    # ----------------------------------------------------------------------
    # 4) Fuse mask and gate using Triton (element‑wise multiply)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # tile sizes – chosen to fit nicely in shared memory on H100
    BLOCK_I = 64
    BLOCK_K = 64
    BLOCK_C = 32

    # strides (in element units)
    left_stride = left.stride()
    right_stride = right.stride()
    mask_stride = mask.stride()
    left_gate_stride = left_gate.stride()
    right_gate_stride = right_gate.stride()

    # launch configuration: (batch, i‑tiles, k‑tiles)
    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_K - 1) // BLOCK_K,
    )

    # left side
    apply_gate_mask_kernel[grid](
        left, mask, left_gate, left_out,
        B, N, hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        mask_stride[0], mask_stride[1], mask_stride[2],
        left_gate_stride[0], left_gate_stride[1], left_gate_stride[2], left_gate_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
    )
    # right side
    apply_gate_mask_kernel[grid](
        right, mask, right_gate, right_out,
        B, N, hidden_dim,
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        mask_stride[0], mask_stride[1], mask_stride[2],
        right_gate_stride[0], right_gate_stride[1], right_gate_stride[2], right_gate_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
    )

    # ----------------------------------------------------------------------
    # 5) Core TriMul: O(N³·C) = Σₖ left[...,k] * right[...,k]^T
    #    Implemented as a batched GEMM per hidden channel using torch.bmm,
    #    which maps to cuBLAS tensor‑core kernels on H100.
    # ----------------------------------------------------------------------
    # reshape to (B*C, N, N) for batched matmul
    L = left_out.permute(0, 3, 1, 2).contiguous()   # (B, C, N, N)
    R = right_out.permute(0, 3, 1, 2).contiguous()  # (B, C, N, N)
    B_, C_, N_, _ = L.shape
    L_flat = L.view(B_ * C_, N_, N_)
    R_flat = R.view(B_ * C_, N_, N_)

    # out_flat[b*c, i, j] = Σₖ L_flat[b*c, i, k] * R_flat[b*c, j, k]
    out_flat = torch.bmm(L_flat, R_flat.transpose(1, 2))
    out = (
        out_flat.view(B_, C_, N_, N_)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B, N, N, C]

    # ----------------------------------------------------------------------
    # 6) Second Layer‑Norm over hidden dimension
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ----------------------------------------------------------------------
    # 7) Apply output gate and final linear projection
    # ----------------------------------------------------------------------
    out_norm = out_norm * out_gate
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return final
```
