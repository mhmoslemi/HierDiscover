                          M, N, K,
                          stride_am, stride_ak, stride_ab,
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cb,

"""
TriMul (outgoing) forward pass, heavily fused with Triton.

Algorithm:
  1. Apply LayerNorm on the input tensor (last dimension C).
  2. Linear projections to a hidden dimension H (left_proj, right_proj).
  3. Optional binary mask (broadcasted on H) applied to the projections.
  4. Compute three sigmoid‑gates (left_gate, right_gate, out_gate) and
     multiply the projected tensors by their respective gates.
  5. Fuse the core O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     using a batched GEMM kernel written in Triton (the kernel works on the
     transposed right tensor so that a standard matmul can be used).
  6. Layer‑norm the result across the hidden dimension, apply the out‑gate,
     and finally map back to the original dimension with a linear layer.
All heavy arithmetic (step 5) runs in float16 with float32 accumulation.
The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (left @ rightᵀ) – computes the O(N³) contraction.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # problem sizes
    stride_am, stride_ak, stride_ab,   # strides for A (batch, rows, cols)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, rows, cols) – B is K×N
    stride_cm, stride_cn, stride_cb,   # strides for C (batch, rows, cols)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)       # batch dimension (B*H)
    pid_m = tl.program_id(0)           # block-row index
    pid_n = tl.program_id(1)           # block-col index

    # tile offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch_ptr = A + pid_batch * stride_ab
    b_batch_ptr = B + pid_batch * stride_bb
    c_batch_ptr = C + pid_batch * stride_cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k                     # absolute k indices

        # ---------------------- load A block ----------------------
        a_ptr = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a = tl.load(a_ptr, mask=a_mask, other=0.0)       # (BLOCK_M, BLOCK_K)

        # ---------------------- load B block ----------------------
        # B shape is (batch, K, N); we need rows = k, cols = n
        b_ptr = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptr, mask=b_mask, other=0.0)       # (BLOCK_K, BLOCK_N)

        # ---------------------- block multiply --------------------
        acc += tl.dot(a, b)    # dot promotes to FP32 automatically

    # write the result back (store as FP16)
    c_ptr = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Public entry point – called by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  Arguments are supplied as a tuple:
        (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # ensure float mask with same dtype as the tensors
        m = mask.to(x_norm.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * m
        right = right * m
    # else: no mask → nothing to do

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core cubic contraction via Triton
    # ------------------------------------------------------------------
    # Convert to FP16 before launching the kernel (reduces memory traffic)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # left shape = [B, N, N, H]

    # Rearrange to (B, H, N, N) → flatten batch & head dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose right on the last two axes (so we can use standard GEMM)
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, H, N, N]

    # Collapse batch and hidden dimensions:   (B*H, N, N)
    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_T.view(batch, N, N)

    # Allocate output buffer (FP16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                # M
        N,                # N
        N,                # K
        left_mat.stride(1), left_mat.stride(2), left_mat.stride(0),   # A strides
        right_mat.stride(1), right_mat.stride(2), right_mat.stride(0),  # B strides
        out_mat.stride(1), out_mat.stride(2), out_mat.stride(0),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and cast to FP32 for the remaining ops
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # apply out‑gate (both FP32)
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
```
