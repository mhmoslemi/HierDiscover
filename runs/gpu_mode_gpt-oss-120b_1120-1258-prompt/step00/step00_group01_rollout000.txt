"""
Custom Triton‑accelerated implementation of the “outgoing” TriMul operator.

The heavy N³ contraction is performed as a batched matrix multiplication
(left @ rightᵀ) using torch.matmul (which internally calls cuBLAS).  
A Triton kernel fuses the mask broadcast, per‑channel gating and element‑wise
multiplication for the left and right projected tensors.  The remaining
operations (LayerNorms, sigmoid‑gates and the final linear projection) are
performed with PyTorch’s highly‑optimised kernels.

This satisfies the requirement of using a Triton kernel while keeping the
overall implementation simple and correct for all supported shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: fuse mask broadcasting, gating and element‑wise multiplication
# ----------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_kernel(
    left_ptr,          # float*  left tensor (B,N,N,H) – in‑place
    right_ptr,         # float*  right tensor (B,N,N,H) – in‑place
    mask_ptr,          # float*  mask tensor (B,N,N)  (0/1) broadcasted over H
    left_gate_ptr,     # float*  left gate tensor (B,N,N,H)
    right_gate_ptr,    # float*  right gate tensor (B,N,N,H)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # ------- decode linear index into (b,i,j,_) coordinates ----------
    # total index = b·(N·N·H) + i·(N·H) + j·H + h
    # We need b,i,j to index the 3‑D mask.
    nnh = N * N * H
    nh  = N * H
    b   = tl.div(offs, nnh)
    rest = offs - b * nnh
    i   = tl.div(rest, nh)
    rest2 = rest - i * nh
    j   = tl.div(rest2, H)
    # (h = rest2 - j*H) is not needed for mask

    # mask index (flattened 3‑D) and load
    mask_idx = (b * N + i) * N + j
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)   # shape (BLOCK_SIZE,)

    # load left/right and gates
    left_val      = tl.load(left_ptr + offs,      mask=mask)
    right_val     = tl.load(right_ptr + offs,     mask=mask)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate_val= tl.load(right_gate_ptr + offs,mask=mask)

    # apply:  tensor = tensor * mask * gate
    new_left  = left_val  * mask_val * left_gate_val
    new_right = right_val * mask_val * right_gate_val

    # store results back in‑place
    tl.store(left_ptr  + offs, new_left,  mask=mask)
    tl.store(right_ptr + offs, new_right, mask=mask)


def _apply_mask_and_gate(left, right, mask, left_gate, right_gate, block_size: int = 65536):
    """
    In‑place Fusion of:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    where mask is broadcasted over the hidden dimension H.
    """
    assert left.shape == right.shape == left_gate.shape == right_gate.shape
    B, N, _, H = left.shape
    total = B * N * N * H

    # make all tensors contiguous for a simple linear layout
    left      = left.contiguous()
    right     = right.contiguous()
    mask      = mask.contiguous()
    left_gate = left_gate.contiguous()
    right_gate= right_gate.contiguous()

    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    _mask_gate_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        B,
        N,
        H,
        BLOCK_SIZE=block_size,
    )
    # tensors are modified in place – nothing to return


# ----------------------------------------------------------------------------- #
# Main entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:         torch.Tensor [B, N, N] (binary or all‑ones)
            - weights:      dict of torch.Tensors with the same keys as the
                            reference PyTorch implementation.
            - config:       dict containing at least "dim", "hidden_dim" and
                            optional "nomask" flag.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------------- #
    (inp, mask, weights, config) = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    device = inp.device
    dtype  = inp.dtype

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (per last dimension)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)    # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 3. Gating values (sigmoid of linear projection)
    # --------------------------------------------------------------------- #
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 4. Apply mask + gating via Triton (in‑place on left/right)
    # --------------------------------------------------------------------- #
    if nomask:
        # No mask – treat it as all‑ones (float) to keep kernel simple
        mask_f = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # Fuse mask, left_gate and right_gate
    _apply_mask_and_gate(left, right, mask_f, left_gate, right_gate)

    # --------------------------------------------------------------------- #
    # 5. N³ contraction: left @ rightᵀ  (batched GEMM)
    # --------------------------------------------------------------------- #
    # Reshape to (B, H, N, N) for batched matrix multiply
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # Right needs to be transposed on the last two dims before the matmul
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # Bring back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 6. Post‑processing: LayerNorm → out_gate → final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate  # element‑wise gating

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)        # (B, N, N, dim)

    # Ensure final dtype is float32 as required
    if out.dtype != torch.float32:
        out = out.float()
    return out
