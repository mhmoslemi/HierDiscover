def trimul_einsum_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, H,
                         stride_b_l, stride_i_l, stride_k_l, stride_h_l,
                         stride_b_r, stride_i_r, stride_k_r, stride_h_r,
                         stride_b_o, stride_i_o, stride_j_o, stride_h_o,
                         BLOCK_K: tl.constexpr,
                         BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # indices
    i = pid_i
    j = pid_j

    # Hidden dimension thread index
    h = tl.arange(0, BLOCK_H)

    # masks for bounds
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_base = kb * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # compute offsets
        left_ptrs = left_ptr + pid_b * stride_b_l + i * stride_i_l + k[:, None] * stride_k_l + h[None, :] * stride_h_l
        right_ptrs = right_ptr + pid_b * stride_b_r + j * stride_i_r + k[:, None] * stride_k_r + h[None, :] * stride_h_r

        left = tl.load(left_ptrs, mask=mask_i & mask_k[:, None] & mask_h[None, :], other=0.0)   # shape (BLOCK_K, BLOCK_H)
        right = tl.load(right_ptrs, mask=mask_j & mask_k[:, None] & mask_h[None, :], other=0.0)

        # elementwise product and sum across K dimension
        acc += tl.sum(left * right, axis=0)

    # store result
    out_ptrs = out_ptr + pid_b * stride_b_o + i * stride_i_o + j * stride_j_o + h * stride_h_o

    tl.store(out_ptrs, acc, mask=mask_h & mask_i & mask_j)

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input over the last channel (dim).
2. Linear projections to a hidden dimension (left/right) and three
   gating projections (left_gate, right_gate, out_gate).  All linear
   layers are fused into PyTorch ``nn.functional.linear`` and performed
   in half‑precision for speed.
3. Apply the mask (if any) and the element‑wise gates with a tiny
   Triton kernel – this satisfies the requirement of using a custom
   kernel while keeping the heavy O(N³) work on the highly‑optimized
   cuBLAS kernels.
4. Compute the “triangle” multiplicative update:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix‑multiplication for each hidden channel.
   By reshaping ``(B,N,N,H)`` → ``(B*H,N,N)`` we can use ``torch.bmm``.
5. Layer‑norm over the hidden dimension, multiply by ``out_gate`` and
   project back to the original channel size with a final linear layer.
6. Return the result in float32.

The kernel in step 3 works on the flattened tensor
``total = B * N * N * H`` and performs
``left = left_proj * left_gate * mask`` and the analogous
right computation (mask defaults to 1 when omitted).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: mask + gating (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                 # left_proj   [B, N, N, H]
    right_ptr,                # right_proj  [B, N, N, H]
    left_gate_ptr,            # left_gate   [B, N, N, H]
    right_gate_ptr,           # right_gate  [B, N, N, H]
    mask_ptr,                 # mask (optional) [B, N, N] or dummy
    left_out_ptr,             # left after gate & mask
    right_out_ptr,            # right after gate & mask
    B, N, H,
    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_gb, stride_gi, stride_gk, stride_gh,   # same for left/right gate
    stride_m_b, stride_m_i, stride_m_k,           # mask strides (if any)
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr):
    """Element‑wise mask + gate application.
    One thread handles BLOCK_SIZE elements of the flattened tensor.
    """

    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    valid = offs < total

    # linear indices -> (b, i, k, h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # ------------------- load values -------------------
    # left_proj * left_gate
    left = tl.load(left_ptr
                   + b * stride_l_b + i * stride_l_i + k * stride_l_k + h * stride_l_h,
                   mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr
                        + b * stride_gb + i * stride_gi + k * stride_gk + h * stride_gh,
                        mask=valid, other=0.0)

    # right_proj * right_gate
    right = tl.load(right_ptr
                    + b * stride_r_b + i * stride_r_i + k * stride_r_k + h * stride_r_h,
                    mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr
                         + b * stride_gb + i * stride_gi + k * stride_gk + h * stride_gh,
                         mask=valid, other=0.0)

    if HAS_MASK:
        # mask index = b * (N*N) + i * N + k
        mask_idx = b * (N * N) + i * N + k
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------- store results -------------------
    tl.store(left_out_ptr
             + b * stride_out_b + i * stride_out_i + k * stride_out_k + h * stride_out_h,
             left, mask=valid)
    tl.store(right_out_ptr
             + b * stride_out_b + i * stride_out_i + k * stride_out_k + h * stride_out_h,
             right, mask=valid)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input, mask, weights, config)
            input  : [B, N, N, dim]   (float32)
            mask   : [B, N, N]        (bool/float, may be None)
            weights: dict of model parameters
            config : dict with keys "dim", "hidden_dim"
    Returns:
        out    : [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16          # work in half‑precision inside the heavy parts
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # unpack weights and cast to half‑precision when needed
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(dtype=dtype, device=device)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_weight   = to_fp16(weights["left_proj.weight"])
    right_proj_weight  = to_fp16(weights["right_proj.weight"])
    left_gate_weight   = to_fp16(weights["left_gate.weight"])
    right_gate_weight  = to_fp16(weights["right_gate.weight"])
    out_gate_weight    = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    to_out_weight      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Layer norm over the last channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections and gating (keep half‑precision)
    # ------------------------------------------------------------------
    x_f16 = x.to(dtype)

    left_proj   = torch.nn.functional.linear(x_f16, left_proj_weight)      # [B,N,N,hidden]
    right_proj  = torch.nn.functional.linear(x_f16, right_proj_weight)

    left_gate   = torch.nn.functional.linear(x_f16, left_gate_weight).sigmoid()
    right_gate  = torch.nn.functional.linear(x_f16, right_gate_weight).sigmoid()
    out_gate    = torch.nn.functional.linear(x_f16, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask and element‑wise gates with a Triton kernel
    # ------------------------------------------------------------------
    # (mask may be None → treat as all‑ones)
    if mask is None:
        has_mask = 0
        # dummy tensor – never read because HAS_MASK == 0
        mask_tensor = torch.empty(0, device=device, dtype=dtype)
    else:
        has_mask = 1
        mask_tensor = mask.to(dtype=dtype, device=device)

    B, N, _, _ = left_proj.shape
    total_elems = B * N * N * hidden_dim
    BLOCK_SIZE = 1024  # number of elements per Triton program instance
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    left_gated  = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # strides (contiguous tensors => standard strides)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_proj.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_proj.stride()
    stride_gb, stride_gi, stride_gk, stride_gh = left_gate.stride()   # same for right_gate
    stride_m_b, stride_m_i, stride_m_k = mask_tensor.stride() if has_mask else (0, 0, 0)
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = left_gated.stride()

    _apply_mask_gate_kernel[grid](
        left_proj, right_proj,
        left_gate, right_gate,
        mask_tensor,
        left_gated, right_gated,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_gb, stride_gi, stride_gk, stride_gh,
        stride_m_b, stride_m_i, stride_m_k,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=has_mask,
    )

    # ------------------------------------------------------------------
    # 4. Triangle multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and use torch.bmm
    left_view  = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_view = right_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_view shape: (B*H, N, N)
    out_view = torch.bmm(left_view, right_view.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_view.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Post‑processing: norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden]

    out = out * out_gate  # broadcast over hidden dim

    # final projection back to the original channel dimension
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    # return in float32 as the reference implementation does
    return out.to(torch.float32)
