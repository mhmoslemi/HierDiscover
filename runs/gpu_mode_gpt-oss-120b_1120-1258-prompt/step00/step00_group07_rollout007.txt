def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"] # shape dim
    norm_bias = weights["norm.bias"]
    input_norm = layernorm_torch(input_tensor, norm_weight, norm_bias, eps)

    # linear projections
    left_proj = F.linear(input_norm, weights["left_proj.weight"])
    right_proj = F.linear(input_norm, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1) # [bs,N,N,1]
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_
    # gates
    left_gate = torch.sigmoid(F.linear(input_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(input_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(input_norm, weights["out_gate.weight"]))
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # shape permutations for GEMM
    bs, N, _, _ = input_tensor.shape # (?) actually dim trending: input_tensor shape [bs,N,N,dim]
    # reshape left_proj: [bs, N, N, hidden_dim] -> [bs, hidden_dim, N, N]
    left = left_proj.permute(0, 3, 1, 2).contiguous()
    right = right_proj.permute(0, 3, 1, 2).contiguous()
    # flatten batch*hidden_dim
    M = bs * hidden_dim
    left_flat = left.view(M, N, N).contiguous()
    right_flat = right.view(M, N, N).contiguous()

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=left_flat.dtype, device=left_flat.device)

    # launch triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    # reshape back
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape [bs,N,N,hidden_dim]
    # to_out_norm
    out = layernorm_torch(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)
    out = out * out_gate
    # final linear
    out = F.linear(out, weights["to_out.weight"])
    return out

def layernorm_torch(x, weight, bias, eps=1e-5):
    # x shape (..., C)
    # compute mean, var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias
```

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Project the normalized tensor to a hidden dimension using three linear
   layers (left_proj, right_proj, gate_proj) and apply a mask (if any).
3. Compute element‑wise sigmoid gates and multiply them into the projected
   tensors.
4. For each batch element and each hidden channel, compute

      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   which is a batched matrix multiplication:
   out[b,:, :,h] = left[b,:, :,h] @ right[b,:, :,h].T

   This heavy O(N³) work is performed by a custom Triton kernel that
   tiles the i, j, k dimensions (BLOCK_M × BLOCK_N × BLOCK_K) and uses
   fp16 arithmetic with fp32 accumulation.
5. Apply a second Layer‑norm over the hidden dimension, multiply by the
   output gate, and finally linearly project back to the original
   dimensionality.

The implementation uses torch for the inexpensive operations (layer‑norm,
linear, sigmoid, masking) and Triton for the dominant multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Simple LayerNorm over the last dimension.
    x : (..., C)   float32
    weight, bias : (C,)
    Returns same shape as x.
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


@triton.jit
def _trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_i, stride_l_k,
    stride_r_m, stride_r_i, stride_r_k,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel for TriMul.
    Computes for each batch‐hidden index m:
        out[m,i,j] = Σ_k left[m,i,k] * right[m,j,k]
    where left and right have shape (M, N, N).
    """
    pid_m = tl.program_id(0)               # which (batch * hidden) we are processing
    pid_i = tl.program_id(1)               # i‑tile
    pid_j = tl.program_id(2)               # j‑tile

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    i_mask = offs_i < N
    j_mask = offs_j < N
    k_mask = offs_k < N

    # ---------- load left tile (i,k) ----------
    left_ptrs = (left_ptr
                 + pid_m * stride_l_m
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
    left = tl.load(left_ptrs,
                   mask=i_mask[:, None] & k_mask[None, :],
                   other=0.0).to(tl.float32)

    # ---------- load right tile (k,j) ----------
    # note: we want right[k,j] (k as row, j as column)
    right_ptrs = (right_ptr
                  + pid_m * stride_r_m
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_i)
    right = tl.load(right_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0).to(tl.float32)

    # ---------- dot product ----------
    acc = tl.dot(left, right)          # (BLOCK_M, BLOCK_N), fp32

    # ---------- store result ----------
    out_ptrs = (out_ptr
                + pid_m * stride_o_m
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [bs, N, N, dim] (float32, on GPU)
        mask         : torch.Tensor [bs, N, N] (float32 or bool) or None
        weights      : dict of torch.Tensors (see below)
        config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (float32, on GPU)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------
    # 1) First LayerNorm over `dim`
    # --------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = _layer_norm(input_tensor, norm_weight, norm_bias, eps=eps)

    # --------------------------------------------------------------------
    # 2) Linear projections (no bias) + optional mask
    # --------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # ensure mask broadcast matches data dtype
        mask = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------------
    # 3) Compute sigmoid gates and apply them
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------------
    # 4) Batched GEMM across the hidden dimension (Triton)
    # --------------------------------------------------------------------
    bs, N, _, _ = input_tensor.shape

    # bring hidden dim to the second axis for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # merge batch and hidden dimensions: M = B * hidden
    M = bs * hidden_dim
    left_flat = left.view(M, N, N).contiguous()    # [M, N, N]
    right_flat = right.view(M, N, N).contiguous()

    # use fp16 for the heavy compute
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # tile sizes (tuned for H100, fp16)
    BLOCK_M = 64   # i dimension
    BLOCK_N = 64   # j dimension
    BLOCK_K = 32   # reduction dimension

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trmul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        M,
        N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # cast back to float32 for the next layers
    out = out_fp16.to(torch.float32)

    # reshape back to original layout [B, N, N, hidden]
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 5) Second LayerNorm over hidden dimension, gating, final projection
    # --------------------------------------------------------------------
    out = _layer_norm(out,
                      weights["to_out_norm.weight"],
                      weights["to_out_norm.bias"],
                      eps=eps)

    out = out * out_gate                     # element‑wise gating (same shape)

    # final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
