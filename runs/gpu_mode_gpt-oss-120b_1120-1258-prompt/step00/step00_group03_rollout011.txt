def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   batch, seq_len, hidden_dim,
                   left_stride_batch, left_stride_i, left_stride_k, left_stride_h,
                   right_stride_batch, right_stride_i, right_stride_k, right_stride_h,
                   out_stride_batch, out_stride_i, out_stride_j, out_stride_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
                   BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)

    # offsets for batch
    batch_offset = pid_b * left_stride_batch

    # base pointers
    a_ptr = left_ptr + batch_offset
    b_ptr = right_ptr + pid_b * right_stride_batch
    c_ptr = out_ptr + pid_b * out_stride_batch

    # Masks
    mask_i = i_idx < seq_len
    mask_j = j_idx < seq_len
    mask_h = tl.arange(0, BLOCK_H) < hidden_dim

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # iterate over k dimension
    for k in range(seq_len):
        # load left vectors for all i and h
        # left ptr: a_ptr + i_idx[:, None] * left_stride_i + k * left_stride_k + tl.arange(0, BLOCK_H)[None, :] * left_stride_h
        left_ptrs = a_ptr + i_idx[:, None] * left_stride_i + k * left_stride_k + tl.arange(0, BLOCK_H)[None, :] * left_stride_h
        left_vec = tl.load(left_ptrs, mask=mask_i[:, None] & mask_h[None, :], other=0.0)  # shape (BLOCK_I, BLOCK_H)

        right_ptrs = b_ptr + j_idx[:, None] * right_stride_i + k * right_stride_k + tl.arange(0, BLOCK_H)[None, :] * right_stride_h
        right_vec = tl.load(right_ptrs, mask=mask_j[:, None] & mask_h[None, :], other=0.0)  # shape (BLOCK_J, BLOCK_H)

        # accumulate
        # Broadcast left_vec across j, right across i
        # left_vec[:, None, :] shape (BLOCK_I, 1, BLOCK_H)
        # right_vec[None, :, :] shape (1, BLOCK_J, BLOCK_H)
        acc += left_vec[:, None, :] * right_vec[None, :, :]

    # store result
    out_ptrs = c_ptr + i_idx[:, None, None] * out_stride_i + j_idx[None, :, None] * out_stride_j + tl.arange(0, BLOCK_H)[None, None, :] * out_stride_h
    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_store)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) # if mask is provided maybe None
    device = input_tensor.device
    eps = 1e-5 # typical for layernorm

    # LayerNorm: x = (x - mean) / sqrt(var + eps) * weight + bias
    # Compute mean/var across last dim (dim)
    # Use torch.nn.functional.layer_norm directly: torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Since we have weight and bias from weights dict.
    # Format: x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections: left_proj (dim -> hidden_dim)
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # shape [hidden_dim, dim]

    # Perform linear: left = torch.nn.functional.linear(x, left_proj_weight) - bias=None
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling: mask may be None or provided.
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Compute gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    out = torch.empty_like(left)  # shape [B,N,N, hidden_dim]

    # Get strides
    left_strides = left.stride()
    right_strides = right.stride()
    out_strides = out.stride()

    right = right.contiguous()
    out = out.contiguous()

    BLOCK_J = 16
    BLOCK_H = 16  # hidden channel tile

    # we will pass hidden_dim to kernel and handle partial tile via mask

    # We'll loop over h_offset in python.

    # We'll adjust pointers: left_ptr = left.data_ptr() + h_offset * left_strides[3] * element_size
    # same for right and out.

    import triton.language as tl

    @triton.jit
    def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                       batch, seq_len, hidden_dim,
                       left_stride_batch, left_stride_i, left_stride_k, left_stride_h,
                       right_stride_batch, right_stride_i, right_stride_k, right_stride_h,
                       out_stride_batch, out_stride_i, out_stride_j, out_stride_h,
                       BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
        pid_b = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)

        i_start = pid_i * BLOCK_I
        j_start = pid_j * BLOCK_J

        i_idx = i_start + tl.arange(0, BLOCK_I)
        j_idx = j_start + tl.arange(0, BLOCK_J)

        # batch offsets
        a_ptr = left_ptr + pid_b * left_stride_batch
        b_ptr = right_ptr + pid_b * right_stride_batch
        c_ptr = out_ptr + pid_b * out_stride_batch

        mask_i = i_idx < seq_len
        mask_j = j_idx < seq_len
        mask_h = tl.arange(0, BLOCK_H) < hidden_dim

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections:
   * left  = X·W_left      (C → H, no bias)
   * right = X·W_right     (C → H, no bias)
3. (Optional) apply a pairwise mask M ∈ {0,1}^{B×N×N}.
4. Compute element‑wise gates (sigmoid):
   left_gate  = sigmoid(X·W_left_gate)
   right_gate = sigmoid(X·W_right_gate)
   out_gate   = sigmoid(X·W_out_gate)
   left  ← left  * left_gate
   right ← right * right_gate
5. Core O(N³·H) operation – “outgoing” TriMul:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a tiled Triton kernel that:
   * processes a (BLOCK_I × BLOCK_J) tile of (i,j) pairs per program,
   * iterates over the sequence dimension k,
   * loads vectors of size BLOCK_H for all i (or j) at each k,
   * broadcasts and accumulates the outer product across i and j.
   The hidden dimension H is tiled with BLOCK_H (e.g. 16) – the kernel
   is launched repeatedly to cover the full hidden size.
6. Apply a second LayerNorm on the hidden result, multiply by `out_gate`,
   and project back to the original channel size C with a final linear layer.

The implementation fuses the O(N³·H) work into a custom Triton kernel,
while the surrounding linear/LayerNorm/gate operations use highly‑optimized
PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, seq_len, hidden_dim,
    left_stride_batch, left_stride_i, left_stride_k, left_stride_h,
    right_stride_batch, right_stride_i, right_stride_k, right_stride_h,
    out_stride_batch, out_stride_i, out_stride_j, out_stride_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape [B, N, N, H] and out the same.
    The kernel processes a (BLOCK_I × BLOCK_J) tile of (i,j) for a
    fixed hidden‑tile of size BLOCK_H.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile row (i)
    pid_j = tl.program_id(2)          # tile col (j)

    # Global i/j indices for this tile
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    i_idx = i_base + tl.arange(0, BLOCK_I)
    j_idx = j_base + tl.arange(0, BLOCK_J)

    # Offsets to the start of the current batch
    a_ptr = left_ptr  + pid_b * left_stride_batch
    b_ptr = right_ptr + pid_b * right_stride_batch
    c_ptr = out_ptr   + pid_b * out_stride_batch

    # Bounds masks
    mask_i = i_idx < seq_len
    mask_j = j_idx < seq_len
    mask_h = tl.arange(0, BLOCK_H) < hidden_dim

    # Accumulator for the tile (float32 for numerical stability)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Iterate over the shared sequence dimension k
    for k in range(seq_len):
        # Load left[i, k, h]  -> shape (BLOCK_I, BLOCK_H)
        left_ptrs = a_ptr + i_idx[:, None] * left_stride_i \
                    + k * left_stride_k \
                    + tl.arange(0, BLOCK_H)[None, :] * left_stride_h
        left_vec = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_h[None, :],
            other=0.0
        )   # (BLOCK_I, BLOCK_H)

        # Load right[j, k, h] -> shape (BLOCK_J, BLOCK_H)
        right_ptrs = b_ptr + j_idx[:, None] * right_stride_i \
                     + k * right_stride_k \
                     + tl.arange(0, BLOCK_H)[None, :] * right_stride_h
        right_vec = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_h[None, :],
            other=0.0
        )   # (BLOCK_J, BLOCK_H)

        # Outer‑product and accumulate:
        # left_vec[:, None, :] -> (BLOCK_I, 1, BLOCK_H)
        # right_vec[None, :, :] -> (1, BLOCK_J, BLOCK_H)
        acc += left_vec[:, None, :] * right_vec[None, :, :]

    # Store the accumulated block
    out_ptrs = c_ptr + i_idx[:, None, None] * out_stride_i \
               + j_idx[None, :, None] * out_stride_j \
               + tl.arange(0, BLOCK_H)[None, None, :] * out_stride_h
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" …

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]    # [hidden_dim, dim]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        # mask shape: [B, N, N] -> broadcast on hidden dim
        mask = mask.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch."

    # Allocate output of the core TriMul (still hidden_dim)
    out_hidden = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # Strides (in element counts, not bytes)
    left_s = left.stride()
    right_s = right.stride()
    out_s = out_hidden.stride()

    # ------------------------------------------------------------------
    # 5. Triton kernel launch (tile over hidden dimension)
    # ------------------------------------------------------------------
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_H = 16  # compile‑time tile size for the hidden channels

    n_block_i = (N + BLOCK_I - 1) // BLOCK_I
    n_block_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B, n_block_i, n_block_j)  # (batch, i‑tiles, j‑tiles)

    element_size = left.element_size()  # bytes per element (float32 → 4)

    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_h = min(BLOCK_H, hidden_dim - h_start)

        # Pointer arithmetic for the hidden slice
        left_ptr = left.data_ptr() + h_start * left_s[3] * element_size
        right_ptr = right.data_ptr() + h_start * right_s[3] * element_size
        out_ptr = out_hidden.data_ptr() + h_start * out_s[3] * element_size

        tri_mul_outgoing_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B,
            N,
            cur_h,
            left_s[0], left_s[1], left_s[2], left_s[3],
            right_s[0], right_s[1], right_s[2], right_s[3],
            out_s[0], out_s[1], out_s[2], out_s[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # [B, N, N, hidden_dim]

    out_hidden = out_hidden * out_gate   # broadcasted element‑wise

    to_out_weight = weights["to_out.weight"]   # [dim, hidden_dim]
    output = torch.nn.functional.linear(out_hidden, to_out_weight)  # [B, N, N, dim]

    return output
