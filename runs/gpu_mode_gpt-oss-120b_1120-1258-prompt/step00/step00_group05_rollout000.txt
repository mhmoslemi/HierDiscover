"""
TriMul “outgoing” kernel (AlphaFold3 style) – forward only.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (last dimension = dim).
2. Linear projections (left/right) and three sigmoid gates (left, right, out)
   – all without bias.
3. Optional mask (broadcasted on the hidden dimension).
4. Fuse left/right with their respective gates.
5. Compute the expensive tensor contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each (batch, hidden) pair:
       out_z = left_z  @  right_zᵀ   with   z = b * hidden_dim + d
   Implemented in Triton for high throughput (fp16 compute, fp32 accumulation).
6. Apply a second LayerNorm over the hidden dimension, multiply by the out‑gate,
   and project back to the original dimension.

Only the mat‑mul (step 5) runs in Triton; everything else uses
torch.nn.functional for simplicity.  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (Z, M, K) @ (Z, N, K)ᵀ → (Z, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to seq_len)
    stride_az, stride_am, stride_ak,   # strides for A (z,m,k)
    stride_bz, stride_bk, stride_bn,   # strides for B (z,k,n)  – note B is logically transposed
    stride_cz, stride_cm, stride_cn,   # strides for C (z,m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)               # batch*hidden index
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block column

    # Offsets of the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers for the current tile
        a_ptrs = a_ptr + pid_z * stride_az + \
                 offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_z * stride_bz + \
                 offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)      # fp32 accumulation

    # Write back the result (store as fp16)
    c_ptrs = c_ptr + pid_z * stride_cz + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 or bool) or None
        - weights      : dict of model parameters (see code)
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    bs, N, N2, dim = x.shape
    assert N == N2, "Input must be square in the sequence dimensions."
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over `dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float (0/1) and broadcastable
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Apply elementwise gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton batched mat‑mul (fp16 compute, fp32 accumulation)
    #    left_z @ right_zᵀ  -> out_z   for each (batch, hidden) pair.
    # ------------------------------------------------------------------
    # Cast to fp16 for speed
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) -> flatten batch*hidden => Z
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    Z = bs * hidden_dim
    left_mat  = left_perm.view(Z, N, N)   # (Z, M, K)
    right_mat = right_perm.view(Z, N, N)  # (Z, N, K) – we will treat as (K, N) inside kernel
    out_mat   = torch.empty_like(left_mat, dtype=torch.float16)  # (Z, M, N)

    # Block sizes – tuned for H100 fp16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        Z,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_mat,               # a_ptr
        right_mat,              # b_ptr (interpreted as K×N)
        out_mat,                # c_ptr
        N, N, N,               # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),   # A strides
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),# B strides (k, n)
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # enough parallelism for H100
    )

    # Convert back to float32 for the remaining ops
    out = out_mat.to(torch.float32)

    # Reshape back to [B, N, N, hidden_dim]
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + output gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # apply out‑gate
    out = F.linear(out, weights["to_out.weight"])   # back to `dim`

    return out
