def mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    hidden_dim: tl.constexpr, total_elems: tl.constexpr, BLOCK: tl.constexpr

    block_start = pid * BLOCK
    offsets = block_start + tl.arange(0, BLOCK)
    mask = offsets < total_elems
    # load left,right,...
    left = tl.load(left_ptr + offsets, mask=mask, other=0.0)
    right = tl.load(right_ptr + offsets, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offsets, mask=mask, other=0.0)

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # we also may have config['epsilon'] optional

    device = input_tensor.device
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-6

    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections and gates
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask and gates via Triton
    if mask is None:

        mask_tensor = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=input_tensor.dtype, device=device)
    else:
        mask_tensor = mask

    # make tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()
    # flatten
    total = left.numel()
    # Ensure it's multiple of BLOCK maybe padding; we can allocate enough blocks
    # Use Triton API for launching

    # batched matmul per hidden_dim

    B, N, _, _ = left.shape  # left shape (B,N,N,hidden_dim)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, hidden_dim, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)
    # compute matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # shape (B*hidden_dim, N, N)

    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    right_gate = right_gate.contiguous()

        left,
        right,
        left_gate,
        right_gate,
        mask_tensor,

        total_elems=total_elements,
        BLOCK=BLOCK,
    )

"""
custom_kernel: Triton‑accelerated forward pass of the **outgoing** Triangle
Multiplicative Update (TriMul) used in AlphaFold‑3 and related protein models.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor `x` over its last dimension (`dim`).
2. Compute three linear projections of `x` (left, right and three gates) using
   the supplied weights. Apply a sigmoid to the three gate outputs.
3. Fuse the mask, left‑gate and right‑gate element‑wise with a Triton kernel.
   The kernel works on the flattened tensors and broadcasts the 3‑D mask across
   the hidden dimension.
4. Perform the core “tri‑multiply”:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication
   `left_d @ right_dᵀ` for every hidden channel `d`.  We reshape the tensors so
   that the hidden dimension becomes part of the batch and call `torch.bmm`,
   which uses cuBLAS on the H100.
5. Apply a second layer‑norm over the hidden dimension, multiply by the
   output‐gate, and finish with a linear projection back to the original
   `dim`.
6. Return the resulting tensor of shape `[B, N, N, dim]`.

Only the mask‑gate fusion is written in Triton; the remaining operations use
high‑performance PyTorch kernels (layer‑norm, linear layers, and batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: in‑place mask + gate multiplication (broadcast mask across
# the hidden dimension).  All tensors are flattened to 1‑D contiguous arrays.
# ----------------------------------------------------------------------------- #
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,                # float32*  [B*N*N*H]
    right_ptr,               # float32*  [B*N*N*H]
    left_gate_ptr,           # float32*  [B*N*N*H]
    right_gate_ptr,          # float32*  [B*N*N*H]
    mask_ptr,                # float32*  [B*N*N]   (broadcasted)
    hidden_dim: tl.constexpr,
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load raw tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask: one mask value per (b,i,j) is reused for all hidden_dim entries
    mask_idx = offs // hidden_dim                     # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Apply mask and gates (order does not matter for element‑wise multiplication)
    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    # Store the gated results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor: torch.Tensor [B, N, N, dim]
        - mask_tensor : torch.Tensor [B, N, N] or None
        - weights     : dict of torch.Tensor containing all linear/LN parameters
        - config      : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 2) Linear projections and gates (all in FP32)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 3) Apply mask and gates with a Triton kernel
    # --------------------------------------------------------------------- #
    if mask_tensor is None or mask_tensor.numel() == 0:
        # No mask → treat as all‑ones (no effect)
        mask_tensor = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    else:
        mask_tensor = mask_tensor.to(dtype=input_tensor.dtype, device=input_tensor.device)

    # Ensure contiguous storage; flatten to 1‑D for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()

    total_elements = left.numel()                     # B * N * N * H
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total_elements, meta["BLOCK"]),)

    # Triton launch – in‑place modification of `left` and `right`
    apply_mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_tensor,
        hidden_dim=hidden_dim,
        total_elems=total_elements,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 4) Core Tri‑multiply: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiplication over the hidden axis.
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape                     # left shape = [B,N,N,H]
    # Bring the hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * hidden_dim, N, N)     # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]

    # Batched GEMM: (B*H) × (N×N) @ (N×N)ᵀ  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5) Output LayerNorm over hidden_dim, gate, and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate                     # element‑wise gating

    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
```
