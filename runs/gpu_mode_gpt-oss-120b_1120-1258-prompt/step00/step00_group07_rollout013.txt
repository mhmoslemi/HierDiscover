def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, H,
                           stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                           stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                           stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                           BLOCK_M: tl.constexpr,
                           BLOCK_N: tl.constexpr,
                           BLOCK_K: tl.constexpr,
                           BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    total = B * num_i * num_j
    # compute coordinates
    b = pid // (num_i * num_j)
    pid_ij = pid % (num_i * num_j)
    i_tile = pid_ij // num_j
    j_tile = pid_ij % num_j

    # offsets within block
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # masks for out-of-bounds

        mask_j = offs_j < N
        mask_k = offs_k < N
        mask_d = tl.arange(0, BLOCK_D) < H

                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + tl.arange(0, BLOCK_D)[None, None, :] * stride_l_h

        right_ptrs = right_ptr + b * stride_r_b \
                    + offs_j[:, None, None] * stride_r_i \
                    + offs_k[None, :, None] * stride_r_k \

    mask_store_j = offs_j < N
    mask_store_d = tl.arange(0, BLOCK_D) < H

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(l_ptr, r_ptr, o_ptr,
                           B, N, H,
                           stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                           stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                           stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                           BLOCK_M: tl.constexpr,
                           BLOCK_N: tl.constexpr,
                           BLOCK_K: tl.constexpr,
                           BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    # compute batch index
    b = pid // (num_i * num_j)
    pid_ij = pid % (num_i * num_j)
    i_tile = pid_ij // num_j
    j_tile = pid_ij % num_j

    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for out-of-bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N
        mask_d = tl.arange(0, BLOCK_D) < H

        # pointers
        l_ptrs = l_ptr + b * stride_l_b \
                 + offs_i[:, None, None] * stride_l_i \
                 + offs_k[None, :, None] * stride_l_k \
                 + tl.arange(0, BLOCK_D)[None, None, :] * stride_l_h

        r_ptrs = r_ptr + b * stride_r_b \
                 + offs_j[:, None, None] * stride_r_i \
                 + offs_k[None, :, None] * stride_r_k \
                 + tl.arange(0, BLOCK_D)[None, None, :] * stride_r_h

        # load tiles
        l_tile = tl.load(l_ptrs,
                         mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                         other=0.0)
        r_tile = tl.load(r_ptrs,
                         mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                         other=0.0)

        # per-channel matmul
        for d in range(0, BLOCK_D):
            # slice corresponding channel
            l_mat = l_tile[:, :, d]        # (M, K)
            r_mat = r_tile[:, :, d]        # (N, K)
            # compute MxN product: l_mat @ r_mat.T
            out_mat = tl.dot(l_mat, r_mat, trans_b=True)  # (M, N)
            acc[:, :, d] += out_mat

    # store accumulator
    o_ptrs = o_ptr + b * stride_o_b \
             + offs_i[:, None, None] * stride_o_i \
             + offs_j[None, :, None] * stride_o_j \
             + tl.arange(0, BLOCK_D)[None, None, :] * stride_o_h

    mask_store_i = offs_i < N
    mask_store_j = offs_j < N
    mask_store_d = tl.arange(0, BLOCK_D) < H

    tl.store(o_ptrs,
             acc,
             mask=mask_store_i[:, None, None] & mask_store_j[None, :, None] & mask_store_d[None, None, :])
```

Better approach: Recognize that the operation is exactly batched matrix multiplication per channel. So we can compute out for each channel using `torch.bmm` across N dimension, but we need to handle hidden_dim channels. We can reshape left and right to combine batch dimension and hidden dimension and use torch.bmm.

Specifically, left shape [B,N,N,H] -> view as [B*H,N,N], right similarly. Then compute for each d: out_d = left_d @ right_d.transpose(-2, -1). This could be computed using a broadcasted batched matmul: `left_view = left.permute(0,3,1,2).reshape(B*H, N, N)`? Actually left shape after permute: [B, H, N, N] -> reshape as (B*H, N, N). Right shape: same. Then out_view = torch.bmm(left_view, right_view.transpose(1,2)) -> shape (B*H, N, N). Then reshape back to [B, N, N, H] by `out = out_view.view(B, H, N, N).permute(0,2,3,1)`. Indeed, `torch.bmm` can handle large batch of matrices efficiently. Since each channel is independent, we can compute all channels in a single batch.

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden idx
    bh = pid_bh
    b = bh // H
    h = bh % H

    # offsets in i and j
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask for boundary i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets
        left_ptrs = left_ptr + b * stride_l_b \
                    + h * stride_l_h \
                    + offs_i[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k

        right_ptrs = right_ptr + b * stride_r_b \
                     + h * stride_r_h \
                     + offs_j[:, None] * stride_r_i \
                     + offs_k[None, :] * stride_r_k

        # load tiles
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)

        # accumulate dot product
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # store output
    out_ptrs = out_ptr + b * stride_o_b \
               + h * stride_o_h \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – highly‑optimised Triton implementation.

The operation consists of
1. Layer‑Norm over the channel dimension.
2. Linear projections + gates (left/right/out) – performed with torch.nn.functional.
3. Optional masking of the projected tensors.
4. The core “tri‑multiplicative” update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per hidden channel.
   We implement it with a custom Triton kernel where each program processes
   a (BLOCK_M × BLOCK_N) tile for a single batch element & hidden channel.
5. Post‑norm, output gate, and final linear projection.

The kernel works for arbitrary batch size (≤2), sequence length (≤1024) and
hidden dimension (any multiple of the block size).  It uses fp32 accumulation
and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled matmul for the “tri‑multiplicative” term.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,           # const float*  [B, N, N, H]
    right_ptr,          # const float*  [B, N, N, H]
    out_ptr,            # float*        [B, N, N, H]

    B, N, H,            # runtime sizes

    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides

    BLOCK_M: tl.constexpr,   # tile rows (i dimension)
    BLOCK_N: tl.constexpr,   # tile cols (j dimension)
    BLOCK_K: tl.constexpr,   # reduction tile (k dimension)
):
    """
    Compute, for a given batch `b` and hidden channel `h`:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Grid layout:
        dim0 (pid_i) : tiles along the i‑axis
        dim1 (pid_j) : tiles along the j‑axis
        dim2 (pid_bh): combined batch‐and‑channel dimension (B*H)
    """
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_bh = tl.program_id(2)         # batch*hidden index

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑channel index from pid_bh
    # ------------------------------------------------------------------
    b = pid_bh // H                   # integer division
    h = pid_bh - b * H                # remainder = channel index

    # ------------------------------------------------------------------
    # Offsets for the output tile (i and j)
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds i/j (when N is not a multiple of BLOCK_*)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile (float32 for best precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Pointer arithmetic for left and right tiles.
        # left[b, i, k, h]   -> stride: (b,i,k,h)
        # right[b, j, k, h]  -> stride: (b,j,k,h)
        # ------------------------------------------------------------------
        left_tile_ptr = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k   # shape (BLOCK_M, BLOCK_K)

        right_tile_ptr = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h \
            + offs_j[:, None] * stride_r_i \
            + offs_k[None, :] * stride_r_k   # shape (BLOCK_N, BLOCK_K)

        # Load tiles with proper masking (out‑of‑bounds entries are zero)
        left_tile = tl.load(
            left_tile_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        right_tile = tl.load(
            right_tile_ptr,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )

        # ------------------------------------------------------------------
        # Accumulate dot product: (M×K) @ (K×N)  (right is transposed)
        # ------------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # ------------------------------------------------------------------
    # Write the computed tile to the output tensor
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr \
        + b * stride_o_b \
        + h * stride_o_h \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j

    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor with shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract model weights (all tensors are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑Norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional masking (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        # broadcast to the channel dimension
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – compute Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Implemented with a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # N == seq_len
    out = torch.empty_like(left)                # same shape as left/right

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides (in elements, not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # Block sizes – chosen to fit comfortably in registers/shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
