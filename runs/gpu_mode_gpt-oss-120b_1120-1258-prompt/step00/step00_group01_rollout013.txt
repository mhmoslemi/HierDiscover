@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_c, stride_i, stride_k,  # from left/right and out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # pid_c, pid_m, pid_n
    pid_c = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets.
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers
        a_ptrs = left_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k

        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b, trans_b=True)  # a (BLOCK_M x BLOCK_K), b (BLOCK_N x BLOCK_K) -> after transpose = (BLOCK_M x BLOCK_N)

    # Write output
    c_ptrs = out_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_c, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_c = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinate offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptr = left_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptr = right_ptr + pid_c * stride_c + offs_j[:, None] * stride_i + offs_k[None, :] * stride_k

        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16).to(tl.float32)
        b = tl.load(b_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        acc += tl.dot(a, b, trans_b=True)

    c_ptr = out_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Computes:
        out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    where left/right are linear projections of the input with gating and optional masking.
    The heavy sum is performed by a fused Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Move weights to correct dtype and device
    # Convert to half for projection and mask
    dtype_half = torch.float16
    dtype_float = torch.float32

    # LayerNorm
    norm_weight = weights["norm.weight"].to(dtype_float).to(device)
    norm_bias = weights["norm.bias"].to(dtype_float).to(device)

    # compute mean & var across dim for each element
    # using torch.nn.functional.layer_norm for simplicity
    x = input_tensor
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # cast to half for projection
    x_h = x.to(dtype_half)

    # Linear weights
    left_proj_w = weights["left_proj.weight"].to(dtype_half).to(device)
    right_proj_w = weights["right_proj.weight"].to(dtype_half).to(device)
    left_gate_w = weights["left_gate.weight"].to(dtype_half).to(device)
    right_gate_w = weights["right_gate.weight"].to(dtype_half).to(device)
    out_gate_w = weights["out_gate.weight"].to(dtype_half).to(device)

    # Projections
    left = F.linear(x_h, left_proj_w)     # (bs, N, N, hidden_dim)
    right = F.linear(x_h, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply mask if present
    if mask is not None:
        mask = mask.to(dtype_half).unsqueeze(-1)  # (bs, N, N, 1)
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for Triton kernel
    bs, N, _, _ = left.shape
    # Permute to [bs, hidden_dim, N, N] and contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden channels
    combined = bs * hidden_dim
    left_flat = left_t.view(combined, N, N)
    right_flat = right_t.view(combined, N, N)

    # Allocate output tensor (float32)
    out_flat = torch.empty((combined, N, N), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_c = left_flat.stride(0)
    stride_i = left_flat.stride(1)
    stride_k = left_flat.stride(2)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (combined,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_einsum_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_c, stride_i, stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K
    )

    # Reshape back to original layout [bs, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_float).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_float).to(device)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # Apply out_gate (now float32)
    out_gate_f = out_gate.to(dtype_float)
    out = out * out_gate_f

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(dtype_float).to(device)
    out = F.linear(out, to_out_w)

    return out

"""
TriMul “outgoing” forward pass with a Triton‐accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections (no bias) → `left` and `right` (shape [..., H])
   and three gating projections (`left_gate`, `right_gate`, `out_gate`).
3. Optional binary mask is applied to `left`/`right` before gating.
4. Fuse the heavy O(N³·H) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication
   `leftₕ @ rightₕᵀ` for every hidden channel `h`.
   A custom Triton kernel tiles the i‑ and j‑axes and accumulates over k,
   loading `left`/`right` in fp16 and accumulating in fp32.
5. Reshape the kernel output back to [B, N, N, H].
6. Apply a second Layer‑norm over the hidden dimension, multiply by
   `out_gate`, and finish with a linear projection back to the original
   feature dimension `C`.

The kernel is launched with a 3‑D grid:
    (combined_batch_hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N)),
where `combined_batch_hidden = B * H`.  Each program instance computes a
`BLOCK_M × BLOCK_N` tile of the output matrix for a single hidden channel.
Boundary handling is performed with masks, and all arithmetic is performed
in fp32 for numerical stability while keeping the memory traffic in fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr,        # const fp16 *
    right_ptr,       # const fp16 *
    out_ptr,         #       fp32 *
    N,               # int32  (sequence length)
    stride_c,        # int32  (stride between different hidden‑channel matrices)
    stride_i,        # int32  (stride between rows)
    stride_k,        # int32  (stride between columns)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one tile of the outgoing TriMul contraction.
    For a fixed hidden channel (identified by program_id(0)):
        out[i,j] = Σₖ left[i,k] * right[j,k]
    """
    pid_c = tl.program_id(0)      # which (batch * hidden) matrix
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile column

    # Row / column offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers into left and right tensors
        a_ptr = left_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptr = right_ptr + pid_c * stride_c + offs_j[:, None] * stride_i + offs_k[None, :] * stride_k

        # Load tiles (fp16 → fp32)
        a = tl.load(a_ptr,
                     mask=mask_i[:, None] & mask_k[None, :],
                     other=0.0,
                     dtype=tl.float16).to(tl.float32)
        b = tl.load(b_ptr,
                     mask=mask_j[:, None] & mask_k[None, :],
                     other=0.0,
                     dtype=tl.float16).to(tl.float32)

        # Accumulate the dot product over the K‑axis
        acc += tl.dot(a, b, trans_b=True)

    # Store the result (still fp32)
    c_ptr = out_ptr + pid_c * stride_c + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑backed forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, C] (float32)
              - mask: torch.Tensor [B, N, N] (bool / float) or None
              - weights: dict of model parameters (float32)
              - config: dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Optional flag – if True we skip any masking altogether
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float32)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32

    # ------------------------------------------------------------------
    # 2) Linear projections (half‑precision) and gating
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)   # fp16 for the heavy linear ops

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Projections (no bias)
    left  = F.linear(x_h, left_proj_w)          # [B, N, N, H] fp16
    right = F.linear(x_h, right_proj_w)         # [B, N, N, H] fp16

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # keep for later

    # ------------------------------------------------------------------
    # 3) Optional mask (binary 0/1) – applied before gating
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # Convert mask to fp16 and broadcast over the hidden dimension
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # Apply element‑wise gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Triton kernel for the O(N³·H) contraction
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Bring hidden channel to the second dimension: [B, H, N, N] (contiguous)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch‑ and hidden‑ dimensions → (B*H, N, N)
    combined = B * hidden_dim
    left_flat  = left_t.view(combined, N, N)   # fp16
    right_flat = right_t.view(combined, N, N)  # fp16

    # Output buffer (float32 for accumulation stability)
    out_flat = torch.empty((combined, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_c = left_flat.stride(0)   # = N * N
    stride_i = left_flat.stride(1)   # = N
    stride_k = left_flat.stride(2)   # = 1

    # Tiling parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        combined,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_einsum_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_c,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, H] float32

    # Apply the output gate (already **float32** after the earlier cast)
    out = out * out_gate.to(torch.float32)

    # Final projection back to the original channel size
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    out = F.linear(out, to_out_w)   # [B, N, N, C] float32

    return out
