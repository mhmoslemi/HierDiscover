   input_tensor, mask, weights, config = data

   batch_size, seq_len, _, dim = input_tensor.shape
   hidden_dim = config["hidden_dim"]

   # Get weight tensors
   # Norm weight and bias
   ...

   # Compute layernorm:
   x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

"""
TriMul (outgoing) implementation with Triton.

The module follows the reference PyTorch version:
1. Layer‑norm over the last dimension (dim).
2. Linear projections (left_proj, right_proj) producing a hidden dimension H.
3. Optional mask is applied to the projected tensors.
4. Gating signals (left_gate, right_gate, out_gate) are computed with a sigmoid.
5. The core “tri‑multiplicative” update:
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is performed by a custom Triton kernel that computes a batch of
   independent matrix‑multiplications (one per hidden slice) in a highly
   tiled fashion (BLOCK_M × BLOCK_N × BLOCK_K).
6. A second layer‑norm over the hidden dimension, multiplication by the
   output gate and a final linear projection back to `dim`.

Only the O(N³·H) part (step 5) is written in Triton; the surrounding
operations use PyTorch for simplicity. The kernel works for any batch size
(≤ 2), sequence length (up to 1024 in the test suite) and hidden dimension
(≤ 128). It returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,                # *float32
    right_ptr,               # *float32
    out_ptr,                 # *float32
    B, N, H,                 # i32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for every batch `b` and hidden slice `h`,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled matrix‑multiply (M×K)·(K×N).
    The third launch dimension indexes (b*H + h).
    """
    # program ids
    pid_m = tl.program_id(0)          # output row block (i)
    pid_n = tl.program_id(1)          # output col block (j)
    pid_batch = tl.program_id(2)      # flattened (b, h)

    # decode batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # offsets within the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: we need right_T[k, j] = right[j, k]
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate partial product
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # store the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul forward pass (outgoing) using Triton for the O(N³·H) term.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N]   (or None)
            - weights: dict of pre‑loaded weight tensors
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm over the last dimension
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects normalized_shape = (dim,)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)          # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating signals
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core Tensor‑Multiplicative update via Triton
    # ------------------------------------------------------------------
    # Ensure contiguous layout (C‑order) for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # Output buffer
    out = torch.empty((batch_size, seq_len, seq_len, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h   = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h       = out.stride()

    # Tiling parameters (tuned for H100, fp32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # smaller K tile reduces register pressure

    grid = (
        triton.cdiv(seq_len, BLOCK_M),   # rows
        triton.cdiv(seq_len, BLOCK_N),   # cols
        batch_size * hidden_dim         # (b, hidden) dimension
    )

    _trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=batch_size,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # typical for fp32 matmul on H100
    )

    # ------------------------------------------------------------------
    # 6) Second layer‑norm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
