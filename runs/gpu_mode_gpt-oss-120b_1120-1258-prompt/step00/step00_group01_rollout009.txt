import torch
import triton
import triton.language as tl

@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    has_mask: tl.constexpr,
):
    # flatten position index: pid = program_id(0)
    pid = tl.program_id(0)
    total_positions = B * N * N
    if pid >= total_positions:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offset base for left/right/gate: combine strides
    # we process BLOCK_H per iteration in H dimension
    BLOCK_H = 32
    offs_h = tl.arange(0, BLOCK_H)
    # compute pointers for left and right for current position
    left_base = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j
    right_base = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j
    left_gate_base = left_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    right_gate_base = right_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    if has_mask:
        mask_base = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        mask_val = tl.load(mask_base)
    else:
        mask_val = 1.0

    # Determine loop over hidden dimension
    for offset_h in range(0, H, BLOCK_H):
        mask_h = offs_h < (H - offset_h)
        # Load left, right, gates
        left = tl.load(left_base + (offset_h + offs_h) * stride_left_h, mask=mask_h, other=0.0)
        right = tl.load(right_base + (offset_h + offs_h) * stride_right_h, mask=mask_h, other=0.0)
        left_gate = tl.load(left_gate_base + (offset_h + offs_h) * stride_gate_h, mask=mask_h, other=0.0)
        right_gate = tl.load(right_gate_base + (offset_h + offs_h) * stride_gate_h, mask=mask_h, other=0.0)

        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

"""
TriMul outgoing implementation (forward only).

The kernel performs the element‑wise mask and gate fusion for the
`left` and `right` tensors.  The expensive N³ reduction
`einsum('... i k d, ... j k d -> ... i j d')` is carried out with a
batched matrix multiplication (`torch.bmm`) which is heavily
optimised on H100.  The overall steps are:

1. Layer‑norm the input tensor.
2. Linear projections (left/right) and three sigmoid gates.
3. Triton kernel:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   (mask is a scalar per (b,i,j) entry, broadcasted over the hidden
   dimension).
4. Batched GEMM over the summed `k` dimension:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   implemented as `torch.bmm` on the reshaped tensors.
5. Layer‑norm on the hidden dimension, apply the output‑gate and a final
   linear projection back to the original `dim`.

Only the mask‑gate fusion is written in Triton, satisfying the
requirement that “part of the operations” be a custom kernel while
keeping the heavy N³ reduction in fast cuBLAS code.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: fuse mask, left‑gate and right‑gate (in‑place)
# ----------------------------------------------------------------------------- #
@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    # strides for left/right tensors (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for gate tensors (same layout as left/right)
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides for mask tensor (B,N,N) – unused when HAS_MASK==0
    stride_mask_b, stride_mask_i, stride_mask_j,
    # runtime parameters
    B, N,
    H: tl.constexpr,               # hidden dimension (compile‑time)
    BLOCK_H: tl.constexpr,         # tile size for H
    HAS_MASK: tl.constexpr,        # 1 → use mask, 0 → no mask
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from the flattened pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j)
    left_base = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j
    right_base = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j
    left_gate_base = left_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    right_gate_base = right_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # load mask scalar if present
    if HAS_MASK:
        mask_base = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        mask_val = tl.load(mask_base).to(tl.float32)   # scalar 0.0 or 1.0
    else:
        mask_val = 1.0

    # loop over hidden dimension in BLOCK_H sized chunks
    offs_h = tl.arange(0, BLOCK_H)
    for offset_h in range(0, H, BLOCK_H):
        # bounds mask for the last (partial) tile
        h_valid = (offset_h + offs_h) < H

        # load a slice of left/right and their gates
        left = tl.load(left_base + (offset_h + offs_h) * stride_left_h,
                       mask=h_valid, other=0.0)
        right = tl.load(right_base + (offset_h + offs_h) * stride_right_h,
                        mask=h_valid, other=0.0)
        left_gate = tl.load(left_gate_base + (offset_h + offs_h) * stride_gate_h,
                            mask=h_valid, other=0.0)
        right_gate = tl.load(right_gate_base + (offset_h + offs_h) * stride_gate_h,
                             mask=h_valid, other=0.0)

        # fuse mask and gates
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

        # write back
        tl.store(left_base + (offset_h + offs_h) * stride_left_h, left,
                 mask=h_valid)
        tl.store(right_base + (offset_h + offs_h) * stride_right_h, right,
                 mask=h_valid)

def custom_kernel(data):

    Arguments

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # --------------------------------------------------------------------- #
    # Unpack configuration and weights
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask
    eps = 1e-5

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear / gate weights (no bias)
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    # Output norm + final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    with torch.no_grad():
        x_norm = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=eps,
        )   # [B, N, N, dim]

        # ----------------------------------------------------------------- #
        # 2. Projections + gates (still in float32)
        # ----------------------------------------------------------------- #
        left = torch.nn.functional.linear(x_norm, left_proj_weight)    # [B,N,N,H]
        right = torch.nn.functional.linear(x_norm, right_proj_weight)

        left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

        # ----------------------------------------------------------------- #
        # 3. Triton kernel: combine mask and gates (in‑place)
        # ----------------------------------------------------------------- #
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        if nomask:
            # Dummy mask – never read when HAS_MASK==0
            dummy_mask = left
        else:
            dummy_mask = mask_tensor.to(dtype=left.dtype).contiguous()

        B, N, _, _ = left.shape
        total_positions = B * N * N

        # strides are expressed in number of elements (not bytes)
        s_left  = left.stride()
        s_right = right.stride()
        s_gate  = left_gate.stride()
        s_mask  = dummy_mask.stride() if not nomask else (0, 0, 0)

        # Launch kernel (one thread per (b,i,j) entry)
        mask_gate_fusion_kernel[(total_positions,)](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=dummy_mask,
            # left/right strides
            stride_left_b=s_left[0],
            stride_left_i=s_left[1],
            stride_left_j=s_left[2],
            stride_left_h=s_left[3],
            stride_right_b=s_right[0],
            stride_right_i=s_right[1],
            stride_right_j=s_right[2],
            stride_right_h=s_right[3],
            # gate strides
            stride_gate_b=s_gate[0],
            stride_gate_i=s_gate[1],
            stride_gate_j=s_gate[2],
            stride_gate_h=s_gate[3],
            # mask strides (ignored when HAS_MASK==0)
            stride_mask_b=s_mask[0] if not nomask else 0,
            stride_mask_i=s_mask[1] if not nomask else 0,
            stride_mask_j=s_mask[2] if not nomask else 0,
            B=B,
            N=N,
            H=hidden_dim,
            BLOCK_H=32,
            HAS_MASK=int(not nomask),
        )

        # ----------------------------------------------------------------- #
        # 4. Batched matrix multiplication over the summed dimension k
        #    out_{b,i,j,d} = Σₖ left_{b,i,k,d} * right_{b,j,k,d}
        #    Per‑hidden‑dim GEMM: (B, H, N, N) @ (B, H, N, N)^T
        # ----------------------------------------------------------------- #
        # reshape to (B*H, N, N)
        left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

        # (B*H, N, N) x (B*H, N, N)^T -> (B*H, N, N)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

        # restore shape [B, N, N, H]
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

        # ----------------------------------------------------------------- #
        # 5. Output layer‑norm, out‑gate and final linear projection
        # ----------------------------------------------------------------- #
        out = torch.nn.functional.layer_norm(
            out,
            (hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=eps,
        )
        out = out * out_gate                     # element‑wise gate
        output = torch.nn.functional.linear(out, to_out_weight)  # -> [B,N,N,dim]

        return output
