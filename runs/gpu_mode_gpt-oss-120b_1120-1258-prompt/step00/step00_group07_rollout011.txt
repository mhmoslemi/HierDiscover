def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6 probably for LayerNorm.

    # Cast input to float32 if necessary:
    x = input_tensor

    # LayerNorm normalizing over last dim dim
    # Use torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-6)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

import torch
import triton
import triton.language as tl

@triton.jit
def triu_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bi, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decode batch and hidden indices
    b_idx = batch_h // H
    h_idx = batch_h % H

    # Base pointers for this (b, h)
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # Offsets inside the M/N tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load sub‑matrix from A (shape BLOCK_M×BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load sub‑matrix from B (shape BLOCK_K×BLOCK_N) – note the transposition
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Tensor‑core dot (accumulate in FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # Store the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    The heavy pairwise reduction ``einsum('... i k d, ... j k d -> ... i j d')``
    is implemented as a batched matrix multiplication ``C = A @ Bᵀ``
    where each hidden‑channel is treated as an independent batch and computed
    with a Tensor‑Core friendly Triton kernel.
    """
    import torch
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (the reference implementation multiplies both
    #    sides by ``mask[..., None]``)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is assumed to be broadcastable to [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating of the projections
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim

    # Allocate output buffer for the reduction result (still FP32)
    out = torch.empty(B, N, N, H, dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch: C = A @ Bᵀ for each (batch, hidden) pair
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bi, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    triu_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ab, stride_ai, stride_ak, stride_ah,
        stride_bb, stride_bi, stride_bk, stride_bh,
        stride_cb, stride_ci, stride_cj, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    # LayerNorm over the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear back to the original feature dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor on the last feature dimension.
2. Project the normalized tensor to a hidden dimension using three weight‑only
   Linear layers: left_proj, right_proj and three gating Linear layers.
   Apply the optional mask and the sigmoid gates.
3. The expensive reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication C = A @ Bᵀ where each hidden channel
   (and each batch) is treated as an independent GEMM:
       A : (B·H, N, N)   = left  (reshape)
       B : (B·H, N, N)   = right (reshape)
   This is performed by a custom Triton kernel that tiles the matrices
   (BLOCK_M×BLOCK_N blocks, reduction over BLOCK_K) and uses Tensor‑Core
   `tl.dot` with FP16 accumulation into FP32.
4. Post‑process the result with a second LayerNorm, the output gate and a
   final Linear projection back to the original feature dimension.

The kernel operates on FP16 inputs (left/right) and writes FP32 outputs,
which matches the reference PyTorch implementation while vastly reducing
the O(N³) work time on H100 GPUs.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (A,B : (B·H, N, N) , C : (B·H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    B, N, H,                           # batch size, sequence length, hidden_dim
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides of A
    stride_bb, stride_bi, stride_bk, stride_bh,   # strides of B
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides of C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # 1) Identify batch & hidden index from the first program ID
    # --------------------------------------------------------------
    batch_h = tl.program_id(0)               # 0 … B*H-1
    pid_m   = tl.program_id(1)               # tile row index
    pid_n   = tl.program_id(2)               # tile col index

    b_idx = batch_h // H                      # original batch
    h_idx = batch_h %  H                      # hidden channel

    # Base pointers for the selected (batch, hidden) slice
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # --------------------------------------------------------------
    # 2) Offsets for the current M×N tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ----------------------------------------------------------------
    # 3) Accumulate over K using Tensor‑Core friendly dot products
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # dtype follows pointer (fp16)

        # B tile (transposed): shape (BLOCK_K, BLOCK_N)
        # Original B is indexed as (b, j, k, h); we need Bᵀ → (k, j)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # fp16

        # Tensor‑core dot – accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ----------------------------------------------------------------
    # 4) Write the tile back to C
    # ----------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim]
            mask         : torch.Tensor, shape [B, N, N] (or None)
            weights      : dict of torch.Tensor containing all model parameters
            config       : dict with "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    # (mask handling flag – if True we skip mask multiplication)
    apply_mask = not config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if apply_mask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] → float32
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating of the projections
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast the gated tensors to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape            # H == hidden_dim
    assert H == hidden_dim, "hidden_dim mismatch"

    # ------------------------------------------------------------------
    # 6) Allocate the output of the reduction (still FP32)
    # ------------------------------------------------------------------
    out = torch.empty(B, N, N, H, dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # 7) Triton GEMM kernel: out = left @ rightᵀ  (batched over B·H)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides (int64) – torch returns Python ints already.
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bi, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Grid: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Post‑processing: second LayerNorm, output gate, final Linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # still [B, N, N, hidden_dim], FP32

    out = out * out_gate                       # apply output gate (sigmoid)

    # Final projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
