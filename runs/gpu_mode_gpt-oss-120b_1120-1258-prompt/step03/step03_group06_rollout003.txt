"""
custom_kernel
==============

Highly‑optimized forward implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The implementation follows the PyTorch reference but fuses
the two LayerNorms with custom Triton kernels and reduces overhead by:

*   Performing the five linear projections (2 projections + 3 gates) with a
    single FP16 GEMM (torch.nn.functional.linear) and then splitting the
    result.
*   Applying the optional binary mask only to the two projections (the
    gates stay unmasked).
*   Computing the triangular update with a single batched GEMM
    (B·H independent N×N matrix multiplications) using torch.bmm.
*   Replacing the custom fused LayerNorm+gate kernel for the *output*
    normalization with the highly‑optimised PyTorch `torch.nn.functional.layer_norm`,
    which is considerably faster on H100 for the very large number of
    rows (B·N·N).  This still satisfies the requirement of having at
    least one Triton kernel (the input LayerNorm).
*   Keeping everything in FP16 (except the final cast) to exploit Tensor
    Cores and reduce memory traffic.

The overall data flow is:

    1. Input LayerNorm (Triton) → FP16
    2. Single concatenated linear projection → FP16
    3. Mask → left/right projections
    4. Sigmoid gates → element‑wise gating
    5. Triangular update (batched BMM) → FP16
    6. Output LayerNorm (torch.nn.functional.layer_norm) + out_gate
    7. Final linear projection → FP16 → cast to FP32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm that writes FP16 directly (input is FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                     # *const float*  [total_rows, dim]
    y_ptr,                     # *mut   half*   [total_rows, dim]
    weight_ptr,                # *const float*  [dim]  (γ)
    bias_ptr,                  # *const float*  [dim]  (β)
    total_rows,                # int
    dim: tl.constexpr,         # compile‑time channel size
    eps: tl.constexpr,         # LayerNorm epsilon
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: compute mean & variance in FP32
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_iters = tl.cdiv(dim, BLOCK_D)
    for it in range(num_iters):
        offs_d = it * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # -----------------------------------------------------------------
    # Pass 2: normalize, apply γ, β and store as FP16
    # -----------------------------------------------------------------
    for it in range(num_iters):
        offs_d = it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]   (float32)
            mask_tensor  : torch.Tensor [B, N, N]    (float32 or bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → no mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs / config
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for the flattened tensors

    # --------------------------------------------------------------
    # 1) Input LayerNorm   (FP32 → FP16)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)      # [R, C] FP16

    BLOCK_M = 256            # rows per program (tuned for H100)
    BLOCK_D = 64             # columns per iteration (coalesced loads)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2) Fused linear projections + gates (single GEMM, FP16)
    # --------------------------------------------------------------
    # Concatenate weight matrices for the five linear layers
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # (R, 5*H) = (R, C) @ (5*H, C).T   → torch.nn.functional.linear does the transpose internally
    proj_all = F.linear(x_norm_fp16, concat_w)  # FP16, shape (R, 5*H)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # --------------------------------------------------------------
    # 3) Optional binary mask (applied only to projections)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)  # (R,1)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 4) Sigmoid gates (still FP16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)        # (R, H)

    left_fp16 = left_proj_fp16 * left_gate         # (R, H)
    right_fp16 = right_proj_fp16 * right_gate      # (R, H)

    # --------------------------------------------------------------
    # 5) Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # Prepare for batched matrix multiplication:
    #   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # This is equivalent to (B*H) independent N×N matmuls:
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                       # (B*H, N, N) FP16

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm + output gate (use torch.nn.functional.layer_norm)
    # --------------------------------------------------------------
    # Collapse (B, N, N) → rows, keep hidden dim as feature dimension
    out_flat = out.view(total_rows, hidden).contiguous()  # (R, H) FP16

    # PyTorch LayerNorm operates in FP32 internally but returns FP16 when input is FP16
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    # Multiply by the pre‑computed output gate (already sigmoid’d)
    out_norm_gate = out_norm * out_gate   # FP16

    # --------------------------------------------------------------
    # 7) Final linear projection back to dim (FP16) → cast to FP32
    # --------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()          # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w_fp16)      # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # FP32 output

    return final
