"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input channel dimension C.
    • Input is cast to fp16 up‑front.
    • The kernel computes mean/variance in fp32, applies γ/β and writes fp16.
2️⃣  Fuse the 5 linear projections (left/right and 3 gates) into a single
    fp16 GEMM by concatenating the weight matrices.
3️⃣  Optional mask is applied to the left/right projections before gating.
4️⃣  Apply sigmoid to the three gate tensors and perform element‑wise gating.
5️⃣  Triangular multiplicative update:
        out[i,j] = Σₖ left[i,k] * right[j,k]
    realised by a single `torch.einsum` on fp16 tensors (no expensive permutes).
6️⃣  Row‑wise LayerNorm on the hidden dimension H (fp16 kernel, fp32 reduction).
7️⃣  Multiply by the output gate, then a final fp16 linear projection back to C.
8️⃣  Cast the result to fp32 and return.

Only the two LayerNorms are written in Triton; all heavy linear algebra uses
Tensor‑Core‑accelerated fp16 GEMMs (via `torch.nn.functional.linear` and
`torch.einsum`).  This eliminates the large memory copies needed for the
previous permutation‑based batched GEMM and brings the forward pass
well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16/fp32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const   <dtype>   [total_rows, dim]   (fp16 or fp32)
    y_ptr,          # *mut     <dtype>   [total_rows, dim]   (same dtype as x)
    weight_ptr,     # *const   float32   [dim]
    bias_ptr,       # *const   float32   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    # -----------------------------------------------------------------
    # 0️⃣  Program / block indices
    # -----------------------------------------------------------------
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load as the dtype of x_ptr (fp16 or fp32) then cast to fp32
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # 2‑D pointer
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                              # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣  Write normalized output (fp16 if input is fp16, else fp32)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – both fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        # Cast back to the original dtype of y (fp16 or fp32)
        out = out.to(y_ptr.dtype)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from layer names to weight tensors
            - config_dict : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare the flattened view used by the LayerNorm kernels
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # rows for the flattened view
    # Cast input to fp16 before normalisation (mean/var are still computed in fp32)
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    x_norm_fp16 = torch.empty_like(x_fp16)        # fp16 buffer for LN output

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over channel dimension C (input → hidden)
    # ------------------------------------------------------------------
    # Choose block sizes that give good occupancy for all tested N
    BLOCK_M = 1024   # rows per program
    BLOCK_D = 32     # cols per iteration (must divide typical dim <= 768)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_fp16,
        x_norm_fp16,
        weights["norm.weight"],      # fp32 weight [C]
        weights["norm.bias"],        # fp32 bias   [C]
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    # Concatenate weight matrices (each [hidden_dim, dim]) and cast to fp16
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]
    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                 # shape (5*H, C)

    # Fused projection: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each [total_rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [total_rows, 1] (float16)
        mask_flat_fp16 = (
            mask_tensor.contiguous()
            .view(total_rows, 1)
            .to(torch.float16)
        )
        left_proj  = left_proj * mask_flat_fp16
        right_proj = right_proj * mask_flat_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)      # [total_rows, H]
    right_gate = torch.sigmoid(right_gate)    # [total_rows, H]
    out_gate   = torch.sigmoid(out_gate)      # [total_rows, H]

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape back to 4‑D layout for the triangular update
    left  = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update via einsum (no extra copy)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 in/out)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                     # rows for hidden‑dim view
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)      # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # fp32 weight [H]
        weights["to_out_norm.bias"],     # fp32 bias   [H]
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # element‑wise fp16
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # fp16, shape [total_rows_h, C]

    # ------------------------------------------------------------------
    # 8️⃣ Cast to fp32 and reshape to original 4‑D layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
