```python
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # 1. First LayerNorm using torch.nn.functional
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for heavy ops
    x_norm_h = x_norm.to(torch.float16)

    # 2. Fused projection
    # Concatenate weight matrices (5*H, D)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    proj_all = torch.nn.functional.linear(x_norm_h, w_concat)  # [R, 5*H]

    H = hidden_dim
    # slice
    left_proj = proj_all[:, 0:H]
    left_gate = proj_all[:, H:2 * H]
    right_proj = proj_all[:, 2 * H:3 * H]
    right_gate = proj_all[:, 3 * H:4 * H]
    out_gate = proj_all[:, 4 * H:5 * H]

    # Apply mask if needed (broadcast)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Sigmoid gates (use half)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Compute gated left/right
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 3. Tri‑multiplicative update (batched GEMM)
    # reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H to batch dimension for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) where k is second dim

    BH = B * H
    left_mat = left_mat.view(BH, N, N)
    right_mat = right_mat.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (BH,N,N) half
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 4. Second LayerNorm + out_gate gating (Triton kernel)
    out_flat = out.view(total_rows, H)               # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)     # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)       # half

    BLOCK_M = 1024  # process many rows per kernel launch
    BLOCK_D = 32    # hidden dim <=128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),
        weights["to_out_norm.bias"].to(torch.float16),
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 5. Final projection
```

```python
"""
TriMul outgoing kernel – high‑performance forward pass.

Algorithm
---------
1. **Row‑wise LayerNorm** (float32) on the full input tensor `x`
   using `torch.nn.functional.layer_norm` (fast C++ kernel).  The
   normalized tensor is cast to FP16.

2. **Fused projection** – concatenate the five weight matrices
   (`left_proj`, `left_gate`, `right_proj`, `right_gate`, `out_gate`)
   and perform a single FP16 GEMM `x_norm @ Wᵀ`.  The result is sliced
   into the three projection streams and three gate streams.

3. **Mask & sigmoid** – optionally mask the two projection streams,
   then apply a half‑precision sigmoid to all three gate tensors and
   multiply the mask‑projected values with the corresponding gates,
   yielding `left` and `right` tensors of shape `[B,N,N,H]`.

4. **Tri‑multiplicative update** – reshape `left/right` to
   `[B,H,N,N]` and compute a batched matrix multiplication
   `out = left @ rightᵀ` with a single call to `torch.bmm`.  The result
   has shape `[B,N,N,H]` (FP16).

5. **Hidden‑dim LayerNorm + output‑gate** – a Triton kernel
   (`layernorm_gate_kernel`) normalises `out` per hidden dimension,
   applies the learned γ/β and multiplies element‑wise by the
   `out_gate` tensor.

6. **Final linear projection** – another FP16 GEMM maps the hidden
   dimension back to the original channel dimension `dim`.  The result
   is cast back to FP32 and returned.

Only the second LayerNorm is implemented in Triton; all large matrix
multiplications use cuBLAS/Tensor‑Core kernels.  Mixed precision is
used throughout to minimise memory traffic while keeping the final
output in float32.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input)
    y_ptr,          #       half*   [total_rows, dim]   (LN output * gate)
    gate_ptr,       # const half*   [total_rows, dim]   (out_gate)
    weight_ptr,     # const half*   [dim]                (γ)
    bias_ptr,       # const half*   [dim]                (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Per‑row LayerNorm (FP32 accumulator) followed by element‑wise
    multiplication with a gate tensor.  All arithmetic is performed in
    FP32 and the result is written back as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply γ/β, multiply gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load input
        in_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(in_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize
        norm = (x - mean[:, None]) * rstd[:, None]

        # γ and β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # store
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: torch.Tensor [B, N, N, D] (float32)
        - mask_tensor : torch.Tensor [B, N, N]   (bool/int, optional)
        - weights_dict: mapping from layer names to torch.Tensor
        - config_dict : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ---------------------------------------------------------
    # 1. First LayerNorm (fast C++ kernel) + cast to fp16
    # ---------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                     # [B,N,N,D] float32
    x_norm_h = x_norm.to(torch.float16)   # half‑precision for the heavy work

    # ---------------------------------------------------------
    # 2. Fused projection: compute left/right projections and gates
    # ---------------------------------------------------------
    # concatenate weights for a single GEMM (shape (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*H, D)

    proj_all = F.linear(x_norm_h, w_concat)   # (R, 5*H) half

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2*H]
    right_proj  = proj_all[:, 2*H:3*H]
    right_gate  = proj_all[:, 3*H:4*H]
    out_gate    = proj_all[:, 4*H:5*H]

    # Optional mask (broadcast over the hidden dimension)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)  # (R,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Half‑precision sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates to the projections
    left = left_proj * left_gate          # (R, H) half
    right = right_proj * right_gate       # (R, H) half

    # ---------------------------------------------------------
    # 3. Tri‑multiplicative update (batched GEMM)
    # ---------------------------------------------------------
    # reshape to [B,N,N,H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H to the batch dimension for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) with k ↔ j

    BH = B * H
    left_mat = left_mat.view(BH, N, N)    # (BH,N,N)
    right_mat = right_mat.view(BH, N, N)  # (BH,N,N)

    out_mat = torch.bmm(left_mat, right_mat)   # (BH,N,N) half

    # reshape back to original layout (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ---------------------------------------------------------
    # 4. Hidden‑dim LayerNorm + output‑gate (Triton)
    # ---------------------------------------------------------
    out_flat = out.view(total_rows, H)               # (R, H) half
    out_gate_flat = out_gate.view(total_rows, H)     # (R, H) half

    out_norm_gate = torch.empty_like(out_flat)       # half

    BLOCK_M = 1024   # number of rows per program
    BLOCK_D = 32     # tile size for the hidden dimension (H ≤ 128)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ   (half)
        weights["to_out_norm.bias"].to(torch.float16),     # β   (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ---------------------------------------------------------
    # 5. Final projection back to the original channel dimension
    # ---------------------------------------------------------
    final_half = F.linear(out_norm_gate, weights["to_out.weight"].to(torch.float16))  # (R, D) half
    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast back to FP32

    return final

    # ---------------------------------------------------------
    left  = left.view(B, N, N, H)   # [B,N,N,H]
    right = right.view(B, N, N, H)  # [B,N,N,H]

    # move hidden dimension to the batch axis for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)  (k ↔ j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # (BH,N,N)
    right_mat = right_mat.view(BH, N, N)   # (BH,N,N)

    out_mat = torch.bmm(left_mat, right_mat)   # (BH,N,N) half

    # back to original layout [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ---------------------------------------------------------
    # 4. Hidden‑dim LayerNorm + output‑gate (Triton)
    # ---------------------------------------------------------
    out_flat      = out.view(total_rows, H)            # (R, H) half
    out_gate_flat = out_gate.view(total_rows, H)       # (R, H) half

    out_norm_gate = torch.empty_like(out_flat)         # half, will hold LN*gate

    BLOCK_M = 1024   # rows processed per program (max 1024 threads)
    BLOCK_D = 32     # tile for hidden dimension (H ≤ 128)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ (half)
        weights["to_out_norm.bias"].to(torch.float16),     # β (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ---------------------------------------------------------
    # 5. Final linear projection back to the original channel dimension
    # ---------------------------------------------------------
    final_half = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16),  # (D, H)
    )                                                # (R, D) half

    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast to FP32
    return final
```
