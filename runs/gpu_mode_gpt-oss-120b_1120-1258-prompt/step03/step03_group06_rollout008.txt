    x_ptr,                     # const   half* [total_rows, dim] (pre‑norm)
    y_ptr,                     # mutable half* [total_rows, dim] (post‑norm⋅gate)
    weight_ptr,                # const   float* [dim]      (γ)
    bias_ptr,                  # const   float* [dim]      (β)
    gate_ptr,                  # const   half*  [total_rows, dim] (out_gate, already sigmoid‑ed)
    total_rows,
    dim: tl.constexpr,         # hidden dimension for this norm
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalize, apply γ, β and multiply by the gate
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]       # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)      # FP32
        out = out * gate_vals                               # apply pre‑computed out_gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32/bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → ignore mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim      = cfg["dim"]
    hidden   = cfg["hidden_dim"]
    eps      = cfg.get("eps", 1e-5)
    nomask   = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP16) – use cuDNN implementation
    # ------------------------------------------------------------------
    inp_fp16 = inp.to(torch.float16)
    x_norm = F.layer_norm(
        inp_fp16,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B,N,N,dim] half, already contiguous

    # ------------------------------------------------------------------
    # 2) Fused linear projections + three gates (single GEMM, FP16)
    # ------------------------------------------------------------------
    # concat of five weight matrices → (5*hidden, dim)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5·H, C)

    # flatten for GEMM
    x_flat = x_norm.view(-1, dim)                # (R, C)
    proj_all = F.linear(x_flat, concat_w)        # (R, 5·H) half

    left_proj_fp16, right_proj_fp16, \
        left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden, dim=1
        )  # each (R, H) half

    # ------------------------------------------------------------------
    # 2a) Optional mask (apply before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] – broadcast over hidden dimension
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still half‑precision)
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)   # (R, H) half, will be used later

    left_fp16  = left_proj_fp16 * left_gate       # (R, H)
    right_fp16 = right_proj_fp16 * right_gate     # (R, H)

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)       # (B,N,N,H)
    right = right_fp16.view(B, N, N, hidden)    # (B,N,N,H)

    # (B·H, N, N) matrices – left as is, right transposed on the k‑dimension
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)    # (B·H, N, N) half
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Second LayerNorm + output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden).contiguous()           # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()               # (C, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w_fp16)           # (R, C) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)       # cast back to FP32

    return final
