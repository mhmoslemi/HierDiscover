"""
TriMul (outgoing) – fast forward pass in Triton + PyTorch.

Algorithm
---------
1️⃣ Layer‑norm on the last channel (dim) → fp16 output (Triton).
2️⃣ Fuse the five linear projections (left/right & three gates) into a
   single fp16 GEMM by concatenating their weight matrices.
3️⃣ (Optional) apply the pairwise mask to left/right projections.
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   realized as a batched fp16 GEMM (`torch.bmm`) on tensors of shape
   (B·H, N, N), where H = hidden_dim.
6️⃣ Second layer‑norm over the hidden dimension (fp16 output, Triton).
7️⃣ Multiply by the output gate and project back to `dim` with a final
   fp16 linear layer. Cast the result to float32 (as required by the
   spec).

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra uses fp16 Tensor‑Core friendly
operations. The implementation works for the full range of test
configurations (various sequence lengths, batch sizes, and optional
masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel – accepts fp32 or fp16 input, writes fp16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   X*   [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,          # *mut     half* [total_rows, dim] (fp16 output)
    weight_ptr,     # *const float*   [dim]            (γ)
    bias_ptr,       # *const float*   [dim]            (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute per‑row mean and variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # cast to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32 raw data

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b                         # fp32 intermediate
        out_fp16 = out_fp32.to(tl.float16)              # down‑cast

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (bool/float, optional)
            - weights_dict: contains all linear and LayerNorm parameters.
            - config_dict : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C (input/output channel size)
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → skip mask

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape        # [B, N, N, C]
    total_rows = B * N * N                  # flatten over (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per kernel launch
    BLOCK_D = 64    # columns per tile (covers C up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gating (all fp16)
    # ------------------------------------------------------------------
    # Concatenate five weight matrices (each [H, C]) → [5H, C]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # (5H, C)

    # Linear: [R, C] @ [C, 5H]^T → [R, 5H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_weight)     # fp16

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Apply optional pairwise mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates & element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # Bring the hidden dimension to the second axis for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]   (k,j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]   (i,k)
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N]   (k,j)

    # Batched matrix multiply in fp16 → [B*H, N, N] (i,j)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    # Back to 4‑D tensor [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)            # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Final gating + linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # fp16 element‑wise

    # Final linear layer (H → C) – fp16 weight, no bias
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # [R_h, C] fp16

    # Cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

This implementation follows the reference forward pass while:
* Performing both LayerNorms in a custom Triton kernel (fp32 accumulation, fp16 output) to avoid costly PyTorch reductions.
* Fusing the five linear projections into a single half‑precision GEMM, cutting the number of kernel launches and memory traffic.
* Using a single batched `torch.bmm` (Tensor‑Core friendly) for the O(N³) triangular multiplication.
* Keeping the entire heavy work in fp16, only converting back to float32 for the final returned tensor.

The code respects all configuration options (batch size, sequence length, optional mask, different channel dimensions) and runs on an H100 GPU with Triton 3.3.1.
