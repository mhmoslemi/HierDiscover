```python
@triton.jit
def layernorm_scaling_kernel(
    x_ptr,   # const float* [total_rows, dim]
    a_ptr,   # float* [total_rows] (rstd)
    b_ptr,   # float* [total_rows] ( -mean * rstd )
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    a = rstd
    b = -mean * rstd

    # store
    tl.store(a_ptr + offs_m, a, mask=mask_m)
    tl.store(b_ptr + offs_m, b, mask=mask_m)
```

"""
TriMul (outgoing) – fused implementation

Algorithm
---------
1️⃣  **Row‑wise statistics** – a tiny Triton kernel computes, for every
    flattened row `x[b,i,k,:]` (size `dim`), the scale `a = 1/√(var+ε)` and
    the bias `b = -mean/√(var+ε)`.  Only two vectors `a` and `b` of length
    `total_rows = B·N·N` are written – the normalized tensor itself is *not*
    materialised.

2️⃣  **One fused linear projection** – the five weight matrices
    (`left_proj`, `left_gate`, `right_proj`, `right_gate`, `out_gate`)
    are concatenated into `W (5·H × D)`.  A single half‑precision GEMM
    (`F.linear`) is executed on the *original* input (also cast to fp16),
    producing `P = X·Wᵀ` of shape `[R,5·H]`.

3️⃣  **Row‑wise de‑normalisation** – using the per‑row factors `a,b` we
    recover the true layer‑normed projection without ever storing the
    normalized tensor:

        proj_norm = a·P + b·w_sum

    where `w_sum = W.sum(dim=1)` (pre‑computed).  The operation is done
    slice‑wise (5 × H) so that memory stays in fp16.

4️⃣  **Gating & masking** – the four gate tensors are passed through a
    sigmoid (float32 → fp16) and `left/right` are masked (if a mask is
    supplied) and gated element‑wise.

5️⃣  **Tri‑multiplicative update** – `left` and `right` are reshaped to
    `[B,N,N,H]`, permuted to `[B,H,N,N]` and batched‑matrix‑multiplied
    (`torch.bmm`) producing `out ∈ fp16`.

6️⃣  **LayerNorm + output gate** – a second Triton kernel fuses the
    hidden‑dim LayerNorm with the element‑wise multiplication by the
    output‑gate (`out_gate`).  The result stays in fp16.

7️⃣  **Final linear projection** – a half‑precision GEMM (`F.linear`)
    maps back to the original channel dimension `dim`.  The final tensor
    is cast to `float32` before being returned.

Only the two LayerNorm‑related parts are written in Triton; all large
matrix‑multiplications use cuBLAS/Tensor‑Cores.  Mixed‑precision
(half‑precision) is used for the bulk of the work while the final
output is `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def row_stats_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    a_ptr,          #       float*   [total_rows]   (scale = 1/√var+ε)
    b_ptr,          #       float*   [total_rows]   (bias = -mean/√var+ε)
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise mean/variance → a = 1/√(var+ε) and b = -mean/√(var+ε)
    Only the two scalars per row are written.  The kernel works in FP32
    accumulators for numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- first pass : sum & sum of squares ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+ε)

    # ---- write a and b ------------------------------------------------------
    a = rstd                           # scale
    b = -mean * rstd                   # bias
    tl.store(a_ptr + offs_m, a, mask=mask_m)
    tl.store(b_ptr + offs_m, b, mask=mask_m)


@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*    [total_rows, hidden_dim]
    y_ptr,          #       half*    [total_rows, hidden_dim] (LN result * gate)
    gate_ptr,       # const half*   [total_rows, hidden_dim] (out_gate)
    weight_ptr,     # const half*   [hidden_dim]                (γ)
    bias_ptr,       # const half*   [hidden_dim]                (β)
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Hidden‑dim LayerNorm (float32 accumulator) followed by
    element‑wise multiplication with an already‑sigmoided gate.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- 1st pass: mean / variance -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        ptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- 2nd pass: normalize, apply γ/β, multiply by gate ------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load X
        x_ptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        x_val = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ / β (stored as half but we promote to FP32 for the arithmetic)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        gate_ptrs = gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        gate = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        out = out * gate

        # store back as half
        y_ptrs = y_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (AlphaFold‑3 style).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/int) – optional
        - weights_dict : mapping from layer names to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1. unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ------------------------------------------------------------------
    # 2. row‑wise statistics (a = 1/√var, b = -mean/√var)
    # ------------------------------------------------------------------
    a = torch.empty(total_rows, dtype=torch.float32, device=device)
    b = torch.empty(total_rows, dtype=torch.float32, device=device)

    X_flat = input_tensor.view(total_rows, dim).contiguous()  # [R,D] float32
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    row_stats_kernel[grid](
        X_flat,
        a,
        b,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # cast a,b to fp16 for the upcoming element‑wise scaling
    a_h = a.to(torch.float16)   # [R]
    b_h = b.to(torch.float16)   # [R]

    # ------------------------------------------------------------------
    # 3. single fused linear (raw projections)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: (5·H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, D)

    # pre‑compute per‑row sums of each weight row: Σ_d w_{row,d}
    w_sum = w_concat.sum(dim=1)             # (5*H)   half

    # cast input to half for the GEMM (Tensor‑Core friendly)
    X_half = X_flat.to(torch.float16)

    # raw projection: [R, D] @ [5·H, D]ᵀ → [R, 5·H]
    proj_raw = F.linear(X_half, w_concat)   # half, shape (R,5*H)

    # ------------------------------------------------------------------
    # 4. de‑normalize (a·P + b·Σw) and split into the five tensors
    # ------------------------------------------------------------------
    # helper to apply the affine transform slice‑wise
    def _norm_slice(raw_slice, sum_slice):
        # raw_slice : [R, H]   half
        # sum_slice : [H]      half
        # a_h, b_h   : [R]     half (broadcasted)
        return raw_slice * a_h[:, None] + b_h[:, None] * sum_slice[None, :]

    H = hidden_dim
    # slice indices
    l0, l1 = 0, H
    g0, g1 = H, 2 * H
    r0, r1 = 2 * H, 3 * H
    rg0, rg1 = 3 * H, 4 * H
    o0, o1 = 4 * H, 5 * H

    # raw slices (half)
    raw_left_proj   = proj_raw[:, l0:l1]
    raw_left_gate   = proj_raw[:, g0:g1]
    raw_right_proj  = proj_raw[:, r0:r1]
    raw_right_gate  = proj_raw[:, rg0:rg1]
    raw_out_gate    = proj_raw[:, o0:o1]

    # corresponding Σw for each slice (half)
    sum_left_proj   = w_sum[l0:l1]
    sum_left_gate   = w_sum[g0:g1]
    sum_right_proj  = w_sum[r0:r1]
    sum_right_gate  = w_sum[rg0:rg1]
    sum_out_gate    = w_sum[o0:o1]

    # apply affine de‑normalisation
    left_proj_norm  = _norm_slice(raw_left_proj,   sum_left_proj)   # [R,H] half
    left_gate_norm  = _norm_slice(raw_left_gate,   sum_left_gate)   # [R,H] half
    right_proj_norm = _norm_slice(raw_right_proj,  sum_right_proj)  # [R,H] half
    right_gate_norm = _norm_slice(raw_right_gate,  sum_right_gate)  # [R,H] half
    out_gate_norm   = _norm_slice(raw_out_gate,    sum_out_gate)    # [R,H] half

    # ------------------------------------------------------------------
    # 5. masking (if required) – applied before gating as in the reference
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)   # [R,1]
        left_proj_norm  = left_proj_norm * mask_h
        right_proj_norm = right_proj_norm * mask_h

    # ------------------------------------------------------------------
    # 6. gated projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_norm.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_norm.float()).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_norm.float()).to(torch.float16)

    left = left_proj_norm * left_gate          # [R, H] half
    right = right_proj_norm * right_gate        # [R, H] half

    # ------------------------------------------------------------------
    # 7. Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)                # [B,N,N,H] half
    right = right.view(B, N, N, H)              # [B,N,N,H] half

    # bring hidden dim to batch axis for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous() # transpose k↔j

    BH = B * H
    left_mat = left_mat.view(BH, N, N)           # [B·H,N,N]
    right_mat = right_mat.view(BH, N, N)         # [B·H,N,N]

    out_mat = torch.bmm(left_mat, right_mat)    # [B·H,N,N] half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 8. Hidden‑dim LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)                # [R,H] half
    out_gate_flat = out_gate.view(total_rows, H)      # [R,H] half

    out_norm_gate = torch.empty_like(out_flat)        # half (final pre‑projection)

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ   (half)
        weights["to_out_norm.bias"].to(torch.float16),     # β   (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gate, w_to_out)          # [R, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
