"""
High‑performance “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣ Fast LayerNorm (torch.nn.functional.layer_norm) on the FP32 input.
2️⃣ Cast to FP16 and run ONE fused GEMM that simultaneously computes the five
   linear projections/gates (left‑proj, left‑gate, right‑proj, right‑gate,
   out‑gate).  The result is kept in‑place – no extra copies are made.
3️⃣ Apply the optional mask (only to the two projections) and the sigmoid
   gates **in‑place** on the slices of the fused GEMM output.
4️⃣ Triangular multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed with a single `torch.einsum` which under the hood uses a
   batched GEMM without any materialised transposes.
5️⃣ Fused LayerNorm + output‑gate (custom Triton kernel) on the hidden
   tensor.
6️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

Only the two LayerNorm‑style kernels are written in Triton; all dense
linear algebra (GEMM, einsum) uses cuBLAS/Tensor‑Cores on H100.  The
implementation minimises memory traffic by re‑using the big GEMM buffer
in‑place and by avoiding unnecessary `contiguous` copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (used for the second LN)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, dim]   (input to LN)
    gate_ptr,       # const half*  [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*  [total_rows, dim]   (output)
    w_ptr,          # const float* [dim]              (γ)
    b_ptr,          # const float* [dim]              (β)
    total_rows,     # i32
    dim: tl.constexpr,          # static (hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance (float32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------- normalize, affine, multiply gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load normalized input
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # LN
        out = out * g_vals                            # apply output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]      (float32 or bool)
        - weights_dict : dict of torch.Tensors containing all model weights
        - config_dict  : dict with keys:
            * dim          – input / output channel dimension
            * hidden_dim   – hidden dimension inside TriMul
            * nomask (opt) – if True the mask is ignored
            * eps (opt)    – layer‑norm epsilon (default 1e‑5)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = config.get("eps", 1e-5)
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape                    # [B, N, N, dim]
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Fast LayerNorm (torch) – reduces overhead vs custom kernel
    # ------------------------------------------------------------------
    # NOTE: we keep the custom kernel for the *second* LN (fused with gate)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                   # FP32, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear algebra
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (proj + gates) into ONE GEMM
    # ------------------------------------------------------------------
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [(5*H), dim]

    proj = F.linear(x_norm_h, weight_concat)           # [R, 5*H]  FP16

    # ------------------------------------------------------------------
    # 3️⃣ Slice and apply mask + sigmoid‑gates *in‑place* (no extra buffers)
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj   = proj[:, 0 * h : 1 * h]               # [R, H]
    left_gate   = proj[:, 1 * h : 2 * h]
    right_proj  = proj[:, 2 * h : 3 * h]
    right_gate  = proj[:, 3 * h : 4 * h]
    out_gate_raw = proj[:, 4 * h : 5 * h]              # [R, H]

    if not nomask:
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)   # [R,1]
        left_proj.mul_(mask_h)
        right_proj.mul_(mask_h)

    # Apply sigmoid‑gates
    left_proj.mul_(torch.sigmoid(left_gate))
    right_proj.mul_(torch.sigmoid(right_gate))

    # The gate that modulates the *second* LayerNorm output
    out_gate = torch.sigmoid(out_gate_raw)            # [R, H]   FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the triangular multiplicative update
    # ------------------------------------------------------------------
    left  = left_proj.view(B, N, N, h)                # [B,N,N,H]
    right = right_proj.view(B, N, N, h)               # [B,N,N,H]

    # Contract over the middle dimension k  (einsum uses an efficient batched GEMM)
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat      = out.view(total_rows, hidden_dim)          # [R,H]   FP16
    out_gate_flat = out_gate.view(total_rows, hidden_dim)     # [R,H]   FP16
    out_norm_gated = torch.empty_like(out_flat)               # destination (FP16)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    )                                                   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
