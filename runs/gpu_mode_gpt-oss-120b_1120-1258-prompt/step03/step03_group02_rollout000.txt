"""
TriMul “outgoing” (AlphaFold‑3) forward pass with Triton‑accelerated LayerNorm.

Algorithm
---------
1️⃣ **LayerNorm (FP32→FP16)** – a tiny Triton kernel that computes μ and σ²
   per row of the flattened `[B·N·N, dim]` tensor and writes the normalized
   result directly in FP16, applying the learned γ/β.

2️⃣ **Fused projection + gate weights** – two concatenated linear layers
   (proj + gate) are executed with a single `torch.nn.functional.linear`
   call each, staying in FP16.

3️⃣ **Mask (optional)** – broadcasted over the hidden dimension and applied
   in‑place on the projected tensors.

4️⃣ **Sigmoid gates** – element‑wise `torch.sigmoid`.

5️⃣ **Tri‑multiplicative update** – for each hidden channel we need

        out_h(i,j) = Σ_k left(i,k,h) · right(j,k,h)

   This is a batched matrix multiplication:
   `left` and `right` are reshaped to `[B·H, N, N]` → `out = left @ rightᵀ`
   using `torch.bmm`.  The operation stays in FP16 (Tensor‑Cores).

6️⃣ **Second LayerNorm (FP16)** – same Triton kernel as step 1.

7️⃣ **Output‑gate & final projection** – the normalized tensor is multiplied
   by the output‑gate, then a final FP16 linear layer projects `hidden_dim →
   dim`.  The result is cast back to FP32.

Only the two LayerNorm passes are written in Triton; all dense linear
algebra (GEMMs) uses PyTorch’s highly‑optimised FP16 kernels.  The routine
handles the presence or absence of a mask and works for all configurations
listed in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (writes FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       half*  [total_rows, dim]  (output)
    w_ptr,          # const float* [dim]  (γ)
    b_ptr,          # const float* [dim]  (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (≤384)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program (e.g. 256)
    BLOCK_D: tl.constexpr,      # columns per inner loop (e.g. 64)
):
    """Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    Internally works in FP32, writes FP16 to cut memory traffic."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)                        # cast to fp16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor[bs, seq_len, seq_len, dim] (fp32)
              - mask_tensor : torch.Tensor[bs, seq_len, seq_len] (bool/float)
              - weights_dict: dict with all linear / norm parameters (fp32)
              - config_dict : {"dim": int, "hidden_dim": int,
                               "nomask": bool (optional)}

    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim),
                           dtype=torch.float16,
                           device=device)

    # Tune block sizes: BLOCK_D must cover dim (max 384); BLOCK_M 256 works well.
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2️⃣ Fuse proj + gate (4 GEMMs total)
    # ------------------------------------------------------------------
    # Build concatenated weight matrices of shape (2*hidden_dim, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, non_blocking=True)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, non_blocking=True)

    # Linear passes (FP16)
    x_norm_flat_h = x_norm_h.view(total_rows, dim)  # for reuse
    left_concat = F.linear(x_norm_flat_h, left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_flat_h, right_concat_w)

    # Slice projections & pre‑gates
    left_proj_h   = left_concat[:, :hidden_dim]                 # [rows, H]
    left_gate_pre = left_concat[:, hidden_dim:]                # [rows, H]
    right_proj_h  = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Output gate (single linear, same shape as other tensors)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_flat_h, weights["out_gate.weight"].to(torch.float16))
    )   # [rows, H]  (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates and multiply
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)    # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h            # [rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # Permute to [B, H, N, N] for the batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)          # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)        # [BH, N, N]

    # out = left @ rightᵀ   → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]  (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(B * N * N, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(B * N * N, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B * N * N,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    out_norm = out_norm * out_gate                    # gated tensor (FP16)

    # Final linear projection hidden_dim → dim (FP16)
    final_h = F.linear(
        out_norm.view(B * N * N, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [rows, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
