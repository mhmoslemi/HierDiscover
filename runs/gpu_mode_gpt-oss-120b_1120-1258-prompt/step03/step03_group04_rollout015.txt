@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     total_rows, dim,
                     eps: tl.constexpr,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean

def layernorm_normonly_kernel(
    x_ptr, y_ptr,
    total_rows, dim,

    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,

def custom_kernel(data):
    """
    Highly‑optimized “outgoing” TriMul forward pass (AlphaFold‑3 style).

    The implementation fuses the first LayerNorm into a custom Triton kernel
    that writes FP16 results directly, folds the layer‑norm parameters into the
    subsequent linear projection to avoid an extra memory pass, and uses
    cuBLAS‑Tensor‑Core GEMMs for the heavy matrix multiplications.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # original channel dimension C
    hidden_dim = config["hidden_dim"] # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (no affine, only mean/var) – output FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                     # [total_rows, C] fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_normonly_kernel[grid](
        x_flat,
        x_norm,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM
    #    (weights are folded with the first LayerNorm affine parameters)
    # ------------------------------------------------------------------
    # concatenate projection weights
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, C)

    # fold first‑norm affine parameters into the linear weight / bias
    norm_w = weights["norm.weight"]   # (C)
    norm_b = weights["norm.bias"]     # (C)

    # weight_scaled = (concat_w * norm_w)  (broadcast over columns)
    weight_scaled = (concat_w * norm_w[None, :]).half().contiguous()   # (5*H, C) fp16
    # bias_concat = concat_w @ norm_b   (shape 5*H)
    bias_concat = torch.mv(concat_w.float(), norm_b.float()).contiguous()  # fp32

    # fused projection (FP16 GEMM)
    proj_all = F.linear(x_norm, weight_scaled, bias_concat)   # [total_rows, 5*H] fp16

    # split into the five tensors
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)  # (total_rows, 1)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)       # (total_rows, H) fp16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)          # (B, N, N, H) fp16
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # (B, H, N, N) layout -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose on k‑axis

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # (B*H, N, N) fp16
    right_mat = right_perm.view(BH, N, N) # (B*H, N, N) fp16

    out_hidden = torch.bmm(left_mat, right_mat)   # (B*H, N, N) fp16

    # reshape to (B, N, N, H)
    out = out_hidden.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (affine) – outputs FP16
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # (total_rows_h, H) fp16
    out_norm = torch.empty_like(out_flat)                  # fp16 output

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_weighted_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate (in‑place) and final linear projection
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)
    out_norm.mul_(out_gate_flat)  # element‑wise gate, stays in fp16

    final_fp16 = F.linear(out_norm, weights["to_out.weight"].half())   # (total_rows_h, C) fp16

    # Cast to FP32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm that only standardises (no affine) and writes FP16.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_normonly_kernel(
    x_ptr,                      # *const float*   [total_rows, dim]
    y_ptr,                      # *mut   half*    [total_rows, dim]
    total_rows, dim,            # runtime args
    eps: tl.constexpr,          # compile‑time constant
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ mean / variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ write normalised FP16 output
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32
        out_fp16 = norm.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  LayerNorm with affine (used for the second LN)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_weighted_kernel(
    x_ptr,                      # *const float*   [rows, dim]
    y_ptr,                      # *mut   half*    [rows, dim]
    weight_ptr,                 # *const float*   [dim]
    bias_ptr,                   # *const float*   [dim]
    total_rows, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * w[None, :] + b[None, :]           # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3).

    1️⃣ Normalise the input with a Triton kernel that writes FP16.
    2️⃣ Fuse the five linear projections (left/right + 3 gates) with the
       first LayerNorm affine parameters → a single FP16 GEMM.
    3️⃣ Optional mask, sigmoid gates and element‑wise gating.
    4️⃣ Batched GEMM implements the triangular multiplicative update.
    5️⃣ Second LayerNorm (affine) via Triton.
    6️⃣ In‑place output‑gate multiplication and final FP16 linear projection.
    7️⃣ Cast to FP32 for the API contract.
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # shapes
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (no affine) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [total_rows, C]  float32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_normonly_kernel[grid](
        x_flat,
        x_norm,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ fused projection (fold first‑norm affine into the weight)
    # --------------------------------------------------------------
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    # concat all five weight matrices → (5*H, C)
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # fp32

    # first‑norm affine parameters
    norm_w = weights["norm.weight"]   # (C)
    norm_b = weights["norm.bias"]     # (C)

    # fold γ (norm.weight) into the linear weight
    weight_scaled = (concat_w * norm_w[None, :]).half().contiguous()   # (5*H, C) fp16

    # bias = W * β (norm.bias)
    bias_concat = torch.mv(concat_w.float(), norm_b.float()).contiguous()  # (5*H,) fp32

    # single fused GEMM (FP16)
    proj_all = F.linear(x_norm, weight_scaled, bias_concat)   # [total_rows, 5*H] fp16

    # split back to the five tensors
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # --------------------------------------------------------------
    # 3️⃣ optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ sigmoid gates & element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)        # (total_rows, H) fp16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_proj_raw * left_gate                # (total_rows, H) fp16
    right = right_proj_raw * right_gate

    # reshape to the original 4‑D layout
    left = left.view(B, N, N, hidden_dim)          # (B, N, N, H) fp16
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # (B, H, N, N) → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose k‑axis

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # (B*H, N, N) fp16
    right_mat = right_perm.view(BH, N, N)  # (B*H, N, N) fp16

    out_hidden = torch.bmm(left_mat, right_mat)   # (B*H, N, N) fp16

    # back to (B, N, N, H)
    out = out_hidden.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm (affine) → FP16
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # (total_rows_h, H) fp16
    out_norm = torch.empty_like(out_flat)                  # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_weighted_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 7️⃣ gating + final projection (all in FP16)
    # --------------------------------------------------------------
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)
    out_norm.mul_(out_gate_flat)               # in‑place gate, stays fp16

    final_fp16 = F.linear(out_norm, weights["to_out.weight"].half())  # (total_rows_h, C) fp16

    # --------------------------------------------------------------
    # 8️⃣ cast back to FP32 for the API contract
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
