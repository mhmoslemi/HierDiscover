@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch dimension
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # col tile

    # Compute offsets for tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to start of batch
    a_ptr_base = a_ptr + pid_b * stride_a_batch
    b_ptr_base = b_ptr + pid_b * stride_b_batch
    c_ptr_base = c_ptr + pid_b * stride_c_batch

    # Load blocks in loop over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for current k tile
        a_ptrs = a_ptr_base + (offs_m[:, None] * stride_a_m) + ((k * BLOCK_K + offs_k)[None, :] * stride_a_k)
        b_ptrs = b_ptr_base + ((k * BLOCK_K + offs_k)[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        # Load A and B blocks with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp32 for accumulation
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)

        # Dot product
        acc += tl.dot(a_fp32, b_fp32)

    # Write result
    c_ptrs = c_ptr_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, C = input_tensor.shape
    dim = config["dim"]  # should equal C

    eps = 1e-5
    nomask = config.get("nomask", True)
    # Cast weights to half where possible
    norm_weight = weights["norm.weight"].half()
    norm_bias = weights["norm.bias"].half()
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    # Fused projection weight
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    concat_weight = torch.cat([left_proj_weight, right_proj_weight,
                               left_gate_weight, right_gate_weight,
                               out_gate_weight], dim=0).half().contiguous()
    to_out_weight = weights["to_out.weight"].half().contiguous()

    # Flatten and layernorm
    total_rows = bs * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64  # increase tile for channels
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # Project all 5 linear layers in one GEMM
    proj_all = F.linear(x_norm, concat_weight)  # fp16 out
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [total_rows, H]

    # Apply mask if needed
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # Gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp = left_proj * left_gate
    right_fp = right_proj * right_gate

    # Reshape to 4-D for batched gemm
    left = left_fp.view(bs, N, N, hidden_dim)   # [B, N, N, H]
    right = right_fp.view(bs, N, N, hidden_dim) # [B, N, N, H]
    out_gate = out_gate.view(bs, N, N, hidden_dim)

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.empty_like(left_mat)

    BLOCK_N = 64
    BLOCK_K = 64

        right_mat.stride(1),

        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    # Second LayerNorm
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat)
    grid2 = (triton.cdiv(total_rows, 128),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        to_out_norm_weight,
        to_out_norm_bias,
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=128,
        BLOCK_D=64,
    )
    out_norm = out_norm.view(bs, N, N, hidden_dim)

    # Apply output gate
    gated = out_norm * out_gate  # fp16

    # Final linear projection H -> C
    final_fp16 = F.linear(gated.view(total_rows, hidden_dim), to_out_weight)  # fp16

    # Cast back to fp32 and reshape
    final = final_fp16.view(bs, N, N, dim).to(torch.float32)

    return final

@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Masks for K bound
        mask_k = cur_k < K

        # Pointers to sub‑tiles
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # compute mean & variance
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])  # x is stored as fp32?
        # Wait x input to layernorm is fp32 because we treat input_tensor as float32; they used x_flat = input_tensor.view(total_rows, dim). We'll pass x_flat (float32). So x_ptr is fp32 pointer.
        # We'll load as float32.

        sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)

"""
Optimized Triton implementation of the “outgoing” TriMul operator
from AlphaFold‑3 (Triangular multiplicative update).

Algorithm
---------
1️⃣  Layer‑norm on the input tensor (fp32 → fp16) – custom Triton kernel
    that computes mean/var in fp32 and writes the normalized result in fp16.
2️⃣  Fuse the 5 linear projections (left/right and three gates) into a single
    GEMM (`F.linear`) by concatenating the weight matrices.  All work is done
    in fp16 (Tensor‑Core friendly).
3️⃣  Optional pair‑wise mask is applied to the left/right projections.
4️⃣  Sigmoid gates are evaluated and applied (still in fp16).
5️⃣  Triangular multiplication (the core `einsum`) is realised as a batched
    matrix‑multiply (`(B·H) × N × N`) using a custom Triton kernel.
    This avoids the overhead of launching many small `torch.bmm` calls.
6️⃣  Second LayerNorm over the hidden dimension – same Triton kernel.
7️⃣  Multiply by the output gate, project back to `dim` with a final linear
    (fp16) and cast the result to float32.

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; the only dense linear‑algebra kernels are the fused
projection (`torch.nn.functional.linear`) and the batched GEMM
implemented in Triton.  The implementation works for any of the
configuration shapes listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm – fp32 accumulation, fp16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const half*   [dim]              (γ, fp16)
    bias_ptr,       # *const half*   [dim]              (β, fp16)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – compute row‑wise mean and variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Batched fp16 GEMM – computes C = A @ B for many (batch, M, N, K)
#   A: (B, M, K)   B: (B, K, N)   C: (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)          # batch index  (0 … B‑1)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)           # fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)           # fp16

        # tl.dot automatically uses tensor‑core fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write back in fp16
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C]  (float32)
            - mask_tensor : [B, N, N]    (bool / float, optional)
            - weights_dict: model weights
            - config_dict : contains "dim", "hidden_dim", and "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack shape / config
    # ------------------------------------------------------------------
    B, N, _, C = input_tensor.shape          # [B, N, N, C]
    dim = config["dim"]                      # C
    hidden_dim = config["hidden_dim"]        # H
    nomask = config.get("nomask", True)

    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # fp32
    x_norm = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Cast LayerNorm weights/bias to half (store‑bandwidth reduction)
    norm_weight = weights["norm.weight"].half().contiguous()
    norm_bias   = weights["norm.bias"].half().contiguous()

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        x_flat,
        x_norm,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output dimension: (5*H, C)
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    # Single GEMM: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj = F.linear(x_norm, concat_weight)    # fp16

    # Split back to the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=1
    )  # each: [total_rows, H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp = left_proj * left_gate      # [total_rows, H]
    right_fp = right_proj * right_gate   # [total_rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors
    left = left_fp.view(B, N, N, hidden_dim)     # [B, N, N, H]
    right = right_fp.view(B, N, N, hidden_dim)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Bring the hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k‑j transposed)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)               # [BH, N, N]   fp16
    right_mat = right_perm.view(BH, N, N)             # [BH, N, N]   fp16

    # Output buffer for the batched matmul
    out_mat = torch.empty_like(left_mat)              # fp16

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_gemm = (BH,
                 triton.cdiv(N, BLOCK_M),
                 triton.cdiv(N, BLOCK_N))

    batched_matmul_fp16_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat)  # fp16

    to_out_norm_weight = weights["to_out_norm.weight"].half().contiguous()
    to_out_norm_bias   = weights["to_out_norm.bias"].half().contiguous()

    grid_ln2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln2](
        out_flat,
        out_norm,
        to_out_norm_weight,
        to_out_norm_bias,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)       # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by output gate and final projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16

    to_out_weight = weights["to_out.weight"].half().contiguous()
    final_fp16 = F.linear(gated.view(total_rows_h, hidden_dim), to_out_weight)  # fp16 -> [total_rows_h, C]

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
