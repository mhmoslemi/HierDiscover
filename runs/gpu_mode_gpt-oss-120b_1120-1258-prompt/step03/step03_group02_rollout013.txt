"""
TriMul “outgoing” implementation (AlphaFold‑3 style) – optimized version.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – a tiny Triton kernel.
2️⃣ Fuse all linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a *single* half‑precision GEMM
   (weight matrix of shape (5·hidden_dim, dim)).
3️⃣ Apply optional pairwise mask (broadcast over hidden_dim) *before*
   the sigmoid gates.
4️⃣ Apply sigmoid‑gates and multiply the gated projections.
5️⃣ Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   This is a batched GEMM (B·hidden_dim matrices of size N×N) performed
   with Torch’s FP16 `torch.bmm`.
6️⃣ Second row‑wise LayerNorm (FP16) – same Triton kernel.
7️⃣ Multiply by the output‑gate and a final FP16 linear projection back
   to `dim`.  The result is cast to float32 for the module output.

The critical speed‑up over the reference implementation is step 2: all
five linear layers are collapsed into one GEMM, cutting the memory traffic
from three reads of the normalized tensor to a single read.  The two
LayerNorms are still executed in Triton to keep the data in FP16 and
avoid extra copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim]  (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- pass 1: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----------- pass 2: write normalized values --------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)                       # cast to half

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 -> FP16)
    # --------------------------------------------------------------
    # Flatten to 2‑D for the kernel
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # Allocate output buffer (FP16)
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Choose block sizes (tuned for typical dims)
    BLOCK_M = 128
    # make BLOCK_D a multiple of 32 that covers dim
    if dim <= 32:
        BLOCK_D = 32
    elif dim <= 64:
        BLOCK_D = 64
    elif dim <= 128:
        BLOCK_D = 128
    else:
        BLOCK_D = 256

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate + out‑gate (single GEMM)
    # --------------------------------------------------------------
    # Concatenate all five weight matrices (each shape [hidden_dim, dim])
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # shape (5*hidden_dim, dim)

    # Single linear pass on the normalized tensor
    proj_all = F.linear(x_norm_h.view(total_rows, dim), weight_concat)   # (total_rows, 5*hidden_dim)

    # Slice the concatenated result
    h = hidden_dim
    left_proj_h      = proj_all[:, 0 * h : 1 * h]          # [rows, h]
    left_gate_pre_h  = proj_all[:, 1 * h : 2 * h]          # [rows, h]
    right_proj_h     = proj_all[:, 2 * h : 3 * h]          # [rows, h]
    right_gate_pre_h = proj_all[:, 3 * h : 4 * h]          # [rows, h]
    out_gate_pre_h   = proj_all[:, 4 * h : 5 * h]          # [rows, h]

    # --------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)

    left_h  = left_proj_h * left_gate_h      # [rows, hidden_dim]
    right_h = right_proj_h * right_gate_h    # [rows, hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, hidden_dim]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Bring hidden dim to batch axis: [B, hidden_dim, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                # batch * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]

    # Batched GEMM: out = left @ rightᵀ   (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, hidden_dim] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) – fuse with final projection later
    # --------------------------------------------------------------
    total_rows_h = total_rows               # same number of rows
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection (hidden_dim → dim)
    # --------------------------------------------------------------
    # Element‑wise gate
    out_norm = out_norm * out_gate_h.view(B, N, N, hidden_dim)

    # Final linear projection (FP16 GEMM)
    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # shape (total_rows_h, dim), dtype=FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
