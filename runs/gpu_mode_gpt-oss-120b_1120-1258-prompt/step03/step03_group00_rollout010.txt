    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_j_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # for i (block row)
    pid_n = tl.program_id(1) # for j (block col)
    pid_bh = tl.program_id(2) # batch id = b*H + h

    h = pid_bh % H

    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
Optimized “outgoing” TriMul kernel (AlphaFold‑3) – pure‑CUDA implementation.

Algorithm
---------
1️⃣  Layer‑norm on the last channel (C) – custom Triton kernel that
    accumulates in fp32 and writes fp16 (used twice, before and after the
    triangular update).

2️⃣  Fuse the five linear projections (left/right + three gates) into a
    single GEMM by concatenating their weight matrices.  All computations
    are done in fp16 (Tensor‑Core friendly).

3️⃣  (Optional) apply the pairwise mask to left/right projections before
    gating.

4️⃣  Apply sigmoid gates (still fp16) and element‑wise gating.

5️⃣  Triangular multiplicative update (`out[b,i,j,h] = Σₖ left[b,i,k,h] *
    right[b,j,k,h]`) – a custom Triton batched‑GEMM that works directly on
    the original `[B,N,N,H]` layout, thus avoiding the expensive
    `permute(...).contiguous()` copies required by `torch.bmm`.

6️⃣  Second Layer‑norm over the hidden dimension (H).

7️⃣  Multiply by the output gate and project back to `dim` with a final
    linear layer (fp16 → fp32 output).

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra uses fused fp16 GEMMs (cuBLAS /
Tensor‑Core).  The implementation works for `B∈{1,2}`, `N∈{128,…,1024}`,
`C∈{128,384,768}` and `H=hidden_dim∈{128}`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel – fp16 output, fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const {fp16,fp32}*  [total_rows, dim]
    y_ptr,          # *mut   fp16*         [total_rows, dim]  (output)
    weight_ptr,     # *const fp32*         [dim]              (γ)
    bias_ptr,       # *const fp32*         [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,               # int (compile‑time)
    eps: tl.constexpr,               # float (compile‑time)
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_D: tl.constexpr,           # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------ pass 1 : mean / var --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp16→fp32 if needed
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------ pass 2 : write normalized --------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32
        out_fp32 = norm * w + b                               # fp32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triangular multiplicative update – custom batched GEMM on [B,N,N,H]
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # all fp16
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_j_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr,    # rows per tile
    BLOCK_N: tl.constexpr,    # cols per tile
    BLOCK_K: tl.constexpr,    # reduction tile
):
    # program ids
    pid_i = tl.program_id(0)          # tile on output row (i)
    pid_j = tl.program_id(1)          # tile on output col (j)
    pid_bh = tl.program_id(2)         # combined batch-head id = b*H + h

    # compute b and h from the combined index
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_b_left  + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base   = out_ptr   + b * stride_b_out   + h * stride_h_out

    # offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k (the shared dimension)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile A[i,k] ------------------------------------------------
        # left[b,i,k,h]
        a_ptrs = left_base + (offs_i[:, None] * stride_i_left) + (offs_k[None, :] * stride_k_left)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # (BM, BK)

        # ---- right tile B_T[k,j] = right[b,j,k,h] -----------------------------
        # right[b,j,k,h] – we load it transposed so that the inner dimension (=k)
        # is the first axis.
        b_ptrs = right_base + (offs_k[:, None] * stride_k_right) + (offs_j[None, :] * stride_j_right)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)               # (BK, BN)

        # GEMM: (BM,BK) @ (BK,BN) -> (BM,BN)
        acc += tl.dot(a, b)

    # store result (fp16)
    out_ptrs = out_base + (offs_i[:, None] * stride_i_out) + (offs_j[None, :] * stride_j_out)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint – TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N]  (may be None)
            - weights_dict : dictionary of model parameters (see sketch)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C], dtype float32, on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape          # [B, N, N, C]
    H = config["hidden_dim"]                 # hidden dimension
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    total_rows = B * N * N                     # rows for flattened view
    x_flat = input_tensor.reshape(total_rows, C).contiguous()
    x_norm_fp16 = torch.empty((total_rows, C), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projections & gates (all fp16)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape (5*H, C))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()          # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)   fp16

    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, H, dim=1
    )  # each (total_rows, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask before gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate          # (total_rows, H)
    right = right_proj * right_gate       # (total_rows, H)

    # reshape back to 4‑D tensors that will be consumed by the custom kernel
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Triangular multiplicative update – custom Triton GEMM
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides for the three tensors (as integer numbers of elements)
    s_b_left, s_i_left, s_k_left, s_h_left = left.stride()
    s_b_right, s_j_right, s_k_right, s_h_right = right.stride()
    s_b_out, s_i_out, s_j_out, s_h_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_matmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_b_left,
        s_i_left,
        s_k_left,
        s_h_left,
        s_b_right,
        s_j_right,
        s_k_right,
        s_h_right,
        s_b_out,
        s_i_out,
        s_j_out,
        s_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dim H
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating and projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16 element‑wise
    # final linear: H → C
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C), fp16

    # cast to fp32 as required by the spec
    final = final_fp16.view(B, N, N, C).to(torch.float32)

    # clean up large temporaries (optional, helps peak memory)
    del x_norm_fp16, proj_all, left, right, out, out_norm, out_norm_fp16, gated, gated_flat

    return final
