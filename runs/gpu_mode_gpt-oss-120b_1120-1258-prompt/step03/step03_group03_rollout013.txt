"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  **Row‑wise LayerNorm (FP32 → FP16)**  
   A fused Triton kernel computes the per‑row mean / variance of the
   FP32 input `x` (shape `[R, C]`, R = B·N·N, C = dim) and writes the
   normalized result directly as FP16 (`x_norm`).  This halves the memory
   traffic of the original two‑step implementation (norm → cast).

2️⃣  **Single fused linear projection**  
   All five weight matrices (`left_proj`, `left_gate`, `right_proj`,
   `right_gate`, `out_gate`) are concatenated → shape `(5·H, C)`.  A
   half‑precision `F.linear(x_norm, W_concat)` gives a tensor of shape
   `[R, 5·H]` containing the three projections and three gates.

3️⃣  **Mask & sigmoid gates** – applied element‑wise on the half‑precision
   tensors (very cheap).

4️⃣  **Tri‑multiplicative update** – reshape the gated projections to
   `[B, N, N, H]`, permute so that the hidden dimension becomes the batch
   axis and perform a batched GEMM (`torch.bmm`).  This yields the
   intermediate tensor `out` of shape `[B, N, N, H]` (FP16).

5️⃣  **Hidden‑dim LayerNorm + output‑gate** – a second Triton kernel
   normalises `out` (still FP16), applies the learned `γ/β` of the
   output‑norm and multiplies by the pre‑computed `out_gate`.  The result
   (`out_norm_gate`) stays in FP16.

6️⃣  **Final linear projection** – a half‑precision linear layer projects
   back to the original channel dimension `dim`.  The result is reshaped
   to `[B, N, N, dim]` and cast to FP32 (the required output type).

Only the two LayerNorm steps are written in Triton; the heavy matrix
multiplications (`F.linear`, `torch.bmm`) use cuBLAS/Tensor‑Core which
is optimal on an H100.  Mixed‑precision (FP16) is used everywhere except
for the statistical accumulation in the LayerNorm kernels (FP32), which
preserves numerical stability while reducing memory bandwidth.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_fp16_out_kernel(
    x_ptr,          # const float*          [total_rows, dim]    (FP32 input)
    y_ptr,          #       half*          [total_rows, dim]    (FP16 output)
    weight_ptr,     # const half*          [dim]                (γ, FP16)
    bias_ptr,       # const half*          [dim]                (β, FP16)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm in FP32 (for statistics) with FP16 output.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Compute per‑row sum and sum‑of‑squares (FP32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        # load FP32 input, cast to FP32 (already FP32) for accumulation
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                           # [BLOCK_M, BLOCK_D] fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                       # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2) Normalise and write FP16 output
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                           # [BLOCK_M, BLOCK_D] fp32

        norm = (x - mean[:, None]) * rstd[:, None]   # fp32

        # γ & β are stored as FP16 → cast to FP32 for the arithmetic
        w = tl.load(weight_ptr + offs_d, mask=mask_d).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d).to(tl.float32)

        out = norm * w[None, :] + b[None, :]         # fp32
        # store as FP16
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,          # const half*          [total_rows, dim]    (FP16 input)
    y_ptr,          #       half*          [total_rows, dim]    (FP16 output)
    gate_ptr,       # const half*          [total_rows, dim]    (FP16 gate)
    weight_ptr,     # const half*          [dim]                (γ, FP16)
    bias_ptr,       # const half*          [dim]                (β, FP16)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    LayerNorm (FP32 accum) + γ/β (FP16) + element‑wise multiplication with a
    gate tensor (FP16).  The result stays in FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Statistics (mean / variance) – FP32 accumulator
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2) Normalise, apply γ/β and multiply by gate
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (FP16 → FP32)
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d).to(tl.float32)

        out = norm * w[None, :] + b[None, :]          # fp32

        gate = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32

        out = out * gate                               # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()                     # [B,N,N,D]  float32
    x_flat = x.view(total_rows, dim)                  # [R,D]  float32
    x_norm_half = torch.empty((total_rows, dim),
                               dtype=torch.float16,
                               device=device)     # output buffer (FP16)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    # cast LayerNorm parameters to FP16 for the kernel
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16   = weights["norm.bias"].to(torch.float16)

    layernorm_fp16_out_kernel[grid](
        x_flat,
        x_norm_half,
        norm_weight_fp16,
        norm_bias_fp16,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 3️⃣ Fused projection & gating (single GEMM)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5*H, D)

    # fused linear:  [R,D] @ (5*H,D)^T → [R,5*H] (FP16)
    proj_all = F.linear(x_norm_half, w_concat)        # half

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2 * H]
    right_proj  = proj_all[:, 2 * H:3 * H]
    right_gate  = proj_all[:, 3 * H:4 * H]
    out_gate    = proj_all[:, 4 * H:5 * H]

    # optional mask (broadcast over hidden dimension)
    if not nomask:
        mask_half = mask_tensor.to(torch.float16).view(total_rows, 1)  # [R,1]
        left_proj = left_proj * mask_half
        right_proj = right_proj * mask_half

    # sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # gated projections
    left  = left_proj * left_gate          # [R, H] half
    right = right_proj * right_gate        # [R, H] half

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)          # [B,N,N,H]
    right = right.view(B, N, N, H)         # [B,N,N,H]

    # bring hidden dim to batch axis for the batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N] (k↔j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)   # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] half

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)               # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)     # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)       # half, final before projection

    # LayerNorm parameters for the hidden dimension
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        to_out_norm_w_fp16,
        to_out_norm_b_fp16,
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final_half = F.linear(out_norm_gate, w_to_out)          # [R, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast back to FP32

    return final
```
