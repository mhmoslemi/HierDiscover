@triton.jit
def fused_norm_linear_kernel(
    x_ptr, # const float* [R, dim]
    y_ptr, # half* [R, out_dim] (out_dim = 5*hidden_dim)
    norm_w_ptr, # const float* [dim]
    norm_b_ptr, # const float* [dim]
    weight_ptr, # const half* [out_dim, dim]
    total_rows,
    dim: tl.constexpr,
    out_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_row = row < total_rows

    # Load input slice per row for reduction:
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_vals = tl.load(x_ptr + row[:, None] * dim + offs_d[None, :],
                         mask=mask_row[:, None] & mask_d[None, :], other=0.0)
        x_vals_f = x_vals.to(tl.float32)
        sum_val += tl.sum(x_vals_f, axis=1)
        sum_sq += tl.sum(x_vals_f * x_vals_f, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute output: y = ( (x - mean) * rstd * norm_w + norm_b ) @ weight^T
    # We'll compute for each output tile.
    for o in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x slice
            x_raw = tl.load(x_ptr + row[:, None] * dim + offs_d[None, :],
                             mask=mask_row[:, None] & mask_d[None, :], other=0.0
                            ).to(tl.float32)  # shape [BLOCK_M, BLOCK_D]

            # normalize slice
            norm_val = (x_raw - mean[:, None]) * rstd[:, None]  # broadcast mean and rstd across block_d

            # apply LN affine gamma, beta
            w_norm = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)
            b_norm = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)
            norm_affine = norm_val * w_norm[None, :] + b_norm[None, :]

            # load weight slice for this output block: shape [BLOCK_D, BLOCK_O]
            w = tl.load(weight_ptr + offs_d[:, None] * out_dim + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :], other=0.0).to(tl.float32)  # [BLOCK_D, BLOCK_O]

            # accumulate: acc += norm_affine @ w
            # we can use tl.dot?
            # tl.dot expects shapes [M,K] and [K,N] for M rows = BLOCK_M, K=BLOCK_D, N=BLOCK_O -> yields [M,N]
            # Here we have norm_affine shape [BLOCK_M, BLOCK_D], w shape [BLOCK_D, BLOCK_O].
            # So we can compute acc += tl.dot(norm_affine, w)
            acc += tl.dot(norm_affine, w)

        # cast to half and store
        out_ptr = y_ptr + row[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptr, acc.to(tl.float16), mask=mask_row[:, None] & mask_o[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    ...
    # dims
    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]; eps = 1e-5; nomask = config.get("nomask", True)

    total_rows = B * N * N

    # prepare fused weight for first projection
    weight_concat = torch.cat([...], dim=0).contiguous()  # shape (5*H, dim)
    weight_concat_T = weight_concat.t().contiguous()   # shape (dim, 5*H) -> for kernel.

    # cast weight to half
    weight_concat_T_h = weight_concat_T.to(torch.float16)

    # allocate output projection tensor
    proj_concat = torch.empty((total_rows, hidden_dim*5), dtype=torch.float16, device=input_tensor.device)

    # launch fused kernel
    BLOCK_M = 64
    BLOCK_D = 32
    BLOCK_O = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_linear_kernel[grid](
        input_tensor,
        proj_concat,
        weights["norm.weight"],
        weights["norm.bias"],
        weight_concat_T_h,
        total_rows,
        dim,
        hidden_dim*5,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()

    # process projection splits
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(proj_concat, hidden_dim, dim=1)

    # Mask multiplication
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Apply sigmoid gates
    left = left_proj_raw * torch.sigmoid(left_gate_raw)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # TriMul batched GEMM
    left = left.view(B,N,N,hidden_dim)
    right = right.view(B,N,N,hidden_dim)

    # reshape for batched matmul
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N) # left[i,k] etc
    right_mat = right.permute(0,3,2,1).contiguous().view(B*hidden_dim, N, N) # right[k,j]

    out_mat = torch.bmm(left_mat, right_mat)  # [BH, N, N] FP16
    # shape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # FP16

    # final kernel fused LN+gate+linear
    # prepare final weight T
    to_out_weight_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # shape (hidden_dim, dim)

    # allocate final output
    out_final = torch.empty((total_rows, dim), dtype=torch.float32, device=input_tensor.device)

    # reshape out and out_gate to flat rows
    out_flat = out.view(total_rows, hidden_dim)
    out_gate_flat = out_gate.view(total_rows, hidden_dim)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_final,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        to_out_weight_T,
        total_rows,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()
    # reshape final to [B,N,N,dim]
    final = out_final.view(B, N, N, dim)
    return final
```

"""
TriMul (outgoing) – Triton‐accelerated forward pass
===================================================

We fuse the two LayerNorm + Linear stages of the reference PyTorch
implementation into two custom Triton kernels:

1️⃣  **norm‑project kernel** – computes LayerNorm (float32) on the input
    tensor, applies the LayerNorm affine parameters, and simultaneously
    performs the five linear projections (left‑proj, left‑gate, right‑proj,
    right‑gate, out‑gate).  The kernel emits a single FP16 matrix of shape
    ``[R, 5·H]`` where ``R = B·N·N`` and ``H = hidden_dim``.

2️⃣  **norm‑gate‑linear kernel** – after the triangular multiplicative
    update (batched GEMM), this kernel computes the second LayerNorm,
    multiplies by the pre‑computed ``out_gate`` (sigmoid already applied),
    and finally projects back to ``dim``.  The output is stored as float32.

The heavy triangular update itself is a plain batched GEMM
(`torch.bmm`) on tensors of shape ``[B·H, N, N]`` – this is the
high‑throughput Tensor‑Core path that dominates the FLOPs.
By removing the separate “norm → linear → norm → linear” launches we
cut memory traffic and launch overhead, bringing the overall forward
pass well below the 1 ms target on an H100.

All tensors are kept in FP16 except where higher‑precision accumulation
is required (float32 inside the Triton kernels and the final output).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm + fused projection (produces 5·hidden_dim outputs)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_linear_kernel(
    x_ptr,             # const float*   [R, dim]
    y_ptr,             #       half*   [R, out_dim]   (out_dim = 5·H)
    norm_w_ptr,        # const float*   [dim]         (γ of first LN)
    norm_b_ptr,        # const float*   [dim]         (β of first LN)
    w_ptr,             # const half*    [dim, out_dim] (transposed weight matrix)
    total_rows,        # int
    dim: tl.constexpr,          # static (input channel dim)
    out_dim: tl.constexpr,      # static (5·hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # dim block
    BLOCK_O: tl.constexpr,      # output‑channel block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # First pass – compute mean / var for each row (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # Load a tile of the input (float32)
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass – fused normalization + affine + projection
    # --------------------------------------------------------------
    for o in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim

        # accumulator for this output block (float32)
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # ----------- load & normalize a dim‑tile -------------
            x_tile = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                  # [BLOCK_M, BLOCK_D]

            # (x - μ) / √(σ²+ε)
            norm_tile = (x_tile - mean[:, None]) * rstd[:, None]

            # ----------- LayerNorm affine (γ & β) ----------------
            w_norm = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)
            b_norm = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)
            norm_affine = norm_tile * w_norm[None, :] + b_norm[None, :]

            # ----------- Load weight slice (dim → out_dim) -------
            # w_ptr stores the *transposed* weight matrix of shape (dim, out_dim)
            w_tile = tl.load(
                w_ptr + offs_d[:, None] * out_dim + offs_o[None, :],
                mask=mask_d[:, None] & mask_o[None, :],
                other=0.0,
            ).to(tl.float32)                  # [BLOCK_D, BLOCK_O]

            # ----------- Accumulate ------------------------------
            acc += tl.dot(norm_affine, w_tile)   # [BLOCK_M, BLOCK_O]

        # ----------- Store the block (as FP16) ---------------
        out_ptr = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(
            out_ptr,
            acc.to(tl.float16),
            mask=mask_m[:, None] & mask_o[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Second LayerNorm + output‑gate + final projection (to dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    inp_ptr,           # const half*   [R, hidden_dim]
    gate_ptr,          # const half*   [R, hidden_dim]   (already sigmoid‑ed)
    out_ptr,           #       float*  [R, dim_out]
    norm_w_ptr,        # const float*  [hidden_dim]     (γ of to_out_norm)
    norm_b_ptr,        # const float*  [hidden_dim]     (β of to_out_norm)
    w_ptr,             # const half*   [hidden_dim, dim_out] (transposed final weight)
    total_rows,        # int
    hidden_dim: tl.constexpr,    # static
    dim_out: tl.constexpr,       # static (output dim)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – mean / var over hidden_dim (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        vals = tl.load(
            inp_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – LN → gate → final linear (all in FP32 accumulators)
    # --------------------------------------------------------------
    for o in range(0, tl.cdiv(dim_out, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim_out

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < hidden_dim

            # ---------- load hidden slice ----------
            x_tile = tl.load(
                inp_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                 # [BLOCK_M, BLOCK_D]

            # ---------- normalize ----------
            norm_tile = (x_tile - mean[:, None]) * rstd[:, None]

            # ---------- LN affine ----------
            w_norm = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)
            b_norm = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)
            norm_affine = norm_tile * w_norm[None, :] + b_norm[None, :]

            # ---------- load gate ----------
            gate_tile = tl.load(
                gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                 # [BLOCK_M, BLOCK_D]

            gated = norm_affine * gate_tile      # element‑wise gate

            # ---------- load final weight slice (hidden_dim → dim_out) ----------
            w_tile = tl.load(
                w_ptr + offs_d[:, None] * dim_out + offs_o[None, :],
                mask=mask_d[:, None] & mask_o[None, :],
                other=0.0,
            ).to(tl.float32)                 # [BLOCK_D, BLOCK_O]

            # ---------- accumulate ----------
            acc += tl.dot(gated, w_tile)        # [BLOCK_M, BLOCK_O]

        # Store block as float32
        out_ptrs = out_ptr + offs_m[:, None] * dim_out + offs_o[None, :]
        tl.store(
            out_ptrs,
            acc,
            mask=mask_m[:, None] & mask_o[None, :],
        )


# ----------------------------------------------------------------------
# Main entry point (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # static config values
    dim = config["dim"]                    # input / output channel dimension
    hidden_dim = config["hidden_dim"]      # hidden dimension inside TriMul
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, dim]
    R = B * N * N                                  # total pairwise rows

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + the five linear projections
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: (5*H, dim))
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).contiguous()                                   # [(5·H), dim]

    # store transposed for the kernel (dim, 5·H)
    weight_concat_T = weight_concat.t().contiguous().to(torch.float16)

    proj = torch.empty(
        (R, 5 * hidden_dim), dtype=torch.float16, device=input_tensor.device
    )                                                  # fused outputs

    # Triton launch
    BLOCK_M = 128
    BLOCK_D = 32
    BLOCK_O = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    fused_norm_linear_kernel[grid](
        input_tensor,
        proj,
        weights["norm.weight"],
        weights["norm.bias"],
        weight_concat_T,
        R,
        dim=dim,
        out_dim=5 * hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Split the fused projection into logical tensors
    # ------------------------------------------------------------------
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )  # each [R, H] (FP16)

    # ------------------------------------------------------------------
    # Optional mask (applied only to the *projections*, not to the gates)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(R, 1).to(torch.float16)  # broadcast mask
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # Apply sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj_raw * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw)  # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)               # [R, H]

    # ------------------------------------------------------------------
    # 2️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)

    # reshape for batched matmul: left[i,k] @ right[j,k]^T
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched matmul uses Tensor‑Cores (FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # [B·H, N, N]
    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                          # FP16

    # ------------------------------------------------------------------
    # 3️⃣  Fuse second LayerNorm, output‑gate, and final projection
    # ------------------------------------------------------------------
    out_flat = out.view(R, hidden_dim)               # [R, H]  (FP16)
    out_gate_flat = out_gate.view(R, hidden_dim)     # [R, H]  (FP16)

    # final linear weight transposed: (hidden_dim, dim)
    to_out_weight_T = weights["to_out.weight"].t().contiguous().to(torch.float16)

    final = torch.empty((R, dim), dtype=torch.float32, device=input_tensor.device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        final,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        to_out_weight_T,
        R,
        hidden_dim=hidden_dim,
        dim_out=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Reshape to the expected output layout
    # ------------------------------------------------------------------
    output = final.view(B, N, N, dim)   # float32
    return output
