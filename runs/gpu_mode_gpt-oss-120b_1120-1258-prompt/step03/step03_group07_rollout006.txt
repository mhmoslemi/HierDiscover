7️⃣  Final linear projection back to the original channel dimension (fp16 GEMM)
    and cast to float32.

Only the two LayerNorm passes are custom Triton kernels; all dense linear‑algebra
leverages cuBLAS/Tensor‑Core GEMMs.  Block sizes are tuned to my‑machine
(H100) – a single pass over the feature dimension (BLOCK_D == feature size) is
used, reducing loop overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float*   [total_rows, dim]  (fp32 input)
    y_ptr,            # *mut   half*    [total_rows, dim]  (fp16 output)
    weight_ptr,       # *const float*   [dim]               (γ)
    bias_ptr,         # *const float*   [dim]               (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per program (set == dim)
):
    """Row‑wise LayerNorm that writes fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean and variance (float32 accumulators)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Since BLOCK_D == dim we have only one iteration
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim
    ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
    vals = tl.load(
        ptr,
        mask=mask_m[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)                                          # [BLOCK_M, BLOCK_D]

    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1/sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values (γ·x̂+β) as fp16
    # --------------------------------------------------------------
    norm = (vals - mean[:, None]) * rstd[:, None]             # [BLOCK_M, BLOCK_D]

    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
    b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
    out = norm * w[None, :] + b[None, :]                      # fp32

    out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
    tl.store(
        out_ptr,
        out.to(tl.float16),
        mask=mask_m[:, None] & mask_d[None, :],
    )


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,            # *const half*   [total_rows, dim]  (pre‑norm tensor)
    gate_ptr,         # *const half*   [total_rows, dim]  (output gate)
    y_ptr,            # *mut   half*   [total_rows, dim]  (LN·γ·β·gate)
    weight_ptr,       # *const float*  [dim]              (γ of LN)
    bias_ptr,         # *const float*  [dim]              (β of LN)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm + gate (γ·x̂+β)·gate, writes fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # One iteration because BLOCK_D == dim
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim
    ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
    vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                   other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized *gate* (γ·x̂+β)·gate as fp16
    # --------------------------------------------------------------
    x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0).to(tl.float32)               # pre‑norm
    g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0).to(tl.float32)               # output gate

    norm = (x - mean[:, None]) * rstd[:, None]               # x̂
    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # LN γ
    b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # LN β

    out = (norm * w[None, :] + b[None, :]) * g               # fp32
    out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
    tl.store(out_ptr, out.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise over C) → fp16
    # --------------------------------------------------------------
    B, N, _, C = input_tensor.shape   # C == dim
    total_rows = B * N * N

    # Flatten to 2‑D for the kernel
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    # Kernel launch parameters – process an entire row in one iteration
    BLOCK_M = 1024                     # rows per program (tuneable)
    BLOCK_D = dim                      # whole channel dimension per iteration

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection to obtain left/right + gates
    # --------------------------------------------------------------
    # Concatenate weight matrices for a single GEMM
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16, shape (total_rows, 5*H)

    # Split back into the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # --------------------------------------------------------------
    # 2a️⃣ Optional mask application (pair‑wise mask)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcastable [total_rows, 1]
        mask_fp16 = mask_tensor.to(dtype=torch.float16).view(total_rows, 1)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 2b️⃣ Sigmoid gates + element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # (total_rows, H)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate          # (total_rows, H)
    right_fp16 = right_proj_fp16 * right_gate

    # Restore 4‑D shapes for the triangular multiplication
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)      # used later

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Reshape to (B*H, N, N) matrices and perform batch bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H, N, N) with k as inner dimension
    out_mat = torch.bmm(left_mat, right_mat)                           # fp16, (B*H, N, N)

    # Back to [B, N, N, H] (half‑precision)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # --------------------------------------------------------------
    total_rows_h = B * N * N                      # same as total_rows
    out_flat = out.view(total_rows_h, hidden_dim)         # (total_rows_h, H)
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)

    fused_fp16 = torch.empty_like(out_flat)                # (total_rows_h, H) fp16

    # Same BLOCK_M as before, BLOCK_D == hidden_dim (full row in one iteration)
    BLOCK_M2 = 1024
    BLOCK_D2 = hidden_dim
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)

    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to C (fp16 GEMM) and cast to fp32
    # --------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()   # (C, H)
    final_fp16 = F.linear(fused_fp16, to_out_weight_fp16)  # (total_rows_h, C) fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
