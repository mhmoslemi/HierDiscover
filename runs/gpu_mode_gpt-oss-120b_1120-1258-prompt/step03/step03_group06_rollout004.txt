"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation
(AlphaFold‑3 / Chai / Protenix).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 accumulation → FP16 output) on the input
   tensor of shape [B·N·N, C].

2️⃣ Fuse the five linear projections (left/right + three gates) into a
   single half‑precision GEMM:
      concat_W = [left_proj; right_proj; left_gate; right_gate; out_gate]
   → splits the result into the five tensors.

3️⃣ (Optional) apply the binary mask to the left/right projections
   (before the sigmoid gates).

4️⃣ Apply sigmoid to the three gate tensors and perform element‑wise
   gating on the left/right projections (still FP16).

5️⃣ Triangular multiplicative update:
        out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   This is exactly a batched matrix‑multiply:
      left  → (B·H, N, N)   with axes (i,k)
      right → (B·H, N, N)   with axes (k,j)   (obtained by a permute)
      out   = torch.bmm(left, right)   # (B·H, N, N)

6️⃣ Fused LayerNorm + output‑gate on the “out” tensor (FP16 → FP16).

7️⃣ Final linear projection back to the original channel dimension
   (half‑precision GEMM) and cast the result to float32.

All heavy element‑wise reductions (LayerNorm) are written in Triton,
while dense linear algebra uses Tensor‑Core‑accelerated cuBLAS via
torch.nn.functional.linear / torch.bmm.  The implementation works for
any batch size ≤ 2, sequence length up to 1024, and channel dimensions
128 / 384 as required by the benchmark.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (float32 stats → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float   *[total_rows, dim]
    y_ptr,          # mut   half    *[total_rows, dim]
    weight_ptr,     # const float   *[dim]   (γ)
    bias_ptr,       # const float   *[dim]   (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- first pass: compute mean & variance (float32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # row‑major
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # ----- second pass: normalize, apply γ, β and store fp16 -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]            # still FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused LayerNorm + output‑gate (both in FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # const half   *[total_rows, dim]  (pre‑norm)
    y_ptr,          # mut   half   *[total_rows, dim]  (post‑norm⋅gate)
    weight_ptr,     # const float  *[dim]   (γ)
    bias_ptr,       # const float  *[dim]   (β)
    gate_ptr,       # const half   *[total_rows, dim] (out_gate, already sigmoid)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- compute mean / variance (float32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalize, apply γ, β, multiply by gate -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        # gate values (already sigmoid'ed) – convert to FP32 for multiplication
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)               # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32/bool; may be dummy)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → mask is ignored
                - eps (float, optional) : LayerNorm epsilon (default 1e-5)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                       # flattened rows for LN

    # ------------------------------------------------------------------
    # 1) Input LayerNorm -> FP16
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)               # [R, C] fp16

    # Choose block sizes that work well for dim up to 384
    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (single half‑precision GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five projection/gate weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # (5*H, C)

    # Linear:   [R, C] @ [5*H, C]^T  →  [R, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)          # (R, 5*H), fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional binary mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → flatten to [R, 1] and cast to half
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).half()
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)          # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)        # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)          # (R, H) – used later

    left_fp16  = left_proj_fp16 * left_gate            # (R, H)
    right_fp16 = right_proj_fp16 * right_gate          # (R, H)

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape tensors to (B, N, N, H)
    left  = left_fp16.view(B, N, N, hidden)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)   # (B, N, N, H)

    # Prepare the two operands for torch.bmm:
    #   left  : (B, H, N, N) with axes (i, k)
    #   right : (B, H, N, N) with axes (k, j)  (permute swaps the last two dims)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, i, k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, k, j)

    # Batched matrix multiplication → (B*H, i, j)
    out_mat = torch.bmm(left_mat, right_mat)                         # fp16
    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Second LayerNorm + final output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()            # (R, H) fp16
    out_norm_gate_fp16 = torch.empty_like(out_flat)                 # (R, H) fp16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                        # already sigmoid'ed, fp16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    # Weight shape for the final projection: (dim, hidden)
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)   # (R, dim) fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
