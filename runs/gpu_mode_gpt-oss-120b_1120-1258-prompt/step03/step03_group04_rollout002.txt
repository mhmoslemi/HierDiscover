"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (dtype float32) using a Triton
   kernel that computes mean/variance in fp32 but writes the normalized
   result directly as fp16 (saves a full‑precision copy).

2️⃣ Fuse the five linear projections (left/right & three gates) into a
   single fp16 GEMM: x_norm @ W_fused^T  →  [B·N·N, 5·H].

3️⃣ Split the fused result, apply the optional mask, sigmoid gates and
   element‑wise gating (all in fp16).

4️⃣ Compute the triangular multiplicative update:
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   as a batched fp16 matrix‑multiply (left @ rightᵀ) for each head.
   The required transposes are realized with a single `permute` + `contiguous`,
   producing a contiguous (B·H, N, N) layout that cuBLAS can consume
   efficiently.

5️⃣ Second row‑wise LayerNorm (hidden dimension) – again via the
   same Triton kernel, outputting fp16.

6️⃣ Multiply by the output gate (fp16) and apply the final linear
   projection back to the original channel dimension (fp16 GEMM).

7️⃣ Cast the final tensor back to float32.

Only the two LayerNorms are written in Triton; all dense linear algebra
uses the highly‑optimized fp16 Tensor‑Core GEMMs from cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut     half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const   float*   [dim]               (γ)
    bias_ptr,       # *const   float*   [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,            # int (compile‑time)
    eps: tl.constexpr,            # float (compile‑time)
    BLOCK_M: tl.constexpr,        # rows processed per program
    BLOCK_D: tl.constexpr,        # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a BLOCK_M × BLOCK_D tile of the input (fp32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                         # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)                  # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣  Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [BLOCK_M, BLOCK_D]

        # (x - mean) / sqrt(var+eps)   (fp32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β   (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w[None, :] + b[None, :]             # fp32
        out_fp16 = out_fp32.to(tl.float16)                    # cast to fp16

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(
            y_ptrs,
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float)
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool (opt.)}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape preparation
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                    # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm: fp32 → fp16 (no extra copy)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)          # [total_rows, C] (fp32)
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single fp16 GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, C])
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    w_fused = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                      # shape (5*H, C)

    # Perform the fused linear: [total_rows, C] @ [C, 5*H]^T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, w_fused)   # fp16

    # Split back into the five tensors (all fp16)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating) – broadcasted over H
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [total_rows, 1] → half
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)     # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)   # [total_rows, H]
    out_gate   = torch.sigmoid(out_gate_fp16)     # [total_rows, H]

    left_fp16  = left_proj_fp16 * left_gate       # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate     # [total_rows, H]

    # Reshape to 4‑D tensors
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update via batched GEMM
    #     out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # (B, H, N, N) layout required for a contiguous batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (transpose k‑axis)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]

    # Batched fp16 matrix multiply (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N] (fp16)

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [total_rows_h, H] (fp16)
    out_norm_fp16 = torch.empty_like(out_flat)               # fp16 output buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # element‑wise fp16

    # Linear projection hidden_dim → dim (fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)      # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)      # [total_rows_h, C]

    # Cast to float32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
