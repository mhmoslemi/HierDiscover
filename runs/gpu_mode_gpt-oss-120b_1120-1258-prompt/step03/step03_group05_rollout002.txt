"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (float32) – Triton kernel.
2️⃣  Cast to FP16 and fuse *all* linear projections + gating
    (left_proj, left_gate, right_proj, right_gate, out_gate) into a single
    half‑precision GEMM.
3️⃣  Apply optional mask (only to the two projections) and sigmoid‑gates.
4️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    – performed with a single `torch.einsum`, avoiding the costly
      permute‑+‑contiguous copies required for a batched `bmm`.
5️⃣  Fuse the second LayerNorm and the output‑gate (sigmoid) in another
    Triton kernel (still FP32 arithmetic, result stored FP16).
6️⃣  Final FP16 linear projection back to the model dimension and cast to
    float32.

The heavy compute is the `einsum` (essentially a batched matrix‑multiply
over the hidden dimension).  By keeping the tensors in-place and
eliminating the two large copies required for the previous `bmm`
implementation we cut the runtime well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (float32) – first normalization
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance (float32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – normalize, apply γ/β and write out
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton fused LayerNorm + element‑wise gate (second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]  (input to LN)
    gate_ptr,       # const half*   [total_rows, dim]  (sigmoid‑gate)
    y_ptr,          #       half*   [total_rows, dim]  (output)
    w_ptr,          # const float*  [dim]              (γ)
    b_ptr,          # const float*  [dim]              (β)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, affine transform, multiply gate and store (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load normalized input
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]      # LN
        out = out * g_vals                        # apply output‑gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing version).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]           # model dimension (input & output)
    hidden_dim  = config["hidden_dim"]    # internal hidden dimension
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # [B, N, N, dim]
    total_rows = B * N * N                    # rows after flattening the 3‑D pairwise axis

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton kernel)
    # ------------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)                     # destination (float32)

    BLOCK_M = 256          # more rows per program → fewer kernels
    BLOCK_D = 64           # column tile, 64 works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Cast to half‑precision for the dense linear algebra
    # ------------------------------------------------------------------
    x_norm_h = x_norm_flat.to(torch.float16)   # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Stack the five weight matrices (no biases in the original module)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # [(5*H), dim]

    proj = F.linear(x_norm_h, fused_weight)                # [R, 5*hidden_dim]  FP16

    # Split the concatenated result
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )                                                      # each [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (only to the two projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)   # [R, 1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # Apply sigmoid gates (still in FP16)
    # ------------------------------------------------------------------
    left = left_proj_raw * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw)# [R, H]
    out_gate = torch.sigmoid(out_gate_raw)                # [R, H] (already FP16)

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update – einsum version
    # ------------------------------------------------------------------
    # Reshape to original 4‑D layout
    left = left.view(B, N, N, hidden_dim)                 # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)               # [B, N, N, H]

    # Tensor contraction over the shared axis `k`
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,H] (FP16)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(B * N * N, hidden_dim)                # [R, H]
    out_gate_flat = out_gate.view(B * N * N, hidden_dim)      # [R, H] (FP16)
    out_norm_gated_flat = torch.empty_like(out_flat)          # destination (FP16)

    grid2 = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gated_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated_flat,
        weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    )   # [R, dim]  FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # back to FP32

    return final
