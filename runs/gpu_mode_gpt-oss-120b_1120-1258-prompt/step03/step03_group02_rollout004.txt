"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton + PyTorch fused
-----------------------------------------------------------------------

Algorithm
---------
1️⃣  Row‑wise LayerNorm (fp32 → fp16) implemented in Triton.
2️⃣  Fuse each projection + gate linear layer into a single fp16 GEMM
    (4 GEMMs total, two for the left/right path and one for the output‑gate).
3️⃣  Optional pairwise mask application (broadcast over the hidden dimension).
4️⃣  Apply sigmoid‑gates and element‑wise multiply the gated projections.
5️⃣  Tri‑multiplicative update:  out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d] .
    Implemented with a single `torch.einsum` (no memory‑layout copies).
6️⃣  Second LayerNorm (fp16) – same Triton kernel as step 1.
7️⃣  Apply the output‑gate (sigmoid) and a final fp16 linear projection back
    to the original `dim`.  Cast the result to `float32`.

Only the two LayerNorms are written in Triton; all dense linear algebra
(batched GEMMs, element‑wise ops) uses PyTorch’s highly‑optimised fp16 kernels.
The implementation works for all test configurations (with/without mask,
different `dim`/`hidden_dim`/`seq_len` values) and returns a
`[bs, seq_len, seq_len, dim]` tensor in `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that reads fp32 and writes fp16
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim]  (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    DIM: tl.constexpr,          # static: number of columns (=dim)
    EPS: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    The kernel works in FP32 internally and writes normalized values as
    FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance (two‑pass algorithm)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < DIM
        ptrs = x_ptr + offs_m[:, None] * DIM + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + EPS)                          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM
        ptrs = x_ptr + offs_m[:, None] * DIM + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * DIM + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point – custom Triton‑enabled TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape                     # B, N, N, dim
    total_rows = B * N * N                               # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    # flatten to 2‑D (rows, dim) – must be contiguous for the kernel
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # allocate FP16 buffer for the normalized tensor
    x_norm_h = torch.empty((total_rows, dim),
                           dtype=torch.float16,
                           device=device)

    # launch the Triton kernel (tuned block sizes)
    BLOCK_M = 256          # rows per program (tuned for H100)
    BLOCK_D = 64           # column chunk size (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()      # ensure kernel completion

    # reshape back to [B, N, N, dim] (still fp16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projections + gates (four GEMMs total)
    # ------------------------------------------------------------------
    # concatenate projection & gate weights for the left path
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # concatenate projection & gate weights for the right path
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # flatten for linear layers
    x_norm_h_flat = x_norm_h.view(total_rows, dim)

    left_concat = F.linear(x_norm_h_flat, left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_h_flat, right_concat_w)

    # split projection and gate tensors
    left_proj_h = left_concat[:, :hidden_dim]               # [rows, H]
    left_gate_pre_h = left_concat[:, hidden_dim:]           # [rows, H]

    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # output‑gate (shared for the whole tensor)
    out_gate_h = torch.sigmoid(
        F.linear(
            x_norm_h_flat,
            weights["out_gate.weight"].to(torch.float16, non_blocking=True)
        )
    )                                                       # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # ------------------------------------------------------------------
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h = left_proj_h * left_gate_h          # [rows, H]
    right_h = right_proj_h * right_gate_h

    # reshape back to 4‑D tensors for the tri‑multiplicative step
    left = left_h.view(B, N, N, hidden_dim)     # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched contraction)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # fp16 result

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) – same Triton kernel
    # ------------------------------------------------------------------
    total_rows_out = B * N * N
    out_flat = out.view(total_rows_out, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_out, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_out,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)
    out_norm = out_norm * out_gate                    # gated hidden representation

    final_h = F.linear(
        out_norm.view(total_rows_out, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )                                                 # [rows_out, dim]

    # reshape to original 4‑D layout and cast to float32 as required
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
