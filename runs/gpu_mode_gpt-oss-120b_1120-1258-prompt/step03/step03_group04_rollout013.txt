    final_ptr, # *mut fp32 [total_rows_h, dim]

    w_out_ptr,  # *const fp16 [hidden_dim, dim] (# weight_T)
    total_rows_h: tl.constexpr,
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows_h

        offs_o = o_start*BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        # re-loop across hidden_dim dimension to compute (normed * out_gate) dot weight
        for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
            offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < hidden_dim

            # load out values
            out_vals = tl.load(out_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                               mask=mask_m[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
            # normalize
            out_norm = (out_vals - mean[:, None]) * rstd[:, None]  # [BLOCK_M,BLOCK_D]

            # apply layernorm gamma & beta
            gamma = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)  # shape BLOCK_D
            beta = tl.load(ln_bias_ptr + offs_d, mask=mask_d, other=0.0)
            out_norm = out_norm * gamma[None, :] + beta[None, :]

            # multiply by out_gate
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                                mask=mask_m[:, None] & mask_d[None, :],
                                other=0.0).to(tl.float32)
            gated = out_norm * gate_vals  # shape [BLOCK_M, BLOCK_D]

            # load weight to_out_T slice: shape (BLOCK_D, BLOCK_O)
            w = tl.load(w_out_ptr + offs_d[:, None] * dim + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)

            acc += tl.dot(gated, w)  # [BLOCK_M, BLOCK_O]
        # store final results as fp32
        out_ptrs = final_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_o[None, :])

"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm + fused linear projection (5× hidden_dim) in one Triton
    kernel.  This eliminates the intermediate LayerNorm output buffer and
    fuses the 5 Linear layers (left/right + 3 gates).  The weight matrix is
    transposed beforehand so the kernel can perform a small GEMM (dim × out).
2️⃣  Optional mask is applied to the left/right projections after the kernel.
3️⃣  Triangular multiplicative update is a batched GEMM
    `out = left @ rightᵀ` performed with Torch’s batched‑GEMM (fp16/Tensor‑Cores).
4️⃣  Row‑wise LayerNorm, out‑gate, and final linear projection are fused in a
    second Triton kernel.  The kernel reads the GEMM result, normalises over
    the hidden dimension, multiplies by the out‑gate and immediately projects
    back to the model dimension.  This avoids an extra out‑norm buffer.

All heavy dense work stays in fp16 (Tensor‑Core friendly) while the two
LayerNorm reductions use fp32 accumulation for numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm + fused Linear (5·hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_linear_kernel(
    # inputs
    x_ptr,               # *const   float32   [total_rows, dim]
    proj_ptr,            # *mut     float16   [total_rows, out_dim]  (out_dim = 5*hidden_dim)
    # LayerNorm parameters
    ln_weight_ptr,       # *const   float32   [dim]
    ln_bias_ptr,         # *const   float32   [dim]
    # Transposed weight matrix for the fused projection
    weight_T_ptr,        # *const   float16   [dim, out_dim] (already transposed)
    # runtime args
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    out_dim: tl.constexpr,
    eps: tl.constexpr,
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)                       # [BLOCK_M]
        sum_sq  += tl.sum(x * x, axis=1)                   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five Linear projections
    # --------------------------------------------------------------
    # Loop over output columns in chunks of BLOCK_O
    for o_start in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o_start * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < out_dim

        # accumulator for the block [BLOCK_M, BLOCK_O]
        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        # iterate over the input dim
        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim

            # normalized input block
            x = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)
            x_norm = (x - mean[:, None]) * rstd[:, None]       # [BLOCK_M, BLOCK_D]

            # weight block (already transposed, shape [BLOCK_D, BLOCK_O])
            w = tl.load(
                weight_T_ptr + offs_d[:, None] * out_dim + offs_o[None, :],
                mask=mask_d[:, None] & mask_o[None, :],
                other=0.0,
            ).to(tl.float32)

            # acc += x_norm @ w   ( (M,D) @ (D,O) -> (M,O) )
            acc += tl.dot(x_norm, w)

        # write the block
        out_ptrs = proj_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_m[:, None] & mask_o[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Row‑wise LayerNorm + out‑gate + final Linear (hidden_dim → dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_outgate_linear_kernel(
    # inputs
    hidden_ptr,          # *const   float16   [total_rows_h, hidden_dim]
    out_gate_ptr,        # *const   float16   [total_rows_h, hidden_dim]
    # LayerNorm parameters for hidden → hidden
    ln_weight_ptr,       # *const   float32   [hidden_dim]
    ln_bias_ptr,         # *const   float32   [hidden_dim]
    # Transposed weight for final projection (hidden_dim, dim)
    weight_T_ptr,        # *const   float16   [hidden_dim, dim]
    # output
    out_ptr,             # *mut     float32   [total_rows_h, dim]
    # runtime args
    total_rows_h: tl.constexpr,
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows_h

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance over hidden_dim (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        h = tl.load(
            hidden_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(h, axis=1)
        sum_sq  += tl.sum(h * h, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Apply LayerNorm, out‑gate and final projection (hidden → dim)
    # --------------------------------------------------------------
    for o_start in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o_start * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < dim

        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < hidden_dim

            # hidden activation
            h = tl.load(
                hidden_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # normalize
            h_norm = (h - mean[:, None]) * rstd[:, None]      # [BLOCK_M, BLOCK_D]

            # γ & β for hidden‑dim LayerNorm
            gamma = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)
            beta  = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)
            h_norm = h_norm * gamma[None, :] + beta[None, :]

            # out‑gate (fp16 → fp32)
            g = tl.load(
                out_gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            gated = h_norm * g                                 # [BLOCK_M, BLOCK_D]

            # weight block for final projection (hidden_dim × dim) – transposed, shape [BLOCK_D, BLOCK_O]
            w = tl.load(
                weight_T_ptr + offs_d[:, None] * dim + offs_o[None, :],
                mask=mask_d[:, None] & mask_o[None, :],
                other=0.0,
            ).to(tl.float32)

            # accumulation
            acc += tl.dot(gated, w)                           # (M,D)·(D,O) → (M,O)

        # write fp32 result
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(
            out_ptrs,
            acc,
            mask=mask_m[:, None] & mask_o[None, :],
        )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask_tensor   : torch.Tensor[B, N, N] (bool/float) – may be omitted (nomask=True)
        - weights_dict  : dict of all model weight tensors (float32)
        - config_dict   : {"dim": C, "hidden_dim": H, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor[B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for flattened [row, C]
    total_rows_h = total_rows                   # after GEMM we still have B·N·N rows

    # --------------------------------------------------------------
    # 1️⃣  Fused LayerNorm + 5×Linear (produces left/right + gates)
    # --------------------------------------------------------------
    x = input_tensor.contiguous()            # [B,N,N,C] – ensures contiguous layout
    x_flat = x.view(total_rows, dim)         # [total_rows, C]

    out_dim = hidden_dim * 5                  # 5 × hidden_dim

    # concatenate the five projection weights and transpose for GEMM
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    w_concat = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).contiguous().half()                     # (5*H, C)

    # Transpose so we can treat it as (C, 5*H) inside the kernel
    w_T = w_concat.t().contiguous()           # (C, 5*H)  fp16

    # output buffer for the 5 projections
    proj_all = torch.empty(total_rows, out_dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    BLOCK_O = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_linear_kernel[grid](
        x_ptr=x_flat,
        proj_ptr=proj_all,
        ln_weight_ptr=weights["norm.weight"],
        ln_bias_ptr=weights["norm.bias"],
        weight_T_ptr=w_T,
        total_rows=total_rows,
        dim=dim,
        out_dim=out_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )

    # split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [total_rows, H] in fp16

    # --------------------------------------------------------------
    # 2️⃣  Optional mask on left/right (applied after projection)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 3️⃣  Apply sigmoid gates (still fp16) and reshape to 4‑D tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = (left_proj * left_gate).view(B, N, N, hidden_dim)    # [B,N,N,H] fp16
    right = (right_proj * right_gate).view(B, N, N, hidden_dim)  # [B,N,N,H] fp16
    out_gate = out_gate.view(B, N, N, hidden_dim)             # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 4️⃣  Triangular multiplicative update: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k‑axis transposed)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)              # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)           # [B*H, N, N] fp16
    del left, right, left_mat, right_mat               # free intermediate memory

    # reshape back to (total_rows_h, hidden_dim) for the second kernel
    out_mat = out_mat.view(total_rows_h, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim).contiguous()

    # --------------------------------------------------------------
    # 5️⃣  Fused LayerNorm (hidden) + out‑gate + final Linear
    # --------------------------------------------------------------
    # final weight (hidden_dim → dim), transposed for the kernel
    w_out_T = weights["to_out.weight"].t().contiguous().half()  # (hidden_dim, dim)

    final = torch.empty(total_rows_h, dim, dtype=torch.float32, device=device)

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    BLOCK_O2 = 64

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)
    fused_ln_outgate_linear_kernel[grid2](
        hidden_ptr=out_mat,
        out_gate_ptr=out_gate_flat,
        ln_weight_ptr=weights["to_out_norm.weight"],
        ln_bias_ptr=weights["to_out_norm.bias"],
        weight_T_ptr=w_out_T,
        out_ptr=final,
        total_rows_h=total_rows_h,
        hidden_dim=hidden_dim,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        BLOCK_O=BLOCK_O2,
    )

    # reshape to original 4‑D layout
    final = final.view(B, N, N, dim)   # [B, N, N, C]
    return final
```
