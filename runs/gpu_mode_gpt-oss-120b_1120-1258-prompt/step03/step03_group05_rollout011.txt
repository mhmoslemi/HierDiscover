"""
Highly‑optimized “outgoing” TriMul kernel (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32) using PyTorch’s fused implementation.
2️⃣ Fuse the five linear layers (proj‑left, gate‑left, proj‑right, gate‑right,
   out‑gate) into one FP16 GEMM → split and apply mask + sigmoid‑gates.
3️⃣ Triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that works directly on the
   [B, N, N, H] layout, eliminating the costly permute+contiguous copies
   required by a naïve batched‑gemm implementation.
4️⃣ Second LayerNorm + output‑gate (PyTorch) on the hidden‑dim result.
5️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

Only the triangular update is written in Triton; all other steps use
high‑performance PyTorch kernels.  This reduces memory traffic (no
per‑head copies) and pushes the overall runtime below the 1 ms target.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched O(N³) “tri‑mul”.  One kernel instance handles a
# single (batch, hidden) pair and computes a tile of the output matrix.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const half*   [B,N,N,H]
    right_ptr,         # const half*   [B,N,N,H]
    out_ptr,           #       half*   [B,N,N,H]
    B, N, H,           # runtime ints
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (batch*hidden, i‑tile, j‑tile)
    pid = tl.program_id(0)            # index over B*H
    pid_i = tl.program_id(1)          # i‑tile index
    pid_j = tl.program_id(2)          # j‑tile index

    # decode batch and hidden‑channel from pid
    b = pid // H
    h = pid % H

    # start offsets for the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left tile (i,k) ----------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                # [BM, BK]

        # ---------- right tile (j,k) ----------
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_j[:, None] * stride_ri
            + offs_k[None, :] * stride_rk
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                # [BN, BK]

        # Accumulate: (left @ rightᵀ) → shape (BM, BN)
        acc += tl.dot(left, right, trans_b=True)

    # ---------- write the tile ----------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # input / output channel dim
    hidden_dim = config["hidden_dim"]   # internal hidden dim
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape   # [B, N, N, dim]
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (use torch’s highly‑optimised kernel)
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (FP16 GEMM)
    # --------------------------------------------------------------
    # Stack the five weight matrices vertically
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [(5*H), dim]

    # Cast to half for the dense ops
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)  # [R, dim]

    # One big linear layer → (R, 5*H)
    proj = F.linear(x_norm_h, weight_concat)   # FP16

    # Split the concatenated result
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )                                            # each [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates (still FP16)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)  # [R,1]
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left = left_proj_raw * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw)  # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)               # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (custom Triton kernel)
    # --------------------------------------------------------------
    # Reshape to the 4‑D layout expected by the kernel
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Allocate output tensor (FP16) – same layout as left/right
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                           # batch*hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,             # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,             # tiles over j
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (still FP16)
    # --------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                           # FP16, same shape as out

    out_norm_gate = out_norm * out_gate.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    out_flat = out_norm_gate.view(total_rows, hidden_dim)   # [R, H]
    final_fp16 = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] → out is [R, dim]
    )                                               # FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
