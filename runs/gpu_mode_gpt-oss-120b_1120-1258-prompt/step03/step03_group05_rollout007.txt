"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  First LayerNorm over the input channels (dim) – we use the
    built‑in torch.nn.functional.layer_norm (fast cuDNN kernel).

2️⃣  Fuse the five linear projections (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a **single** FP16 GEMM:
        Y = X_norm @ W_fused.T
    where W_fused stacks the five weight matrices.
    Afterwards split Y, apply the optional mask and the sigmoid‑gates.

3️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched GEMM (torch.bmm) with batch dimension
    B·hidden_dim.

4️⃣  Fuse the *second* LayerNorm, the output‑gate and the final
    linear projection back to `dim` into ONE Triton kernel.
    For each row `r` (r ∈ [0, B·N·N)):
        • compute mean/var over the hidden dimension (float32)
        • normalize, apply affine (γ/β), multiply with the pre‑computed
          out‑gate (sigmoid)
        • accumulate the dot‑product with the transposed output weight
          matrix (to_out.weight.T) → final tensor of shape (dim,).

    This eliminates a second LayerNorm kernel **and** a costly
    (R × hidden_dim × dim) GEMM, cutting memory traffic dramatically.

5️⃣  Cast the FP16 result back to float32 and return.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: fused LayerNorm + output‑gate + final projection

@triton.jit
def ln_gate_proj_kernel(
    x_ptr,          # const half*  [total_rows, hidden_dim]   (tri‑update output)
    gate_ptr,       # const half*  [total_rows, hidden_dim]   (sigmoid‑out_gate)
    out_ptr,        #       half*  [total_rows, out_dim]      (final result)
    gamma_ptr,      # const float* [hidden_dim]              (to_out_norm.weight)
    beta_ptr,       # const float* [hidden_dim]              (to_out_norm.bias)
    w_out_ptr,      # const half*  [hidden_dim, out_dim]     (to_out.weight.T)
    total_rows: tl.constexpr,               # int
    hidden_dim: tl.constexpr,               # int (H)
    out_dim: tl.constexpr,                  # int (C)
    eps: tl.constexpr,                      # float
    BLOCK_M: tl.constexpr,                  # rows per program
    BLOCK_K: tl.constexpr,                  # hidden‑dim tile
    BLOCK_N: tl.constexpr,                  # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ mean / variance over the hidden dimension (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        x = tl.load(
            x_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalization, affine, gate and accumulate projection
    # --------------------------------------------------------------
    out_acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    out_final_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=input_tensor.device)

    # Triton launch configuration
    BLOCK_M = 128                # rows per program
    BLOCK_K = 32                 # hidden‑dim tile
    BLOCK_N = 64                 # output‑dim tile (covers dim up to 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    ln_gate_proj_kernel[grid](
        out_flat,                # x_ptr
        out_gate_flat,           # gate_ptr
        out_final_fp16,          # out_ptr
        weights["to_out_norm.weight"],   # γ   (float32)
        weights["to_out_norm.bias"],     # β   (float32)
        w_out_T,                 # transposed output weight (half)
        total_rows,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )
    torch.cuda.synchronize()

    # ------------------- 5️⃣ reshape & cast back -------------------
    out_final = out_final_fp16.view(B, N, N, dim).to(torch.float32)  # float32 as required
    return out_final
