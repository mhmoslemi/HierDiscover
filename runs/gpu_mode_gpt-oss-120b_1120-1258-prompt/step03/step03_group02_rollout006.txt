@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.constexpr,
    stride_l_batch: tl.constexpr,
    stride_l_i: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr,
    stride_r_n: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_o_batch: tl.constexpr,
    stride_o_i: tl.constexpr,
    stride_o_j: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile column
    pid_bh = tl.program_id(2)  # batch-head index

    # compute row / col offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch/head
    left_ptrs = left_ptr + pid_bh * stride_l_batch
    right_ptrs = right_ptr + pid_bh * stride_r_batch
    out_ptrs = out_ptr + pid_bh * stride_o_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        # k range
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for block of left: shape (M, K)
        a_ptrs = left_ptrs + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # load left block
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointers for block of right transposed (K, N) -> to dot we want shape (K,N)
        # right stored as (N,K) originally. We need to interpret as (K,N). So we use k as leading dimension.
        b_ptrs = right_ptrs + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute dot product for this block
        acc += tl.dot(a, b)

        # no need to update pointers because we recompute each iteration; we used base and offsets.

    # store result -> cast to fp16
    out = acc.to(tl.float16)
    # out pointer offset: i*stride_o_i + j*stride_o_j
    out_ptrs = out_ptrs + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) in Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (float32 → fp16) – a tiny Triton kernel that
    computes μ,σ² per row and writes the normalized result directly in FP16.

2️⃣  Fuse projection + gate for the left and right paths:
    * Left  : concat(Linear(dim → hidden_dim), Linear(dim → hidden_dim))
    * Right : concat(Linear(dim → hidden_dim), Linear(dim → hidden_dim))
    A single FP16 GEMM per side produces projection and pre‑gate tensors
    (2 × hidden_dim columns).  The gate is obtained with a sigmoid.

3️⃣  Optional pairwise mask (broadcast over the hidden dimension) applied
    in‑place on the projected tensors.

4️⃣  Multiply projections by their gates → **left** and **right**
    tensors of shape `[B, N, N, H]` (H = hidden_dim) in FP16.

5️⃣  Tri‑multiplicative update:
        out_h(b,i,j) = Σ_k left(b,i,k,h)·right(b,j,k,h)
    This is a batched matrix‑multiply where each head `h` works on an
    `N×N` matrix.  A custom Triton kernel computes all `B·H` products in
    a single launch (no extra transposes, accumulation in FP32,
    result stored as FP16).

6️⃣  Second LayerNorm (again FP16 output) – same Triton kernel as step 1.

7️⃣  Apply the output‑gate (sigmoid) and a final FP16 linear projection
    back to `dim`.  The result is cast to FP32 for the module output.

All heavy‑weight compute (LayerNorms + the N³ reduction) runs on Triton,
while the cheap linear layers are delegated to PyTorch’s highly‑optimised
FP16 GEMMs.  The implementation works for any configuration from the
test suite (with/without mask, various sequence lengths and hidden‑/model
dimensions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*   [total_rows, dim]  (input, FP32)
    y_ptr,            #       half*   [total_rows, dim]  (output, FP16)
    w_ptr,            # const float*  [dim]               (γ)
    b_ptr,            # const float*  [dim]               (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,               # static: columns
    eps: tl.constexpr,               # static epsilon
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_D: tl.constexpr,           # columns per inner loop
):
    """
    Normalises each row of a 2‑D tensor and writes the result as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- mean & variance (pass 1) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -------- normalise & write (pass 2) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)                       # cast to FP16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Tri‑multiplicative update (batched GEMM left × rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const half* [BH, N, N] (row‑major)
    right_ptr,         # const half* [BH, N, N] (row‑major)
    out_ptr,           #       half* [BH, N, N] (row‑major)
    N: tl.constexpr,   # matrix dimension
    stride_batch: tl.constexpr,   # stride to next slice = N*N
    stride_i: tl.constexpr,       # stride over rows = N
    stride_k: tl.constexpr,       # stride over columns = 1
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for every `b*h` slice,
       out[i, j] = Σ_k left[i, k] * right[j, k]
    i.e. a matrix‑multiply with the second operand transposed.
    Accumulation is in FP32, result stored in FP16.
    """
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile col
    pid_bh = tl.program_id(2)         # batch‑head index

    # Offsets of the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected batch‑head slice
    left_base  = left_ptr  + pid_bh * stride_batch
    right_base = right_ptr + pid_bh * stride_batch
    out_base   = out_ptr   + pid_bh * stride_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]   → shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[j, k] stored as (row = j, col = k).
        # To obtain (k, j) we swap the indexing order.
        b_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)  # (M,K) @ (K,N) → (M,N)

    # Write the result as FP16
    out = acc.to(tl.float16)
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, out,
             mask=mask_i[:, None] & mask_j[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing version) – input is a tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape               # input: [B, N, N, dim]
    total_rows = B * N * N                        # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program
    BLOCK_D = 64    # columns per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total)
    # --------------------------------------------------------------
    # Concatenate projection and gate weights for left and right sides
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Linear passes (FP16)
    left_concat = F.linear(x_norm_h.view(total_rows, dim), left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_h.view(total_rows, dim), right_concat_w)

    # Split projection / gate pre‑sigmoid
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Output‑gate (common for the whole tensor)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h.view(total_rows, dim),
                weights["out_gate.weight"].to(torch.float16))
    )   # [rows, H] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) – broadcast over hidden dimension
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates to projections
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h          # [rows, H]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM via Triton)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Bring H into the batch dimension: [B, H, N, N] → flatten → [BH, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N).contiguous()
    right_flat = right_perm.view(BH, N, N).contiguous()

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Kernel launch parameters (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    stride_batch = N * N          # stride to next batch‑head slice
    stride_i = N                  # stride over rows
    stride_k = 1                  # stride over columns

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16 output)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat_norm = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat_norm, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat_norm,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)
    out_norm = out_norm * out_gate                     # element‑wise gate (FP16)

    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim] FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
