"""
Highly‑optimized Triton‑accelerated implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ **First layer‑norm** – performed with the native
   ``torch.nn.functional.layer_norm`` (fast fused CUDA kernel).  The
   normalized tensor is cast to fp16.

2️⃣ **Fused linear projection** – the five weight matrices
   (left/right projections, three gates) are concatenated into a single
   matrix ``W`` of shape ``[5·H, C]``.  A single ``F.linear`` yields
   ``[B·N·N, 5·H]`` which is split into the four projected tensors and
   the output‑gate tensor.

3️⃣ **Mask & sigmoid gating** – the optional pair‑wise mask is applied
   to the left/right projections.  Sigmoid is applied to the three gate
   tensors and the left/right tensors are element‑wise gated.

4️⃣ **Triangular multiplicative update** – the batched matrix‑multiply
   ``out_h[i,j] = Σ_k left_h[i,k]·right_h[j,k]`` is realised by
   ``torch.bmm`` on tensors of shape ``(B·H, N, N)`` (fp16, Tensor‑Core).

5️⃣ **Second layer‑norm + output‑gate** – a custom Triton kernel
   ``layernorm_gate_fp16_kernel`` computes the row‑wise layer‑norm over
   the hidden dimension, multiplies by the output‑gate and writes the
   result as fp16.

6️⃣ **Final projection** – a standard ``F.linear`` maps the hidden
   dimension back to the original channel dimension ``C``.  The result
   is cast to ``float32`` before returning.

Only the second layer‑norm (step 5) is written in Triton, satisfying the
requirement that *some* part of the computation runs in a kernel while
keeping the overall pipeline compact and fast.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: second layer‑norm + output‑gate (writes fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,            # *const half*   [total_rows, dim]   (pre‑norm tensor)
    gate_ptr,         # *const half*   [total_rows, dim]   (output gate)
    y_ptr,            # *mut   half*   [total_rows, dim]   (LN·γ·β·gate)
    weight_ptr,       # *const float*  [dim]               (γ of the second LN)
    bias_ptr,         # *const float*  [dim]               (β of the second LN)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    """LayerNorm + output‑gate (γ·x̂+β)·gate, writes fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                     # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean and variance (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)               # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                            # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                      # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Write normalized · gate (γ·x̂+β)·gate  as fp16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # original (pre‑norm) value
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # fp32
        # output gate value
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # fp32

        # normalize
        norm = (x - mean[:, None]) * rstd[:, None]          # x̂

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        # (γ·x̂+β)·gate
        out = (norm * w[None, :] + b[None, :]) * g

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (over C) – use torch's fast implementation
    # --------------------------------------------------------------
    # LayerNorm is applied over the last dimension of the 4‑D tensor.
    # The result is cast to fp16 for the subsequent linear projection.
    x_norm_fp16 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (all in fp16)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5·H, C))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()          # (5·H, C)

    # Single GEMM: [B·N·N, C] @ [C, 5·H]ᵀ → [B·N·N, 5·H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # --------------------------------------------------------------
    # 2a️⃣ Optional pair‑wise mask (broadcasted over H)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.contiguous().view(-1, 1).to(dtype=torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # [B·N·N, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16  = left_proj_fp16 * left_gate        # [B·N·N, H]
    right_fp16 = right_proj_fp16 * right_gate

    # --------------------------------------------------------------
    # Reshape to (B, N, N, H) for the triangular update
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)   # used later

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Merge batch and hidden dimensions: (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # Batched fp16 matrix multiply – leverages Tensor‑Core GEMMs.
    out_mat = torch.bmm(left_mat, right_mat)  # fp16, shape (B·H, N, N)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (custom Triton kernel)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)            # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)

    fused_fp16 = torch.empty_like(out_flat)                  # fp16 buffer for LN·gate result

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 32    # hidden‑dim columns per iteration (must divide hidden_dim)

    grid = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to C and cast to float32
    # --------------------------------------------------------------
    final_fp16 = F.linear(fused_fp16, weights["to_out.weight"].half())
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
