Implementation skeleton:

```
@triton.jit
def fused_norm_proj_kernel(
    inp_ptr,       # *const float* [total_rows, dim]  (FP32)
    out_ptr,       # *mut half* [total_rows, out_feat] (FP16)
    weight_ptr,    # *const half* [dim, out_feat] (FP16) transposed weight 
    gamma_ptr,     # *const float* [dim] maybe not needed? Actually proj is linear, no gamma.
    # we don't need gamma for linear conv.
    total_rows, dim, out_feat: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # compute projection per output block
    num_out_blocks = tl.cdiv(out_feat, BLOCK_O)
    # allocate accumulator memory for each out block
    # We allocate local buffers for each block within loop; we could allocate a fixed small array at a time.
    for ob in range(num_out_blocks):
        # compute offsets for out
        offs_o = ob * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_feat

        # accumulator for this out block
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        # loop over dim tiles
        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load input tile
            in_ptrs = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
            in_vals_fp32 = tl.load(in_ptrs,
                                   mask=mask_m[:, None] & mask_d[None, :],
                                   other=0.0).to(tl.float32)

            # normalize
            norm_tile = (in_vals_fp32 - mean[:, None]) * rstd[:, None]

            # load weight tile
            w_ptrs = weight_ptr + offs_d[:, None] * out_feat + offs_o[None, :]
            w = tl.load(w_ptrs,
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)

            # accumulate: dot product
            acc += tl.dot(norm_tile, w)  # shape [BLOCK_M, BLOCK_O]

        # store accumulator
        out_ptrs = out_ptr + offs_m[:, None] * out_feat + offs_o[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])
```

@triton.jit
def fused_layernorm_gate_linear_kernel(
    x_ptr,      # const half* [R, hidden] (pre-norm)
    gate_ptr,   # const half* [R, hidden] (out_gate)
    y_ptr,      # *mut half* [R, dim] (output)
    weight_norm_ptr,   # const float* [hidden] (LN gamma)
    bias_norm_ptr,     # const float* [hidden] (LN beta)
    weight_out_ptr,    # const half* [hidden, dim] (final projection)
    total_rows,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and var across hidden dimension
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Number of output blocks
    num_out_blocks = tl.cdiv(dim, BLOCK_D)
    # Allocate accumulator for each out block
    for ob in range(num_out_blocks):
        offs_d = ob * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        # Loop over hidden tiles to compute matmul
        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden
            # Load x tile (pre-norm)
            x_ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # Normalize
            norm = (x_vals - mean[:, None]) * rstd[:, None]

            # Apply LN weight & bias
            gamma = tl.load(weight_norm_ptr + offs_h, mask=mask_h, other=0.0)
            beta = tl.load(bias_norm_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]   # shape [M, H_tile]

            # Load gate values
            gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
            # Apply gate
            gated = norm * gate_vals

            # Load final weight tile (hidden_tile x dim_tile)
            w_ptrs = weight_out_ptr + offs_h[:, None] * dim + offs_d[None, :]
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # Accumulate
            acc += tl.dot(gated, w)  # shape [M, D_tile]

        # Store output block
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])
```

"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. **Fused LayerNorm + linear projection**  
   - One Triton kernel computes the row‑wise LayerNorm (FP32 accumulation,
     FP16 output) and simultaneously multiplies the normalized vectors with a
     concatenated weight matrix (dim → 5·hidden).  This replaces a separate
     LayerNorm + five `torch.nn.functional.linear` calls, halving memory traffic.

2. **Mask & gating**  
   - Optional binary mask is applied only to the left/right projections.  
   - Sigmoid is computed with PyTorch (tiny cost) and the gated values are
     produced by an element‑wise multiply.

3. **Triangular multiplicative update**  
   - The core operation `out[i,j] = Σ_k left[i,k] * right[j,k]` is performed by
     a batched half‑precision matrix multiplication (`torch.bmm`).  This uses
     Tensor‑Core kernels on the H100.

4. **Fused final LayerNorm + output‑gate + linear projection**  
   - A second Triton kernel normalises the per‑row hidden vectors, applies the
     learned LayerNorm scale/bias, multiplies by the pre‑computed `out_gate`,
     and finally projects the result to the original channel dimension
     (`hidden → dim`) in a single GEMM.  The kernel writes the final tensor in
     FP16; the host casts it to FP32 for the return value.

Both kernels operate on contiguous 2‑D views of the tensors, use only FP16
stores (except for the LayerNorm accumulators) and keep the hidden‑dimension
work‑loads small enough to stay within register limits.  The implementation
meets the requirement of having at least one custom Triton kernel while
significantly reducing memory passes compared with the reference PyTorch
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fused LayerNorm + projection (input → left/right/gates)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_kernel(
    inp_ptr,          # *const float*   [total_rows, dim]   (FP32 input)
    out_ptr,          # *mut   half*    [total_rows, out_feat] (FP16 output)
    weight_ptr,       # *const half*    [dim, out_feat]   (FP16, transposed weight)
    total_rows,       # int
    dim: tl.constexpr,
    out_feat: tl.constexpr,   # 5 * hidden_dim
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # input‑dim tile
    BLOCK_O: tl.constexpr,    # output‑feat tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # First pass: compute mean & variance (float32 reduction)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Second pass: normalise and multiply with weight matrix
    # --------------------------------------------------------------
    num_out_blocks = tl.cdiv(out_feat, BLOCK_O)

    for ob in range(num_out_blocks):
        offs_o = ob * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < out_feat

        # accumulator for this output block (FP32)
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # input tile again (needed for normalisation)
            inp_tile_ptr = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
            inp_tile = tl.load(inp_tile_ptr,
                               mask=mask_m[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)

            # normalised input
            norm_tile = (inp_tile - mean[:, None]) * rstd[:, None]   # [M, D_tile]

            # corresponding weight tile
            w_ptr = weight_ptr + offs_d[:, None] * out_feat + offs_o[None, :]
            w_tile = tl.load(w_ptr,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0).to(tl.float32)                # [D_tile, O_tile]

            acc += tl.dot(norm_tile, w_tile)                         # [M, O_tile]

        # write block result
        out_ptrs = out_ptr + offs_m[:, None] * out_feat + offs_o[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 2) Fused final LayerNorm + out_gate * linear projection (hidden → dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_linear_kernel(
    x_ptr,            # *const half*   [total_rows, hidden]      (pre‑norm)
    gate_ptr,         # *const half*   [total_rows, hidden]      (sigmoid(out_gate))
    out_ptr,          # *mut   half*   [total_rows, dim]        (final FP16)
    ln_weight_ptr,    # *const float*  [hidden]                 (γ of to_out_norm)
    ln_bias_ptr,      # *const float*  [hidden]                 (β of to_out_norm)
    out_weight_ptr,   # *const half*   [hidden, dim]            (to_out.weight.T)
    total_rows,       # int
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_H: tl.constexpr,    # hidden‑dim tile
    BLOCK_D: tl.constexpr,    # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Mean + variance over hidden dimension (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # MatMul: (normed * gate)  @  out_weight   →  final output
    # --------------------------------------------------------------
    num_out_blocks = tl.cdiv(dim, BLOCK_D)

    for ob in range(num_out_blocks):
        offs_d = ob * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # accumulator for this output block
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # load pre‑norm hidden slice
            x_tile_ptr = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
            x_tile = tl.load(x_tile_ptr,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)         # [M, H_tile]

            # normalise
            norm_tile = (x_tile - mean[:, None]) * rstd[:, None]  # [M, H_tile]

            # apply to_out_norm γ / β
            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
            beta  = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)
            norm_tile = norm_tile * gamma[None, :] + beta[None, :]  # [M, H_tile]

            # load gate slice and apply
            gate_tile_ptr = gate_ptr + offs_m[:, None] * hidden + offs_h[None, :]
            gate_tile = tl.load(gate_tile_ptr,
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)            # [M, H_tile]
            gated_tile = norm_tile * gate_tile                       # [M, H_tile]

            # load corresponding slice of final weight (hidden_tile × dim_tile)
            w_ptr = out_weight_ptr + offs_h[:, None] * dim + offs_d[None, :]
            w_tile = tl.load(w_ptr,
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)               # [H_tile, D_tile]

            # accumulate matmul result
            acc += tl.dot(gated_tile, w_tile)                       # [M, D_tile]

        # write final block
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry‑point: forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32/bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → no mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows = B·N·N

    # ------------------------------------------------------------------
    # 1) Fused LayerNorm + Projection (produces left/right + gates)
    # ------------------------------------------------------------------
    # Concatenate the five projection/gate weight matrices (shape (5*H, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                  # (5*H, dim)

    # Store as (dim, 5*H) for convenient row‑wise access inside Triton
    proj_weight_T = concat_w.t().contiguous()   # (dim, 5*H)

    out_feat = 5 * hidden                         # = size of concatenated output

    inp_flat = inp.view(total_rows, dim).contiguous()  # [R, C] float32
    proj_out = torch.empty(total_rows, out_feat,
                           dtype=torch.half,
                           device=device)             # [R, 5H] FP16

    # kernel launch configuration
    BLOCK_M = 64          # rows per program
    BLOCK_D = 32          # tile over C (input dim)
    BLOCK_O = 64          # tile over output features (5*H)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_norm_proj_kernel[grid](
        inp_flat,                # input FP32
        proj_out,                # output FP16
        proj_weight_T,           # weight (dim, 5*H) FP16
        total_rows,
        dim,
        out_feat,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()

    # split to the five individual tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_out, hidden, dim=1
    )  # each (R, H) FP16

    # ------------------------------------------------------------------
    # 2) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).half()
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 3) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # FP16
    right_gate = torch.sigmoid(right_gate_fp16)    # FP16
    out_gate = torch.sigmoid(out_gate_fp16)        # FP16 (used later)

    left_fp16 = left_proj_fp16 * left_gate          # (R, H)
    right_fp16 = right_proj_fp16 * right_gate       # (R, H)

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)          # (B,N,N,H)
    right = right_fp16.view(B, N, N, hidden)        # (B,N,N,H)

    # reshape to (B*H, N, N) and perform batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)       # (B*H, N, N)  FP16
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 5) Fused final LayerNorm + out_gate + linear projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()  # (R, H) FP16
    final_fp16 = torch.empty(total_rows, dim,
                             dtype=torch.half,
                             device=device)               # (R, dim) FP16

    # final linear weight: (dim, hidden) → we need (hidden, dim) for kernel
    to_out_weight_T = weights["to_out.weight"].half().t().contiguous()  # (H, dim)

    BLOCK_H = 32   # tile over hidden dimension
    BLOCK_D = 32   # tile over output dim
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_linear_kernel[grid2](
        out_flat,                     # pre‑norm hidden (FP16)
        out_gate,                     # sigmoid‑ed out_gate (FP16)
        final_fp16,                   # result buffer (FP16)
        weights["to_out_norm.weight"],# LN γ  (FP32, size H)
        weights["to_out_norm.bias"],  # LN β  (FP32, size H)
        to_out_weight_T,              # final projection weight (H, dim) FP16
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Cast back to FP32 and reshape to original layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
