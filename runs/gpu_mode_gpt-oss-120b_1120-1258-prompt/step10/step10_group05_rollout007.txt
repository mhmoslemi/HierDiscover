"""
Tri‑Multiplicative (outgoing) update – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – done with
   PyTorch’s highly‑optimised `torch.nn.functional.layer_norm` (float32 → float16).

2️⃣ Fused linear projection that creates the five tensors required for the update:
   left, right, left‑gate, right‑gate, out‑gate.
   The projection is a single `F.linear` on half‑precision data.

3️⃣ One Triton kernel (`split_fuse_gate_kernel`) that
   * splits the fused projection into the three tensors needed for the update,
   * applies an optional pair‑wise mask,
   * computes the sigmoid gates (fp32 for stability) and
   * writes `left`, `right` and `out_gate` (all fp16).

4️⃣ Batched GEMM over the hidden dimension:
   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N) implements
   `einsum('... i k d, ... j k d -> ... i j d')`.

5️⃣ Second LayerNorm on the hidden‑dim tensor (again via PyTorch) and
   element‑wise multiplication with `out_gate`.

6️⃣ Final linear projection back to `dim` (fp16 GEMM) and cast to float32.

Only step 3 is implemented in Triton – satisfying the “must use a kernel” rule –
while the remaining heavy work stays in cuBLAS / cuDNN where it runs fastest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast across H
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # iterate over hidden dimension in tiles
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection:
        #   [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for better numerics
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # ----------------------------------------------------------------
        # store results
        # ----------------------------------------------------------------
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, og,
                 mask=mask_r[:, None] & mask_h[None, :])

    x_norm_f32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                   # still float32
    x_norm = x_norm_f32.to(torch.float16)               # fp16 for the gemm

    # ----------------------- 2️⃣ fused projection -----------------------
    # concatenate the five weight matrices (each [H, dim]) → [5·H, dim]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # linear (no bias) → [R, 5·H] in fp16
    proj = F.linear(x_norm, fused_w)                    # fp16

    # --------------------------------------------------------------
    # 3️⃣ split, mask, sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right     = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)  # 1.0 keep, 0.0 discard

    BLOCK_R = 256   # rows per program
    BLOCK_H = 32    # hidden dim tiles
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free temporary tensors
    del proj, mask_flat, x_norm, x_norm_f32

    # ----------------------- 4️⃣ batched GEMM (tri‑multiplicative) -----------------------
    # reshape to [B, H, N, N] for batched matmul
    left  = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left.view(B * hidden_dim, N, N)      # [B·H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ----------------------- 5️⃣ second LayerNorm + out‑gate (float32 LN) -----------------------
    out_flat = out.view(R, hidden_dim).contiguous()          # [R, H] fp16
    # layer‑norm in float32 for numerical stability, then cast back
    out_norm_f32 = F.layer_norm(
        out_flat.float(),
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # float32
    out_norm = out_norm_f32.to(torch.float16) * out_gate       # fp16 * fp16 → fp16
    del out_flat, out_gate, out, out_norm_f32

    # ----------------------- 6️⃣ final projection back to `dim` -----------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # [R, dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)    # cast to float32

    return final
```
