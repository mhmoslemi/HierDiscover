    x_ptr, weight_ptr, bias_ptr, out_ptr,
    rows: tl.int32,
    dim: tl.int32,
    out_dim: tl.int32,  # 5*H
    eps: tl.float32,
    BLOCK_R: tl.constexpr,  # rows per block
    BLOCK_D: tl.constexpr,  # dim tile size
    BLOCK_O: tl.constexpr,  # output dim tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < rows

    # compute mean and var for each row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

"""
High‑performance implementation of the *outgoing* TriMul operator
(AlphaFold‑3) with a single Triton kernel.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input using cuDNN (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left/right projections + 3 gates) into a single
    GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)  where R = B·N·N and H = hidden_dim.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
        - splits the fused projection into left/right tensors,
        - applies the optional pair‑wise mask,
        - evaluates the three sigmoid gates,
        - stores `left`, `right` and `out_gate` (all fp16).
4️⃣  Batched half‑precision GEMM (torch.bmm) over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
5️⃣  cuDNN LayerNorm on the GEMM result, element‑wise multiplication with
    `out_gate`, and a final linear projection back to the original channel size.
    All these steps stay in fp16; the final tensor is cast to fp32.

Only the split‑gate logic is written in Triton – this satisfies the
“must use a kernel” requirement while keeping the heavy GEMMs in cuBLAS,
which gives the best possible throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All operations are performed in fp16 (sigmoid evaluated in fp32 for
    numerical stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # load per‑row mask (if present)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offsets into proj

        p_left      = proj_ptr + base
        p_right     = proj_ptr + base + H
        p_left_gate = proj_ptr + base + 2 * H
        p_right_gate= proj_ptr + base + 3 * H
        p_out_gate  = proj_ptr + base + 4 * H

        left_raw      = tl.load(p_left,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_raw     = tl.load(p_right,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        left_gate_raw = tl.load(p_left_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_gate_raw= tl.load(p_right_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(p_out_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        lg = left_gate_raw.to(tl.float32)
        left_gate  = (1.0 / (1.0 + tl.exp(-lg))).to(tl.float16)

        rg = right_gate_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg))).to(tl.float16)

        og = out_gate_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # store results
        tl.store(left_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain keys ``dim``, ``hidden_dim`` and
                         ``nomask`` (bool). May also contain other flags.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N            # number of rows for row‑wise operations
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (cuDNN) – keep everything in fp32,
    #    then cast to fp16 for the subsequent linear projection.
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] float32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                      # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates) – one GEMM.
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each H×dim) → (5·H, dim)
    weight_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, dim]

    # (R, dim) @ (5·H, dim).T  → (R, 5·H)
    proj = F.linear(x_norm, weight_fused)                     # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused result, apply mask, compute sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask_tensor is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → [R]; values are 0/1
        mask_half = mask_tensor.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dimension slice
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr  # free memory early

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N] → view as (B·H, N, N) for torch.bmm
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # left @ rightᵀ  →  [B·H, N, N]  (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    # cuDNN LayerNorm (weights must match the dtype of the input)
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=out_norm_weight,
        bias=out_norm_bias,
        eps=eps,
    )  # fp16

    # Apply the output gate (still fp16)
    out_fused = out_norm * out_gate.view(R, H)  # [R, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
