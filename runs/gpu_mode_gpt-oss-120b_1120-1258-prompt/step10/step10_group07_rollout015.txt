kernel fused_ln_proj(
    x_ptr,
    y_ptr, # output (R, 5*H)
    norm_weight_ptr,
    norm_bias_ptr,
    proj_weight_ptr, # [5*H, D] in float16
    rows, D, O,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr
):
    pid = tl.program_id(0) # ranges over rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # Compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    # we also load raw values for later reuse; maybe store per block in local memory? We can compute normalized on-the-fly each d block.

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Load layernorm weight and bias (per dim)
    # Determine shape (D)
    # We'll compute normalized values and multiply with projection weights.

    for o_off in range(0, tl.cdiv(O, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < O

        # init accumulator
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)  # accumulate in fp32

        # iterate over d dimension
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # load x values
            x_ptrs = x_ptr + offs_m[:, None] * D + offs_d[None, :]
            x_vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # apply norm: (x - mean) * rstd * weight_norm + bias_norm
            w_norm = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
            b_norm = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

            # normalized: (x_vals - mean[:,None]) * rstd[:,None] * w_norm + b_norm
            normed = (x_vals - mean[:, None]) * rstd[:, None] * w_norm[None, :] + b_norm[None, :]  # fp32
            # convert to fp16 if needed for multiplication with proj weight
            normed_fp16 = normed.to(tl.float16)

            # load projection weight slice: each output feature o maps to D dimensions
            # weight matrix stored row-major [O, D] (that's consistent with .view weight shape)
            # We need slice of weight rows for output block and cols for d block.
            proj_ptrs = proj_weight_ptr + offs_o[:, None] * D + offs_d[None, :]  # shape [BLOCK_O, BLOCK_D]

            proj_vals = tl.load(proj_ptrs, mask=mask_o[:, None] & mask_d[None, :], other=0.0)  # fp16

            # dot product: acc[m,o] += sum_{d} normed_fp16[m,d] * proj_vals[o,d]
            # Using tl.dot: dot between dim=BLOCK_D vectors for each row m and each output o separate? tl.dot expects 2D arr: [M, K] dot [K, N] -> [M,N].
            # We can do: acc += tl.dot(normed_fp16, proj_vals)  # normed_fp16 shape [BLOCK_M, BLOCK_D], proj_vals shape [BLOCK_O, BLOCK_D] -> need transpose.
            # Actually we need normed_fp16 [M,K] dot proj_vals^T [K,N]? The weight slice is [O,K] (i.e., rows are output features). For dot we need [M,K] @ [K,N] => [M,N], so we need proj_vals_T = tl.load(proj_ptrs) transposed.
            # Use tl.dot(normed_fp16, proj_vals.T) but we can load weight as [K, O] using Expose. Simpler: use tl.dot. We need to make shape [BLOCK_M, BLOCK_D] and [BLOCK_D, BLOCK_O] multiply to [BLOCK_M, BLOCK_O].
            # So we should load weight slice as shape [BLOCK_D, BLOCK_O] (transposed).
            proj_ptrs_T = proj_weight_ptr + offs_d[:, None] * O + offs_o[None, :]  # [BLOCK_D, BLOCK_O]
            proj_vals_T = tl.load(proj_ptrs_T, mask=mask_d[:, None] & mask_o[None, :], other=0.0)  # fp16
            # dot
            acc += tl.dot(normed_fp16, proj_vals_T)  # yields fp32

        # after d loop, store acc (fp32) to y
        y_ptrs = y_ptr + offs_m[:, None] * O + offs_o[None, :]
        # Cast to fp16 for output
        tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

@triton.jit
def row_center_kernel(x_ptr, centered_ptr, inv_std_ptr, rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)  # 1/std

    # store inv_std (rstd)
    tl.store(inv_std_ptr + offs_m, rstd, mask=mask_m)

    # second pass: compute centered values (x - mean)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        centered = (vals - mean[:, None]).to(tl.float16)
        out_ptr = centered_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, centered, mask=mask_m[:, None] & mask_d[None, :])
```

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    # Flatten input
    x_flat = x.view(R, dim).contiguous()  # fp32
    # allocate buffers
    centered = torch.empty_like(x_flat, dtype=torch.float16, device=device)
    inv_std = torch.empty(R, dtype=torch.float32, device=device)
    # Launch rowcenter kernel
    BLOCK_M = 128  # maybe tune
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    row_center_kernel[grid](
        x_flat,
        centered,
        inv_std,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # Precompute scaled weights and biases
    norm_weight = weights["norm.weight"]   # shape [dim] fp32
    norm_bias = weights["norm.bias"]       # shape [dim] fp32

    # helper
    def scale_weight(w):
        return (w * norm_weight[None, :]).to(torch.float16)  # (H, D)
    def bias_term(w):
        # w shape (H,D)
        # bias = norm_bias @ w.T
        return torch.matmul(norm_bias, w.t()).to(torch.float16)  # (H,)

    left_proj_scaled = scale_weight(weights["left_proj.weight"])
    right_proj_scaled = scale_weight(weights["right_proj.weight"])
    left_gate_scaled = scale_weight(weights["left_gate.weight"])
    right_gate_scaled = scale_weight(weights["right_gate.weight"])
    out_gate_scaled = scale_weight(weights["out_gate.weight"])

    left_proj_bias = bias_term(weights["left_proj.weight"])
    right_proj_bias = bias_term(weights["right_proj.weight"])
    left_gate_bias = bias_term(weights["left_gate.weight"])
    right_gate_bias = bias_term(weights["right_gate.weight"])
    out_gate_bias = bias_term(weights["out_gate.weight"])

    # concatenate
    weight_concat_scaled = torch.cat([left_proj_scaled, right_proj_scaled,
                                      left_gate_scaled, right_gate_scaled,
                                      out_gate_scaled], dim=0)  # (5*H, dim)
    bias_concat = torch.cat([left_proj_bias, right_proj_bias,
                             left_gate_bias, right_gate_bias,
                             out_gate_bias], dim=0)  # (5*H,)

    # linear
    proj = torch.nn.functional.linear(centered, weight_concat_scaled, bias=None)  # (R,5*H)
    # apply inv_std scaling and add bias
    inv_std_fp16 = inv_std.to(torch.float16)
    proj = proj * inv_std_fp16[:, None] + bias_concat[None, :]

    # split
    slice0 = slice(0, H)
    slice1 = slice(H, 2*H)
    slice2 = slice(2*H, 3*H)
    slice3 = slice(3*H, 4*H)
    slice4 = slice(4*H, 5*H)

    left = proj[:, slice0]
    right = proj[:, slice1]
    left_gate = proj[:, slice2]
    right_gate = proj[:, slice3]
    out_gate = proj[:, slice4]

    # sigmoid gates in fp32 for stability
    left_gate_sig = torch.sigmoid(left_gate.float()).to(torch.float16)
    right_gate_sig = torch.sigmoid(right_gate.float()).to(torch.float16)
    out_gate_sig = torch.sigmoid(out_gate.float()).to(torch.float16)

    # mask
    if cfg.get("nomask", True) or mask is None:
        mask_flat = torch.ones(R, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)

    # Apply mask to left/right before gating or after gating; multiplication order doesn't matter
    left = left * mask_flat[:, None] * left_gate_sig
    right = right * mask_flat[:, None] * right_gate_sig
    out_gate_sig = out_gate_sig  # unchanged

    # reshape for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # compute out
    out_mat = torch.matmul(left, right.transpose(-1, -2))  # (B, H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # second layernorm (use torch.nn.functional.layer_norm)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=[H],
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # multiply with out_gate_sig (shape [R, H]) - need to reshape to match out_norm shape
    out_gate_sig_reshape = out_gate_sig.view(B, N, N, H)
    out_fused = out_norm * out_gate_sig_reshape  # fp32 * fp16? out_norm is float32 (by default layer_norm returns fp32)
    # ensure out_fused is float16 for final linear if to_out.weight is half
    out_fused_half = out_fused.to(torch.float16)

    # final projection
    final_half = torch.nn.functional.linear(out_fused_half, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

def row_center_kernel(
    x_ptr,            # const   *[rows, dim] (float32)
    out_ptr,          # mutable *[rows, dim] (float16)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute sum, sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim

    rstd = tl.rsqrt(var + eps)  # 1/std

    # compute normalized centered values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):

"""
TriMul (outgoing) – fused *Row‑wise LayerNorm + 5‑way Linear* (Triton + Torch)

Algorithm
---------
1️⃣  Row‑wise LayerNorm (μ, σ²) is computed by a tiny Triton kernel.
    The kernel writes the *centered & normalized* tensor
        X̂ = (X – μ) * rsqrt(σ²+ε)          (float16)
    which is exactly the input needed for the following linear layers
    (the γ/β of LayerNorm are folded into the weight/bias of the linear).

2️⃣  The five Linear maps (left/right projections + three gates) are
    concatenated into a single GEMM:
        Y = X̂ @ Ŵᵀ  +  b̂                (fp16)
    where
        Ŵ = W ⊙ γ   (γ broadcast over the input dimension)
        b̂ = β ⋅ Wᵀ   (β · weightᵀ, a per‑output‑feature constant)

3️⃣  The GEMM output `Y` is sliced into the five tensors
    (left, right, left‑gate, right‑gate, out‑gate).  The three gates are
    passed through a sigmoid (computed in fp32 for numerical stability),
    the mask (if any) is applied, and `left`/`right` are multiplied by their
    respective gate values.

4️⃣  Batched FP16 matmul (Tensor‑Core) implements the pair‑wise update:
        out = left @ rightᵀ   →  (B·H, N, N)

5️⃣  A second LayerNorm on `out` is performed with PyTorch’s cuDNN kernel,
    followed by element‑wise multiplication with the `out_gate`.

6️⃣  Final Linear projection back to `dim` (fp16 → fp32) completes the
    forward pass.

Only the row‑wise LayerNorm is a custom Triton kernel; all heavy matrix
multiplications use cuBLAS/cuDNN for maximum throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 0: row‑wise mean / variance & write (X‑μ)·rsqrt(σ²+ε)  (fp16)
# ----------------------------------------------------------------------
@triton.jit
def row_center_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    out_ptr,        # mutable *[rows, dim]   (float16)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean and variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write (X‑μ) * rstd  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = ((vals - mean[:, None]) * rstd[:, None]).to(tl.float16)
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, norm,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of rows (= B·N·N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣  Flatten input
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()      # [R, dim]  float32

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise (X‑μ)·rsqrt(σ²+ε)  (fp16)   <-- custom Triton kernel
    # ------------------------------------------------------------------
    x_norm_centered = torch.empty_like(x_flat, dtype=torch.float16, device=device)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    row_center_kernel[grid](
        x_flat,
        x_norm_centered,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat   # free original fp32 tensor

    # ------------------------------------------------------------------
    # 2️⃣  Fuse γ/β into the five linear maps
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]      # [dim]  fp32
    norm_b = weights["norm.bias"]        # [dim]  fp32

    # helper to fold LayerNorm γ into a weight matrix (W ⊙ γ) and compute β·Wᵀ
    def fold_ln(w):
        # w : [out, dim] fp32
        w_scaled = (w * norm_w[None, :]).to(torch.float16)                     # ⊙ γ
        b_scaled = torch.mv(w, norm_b).to(torch.float16)                       # β·Wᵀ
        return w_scaled, b_scaled

    w_l,   b_l   = fold_ln(weights["left_proj.weight"])
    w_r,   b_r   = fold_ln(weights["right_proj.weight"])
    w_lg,  b_lg  = fold_ln(weights["left_gate.weight"])
    w_rg,  b_rg  = fold_ln(weights["right_gate.weight"])
    w_og,  b_og  = fold_ln(weights["out_gate.weight"])

    # concatenate for a single GEMM
    weight_cat = torch.cat([w_l, w_r, w_lg, w_rg, w_og], dim=0)    # [(5·H), dim]  fp16
    bias_cat   = torch.cat([b_l, b_r, b_lg, b_rg, b_og], dim=0)    # [(5·H),]      fp16

    # ------------------------------------------------------------------
    # 3️⃣  Single GEMM: (R,dim) @ (5·H,dim)ᵀ  →  (R,5·H)
    # ------------------------------------------------------------------
    proj = F.linear(x_norm_centered, weight_cat, bias_cat)          # fp16, shape [R,5·H]
    del x_norm_centered, weight_cat, bias_cat

    # ------------------------------------------------------------------
    # 4️⃣  Slice the five parts
    # ------------------------------------------------------------------
    slice0 = slice(0, H)
    slice1 = slice(H, 2 * H)
    slice2 = slice(2 * H, 3 * H)
    slice3 = slice(3 * H, 4 * H)
    slice4 = slice(4 * H, 5 * H)

    left_raw      = proj[:, slice0]   # [R, H]
    right_raw     = proj[:, slice1]
    left_gate_raw = proj[:, slice2]
    right_gate_raw= proj[:, slice3]
    out_gate_raw  = proj[:, slice4]
    del proj

    # ------------------------------------------------------------------
    # 5️⃣  Sigmoids (fp32 → fp16) – numerical‑stable
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.float()).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣  Apply mask (if present) and combine with gates
    # ------------------------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        mask_flat = torch.ones(R, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]

    # broadcast mask over hidden dimension
    mask_exp = mask_flat[:, None]                       # [R,1]

    left  = left_raw  * mask_exp * left_gate
    right = right_raw * mask_exp * right_gate
    # out_gate stays as a separate tensor for the later multiplication
    del left_raw, right_raw, left_gate_raw, right_gate_raw, left_gate, right_gate, mask_exp

    # ------------------------------------------------------------------
    # 7️⃣  Batched matmul over the hidden dimension (FP16 → Tensor‑Core)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.matmul(left, right.transpose(-1, -2))            # fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # [B, N, N, H]
    del left, right, out_mat

    # ------------------------------------------------------------------
    # 8️⃣  Second LayerNorm (cuDNN) + multiplication with out_gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=[H],
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # float32

    # out_gate has shape [R, H] → reshape to match out_norm
    out_gate_reshape = out_gate.view(B, N, N, H).float()
    out_fused = out_norm * out_gate_reshape                       # fp32
    del out, out_gate, out_gate_reshape, out_norm

    # ------------------------------------------------------------------
    # 9️⃣  Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(out_fused.to(torch.float16),
                          weights["to_out.weight"].to(torch.float16))  # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
