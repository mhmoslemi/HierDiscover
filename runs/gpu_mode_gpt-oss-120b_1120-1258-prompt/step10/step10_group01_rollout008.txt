"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm of the 4‑D input (torch.nn.functional.layer_norm,
   fp32 → fp16).

2️⃣ Fuse the five linear maps (left/right proj + three gates) into one
   GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H),  R = B·N·N.
   The weight matrix is built once and kept in fp16.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   • reads the fused projection,
   • (optionally) multiplies by the pair‑wise mask,
   • evaluates the three sigmoid gates (fp32 → fp16),
   • writes three tensors  left, right, out_gate  of shape (R, H).

4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   Implemented with `torch.bmm`, which uses cuBLAS Tensor‑Cores.

5️⃣ Row‑wise LayerNorm on the result, element‑wise multiplication with
   `out_gate` (torch.nn.functional.layer_norm, fp16).

6️⃣ Final projection back to the original `dim`:
        out_fused @ to_out.weightᵀ
   (again a single GEMM in fp16).

All heavy linear algebra is performed by cuBLAS; only the mask / gating
stage is handcrafted in Triton.  The implementation stays fully
autograd‑free (forward‑only) and returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel  : split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (r indexes B·N·N):
        left  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic is fp16 except the sigmoid (computed in fp32 for
    numerical stability).  The layout of `proj` is
        [ left , right , left_gate , right_gate , out_gate ].
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- optional mask (scalar per row) ---------------------------------
    if HAS_MASK:
        # mask values are {0.,1.} stored as fp16
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                     # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- iterate over hidden dimension ----------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointers inside the fused projection
        # layout: [ left , right , left_gate , right_gate , out_gate ]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(right_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(right_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(out_gate_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # ----- sigmoid (stable, fp32) --------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store --------------------------------------------------------
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, [B, N, N, dim], dtype float32
        mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
        weights_dict : dict with all Linear/LayerNorm params (float32)
        config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim        = cfg["dim"]            # input channel dimension
    hidden_dim = cfg["hidden_dim"]    # hidden dimension (H)
    eps        = 1e-5
    device     = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                     # number of rows in the flattened view
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch, fp32 → fp16)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x_norm = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # shape [B, N, N, dim] , fp16

    # flatten to (R, dim) for the fused linear projection
    x_norm_flat = x_norm.view(R, dim).contiguous()
    del inp, x_norm   # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection → (R, 5*H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
            w["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)               # (5*H, dim)

    proj = F.linear(x_norm_flat, fused_w)   # (R, 5*H), fp16
    del fused_w, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # no mask → compile kernel with HAS_MASK=0
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # mask stored as {0.,1.} – flatten to (R,)
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16

    # launch configuration
    BLOCK_R = 128                     # rows per program
    BLOCK_H = 32                      # hidden dim chunk (must divide H≤384)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update  (batched GEMM over H)
    # ------------------------------------------------------------------
    # shape to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)       # [B·H, N, N]

    # left @ rightᵀ  (half‑precision Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + gate multiplication (torch)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                 # fp16

    # apply out_gate (need the same layout)
    out_gate_reshaped = out_gate.view(B, N, N, H)
    out_norm = out_norm * out_gate_reshaped
    del out_gate, out_gate_reshaped, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out_flat = out_norm.view(R, H).contiguous()
    final_half = F.linear(
        out_flat,
        w["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
