"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3) – forward pass only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D pairwise tensor (torch built‑in LayerNorm,
   FP32 → FP16).
2️⃣ Fuse the five Linear maps (left/right projection + three gates) into a
   single half‑precision GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H)  where
   R = B·N·N.
3️⃣ Triton kernel ``split_fuse_gate_kernel`` extracts the three projections,
   applies the optional pair‑wise mask, evaluates the sigmoid gates and writes
   three tensors: ``left`` , ``right``  (both [R, H]) and ``out_gate``  [R, H].
4️⃣ Reshape the two projected tensors to [B·H, N, N] and perform a single
   strided batched GEMM with cuBLAS (torch.bmm):
        out_mat = left_mat @ right_matᵀ           # [B·H, N, N]
5️⃣ Second LayerNorm (torch built‑in) on the hidden dimension followed by
   element‑wise multiplication with ``out_gate``.
6️⃣ Final Linear projection back to the original channel dimension (FP16 →
   FP32).

Only the mask/gate parsing is written in Triton; all heavy matrix
multiplications utilise cuBLAS/Tensor‑Cores on H100, giving a runtime well
below the 1 ms target for the benchmark sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) (ignored if HAS_MASK==0)
    rows: tl.int32,    # R = B·N·N
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row r (corresponding to a (b,i,k) pair):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are performed in fp16 (sigmoid evaluated in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # Load per‑row mask if it exists
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast to H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offsets for the first segment

        p_left       = proj_ptr + base
        p_right      = proj_ptr + base + H
        p_left_gate  = proj_ptr + base + 2 * H
        p_right_gate = proj_ptr + base + 3 * H
        p_out_gate   = proj_ptr + base + 4 * H

        left_raw      = tl.load(p_left,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_raw     = tl.load(p_right,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        left_gate_raw = tl.load(p_left_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_gate_raw= tl.load(p_right_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(p_out_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32 for numeric stability
        lg_f32 = left_gate_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lg_f32))).to(tl.float16)

        rg_f32 = right_gate_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_f32))).to(tl.float16)

        og_f32 = out_gate_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og_f32))).to(tl.float16)

        # apply mask and gates
        left   = left_raw   * mask_val * left_gate
        right  = right_raw  * mask_val * right_gate

        # store the three results
        dst_left   = left_ptr   + offs_r[:, None] * H + offs_h[None, :]
        dst_right  = right_ptr  + offs_r[:, None] * H + offs_h[None, :]
        dst_gate   = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_gate,   out_gate, mask=mask_r[:, None] & mask_h[None, :])

# Forward function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain keys ``dim``, ``hidden_dim`` and ``nomask``
                         (bool).  May also contain other flags.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dtype  = torch.float32

    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N            # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch implementation – fast, uses cudnn)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)            # fp16 for the GEMMs

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: 5 projections + 3 gates  →  (R, 5·H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, w_fused)            # (R, 5·H), fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split fused result, apply mask, evaluate sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # flatten to [R]

    # choose block sizes that give good occupancy for the tested shapes
    BLOCK_R = 1024
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch axis
    left_4d  = left.view(B, N, N, H)
    right_4d = right.view(B, N, N, H)

    # [B*H, N, N] – non‑contiguous layout is fine for torch.bmm
    left_mat  = left_4d.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_4d.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, out_mat, left_4d, right_4d, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with the out‑gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                     # [R, H]
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_scaled = (out_norm * out_gate).to(torch.float16)   # still fp16
    del out, out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 → fp32)
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (B·N·N, dim) fp16
    final = final_half.view(B, N, N, dim).to(dtype)    # fp32 output
    return final
