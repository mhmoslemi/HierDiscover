"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D input  (FP32 → FP16).  
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a single
   GEMM:  X_norm @ W_fusedᵀ  →  (R × 5·H)  where  R = B·N·N .
3️⃣ One Triton kernel splits the fused result, applies the optional pair‑wise mask,
   evaluates the sigmoid gates and writes three tensors (left, right, out_gate)
   in FP16 (sigmoid computed in FP32 for stability).
4️⃣ Batched FP16 GEMM over the hidden dimension:
        out = left @ rightᵀ    (B·H independent N×N matmuls)
   Implemented with `torch.matmul` → cuBLAS/Tensor‑Cores.
5️⃣ A second Triton kernel fuses the second LayerNorm, the out‑gate multiplication
   and the final linear projection back to `dim`.  All ops stay in FP16 with FP32
   accumulation for the statistics and the final matmul.
6️⃣ Return the result cast to FP32.

All kernels are written for Triton 3.3.1 and are intended for H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]      (float32)
    y_ptr,          # mutable *[rows, dim]      (float16)
    weight_ptr,     # const   *[dim]           (float32)   γ
    bias_ptr,       # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2: split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    Ops are FP16; sigmoid evaluated in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # mask per row (broadcast over H) – if none, use 1.0
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # write out
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3: fused second LayerNorm + out‑gate multiplication + final projection
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_proj_kernel(
    x_ptr,            # const   *[rows, H]          (float16)  – output of the batched GEMM
    gate_ptr,         # const   *[rows, H]          (float16)  – out_gate
    y_ptr,            # mutable *[rows, dim]       (float16)  – final result
    ln_weight_ptr,    # const   *[H]               (float32)  – γ of to_out_norm
    ln_bias_ptr,      # const   *[H]               (float32)  – β of to_out_norm
    proj_weight_ptr,  # const   *[dim, H]          (float32)  – to_out.weight
    proj_bias_ptr,    # const   *[dim]             (float32)  – to_out.bias (zero in our case)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    """
    For each row r:
        y = LayerNorm(x)               # uses ln_weight/ln_bias (γ,β)
        y = y * gate                 # element‑wise out‑gate
        y = y @ proj_weightᵀ + proj_bias   # final linear projection
    Everything stays in FP16.  Accumulations are in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute mean / variance of x (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H
        ptr = x_ptr + offs_m[:, None] * H + offs_h[None, :]  # [M,H]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ fused norm + gate + projection (FP32 accum)
    # --------------------------------------------------------------
    # allocate output accumulator (M x BLOCK_O) in FP32
    for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)      # [BLOCK_O]
        mask_o = offs_o < dim

        # accumulator for this output block
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        # iterate over hidden dimension blocks
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
            mask_h = offs_h < H

            # load slice of x and gate (both [M, BLOCK_H])
            x_slice = tl.load(x_ptr + offs_m[:, None] * H + offs_h[None, :],
                              mask=mask_m[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)   # fp32

            # layer‑norm: (x - μ) * rstd
            x_norm = (x_slice - mean[:, None]) * rstd[:, None]

            # apply ln weight & bias (γ/β) – both fp32
            w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ
            b_ln = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β
            x_norm = x_norm * w_ln[None, :] + b_ln[None, :]                # fp32

            # multiply by out‑gate
            gate_slice = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                                 mask=mask_m[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)                # fp32
            gated = x_norm * gate_slice                                   # fp32

            # load projection weight slice: shape [BLOCK_O, BLOCK_H]
            w_proj = tl.load(proj_weight_ptr + offs_o[:, None] * H + offs_h[None, :],
                             mask=mask_o[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)                  # fp32

            # dot‑product: acc += gated @ w_projᵀ
            acc += tl.dot(gated, w_proj)                                 # (M,O)

        # add bias for this output block
        b_proj = tl.load(proj_bias_ptr + offs_o, mask=mask_o, other=0.0)   # fp32
        acc += b_proj[None, :]

        # store result as FP16
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptr, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates (5 * H rows)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, w_concat)       # fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).reshape(R)  # flatten
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.matmul(left, right.transpose(-1, -2))           # fp16

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication + final projection
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    norm_gate_proj_kernel[grid2](
        out_flat,                                   # x_ptr
        out_gate.view(R, H),                        # gate_ptr
        final_half,                                 # y_ptr (final output)
        weights["to_out_norm.weight"],   # ln_weight (γ)  fp32
        weights["to_out_norm.bias"],     # ln_bias   (β)  fp32
        weights["to_out.weight"],        # proj_weight (dim, H) fp32
        weights["to_out.bias"],          # proj_bias (dim)      fp32
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_O=32,                     # split dim into 32‑wide tiles
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # Cast back to FP32 and reshape to the original 4‑D layout
    # ------------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
