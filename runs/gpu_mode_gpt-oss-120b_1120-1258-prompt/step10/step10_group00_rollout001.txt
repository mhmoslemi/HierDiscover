TriMul (outgoing) – high‑performance Triton + cuBLAS implementation

Algorithm (forward only)
------------------------
1️⃣ Row‑wise LayerNorm of the 4‑D input (fp32→fp16) – Triton.
2️⃣ Fuse the three linear projections (left/right) and the three gates
   into a single GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)  (R = B·N·N).
3️⃣ Triton kernel splits the fused result, applies the optional pair‑wise
   mask, evaluates sigmoid gates and writes three tensors:
   left, right, out_gate  (all fp16).
4️⃣ Batched half‑precision matrix multiplication with cuBLAS/Tensor‑Cores:
   out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ   → (R, H).
5️⃣ Second LayerNorm + element‑wise multiplication with out_gate – Triton.
6️⃣ Final linear projection back to `dim` (fp16→fp32) – torch.nn.functional.linear.

All heavy MatMul work is delegated to cuBLAS, while the two LayerNorms and
the projection‑split+gate kernel are hand‑written in Triton for maximal
throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (input fp32, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (fp32)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    w_ptr,          # const   *[dim]        (fp32) γ
    b_ptr,          # const   *[dim]        (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β

        out = norm * w[None, :] + b[None, :]                   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # optional mask (broadcasted later)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout per row: [L, R, L_gate, R_gate, O_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offsets into proj

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        left_dst_ptr      = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr     = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,          # const   *[rows, dim] (fp16)
    gate_ptr,       # const   *[rows, dim] (fp16) – out_gate
    y_ptr,          # mutable *[rows, dim] (fp16)
    w_ptr,          # const   *[dim]      (fp32) γ
    b_ptr,          # const   *[dim]      (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw values
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β
        norm = norm * w[None, :] + b[None, :]                  # fp32

        # load gate (out_gate)
        gptr = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(gptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : dict with keys (all float32):
            - "norm.weight", "norm.bias"
            - "left_proj.weight", "right_proj.weight"
            - "left_gate.weight", "right_gate.weight", "out_gate.weight"
            - "to_out_norm.weight", "to_out_norm.bias"
            - "to_out.weight"
        * config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512          # more rows per program reduces launch overhead
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (fp32)
        weights["norm.bias"],      # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # concatenate weight matrices (out_features, in_features)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # (R, dim) @ (5*H, dim).T → (R, 5*H)    using fp16 GEMM
    proj = F.linear(x_norm, w_fused)        # fp16
    del w_fused, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates
    # ------------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # compile‑time flag disables mask reads
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
    else:
        # flatten mask and convert to fp16 (0/1)
        mask_flat = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_flat
        HAS_MASK = 1

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (fp16, Tensor‑Core)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.reshape(B * H, N, N)    # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)   # [B·H, N, N]

    # out_mat[i] = left_mat[i] @ right_mat[i].T   → [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiply with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()               # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # weight shape for Linear(dim_out=dim, dim_in=H) → [dim, H]
    final_fp16 = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
