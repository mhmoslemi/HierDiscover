sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)

    mask_h = offs_h < hidden_dim

    left_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    right_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    lg_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    rg_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
    og_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    # loop over dim tiles
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x slice
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_tile = tl.load(x_ptrs,

                            mask=mask_d,
                            other=0.0).to(tl.float32)

        # compute base pointer for weight matrices: weight_ptr + offs_d * hidden_dim + offs_h
        # left_proj
        left_w_ptr = left_proj_w_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]
        left_w = tl.load(left_w_ptr,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # right_proj
        right_w_ptr = right_proj_w_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]
        right_w = tl.load(right_w_ptr,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # left_gate
        left_gate_w_ptr = left_gate_w_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]
        left_gate_w = tl.load(left_gate_w_ptr,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # right_gate
        right_gate_w_ptr = right_gate_w_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]
        right_gate_w = tl.load(right_gate_w_ptr,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # out_gate
        out_gate_w_ptr = out_gate_w_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]
        out_gate_w = tl.load(out_gate_w_ptr,
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        left_acc += tl.dot(norm_tile, left_w)
        right_acc += tl.dot(norm_tile, right_w)
        lg_acc += tl.dot(norm_tile, left_gate_w)
        rg_acc += tl.dot(norm_tile, right_gate_w)
        og_acc += tl.dot(norm_tile, out_gate_w)

    # sigmoid function: 1/(1+exp(-x))

    # right = right_acc * rg_sig * mask

    right_fp = right_acc.to(tl.float16)

        mask_tile = mask_vals_f32[:, None]  # shape [BLOCK_M, 1]
    else:
        mask_tile = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float32)

    right_fp = right_fp * rg_sig * mask_tile

    right_ptrs = right_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]

    tl.store(right_ptrs, right_fp, mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptrs, og_sig, mask=mask_m[:, None] & mask_h[None, :])

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N

    x_flat = x.view(R, dim).contiguous()

    left_proj_w = weights["left_proj.weight"].to(torch.float16).t().contiguous()   # [dim, H]
    right_proj_w = weights["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w = weights["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w = weights["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w = weights["out_gate.weight"].to(torch.float16).t().contiguous()

    # Allocate output arrays
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:

        mask_flat = mask.to(torch.float16).reshape(R)   # [R]

"""
Tri‑Multiplicative (outgoing) update – fused Triton implementation.

Algorithm (overview)
--------------------
1️⃣  Row‑wise LayerNorm + five fused linear projections (left, right,
    left‑gate, right‑gate, out‑gate) in a single Triton kernel.
    The kernel:
      * computes mean / variance of each input row,
      * normalises the row,
      * for each hidden‑dim block evaluates 5 matrix‑vector products
        (norm × W) using `tl.dot`,
      * applies sigmoid to the three gate results,
      * multiplies `left/right` by their gates and an optional mask,
      * writes three fp16 tensors: `left`, `right`, `out_gate`.

2️⃣  Batched GEMM (`B·H` matrices of shape `N×N`) implements the
    `einsum('... i k d, ... j k d -> ... i j d')` step.
    This is performed with `torch.bmm`, which on H100 uses Tensor‑cores.

3️⃣  Row‑wise LayerNorm on the hidden‑dim tensor, merged with the final
    out‑gate multiplication (`to_out_norm * out_gate`).  Implemented by a
    second Triton kernel (identical to the reference `layernorm_gate_kernel`).

4️⃣  Final linear projection `hidden_dim → dim` (fp16 GEMM) using
    `torch.nn.functional.linear`.  The result is cast back to float32.

Only the two Triton kernels constitute the custom‑kernel part; the heavy
matrix‑multiplications stay in cuBLAS where they are optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm + 5 fused linear projections + gates
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr,                     # const   *[rows, dim]       (float32)
    left_ptr,                  # mutable *[rows, hidden]   (float16)
    right_ptr,                 # mutable *[rows, hidden]   (float16)
    out_gate_ptr,              # mutable *[rows, hidden]   (float16)
    mask_ptr,                  # const   *[rows]           (float16) – ignored if HAS_MASK=0
    norm_weight_ptr,           # const   *[dim]            (float32) – LayerNorm γ
    norm_bias_ptr,             # const   *[dim]            (float32) – LayerNorm β
    left_proj_w_ptr,           # const   *[dim, hidden]    (float16)
    right_proj_w_ptr,          # const   *[dim, hidden]    (float16)
    left_gate_w_ptr,           # const   *[dim, hidden]    (float16)
    right_gate_w_ptr,          # const   *[dim, hidden]    (float16)
    out_gate_w_ptr,            # const   *[dim, hidden]    (float16)
    rows: tl.int32,
    dim: tl.int32,
    hidden: tl.int32,
    eps: tl.float32,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # dim tile
    BLOCK_H: tl.constexpr,     # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1) mean & variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) optional unary mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_row = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # [BLOCK_M] fp16
    else:
        mask_row = tl.full([BLOCK_M], 1.0, dtype=tl.float16)           # broadcast=1

    # --------------------------------------------------------------
    # 3) loop over hidden‑dim blocks
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # accumulators for the five outputs (fp32)
        left_acc   = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        lgate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        rgate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        ogate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        # ----------------------------------------------------------
        #   loop over dim tiles and accumulate dot products
        # ----------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # ---- load raw input tile -------------------------------------------------
            x_tile = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # ---- normalise the tile ---------------------------------------------------
            gamma = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
            beta  = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

            norm_tile = ((x_tile - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

            # ---- load weight tiles (dim × hidden) ------------------------------------
            # layout of each weight matrix: [dim, hidden] (contiguous)
            w_left  = tl.load(
                left_proj_w_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_right = tl.load(
                right_proj_w_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_lgate = tl.load(
                left_gate_w_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_rgate = tl.load(
                right_gate_w_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_ogate = tl.load(
                out_gate_w_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # ---- accumulate dot products ------------------------------------------------
            left_acc  += tl.dot(norm_tile, w_left)
            right_acc += tl.dot(norm_tile, w_right)
            lgate_acc += tl.dot(norm_tile, w_lgate)
            rgate_acc += tl.dot(norm_tile, w_rgate)
            ogate_acc += tl.dot(norm_tile, w_ogate)

        # --------------------------------------------------------------
        # 4) apply sigmoid gates, mask and store results (fp16)
        # --------------------------------------------------------------
        # sigmoid in fp32 for stability, then cast to fp16
        lgate_sig = (1.0 / (1.0 + tl.exp(-lgate_acc))).to(tl.float16)
        rgate_sig = (1.0 / (1.0 + tl.exp(-rgate_acc))).to(tl.float16)
        ogate_sig = (1.0 / (1.0 + tl.exp(-ogate_acc))).to(tl.float16)

        # mask broadcast to (BLOCK_M, BLOCK_H)
        mask_tile = mask_row[:, None]    # fp16

        left_fp = left_acc.to(tl.float16) * lgate_sig * mask_tile
        right_fp = right_acc.to(tl.float16) * rgate_sig * mask_tile
        ogate_fp = ogate_sig                       # no mask on out‑gate

        # store
        left_ptrs = left_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        right_ptrs = right_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        og_ptrs = out_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :]

        tl.store(left_ptrs,  left_fp,  mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right_fp, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(og_ptrs,    ogate_fp, mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣  Second LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # normalized * weight + bias, then multiply by gate (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr   + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr+ offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        # layer‑norm parameters
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals
        y_fp16 = y_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters
                         (all tensors are float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device
    B, N, _, _   = x.shape
    R = B * N * N            # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + five projections + gates (single kernel)
    # ------------------------------------------------------------------
    # flatten input to [R, dim]
    x_flat = x.view(R, dim).contiguous()                     # fp32

    # transpose weights so that they are stored as [dim, hidden]
    left_proj_w   = weights["left_proj.weight"].to(torch.float16).t().contiguous()
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).t().contiguous()

    # allocate outputs
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling
    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)     # [R]

    # launch parameters (tuned for H100)
    BLOCK_M = 64   # rows per program
    BLOCK_D = 32   # dim tile
    BLOCK_H = 32   # hidden‑dim tile

    grid = (triton.cdiv(R, BLOCK_M),)

    fused_norm_proj_gate_kernel[grid](
        x_flat,
        left,
        right,
        out_gate,
        mask_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        dim,
        H,
        eps,
        HAS_MASK=HAS_MASK,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,          # H100 benefits from more warps
    )
    del x_flat, mask_flat

    # ------------------------------------------------------------------
    # 2️⃣  Batched GEMM over the hidden dimension (i‑k × j‑k → i‑j)
    # ------------------------------------------------------------------
    # reshape for batch GEMM:  [B, N, N, H] → [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 3️⃣  Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                     # [R, H] fp16
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2 = 64
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(R, BLOCK_M2),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 4️⃣  Final linear projection back to `dim` (fp16 GEMM) and cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, hidden_dim]
        bias=None,
    )   # → [R, dim]  (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
