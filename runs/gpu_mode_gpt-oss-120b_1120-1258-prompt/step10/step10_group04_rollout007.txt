"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass

The implementation follows the reference PyTorch module but moves the three
expensive per‑row operations to Triton kernels:

1️⃣ Row‑wise LayerNorm on the input tensor  [B,N,N,dim] → fp16.
2️⃣ A *single* linear projection that produces the five tensors required for the
   outgoing update (left, right, left_gate, right_gate, out_gate).  The weight
   matrix is concatenated along the output dimension (shape = [5*H, dim]).
3️⃣ One Triton kernel splits the fused projection, applies the optional pair‑wise
   mask and the three sigmoid gates, yielding ``left`` , ``right`` and
   ``out_gate`` (all fp16).
4️⃣ The true multiplicative update is a batched GEMM:
   ``out = (B·H,N,N) @ (B·H,N,N)ᵀ``   (left @ rightᵀ)  – performed by cuBLAS.
5️⃣ A second Triton kernel fuses a row‑wise LayerNorm on the GEMM result with
   the multiplication by the ``out_gate``.
6️⃣ Final linear projection back to the original ``dim`` (fp16 GEMM) and a
   cast to float32.

All heavy arithmetic is in float16; the two LayerNorm kernels compute the
mean/variance in float32 for numeric stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----------------- mean / variance (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------- write normalized values (fp16) -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # optional pair‑wise mask (broadcasted over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # process hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # offsets for the five contiguous slices: [L,R,Lg,Rg,Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        left_raw        = tl.load(proj_ptr + base,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
        right_raw       = tl.load(proj_ptr + base + H,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
        left_gate_raw   = tl.load(proj_ptr + base + 2 * H,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
        right_gate_raw  = tl.load(proj_ptr + base + 3 * H,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
        out_gate_raw    = tl.load(proj_ptr + base + 4 * H,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)

        # sigmoid (fp32 for stability)
        left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :], left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :], right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :], out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- normalize, apply LN weight/bias and gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    torch.cuda.empty_cache()           # safety, not strictly needed
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    rows       = B * N * N          # total number of rows for row‑wise ops
    H          = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(rows, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512        # larger block reduces launch overhead on H100
    BLOCK_D = 64
    grid_ln1 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],        # γ (float32)
        weights["norm.bias"],          # β (float32)
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates  →  [R, 5*H] (fp16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices once (outside of the kernel)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [R, 5*H] fp16
    del w_fused, x_norm                     # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(rows, H, dtype=torch.float16, device=device)
    right     = torch.empty(rows, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, H, dtype=torch.float16, device=device)

    if nomask:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask is boolean/float, broadcasted over hidden dim
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(rows, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N]  – we deliberately avoid extra permutes by
    # using a strided view + a single .transpose that makes the data
    # contiguous for the GEMM.
    # left:  [rows, H] → [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H to the batch dimension (contiguous layout required by bmm)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten batch & head
    left_mat  = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)      # [B·H, N, N]

    # batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(rows, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 GEMM) + cast to fp32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [rows, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
