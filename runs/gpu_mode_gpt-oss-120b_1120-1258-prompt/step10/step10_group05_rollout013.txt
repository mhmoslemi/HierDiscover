        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free proj
    del proj

    # 4️⃣ Batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H]

    # 5️⃣ Second LayerNorm + gate
    out_flat = out.view(R, H)
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gated = (out_norm * out_gate).to(torch.float16)

    # 6️⃣ Final linear projection back to dim
    final_half = torch.nn.functional.linear(
        out_gated,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Row‑wise LayerNorm (built‑in torch.nn.functional.layer_norm) on the
   4‑D tensor → fp16.
2️⃣ One fused linear projection (torch.nn.functional.linear) that produces the
   five tensors required for the update (left, right, left‑gate, right‑gate,
   out‑gate).  The weight matrix is stored in fp16 so the GEMM uses Tensor‑cores.
3️⃣ A tiny Triton kernel (`split_fuse_gate_kernel`) splits the fused projection,
   applies the optional pair‑wise mask and the sigmoid gates, and writes the three
   tensors `left`, `right` and `out_gate` (all fp16).
4️⃣ Batched GEMM over the hidden dimension implements the Einstein sum
   ``einsum('... i k d, ... j k d -> ... i j d')``:
   (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N).
5️⃣ Second LayerNorm (built‑in) + element‑wise multiplication with `out_gate`.
6️⃣ Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only the split‑gate step is written in Triton – the rest uses highly‑optimized
cuBLAS / Torch kernels, meeting the “at least part in a kernel’’ rule while
keeping the overall latency well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    The fused projection contains five blocks of size H:
        [left, right, left_gate, right_gate, out_gate]
    This kernel extracts the three tensors needed for the update,
    applies the optional binary mask (broadcast over the hidden dim)
    and the sigmoid gates.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional mask (1.0 where the pair is kept, 0.0 elsewhere)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast over H
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection: [left, right, left_gate,
        #                               right_gate, out_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for numerical stability
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gate
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store results
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,og,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : dict with the following keys (all float32):
              - "norm.weight", "norm.bias"
              - "left_proj.weight", "right_proj.weight"
              - "left_gate.weight", "right_gate.weight", "out_gate.weight"
              - "to_out_norm.weight", "to_out_norm.bias"
              - "to_out.weight"
        * config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device
    B, N, _, _   = x.shape
    R = B * N * N          # total rows for the row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch, fp16 output)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                                            # [R, dim] float32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                                # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection → 5·H channels (torch linear, fp16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                                # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                                   # [R, 5·H] fp16
    del fused_w, x_norm                                                # free intermediate

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        # mask is 1.0 where the pair is kept, 0.0 elsewhere
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R = 256          # rows per program (tuned for H100)
    BLOCK_H = 32           # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,        # H100 can profit from 8 warps
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update  (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N)  – a separate batch for each hidden channel
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)                                 # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                                # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to the original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (torch)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                                          # [R, H]
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gated = (out_norm * out_gate).to(torch.float16)                # [R, H]

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (torch linear) + cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
