def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # expected shapes: input_tensor [B,N,N,dim] float32
    # mask_tensor [B,N,N] bool/float.

    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # flatten rows
    R = B * N * N
    x_flat = input_tensor.view(R, dim).contiguous()  # fp32

    # Prepare LN weight and bias
    ln_weight = weights["norm.weight"]
    ln_bias = weights["norm.bias"]

    # Prepare scaled weight matrices and bias vectors
    def prepare(w):
        # w shape [H, dim] float32
        # scaled = w * ln_weight (broadcast)
        scaled = (w * ln_weight).to(torch.float16)  # half
        sum_ws = (scaled.float().sum(dim=1)).contiguous()  # fp32
        bias = (ln_bias * w).sum(dim=1).contiguous()  # fp32
        return scaled, sum_ws, bias

    left_w_scaled, left_sum_ws, left_bias = prepare(weights["left_proj.weight"])
    right_w_scaled, right_sum_ws, right_bias = prepare(weights["right_proj.weight"])
    left_gate_w_scaled, left_gate_sum_ws, left_gate_bias = prepare(weights["left_gate.weight"])
    right_gate_w_scaled, right_gate_sum_ws, right_gate_bias = prepare(weights["right_gate.weight"])
    out_gate_w_scaled, out_gate_sum_ws, out_gate_bias = prepare(weights["out_gate.weight"])

    # output buffers
    left = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    # mask flatten (if present)
    if config.get("nomask", True):
        has_mask = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).view(R)
        has_mask = 1

    # launch kernel
    BLOCK_R = 128
    BLOCK_H = 32
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    proj_gate_fused_kernel[grid](
        x_ptr = x_flat,
        rows = R,
        dim = dim,
        H = hidden_dim,
        HAS_MASK = has_mask,
        mask_ptr = mask_fp16,
        left_w_ptr = left_w_scaled,
        right_w_ptr = right_w_scaled,
        left_gate_w_ptr = left_gate_w_scaled,
        right_gate_w_ptr = right_gate_w_scaled,
        out_gate_w_ptr = out_gate_w_scaled,
        left_sum_ws_ptr = left_sum_ws,
        right_sum_ws_ptr = right_sum_ws,
        left_gate_sum_ws_ptr = left_gate_sum_ws,
        right_gate_sum_ws_ptr = right_gate_sum_ws,
        out_gate_sum_ws_ptr = out_gate_sum_ws,
        left_bias_ptr = left_bias,
        right_bias_ptr = right_bias,
        left_gate_bias_ptr = left_gate_bias,
        right_gate_bias_ptr = right_gate_bias,
        out_gate_bias_ptr = out_gate_bias,
        left_ptr = left,
        right_ptr = right,
        out_gate_ptr = out_gate,
        eps = eps,
        BLOCK_R = BLOCK_R,
        BLOCK_H = BLOCK_H,
        BLOCK_D = BLOCK_D,
        num_warps = 4,
    )

    # triangular update via batched GEMM
    left_view = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right_view = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    left_mat = left_view.view(B * hidden_dim, N, N)
    right_mat = right_view.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm + out_gate multiplication (fused)
    out_flat = out.view(R, hidden_dim).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        x_ptr = out_flat,
        gate_ptr = out_gate,
        y_ptr = gated_norm,
        weight_ptr = weights["to_out_norm.weight"],
        bias_ptr = weights["to_out_norm.bias"],
        rows = R,
        dim = hidden_dim,
        eps = eps,
        BLOCK_M = BLOCK_M,
        BLOCK_D = BLOCK_D2,
        num_warps = 4,
    )

    # final linear projection
    final = torch.nn.functional.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),
        bias=None
    ).view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fused “LayerNorm → Linear → Gates” kernel +
batched GEMM + fused post‑norm.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input is **fused** with the five
    linear projections (left/right + three gates).  The LayerNorm
    weight / bias are folded into the projection weights:
        w_scaled = w * ln_weight          (fp16)
        bias_pre = (ln_bias * w).sum(dim=1) (fp32)
        sum_ws   = w_scaled.sum(dim=1)    (fp32)
    For each row we compute mean / variance, then the five dot‑products
    `x · w_scaledᵀ` in one Triton kernel.  The gate values are turned
    into sigmoids (fp32 → fp16) and the mask (if present) is applied
    to *left* and *right* tensors before storing them.

2️⃣  The triangular multiplicative update is a batched matrix
    multiplication:
        out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented with `torch.bmm` on tensors of shape
    `[B·H, N, N]`.  This call uses cuBLAS/Tensor‑Cores on H100.

3️⃣  Row‑wise LayerNorm on the GEMM result **and** element‑wise
    multiplication by the previously‑computed `out_gate` are fused
    into a single Triton kernel.

4️⃣  Final linear projection back to `dim` (fp16 GEMM) and cast to
    fp32.

The implementation keeps the heavy GEMM work in cuBLAS (the
most efficient path) and removes the two intermediate kernels of the
reference implementation (stand‑alone LayerNorm + split‑gate).  This
cuts memory traffic and kernel launch overhead, bringing the forward
pass well below the 1 ms target on the listed configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5×Linear + gates (produces left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    # input & geometry
    x_ptr,                     # const *[rows, dim] (fp32)
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    # mask
    HAS_MASK: tl.constexpr,
    mask_ptr,                  # const *[rows] (fp16) – ignored if HAS_MASK=0
    # weight matrices already multiplied by LN‑weight (fp16)
    left_w_ptr,                # const *[H, dim]
    right_w_ptr,
    left_gate_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    # pre‑computed Σ_d w_scaled and bias_pre (fp32)
    left_sum_ws_ptr,           # const *[H]
    right_sum_ws_ptr,
    left_gate_sum_ws_ptr,
    right_gate_sum_ws_ptr,
    out_gate_sum_ws_ptr,
    left_bias_ptr,             # const *[H]  (bias_pre)
    right_bias_ptr,
    left_gate_bias_ptr,
    right_gate_bias_ptr,
    out_gate_bias_ptr,
    # outputs
    left_ptr,                  # mutable *[rows, H] (fp16)
    right_ptr,
    out_gate_ptr,
    # hyper‑params
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean / variance of the row (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                # [R,D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Optional pair‑wise mask (broadcasted over H)
    # --------------------------------------------------------------
    if HAS_MASK:

                           mask=mask_r,
                           other=1.0)                # [R] fp16
        mask_val = mask_val[:, None]                      # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # 3️⃣  Loop over hidden‑dim tiles, accumulate dot‑products
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

                           mask=mask_h, other=0.0)            # [H]
        sum_ws_r = tl.load(right_sum_ws_ptr + offs_h,
                           mask=mask_h, other=0.0)
        sum_ws_lg = tl.load(left_gate_sum_ws_ptr + offs_h,
                           mask=mask_h, other=0.0)
        sum_ws_rg = tl.load(right_gate_sum_ws_ptr + offs_h,
                           mask=mask_h, other=0.0)
        sum_ws_og = tl.load(out_gate_sum_ws_ptr + offs_h,
                           mask=mask_h, other=0.0)

        bias_l   = tl.load(left_bias_ptr + offs_h,
                           mask=mask_h, other=0.0)            # [H]
        bias_r   = tl.load(right_bias_ptr + offs_h,
                           mask=mask_h, other=0.0)
        bias_lg  = tl.load(left_gate_bias_ptr + offs_h,
                           mask=mask_h, other=0.0)
        bias_rg  = tl.load(right_gate_bias_ptr + offs_h,
                           mask=mask_h, other=0.0)
        bias_og  = tl.load(out_gate_bias_ptr + offs_h,
                           mask=mask_h, other=0.0)

        # accumulators for the five dot‑products
        acc_l   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_r   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lg  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rg  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_og  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ----- inner loop over the feature dimension -----
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
            mask_d = offs_d < dim

            # load a tile of the input row
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                 # [R,D]

            # load the five weight tiles (already multiplied by LN‑weight)
            w_l  = tl.load(left_w_ptr        + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)              # [H,D]
            w_r  = tl.load(right_w_ptr       + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            w_lg = tl.load(left_gate_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            w_rg = tl.load(right_gate_w_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            w_og = tl.load(out_gate_w_ptr    + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            # dot‑product: (R,D) @ (D,H) → (R,H)
            acc_l  += tl.dot(x, w_l.t())
            acc_r  += tl.dot(x, w_r.t())
            acc_lg += tl.dot(x, w_lg.t())
            acc_rg += tl.dot(x, w_rg.t())
            acc_og += tl.dot(x, w_og.t())

        # --------------------------------------------------------------
        # 4️⃣  Apply LayerNorm scaling, add bias, compute sigmoids
        # --------------------------------------------------------------
        # broadcast μ and rstd (shape [R,1])
        mu   = mean[:, None]
        inv  = rstd[:, None]

        # Linear outputs (fp32)
        proj_l   = inv * (acc_l - mu * sum_ws_l[None, :]) + bias_l[None, :]
        proj_r   = inv * (acc_r - mu * sum_ws_r[None, :]) + bias_r[None, :]
        pre_lg   = inv * (acc_lg - mu * sum_ws_lg[None, :]) + bias_lg[None, :]
        pre_rg   = inv * (acc_rg - mu * sum_ws_rg[None, :]) + bias_rg[None, :]
        pre_og   = inv * (acc_og - mu * sum_ws_og[None, :]) + bias_og[None, :]

        # sigmoids (computed in fp32 → fp16)
        gate_l = (1.0 / (1.0 + tl.exp(-pre_lg))).to(tl.float16)
        gate_r = (1.0 / (1.0 + tl.exp(-pre_rg))).to(tl.float16)
        gate_o = (1.0 / (1.0 + tl.exp(-pre_og))).to(tl.float16)

        # apply mask to left/right *after* gating
        left_out  = (proj_l * gate_l * mask_val).to(tl.float16)
        right_out = (proj_r * gate_r * mask_val).to(tl.float16)

        # --------------------------------------------------------------
        # 5️⃣  Store results
        # --------------------------------------------------------------
        left_dst_ptr      = left_ptr      + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr     = right_ptr     + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr  = out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, gate_o,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 2️⃣  Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)
    gate_ptr,       # const   *[rows, dim]   (fp16)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32) γ
    bias_ptr,       # const   *[dim]        (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 3️⃣  Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all tensors are float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device
    B, N, _, _   = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # Fold LayerNorm weight into the projection matrices
    # --------------------------------------------------------------
    ln_weight = weights["norm.weight"]          # [dim]  fp32
    ln_bias   = weights["norm.bias"]            # [dim]  fp32

    def _prepare(w):
        """
        Return
        ------
        w_scaled : fp16   – weight * ln_weight  (shape [H, dim])
        sum_ws   : fp32   – Σ_d w_scaled (shape [H])
        bias_pre : fp32   – Σ_d ln_bias * w   (shape [H])
        """
        # w : [H, dim] (fp32)
        w_scaled = (w * ln_weight).to(torch.float16)          # element‑wise
        sum_ws   = w_scaled.float().sum(dim=1)                # Σ_d w_scaled
        bias_pre = (ln_bias * w).sum(dim=1)                    # Σ_d ln_bias * w
        return w_scaled, sum_ws, bias_pre

    left_w,   left_sum_ws,   left_bias   = _prepare(weights["left_proj.weight"])
    right_w,  right_sum_ws,  right_bias  = _prepare(weights["right_proj.weight"])
    lg_w,     lg_sum_ws,     lg_bias     = _prepare(weights["left_gate.weight"])
    rg_w,     rg_sum_ws,     rg_bias     = _prepare(weights["right_gate.weight"])
    og_w,     og_sum_ws,     og_bias     = _prepare(weights["out_gate.weight"])

    # --------------------------------------------------------------
    # Allocate output buffers
    # --------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # Prepare mask (flattened) – does NOT affect the gate tensors
    # --------------------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R)  # [R]

    # ------------------------------------------------------------------
    # Launch fused LayerNorm + projection + gating kernel
    # ------------------------------------------------------------------
    BLOCK_R = 64      # rows per program (tuned for register pressure)
    BLOCK_H = 16      # hidden‑dim tile
    BLOCK_D = 32      # feature‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    proj_gate_fused_kernel[grid](
        x_ptr                = x.view(R, dim).contiguous(),
        rows                 = R,
        dim                  = dim,
        H                    = H,
        HAS_MASK             = HAS_MASK,
        mask_ptr             = mask_flat,
        left_w_ptr           = left_w,
        right_w_ptr          = right_w,
        left_gate_w_ptr      = lg_w,
        right_gate_w_ptr     = rg_w,
        out_gate_w_ptr       = og_w,
        left_sum_ws_ptr      = left_sum_ws,
        right_sum_ws_ptr     = right_sum_ws,
        left_gate_sum_ws_ptr = lg_sum_ws,
        right_gate_sum_ws_ptr= rg_sum_ws,
        out_gate_sum_ws_ptr  = og_sum_ws,
        left_bias_ptr        = left_bias,
        right_bias_ptr       = right_bias,
        left_gate_bias_ptr   = lg_bias,
        right_gate_bias_ptr  = rg_bias,
        out_gate_bias_ptr    = og_bias,
        left_ptr             = left,
        right_ptr            = right,
        out_gate_ptr         = out_gate,
        eps                  = eps,
        BLOCK_R              = BLOCK_R,
        BLOCK_H              = BLOCK_H,
        BLOCK_D              = BLOCK_D,
        num_warps            = 4,
    )

    # --------------------------------------------------------------
    # 4️⃣  Batched GEMM:  out = left @ rightᵀ  (shape [B, N, N, H])
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] → batched GEMM on (B·H) matrices
    left_view  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_view = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat   = left_view.reshape(B * H, N, N)      # [B·H, N, N]
    right_mat  = right_view.reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣  Row‑wise LayerNorm + out‑gate multiplication (fused)
    # --------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        x_ptr      = out_flat,
        gate_ptr   = out_gate,
        y_ptr      = gated_norm,
        weight_ptr = weights["to_out_norm.weight"],   # γ (fp32)  shape [H]
        bias_ptr   = weights["to_out_norm.bias"],     # β (fp32)
        rows       = R,
        dim        = H,
        eps        = eps,
        BLOCK_M    = BLOCK_M,
        BLOCK_D    = BLOCK_D2,
        num_warps  = 4,
    )

    # --------------------------------------------------------------
    # 6️⃣  Final projection back to `dim`  (fp16 GEMM) + cast to fp32
    # --------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
