def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape

    # 1. First layernorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps).to(torch.float16)

    # 2. Fused projection GLU: combine 5 linear transforms.
    # weight concatenated shape [5*H, dim]
    w_concat = torch.cat([...], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, w_concat) # (B,N,N,5*H) as half
    # (R, 5*H) flatten
    R = B*N*N
    proj = proj.view(R, 5*hidden_dim)

    # 3. Split, mask, gates
    left = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_flat = mask.view(R).to(torch.float16)
        mask_ptr = mask_flat

    BLOCK_R = 256  # choose bigger for throughput
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # 4. Batched matmul
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    # matmul: (B*H,N,N) @ (B*H,N,N)^T
    out_mat = torch.matmul(left, right.transpose(-1, -2))  # half
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 5. Second layernorm + gate multiplication
    # layernorm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    # Multiply with out_gate (flattened)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out_fused = (out_norm * out_gate).to(torch.float16)  # keep half

    # 6. Final linear projection back to dim
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # [dim, hidden_dim]
    final_half = torch.nn.functional.linear(out_fused, to_out_weight)  # half
    final = final_half.to(torch.float32)

    return final
```

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    rows, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < rows

    # load mask value per row, broadcast over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]  # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # compute linear offsets for each of the 5 slices
        # stride between rows = 5*H, stride between cols = 1
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

                             mask=mask_r[:, None] & mask_h[None, :],

"""
Optimized outgoing TriMul (TriMul‑out) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim).  Implemented with
   torch.nn.functional.layer_norm (highly tuned in PyTorch) → fp16.

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM:   X_norm @ W_fusedᵀ   →   (R, 5·H)   where R = B·N·N.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * splits the fused projection into left, right and three gate tensors,
   * applies the optional pair‑wise mask,
   * evaluates sigmoid gates (fp32 → fp16),
   * writes three fp16 tensors: left, right, out_gate.

4️⃣ Batched FP16 GEMM over the hidden dimension:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented with a single torch.matmul on tensors shaped
   (B·H, N, N) → fp16 (Tensor‑Core).

5️⃣ Second LayerNorm + out‑gate multiplication.
   LayerNorm (torch) → fp16, then element‑wise multiply with out_gate.

6️⃣ Final fp16 linear projection back to `dim` and cast to float32.

Only step‑3 is custom Triton code; all dense GEMMs use cuBLAS/Tensor‑Core.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Kernel 3 (the only custom Triton kernel): split fused projection,
# apply (optional) mask, and compute sigmoid gates.
# ---------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per program
):
    """ (R,5H) → left, right, out_gate  with optional mask & sigmoid. """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ----------------------------------------------------------------
    # mask value per row (broadcast over the hidden dimension)
    # ----------------------------------------------------------------
    if HAS_MASK:
        # mask is stored as fp16 (0.0 / 1.0).  Load with a mask to avoid OOB.
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]                     # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----------------------------------------------------------------
    # iterate over hidden dimension blocks
    # ----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused matrix:
        #   [ left_proj | right_proj | left_gate | right_gate | out_gate ]
        # each slice has size H.

    # ------------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling -------------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        # flatten to (R,) and cast to fp16 (0.0 / 1.0)
        mask_flat = mask.view(R).to(torch.float16)
        mask_ptr = mask_flat

    # launch kernel -------------------------------------------------------
    BLOCK_R = 128          # rows per program (tuned for H100)
    BLOCK_H = 64           # hidden dim per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched FP16 GEMM over the hidden dimension (out = left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for a single batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)  (Tensor‑Core fp16)
    out_mat = torch.matmul(left, right.transpose(-1, -2))
    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate (both fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                             # fp16

    out_gate = out_gate.view(B, N, N, H)          # reshape to broadcast
    fused = (out_norm * out_gate).to(torch.float16)
    del out_norm, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` and cast to float32
    # ------------------------------------------------------------------
    final_half = F.linear(
        fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
