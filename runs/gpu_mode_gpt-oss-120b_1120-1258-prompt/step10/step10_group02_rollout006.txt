"""
High‑performance outgoing Tri‑Multiplicative update (TriMul) for AlphaFold3.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) using
    :func:`torch.nn.functional.layer_norm` – this avoids a custom kernel
    and benefits from cuDNN’s highly‑tuned implementation.

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single matmul:
        (B·N·N, dim) @ (5·H, dim)ᵀ  →  (B·N·N, 5·H)
    The weight matrix is concatenated and cast to fp16 so that Tensor‑Core
    GEMM is used.

3️⃣  Triton kernel ``split_fuse_gate_kernel``:
        * extracts the five 𝑯‑dimensional slices,
        * applies the optional pair‑wise mask,
        * evaluates the three sigmoid gates (in FP32 for stability),
        * writes three fp16 tensors: ``left``, ``right`` and ``out_gate``.

4️⃣  Batched GEMM over the hidden dimension:
        left[b·H+i] @ right[b·H+i]ᵀ    (fp16)
    Implemented with ``torch.bmm`` on a strided‑batched view → cuBLAS.

5️⃣  Triton kernel ``layernorm_mul_kernel`` fuses a second LayerNorm over
    the hidden dimension (fp16 → fp16) and the element‑wise multiplication
    by ``out_gate``.

6️⃣  Final linear projection back to the model dimension (fp16 → fp32).

All heavy linear algebra is delegated to cuBLAS/Tensor‑Core; only the
mask‑gate split and the second LayerNorm+gate multiplication are
implemented in Triton.  The implementation works for any of the test
configurations listed in the problem statement.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (unused if HAS_MASK==0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are fp16 (sigmoid evaluated in fp32 for numerical stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load mask if present (broadcast later)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)         # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the concatenated 5·H matrix
        row_off = offs_r[:, None] * (5 * H)                      # [R,1]
        col_off = offs_h[None, :]                                # [1,H]

        # left projection
        left_raw = tl.load(proj_ptr + row_off + col_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        # right projection
        right_raw = tl.load(proj_ptr + row_off + H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        # left gate
        lgate_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        # right gate
        rgate_raw = tl.load(proj_ptr + row_off + 3 * H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        # out gate
        ogate_raw = tl.load(proj_ptr + row_off + 4 * H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        lgate_f32 = lgate_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lgate_f32))).to(tl.float16)

        rgate_f32 = rgate_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rgate_f32))).to(tl.float16)

        ogate_f32 = ogate_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-ogate_f32))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # Store results
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + multiply by out_gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)
    y_ptr,            # mutable *[rows, dim]   (fp16)
    weight_ptr,       # const   *[dim]        (fp32)   γ
    bias_ptr,         # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean / variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply γ/β, multiply with gate, store (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load normalized input
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x-μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]        # fp32

        # γ, β (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * g
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with linear / layer‑norm parameters
                             (all tensors are float32, Linear layers have bias=False)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) – use cuDNN implementation
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # shape [B, N, N, dim], float32
    x_norm_h = x_norm.half()            # fp16 for the heavy GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection to obtain all five heads at once
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # shape [5·H, dim] (fp16)

    # (R, dim) @ (5·H, dim).T  →  (R, 5·H)
    proj = F.linear(x_norm_h, w_concat)  # bias=None
    del x_norm_h, w_concat

    # ------------------------------------------------------------------
    # 3️⃣ Split the concatenated projection, apply mask & sigmoid gates
    # ------------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # Determine whether we have a mask
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_half = mask.to(torch.float16).reshape(R)   # flatten
        mask_ptr = mask_half

    BLOCK_R = 128          # rows per program
    BLOCK_H = 32           # hidden dim chunk per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:
    #     out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → [B, H, N, N]
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # collapse batch & hidden dimension -> strided‑batched GEMM
    left_mat  = left .view(B * H, N, N)           # [B·H, N, N]
    right_mat = right.view(B * H, N, N)           # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16
    del left, right, left_mat, right_mat

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication by out_gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] fp16
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
