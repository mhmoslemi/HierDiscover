Pseudo:

```
@triton.jit
def final_proj_kernel(
    out_ptr,          # const *[rows, H] (fp16)
    gate_ptr,         # const *[rows, H] (fp16)
    weight_norm_ptr,  # const *[H] (fp32)
    bias_norm_ptr,    # const *[H] (fp32)
    w_out_ptr,        # const *[dim, H] (fp16)
    out_final_ptr,    # mutable *[rows, dim] (fp16)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # compute mean, var for each row across H

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # loop over H in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean

"""
TriMul (outgoing) – fast Triton‑only forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) – Triton.
2️⃣ Fuse the five linear maps (left/right projection + three gates) into one
   GEMM:  X_norm @ W_fused.T  → (R, 5·H)  (R = B·N·N).
3️⃣ Split the fused result, apply the optional 2‑D mask, evaluate the sigmoid
   gates and write three tensors (left, right, out_gate) – Triton.
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i].T
   (uses torch.bmm / cuBLAS for maximal Tensor‑Core throughput).
5️⃣ Fuse the second LayerNorm, the gate multiplication and the final
   linear projection into a single Triton kernel:
        out_norm = LayerNorm(out)            (γ/β = to_out_norm)
        out_gate  = out_norm * out_gate
        out_final = out_gate @ to_out.T       (to_out : [dim, H])
   This eliminates the extra write‑read of the intermediate `(R, H)` tensor.
6️⃣ Reshape to `[B, N, N, dim]` and cast back to FP32.

All heavy compute (steps 1, 3, 5) is performed by custom Triton kernels,
while the huge `N³·H` contraction of step 4 uses cuBLAS (torch.bmm) which
already saturates the H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]      (float32)
    y_ptr,          # mutable *[rows, dim]      (float16)
    weight_ptr,     # const   *[dim]           (float32)   γ
    bias_ptr,       # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)              # [BLOCK_R]
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout per row: [L, R, L_gate, R_gate, O_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # offset into proj

        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw     = tl.load(left_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(right_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw = tl.load(left_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw = tl.load(right_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (fp32 for stability, then back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # const   *[rows, H]      (float16)
    gate_ptr,         # const   *[rows, H]      (float16)
    norm_w_ptr,       # const   *[H]           (float32)   (γ of to_out_norm)
    norm_b_ptr,       # const   *[H]           (float32)   (β of to_out_norm)
    proj_w_ptr,       # const   *[dim, H]      (float16)   (to_out weight)
    out_ptr_final,    # mutable *[rows, dim]   (float16)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- mean & variance of `out` (fp32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- produce final output block‑wise ----------
    for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)      # [BLOCK_O]
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
            mask_h = offs_h < H

            # load `out` and `gate`
            out_vals  = tl.load(out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)

            # LayerNorm scaling (γ/β)
            w_n = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            b_n = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

            # normalize + γ/β
            normed = (out_vals - mean[:, None]) * rstd[:, None]   # (row, H)
            normed = normed * w_n[None, :] + b_n[None, :]

            # apply out‑gate
            normed = normed * gate_vals

            # load slice of final projection weight: shape [BLOCK_O, BLOCK_H]
            w_proj = tl.load(proj_w_ptr + offs_o[:, None] * H + offs_h[None, :],
                             mask=mask_o[:, None] & mask_h[None, :],
                             other=0.0)                               # fp16
            w_proj_f32 = w_proj.to(tl.float32)

            # acc += normed @ w_proj.T   → (BLOCK_R, BLOCK_O)
            acc += tl.dot(normed, w_proj_f32, trans_b=True)

        # store the result
        out_ptr_store = out_ptr_final + offs_r[:, None] * dim + offs_o[None, :]
        tl.store(out_ptr_store, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])

# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : contains the model weights (all float32)
        * config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (float32)
        weights["norm.bias"],         # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: left/right + 3 gates → (R, 5·H)
    # --------------------------------------------------------------
    # concatenate weight matrices (out_features, in_features); cast to fp16
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # (R, dim) @ (5·H, dim)^T → (R, 5·H)
    proj = F.linear(x_norm, w_fused)                     # fp16
    del w_fused, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        # no mask – compile‑time constant disables loading
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        # mask ∈ {0,1}; broadcast to rows and convert to fp16
        mask_flat = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (half‑precision)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N]  (contiguous for cuBLAS)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # out_mat[i] = left_mat[i] @ right_mat[i].T   → [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    out_flat      = out.view(R, H).contiguous()                # (R, H)
    out_gate_flat = out_gate.view(R, H).contiguous()          # (R, H)
    final_flat    = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Cast the final linear weight once
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16).contiguous()

    BLOCK_R = 128
    BLOCK_H = 32
    BLOCK_O = 64
    grid_final = (triton.cdiv(R, BLOCK_R),)
    final_fused_kernel[grid_final](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        to_out_weight_fp16,              # [dim, H] (fp16)
        final_flat,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out, out_gate, to_out_weight_fp16

    # --------------------------------------------------------------
    # 6️⃣ Reshape to final shape and cast to FP32
    # --------------------------------------------------------------
    final = final_flat.view(B, N, N, dim).to(torch.float32)
    return final
