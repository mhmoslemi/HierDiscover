"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input `[B,N,N,dim]` → fp16.
2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   GEMM (`x_norm @ W_fused.T`).  The weight matrix is concatenated along the
   output dimension → shape `[5*H, dim]`.
3️⃣ Triton kernel splits the fused tensor into `left`, `right` and the
   `out_gate`, applies the optional pair‑wise mask and sigmoid gates.
4️⃣ Batched GEMM (`torch.bmm`) computes the tri‑multiplicative update:
   `(B·H,N,N) @ (B·H,N,N).T` → `(B·H,N,N)`.
5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused) on the GEMM result.
6️⃣ Final linear projection back to `dim` (fp16 GEMM) and cast to fp32.

Only the two LayerNorm kernels and the split‑gate kernel are implemented in
Triton; the heavy matrix multiplications use cuBLAS via PyTorch which already
exploits Tensor‑Cores on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values: y = γ·(x‑μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load optional pair‑wise mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five contiguous slices: [L,R,Lg,Rg,Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # ---- sigmoid (computed in fp32 for numeric stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask & gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store results ----
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm  (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # dim tile
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (produces [R, 5*H] in fp16)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [R, 5*H] fp16
    del w_fused, x_norm                     # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask values are 0/1 → float16, flatten to [R]
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim tile
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B,N,N,H] → bring H into batch dimension
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N).T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) + cast to fp32
    # --------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

**Explanation of key design choices**

* **Row‑wise LayerNorm kernels** compute mean/variance in FP32 for numerical stability and write the normalized result directly in FP16 to reduce memory traffic.
* **Fusion of five linear layers** into a single GEMM (`x_norm @ W_fused.T`) removes three separate kernel launches and leverages cuBLAS/Tensor‑Cores.
* **`split_fuse_gate_kernel`** simultaneously extracts `left`, `right` and `out_gate`, applies the optional pair‑wise mask and computes the three sigmoid gates, keeping all work in registers and shared memory.
* **Tri‑multiplicative update** is expressed as a batched matrix multiplication (`torch.bmm`). By reshaping to `[B·H, N, N]` we let cuBLAS handle the O(N³) work efficiently.
* **`layernorm_gate_kernel`** fuses the second LayerNorm with the element‑wise multiplication by the `out_gate`, again using FP32 accumulators for the statistics.
* The final projection back to `dim` uses a standard FP16 GEMM and the result is cast back to FP32, matching the original PyTorch module’s output dtype.

The kernels are written for Triton 3.3.1 and are tuned (block sizes, 8 warps) for the H100, striking a balance between occupancy and shared‑memory usage while staying well within the problem constraints.
