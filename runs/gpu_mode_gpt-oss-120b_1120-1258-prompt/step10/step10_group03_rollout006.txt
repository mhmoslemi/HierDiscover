        BLOCK_H=BLOCK_H,
        num_warps=4
    )
    # free proj
    del proj

    # 4. Tri-mul update
    # left/right shape [B,N,N,hidden_dim]
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

"""
TriMul outgoing update (forward only) – fast Triton + torch implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (B,N,N,dim) using the built‑in PyTorch LayerNorm
   (highly‑optimised on H100).  Result is cast to FP16.

2️⃣ Fuse the five linear maps (left‑proj, right‑proj, left‑gate, right‑gate,
   out‑gate) into a single GEMM:
        X_norm @ W_fusedᵀ → (R,5·H)   where R = B·N·N, H = hidden_dim.
   The weight matrix is pre‑concatenated and stored as FP16.

3️⃣ One custom Triton kernel (`split_fuse_gate_kernel`) splits the fused
   projection, applies the optional pair‑wise mask, evaluates three sigmoid
   gates (in FP32 for stability) and writes three FP16 tensors:
        left  = (proj_left  * mask) * σ(proj_left_gate)
        right = (proj_right * mask) * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)

4️⃣ Tri‑multiplicative update – batched GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   Implemented with `torch.bmm` (cuBLAS tensor‑core kernel).

5️⃣ Second LayerNorm on the hidden‑dim tensor (torch.nn.functional.layer_norm)
   followed by element‑wise multiplication with `out_gate`.

6️⃣ Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only the split‑/gate kernel is written in Triton, satisfying the requirement
to implement a custom kernel while delegating the heavy matrix‑multiplies to
cuBLAS for peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: split fused projection, mask, sigmoid gates
# ---------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic is in fp16, sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask if present, broadcast later
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate] each H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # offsets in proj

        # pointers to the five slices
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # load raw values (fp16)
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # store results
        left_dst_ptr     = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ---------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors; Linear layers have bias=False)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    nomask      = cfg.get("nomask", True)
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                        # total number of pairwise rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use highly‑optimised torch implementation)
    # ------------------------------------------------------------------
    # Normalise over the last dimension and cast to FP16 for the GEMM.
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                       # [B, N, N, dim] FP16

    # Flatten to (R, dim) for the fused linear projection.
    x_norm_flat = x_norm.view(R, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates (5·H outputs)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, dim)

    # (R, dim) @ (5·H, dim)ᵀ  →  (R, 5·H)  (FP16 GEMM via cuBLAS)
    proj = F.linear(x_norm_flat, w_fused)       # FP16
    del w_fused, x_norm, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        # No mask – compile‑time flag disables mask loading.
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
        mask_ptr = dummy_mask
    else:
        # mask shape [B, N, N] → [R]; cast to FP16 (0 / 1 values)
        mask_half = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R = 128                 # rows per program
    BLOCK_H = 32                  # hidden‑dim chunk
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] then bring H into the batch axis for bmm.
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # out_mat[b·H] = left_mat[b·H] @ right_mat[b·H]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Back‑to [B, N, N, H] (FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # ------------------------------------------------------------------
    # Reshape out_gate to match out.
    out_gate_reshaped = out_gate.view(B, N, N, H).contiguous()
    # LayerNorm on the hidden dimension (float32 for numeric stability)
    out_float = out.float()
    out_norm = torch.nn.functional.layer_norm(
        out_float,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm_half = out_norm.half()

    # Element‑wise multiplication with out_gate (FP16)
    out_fused = out_norm_half * out_gate_reshaped
    del out, out_float, out_norm, out_norm_half, out_gate, out_gate_reshaped

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (FP16 GEMM) → FP32 output
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused.view(R, H),                     # (R, H)
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )                                            # (R, dim) FP16

    final = final_half.view(B, N, N, dim).float()
    return final
