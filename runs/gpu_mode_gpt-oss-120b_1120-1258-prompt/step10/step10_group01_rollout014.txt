Pseudo steps:

```
pid = tl.program_id(0)
row_start = pid*BLOCK_R
offs_r = row_start + tl.arange(0, BLOCK_R)
mask_r = offs_r < R

# First pass: mean/var
sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

for d_off ...:
    offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim
    x = tl.load(x_ptr + offs_r[:, None]*dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
    sum_val += tl.sum(x, axis=1)
    sum_sq += tl.sum(x * x, axis=1)

mean = sum_val / dim
var = sum_sq / dim - mean * mean
rstd = tl.rsqrt(var + eps)

# Optional mask value retrieval
if HAS_MASK:
    mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
    mask_val = mask_val[:, None]  # broadcast

# Output accumulators for left,right,gates

acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
acc_left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
acc_right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
acc_out_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

# Loop over hidden blocks
for h_off ...:
    offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # initialize block accumulators for each projection (set to zero each hidden block)
    acc_left_block = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right_block = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_left_gate_block = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right_gate_block = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_out_gate_block = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # Loop over d for GEMM accumulate
    for d_off ...:
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # Load normalized x tile
        x = tl.load(x_ptr + offs_r[:, None]*dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        # Normalize
        x_norm = (x - mean[:, None]) * rstd[:, None]   # fp32

        # Load weight tiles
        w_left = tl.load(w_left_ptr + offs_d[:, None]*H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        w_right = tl.load(w_right_ptr + offs_d[:, None]*H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        w_left_gate = tl.load(w_left_gate_ptr + offs_d[:, None]*H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        w_right_gate = tl.load(w_right_gate_ptr + offs_d[:, None]*H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        w_out_gate = tl.load(w_out_gate_ptr + offs_d[:, None]*H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # accumulate dot products
        acc_left_block += tl.dot(x_norm, w_left)
        acc_right_block += tl.dot(x_norm, w_right)
        acc_left_gate_block += tl.dot(x_norm, w_left_gate)
        acc_right_gate_block += tl.dot(x_norm, w_right_gate)
        acc_out_gate_block += tl.dot(x_norm, w_out_gate)

    # after inner loop over d, we have raw projection values for this hidden block
    # apply mask (if present) and sigmoid gates
    left_gate = (1.0 / (1.0 + tl.exp(-acc_left_gate_block))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-acc_right_gate_block))).to(tl.float16)
    out_gate_val = (1.0 / (1.0 + tl.exp(-acc_out_gate_block))).to(tl.float16)
    # apply mask
    if HAS_MASK:
        left_raw = acc_left_block.to(tl.float16) * mask_val
        right_raw = acc_right_block.to(tl.float16) * mask_val
    else:
        left_raw = acc_left_block.to(tl.float16)
        right_raw = acc_right_block.to(tl.float16)

    # Apply gates
    left = left_raw * left_gate
    right = right_raw * right_gate

    # Store left, right, out_gate for current block of hidden dims
    left_ptr_tile = left_ptr + offs_r[:, None] * H + offs_h[None, :]
    right_ptr_tile = right_ptr + offs_r[:, None] * H + offs_h[None, :]
    out_gate_ptr_tile = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

    tl.store(left_ptr_tile, left, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr_tile, right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr_tile, out_gate_val, mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fused layer‑norm, projection, gating and final projection.

The forward pass consists of three main steps:
1️⃣ Fuse the first LayerNorm (over the input channel “dim”) with the five linear
   maps (left/right projections + three gates).  The kernel computes the row‑wise
   mean/variance, normalises the input and simultaneously performs a small GEMM
   (`x_norm @ Wᵀ`) for the five weight matrices.  The optional pair‑wise mask is
   applied and the sigmoid gates are evaluated (fp32 for stability).  The kernel
   writes three tensors of shape [R, H] (left, right, out_gate), where
   R = B·N·N and H = hidden_dim.

2️⃣ Batched matrix‑multiplication over the hidden dimension:
   `out = left @ rightᵀ`   (shape [B·H, N, N]).
   This heavy O(H·N³) work is performed by cuBLAS (`torch.bmm`), which on an H100
   reaches near‑peak Tensor‑Core throughput.

3️⃣ Fuse the second LayerNorm (over the hidden dimension), the element‑wise
   multiplication with `out_gate` and the final linear projection back to `dim`.
   The kernel computes the row‑wise mean/variance of the GEMM result,
   normalises it, applies the learned γ/β, multiplies by `out_gate` and
   finally performs the hidden‑→‑dim GEMM in‑kernel, producing the final
   float32 tensor of shape [B, N, N, dim].

All heavy linear algebra is executed in fp16 (Tensor‑Core) while reductions
and sigmoid are kept in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 0 – first LayerNorm + five projections + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,               # const  *[R, dim]      fp32 (input)
    mask_ptr,            # const  *[R]           fp16 (optional)
    left_ptr,            # mutable*[R, H]       fp16
    right_ptr,           # mutable*[R, H]       fp16
    out_gate_ptr,        # mutable*[R, H]       fp16
    w_left_ptr,          # const  *[dim, H]     fp16
    w_right_ptr,         # const  *[dim, H]     fp16
    w_left_gate_ptr,     # const  *[dim, H]     fp16
    w_right_gate_ptr,    # const  *[dim, H]     fp16
    w_out_gate_ptr,      # const  *[dim, H]     fp16
    R, dim, H,           # shape parameters
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 1) Row‑wise mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d = d_off * BLOCK_D
        offs_d = d + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + 1e-5)

    # ------------------------------------------------------------------
    # 2) Optional mask (scalar per row)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 3) Loop over hidden‑dim blocks and accumulate the five GEMMs
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h = h_off * BLOCK_H
        offs_h = h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for this hidden block (fp32)
        acc_left       = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_left_gate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            d = d_off * BLOCK_D
            offs_d = d + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # ------------ normalised input tile (fp16) ------------
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
            x_norm = ((x - mean[:, None]) * rstd[:, None]).to(tl.float16)   # fp16

            # ------------ weight tiles (fp16) ----------------------
            w_left       = tl.load(w_left_ptr       + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :], other=0.0)
            w_right      = tl.load(w_right_ptr      + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :], other=0.0)
            w_left_gate  = tl.load(w_left_gate_ptr  + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :], other=0.0)
            w_right_gate = tl.load(w_right_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :], other=0.0)
            w_out_gate   = tl.load(w_out_gate_ptr   + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :], other=0.0)

            # ------------ GEMM accumulation (fp32) ----------------
            acc_left       += tl.dot(x_norm, w_left)
            acc_right      += tl.dot(x_norm, w_right)
            acc_left_gate  += tl.dot(x_norm, w_left_gate)
            acc_right_gate += tl.dot(x_norm, w_right_gate)
            acc_out_gate   += tl.dot(x_norm, w_out_gate)

        # ------------ sigmoid (stable fp32) -----------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        # ------------ apply mask & gates -------------------------
        left  = acc_left.to(tl.float16) * mask_val * left_gate
        right = acc_right.to(tl.float16) * mask_val * right_gate

        # ------------ write results -------------------------------
        left_ptr_tile  = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_ptr_tile = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_ptr_tile   = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_ptr_tile,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr_tile, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_ptr_tile,   out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 1 – second LayerNorm + out_gate * final linear projection
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_proj_kernel(
    x_ptr,          # const  *[R, H]        fp16  (output of the batched GEMM)
    gate_ptr,       # const  *[R, H]        fp16  (out_gate)
    w_ptr,          # const  *[H, dim]      fp16  (to_out.weight.T)
    gamma_ptr,      # const  *[H]           fp32  (to_out_norm.weight)
    beta_ptr,       # const  *[H]           fp32  (to_out_norm.bias)
    y_ptr,          # mutable*[R, dim]      fp16  (final output)
    R, H, dim,      # shape parameters
    eps: tl.float32 = 1e-5,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 1) Compute mean & variance over the hidden dim (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h = h_off * BLOCK_H
        offs_h = h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        xv = tl.load(x_ptr + offs_r[:, None] * H + offs_h[None, :],
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0).to(tl.float32)
        sum_val += tl.sum(xv, axis=1)
        sum_sq  += tl.sum(xv * xv, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Produce final output: (LN(x) * gamma + beta) * gate   @   W
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d = d_off * BLOCK_D
        offs_d = d + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # accumulator for this dim‑tile (fp32)
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            h = h_off * BLOCK_H
            offs_h = h + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load hidden slice and gate slice
            xv   = tl.load(x_ptr   + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

            # layer‑norm affine transform
            xv_norm = (xv - mean[:, None]) * rstd[:, None]               # fp32
            gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
            beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)  # fp32
            xv_scaled = xv_norm * gamma[None, :] + beta[None, :]          # fp32

            # apply out‑gate
            xv_gated = xv_scaled * gate                                   # fp32

            # weight slice for this hidden block and output‑dim block
            w = tl.load(w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                         # (BLOCK_H, BLOCK_D)

            # accumulate matmul
            acc += tl.dot(xv_gated, w)                                    # (BLOCK_R, BLOCK_D)
        # store the final fp16 result
        y_ptr_tile = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(y_ptr_tile, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping of all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total rows (B·N·N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # Prepare weight tensors (transpose to [dim, hidden] for GEMM)
    # ------------------------------------------------------------------
    w_left_T      = w["left_proj.weight"].to(torch.float16).t().contiguous()
    w_right_T     = w["right_proj.weight"].to(torch.float16).t().contiguous()
    w_left_gate_T = w["left_gate.weight"].to(torch.float16).t().contiguous()
    w_right_gate_T= w["right_gate.weight"].to(torch.float16).t().contiguous()
    w_out_gate_T  = w["out_gate.weight"].to(torch.float16).t().contiguous()
    w_to_out_T    = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if cfg.get("nomask", True):
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)   # dummy
        has_mask  = 0
    else:
        mask_float = mask.to(torch.float16).reshape(R)   # [R]
        mask_flat = mask_float
        has_mask  = 1

    # ------------------------------------------------------------------
    # Allocate output tensors for the first stage
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Stage 0 – fused LayerNorm + five projections + mask + sigmoids
    # ------------------------------------------------------------------
    BLOCK_R = 64        # rows per program (tuned for H100)
    BLOCK_H = 32        # hidden‑dim chunk
    BLOCK_D = 32        # input‑dim chunk
    grid0 = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid0](
        x,
        mask_flat,
        left,
        right,
        out_gate,
        w_left_T,
        w_right_T,
        w_left_gate_T,
        w_right_gate_T,
        w_out_gate_T,
        R, dim, H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # Stage 1 – batched GEMM over hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] (contiguous) for torch.bmm
    left_4d  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_4d = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_4d.view(B * H, N, N)     # [B·H, N, N]
    right_mat = right_4d.view(B * H, N, N)    # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  fp16
    del left_mat, right_mat, left_4d, right_4d, left, right   # free memory

    # reshape back to [R, H] for the next kernel
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)  # fp16
    del out_mat

    # ------------------------------------------------------------------
    # Stage 2 – fused second LayerNorm + out_gate * final linear projection
    # ------------------------------------------------------------------
    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R2 = 64

    BLOCK_D2 = 64    # output‑dim chunk (fits nicely for dim up to 384)
    grid1 = (triton.cdiv(R, BLOCK_R2),)

    ln_gate_proj_kernel[grid1](
        out_flat,

        w["to_out_norm.bias"],
        final_half,
        R, H, dim,
        eps=eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate

    # ------------------------------------------------------------------
    # Reshape to the expected output layout and cast to float32
    # ------------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
