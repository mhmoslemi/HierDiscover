@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr,      # const *[rows, dim] float32
    left_ptr,   # mutable *[rows, H] float16
    right_ptr,  # mutable *[rows, H] float16
    out_gate_ptr, # mutable *[rows, H] float16
    weight_l_ptr, # const *[H, dim] float16
    weight_r_ptr,
    weight_lg_ptr,
    weight_rg_ptr,
    weight_og_ptr,
    eps: tl.float32,
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    mask_ptr: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Compute mean & var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k*BLOCK_K + tl.arange(0,BLOCK_K)
        mask_k = offs_k < dim
        x = tl.load(x_ptr + offs_r[:,None]*dim + offs_k[None,:],
                    mask=mask_r[:,None] & mask_k[None,:],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulators for each branch
    acc_l = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_r = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_og = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # For each tile of K compute normalized slice and dot with weight slices
    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k*BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim
        # load x slice and normalize
        x = tl.load(x_ptr + offs_r[:,None]*dim + offs_k[None,:],
                    mask=mask_r[:,None] & mask_k[None,:],
                    other=0.0).to(tl.float32)
        x_norm = (x - mean[:,None]) * rstd[:,None]

        # weight slices: shape [BLOCK_K, BLOCK_H]
        w_l = tl.load(weight_l_ptr + offs_k[:,None]*H + tl.arange(0, BLOCK_H)[None,:],
                      mask=mask_k[:,None] & (tl.arange(0,BLOCK_H) < H), other=0.0).to(tl.float32)
        w_r = tl.load(weight_r_ptr + offs_k[:,None]*H + tl.arange(0, BLOCK_H)[None,:],
                      mask=mask_k[:,None] & (tl.arange(0,BLOCK_H) < H), other=0.0).to(tl.float32)
        w_lg = tl.load(weight_lg_ptr + offs_k[:,None]*H + tl.arange(0, BLOCK_H)[None,:],
                      mask=mask_k[:,None] & (tl.arange(0,BLOCK_H) < H), other=0.0).to(tl.float32)
        w_rg = tl.load(weight_rg_ptr + offs_k[:,None]*H + tl.arange(0, BLOCK_H)[None,:],
                      mask=mask_k[:,None] & (tl.arange(0,BLOCK_H) < H), other=0.0).to(tl.float32)
        w_og = tl.load(weight_og_ptr + offs_k[:,None]*H + tl.arange(0, BLOCK_H)[None,:],
                      mask=mask_k[:,None] & (tl.arange(0,BLOCK_H) < H), other=0.0).to(tl.float32)

        # accumulate
        acc_l += tl.dot(x_norm, w_l)   # (BLOCK_R, BLOCK_H)
        acc_r += tl.dot(x_norm, w_r)
        acc_lg += tl.dot(x_norm, w_lg)
        acc_rg += tl.dot(x_norm, w_rg)
        acc_og += tl.dot(x_norm, w_og)

    # After loop, apply sigmoid on gate accumulators
    # convert to fp16 for storage later
    sig_lg = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)

@triton.jit
def fused_ln_proj_gate_kernel(
    x_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    weight_l_ptr,
    weight_r_ptr,
    weight_lg_ptr,
    weight_rg_ptr,
    weight_og_ptr,
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    mask_ptr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Compute mean+var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k*BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_k[None, :],
                    mask=mask_r[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # optional mask broadcast
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)[:, None]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # accumulators
    acc_l = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_r = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_og = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    H_range = tl.arange(0, BLOCK_H)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k*BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim
        # load x slice
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_k[None, :],
                    mask=mask_r[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        x_norm = (x - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_K]

        # load weight slices
        # weight layout: [dim, H]
        w_l = tl.load(weight_l_ptr + offs_k[:, None] * H + H_range[None, :],
                      mask=mask_k[:, None] & (H_range[None, :] < H), other=0.0).to(tl.float32)
        w_r = tl.load(weight_r_ptr + offs_k[:, None] * H + H_range[None, :],
                      mask=mask_k[:, None] & (H_range[None, :] < H), other=0.0).to(tl.float32)
        w_lg = tl.load(weight_lg_ptr + offs_k[:, None] * H + H_range[None, :],
                       mask=mask_k[:, None] & (H_range[None, :] < H), other=0.0).to(tl.float32)
        w_rg = tl.load(weight_rg_ptr + offs_k[:, None] * H + H_range[None, :],
                       mask=mask_k[:, None] & (H_range[None, :] < H), other=0.0).to(tl.float32)
        w_og = tl.load(weight_og_ptr + offs_k[:, None] * H + H_range[None, :],
                       mask=mask_k[:, None] & (H_range[None, :] < H), other=0.0).to(tl.float32)

        acc_l += tl.dot(x_norm, w_l)   # (BLOCK_R, BLOCK_H)
        acc_r += tl.dot(x_norm, w_r)
        acc_lg += tl.dot(x_norm, w_lg)
        acc_rg += tl.dot(x_norm, w_rg)
        acc_og += tl.dot(x_norm, w_og)

    # apply sigmoid to gate accumulators (convert to fp16)
    left_gate = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)

    # apply mask and gates to left/right (left/right raw are fp16)
    left_raw = acc_l.to(tl.float16)
    right_raw = acc_r.to(tl.float16)

    left = left_raw * m * left_gate
    right = right_raw * m * right_gate

    # store
    offs_h = tl.arange(0, BLOCK_H)
    valid_h = offs_h < H

    left_ptrs = left_ptr + offs_r[:, None] * H + offs_h[None, :]
    right_ptrs = right_ptr + offs_r[:, None] * H + offs_h[None, :]
    out_gate_ptrs = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

    tl.store(left_ptrs, left, mask=mask_r[:, None] & valid_h[None, :])
    tl.store(right_ptrs, right, mask=mask_r[:, None] & valid_h[None, :])
    tl.store(out_gate_ptrs, out_gate, mask=mask_r[:, None] & valid_h[None, :])
```

    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; eps=1e-5; device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # Flatten input
    x_flat = x.view(R, dim).contiguous()  # float32

    # Prepare weights (transpose to [dim, H] for each)
    left_w_T = weights["left_proj.weight"].t().to(torch.float16).contiguous()
    right_w_T = weights["right_proj.weight"].t().to(torch.float16).contiguous()
    left_gate_w_T = weights["left_gate.weight"].t().to(torch.float16).contiguous()

    out_gate_w_T = weights["out_gate.weight"].t().to(torch.float16).contiguous()

    # allocate output buffers
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask flatten
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_flat

    # launch kernel
    BLOCK_R = 128
    BLOCK_H = 32
    BLOCK_K = 32
    grid = (triton.cdiv(R, BLOCK_R),)

        right_gate_w_T,
        out_gate_w_T,

        dim,
        H,
        eps,
        mask_ptr,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Tri-multiplicative update
    # reshape and permute
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # final linear
    final_half = F.linear(gated_norm,
                          weights["to_out.weight"].t().to(torch.float16), bias=None)
    final = final_half.view(B, N, N, dim).to(torch.float32)

"""
OutgoingTriMul – Triton‐accelerated forward pass
------------------------------------------------
1️⃣  Row‑wise LayerNorm (float32 → float16) **and** the five linear
    projections (left/right + three gates) are fused into a single
    Triton kernel.  The kernel:
        • computes per‑row mean/variance,
        • normalises the input,
        • performs 5 × (dim × hidden) matmul tiles,
        • applies the optional pair‑wise mask,
        • applies sigmoid on the three gate tensors.
    Output → left, right (both multiplied by their gates) and out_gate.

2️⃣  Tri‑multiplicative update:   left @ rightᵀ   (batched half‑precision
    GEMM, cuBLAS/Tensor‑Cores).

3️⃣  Row‑wise LayerNorm on the GEMM result + out‑gate multiplication
    (fused in a second Triton kernel).

4️⃣  Final linear projection back to ``dim`` (half‑precision GEMM) and
    cast to ``float32``.

Only two Triton kernels are launched, dramatically reducing memory traffic
and kernel‑launch overhead compared with a naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5 linear projections + gating (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    # inputs
    x_ptr,                    # const  *[rows, dim]      (float32)
    # outputs
    left_ptr,                 # mutable *[rows, H]       (float16)
    right_ptr,                # mutable *[rows, H]       (float16)
    out_gate_ptr,             # mutable *[rows, H]       (float16)
    # weight matrices (already transposed: [dim, H])
    w_l_ptr,                  # const  *[dim, H] (float16)
    w_r_ptr,                  # const  *[dim, H] (float16)
    w_lg_ptr,                 # const  *[dim, H] (float16)
    w_rg_ptr,                 # const  *[dim, H] (float16)
    w_og_ptr,                 # const  *[dim, H] (float16)
    # scalars
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    # optional mask (flattened, 0/1)
    mask_ptr,                 # const  *[rows] (float16)
    HAS_MASK: tl.constexpr,
    # compile‑time tile sizes
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32) for LayerNorm
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < dim
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [R,K]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Load optional pair‑wise mask (broadcasted over H)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)[:, None]  # [R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # 3️⃣  Accumulate the five matmul results in fp32
    # --------------------------------------------------------------
    acc_l  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_r  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_og = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    h_range = tl.arange(0, BLOCK_H)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < dim

        # load and normalise a tile of the input
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [R,K]
        x_norm = (x - mean[:, None]) * rstd[:, None]   # [R,K]

        # load the corresponding weight tiles (layout: [dim, H])
        wl = tl.load(
            w_l_ptr + offs_k[:, None] * H + h_range[None, :],
            mask=mask_k[:, None] & (h_range[None, :] < H),
            other=0.0,
        ).to(tl.float32)                               # [K,H]
        wr = tl.load(
            w_r_ptr + offs_k[:, None] * H + h_range[None, :],
            mask=mask_k[:, None] & (h_range[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        wlg = tl.load(
            w_lg_ptr + offs_k[:, None] * H + h_range[None, :],
            mask=mask_k[:, None] & (h_range[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        wrg = tl.load(
            w_rg_ptr + offs_k[:, None] * H + h_range[None, :],
            mask=mask_k[:, None] & (h_range[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        wog = tl.load(
            w_og_ptr + offs_k[:, None] * H + h_range[None, :],
            mask=mask_k[:, None] & (h_range[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        # GEMM tiles
        acc_l  += tl.dot(x_norm, wl)    # (R,H)
        acc_r  += tl.dot(x_norm, wr)
        acc_lg += tl.dot(x_norm, wlg)
        acc_rg += tl.dot(x_norm, wrg)
        acc_og += tl.dot(x_norm, wog)

    # --------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates (fp32 → fp16) and mask
    # --------------------------------------------------------------
    left_gate  = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)

    left  = acc_l.to(tl.float16) * m * left_gate
    right = acc_r.to(tl.float16) * m * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Store results
    # --------------------------------------------------------------
    valid_h = h_range < H

    left_ptrs  = left_ptr  + offs_r[:, None] * H + h_range[None, :]
    right_ptrs = right_ptr + offs_r[:, None] * H + h_range[None, :]
    og_ptrs    = out_gate_ptr + offs_r[:, None] * H + h_range[None, :]

    tl.store(left_ptrs, left,
             mask=mask_r[:, None] & valid_h[None, :])
    tl.store(right_ptrs, right,
             mask=mask_r[:, None] & valid_h[None, :])
    tl.store(og_ptrs, out_gate,
             mask=mask_r[:, None] & valid_h[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        g_vals = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            y_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing the same keys as the reference
                         PyTorch implementation (all float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + 5 linear projections + gates
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                          # [R, dim] FP32

    # Transpose & convert weights once (shape: [dim, H])
    w_l_T  = weights["left_proj.weight"].t().to(torch.float16).contiguous()
    w_r_T  = weights["right_proj.weight"].t().to(torch.float16).contiguous()
    w_lg_T = weights["left_gate.weight"].t().to(torch.float16).contiguous()
    w_rg_T = weights["right_gate.weight"].t().to(torch.float16).contiguous()
    w_og_T = weights["out_gate.weight"].t().to(torch.float16).contiguous()

    # Allocate output buffers
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # Optional pair‑wise mask (flattened, FP16)
    if cfg.get("nomask", True):
        has_mask = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1

    # Kernel launch
    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim tile
    BLOCK_K = 32    # input dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    fused_ln_proj_gate_kernel[grid](
        x_flat,
        left,
        right,
        out_gate,
        w_l_T,
        w_r_T,
        w_lg_T,
        w_rg_T,
        w_og_T,
        R,
        dim,
        H,
        eps,
        mask_fp16,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Free resources we no longer need
    del x_flat, w_l_T, w_r_T, w_lg_T, w_rg_T, w_og_T, mask_fp16

    # ------------------------------------------------------------------
    # 2️⃣  Tri‑multiplicative update (batched GEMM, FP16)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for the batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣  Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Release unused tensors
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 4️⃣  Final linear projection back to `dim` (FP16 GEMM) + cast to FP32
    # ------------------------------------------------------------------
    # weight shape expected by F.linear: [out_features, in_features] = [dim, H]
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
        bias=None,
    )   # → [R, dim] (FP16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
