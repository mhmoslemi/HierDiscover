def custom_kernel(data):
    x, mask, weights, cfg = data

    dim = cfg['dim']
    hidden_dim = cfg['hidden_dim']
    eps = 1e-5

    B,N,_,_ = x.shape
    R = B*N*N
    device = x.device

    # 1. First layernorm using torch.nn.functional.layer_norm
    # flatten to (R, dim)
    x_flat = x.view(R, dim)
    # F.layer_norm expects weight bias shape (dim,)
    # weights of type float32. Accept them.
    # output float32 then convert to half
    x_norm_fp32 = torch.nn.functional.layer_norm(x_flat, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # 2. Linear projection fused weight
    fused_w = torch.cat([...], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = torch.nn.functional.linear(x_norm, fused_w)   # [R,5*H] fp16

    # 3. Split & gating kernel using Triton
    left = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    right = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    # mask flatten
    if cfg.get('nomask', True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_ptr = mask.to(torch.float16).reshape(R)

    # Launch kernel
    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # 4. Batched GEMM over hidden dimension
    # Reshape for bmm; `left` shape (B,N,N,hidden_dim)
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    # Now shape (B*hidden_dim, N, N)
    left_mat = left.view(B*hidden_dim, N, N)
    right_mat = right.view(B*hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    del left_mat, right_mat, left, right

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(R, hidden_dim).contiguous()
    del out_mat, out

    # 5. Second layernorm + gating
    out_norm_fp32 = torch.nn.functional.layer_norm(out_flat, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)
    out_norm = out_norm_fp32.to(torch.float16)

    gated = out_norm * out_gate
    del out_norm, out_gate

    # 6. Final projection
    final_half = torch.nn.functional.linear(gated, weights['to_out.weight'].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

    bias_norm_ptr,   # const *[H] fp32   to_out_norm.bias beta
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    block_row_start = pid * BLOCK_R
    offs_r = block_row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

@triton.jit
def fused_norm_linear_gate_kernel(
    x_ptr,            # const *[rows, H]   fp16
    gate_ptr,         # const *[rows, H]   fp16
    weight_ptr,       # const *[dim, H]   fp16   to_out.weight
    out_ptr,          # mutable *[rows, dim] fp16
    norm_weight_ptr,  # const *[H]        fp32   to_out_norm.weight
    norm_bias_ptr,    # const *[H]        fp32   to_out_norm.bias
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # compute mean and variance across hidden dimension H
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_o in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_o * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        x_ptrs = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # now iterate over output dim blocks
    for d_o in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_o * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # accumulator for this block of dims
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h_o in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_o * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load x and gate slices
            x_ptrs = x_ptr + offs_r[:, None] * H + offs_h[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            gate_ptrs = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            gate_vals = tl.load(gate_ptrs,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # normalize x
            norm = (x_vals - mean[:, None]) * rstd[:, None]  # (BLOCK_R, BLOCK_H)

            # load LN weight and bias for hidden dim
            w_norm = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
            b_norm = tl.load(norm_bias_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

            # apply LN and gate: ((norm * gamma + beta) * gate)
            y = (norm * w_norm[None, :] + b_norm[None, :]) * gate_vals  # (BLOCK_R, BLOCK_H)

            # load weight slice (transposed) for output dims
            # weight shape [dim, H]; we want [BLOCK_H, BLOCK_D] slice, i.e., rows = hidden dim
            w_ptrs = weight_ptr + offs_h[:, None] + offs_d[None, :] * H   # [BLOCK_H, BLOCK_D]
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)   # fp32

            # accumulate: acc += y @ w   (y: [BLOCK_R, BLOCK_H], w: [BLOCK_H, BLOCK_D])
            acc += tl.dot(y, w)   # (BLOCK_R, BLOCK_D)

        # write out
        out_ptrs = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
        out_vals = acc.to(tl.float16)
        tl.store(out_ptrs,
                 out_vals,
                 mask=mask_r[:, None] & mask_d[None, :])

"""
Tri‑Multiplicative (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16).
2️⃣ One fused linear projection (fp16) that produces the five tensors
   needed for the update (left, right, three gates).
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * extracts left, right and out_gate,
   * applies the optional pair‑wise mask,
   * applies sigmoid on the three gates.
4️⃣ Tri‑multiplicative contraction implemented with a single
   `torch.einsum` (equivalent to B·H batched GEMMs).
5️⃣ Triton kernel `fused_norm_linear_gate_kernel`:
   * row‑wise LayerNorm on the intermediate tensor,
   * multiplies the result by `out_gate`,
   * folds the final linear projection (to_out.weight) into the same kernel.
   The kernel fuses the two memory‑bound steps (norm + gate × weight)
   into a single pass over the data.
6️⃣ Cast the fp16 result back to fp32.

Only the two Triton kernels are required for the “must‑use‑kernel” rule.
All heavy matrix multiplies stay in cuBLAS (torch.nn.functional.linear /
torch.einsum), which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # mean & variance (fp32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # write normalised values (γ·(x‑μ)·rstd + β) as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional unary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection:
        #   [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for numerical stability
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse the 2nd LayerNorm, out‑gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_linear_gate_kernel(
    x_ptr,            # const   *[rows, H]               (fp16)   – post‑tri‑mul tensor
    gate_ptr,         # const   *[rows, H]               (fp16)   – out_gate
    weight_ptr,       # const   *[dim, H]               (fp16)   – to_out.weight
    out_ptr,          # mutable *[rows, dim]           (fp16)   – final result
    ln_weight_ptr,    # const   *[H]                    (float32) – to_out_norm.weight (γ)
    ln_bias_ptr,      # const   *[H]                    (float32) – to_out_norm.bias   (β)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        x_ptrs = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        sum_val += tl.sum(x_vals, axis=1)
        sum_sq  += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # row‑wise LN + gate * weight (fused)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # accumulator for this dim‑block
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load x and gate slices
            x_ptrs = x_ptr + offs_r[:, None] * H + offs_h[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            g_ptrs = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            g_vals = tl.load(g_ptrs,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # normalise x
            norm = (x_vals - mean[:, None]) * rstd[:, None]          # fp32

            # layer‑norm parameters (γ,β)
            w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
            b_ln = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # fp32

            # apply LN and out‑gate
            y = (norm * w_ln[None, :] + b_ln[None, :]) * g_vals      # fp32, shape [BLOCK_R, BLOCK_H]

            # load weight slice (transposed so that we can use tl.dot)
            # weight shape [dim, H] → want [BLOCK_H, BLOCK_D]
            w_ptrs = weight_ptr + offs_h[:, None] + offs_d[None, :] * H   # [BLOCK_H, BLOCK_D]
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                       # fp32

            # acc += y @ w   (y: [BLOCK_R, BLOCK_H], w: [BLOCK_H, BLOCK_D])
            acc += tl.dot(y, w)                                        # fp32

        # write final fp16 result
        out_ptrs = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

# Custom forward for the *outgoing* TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N            # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (produces left, right, three gates)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, dim]

    # linear (no bias) – fp16
    proj = F.linear(x_norm, fused_w)                      # [R, 5*H]
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R = 1024          # max 1024 threads per block
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative contraction (einsum ≡ batched GEMMs)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] – no copy, just a view
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # torch.einsum uses optimized cuBLAS kernels internally and avoids
    # the explicit permute‑contiguous copies of the earlier implementation.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H]
    del left, right

    # flatten for the fused kernel (row‑wise order matches out_gate)
    out_flat = out.view(R, H).contiguous()
    del out

    # --------------------------------------------------------------
    # 5️⃣ Fuse 2nd LayerNorm, out‑gate multiplication and final projection
    # --------------------------------------------------------------
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R2 = 256           # rows per program
    BLOCK_D2 = 32            # output‑dim tile
    grid_fused = (triton.cdiv(R, BLOCK_R2),)
    fused_norm_linear_gate_kernel[grid_fused](
        out_flat,
        out_gate,
        weights["to_out.weight"].to(torch.float16),        # [dim, H]
        final_fp16,
        weights["to_out_norm.weight"],                    # γ (float32)
        weights["to_out_norm.bias"],                      # β (float32)
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Cast back to float32 and reshape to [B, N, N, dim]
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
