"""
Tri‑Multiplicative (outgoing) update – fast forward pass.

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) – use PyTorch’s
    fused `F.layer_norm` (fast on H100).  
2️⃣ One linear projection that produces the five tensors required
   for the update (left, right, left‑gate, right‑gate, out‑gate).  
   The projection is a half‑precision GEMM (cuBLAS/Tensor‑Cores).  

3️⃣ A single Triton kernel (`split_fuse_gate_kernel`)   
   * splits the fused projection into left/right/out_gate,  
   * applies the optional pair‑wise mask,  
   * applies the three sigmoid gates (in FP32 for stability).  

4️⃣ Batched GEMM over the hidden dimension (the “einsum”):
   `(B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)` – performed with
   `torch.bmm` (cuBLAS/Tensor‑Cores).  

5️⃣ Second LayerNorm + out‑gate multiplication – again use PyTorch’s
   fused `F.layer_norm` in FP32 and an element‑wise product with the
   gate (both tensors promoted to FP32).  

6️⃣ Final linear projection back to `dim` (half‑precision GEMM) and
   cast the result to FP32.

Only step 3 is implemented in Triton, satisfying the “at least one
custom kernel” rule while the expensive reductions (layer‑norm) use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional unary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast to H
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_raw   = tl.load(right_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        left_g_raw  = tl.load(left_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store results
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, og,
                 mask=mask_r[:, None] & mask_h[None, :])

    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16) using PyTorch fused kernel
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.view(R, dim)
    # fused layer‑norm (returns float32)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # cast to half for the following GEMM
    x_norm = x_norm_fp32.half()                                 # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5*H channels (left, right, gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim]) → [5*H, dim]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5*H, dim]

    # Linear (no bias) – half‑precision GEMM
    proj = F.linear(x_norm, fused_w)                             # [R, 5*H]
    del fused_w, x_norm, x_flat  # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        # flatten mask to [R] and cast to fp16 (1.0 keep, 0.0 discard)
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128          # rows per program
    BLOCK_H = 32           # hidden dim per tile
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr   # release

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] then permute to [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)  – half‑precision, Tensor Cores
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (torch implementation)
    # ------------------------------------------------------------------
    out_flat_fp16 = out.view(R, H).contiguous()                # [R, H] fp16
    # promote to fp32 for numerically‑stable layer‑norm
    out_flat_fp32 = out_flat_fp16.to(torch.float32)

    out_norm_fp32 = F.layer_norm(
        out_flat_fp32,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply out_gate (cast gate to fp32)
    gated_fp32 = out_norm_fp32 * out_gate.to(torch.float32)
    gated_fp16 = gated_fp32.half()                             # [R, H] fp16
    del out_flat_fp16, out_flat_fp32, out_norm_fp32, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) and cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_fp16,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
