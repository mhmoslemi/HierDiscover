"""
Outgoing TriMul (AlphaFold3) – highly‑optimized Triton/PyTorch forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16) using
   the highly‑optimized cuDNN implementation (torch.nn.functional.layer_norm).

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   matrix‑multiply with a concatenated weight matrix `[5·H, dim]`.  The result
   has shape `[R, 5·H]` (R = B·N·N) and is in fp16.

3️⃣ A small Triton kernel (`split_fuse_gate_kernel`) splits the fused tensor
   into `left`, `right` and `out_gate`, applies the optional pair‑wise mask,
   and computes the three sigmoid gates.  All operations stay in fp16.

4️⃣ The true tri‑multiplicative update is a batched GEMM:
   `out = (left @ rightᵀ)` for each hidden channel.
   We reshape to `[B·H, N, N]` and use `torch.bmm`, which runs on Tensor‑Cores.

5️⃣ Row‑wise LayerNorm on the GEMM result (float16 → float16) using cuDNN,
   then element‑wise multiply by `out_gate`.

6️⃣ Final linear projection back to the original `dim` using a half‑precision
   GEMM (`torch.nn.functional.linear`).  The output is cast back to float32.

Only the split‑gate operation is written in Triton, satisfying the
requirement to use a custom kernel while letting cuDNN / cuBLAS handle the
heavy linear algebra for maximal throughput on an H100.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(

    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load optional pair‑wise mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five contiguous slices: [L,R,Lg,Rg,Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        left_raw     = tl.load(proj_ptr + base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw= tl.load(proj_ptr + base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw=tl.load(proj_ptr + base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # -------- sigmoid (computed in fp32 for numeric stability) --------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----------------- apply mask & gates -----------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----------------- store results -----------------
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
          All tensors are float32.
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N          # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 input → fp16 output) via cuDNN
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim)                     # [R, dim] float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # float32
    x_norm = x_norm_fp32.to(torch.float16)         # [R, dim] fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (produces [R, 5*H] in fp16)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]
    proj = F.linear(x_norm, w_fused)               # [R, 5*H] fp16
    del w_fused, x_norm                           # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and sigmoid gates (Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # flatten mask to [R] and convert to fp16 (0/1)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    # Triton launch configuration
    BLOCK_R = 128   # rows per program
    BLOCK_H = 64    # hidden dimension tile
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched half‑precision GEMM)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] (contiguous)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # fuse batch and hidden dim for torch.bmm
    left_mat = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # out_mat = left_mat @ right_matᵀ   → shape [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H)                                         # [R, H]
    del left_mat, right_mat, left, right, out, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (cuDNN)
    # --------------------------------------------------------------
    out_norm_fp32 = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32
    out_norm = out_norm_fp32.to(torch.float16)   # [R, H] fp16
    out_norm_gated = out_norm * out_gate         # element‑wise, fp16
    del out_flat, out_gate, out_norm

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (half‑precision GEMM) + cast to fp32
    # --------------------------------------------------------------
    final_half = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
