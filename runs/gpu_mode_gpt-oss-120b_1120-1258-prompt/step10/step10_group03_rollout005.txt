"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ One cuBLAS GEMM (R×dim)·(5·H×dim)ᵀ (fp16) produces five projected
   tensors (left, right, left‑gate, right‑gate, out‑gate) in a single
   matrix of shape (R,5·H).
3️⃣ Triton kernel splits the fused projection, optionally masks the
   pairwise tensors, evaluates the three sigmoid gates and writes three
   half‑precision tensors:
        left   = proj_left   * mask * sigmoid(proj_left_gate)
        right  = proj_right  * mask * sigmoid(proj_right_gate)
        out_g  = sigmoid(proj_out_gate)
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ          (torch.bmm)
5️⃣ Row‑wise LayerNorm on `out` followed by element‑wise multiplication
   with `out_g` (single Triton kernel).
6️⃣ Final linear projection back to `dim` (cuBLAS GEMM, fp16→fp32).

Only the two LayerNorm passes and the split‑gate step are written in
Triton; the heavy matrix multiplications use cuBLAS‑tensor‑core kernels,
which gives near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_to_fp16_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- mean & variance (fp32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- write normalized values --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_gate_mask_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]      (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left   = (proj_left   * mask) * sigmoid(proj_left_gate)
        right  = (proj_right  * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic in fp16, sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask if present (broadcast later)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Process hidden dim in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for each of the five slices inside proj
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for numeric stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        left_dst_ptr     = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise LayerNorm on `out` and multiplication with `out_gate`
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)  (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)  (result)
    weight_ptr,       # const   *[dim]        (float32) γ
    bias_ptr,         # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm on `x` (fp16) followed by element‑wise multiplication
    with `gate`.  All intermediate arithmetic in fp32, the final result is
    stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------ mean / variance (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------ normalize, apply γ/β, multiply by gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        norm = norm * w[None, :] + b[None, :]                     # γ·norm + β

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper for block‑size selection
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Simple heuristic for Triton block sizes."""
    BLOCK_M = 256 if rows >= 256 else max(1, rows)
    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Custom forward kernel – entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear / layer‑norm parameters
                         (all tensors are float32, Linear layers have no bias)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    nomask      = cfg.get("nomask", True)
    device      = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_fp32_to_fp16_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate weight matrices → shape [5*H, dim] (float16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear: (R, dim) @ (5*H, dim).T → (R, 5*H)   (cuBLAS fp16 GEMM)
    proj = F.linear(x_norm, w_fused)        # fp16
    del w_fused, x_norm                     # free memory

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        # No mask ⇒ compile‑time constant 0, dummy pointer.
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
        mask_ptr = dummy_mask
    else:
        # Convert mask to 0/1 float16 and flatten to [R]
        mask_half = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R = 128          # rows per program
    BLOCK_H = 64           # hidden dim per program (covers H up to 128 efficiently)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_gate_mask_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # Reshape to [B, H, N, N] (contiguous) → view as [B*H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # bmm: (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
        num_stages=2,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
