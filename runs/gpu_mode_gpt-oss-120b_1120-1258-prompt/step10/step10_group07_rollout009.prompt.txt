import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton + cuBLAS implementation
======================================================

Algorithm
---------
1️⃣  Row‑wise LayerNorm of the 4‑D input (B·N·N × dim) – fp32 → fp16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (R × 5·H)  where  R = B·N·N .
3️⃣  Triton kernel splits the fused result, applies the optional pair‑wise mask,
    evaluates sigmoid gates and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
   All in FP16 (sigmoid computed in FP32 for stability).
4️⃣  Batched FP16 GEMM over the hidden dimension:
        out = left_mat @ right_matᵀ   (B·H independent N×N matmuls)
   Implemented with `torch.matmul` → cuBLAS/Tensor‑Cores.
5️⃣  Fuse the second LayerNorm with the output‑gate multiplication in a single
    Triton kernel.
6️⃣  Final FP16 linear projection back to `dim` and cast to FP32.

Only the three Triton kernels are custom; the heavy matrix multiplications use
cuBLAS (via `torch.nn.functional.linear` / `torch.matmul`) which gives
near‑peak Tensor‑Core throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (fp32 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]      (float32)
    y_ptr,          # mutable *[rows, dim]      (float16)
    weight_ptr,     # const   *[dim]           (float32)   γ
    bias_ptr,       # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2: split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are in FP16; sigmoid evaluated in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # load (or create) mask value per row
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                         # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        # raw projection slices
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # write out
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3: second LayerNorm + multiplication with out_gate (fp16 in/out)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16) – out_gate
    y_ptr,            # mutable *[rows, dim]   (float16) – result
    weight_ptr,       # const   *[dim]        (float32) γ
    bias_ptr,         # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply with gate → store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]          # fp32

        # apply γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # load gate
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()         # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates (5 * H rows)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, w_concat)       # fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).reshape(R)  # flatten
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_mat = torch.matmul(left, right.transpose(-1, -2))           # fp16

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication by out_gate (fused kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
