"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened input.
   Implemented as a custom Triton kernel for maximum throughput.

2️⃣ Fuse the five linear maps (left/right projection + three gates) into a
   single FP16 GEMM:
        X_norm @ W_fused.T  →  (R, 5·H)
   where R = B·N·N and H = hidden_dim.

3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   evaluate sigmoid gates and reshape to (B, H, N, N) layout.

4️⃣ Batched half‑precision matrix multiplication (torch.bmm) over the hidden
   dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   → out has shape (B, N, N, H) in FP32.

5️⃣ Apply a second LayerNorm on the hidden axis, multiply by the out‑gate
   (sigmoid), and finally project back to the original channel dimension
   with a FP32 linear layer.

Only the first LayerNorm is written in Triton; the remaining operations use
high‑performance PyTorch primitives (the heavy GEMM is delegated to cuBLAS).

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns a tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   γ
    b_ptr,          # const   *[dim]        (float32)   β
    rows,           # tl.int32
    dim,            # tl.int32
    eps,            # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_dblocks = tl.cdiv(dim, BLOCK_D)
    for db in range(0, num_dblocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized output ----------
    for db in range(0, num_dblocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)
        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                       # total positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16) using Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (FP32)
        weights["norm.bias"],     # β (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Free the original tensor early
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left/right proj + three gates)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # (5·H, dim)

    # Linear: X_norm (R, dim) @ W_fused.T (dim, 5·H) → (R, 5·H)
    proj = F.linear(x_norm_fp16, fused_w)               # FP16
    proj = proj.to(torch.float32)                       # work in FP32 afterwards
    del fused_w, x_norm_fp16

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask & sigmoid gates, reshape
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)               # [R, 5, H]

    left_raw      = proj[:, 0, :]           # [R, H]
    right_raw     = proj[:, 1, :]
    left_gate_raw = proj[:, 2, :]
    right_gate_raw= proj[:, 3, :]
    out_gate_raw  = proj[:, 4, :]

    # Optional pair‑wise mask (broadcast over H)
    if mask is not None and not cfg.get("nomask", True):
        mask = mask.to(device).to(torch.float32)
        mask_flat = mask.view(R, 1)         # [R, 1]
        left_raw   = left_raw * mask_flat
        right_raw  = right_raw * mask_flat
    # Apply sigmoid gates
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # Reshape to (B, H, N, N) layout required for batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)                 # (B·H, N, N)
    right_mat = right.reshape(B * H, N, N)                # (B·H, N, N)

    # out_mat shape: (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden axis, out‑gate, final projection
    # --------------------------------------------------------------
    # LayerNorm over the last dimension (hidden_dim)
    out_ln = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # still FP32

    # Apply out‑gate
    out_gated = out_ln * out_gate

    # Final linear projection back to dim
    out_flat = out_gated.view(R, hidden_dim)           # [R, H]
    final = F.linear(out_flat, weights["to_out.weight"])   # (R, dim), FP32

    final = final.view(B, N, N, dim)                   # [B, N, N, dim]

    return final
```
