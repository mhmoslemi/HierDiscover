"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The implementation follows the reference PyTorch module but
fuses the two LayerNorms, the linear projections and the sigmoid gates in
custom Triton kernels.  The heavy batched GEMM (half‑precision Tensor‑Core)
remains in PyTorch (cuBLAS) where it is already optimal.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input  (B·N·N × dim) → fp16.
2️⃣  One fused Linear produces the five tensors needed for left/right
    projections and the three gates (shape (R, 5·H),  R=B·N·N).
3️⃣  `split_fuse_gate_fused_kernel`:
    * decodes (b,i,j) from a flat row index,
    * optionally applies the pair‑wise mask,
    * computes sigmoid gates (fp32 → fp16),
    * writes **left** and **right** directly in (B·H, N, N) layout,
    * stores the **out_gate** tensor (R, H) for later use.
4️⃣  Batched half‑precision GEMM (cuBLAS/Tensor‑Cores):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
5️⃣  `layernorm_mul_strided_kernel`:
    * computes a second LayerNorm over the hidden dimension H
      (strided by N²),
    * multiplies the normalized values by the previously computed `out_gate`,
      and writes the fused result (R, H) (fp16).
6️⃣  Final linear projection back to the original dimension `dim`.

Only the two LayerNorms and the split/fuse kernel are written in Triton;
the remaining linear algebra uses highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)  γ
    bias_ptr,       # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program (≤1024)
    BLOCK_D: tl.constexpr,   # dim chunk size (multiple of 32)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # accumulate mean / variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]     # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # write normalized values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]     # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split / fuse projection, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,j) from flat row index
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # optional mask (broadcast across hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base offset into concatenated projection tensor (row‑major)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)                                # fp16
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid – computed in fp32 for numerical stability
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # write left/right in (B·H, N, N) layout
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (still (rows, H))
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (float16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (float16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (float16)   (fused output)
    weight_ptr,     # const   *[H]         (float32)   γ
    bias_ptr,       # const   *[H]         (float32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from flat row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset of the first hidden slice (h = 0) within the strided tensor
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ mean / variance over H (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply by out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β fp32

        norm = norm * w[None, :] + b[None, :]               # fp32

        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # fp32

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters.
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and a bool
                         ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                   # [R, C] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 1024        # rows per program (tuned for H100)
    BLOCK_D_LN = 64          # dim chunk (multiple of 32 works well)
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ float32, shape [C]
        weights["norm.bias"],     # β float32, shape [C]
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=8,
        num_stages=3,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5*H, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, C]

    # (R,5*H) fp16
    proj = F.linear(x_norm, concat_w)                     # fused linear
    del concat_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask values are expected to be 0/1; flatten to [R]
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 1024   # rows per program (balanced for all test sizes)
    BLOCK_H = 32     # hidden dim block (fits H=128 well)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (half‑precision TensorCore)
    # ------------------------------------------------------------------
    # left/right are already (B*H, N, N) in fp16
    out_mat = torch.bmm(left, right.transpose(1, 2))      # (B*H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_mul_strided_kernel[grid_ln2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ float32, shape [H]
        weights["to_out_norm.bias"],     # β float32, shape [H]
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_H=BLOCK_D_LN,              # reuse same dim block size
        num_warps=8,
        num_stages=3,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # weight shape: [dim, hidden_dim] – already float32, convert to fp16
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
