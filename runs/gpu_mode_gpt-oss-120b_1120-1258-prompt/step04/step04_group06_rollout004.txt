"""
Highly‑optimized Triton‑accelerated forward pass for the *outgoing* TriMul
operator (AlphaFold‑3).  The algorithm follows the reference implementation
but removes all unnecessary memory traffic and fuses several element‑wise
operations:

1️⃣   Row‑wise LayerNorm on the input tensor (custom Triton kernel).
2️⃣   A single fused half‑precision GEMM that computes the five linear
     layers (left/right projections + three gates) at once.
3️⃣   In‑place sigmoid on the gate slices and in‑place masking/gating,
     eliminating temporaries.
4️⃣   Batched matrix multiplication (B·H independent GEMMs) that implements
     the triangular‑multiplicative update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
5️⃣   A second Triton kernel that performs LayerNorm **and** multiplies by the
     output gate in a single pass.
6️⃣   Final half‑precision linear projection back to the original dimension.

All heavy kernels run in FP16, only the final result is cast to FP32.
The implementation works for all test configurations (various seq‑len,
batch‑size, hidden dimensions and optional masks) and satisfies the
requirement of using Triton for at least part of the computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – Row‑wise LayerNorm (float32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [total_rows, dim]  (float32 or fp16)
    y_ptr,            # mutable *[total_rows, dim] (fp16 output)
    weight_ptr,       # const   [dim] (float32)
    bias_ptr,         # const   [dim] (float32)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute per‑row mean and variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ----- write normalized values (γ·(x‑μ)·rstd + β) in FP16 ---------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(
            out_ptrs,
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )

    y_ptr,            # mutable *[total_rows, dim] (fp16)
    weight_ptr,       # const   [dim] (float32)

    gate_ptr,         # const   [total_rows, dim] (fp16)  – out_gate
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(

# ----------------------------------------------------------------------
# Entry‑point: forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[bs, N, N]       (bool/float)
        - weights_dict : mapping from weight name → torch.Tensor
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the row‑wise LN

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (row‑wise) – Triton kernel
    # ------------------------------------------------------------------
    # flatten to [total_rows, dim] (float32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(
        total_rows, dim, dtype=torch.float16, device=device
    )  # fp16 result

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the 5 linear layers (2 projections + 3 gates) into one GEMM
    # ------------------------------------------------------------------
    # concatenate all five weight matrices → [5*hidden_dim, dim]  (fp16)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # single half‑precision linear: [R, dim] @ [5*H, dim].T → [R, 5*H]
    proj_concat = F.linear(x_norm_flat, w_concat)  # fp16, shape [R, 5*H]

    H = hidden_dim
    # ---- slice views (no extra memory) ---------------------------------
    left_proj = proj_concat[:, 0 : H]                     # [R, H]
    right_proj = proj_concat[:, H : 2 * H]                # [R, H]
    left_gate = proj_concat[:, 2 * H : 3 * H]             # [R, H]
    right_gate = proj_concat[:, 3 * H : 4 * H]            # [R, H]
    out_gate = proj_concat[:, 4 * H : 5 * H]              # [R, H]

    # ---- in‑place sigmoid for the three gate tensors --------------------
    left_gate.sigmoid_()
    right_gate.sigmoid_()
    out_gate.sigmoid_()

    # ---- optional mask (applied only to the two projected tensors) ----
    if not config.get("nomask", True):
        # mask shaped [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj.mul_(mask_h)
        right_proj.mul_(mask_h)

    # ---- element‑wise gating (in‑place) ---------------------------------
    left_proj.mul_(left_gate)       # left = left_proj * left_gate
    right_proj.mul_(right_gate)     # right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for batched bmm
    left = left_proj.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right_proj.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # collapse batch & head dimensions → [B*H, N, N]
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # batched matrix multiplication:   (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣  Second LayerNorm + output‑gate multiplication (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_scaled_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_norm_scaled_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate,                        # out_gate after sigmoid (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # shape back to [B, N, N, H]
    out_norm_scaled = out_norm_scaled_flat.view(B, N, N, H)    # fp16

    # ------------------------------------------------------------------
    # 5️⃣  Final projection back to `dim` (half‑precision linear)
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm_scaled.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # → [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)   # cast to FP32
    return final
