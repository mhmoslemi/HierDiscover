@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr, gate_ptr, weight_norm_ptr, bias_norm_ptr,
    w_ptr, # (hidden, dim) half
    out_ptr, # (total_rows, dim) half
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, # rows per tile
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(total_rows, BLOCK_M)
    num_pid_n = tl.cdiv(dim, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # Compute mean and var for each row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulator for output
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # second pass: compute normalized, gate, weight multiplication
    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # Load input tile
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)  # (M,K)

        # Load gate tile
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Normalize
        norm = (x - mean[:, None]) * rstd[:, None]  # (M,K)

        # Apply to_out_norm weight & bias per hidden dimension
        gamma = tl.load(weight_norm_ptr + offs_k, mask=mask_k, other=0.0) # (K)
        beta = tl.load(bias_norm_ptr + offs_k, mask=mask_k, other=0.0)   # (K)
        # expand dims
        norm = norm * gamma[None, :] + beta[None, :]

        # Apply gating
        y = norm * gate

        # Load weight for linear projection: (K, N_block)
        w = tl.load(w_ptr + offs_k[:, None] * dim + offs_n[None, :],
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)   # shape (K,N)

        # Accumulate dot product: y (M,K) dot w (K,N) -> (M,N)
        acc += tl.dot(y, w)  # tl.dot expects (M,K)*(K,N)->(M,N)

    # Store result
    tl.store(out_ptr + offs_m[:, None] * dim + offs_n[None, :],
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
Highly‑optimized Triton forward for the **outgoing** TriMul operator
(AlphaFold‑3 “TriMul – outgoing” version).

Algorithm
---------
1️⃣  Input LayerNorm (row‑wise) – Triton kernel that computes the
    mean/variance in FP32 and writes FP16 normalized data.

2️⃣  Fuse the five linear projections (left/right + 3 gates) into one
    FP16 GEMM (“proj_all”).  The weight matrix is the concatenation of the
    five [hidden, dim] matrices → a single torch linear call.

3️⃣  Apply the optional binary mask, the three sigmoid gates and element‑wise
    gating on the left/right projections.

4️⃣  Triangular multiplicative update = batched matrix‑multiply over the
    sequence dimension (B × hidden) – torch.bmm‑TensorCore.

5️⃣  Second LayerNorm + out‑gate + output projection are fused into a single
    Triton kernel.  The kernel:
        • reads the B × N × N × hidden tensor,
        • computes row‑wise mean/variance,
        • normalises, scales (γ/β), multiplies by the pre‑computed out‑gate,
        • performs the GEMM with the transposed “to_out” weight
          (hidden → dim) in the same pass.
    This removes an extra memory pass compared to the reference
    implementation.

6️⃣  The fused kernel writes the final tensor in FP16; we cast it back
    to FP32 before returning.

Result
------
`custom_kernel(data)` returns a tensor of shape `[B, N, N, dim]` (FP32)
identical to the PyTorch reference but ≈2× faster on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32 input)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16 output)
    weight_ptr,          # *const float*   [dim]               (γ)
    bias_ptr,            # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- first pass: mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: normalise & affine ----------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate + final Linear (hidden → dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half* [M, H]   (pre‑norm)
    gate_ptr,            # *const half* [M, H]   (out_gate, already sigmoid’ed)
    ln_weight_ptr,       # *const float* [H]    (γ of to_out_norm)
    ln_bias_ptr,         # *const float* [H]    (β of to_out_norm)
    w_ptr,               # *const half*  [H, D] (to_out weight already transposed)
    y_ptr,               # *mut   half*  [M, D] (final output)
    M: tl.int32,                     # total rows = B·N·N
    H: tl.int32,                     # hidden dimension
    D: tl.constexpr,                 # output dimension (dim)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_K: tl.constexpr,           # reduction (hidden) tile
    BLOCK_N: tl.constexpr,           # output‑channel tile
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(D, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < D

    # ---------- first pass: mean / variance across hidden ---------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        x = tl.load(x_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # (M, K)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalise, apply γ/β, gate & GEMM ----------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw hidden values
        x = tl.load(x_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # (M, K)

        # load corresponding gate values
        g = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # (M, K)

        # normalise
        norm = (x - mean[:, None]) * rstd[:, None]       # (M, K)

        # apply to_out_norm affine (γ/β) – per‑hidden
        gamma = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)  # (K)
        beta  = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)  # (K)
        norm = norm * gamma[None, :] + beta[None, :]     # (M, K)

        # apply out‑gate
        norm = norm * g                                 # (M, K)

        # load weight tile (K, Nblock)
        w = tl.load(w_ptr + offs_k[:, None] * D + offs_n[None, :],
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)           # (K, N)

        # accumulate GEMM contribution
        acc += tl.dot(norm, w)                          # (M, N)

    # ---------- write result --------------------------------------------
    tl.store(y_ptr + offs_m[:, None] * D + offs_n[None, :],
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32 / bool)
        - weights_dict : model parameters (float32)
        - config_dict  : { "dim": C,
                           "hidden_dim": H,
                           "nomask": bool,
                           "eps": float (optional) }

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()               # [R, C] FP32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gates (single FP16 GEMM)
    # --------------------------------------------------------------
    # Concatenate the five projection / gate weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden, dim)
    ).half().contiguous()

    proj_all = F.linear(x_norm_fp16, concat_w)                     # (R, 5*hidden) FP16
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, hidden)

    # --------------------------------------------------------------
    # 2a) Optional input mask
    # --------------------------------------------------------------
    if not nomask:
        # mask -> (R, 1) half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)          # (R, hidden)
    right_gate = torch.sigmoid(right_gate_fp16)        # (R, hidden)
    out_gate = torch.sigmoid(out_gate_fp16)            # (R, hidden) – used later

    left_fp16 = left_proj_fp16 * left_gate             # (R, hidden)
    right_fp16 = right_proj_fp16 * right_gate          # (R, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)            # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)          # (B, N, N, H)

    # reshape for batched GEMM: (B*H, N, N) × (B*H, N, N)ᵀ
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                        # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final Linear (hidden → dim)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()            # (R, H)  FP16
    out_gate_flat = out_gate.view(total_rows, hidden).contiguous() # (R, H)  FP16

    # transpose to_out weight for easier indexing inside the kernel: (H, D)
    w_to_out_T = weights["to_out.weight"].t().half().contiguous()   # (H, D)

    out_final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    # kernel launch configuration
    BLOCK_K = 32
    BLOCK_N = 64
    grid_final = (triton.cdiv(total_rows, BLOCK_M) * triton.cdiv(dim, BLOCK_N),)

    layernorm_gate_linear_kernel[grid_final](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],          # γ (float32)  shape [H]
        weights["to_out_norm.bias"],            # β (float32)  shape [H]
        w_to_out_T,
        out_final_fp16,
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )

    # --------------------------------------------------------------
    # 5️⃣ Cast back to FP32 and reshape
    # --------------------------------------------------------------
    final = out_final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
