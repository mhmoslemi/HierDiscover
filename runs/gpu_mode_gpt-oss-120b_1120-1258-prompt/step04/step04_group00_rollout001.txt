def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    device = x.device
    rows = B * N * N

    # 1. First LayerNorm (FP16)
    total_rows = rows
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2. Fused projection + gate + out_gate linear
    # Concatenate all weight matrices
    # left_proj, left_gate, right_proj, right_gate, out_gate => shapes (H, dim)
    # Build weight_concat total of shape (5*H, dim)
  
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # Concatenate in order
    weight_concat = torch.cat([w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=0)
    weight_concat = weight_concat.to(torch.float16)

    # Linear (rows, dim) @ (5H, dim).T = (rows, 5H)
    fused_out = torch.nn.functional.linear(x_norm_h, weight_concat)  # bias None

    # split
    H = hidden_dim
    offset0 = 0
    left_proj_h = fused_out[:, offset0: offset0+H]          # [rows, H]
    offset0 += H
    left_gate_pre_h = fused_out[:, offset0: offset0+H]
    offset0 += H
    right_proj_h = fused_out[:, offset0: offset0+H]
    offset0 += H
    right_gate_pre_h = fused_out[:, offset0: offset0+H]
    offset0 += H
    out_gate_pre_h = fused_out[:, offset0: offset0+H]   # final H

    # apply mask if needed
    if not nomask:
        mask_h = mask.view(rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # gates
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h = torch.sigmoid(out_gate_pre_h)

    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # reshape to (B, N, N, H)
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_mat = left_perm.view(BH, N, N)          # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # batched GEMM: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    out = (

        .permute(0, 2, 3, 1)
        .contiguous()

        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm_h = out_norm_h.view(B, N, N, H)

    out_norm_h = out_norm_h * out_gate_h.view(B, N, N, H)

        weights["to_out.weight"].to(torch.float16)
    )  # shape (rows, dim)

    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul “outgoing” forward pass – optimized Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input tensor.
2️⃣  Fuse **all** linear projections + gates (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`, `out_gate`) into a **single** GEMM:
        Y = norm(X) @ Wᵀ   with   W ∈ ℝ^{5·H×C}
    The five slices of Y give the projected tensors and the pre‑sigmoid gates.
3️⃣  Apply optional pair‑wise mask (broadcast over the hidden dimension) and
    the three sigmoid gates.
4️⃣  Compute the Tri‑multiplicative update:
        out_h(i,j) = Σₖ left(i,k,h)·right(j,k,h)
    This is a batched GEMM (B·H matrices of size N×N) performed with
    `torch.bmm`, which on H100 uses Tensor‑Cores.
5️⃣  Row‑wise LayerNorm on the hidden dimension (H) of the update.
6️⃣  Apply the output‑gate (`out_gate`) and a final linear projection
    (H → C).  The result is cast back to FP32.

Only the LayerNorms are hand‑written in Triton; all dense linear layers
and the O(N³) update use highly‑optimised torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor.
    The mean/variance are computed in FP32, the normalized result is
    written as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1 : compute mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------- Pass 2 : write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # weight
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # bias

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(
            out_ptrs,
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    rows = B * N * N
    x_flat = input_tensor.view(rows, C).contiguous()          # [rows, C] FP32

    x_norm_h = torch.empty(
        (rows, C), dtype=torch.float16, device=device
    )  # FP16 normalized output

    # launch LayerNorm kernel
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        rows,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all projections & gates into a single matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # ------------------------------------------------------------------
    # concatenate weight matrices: each is (H, C)
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)                     # (5H, C)

    fused_out = F.linear(x_norm_h, weight_concat)              # [rows, 5H]

    # slice the concatenated output
    offset = 0
    left_proj_h      = fused_out[:, offset : offset + H];        offset += H
    left_gate_pre_h  = fused_out[:, offset : offset + H];        offset += H
    right_proj_h     = fused_out[:, offset : offset + H];        offset += H
    right_gate_pre_h = fused_out[:, offset : offset + H];        offset += H
    out_gate_pre_h   = fused_out[:, offset : offset + H];        offset += H

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)               # [rows, H]

    left_h  = left_proj_h  * left_gate_h      # [rows, H]
    right_h = right_proj_h * right_gate_h     # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Reshape and perform the Tri‑multiplicative update
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # bring hidden dimension into batch axis → (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H                         # effective batch for GEMM
    left_mat  = left_perm.view(BH, N, N)      # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)     # [BH, N, N]

    # batched GEMM: (N×N) @ (N×N)ᵀ → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    rows_h = rows                       # B·N·N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_h.view(B, N, N, H)   # gated

    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )   # [rows_h, C] (FP16)

    # Cast back to FP32 as required by the reference module
    final = final_h.view(B, N, N, C).to(torch.float32)

    return final
