def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, H,
                           stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                           stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)  # batch_h
    pid1 = tl.program_id(1)  # tile_i
    pid2 = tl.program_id(2)  # tile_j

    batch_h = pid0
    batch = batch_h // H
    h = batch_h % H

    i_offset = pid1 * BLOCK_I
    j_offset = pid2 * BLOCK_J

    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # K loop tiling:
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_outer * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i,k)
        # shape: BLOCK_I x BLOCK_K
        left_ptrs = (batch * stride_left_b
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k
                     + h * stride_left_h)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (j,k) -> we need shape (BLOCK_K, BLOCK_J)
        # So load as (offs_k[:, None], offs_j[None, :]) but our memory layout is j,i? Wait.
        # right tensor dims: (batch, j, k, h). So stride for j dimension = stride_right_j (N*H)
        # stride for k dimension = stride_right_k (=H)
        # So pointer: batch*stride_b + offs_j[:, None]*stride_j + offs_k[None,:]*stride_k + h*stride_h

        right_ptrs = (batch * stride_right_b
                      + offs_j[:, None] * stride_right_j
                      + offs_k[None, :] * stride_right_k
                      + h * stride_right_h)
        # This loads shape (BLOCK_J, BLOCK_K)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Now right needs transposition to (BLOCK_K, BLOCK_J) to match dot.
        # We can do tl.dot(left, right, trans_b=True) or compute dot with transposition.
        acc += tl.dot(left, right, trans_b=True)  # left: I x K, right: J x K -> dot trans_b => left @ right.T => I x J
    # Normalize? No.

    # Store to out pointer: out[batch, i, j, h] shape (B,N,N,H)
    # pointer: batch*stride_out_b + offs_i[:, None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
    out_ptrs = (batch * stride_out_b
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j
                + h * stride_out_h)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Highly‑optimized Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣  Input LayerNorm (per‑row) – Triton kernel that reads FP32,
    computes mean/var in FP32 and writes FP16.

2️⃣  Fuse the five linear projections (left, right, three gates) into a
    single half‑precision GEMM (`torch.nn.functional.linear`).  This avoids
    five separate matmuls and keeps the data in FP16.

3️⃣  Apply the optional mask (broadcast on the hidden dimension).

4️⃣  Compute sigmoid gates (still FP16) and gate the left/right projections.

5️⃣  **Triangular multiplicative update** – a single Triton kernel that
    computes, for every batch b and hidden channel h,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel tiles the (i,j) plane (64×64) and the reduction axis k
    (32).  Each thread‑block processes one (batch×channel) pair, so the
    whole (B·H) matrix‑multiplications are fused into one launch, avoiding
    the massive launch‑overhead of `torch.bmm`.

6️⃣  Fused LayerNorm + output‑gate – another Triton kernel that normalises
    the result, applies the learned scale/shift and multiplies by the
    pre‑computed `out_gate`.

7️⃣  Final linear projection back to `dim` (half‑precision GEMM) and cast
    to FP32.

Only the two LayerNorm steps and the O(N³) contraction are written in
Triton; all dense linear algebra uses cuBLAS/Tensor‑Cores which is already
optimal on an H100.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static dimension (≤ 384)
    eps: tl.constexpr,   # LayerNorm epsilon
    BLOCK_M: tl.constexpr,  # rows per program (e.g. 64)
    BLOCK_D: tl.constexpr,  # columns processed per iteration (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) TriMul contraction (outgoing) – fused over all hidden channels
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # *const half*  [B, N, N, H]
    right_ptr,  # *const half*  [B, N, N, H]
    out_ptr,    # *mut   half*  [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,  # tile size on i dimension
    BLOCK_J: tl.constexpr,  # tile size on j dimension
    BLOCK_K: tl.constexpr,  # tile size on k dimension
):
    # program ids:
    # pid0 : batch*H (each hidden channel of each batch)
    # pid1 : i‑tile
    # pid2 : j‑tile
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # decode batch and hidden channel
    batch_h = pid0
    batch = batch_h // H
    h = batch_h % H

    # start offsets for this tile
    i_offset = pid1 * BLOCK_I
    j_offset = pid2 * BLOCK_J

    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k tiles
    # ------------------------------------------------------------------
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_outer * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i,k) ------------------------------------
        left_ptrs = (batch * stride_l_batch
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k
                     + h * stride_l_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # (BLOCK_I, BLOCK_K)

        # ----- load right tile (j,k) -----------------------------------
        # right layout is (batch, j, k, h)
        right_ptrs = (batch * stride_r_batch
                      + offs_j[:, None] * stride_r_j
                      + offs_k[None, :] * stride_r_k
                      + h * stride_r_h)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)         # (BLOCK_J, BLOCK_K)

        # ----- accumulate (i,k)·(j,k)^T --------------------------------
        # tl.dot(A, B, trans_b=True) computes A @ B.T
        acc += tl.dot(left, right, trans_b=True)               # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Store the result (FP16)
    # ------------------------------------------------------------------
    out_ptrs = (batch * stride_o_batch
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)

        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):

    to_out_w_fp16 = weights["to_out.weight"].half()                # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)       # (R, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)        # FP32 output
    return final
```
