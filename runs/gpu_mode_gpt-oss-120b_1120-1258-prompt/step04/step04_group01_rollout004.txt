@triton.jit
def rowwise_stats_kernel(x_ptr, mean_ptr, rstd_ptr,
                         total_rows, dim,
                         eps,
                         BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

@triton.jit
def rowwise_stats_kernel(
    x_ptr,   # const float32*
    mean_ptr,   # mutable float32*
    rstd_ptr,   # mutable float32*
    total_rows,
    dim,
    stride_xm,      # stride for rows (dim)
    stride_xd,      # stride for columns (1)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        val = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
                     mask=mask_m[:, None] & mask_d[None, :],
                     other=0.0)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass
================================================

The implementation removes the intermediate LayerNorm output tensor
and fuses the first LayerNorm with the five dense projections
(left/right projection + three gates).  The fusion is based on the
identity

    LN(x) @ W = rstd * (x @ (γ·W)) – (μ·rstd)·(γ·W).sum(dim=0) + (β·W).sum(dim=0)

where *γ* and *β* are the LayerNorm affine parameters and (μ,rstd) are the
row‑wise mean and inverse‑std.  We therefore:

1.   Compute per‑row mean and rstd **once** with a tiny Triton kernel that
     also copies the FP32 input to an FP16 buffer (used later for the GEMM).
2.   Pre‑scale the projection weight matrix with γ and pre‑compute the
     constant offsets (γ·W).sum and (β·W).sum.
3.   Perform a single batched FP16 GEMM on the FP16 copy of the input.
4.   Apply the per‑row scaling / offset to obtain the five concatenated
     tensors (left‑proj, right‑proj, left‑gate, right‑gate, out‑gate).
5.   Apply the mask, sigmoid gates and the triangular multiplicative update
     (batched GEMM left @ rightᵀ).
6.   A second row‑wise LayerNorm (still implemented as a Triton kernel)
     is applied to the hidden tensor, followed by the element‑wise output
     gate and the final linear projection back to the model dimension.

All heavy matrix multiplications run in FP16 Tensor‑Core mode, while
statistics are computed in FP32 for numerical stability.
The final tensor is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# 1️⃣ Row‑wise mean / rstd + FP32→FP16 copy (used for the first projection)
# -------------------------------------------------------------------------
@triton.jit
def stats_and_fp16_copy_kernel(
    x_ptr,               # const fp32 *   [R, dim]
    x_fp16_ptr,          # mutable fp16 * [R, dim] (copy of x in fp16)
    mean_ptr,            # mutable fp32 * [R]
    rstd_ptr,            # mutable fp32 * [R]
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps: tl.constexpr,   # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # accumulate statistics
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to the current block of the source tensor (fp32)
        src_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val_f32 = tl.load(src_ptr,
                          mask=mask_m[:, None] & mask_d[None, :],
                          other=0.0)                         # fp32

        # statistics
        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

        # write the same values as fp16 into the temporary buffer
        dst_ptr = x_fp16_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(dst_ptr,
                 val_f32.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr   + offs_m, rstd, mask=mask_m)


# -------------------------------------------------------------------------
# 2️⃣ Simple row‑wise LayerNorm used for the second (hidden) normalization
# -------------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # const   *  [R, dim]   (fp16)
    y_ptr,           # mutable *  [R, dim]   (fp16)
    weight_ptr,      # const   *  [dim]      (fp32)
    bias_ptr,        # const   *  [dim]      (fp32)
    total_rows,      # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass – mean & variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0)                     # fp16
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass – write normalized + affine
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d,
                    mask=mask_d,
                    other=0.0)                # gamma, fp32
        b = tl.load(bias_ptr + offs_d,
                    mask=mask_d,
                    other=0.0)                # beta, fp32
        out = norm * w[None, :] + b[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# -------------------------------------------------------------------------
# Custom forward kernel for the **outgoing** TriMul operator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   (float32)
            mask_tensor  : [B, N, N]        (bool/int, optional)
            weights      : dict of model weight tensors
            config       : dict with keys:
                - dim          (int)
                - hidden_dim   (int)
                - nomask (bool, optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # e.g. 128 or 384
    hidden_dim = config["hidden_dim"]   # always 128 in the tests
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean / rstd and copy the input to FP16 once
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()    # [R, dim] fp32
    x_fp16 = torch.empty_like(x_flat, dtype=torch.float16)      # buffer for GEMM
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # columns per program (fits dim≤384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    stats_and_fp16_copy_kernel[grid](
        x_flat,                 # fp32 source
        x_fp16,                 # fp16 destination
        mean,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse LayerNorm affine (γ,β) into the weight matrix of the five
    #     projections + three gates.
    # --------------------------------------------------------------
    # Concatenated weight matrix W ∈ ℝ[5·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5H, dim]

    # ---- pre‑compute γ·W and the constant offset vectors
    # γ and β from the first LayerNorm
    gamma = weights["norm.weight"]                     # [dim] fp32
    beta  = weights["norm.bias"]                       # [dim] fp32

    # Wᵀ  (dim × 5H) in FP32 for the small algebraic steps
    w_T_fp32 = w_concat.t().float()                    # [dim, 5H]

    # γ·Wᵀ  (dim × 5H)
    gamma_w_T = w_T_fp32 * gamma[:, None]              # broadcast → fp32

    # Weight used in the GEMM (γ·W) transposed back to (5H × dim) fp16
    scaled_weight = gamma_w_T.t().half()               # [5H, dim] fp16

    # Offset vectors (γ·W) summed over the input dimension and (β·W)
    offset_vec   = gamma_w_T.sum(dim=0)                # [5H] fp32   (= γ·W).sum(dim=0)
    bias_vec     = (w_T_fp32 * beta[:, None]).sum(dim=0)   # [5H] fp32   (= β·W).sum(dim=0)

    # --------------------------------------------------------------
    # 3️⃣ Linear projection on the FP16 copy (single GEMM)
    # --------------------------------------------------------------
    proj_fp16 = F.linear(x_fp16, scaled_weight)        # [R, 5H] fp16
    proj_fp32 = proj_fp16.float()                      # cast back to fp32 for scaling

    # --------------------------------------------------------------
    # 4️⃣ Apply the per‑row LN scaling + the constant offsets
    # --------------------------------------------------------------
    # rstd, mean : [R]  (float32)
    scale = rstd.unsqueeze(-1)                         # [R,1]
    # term1 = rstd * (x @ (γ·W))
    term1 = proj_fp32 * scale

    # term2 = -(μ·rstd) * offset_vec
    mu_rstd = (mean * rstd).unsqueeze(-1)              # [R,1]
    term2 = mu_rstd * offset_vec                       # broadcast → [R,5H]

    # final concatenated tensor after the first LN & linear
    out_concat = term1 - term2 + bias_vec               # [R,5H]  fp32
    out_concat = out_concat.to(torch.float16)           # back to fp16 for the rest

    # --------------------------------------------------------------
    # 5️⃣ Split concatenated tensor into the five components
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj   = out_concat[:, 0:H]
    right_proj  = out_concat[:, H:2*H]
    left_gate_p = out_concat[:, 2*H:3*H]
    right_gate_p= out_concat[:, 3*H:4*H]
    out_gate_p  = out_concat[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 6️⃣ Sigmoid gates
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 7️⃣ Optional pairwise mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).view(total_rows, 1)   # [R,1]
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply gates
    # --------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # reshape back to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 9️⃣ Triangular multiplicative update (batched matmul)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_perm = torch.matmul(
        left_perm,
        right_perm.transpose(-1, -2)
    )                                                    # [B, H, N, N]

    out = out_perm.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]

    # --------------------------------------------------------------
    # 10️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (fp32) for hidden dim
        weights["to_out_norm.bias"],     # β (fp32) for hidden dim
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, H)

    # --------------------------------------------------------------
    # 11️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (FP16)

    out_norm_flat2 = out_norm.view(total_rows, H).contiguous()
    final_fp16 = F.linear(
        out_norm_flat2,
        weights["to_out.weight"].to(torch.float16)    # [dim, H]
    )                                                  # [R, dim]

    # Cast back to FP32 for downstream use
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
