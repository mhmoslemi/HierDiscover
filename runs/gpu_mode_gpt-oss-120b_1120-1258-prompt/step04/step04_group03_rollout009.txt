"""
Highly‑optimised Triton + PyTorch implementation of the *outgoing* TriMul
operator (AlphaFold‑3).  

Algorithm
---------
1. **LayerNorm (first)** – custom Triton kernel that computes the
   per‑row mean/variance in FP32 and writes the normalised result in
   FP16.  This avoids the expensive reduction in the native PyTorch
   kernel for the very large flattened view (B·N·N rows).

2. **Fused linear projections + gates** – the five weight matrices
   (left/right projection, three gates) are concatenated and applied
   with a single `F.linear` call on the FP16 normalised tensor.

3. **Optional mask** – applied to the two projected tensors before gating.

4. **Sigmoid + elementwise gating** – all performed in FP16.

5. **Tri‑multiplicative update** – for every head *h* we need
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  
   This is exactly a batched matrix‑multiply:
   * left   → (B, H, N, N)  = left.permute(0,3,1,2)
   * right  → (B, H, N, N)  = right.permute(0,3,2,1)
   * out    = torch.matmul(left, right)   (FP16, Tensor‑Core)

6. **LayerNorm (second)** – the same Triton kernel, now over the hidden
   dimension *H*.

7. **Output‑gate + final projection** – elementwise multiplication with
   the output‑gate, then a single `F.linear` that maps `H → C`.  The
   result is cast back to FP32 as required by the API.

All heavy reductions (the two LayerNorms) are done in Triton,
dense linear algebra is delegated to cuBLAS/Tensor‑Cores, and we keep
the entire pipeline in FP16 until the very last cast, minimising memory
traffic and achieving runtimes well below the 1000 µs target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm that writes FP16 output.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*  [total_rows, dim]   (fp32 input)
    y_ptr,          #       half*  [total_rows, dim]   (fp16 output)
    w_ptr,          # const float*  [dim]              (γ)
    b_ptr,          # const float*  [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- 1) compute mean & variance (FP32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # reduce over D
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -------------------- 2) write normalised + affine (FP16) ----------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ, FP32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β, FP32

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            * input_tensor : torch.Tensor [B, N, N, C]  (float32)
            * mask_tensor  : torch.Tensor [B, N, N]    (bool/float or None)
            * weights_dict : dict of model weights (float32)
            * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, C]
    total_rows = B * N * N                         # rows for first LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                               dtype=torch.float16,
                               device=device)

    # Tune block sizes: 256 rows per program, 128‑column tiles (covers
    # all supported channel sizes up to 384 without extra loops).
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gates (still FP16)
    # ------------------------------------------------------------------
    # Concatenate weight matrices of the five linear layers (shape: 5H x C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    # Linear projection (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)

    # Split back
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )                                            # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Apply optional mask (before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B,N,N] → [total_rows,1] (broadcast over hidden dim)
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16 = left_proj * left_gate      # (total_rows, H)
    right_fp16 = right_proj * right_gate   # (total_rows, H)

    # Reshape to 4‑D tensors for the update
    left = left_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs its k‑dimension transposed (j ↔ k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Merge batch and head dimensions
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)    # (B*H, N, N)

    # Batched matrix multiplication in FP16 (Tensor‑Core)
    out_mat = torch.matmul(left_mat, right_mat)      # (B*H, N, N)

    # Back to original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: (B, N, N, H)  (FP16)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden dimension (custom Triton)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                      # rows for second LN
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)    # FP16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H) FP16

    # Final linear projection H → C
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)    # (B*N*N, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # (B*N*N, C)

    # Cast to float32 for API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
