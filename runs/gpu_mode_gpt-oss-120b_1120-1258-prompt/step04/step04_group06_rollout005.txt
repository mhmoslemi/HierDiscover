"""
custom_kernel – Optimised forward pass for the “outgoing” TriMul operator
=======================================================================

Algorithm (FP16‑centric)
------------------------
1️⃣  Row‑wise LayerNorm on the input tensor (shape B·N·N × dim).  A small
    Triton kernel computes per‑row mean/variance in FP32 and writes the
    normalised result in FP16.

2️⃣  Fuse the five linear layers (left/right projections + three gates) into
    one half‑precision GEMM:
        [R, dim] @ [5·hidden, dim]ᵀ → [R, 5·hidden]   (R = B·N·N)
    The result is sliced into the three projections and three gate tensors.

3️⃣  Apply the optional mask **after** the gate‑multiplication – this needs only
    one extra element‑wise multiply per tensor.

4️⃣  Tri‑multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    This is exactly a batched matrix‑multiply of shape
    (B·hidden, N, N) @ (B·hidden, N, N)ᵀ.  We use a cuBLAS‐accelerated
    ``torch.bmm`` (Tensor‑Core FP16) after a cheap ``permute``/``contiguous``
    that reorganises the data to [B·hidden, N, N].

5️⃣  Second LayerNorm (over the hidden dimension) – the same Triton kernel as
    step 1, now with hidden‑size = hidden_dim.

6️⃣  Multiply by the output‑gate, then a final half‑precision linear projection
    back to ``dim``.  The result is cast to FP32 to match the reference
    implementation.

Only the two LayerNorms are written in Triton; all heavy dense linear
operations rely on cuBLAS/Tensor‑Core kernels, which gives the required
speed‑up for the N³ step while keeping the implementation simple and
numerically faithful.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [total_rows, dim]   (float32 or fp16)
    y_ptr,            # mutable *[total_rows, dim]   (fp16 output)
    weight_ptr,       # const *  [dim]               (float32)
    bias_ptr,         # const *  [dim]               (float32)
    total_rows,       # int   (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / √(σ²+eps) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute row‑wise mean and variance (accumulate in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalised values γ·(x−μ)·rstd + β (store as FP16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor with shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device

    B, N, _, _ = input_tensor.shape          # input is [B, N, N, dim]
    total_rows = B * N * N                    # rows for row‑wise layer‑norms

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    # flat view = [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Choose a block size that keeps the number of launched programs modest.
    BLOCK_M = 1024 if total_rows > 256 * 256 else 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (FP32)
        weights["norm.bias"],     # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear layers (2 projections + 3 gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices: shape [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # FP16 for Tensor‑Core GEMM

    # Single GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H] (half‑precision)
    proj_concat = F.linear(x_norm, w_concat)   # shape [R, 5*hidden]

    H = hidden_dim
    left_proj   = proj_concat[:, 0:H]
    right_proj  = proj_concat[:, H:2*H]
    left_gate_p = proj_concat[:, 2*H:3*H]
    right_gate_p= proj_concat[:, 3*H:4*H]
    out_gate_p  = proj_concat[:, 4*H:5*H]

    # Sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (applied only to the projected tensors)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1]  (FP16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Element‑wise gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # Back to 4‑D tensors
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣  Tri‑multiplicative update – batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Re‑order to [B, H, N, N] and fuse batch & head dimensions → [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)         # [B·H, N, N]

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original order → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣  Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise (FP16)

    # Final linear projection: [R, H] → [R, dim] (half‑precision)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # shape [dim, H]
    )   # -> [R, dim] (FP16)

    # Cast to float32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
