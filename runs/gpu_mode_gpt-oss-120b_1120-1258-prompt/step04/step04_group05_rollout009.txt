"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 mean/var, fp16 output) on the input tensor.
2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM by concatenating the weight matrices.
3️⃣ Optional pairwise mask applied before the gates.
4️⃣ Sigmoid gates and element‑wise gating (still in fp16).
5️⃣ **Outgoing triangular multiplicative update**:
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   Implemented as a custom Triton batched GEMM that computes
   `left @ rightᵀ` for every hidden slice in a single launch (no
   intermediate materialisation of the transposed tensor).
6️⃣ Row‑wise LayerNorm on the hidden dimension (fp32 reduce, fp16 output).
7️⃣ Multiply by the output gate and a final half‑precision linear projection
   back to the original channel dimension.
8️⃣ Cast result to float32.

The implementation keeps the heavy linear work (steps 2 & 7) in cuBLAS
because it is already close‑to‑optimal, while the two LayerNorms and the
core N³ update are hand‑written in Triton to avoid extra memory traffic
and to achieve better occupancy on H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,          # *mut    fp16*   [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]               (γ)
    bias_ptr,       # *const float*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton batched GEMM for the outgoing TriMul update:
#   C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (i.e. A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,                # *fp16 tensors
    batch, N: tl.constexpr,
    stride_am, stride_ai, stride_ak,    # strides of A  (batch, i, k)
    stride_bm, stride_bj, stride_bk,    # strides of B  (batch, j, k)
    stride_cm, stride_ci, stride_cj,    # strides of C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch dimension (B * hidden_dim)
    pid_m = tl.program_id(1)              # tile over i
    pid_n = tl.program_id(2)              # tile over j

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # batch offset (common for A/B/C)
    a_batch_off = pid_batch * stride_am
    b_batch_off = pid_batch * stride_bm
    c_batch_off = pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: A[batch, i, k]
        a_ptrs = A_ptr + a_batch_off \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # B tile (transposed on the fly):
        # we need B_T[k, j] = B[batch, j, k]
        b_ptrs = B_ptr + b_batch_off \
                 + offs_n[None, :] * stride_bj \
                 + offs_k[:, None] * stride_bk
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        acc += tl.dot(a, b)                 # fp32 accumulation

    # Store result as fp16
    c_ptrs = C_ptr + c_batch_off \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool / float)
            - weights_dict : model weights (float32 tensors)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # flatten over (B, i, j)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)         # fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)                       # fp16 buffer

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + 3 gates)
    # --------------------------------------------------------------
    # Concatenate weight matrices (each [H, C])
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat([w_left, w_right, w_lgate,
                          w_rgate, w_outgate], dim=0).half().contiguous()   # (5*H, C)

    # Fused GEMM: [total_rows, C] @ [C, 5*H]ᵀ → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16

    # Split back
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each [total_rows, H] (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to 4‑D layout for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Outgoing triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Move hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N]  (fp16)
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N]  (fp16)

    out_mat = torch.empty_like(left_mat)                # fp16 result

    # strides for the Triton kernel (batch, i, k) / (batch, j, k)
    stride_am, stride_ai, stride_ak = left_mat.stride()
    stride_bm, stride_bj, stride_bk = right_mat.stride()
    stride_cm, stride_ci, stride_cj = out_mat.stride()

    BLOCK_M_T = 64   # tile size for output rows (i)
    BLOCK_N_T = 64   # tile size for output cols (j)
    BLOCK_K_T = 32   # reduction tile size (k)

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M_T),
        triton.cdiv(N, BLOCK_N_T),
    )
    batched_trimul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M_T,
        BLOCK_N=BLOCK_N_T,
        BLOCK_K=BLOCK_K_T,
    )

    # Back to original [B, N, N, H] layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [total_rows, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate            # fp16 element‑wise

    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)      # [total_rows, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # fp16 → [total_rows, C]

    # --------------------------------------------------------------
    # 8️⃣ Cast to float32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
