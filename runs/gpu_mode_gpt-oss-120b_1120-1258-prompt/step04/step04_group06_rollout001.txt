"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – custom Triton
    kernel (`layernorm_kernel`).  The kernel works on fp16 data, does the
    reduction in fp32 and writes the normalized result in fp16.

2️⃣  Fuse the five linear layers (left/right projections + three gates) into a
    single half‑precision GEMM:
    `[R, dim] @ [5·hidden, dim].T → [R, 5·hidden]` where `R = B·N·N`.

3️⃣  Apply optional mask (only to the two projected tensors) and the three
    sigmoid gates.

4️⃣  Triangular multiplicative update:
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    This is exactly a batched matrix‑multiply of shape
    `(B·hidden, N, N) @ (B·hidden, N, N).T`.  We use `torch.bmm`,
    which dispatches to cuBLAS Tensor‑Core kernels on an H100.

5️⃣  Second LayerNorm over the hidden dimension – same Triton kernel as step‑1.

6️⃣  Multiply by the output gate, project back to `dim` with a final half‑precision
    linear layer and cast the result to float32.

All heavy arithmetic is performed in FP16 (Tensor‑Core friendly), only the
normalisation accumulates in FP32 for numerical stability.  At least one
operation (both LayerNorms) is executed by a custom Triton kernel as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input can be fp16/ fp32, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [rows, dim]   (fp16 or fp32)
    y_ptr,            # mutable *[rows, dim]   (fp16 output)
    weight_ptr,       # const *  [dim]         (fp32)
    bias_ptr,         # const *  [dim]         (fp32)
    rows,             # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / √(σ²+eps) * γ + β
    - Computation is done in FP32, result stored as FP16.
    - Works for any `dim` that fits in the block size.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    # row indices covered by this program
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension in tiles of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load a (BLOCK_M, BLOCK_D) tile.  Load as fp16 and promote to fp32.
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized output
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        # (x-μ) * rstd
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) are fp32 scalars per feature
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out_fp32 = norm * w[None, :] + b[None, :]
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward op
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same CUDA device as inputs).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input feature dimension
    hidden = config["hidden_dim"]      # hidden dimension (H)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # = R

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – custom Triton kernel
    # --------------------------------------------------------------
    # Flatten to [R, dim] and allocate fp16 buffer for the normalized tensor
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Kernel launch parameters – tuned for H100
    BLOCK_M = 1024          # rows per program (≈ 1 Mi rows → ~1024 blocks)
    BLOCK_D = 64           # feature tile size (fits dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (fp32)
        weights["norm.bias"],      # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    # Concatenate weight matrices → shape [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16 for Tensor‑Core GEMM

    # Single GEMM: [R, dim] @ [5*hidden, dim].T → [R, 5*hidden]
    proj = F.linear(x_norm, w_concat)       # fp16 result, shape [R, 5*H]

    H = hidden
    left_proj   = proj[:, 0:H]                     # [R, H]
    right_proj  = proj[:, H:2*H]                   # [R, H]
    left_gate_p = proj[:, 2*H:3*H]                 # [R, H]
    right_gate_p = proj[:, 3*H:4*H]                # [R, H]
    out_gate_p  = proj[:, 4*H:5*H]                 # [R, H]

    # 3️⃣ Sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (applied only to projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gate the projections
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape to 4‑D tensors for the N³ update
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Move hidden dimension into the batch dimension → [B*H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched mat‑mul: left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original ordering → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (custom Triton)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise scaling (fp16)

    # Final linear projection: [R, H] → [R, dim]
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    # Cast to float32 to match the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
