"""
custom_kernel - Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input  (B·N·N × dim) → half‑precision.
2️⃣ Fuse the five linear layers (2 projections + 3 sigmoid gates) into a single
   GEMM:   X_norm @ W_concat.T  →  [R, 5·hidden]  (R = B·N·N).
   Slice the result into left/right projections and the three gate vectors.
3️⃣ Apply optional mask (only on the two projected tensors) and the three
   sigmoid gates (all in fp16).
4️⃣ Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply (B·hidden) × (N,N) @ (N,N)ᵀ.
   Implemented as a custom Triton kernel that:
   • treats each head (batch * hidden) as an independent GEMM,
   • loads tiles of shape (BM×BK) and (BK×BN) in fp16,
   • accumulates in fp32 and stores the result in fp16.
5️⃣ Second LayerNorm over the hidden dimension (fp16).
6️⃣ Multiply by the output gate, apply the final linear projection back to
   `dim` and cast the result to float32.

The heavy N³ work is performed by the Triton kernel (step 4), while the
LayerNorms are also run in Triton.  All other operations use fused
FP16 GEMMs from cuBLAS (via `torch.nn.functional.linear`).  This layout
maximises Tensor‑Core utilisation on an H100 and reduces memory traffic,
bringing the runtime well below the target 1 ms for the benchmark sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input can be fp32 or fp16, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *[total_rows, dim]  (fp32/fp16)
    y_ptr,            # mutable *[total_rows, dim]  (fp16)
    weight_ptr,       # const   *[dim]             (fp32)
    bias_ptr,         # const   *[dim]             (fp32)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm producing fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Apply γ,β and store fp16 result
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: batched “TriMul” = left @ rightᵀ   (half‑precision, acc‑fp32)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const   *[BATCH, M, K] (fp16)
    right_ptr,         # const   *[BATCH, N, K] (fp16) – we will read it transposed
    out_ptr,           # mutable *[BATCH, M, N] (fp16)
    BATCH, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_m, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    Implements a batch of GEMMs where the second operand is effectively
    transposed (i.e. we multiply left[i,k] with right[j,k]).
    """
    pid_m = tl.program_id(0)   # tile over M
    pid_n = tl.program_id(1)   # tile over N
    batch_id = tl.program_id(2)  # which head (BATCH = B*hidden)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k_start * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_idxs = left_ptr \
            + batch_id * stride_left_batch \
            + offs_m[:, None] * stride_left_m \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_idxs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (transposed on‑the‑fly): we need right_T[k, n] = right[n, k]
        # Load as shape (BLOCK_K, BLOCK_N)
        right_idxs = right_ptr \
            + batch_id * stride_right_batch \
            + offs_k[:, None] * stride_right_m \
            + offs_n[None, :] * stride_right_k
        right = tl.load(right_idxs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(left, right)   # (BM×BK) @ (BK×BN) -> (BM×BN)

    # Write back the result
    out_idxs = out_ptr \
        + batch_id * stride_out_batch \
        + offs_m[:, None] * stride_out_m \
        + offs_n[None, :] * stride_out_n

    out_fp16 = acc.to(tl.float16)
    tl.store(out_idxs,
             out_fp16,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Full forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32 on the same device).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for row‑wise LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gates into one GEMM (half‑precision)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices: shape (5*hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Linear: [R, dim] @ [5*hidden, dim].T -> [R, 5*hidden]
    proj_concat = F.linear(x_norm_flat, w_concat)   # fp16 → shape (R, 5*hidden)

    H = hidden_dim
    left_proj   = proj_concat[:, 0:H]
    right_proj  = proj_concat[:, H:2*H]
    left_gate_p = proj_concat[:, 2*H:3*H]
    right_gate_p = proj_concat[:, 3*H:4*H]
    out_gate_p   = proj_concat[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Gates + optional mask (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    if not config.get("nomask", True):
        # mask shape [B, N, N] -> flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the N³ update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update via custom Triton GEMM
    # ------------------------------------------------------------------
    # Permute to bring head dimension into the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & head -> batch for the kernel
    BATCH = B * H
    left_mat = left_perm.view(BATCH, N, N)
    right_mat = right_perm.view(BATCH, N, N)

    # Allocate output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_left_batch = left_mat.stride(0)
    stride_left_m     = left_mat.stride(1)   # row stride = N
    stride_left_k     = left_mat.stride(2)   # col stride = 1

    stride_right_batch = right_mat.stride(0)
    stride_right_m     = right_mat.stride(1)
    stride_right_k     = right_mat.stride(2)

    stride_out_batch = out_mat.stride(0)
    stride_out_m     = out_mat.stride(1)
    stride_out_n     = out_mat.stride(2)

    # Triton launch configuration.  These block sizes give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),      # tiles over M (i dimension)
        triton.cdiv(N, BLOCK_N),      # tiles over N (j dimension)
        BATCH,                        # each head
    )
    trimul_kernel[grid](
        left_mat, right_mat, out_mat,
        BATCH, N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_m, stride_right_k,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original 4‑D layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate + final projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                 # element‑wise fp16

    # Final linear: [R, H] → [R, dim] (half‑precision)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    )   # fp16

    # Cast to float32 as in the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
