    stride_bh, stride_bk, stride_bj,

        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is expressed as a custom Trit‑kernel that computes a batch of
   matrix‑multiplications with the hidden dimension folded into the batch
   axis (BH = B·hidden).  No extra permutes or copies are performed.
6️⃣ LayerNorm on the hidden dimension (FP16 → FP16) using another Triton
   kernel.
7️⃣ Multiply by the output‑gate (sigmoid) and a final half‑precision linear
   projection (hidden → dim).  The result is cast back to FP32.

Only the two LayerNorm kernels and the core Tri‑multiplication kernel are
implemented in Triton; all dense linear layers use PyTorch’s highly‑optimised
FP16 Tensor‑Core kernels.  The implementation works for all test shapes
listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: 2‑D row‑wise LayerNorm (writes FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*   [total_rows, dim]
    y_ptr,            #       half*   [total_rows, dim] (output)
    w_ptr,            # const float*  [dim]   (γ)
    b_ptr,            # const float*  [dim]   (β)
    total_rows,       # int   runtime number of rows
    dim,              # tl.constexpr number of columns
    eps,              # tl.constexpr epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                         # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean / variance (FP32) ---------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)                   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                    # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values (FP16) --------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched tri‑multiplicative update (outgoing)
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const half*  [B, N, N, H]  (contiguous)
    right_ptr,         # const half*  [B, N, N, H]  (contiguous)
    out_ptr,           #       half*  [B, N, N, H]  (output)
    batch,             # int  B
    N,                 # int  sequence length
    hidden_dim,        # int  H
    stride_batch,      # int  stride over B dimension (N*N*H)
    stride_i,          # int  stride over the first N dim (N*H)
    stride_k,          # int  stride over the second N dim (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids:
    pid_z = tl.program_id(0)               # combined (batch, hidden)
    pid_i = tl.program_id(1)               # tile over rows   (i)
    pid_j = tl.program_id(2)               # tile over cols   (j)

    # decompose combined id
    b_id = pid_z // hidden_dim            # batch index 0 .. B-1
    h_id = pid_z % hidden_dim             # hidden index 0 .. H-1

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offset for the (b_id, h_id) slice
    base = b_id * stride_batch + h_id     # h_id uses stride_h == 1

    # accumulator
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left matrix tile A[i,k]   (shape BLOCK_M × BLOCK_K)
        a_ptrs = left_ptr + base \
                 + offs_i[:, None] * stride_i \
                 + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right matrix tile B_T[k,j] = right[b,j,k,h] (shape BLOCK_K × BLOCK_N)
        b_ptrs = right_ptr + base \
                 + offs_k[:, None] * stride_k \
                 + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)               # FP32 accumulation

    # write output tile out[i,j]
    out_ptrs = out_ptr + base \
               + offs_i[:, None] * stride_i \
               + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask_tensor  : torch.Tensor [B, N, N]  (float32) – may be None/unused
        weights_dict : mapping of all model weights (float32)
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty_like(x_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64 if dim <= 256 else 128 if dim <= 384 else 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_ptr=x_flat,
        y_ptr=x_norm,
        w_ptr=weights["norm.weight"],
        b_ptr=weights["norm.bias"],
        total_rows=total_rows,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm = x_norm.view(B, N, N, dim)          # FP16

    # --------------------------------------------------------------
    # 2️⃣ fused projection + gate (single matmul per side)
    # --------------------------------------------------------------
    # concatenate proj & gate weights (dim → 2·hidden)
    left_w_cat = torch.cat([weights["left_proj.weight"],
                            weights["left_gate.weight"]], dim=0).to(torch.float16, device=device)
    right_w_cat = torch.cat([weights["right_proj.weight"],
                             weights["right_gate.weight"]], dim=0).to(torch.float16, device=device)

    rows = total_rows
    x_flat = x_norm.view(rows, dim)             # [rows, dim] FP16

    left_all = F.linear(x_flat, left_w_cat)     # [rows, 2·H]
    right_all = F.linear(x_flat, right_w_cat)   # [rows, 2·H]

    left_proj = left_all[:, :hidden_dim]        # [rows, H]
    left_gate_pre = left_all[:, hidden_dim:]    # [rows, H]
    right_proj = right_all[:, :hidden_dim]
    right_gate_pre = right_all[:, hidden_dim:]

    # out‑gate (shared for the whole pair)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_flat, out_gate_w)  # [rows, H]
    out_gate = torch.sigmoid(out_gate_pre)       # [rows, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, device=device)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate      # [rows, H] FP16
    right_h = right_proj * right_gate   # [rows, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] – already contiguous because rows = B·N·N
    left_h = left_h.view(B, N, N, hidden_dim).contiguous()
    right_h = right_h.view(B, N, N, hidden_dim).contiguous()
    out_h = torch.empty_like(left_h)                     # will hold FP16 result

    # strides of the [B, N, N, H] tensors
    stride_batch = left_h.stride(0)   # N*N*H
    stride_i = left_h.stride(1)       # N*H
    stride_k = left_h.stride(2)       # H

    # launch Triton kernel: grid = (B*H, ceil(N/BM), ceil(N/BN))
    BLOCK_M_T = 64
    BLOCK_N_T = 64
    BLOCK_K_T = 64
    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M_T),
            triton.cdiv(N, BLOCK_N_T))
    trimul_outgoing_kernel[grid](
        left_ptr=left_h,
        right_ptr=right_h,
        out_ptr=out_h,
        batch=B,
        N=N,
        hidden_dim=hidden_dim,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_k=stride_k,
        BLOCK_M=BLOCK_M_T,
        BLOCK_N=BLOCK_N_T,
        BLOCK_K=BLOCK_K_T,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm on hidden dimension
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out_h.view(rows_h, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_D2 = 64 if hidden_dim <= 256 else 128
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        x_ptr=out_flat,
        y_ptr=out_norm,
        w_ptr=weights["to_out_norm.weight"],
        b_ptr=weights["to_out_norm.bias"],
        total_rows=rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ output‑gate & final projection
    # --------------------------------------------------------------
    out_gate_view = out_gate.view(B, N, N, hidden_dim)   # FP16
    out_gated = out_norm * out_gate_view                # FP16

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    final_h = F.linear(out_gated.view(rows_h, hidden_dim), to_out_w)  # [rows_h, dim] FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
