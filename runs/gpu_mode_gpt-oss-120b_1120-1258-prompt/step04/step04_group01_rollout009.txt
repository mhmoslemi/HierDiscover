def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    d_base = tl.arange(0, BLOCK_D)  # for this tile

    mask_i = i_base < N
    mask_j = j_base < N
    mask_d = d_base < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load left tile: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        # Build pointer offsets for each (i,k,d)
        left_ptrs = left_ptr + pid_b*stride_left_b \
                    + i_base[:, None, None] * stride_left_i \
                    + k_base[None, :, None] * stride_left_k \
                    + d_base[None, None, :] * stride_left_d

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # Load right tile similarly but j dimension
        right_ptrs = right_ptr + pid_b*stride_right_b \
                    + j_base[:, None, None] * stride_right_j \
                    + k_base[None, :, None] * stride_right_k \
                    + d_base[None, None, :] * stride_right_d

        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # Multiply & accumulate across K dimension: For each k, we need outer product across i and j for each d.
        # This can be done via broadcasting:
        # left_i_k_d: (M, K, D)
        # right_j_k_d: (N, K, D)
        # Compute sum over k dimension:
        # acc[i,j,d] += sum_k left[i,k,d] * right[j,k,d]

        # We can compute with an inner loop over k:
        for kk in range(BLOCK_K):
            # shape (M, D) and (N, D) for this k
            l = left[:, kk, :]   # (M, D)
            r = right[:, kk, :]  # (N, D)
            # expand dims and multiply
            acc += l[:, None, :] * r[None, :, :]  # (M,N,D)

    # Write output
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + i_base[:, None, None] * stride_out_i \
               + j_base[None, :, None] * stride_out_j \
               + d_base[None, None, :] * stride_out_d

    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f = val.to(tl.float32)

        sum_val += tl.sum(val_f, axis=1)
        sum_sq += tl.sum(val_f * val_f, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # fused linear (5 * hidden_dim) outputs
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm, w_concat)   # [R, 5*H] fp16

    H = hidden_dim
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(proj, H, dim=1)

    # optional mask (broadcast over hidden dim)
    if not config.get("nomask", True):
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16)  # [R,1]
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # apply gates
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # reshape back to [B,N,N,H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Triangular multiplicative update (einsum, works on original layout)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # fp16

    # second LN over hidden dim
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)

    # gating and final projection
    out_norm = out_norm * out_gate   # elementwise fp16

    out_flat2 = out_norm.view(total_rows, H)
    final = F.linear(out_flat2, weights["to_out.weight"].to(torch.float16))

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The implementation follows the reference PyTorch module but speeds up the two
LayerNorms with a custom Triton kernel and removes the expensive “transpose‑and‑contiguous”
steps that were required for a cuBLAS batched GEMM.  The heavy
`einsum('b i k d, b j k d -> b i j d')` operates directly on the original
layout `[B, N, N, H]` (no copies, no extra transposes) and therefore avoids the
~500 MiB of memory traffic that dominated the previous version.
All linear layers are fused into a single half‑precision GEMM to keep the
working set in FP16, the final output is returned as FP32.

The kernel hierarchy:
1️⃣  Row‑wise LayerNorm (FP32 → FP16) – Triton.
2️⃣  Fused projection + three gates (single FP16 GEMM via `F.linear`).
3️⃣  Optional pairwise mask (applied only to the two projections).
4️⃣  Triangular multiplicative update via a single `torch.einsum`
    (operates on the original `[B,N,N,H]` layout, no transposes).
5️⃣  Second LayerNorm (again Triton).
6️⃣  Element‑wise output gate → final FP16 linear projection → FP32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *  [rows, dim]   (fp32 or fp16)
    y_ptr,          # mutable *  [rows, dim]   (fp16)
    w_ptr,          # const   *  [dim]         (fp32)  – gamma
    b_ptr,          # const   *  [dim]         (fp32)  – beta
    rows,           # int (runtime)
    dim: tl.constexpr,          # compile‑time
    eps: tl.constexpr,          # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # -----------------------------------------------------------------
    # 1st pass – compute mean & variance per row
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load as fp16 if the source is fp16, then promote to fp32
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass – write normalized value * gamma + beta (store fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # model dimension (e.g. 128 or 384)
    hidden_dim = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1) first LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (fp32)
        x_norm,                     # output pointer (fp16)
        weights["norm.weight"],     # gamma (float32)
        weights["norm.bias"],       # beta  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) fuse five linear layers (proj + three gates) into a single GEMM
    # --------------------------------------------------------------
    # concat: [left_proj; right_proj; left_gate; right_gate; out_gate]  → (5·H, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # FP16 for speed

    proj = F.linear(x_norm, w_concat)       # [R, 5·H]  (FP16)

    H = hidden_dim
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, H, dim=1
    )

    # --------------------------------------------------------------
    # 3) optional pairwise mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16)   # [R,1]
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4) apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_proj * left_gate          # FP16
    right = right_proj * right_gate       # FP16

    # --------------------------------------------------------------
    # 5) reshape to 4‑D tensor and perform the triangular multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # einsum works directly on the original layout – no transposes needed
    # out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)   # FP16

    # --------------------------------------------------------------
    # 6) second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7) output gate + final projection back to original dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (FP16)

    out_norm_flat = out_norm.view(total_rows, H)
    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]  (FP16)
    )                                                  # [R, dim]

    # cast to FP32 for downstream consumption
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
