"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑Norm over the last channel (dim).  A custom Triton kernel
    computes mean/variance in fp32 and writes the normalized tensor
    in fp16 (to save bandwidth).

2️⃣ Five linear layers (left/right projections + 3 gates) are fused
    into a single `F.linear` call by concatenating the weight matrices.
    All arithmetic is performed in fp16.

3️⃣ Optional pairwise mask is applied to the two projections before
    the element‑wise sigmoid gates.

4️⃣ The three sigmoid gates are evaluated and applied.

5️⃣ Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a single batched GEMM using `torch.matmul` on the
    view `(B,H,N,N)`.  No intermediate copies are made – the permutes
    only change strides, and cuBLAS handles the resulting strides
    efficiently.

6️⃣ Second Layer‑Norm (over the hidden dimension) – same Triton kernel.

7️⃣ Multiply by the output gate, project back to `dim` with a final
    FP16 linear layer and cast the result to fp32 (the public API contract).

The implementation works for all supported batch‑size, sequence‑length
and channel configurations and runs on H100 (Triton 3.3.1).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – row‑wise mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized + affine (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        * input_tensor : torch.Tensor, [B, N, N, C] (float32)
        * mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
        * weight_dict  : mapping of all model parameters (float32)
        * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256            # tuned for H100
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gate‑linear layers
    # ------------------------------------------------------------------
    # shape (5*H, C) – everything is fp16
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    # [total_rows, 5*H]  –   fp16 GEMM
    proj_all = F.linear(x_norm_fp16, concat_w)    # (total_rows, 5*H)

    # split back
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each : (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # element‑wise gating
    left_fp16 = left_proj * left_gate       # (total_rows, H)
    right_fp16 = right_proj * right_gate    # (total_rows, H)

    # reshape to 4‑D tensors needed for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM, no extra copies)
    # ------------------------------------------------------------------
    # View as (B, H, N, N) without materialising a copy.
    # torch.matmul can handle the non‑contiguous stride pattern efficiently.
    left_perm = left.permute(0, 3, 1, 2)     # (B, H, N, N)   – view only
    right_perm = right.permute(0, 3, 2, 1)   # (B, H, N, N)   – view only

    # Batched GEMM → (B, H, N, N)
    out = torch.matmul(left_perm, right_perm)

    # Rearrange back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H)  fp16

    # final linear projection H → C (still fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # (total_rows_h, C)

    # cast back to fp32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
