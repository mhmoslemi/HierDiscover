"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1. Row‑wise LayerNorm (custom Triton kernel) on the 4‑D input
   → FP16 tensor `x_norm`.

2. Fuse the five linear layers (left/right projection + three gates)
   into one half‑precision GEMM:
   `[R, dim] @ [5*H, dim].T → [R, 5*H]` where `R = B·N·N`.

3. Slice the concatenated result, apply the optional mask,
   compute the three sigmoid gates and obtain the gated tensors
   `left` and `right` (shape `[B,N,N,H]`).

4. Rearrange to `[B*H,N,N]` (contiguous) and perform the
   batched matrix multiplication `left @ rightᵀ`
   → `[B*H,N,N]`.  Reshape back to `[B,N,N,H]`.

5. Second LayerNorm over the hidden dimension (use
   `torch.nn.functional.layer_norm`, which is highly tuned).

6. Apply the output gate, project back to the original
   `dim` with a final half‑precision linear layer and cast
   to float32.

All heavy arithmetic (LayerNorm, GEMMs, and the O(N³) update) runs in
FP16, exploiting Tensor‑Core throughput on an H100.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim] (FP32)
    y_ptr,          # mutable *[total_rows, dim] (FP16)
    weight_ptr,     # const   *[dim] (FP32)   – γ
    bias_ptr,       # const   *[dim] (FP32)   – β
    total_rows,     # i32
    dim,            # i32 (compile‑time)
    eps,            # f32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per loop (tile over dim)
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(var + eps)) * γ + β
    μ and var are computed per row.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance (float32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                         # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)     # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized result (float16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                         # FP32 load
        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (custom Triton)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # FP32 [R, dim]
    x_norm_flat = torch.empty(total_rows,
                              dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256            # rows per program – tuned for our R
    BLOCK_D = 64             # cols per inner tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],   # γ
        weights["norm.bias"],     # β
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers into one GEMM (FP16)
    # ------------------------------------------------------------------
    # Concatenate weights: shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # Linear: (R, dim) @ (dim, 5*H).T → (R, 5*H)
    out_concat = F.linear(x_norm_flat, w_concat)                # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply mask, compute sigmoid gates
    # ------------------------------------------------------------------
    # Slices (all shape [R, H])
    left_proj   = out_concat[:, 0:H]
    right_proj  = out_concat[:, H:2*H]
    left_gate_p = out_concat[:, 2*H:3*H]
    right_gate_p= out_concat[:, 3*H:4*H]
    out_gate_p  = out_concat[:, 4*H:5*H]

    # sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)   # [R,1]
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates and reshape to [B, N, N, H]
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).view(B, N, N, H)    # FP16
    right = (right_proj * right_gate).view(B, N, N, H) # FP16

    # Rearrange to (B*H, N, N) – each head becomes its own batch.
    # .contiguous() forces the needed layout for the GEMM.
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))   # (B*H,N,N)

    # Back to [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (use functional LN)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                                       # FP16
    out_flat = out.view(total_rows, H)                         # [R, H]

    final_half = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16)            # [dim, H]
    )                                                          # [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)    # FP32 output
    return final
