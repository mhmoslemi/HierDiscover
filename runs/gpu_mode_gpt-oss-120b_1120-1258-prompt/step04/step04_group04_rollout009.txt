"""
TriMul (outgoing) – fast forward pass in Triton + PyTorch.

Algorithm
---------
1. Layer‑norm the input (fp16 output) with a custom *one‑pass* Triton
   kernel that loads each element only once (stores it in shared memory
   for the second reduction step).
2. Fuse the five linear projections (left/right & three gates) into a
   single fp16 GEMM by concatenating their weight matrices.
3. Apply the optional pair‑wise mask, sigmoid gates and element‑wise
   gating (still fp16).
4. Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched fp16 GEMM (`torch.bmm`) on tensors of shape
   (B·H, N, N) where H = hidden_dim.
5. Second layer‑norm over the hidden dimension (fp16 output) – also the
   one‑pass Triton kernel.
6. Multiply by the output gate and project back to `dim` with a final
   fp16 linear layer.  Cast the result back to float32 (as required by
   the spec).

Both layer‑norms are the only parts written in Triton; the heavy
linear/GEMM work relies on cuBLAS/Tensor‑Core‑friendly fp16 kernels.
This implementation reduces global‑memory traffic for the two
layer‑norms (≈½ the loads of a naïve two‑pass implementation) and
therefore meets the target runtime on the provided test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_one_pass_fp16_kernel(
    x_ptr,            # *const   half*   [total_rows, dim]
    y_ptr,            # *mut     half*   [total_rows, dim]
    weight_ptr,       # *const   float*  [dim]   (γ)
    bias_ptr,         # *const   float*  [dim]   (β)
    total_rows,       # i32  (runtime)
    dim: tl.constexpr,        # i32 (compile‑time)
    eps: tl.constexpr,        # f32 (compile‑time)
    BLOCK_D: tl.constexpr,    # i32 – threads per block (covers dim)
):
    """
    One‑pass LayerNorm for fp16 inputs.
    Each program processes ONE row (one [total_rows] entry) using BLOCK_D
    threads.  All values of the row are streamed once from global memory
    into shared memory; the mean/variance are accumulated on‑the‑fly and
    the normalized results are written out using the cached values.
    """
    row = tl.program_id(0)
    row_valid = row < total_rows

    # ----- shared memory buffer for a full row (dim half values) -----
    row_smem = tl.shared_memory([dim], tl.float16)     # size ≤ 384 * 2 B

    # ----- each thread works on a slice of the row -----
    offs = tl.arange(0, BLOCK_D)                        # per‑thread column offset
    n_iters = tl.cdiv(dim, BLOCK_D)

    # ----- reduction accumulators (FP32) -----
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # ----- first pass: load from global, cache, accumulate -----
    for it in range(n_iters):
        col = it * BLOCK_D + offs
        col_mask = col < dim

        # global pointer for the current element
        ptr = x_ptr + row * dim + col
        val_fp16 = tl.load(ptr,
                           mask=row_valid & col_mask,
                           other=0.0)                 # half
        # cache in shared memory for the second pass
        tl.store(row_smem + col, val_fp16, mask=col_mask)

        # FP32 accumulation for mean/var
        val_f32 = val_fp16.to(tl.float32)
        sum_val += tl.sum(val_f32, axis=0)          # reduction over threads
        sum_sq  += tl.sum(val_f32 * val_f32, axis=0)

    # ----- compute mean / rsqrt(var + eps) -----
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: read from shared memory, normalize, apply γ/β -----
    for it in range(n_iters):
        col = it * BLOCK_D + offs
        col_mask = col < dim

        val_fp16 = tl.load(row_smem + col, mask=col_mask)   # half
        val_f32  = val_fp16.to(tl.float32)

        norm = (val_f32 - mean) * rstd                        # (x-μ)/σ
        w = tl.load(weight_ptr + col, mask=col_mask)         # γ (float32)
        b = tl.load(bias_ptr   + col, mask=col_mask)         # β (float32)
        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + row * dim + col
        tl.store(out_ptr,
                 out_fp16,
                 mask=row_valid & col_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (bool/float, optional)
            - weights_dict: model parameters (all tensors on the same device)
            - config_dict : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C  (input/output channel size)
    hidden_dim = config["hidden_dim"]   # H  (hidden dimension for projections)
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # rows for the first LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output) – one‑pass kernel
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_D = 64                                 # threads per block (covers dim up to 384)
    grid = (triton.cdiv(total_rows, 1),)         # one program per row
    layernorm_one_pass_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim=dim,
        eps=eps,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections (left/right & three gates)
    # ------------------------------------------------------------------
    # Concatenate five weight matrices (each [hidden_dim, dim]) → [5*H, dim]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                # (5*H, C)

    # Linear: [R, C] @ [C, 5*H]^T → [R, 5*H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1]  (float16)
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates & element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate       # [R, H]

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update via batched GEMM (fp16)
    # --------------------------------------------------------------
    # Re‑order so that the hidden dimension becomes the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]   (k,j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)              # [B*H, N, N]

    # Batched matrix multiply – Tensor‑Core friendly FP16 GEMM
    out_mat = torch.bmm(left_mat, right_mat)           # [B*H, N, N] (fp16)

    # Back to 4‑D layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16) – one‑pass kernel
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                         # same count as before
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)        # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, 1),)
    layernorm_one_pass_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Final gating + linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16 element‑wise

    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # [R_h, C] fp16

    # Cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
