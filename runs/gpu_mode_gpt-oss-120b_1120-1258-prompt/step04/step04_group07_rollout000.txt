"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.
---------------------------------------------------------

We implement the forward of the “outgoing” TriMul module from AlphaFold‑3.
The heavy parts are:

1️⃣ Row‑wise LayerNorm on the 4‑D input tensor – custom Triton kernel.
2️⃣ Fuse the five linear layers (left/right projection + three gates) into a
   single half‑precision GEMM.
3️⃣ Apply optional mask, sigmoid gates and reshape to [B,N,N,hidden].
4️⃣ Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   = (left @ rightᵀ) for every head, realised as one batched FP16 matrix
   multiplication (torch.bmm) over the combined B·hidden dimension.
5️⃣ Second LayerNorm (again Triton), element‑wise output‑gate and final
   FP16 linear projection back to the original dimension.
6️⃣ Cast the result back to float32.

All arithmetic (except the final cast) is performed in FP16, leveraging
Tensor‑Core throughput on H100.  The only custom kernel is the
row‑wise LayerNorm, which is used twice (pre‑ and post‑update).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const *  [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,          # mutable *[total_rows, dim]   (fp16 output)
    w_ptr,          # const *  [dim]               (fp32)
    b_ptr,          # const *  [dim]               (fp32)
    total_rows,     # int64   (runtime)
    dim,            # int32   (compile‑time)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128‑256)
    BLOCK_D: tl.constexpr,   # columns per tile (e.g. 32‑64)
):
    """Row‑wise LayerNorm.
    Computes (x - μ) / sqrt(σ²+eps) * w + b  →  fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # 1st pass: accumulate sum and sum‑of‑squares for mean / var.
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in tiles of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to the tile
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load with masks (out‑of‑bounds → 0)
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)            # sum per row
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)  # sum‑sq per row

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # -------------------------------------------------------------
    # 2nd pass: write normalized result (apply weight & bias)
    # -------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        # (x - μ) * rstd
        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        # γ & β (weight & bias) are FP32 → broadcast to rows
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on same device).
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    total_rows = B * N * N                           # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp16 output
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()    # [R, dim]  (fp32)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # launch parameters (tuned for dim ≤ 384)
    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (proj + three gates) into one GEMM
    # --------------------------------------------------------------
    # weight shape: [5*hidden, dim] (FP16)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [hidden, dim]
            weights["right_proj.weight"],  # [hidden, dim]
            weights["left_gate.weight"],   # [hidden, dim]
            weights["right_gate.weight"],  # [hidden, dim]
            weights["out_gate.weight"],    # [hidden, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # Linear: [R, dim] @ [5*hidden, dim].T  →  [R, 5*hidden] (FP16)
    proj_concat = F.linear(x_norm_fp16, w_concat)   # bias=False

    H = hidden
    left_proj   = proj_concat[:, 0:H]                     # [R, H]
    right_proj  = proj_concat[:, H:2*H]
    left_gate_p = proj_concat[:, 2*H:3*H]
    right_gate_p= proj_concat[:, 3*H:4*H]
    out_gate_p  = proj_concat[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoids (still FP16) and optional mask
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)      # [R, H]
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)       # [R, H]

    if not cfg.get("nomask", True):
        # mask: [B,N,N] → flatten → [R,1] (FP16)
        mask_h = mask.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply gates & reshape to 4‑D tensors
    # --------------------------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # will be used after 2nd LN

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update  (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Convert to shape [B*H, N, N] for torch.bmm
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection and cast back to FP32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (FP16)

    # Final linear: [R, H] @ [dim, H].T  →  [R, dim] (FP16)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # bias=False

    # Cast to FP32 and restore original layout
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
