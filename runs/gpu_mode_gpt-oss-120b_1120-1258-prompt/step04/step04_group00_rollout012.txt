Pseudo:

@triton.jit
def apply_gate_kernel(proj_ptr, gate_pre_ptr, out_ptr,
                     mask_ptr,   # maybe
                     rows, hidden_dim,
                     has_mask: tl.constexpr,
                     BLOCK_M: tl.constexpr,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load proj and gate_pre
        proj = tl.load(proj_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        gate_pre = tl.load(gate_pre_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        # compute sigmoid
        gate = 1.0 / (1.0 + tl.exp(-gate_pre))  # approximate

        # optionally mask
        if has_mask:
            mask_val = tl.load(mask_ptr + offs_m,
                               mask=mask_m,
                               other=1.0)[:, None]   # shape [BLOCK_M,1]
            proj = proj * mask_val

        out_val = proj * gate

        # store
        tl.store(out_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                 out_val, mask=mask_m[:, None] & mask_d[None, :])

Note: mask_ptr is shape rows (or rows,1). We'll store mask_val repeated across hidden_dim.

@triton.jit
def apply_gate_kernel(proj_ptr, gate_pre_ptr, out_ptr, mask_ptr,
                      rows, hidden_dim,
                      HAS_MASK: tl.constexpr,
                      BLOCK_M: tl.constexpr,
                      BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < rows

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # compute pointer offsets
        proj = tl.load(proj_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        gate_pre = tl.load(gate_pre_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)

        # sigmoid gate
        gate = 1.0 / (1.0 + tl.exp(-gate_pre))

        # apply mask
        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_m,
                               mask=mask_m,
                               other=1.0)[:, None]  # [BLOCK_M, 1]
            proj = proj * mask_val

        out = proj * gate

        # store
        tl.store(out_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                 out,
                 mask=mask_m[:, None] & mask_d[None, :])

"""
custom_kernel – TriMul “outgoing” (AlphaFold‑3) forward pass

Algorithm (all heavy arithmetic is FP16, LayerNorm is computed in FP32 for
numerical stability):
1️⃣  Row‑wise LayerNorm on the input [B,N,N,D] → FP16.
2️⃣  Two fused Linear layers produce (proj , gate_pre) for the left and right
    paths (weights are concatenated → one GEMM per path).
3️⃣  A tiny Triton kernel fuses:   proj * mask (optional) * sigmoid(gate_pre)
    → left / right tensors of shape [rows, H] (rows = B·N·N).
4️⃣  Out‑gate is computed by a regular Linear + torch.sigmoid.
5️⃣  Tri‑multiplicative update:
        out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
    implemented as a batched GEMM (B·H independent N×N → N×N
    mat‑multiplications) using Torch’s Tensor‑Core kernels.
6️⃣  Second LayerNorm fused with the out‑gate (one Triton kernel).
7️⃣  Final Linear (hidden → D) and cast to float32.

Only the LayerNorms and the element‑wise gating are written in Triton;
the matrix multiplications use cuBLAS/Tensor‑Core kernels for maximal
throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const half*   [total_rows, dim] – input
    y_ptr,          #       half*   [total_rows, dim] – normalized output
    w_ptr,          # const half*  [dim]   (γ)
    b_ptr,          # const half*  [dim]   (β)
    total_rows,     # int – number of rows
    dim: tl.constexpr,          # static number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)                # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized result (γ,β are half → cast to FP32)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w + b                     # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,          # const half*   [rows, dim] – input
    y_ptr,          #       half*   [rows, dim] – normalized & gated output
    w_ptr,          # const half*  [dim]   (γ)
    b_ptr,          # const half*  [dim]   (β)
    gate_ptr,       # const half*  [rows, dim] – element‑wise gate (optional)
    rows,           # int – number of rows
    dim: tl.constexpr,          # static columns
    eps: tl.constexpr,          # static epsilon
    HAS_GATE: tl.constexpr,     # compile‑time bool, 1 → multiply by gate
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------- mean / var (FP32) -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------- normalized + (γ,β) + optional gate -------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w + b                 # FP32

        if HAS_GATE:
            g = tl.load(
                gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=1.0
            ).to(tl.float32)               # half → fp32
            out = out * g

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def apply_gate_kernel(
    proj_ptr,       # const half*   [rows, dim]  – projection (already masked)
    gate_pre_ptr,   # const half*   [rows, dim]  – pre‑sigmoid gate
    out_ptr,        #       half*   [rows, dim]  – result = proj * sigmoid(gate_pre)
    mask_ptr,       # const half*   [rows]       – optional pairwise mask
    rows,           # int – number of rows
    dim: tl.constexpr,          # static columns (hidden_dim)
    HAS_MASK: tl.constexpr,     # compile‑time bool, 1 → apply mask
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load projection and gate pre‑sigmoid
        proj = tl.load(
            proj_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )
        gate_pre = tl.load(
            gate_pre_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )

        # sigmoid
        gate = 1.0 / (1.0 + tl.exp(-gate_pre))

        # optional mask (broadcast over dim)
        if HAS_MASK:
            m = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)[:, None]  # [BLOCK_M,1]
            proj = proj * m

        out = proj * gate

        tl.store(
            out_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out,
            mask=mask_m[:, None] & mask_d[None, :]
        )

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, D]  (float32)
        mask_tensor  : torch.Tensor [B, N, N]   (float32) – optional
        weights_dict : mapping containing all linear / layer‑norm parameters
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (default True)
        }

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    rows = B * N * N               # total pairwise positions

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    # Convert input to FP16 (the kernel works on half‑precision data)
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    x_flat = x_fp16.view(rows, dim)                     # [rows, dim]

    # LayerNorm parameters – store as FP16 to cut memory traffic
    norm_w = weights["norm.weight"].to(torch.float16, non_blocking=True)
    norm_b = weights["norm.bias"].to(torch.float16, non_blocking=True)

    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        norm_w,
        norm_b,
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape to 4‑D tensor
    x_norm = x_norm_fp16.view(B, N, N, dim)   # [B,N,N,D] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate (one GEMM per side)
    # ------------------------------------------------------------------
    # concatenate proj‑ and gate‑weights (each: [H, D])
    left_w   = torch.cat([weights["left_proj.weight"],
                          weights["left_gate.weight"]], dim=0).to(
        torch.float16, non_blocking=True)                # [2*H, D]
    right_w  = torch.cat([weights["right_proj.weight"],
                          weights["right_gate.weight"]], dim=0).to(
        torch.float16, non_blocking=True)                # [2*H, D]

    # linear: (rows, D) @ (2*H, D).T → (rows, 2*H)
    left_cat  = F.linear(x_norm.view(rows, dim), left_w)   # [rows, 2*H]
    right_cat = F.linear(x_norm.view(rows, dim), right_w)  # [rows, 2*H]

    # split into projection / pre‑gate tensors
    left_proj   = left_cat[:, :hidden_dim]                # [rows, H]
    left_gatepr = left_cat[:, hidden_dim:]                # [rows, H]
    right_proj  = right_cat[:, :hidden_dim]
    right_gatepr= right_cat[:, hidden_dim:]

    # ------------------------------------------------------------------
    # 3️⃣ Mask handling (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
    else:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection * sigmoid(gate) (+ optional mask) via Triton
    # ------------------------------------------------------------------
    left_h  = torch.empty_like(left_proj)   # [rows, H]
    right_h = torch.empty_like(right_proj)

    grid_gate = (triton.cdiv(rows, BLOCK_M),)

    apply_gate_kernel[grid_gate](
        left_proj,
        left_gatepr,
        left_h,
        mask_fp16,
        rows,
        hidden_dim,
        HAS_MASK=0 if nomask else 1,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    apply_gate_kernel[grid_gate](
        right_proj,
        right_gatepr,
        right_h,
        mask_fp16,
        rows,
        hidden_dim,
        HAS_MASK=0 if nomask else 1,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # Release intermediate cat tensors
    del left_cat, right_cat, left_proj, left_gatepr, right_proj, right_gatepr
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 5️⃣ Out‑gate (linear + sigmoid) – no mask
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_pre = F.linear(x_norm.view(rows, dim), out_gate_w)   # [rows, H]
    out_gate = torch.sigmoid(out_gate_pre)                       # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_h.view(B, N, N, hidden_dim)          # [B,N,N,H]
    right = right_h.view(B, N, N, hidden_dim)

    # bring H to batch dimension → [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # out_mat[b*h, i, j] = Σₖ left[i,k]·right[j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    rows_h = rows                       # same as B·N·N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # LayerNorm parameters for the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)

    grid_ln2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_gate_kernel[grid_ln2](
        out_flat,
        out_norm_gated,
        to_out_norm_w,
        to_out_norm_b,
        out_gate.view(rows_h, hidden_dim),   # gate pointer
        rows_h,
        hidden_dim,
        eps,
        HAS_GATE=1,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    final_fp16 = F.linear(out_norm_gated.view(rows_h, hidden_dim), to_out_w)  # [rows, D]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
