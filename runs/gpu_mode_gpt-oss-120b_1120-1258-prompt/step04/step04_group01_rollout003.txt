"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32→FP16) on the input tensor.
2️⃣  Fused linear projection (5 * hidden_dim outputs) using a single FP16 GEMM.
    →  left_proj, right_proj, left_gate, right_gate, out_gate.
3️⃣  Optional pair‑wise mask applied only to the two projections.
4️⃣  Apply sigmoid gates and obtain gated tensors *left* and *right*.
5️⃣  Batched matrix multiplication:
        out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    Implemented with a single torch.matmul on non‑contiguous views
    (no extra copies).
6️⃣  Fuse the second LayerNorm, the output‑gate, and a write‑out to a
    contiguous buffer (R = B·N·N rows, H = hidden_dim columns) with a
    custom Triton kernel.  This kernel:
      • reads the non‑contiguous result of the matmul,
      • computes per‑row mean/var,
      • applies γ/β,
      • multiplies by *out_gate*,
      • writes the final hidden representation in FP16, layout (R, H).
7️⃣  Final linear projection back to the model dimension (FP16 GEMM) and
    cast to FP32.

All heavy kernels run on the GPU (H100) and use FP16 Tensor‑Core
operations; only two small Triton kernels (first LayerNorm & fused
LayerNorm‑gate) are required.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   *  [total_rows, dim]   (fp32)
    y_ptr,                # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,           # const   *  [dim]               (fp32)
    bias_ptr,             # const   *  [dim]               (fp32)
    total_rows,           # int     (runtime)
    dim,                  # int     (compile‑time)
    eps,                  # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # pass 1 – compute mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # pass 2 – write normalized data (γ/β) as fp16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6️⃣  Fused second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    # Input tensor: out_perm shape (B, H, N, N) – non‑contiguous
    src_ptr,                 # const   *  [B, H, N, N] (fp16)
    # Gate tensor: out_gate shape (R, H) – contiguous
    gate_ptr,                # const   *  [R, H]    (fp16)
    # Output buffer: hidden after LN & gate, shape (R, H) – contiguous
    dst_ptr,                 # mutable *  [R, H]    (fp16)
    # LayerNorm parameters (γ/β) over the hidden dimension H
    ln_weight_ptr,           # const   *  [H]       (fp32)
    ln_bias_ptr,             # const   *  [H]       (fp32)
    # Shape / stride information
    total_rows,              # int (runtime)  = B·N·N
    hidden_dim,              # int (compile‑time)
    seq_len,                 # int (runtime)  = N
    stride_b,                # int (runtime)  stride on dim 0 (B) of src
    stride_h,                # int (runtime)  stride on dim 1 (H) of src
    stride_i,                # int (runtime)  stride on dim 2 (N) of src
    stride_j,                # int (runtime)  stride on dim 3 (N) of src
    eps,                     # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- decode (b,i,j) from linear row index -----------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # pass 1 – mean & variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # Load a tile of the hidden vector (strided by stride_h)
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # pass 2 – normalize, apply γ/β, multiply with out_gate and store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # Load raw hidden values
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        # Multiply with out_gate (already fp16 → promote)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # model dimension (e.g., 128 or 384)
    hidden_dim   = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps          = 1e-5
    device       = input_tensor.device
    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N                 # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64                         # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) One fused linear (5 * hidden_dim outputs)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices and cast once to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Linear: [R, dim] @ [dim, 5*H]ᵀ → [R, 5*H]
    proj_out = F.linear(x_norm, w_concat)   # fp16, no bias

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate   = proj_out[:, 2*H:3*H]
    right_gate  = proj_out[:, 3*H:4*H]
    out_gate    = proj_out[:, 4*H:5*H]

    # Sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate  = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 3) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left  = left_proj * left_gate      # [R, H]
    right = right_proj * right_gate    # [R, H]

    # Reshape back to (B, N, N, H) – still contiguous because H is the last dim
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    # out_gate stays flat (R, H) for later use

    # --------------------------------------------------------------
    # 5) Batched matrix multiplication (no extra copies)
    #    out_perm: (B, H, N, N) = left @ rightᵀ
    # --------------------------------------------------------------
    # Use non‑contiguous permutes – torch.matmul accepts strided tensors
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N) view
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N) view
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # --------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Result is a contiguous (R, H) fp16 tensor `hidden_out`.
    # --------------------------------------------------------------
    # Prepare stride information for the source tensor `out_perm`
    stride_b = out_perm.stride(0)   # stride for batch dimension B
    stride_h = out_perm.stride(1)   # stride for hidden dimension H  (= N*N)
    stride_i = out_perm.stride(2)   # stride for dim i (N)
    stride_j = out_perm.stride(3)   # stride for dim j (1)

    hidden_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid_ln2](
        out_perm,                               # src (non‑contig)
        out_gate,                               # gate (contig, [R,H])
        hidden_out,                             # dst (contig)
        weights["to_out_norm.weight"],          # γ (fp32)
        weights["to_out_norm.bias"],            # β (fp32)
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 7) Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    # Convert final weight once to fp16 (shape [dim, H])
    w_out = weights["to_out.weight"].to(torch.float16)

    final_half = F.linear(hidden_out, w_out)   # [R, dim], fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
