"""
custom_kernel – Outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input tensor (shape B·N·N × dim).  
   Implemented in Triton (fp32 accumulation → fp16 output) to avoid a
   costly torch.nn.LayerNorm call on the huge flattened tensor.

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single half‑precision GEMM:
        [R, dim] @ [5·hidden, dim].T → [R, 5·hidden]   (R = B·N·N)

3️⃣ Split the GEMM result into the projected tensors and gate tensors,
   apply the optional mask (only on the two projections) and the sigmoid
   gates.

4️⃣ Triangular multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of size (B·hidden, N, N) ×
   (B·hidden, N, N)ᵀ.  The right matrix is pre‑transposed so we can use
   a single torch.bmm call (Tensor‑Core accelerated).

5️⃣ Second LayerNorm over the hidden dimension (again Triton, fp16).

6️⃣ Apply the output gate, a final half‑precision linear projection back to
   the original `dim`, and cast the result to float32 (as in the reference).

All heavy compute uses FP16/Tensor‑Cores; only two custom Triton kernels
are required (the two LayerNorms).  The implementation works for the
full range of test configurations (different batch sizes, sequence lengths,
mask presence and data distributions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 accumulation → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32 or fp16)
    y_ptr,            # mutable * [total_rows, dim]   (fp16 output)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       #   int   (runtime)
    dim,              #   int   (constexpr)
    eps,              #   float (constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / sqrt(σ²+eps) * γ + β
    where μ,σ² are computed per row.
    The output dtype is fp16; accumulation is done in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute row‑wise mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N]      (float32 or bool)
            - weights_dict : dict of model parameters (float32)
            - config_dict  : {
                  "dim"        : int,
                  "hidden_dim" : int,
                  "nomask"     : bool (optional, default True)
              }

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # flattened rows for LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Tune block sizes: larger BLOCK_M reduces kernel launches.
    BLOCK_M = 1024
    BLOCK_D = 64   # works for dim up to 384 (64 * 6 = 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse 5 linear layers (2 proj + 3 gates) into a single GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: shape [5*hidden_dim, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # half‑precision GEMM weight

    # Linear: [R, dim] @ [5*hidden, dim].T → [R, 5*hidden]
    proj = F.linear(x_norm, w_concat)   # fp16, shape (R, 5*hidden_dim)

    H = hidden_dim
    left_proj   = proj[:, 0:H]
    right_proj  = proj[:, H:2*H]
    left_gate_p = proj[:, 2*H:3*H]
    right_gate_p = proj[:, 3*H:4*H]
    out_gate_p  = proj[:, 4*H:5*H]

    # ---------------------------------
    # 3️⃣ Sigmoid gates (fp16)
    # ---------------------------------
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # ---------------------------------
    # 3️⃣ Optional mask (only on projections)
    # ---------------------------------
    if not config.get("nomask", True):
        # mask → [R, 1]  (broadcast over hidden dim)
        mask_half = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_half
        right_proj = right_proj * mask_half

    # ---------------------------------
    # 4️⃣ Apply gates
    # ---------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ---------------------------------
    # 5️⃣ Reshape for batched GEMM (tri‑mul update)
    # ---------------------------------
    # From [R, H] to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Permute to bring H into the batch dimension.
    # left_perm : (B, H, N, N)   (row‑major)
    # right_perm_T : (B, H, N, N) where each matrix is already transposed
    left_perm      = left.permute(0, 3, 1, 2).contiguous()
    right_perm_T   = right.permute(0, 3, 2, 1).contiguous()   # transpose N‑dims

    # Collapse batch & head dimensions → (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm_T.view(B * H, N, N)

    # Batched matrix multiplication: L @ R   (both fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # shape (B*H, N, N)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, H)   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate & final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate           # element‑wise fp16

    # Final linear projection: [R, H] → [R, dim] (half‑precision)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden)
    )   # fp16, shape (R, dim)

    # Cast to float32 as in the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
