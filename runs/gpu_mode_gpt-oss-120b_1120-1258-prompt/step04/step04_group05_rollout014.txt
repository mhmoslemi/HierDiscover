5️⃣  A second row‑wise LayerNorm (again Triton) brings the hidden
    representation to a stable scale.

6️⃣  Multiply by the *output* gate, then a final fp16 linear
    projection back to the original channel dimension.

All operations stay in fp16 except the two LayerNorm reductions
( fp32 accumulation for numerical stability ).  The final tensor is
returned in float32 as required.

The only Triton code is the fused LayerNorm kernel; the heavy
matrix‑multiply work relies on cuBLAS Tensor‑Core kernels, which
are optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (fp32 reduction → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut    float16* [total_rows, dim]   (fp16 output)
    w_ptr,          # *const  float*   [dim]               (γ)
    b_ptr,          # *const  float*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1. mean / variance (fp32) – one pass over the channel dim
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [row, col]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        sum_val += tl.sum(vals, axis=1)        # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2. write normalized output (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict : mapping of weight names to tensors (float32)
            - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    device = input_tensor.device
    dim = cfg["dim"]               # C
    hidden_dim = cfg["hidden_dim"] # H
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    # Use the Triton row‑wise LayerNorm defined above.
    x_fp32 = input_tensor  # already fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Run kernel
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_fp32,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Split the five linear projections into two fused GEMMs.
    #    – left/right (2·H) are *masked* before the projection,
    #    – three gates (3·H) are unmasked.
    # ------------------------------------------------------------------
    # Convert all weight matrices once to fp16.
    w_left   = weights["left_proj.weight"].half()
    w_right  = weights["right_proj.weight"].half()
    w_lgate  = weights["left_gate.weight"].half()
    w_rgate  = weights["right_gate.weight"].half()
    w_outgate= weights["out_gate.weight"].half()

    # Concatenate for the two fused GEMMs
    w_lr = torch.cat([w_left, w_right], dim=0)                     # (2·H, C)
    w_gate = torch.cat([w_lgate, w_rgate, w_outgate], dim=0)      # (3·H, C)

    # Flatten input to (total_rows, C) and optionally apply the mask **before**
    # the projection (scalar per row, so it commutes with the linear layer).
    x_flat = x_norm_fp16.view(total_rows, dim)                     # fp16

    if not nomask:
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        x_masked = x_flat * mask_fp16
    else:
        x_masked = x_flat

    # ---- left / right projections (masked) --------------------------------
    proj_lr = F.linear(x_masked, w_lr)          # (total_rows, 2·H)
    left_proj_fp16, right_proj_fp16 = torch.split(
        proj_lr, hidden_dim, dim=1
    )                                            # each (total_rows, H)

    # ---- gate projections (unmasked) --------------------------------------
    proj_gate = F.linear(x_flat, w_gate)        # (total_rows, 3·H)
    left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_gate, hidden_dim, dim=1
    )                                            # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates and element‑wise gating (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate        # (total_rows, H)
    right = right_proj_fp16 * right_gate     # (total_rows, H)

    # Restore the 4‑D layout needed for the triangular update
    left = left.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim) # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (outgoing):
    #     out_h[i, j] = Σ_k left[i, k, h] * right[j, k, h]
    # --------------------------------------------------------------
    # Bring the hidden dimension into the batch dimension.
    # The contiguity operation is required because the hidden axis is not
    # stored contiguously for each (i, j) pair.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                  # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                 # (BH, N, N)

    # Batched GEMM: left @ rightᵀ   →   (BH, N, N)   (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # shape [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16 output) – same Triton kernel
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                # element‑wise (fp16)

    gated_flat = gated.view(total_rows, hidden_dim)   # (total_rows, H)
    final_fp16 = F.linear(gated_flat, weights["to_out.weight"].half())  # (total_rows, C)

    # --------------------------------------------------------------
    # 7️⃣ Cast back to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
