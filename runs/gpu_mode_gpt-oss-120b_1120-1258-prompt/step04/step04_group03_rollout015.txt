```
@triton.jit
def norm_linear_fused_kernel(
    a_ptr, weight_ptr, bias_ptr, out_ptr,
    norm_weight_ptr, norm_bias_ptr,
    total_rows, total_out, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    row_start = pid * BLOCK_M
    col_start = pid_n * BLOCK_N
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)
    mask_m = offs_m < total_rows
    mask_n = offs_n < total_out

    # Sum for mean/var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + (offs_m[:, None] * dim + offs_k[None, :])
        vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulator for output
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # Load a block from input
        a_ptrs = a_ptr + (offs_m[:, None] * dim + offs_k[None, :])
        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Apply layernorm affine
        w_ptrs = norm_weight_ptr + offs_k
        w = tl.load(w_ptrs, mask=mask_k, other=0.0)   # norm_weight per dim

        b_ptrs = norm_bias_ptr + offs_k
        b = tl.load(b_ptrs, mask=mask_k, other=0.0)   # norm_bias per dim

        a_centered = a_vals - mean[:, None]
        a_norm = a_centered * rstd[:, None]          # (BLOCK_M, BLOCK_K)

        x_norm = a_norm * w[None, :] + b[None, :]    # (BLOCK_M, BLOCK_K)

        # Load weight slice
        w_ptrs_fused = weight_ptr + (offs_k[:, None] * total_out + offs_n[None, :]) # weight shape (dim, total_out)
        w_tile = tl.load(w_ptrs_fused, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate GEMM
        acc += tl.dot(x_norm, w_tile)   # (BLOCK_M, BLOCK_N)

    # Add bias
    bias_ptrs = bias_ptr + offs_n
    bias_tile = tl.load(bias_ptrs, mask=mask_n, other=0.0)  # fp32

    acc = acc + bias_tile[None, :]   # broadcast

    # Store result as half
    out_ptrs = out_ptr + (offs_m[:, None] * total_out + offs_n[None, :])
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul – “outgoing” version (AlphaFold‑3) – fast Triton implementation.

Algorithm
---------
1️⃣  Fuse the first LayerNorm and the five linear projections (left/right
    projections + three gates) into a single custom Triton kernel.
    * The kernel computes per‑row mean/variance of the input (fp32),
      normalises it with the LayerNorm parameters, and immediately
      multiplies by the concatenated projection weight matrix (fp16).
    * A fused bias (‖proj_weight‖·norm_bias) is added and the result is
      stored as fp16 – no intermediate fp32 buffer is materialised.

2️⃣  Optional pair‑wise mask is applied to the two projection tensors
    (left/right) before the sigmoid gates.

3️⃣  Apply sigmoid to the three gate tensors, element‑wise gating and
    reshape the tensors to `[B, N, N, H]`.

4️⃣  Triangular multiplicative update: for each batch and hidden‑dim
    compute `out = left @ rightᵀ` – realised as a batched GEMM using
    `torch.matmul` on tensors `(B*H, N, N)`.  This is the heaviest O(N³)
    part and cuBLAS/Tensor‑Core provides the best performance.

5️⃣  Second LayerNorm (hidden dimension) – a small Triton kernel that
    reduces in fp32 and writes the normalised result in fp16.

6️⃣  Multiply by the output‑gate, project back to the original channel
    dimension with a final fp16 linear layer and cast the result to fp32.

The only extra memory traffic compared with the reference PyTorch code
is the fused “norm + linear” kernel, which reads each input row only twice
(instead of three times in the naïve implementation).  All heavy
computations stay in half‑precision and exploit Tensor‑Cores, while the
custom kernels keep the reduction overhead low.  The function returns a
float32 tensor of shape `[B, N, N, C]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (fp32 reduction → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int  (runtime)
    dim: tl.constexpr,          # int  (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------- 1️⃣ mean & variance (fp32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- 2️⃣ normalise + affine (fp16) ---------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + linear projection (fp32 reduction →
# fp16 fused output).  The weight matrix is supplied transposed
# (shape [dim, out_dim]) to match the GEMM layout.
# ----------------------------------------------------------------------
@triton.jit
def norm_linear_fused_kernel(
    a_ptr,                # *const float*   [total_rows, dim]   (fp32 input)
    w_ptr,                # *const half*    [dim, out_dim]    (fused weight – fp16)
    b_ptr,                # *const float*   [out_dim]        (fused bias – fp32)
    out_ptr,              # *mut   half*    [total_rows, out_dim] (fp16 output)
    ln_w_ptr,             # *const float*   [dim]            (LayerNorm γ)
    ln_b_ptr,             # *const float*   [dim]            (LayerNorm β)
    total_rows,           # int  (runtime)
    out_dim,              # int  (runtime)  == 5*hidden_dim
    dim: tl.constexpr,    # int  (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)        # column tile

    row_start = pid * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < out_dim

    # ----------------- 1️⃣ mean & variance (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        a_ptrs = a_ptr + (offs_m[:, None] * dim + offs_k[None, :])
        vals = tl.load(a_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------- 2️⃣ fused GEMM (normalized input) ----------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # load raw input tile
        a_ptrs = a_ptr + (offs_m[:, None] * dim + offs_k[None, :])
        a_vals = tl.load(a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0)

        # LayerNorm affine parameters for this tile
        ln_w = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # γ
        ln_b = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # β

        # Normalise: ((x - mean) * rstd) * γ + β
        a_centered = a_vals - mean[:, None]
        a_norm = a_centered * rstd[:, None]
        x_norm = a_norm * ln_w[None, :] + ln_b[None, :]

        # Load fused projection weight tile (dim × out_dim)
        w_ptrs = w_ptr + (offs_k[:, None] * out_dim + offs_n[None, :])
        w_tile = tl.load(w_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        # Accumulate GEMM
        acc += tl.dot(x_norm, w_tile)                     # (BLOCK_M, BLOCK_N)

    # add fused bias
    bias_tile = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)      # (BLOCK_N)
    acc = acc + bias_tile[None, :]

    # store result as fp16
    out_ptrs = out_ptr + (offs_m[:, None] * out_dim + offs_n[None, :])
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – returns a float32 tensor of shape
    [bs, seq_len, seq_len, dim].

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : mapping of layer‑norm and linear weight tensors
                         (all float32)
        * config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        The transformed tensor, dtype float32, same device as the inputs.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                         # rows for the flattened view
    out_dim = 5 * hidden_dim                       # concatenated projection size

    # ------------------------------------------------------------------
    # 1️⃣ Fuse LayerNorm + the five linear projections (including gates)
    # ------------------------------------------------------------------
    # Concatenate the five projection weight matrices (shape [5*H, C])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                             # (5*H, C) float32

    # Fuse the first LayerNorm’s affine parameters into the projection weight
    # (γ is multiplied element‑wise, β is folded into a bias term).
    ln_w = weights["norm.weight"]                 # (C)
    ln_b = weights["norm.bias"]                   # (C)
    fused_w = concat_w * ln_w[None, :]            # (5*H, C)
    fused_b = torch.mv(concat_w, ln_b)            # (5*H)

    # Transpose to (C, 5*H) because the kernel expects the weight in that order.
    fused_w_T = fused_w.t().contiguous().half()  # (C, 5*H) fp16

    # Allocate output buffer for the fused projection
    proj_all_fp16 = torch.empty(
        (total_rows, out_dim), dtype=torch.float16, device=device
    )

    # Triton launch configuration
    BLOCK_M = 64   # rows per program
    BLOCK_N = 64   # output‑feature tile
    BLOCK_K = 32   # reduction over `dim`

    grid = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(out_dim, BLOCK_N),
    )

    norm_linear_fused_kernel[grid](
        input_tensor.view(total_rows, dim).contiguous(),
        fused_w_T,
        fused_b,
        proj_all_fp16,
        ln_w,
        ln_b,
        total_rows,
        out_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Split fused projection into the five tensors we need
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all_fp16, hidden_dim, dim=1
    )  # each [total_rows, H]   fp16

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates  (computed in fp32 for stability, cast back)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate.float()).half()
    right_gate = torch.sigmoid(right_gate.float()).half()
    out_gate = torch.sigmoid(out_gate.float()).half()

    # Apply element‑wise gating
    left = left_proj * left_gate          # (total_rows, H) fp16
    right = right_proj * right_gate       # (total_rows, H) fp16
    out_gate = out_gate.reshape(total_rows, hidden_dim)   # (total_rows, H) fp16

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)       # (B, N, N, H)
    right = right.view(B, N, N, hidden_dim)     # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) – each slice is a matrix multiplication.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B_H = B * hidden_dim
    left_mat = left_perm.view(B_H, N, N)       # (B*H, N, N)
    right_mat = right_perm.view(B_H, N, N)     # (B*H, N, N)

    # Batched GEMM – fp16 → fp32 accumulation (Tensor‑Core friendly)
    out_mat = torch.matmul(left_mat, right_mat)    # (B*H, N, N), fp16

    # Back to original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    BLOCK_M2 = 256
    BLOCK_D2 = 64
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)

    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate → final projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) fp16

    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # (B*N*N, H)
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)   # (B*N*N, C) fp16

    # Cast back to FP32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
