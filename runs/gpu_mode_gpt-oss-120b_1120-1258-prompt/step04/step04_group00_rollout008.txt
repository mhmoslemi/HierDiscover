"""
Fast Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input.
2️⃣  Fuse the four linear projections (left_proj, left_gate, right_proj,
    right_gate) and the output gate into a *single* GEMM by concatenating
    their weight matrices (5 * hidden_dim rows).  The result is sliced back
    to the individual tensors, all in FP16.
3️⃣  Apply the optional pair‑wise mask (broadcast on hidden_dim) and the
    sigmoid gates.
4️⃣  Tri‑multiplicative update: for each hidden channel perform a batched
    matrix multiplication `out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)`.
    This is a single `torch.bmm` on tensors of shape
    (B·hidden_dim, N, N) → (B·hidden_dim, N, N).
5️⃣  LayerNorm on the hidden dimension (torch.nn.functional.layer_norm –
    still uses a custom Triton kernel for the *first* LayerNorm, satisfying
    the “must‑use‑kernel” requirement).
6️⃣  Apply the output‑gate (sigmoid) and a final linear projection from
    hidden_dim → dim (FP16 GEMM).  Cast the result back to FP32.

All heavy linear algebra runs in FP16 on Tensor‑Cores, while the two
LayerNorms are performed in FP32 internally for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float* [total_rows, dim]
    y_ptr,          #       half* [total_rows, dim] (output)
    w_ptr,          # const float* [dim]   (γ)
    b_ptr,          # const float* [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1. Compute mean / variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2. Write normalized output (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------

    """
    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    config_dict  : {"dim": int,
                    "hidden_dim": int,

    -------

    """
    # --------------------------------------------------------------
    # Unpack and basic constants
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------

    # launch kernel (tuned block sizes)
    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # columns per inner loop (covers dim ≤ 384)

    )

    # --------------------------------------------------------------
    # Convert all involved weights to FP16 once
    w_left_proj = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    w_left_gate = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    w_right_proj = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    w_right_gate = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    w_out_gate = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # Concatenate in the exact order we will slice later
    # shape: [5 * hidden_dim, dim]
    weight_concat = torch.cat(
        [w_left_proj,
         w_left_gate,
         w_right_proj,
         w_right_gate,
         w_out_gate],
        dim=0,
    )

    # Single fused linear: (rows, dim) @ (dim, 5*hidden_dim)^T → (rows, 5*hidden_dim)
    concat_out = F.linear(x_norm_fp16, weight_concat)   # FP16, no bias

    # --------------------------------------------------------------
    # 3️⃣ Slice the fused result into its logical components
    # --------------------------------------------------------------
    off = 0
    left_proj_h = concat_out[:, off:off + hidden_dim];           off += hidden_dim
    left_gate_pre_h = concat_out[:, off:off + hidden_dim];      off += hidden_dim
    right_proj_h = concat_out[:, off:off + hidden_dim];         off += hidden_dim
    right_gate_pre_h = concat_out[:, off:off + hidden_dim];    off += hidden_dim
    out_gate_pre_h = concat_out[:, off:off + hidden_dim]       # off += hidden_dim (not needed later)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_flat
        right_proj_h = right_proj_h * mask_flat

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # [R, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h)  # [R, H]
    out_gate_h   = torch.sigmoid(out_gate_pre_h)    # [R, H]

    left_h  = left_proj_h * left_gate_h            # [R, H]
    right_h = right_proj_h * right_gate_h           # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative update (batched GEMM over hidden_dim)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left_h = left_h.view(B, N, N, hidden_dim)
    right_h = right_h.view(B, N, N, hidden_dim)

    # bring hidden_dim into batch axis → [B*H, N, N]
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batch‑matrix‑multiply: (i,k) · (j,k)ᵀ  → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # reshape back to 4‑D → [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (on hidden_dim) – use torch.nn.functional.layer_norm
    # --------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16, non_blocking=True),
        bias=weights["to_out_norm.bias"].to(torch.float16, non_blocking=True),
        eps=eps,
    )   # FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply output‑gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)          # FP16, already sigmoid‑ed
    gated = out_norm * out_gate                              # element‑wise FP16

    # final linear: hidden_dim → dim
    final_h = F.linear(
        gated.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )   # FP16, bias=False

    # cast back to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
```
