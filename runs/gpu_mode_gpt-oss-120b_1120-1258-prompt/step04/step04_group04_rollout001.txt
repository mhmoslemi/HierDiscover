"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Fuse the first LayerNorm (LN) into a small Triton kernel that
   reads fp32 input, computes per‑row mean/var in fp32, writes
   the normalized result as fp16.  This avoids the two‑pass
   PyTorch LN and gives us an fp16 tensor that can be fed
   directly to the linear projections.

2️⃣ Perform a single fused linear projection for the five
   weight matrices (left_proj, right_proj, left_gate,
   right_gate, out_gate) by concatenating the weight tensors.
   The operation is a standard fp16 GEMM (`torch.nn.functional.linear`).

3️⃣ Optional pair‑wise mask is applied *in‑place* on the left/right
   projections (still fp16).  Gates are computed with a fast
   sigmoid (`torch.sigmoid`) and multiplied into the projected
   tensors.

4️⃣ Triangular multiplicative update:
   For each hidden channel we need a matrix product
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realised as a batched fp16 GEMM on tensors of shape
   (B·H, N, N) where H = hidden_dim.  The transposes are performed
   by simple view/permute operations that are *no‑copy*.

5️⃣ Second LayerNorm over the hidden dimension (same Triton kernel
   as in step‑1).  The kernel writes fp16 results directly to the
   final buffer, eliminating an intermediate allocation.

6️⃣ Apply the output gate, then a final fp16 linear projection back
   to the original channel dimension (dim).  The result is cast to
   float32 as required by the specification.

All heavy reductions (the two LayerNorms) are performed by Triton,
the dense linear algebra uses Tensor‑Core‑friendly fp16 GEMM
operations from cuBLAS.  The implementation works for every
configuration listed in the test suite (variable batch size,
sequence length, presence/absence of mask, and different channel
sizes).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const   *fp32*  [total_rows, dim]
    y_ptr,          # mutable *fp16*  [total_rows, dim]
    weight_ptr,     # const   *fp32*  [dim]   (γ)
    bias_ptr,       # const   *fp32*  [dim]   (β)
    total_rows,     # i32 runtime
    DIM: tl.constexpr,          # compile‑time channel size
    EPS: tl.constexpr,          # compile‑time epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile (should divide DIM)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute mean and variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        # Load a tile sized [BLOCK_M, BLOCK_D]
        ptr = x_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        ptr = x_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # Apply γ (weight) and β (bias) – both stored as fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device).contiguous()  # (5H, C)

    proj_all = F.linear(x_norm_fp16, concat_weight)            # (R, 5H) fp16

    # Split into the five tensors, each (R, H)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (broadcast over hidden dim) – in‑place
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] for broadcasting
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Gates (sigmoid) and element‑wise gating – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape for the triangular multiplicative update
    # ------------------------------------------------------------------
    # Shapes: [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)

    # Bring the hidden dimension to the batch axis for a single bmm call.
    # left_perm   → (B, H, N, N) where axis order = (batch, channel, i, k)
    # right_perm  → (B, H, N, N) where axis order = (batch, channel, j, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # swap i ↔ j for the second operand

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # (BH, N, N)  – (i,k)
    right_mat = right_perm.view(BH, N, N)    # (BH, N, N)  – (j,k)

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (fp16 Tensor Cores) → out shape (BH, N, N)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # fp16, uses cuBLAS

    # Reshape back to 4‑D: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden_dim (fp16) – Triton kernel again
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                     # rows = B·N·N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        DIM=hidden_dim,
        EPS=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating by out_gate and final linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # fp16 element‑wise

    # Final linear: hidden_dim → dim (no bias)
    to_out_weight_fp16 = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    gated_flat = gated.view(total_rows_h, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # (R_h, C) fp16

    # Cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
```
