"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Layer‑Norm over the last channel (dim) – custom Triton kernel.
    The kernel reduces in fp32, writes the normalized result in fp16.

2️⃣  Fuse the five linear layers (left/right projections + three gates)
    into a single half‑precision GEMM by concatenating their weight
    matrices.  The GEMM reads the normalized tensor only once.

3️⃣  Optional pair‑wise mask is applied **after** the two projections
    (exactly as in the reference implementation).

4️⃣  Apply the three sigmoid gates and element‑wise gating.

5️⃣  Triangular multiplicative update: for every head h we need
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is realised as a batched half‑precision matrix multiply with
    shape (B·H, N, N) × (B·H, N, N) → (B·H, N, N).

6️⃣  Second Layer‑Norm (over the hidden dimension) – same Triton kernel
    (the Tensor‑Core friendly half‑precision version).

7️⃣  Multiply by the output gate and project back to `dim` with a final
    half‑precision linear layer.  The result is cast to fp32 for the
    API contract.

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra is delegated to cuBLAS/Tensor‑Core
via regular PyTorch ops.  The implementation works for any supported
batch‑size, sequence‑length and channel dimensions and respects the
`nomask` flag (skip mask handling when it is `True`).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (fp32 reduction → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int  (runtime)
    dim: tl.constexpr,          # int  (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣  Compute per‑row mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Loop over the channel dimension in tiles
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load one tile (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2️⃣  Write normalized + affine (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load the same tile again (still fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # Normalisation
        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        # Affine parameters (γ, β) – fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of model weights (all float32)
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C], dtype float32, same device as inputs.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape constants
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # flatten across (i,j) pairs

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Tune block sizes – small dim (≤384) fits nicely in BLOCK_D = 64
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (still fp16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # (5*H, C)

    # Linear projection on the normalized tensor (FP16 GEMM)
    proj_all = F.linear(x_norm_fp16, concat_w)          # (total_rows, 5*H)

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate  = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16  = left_proj * left_gate   # (total_rows, H)
    right_fp16 = right_proj * right_gate # (total_rows, H)

    # Reshape to 4‑D tensors for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange so that the summed‑over dimension is the inner dimension:
    #   left  : (B, H, N, N)  with axes (b, h, i, k)
    #   right : (B, H, N, N)  with axes (b, h, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Merge batch and head dimensions for a strided‑batched GEMM
    B_H = B * hidden_dim
    left_mat  = left_perm.view(B_H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B_H, N, N)         # (B*H, N, N)

    # Half‑precision matmul → (B*H, N, N)
    out_mat = torch.matmul(left_mat, right_mat)    # fp16

    # Restore the original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) fp16

    # Final linear projection H → C (FP16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C)

    # Cast to FP32 for the public API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
