"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape [B, N, N, C]).
   Implemented in a custom Triton kernel that writes the
   normalized result directly in fp16.

2️⃣ Fuse the five dense layers (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single half‑precision
   GEMM:
        X_norm_flat @ W_concat.T   →   [R, 5·H]   (R = B·N·N)

3️⃣ Slice the concatenated result, apply the optional mask
   (broadcasted over the hidden dimension) and the sigmoid gates.
   This yields the gated tensors `left` and `right` of shape
   [B, N, N, H].

4️⃣ Triangular multiplicative update (the “outgoing” version):
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix‑multiplication
   `left @ rightᵀ` for every hidden channel.  We use
   `torch.bmm` on the reshaped tensors (B·H, N, N) – the most
   efficient atomic implementation on an H100.

5️⃣ Second LayerNorm over the hidden dimension (again a Triton kernel).

6️⃣ Apply the output gate, project back to the original dimension
   with a final half‑precision linear layer and cast the result
   to float32.

All heavy arithmetic runs in fp16, exploiting Tensor‑Core
throughput.  The only custom kernels are the two
row‑wise LayerNorms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (no bias in the reduction step)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (fp32)
    y_ptr,          # mutable *[total_rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]               (fp32)  γ
    bias_ptr,       # const   *[dim]               (fp32)  β
    total_rows,     #   int   (runtime)
    dim,            #   int   (compile‑time)
    eps,            #   float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # tile over the feature dimension
):
    pid = tl.program_id(0)                     # one program per row‑tile
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------- 1st pass: mean & variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1/√(var+eps)

    # --------------------- 2nd pass: write normalized result ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# TriMul forward – custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast Triton implementation of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (bool/float, may be dummy)
        - weights_dict : str → torch.Tensor (float32) containing all model parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32), on the same device as the inputs.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise, fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Triton launch config – tuned for typical dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                                   # input fp32
        x_norm,                                   # output fp16
        weights["norm.weight"],                   # γ (fp32)
        weights["norm.bias"],                     # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (proj + three gates) into one GEMM
    # ------------------------------------------------------------------
    # Build concatenated weight matrix of shape [5·H, C] (half‑precision)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, C]

    # Single half‑precision linear transformation → [R, 5·H]
    proj_concat = F.linear(x_norm, w_concat)            # bias‑free

    H = hidden_dim
    # Slice the concatenated result
    left_proj      = proj_concat[:, 0:H]                     # [R, H]
    right_proj     = proj_concat[:, H:2*H]                   # [R, H]
    left_gate_pre  = proj_concat[:, 2*H:3*H]                 # [R, H]
    right_gate_pre = proj_concat[:, 3*H:4*H]                 # [R, H]
    out_gate_pre   = proj_concat[:, 4*H:5*H]                 # [R, H]

    # Sigmoids – stay in half‑precision
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [R, 1] (half)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates and reshape to [B, N, N, H]
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # kept for later

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) and perform BMM with transposed right
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))  # (B·H, N, N)

    # Restore original shape [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)          # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection, and cast to float32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),             # [R, H]
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
