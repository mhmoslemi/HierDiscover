@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,
    eps: tl.constexpr, # not needed
):
    pid_z = tl.program_id(0)
    pid_x = tl.program_id(1)
    pid_y = tl.program_id(2)

    # Determine batch index
    b = pid_z

    # compute tile offsets
    offs_m = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers for left and right
    # left: batch b, rows i=offs_m, cols k (k tiles)
    # right: batch b, rows k, cols j=offs_n

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # load left tile
        a_ptrs = left_ptr + b * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)

        # load right tile
        b_ptrs = right_ptr + b * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # compute dot product for each sub-tile
        acc += tl.dot(a, b)   # shape (BLOCK_M, BLOCK_N)

    # Apply mask if needed? Not needed here.

    # write result
    c_ptrs = out_ptr + b * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
Custom Triton-accelerated implementation of the outgoing TriMul operator.
We fuse the first LayerNorm together with the five linear projections
(including the three gating matrices) into two Triton kernels:
 - `fused_ln_proj_kernel` computes the normalized input and produces
   `left` and `right` projections (with optional binary mask applied).
 - `fused_ln_gate_kernel` computes the same normalized input again
   (its cost is tiny compared to the O(N³) triangular update) and
   produces the three gate tensors.
The triangular multiplicative update is performed with a single batched
matrix‑multiply (`torch.bmm`), which lets cuBLAS sweep the N³‑work using
Tensor‑Cores.  Finally we apply a second fused LayerNorm+output‑gate
kernel and a small FP16 linear projection back to the original channel
dimension.  The whole forward pass runs entirely in FP16 (except the
LayerNorm reductions that stay in FP32 for numerical stability) and
returns FP32 as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: LayerNorm + left/right projections (mask applied inside)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_kernel(
    x_ptr,                 # *const float*   [total_rows, dim]
    left_ptr,              # *mut   half*    [total_rows, hidden]
    right_ptr,             # *mut   half*    [total_rows, hidden]
    mask_ptr,              # *const half*   [total_rows] (optional)
    norm_weight_ptr,       # *const float*  [dim]
    norm_bias_ptr,         # *const float*  [dim]
    w_left_ptr,            # *const half*   [dim, hidden]
    w_right_ptr,           # *const half*   [dim, hidden]
    total_rows: tl.int32,
    dim: tl.int32,
    hidden: tl.constexpr,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    apply_mask: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # Compute mean & variance (FP32) – two‑pass reduction over dim
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                        # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalize, apply γ/β, then GEMM with left/right weights
    # ---------------------------------------------------------------
    left_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float32)
    right_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load raw data tile
        ptrs_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        # Normalize and apply LayerNorm affine parameters (γ/β)
        norm = (vals - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_D]

        w = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)   # [BLOCK_D]
        b = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)   # [BLOCK_D]

        norm_affine = norm * w[None, :] + b[None, :]    # [BLOCK_M, BLOCK_D]

        # Load weight tiles (dim -> hidden)
        w_left = tl.load(w_left_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                         mask=mask_d[:, None], other=0.0).to(tl.float32)   # [BLOCK_D, hidden]
        w_right = tl.load(w_right_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0).to(tl.float32) # [BLOCK_D, hidden]

        # GEMM – accumulate into per‑row registers
        left_acc  += tl.dot(norm_affine, w_left)    # (BLOCK_M, hidden)
        right_acc += tl.dot(norm_affine, w_right)

    # Optional mask multiplication (binary mask broadcast on hidden dim)
    if apply_mask:
        mask_vals = tl.load(mask_ptr + offs_m,
                            mask=mask_m,
                            other=0.0).to(tl.float32)               # [BLOCK_M]
        mask_vals = mask_vals[:, None]                                 # broadcast
        left_acc  *= mask_vals
        right_acc *= mask_vals

    # Store results (half‑precision)
    offs_h = tl.arange(0, hidden)
    tl.store(left_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
             left_acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(right_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             right_acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))


# ----------------------------------------------------------------------
# Kernel 2: LayerNorm + gates (left_gate, right_gate, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    x_ptr,                 # *const float*   [total_rows, dim]
    left_gate_ptr,         # *mut   half*    [total_rows, hidden]
    right_gate_ptr,        # *mut   half*    [total_rows, hidden]
    out_gate_ptr,          # *mut   half*    [total_rows, hidden]
    norm_weight_ptr,       # *const float*  [dim]
    norm_bias_ptr,         # *const float*  [dim]
    w_lgate_ptr,           # *const half*   [dim, hidden]
    w_rgate_ptr,           # *const half*   [dim, hidden]
    w_outgate_ptr,         # *const half*   [dim, hidden]
    total_rows: tl.int32,
    dim: tl.int32,
    hidden: tl.constexpr,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass – mean & variance (same as in proj kernel)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass – normalized affine, then GEMM with gate weights
    # ---------------------------------------------------------------
    lgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float32)
    rgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float32)
    outgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # Normalize + LN affine
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        norm_affine = norm * w[None, :] + b[None, :]

        # Gate weight tiles
        w_lgate = tl.load(w_lgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0).to(tl.float32)
        w_rgate = tl.load(w_rgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0).to(tl.float32)
        w_outgate = tl.load(w_outgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                            mask=mask_d[:, None], other=0.0).to(tl.float32)

        lgate_acc   += tl.dot(norm_affine, w_lgate)
        rgate_acc   += tl.dot(norm_affine, w_rgate)
        outgate_acc += tl.dot(norm_affine, w_outgate)

    # Store results
    offs_h = tl.arange(0, hidden)
    tl.store(left_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             lgate_acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(right_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             rgate_acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(out_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             outgate_acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))

    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass – normalise, apply γ/β, multiply by gate and store
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        out = out * g

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator with Triton‑fused
    LayerNorm+linear stages.  Arguments:

        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]                # input channel dimension C
    hidden = cfg["hidden_dim"]      # hidden dimension H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 0) Prepare tensors
    # ------------------------------------------------------------------
    # flatten input: [R, C] float32
    x = inp.view(total_rows, dim).contiguous()

    # Pre‑transpose linear weights to shape (dim, hidden) and cast to FP16
    w_left = weights["left_proj.weight"].transpose(0, 1).contiguous().half()
    w_right = weights["right_proj.weight"].transpose(0, 1).contiguous().half()
    w_lgate = weights["left_gate.weight"].transpose(0, 1).contiguous().half()
    w_rgate = weights["right_gate.weight"].transpose(0, 1).contiguous().half()
    w_outgate = weights["out_gate.weight"].transpose(0, 1).contiguous().half()

    # Allocate outputs of the two fused kernels
    left_proj = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_proj = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    left_gate_raw = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_gate_raw = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    out_gate_raw = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    # Optional mask (binary) cast to half
    if not nomask:
        mask_fp16 = mask.view(total_rows).to(torch.float16).contiguous()
        apply_mask = 1
    else:
        # dummy tensor (will never be dereferenced because apply_mask==0)
        mask_fp16 = torch.empty(0, dtype=torch.half, device=device)
        apply_mask = 0

    # ------------------------------------------------------------------
    # 1️⃣ Fuse first LayerNorm + left/right projections (+mask)
    # ------------------------------------------------------------------
    BLOCK_M = 32                      # rows per program
    BLOCK_D = 32                      # dim tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_ln_proj_kernel[grid](
        x,
        left_proj,
        right_proj,
        mask_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        w_left,
        w_right,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        apply_mask=apply_mask,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse first LayerNorm + three gate projections
    # ------------------------------------------------------------------
    fused_ln_gate_kernel[grid](
        x,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        weights["norm.weight"],
        weights["norm.bias"],
        w_lgate,
        w_rgate,
        w_outgate,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # (R, H) half
    right_gate = torch.sigmoid(right_gate_raw)        # (R, H) half
    out_gate = torch.sigmoid(out_gate_raw)            # (R, H) half – used later

    left = left_proj * left_gate                       # (R, H) half
    right = right_proj * right_gate                    # (R, H) half

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)                 # (B, N, N, H)
    right = right.view(B, N, N, hidden)               # (B, N, N, H)

    # reshape for batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                          # (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid’ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w)   # (R, dim) half

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

Highly‑optimized Triton implementation of the **outgoing** TriMul operator.
The first LayerNorm and all five linear projections (left/right + three gates)
are fused into two custom Triton kernels, removing the large intermediate
tensor that would otherwise be created by a separate `torch.nn.Linear`.
The triangular multiplicative update is performed with a single batched
matrix‑multiply (`torch.bmm`) that lets cuBLAS/Tensor‑Core handle the
N³‑heavy work efficiently.  A third fused kernel implements the second
LayerNorm together with the output gate.  All heavy work stays in FP16
(except the LayerNorm reductions that use FP32 for numerical stability);
the final result is cast back to FP32 as required by the model.
"""

import triton.language as tl

# Kernel 1: LayerNorm + left/right projections (optional mask applied)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_kernel(
    x_ptr,                 # *const float*   [total_rows, dim]
    left_ptr,              # *mut   half*    [total_rows, hidden]
    right_ptr,             # *mut   half*    [total_rows, hidden]
    mask_ptr,              # *const half*   [total_rows] (optional)
    norm_weight_ptr,       # *const float*  [dim]
    norm_bias_ptr,         # *const float*  [dim]
    w_left_ptr,            # *const half*   [dim, hidden]
    w_right_ptr,           # *const half*   [dim, hidden]
    total_rows: tl.int32,
    dim: tl.int32,
    hidden: tl.constexpr,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    apply_mask: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # First pass: mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, LayerNorm affine, then GEMM with left/right
    # ------------------------------------------------------------------
    left_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float16)
    right_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float16)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw data tile
        ptrs_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        # normalize (FP32) → half for the matmul
        norm_fp16 = ((vals - mean[:, None]) * rstd[:, None]).to(tl.float16)  # [M, D]

        # LayerNorm affine (γ/β) in half
        w = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float16)
        b = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float16)
        norm_affine = norm_fp16 * w[None, :] + b[None, :]                     # [M, D]

        # weight tiles (dim → hidden)
        w_left = tl.load(w_left_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                         mask=mask_d[:, None], other=0.0)                     # [D, H]
        w_right = tl.load(w_right_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0)                    # [D, H]

        left_acc  += tl.dot(norm_affine, w_left)   # (BLOCK_M, hidden)
        right_acc += tl.dot(norm_affine, w_right)

    # optional mask (binary, broadcast on hidden dim)
    if apply_mask:
        mask_vals = tl.load(mask_ptr + offs_m,
                            mask=mask_m,
                            other=0.0).to(tl.float16)               # [BLOCK_M]
        mask_vals = mask_vals[:, None]                                 # broadcast
        left_acc  *= mask_vals
        right_acc *= mask_vals

    # store results
    offs_h = tl.arange(0, hidden)
    tl.store(left_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
             left_acc,
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(right_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             right_acc,
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))


# ----------------------------------------------------------------------
# Kernel 2: LayerNorm + three gate projections
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    x_ptr,                 # *const float*   [total_rows, dim]
    left_gate_ptr,         # *mut   half*    [total_rows, hidden]
    right_gate_ptr,        # *mut   half*    [total_rows, hidden]
    out_gate_ptr,          # *mut   half*    [total_rows, hidden]
    norm_weight_ptr,       # *const float*  [dim]
    norm_bias_ptr,         # *const float*  [dim]
    w_lgate_ptr,           # *const half*   [dim, hidden]
    w_rgate_ptr,           # *const half*   [dim, hidden]
    w_outgate_ptr,         # *const half*   [dim, hidden]
    total_rows: tl.int32,
    dim: tl.int32,
    hidden: tl.constexpr,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # First pass – mean & variance (same as projection kernel)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalize, affine, then GEMM with gate weights
    # ------------------------------------------------------------------
    lgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float16)
    rgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float16)
    outgate_acc = tl.zeros([BLOCK_M, hidden], dtype=tl.float16)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # normalize → half
        norm_fp16 = ((vals - mean[:, None]) * rstd[:, None]).to(tl.float16)

        # affine
        w = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float16)
        b = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float16)
        norm_affine = norm_fp16 * w[None, :] + b[None, :]

        # gate weight tiles
        w_lgate = tl.load(w_lgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0)
        w_rgate = tl.load(w_rgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                          mask=mask_d[:, None], other=0.0)
        w_outgate = tl.load(w_outgate_ptr + offs_d[:, None] * hidden + tl.arange(0, hidden)[None, :],
                            mask=mask_d[:, None], other=0.0)

        lgate_acc   += tl.dot(norm_affine, w_lgate)
        rgate_acc   += tl.dot(norm_affine, w_rgate)
        outgate_acc += tl.dot(norm_affine, w_outgate)

    # store gate results
    offs_h = tl.arange(0, hidden)
    tl.store(left_gate_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
             lgate_acc,
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(right_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
             rgate_acc,
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))
    tl.store(out_gate_ptr   + offs_m[:, None] * hidden + offs_h[None, :],
             outgate_acc,
             mask=mask_m[:, None] & (offs_h[None, :] < hidden))


# ----------------------------------------------------------------------
# Kernel 3: fused LayerNorm + out‑gate (post‑tri‑mul)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                # *const half*   [total_rows, hidden]
    y_ptr,                # *mut   half*   [total_rows, hidden]
    weight_ptr,           # *const float*  [hidden]
    bias_ptr,             # *const float*  [hidden]
    gate_ptr,             # *const half*   [total_rows, hidden] (already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass – mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass – normalise, apply γ/β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]   # FP32

        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        out = out * g

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]     (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model weights (float32)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"    : bool,
              "eps"       : float (optional)
          }

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # Flatten input (float32) – shape (R, C)
    # ------------------------------------------------------------------
    x = inp.view(total_rows, dim).contiguous()   # [R, C] float32

    # ------------------------------------------------------------------
    # Pre‑transpose weights to (dim, hidden) and cast to fp16
    # ------------------------------------------------------------------
    w_left      = weights["left_proj.weight"].transpose(0, 1).contiguous().half()
    w_right     = weights["right_proj.weight"].transpose(0, 1).contiguous().half()
    w_lgate     = weights["left_gate.weight"].transpose(0, 1).contiguous().half()
    w_rgate     = weights["right_gate.weight"].transpose(0, 1).contiguous().half()
    w_outgate   = weights["out_gate.weight"].transpose(0, 1).contiguous().half()

    # ------------------------------------------------------------------
    # Allocate outputs of the fused kernels
    # ------------------------------------------------------------------
    left_proj  = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_proj = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    left_gate_raw  = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_gate_raw = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    out_gate_raw   = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    # ------------------------------------------------------------------
    # Optional mask (binary) – broadcast on hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.view(total_rows).to(torch.float16).contiguous()
        apply_mask = 1
    else:
        mask_fp16 = torch.empty(0, dtype=torch.half, device=device)  # dummy
        apply_mask = 0

    # ------------------------------------------------------------------
    # Launch fused kernels
    # ------------------------------------------------------------------
    BLOCK_M = 32    # rows per program – chosen to keep register pressure low
    BLOCK_D = 32    # tile size along the feature dimension
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_ln_proj_kernel[grid](
        x,
        left_proj,
        right_proj,
        mask_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        w_left,
        w_right,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        apply_mask=apply_mask,
    )

    fused_ln_gate_kernel[grid](
        x,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        weights["norm.weight"],
        weights["norm.bias"],
        w_lgate,
        w_rgate,
        w_outgate,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # (R, H) half
    right_gate = torch.sigmoid(right_gate_raw)  # (R, H) half
    out_gate   = torch.sigmoid(out_gate_raw)    # (R, H) half (used later)

    left  = left_proj * left_gate               # (R, H) half
    right = right_proj * right_gate             # (R, H) half

    # ------------------------------------------------------------------
    # Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)          # (B, N, N, H)
    right = right.view(B, N, N, hidden)         # (B, N, N, H)

    # reshape for batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate = torch.empty_like(out_flat)      # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                   # already sigmoid’ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()               # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w)           # (R, dim) half

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast back to FP32
    return final
