import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Kernel for fused layernorm + out_gate (hidden dimension)
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,       # const half* [total_rows, hidden]
    y_ptr,       # *mut half* [total_rows, hidden]
    weight_ptr,  # const float* [hidden] (γ)
    bias_ptr,    # const float* [hidden] (β)
    gate_ptr,    # const half* [total_rows, hidden] (sigmoid(out_gate))
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: norm, γβ, gate ----------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x-mean)/sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]     # FP32

        # multiply by gate (already sigmoid'd)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)

        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Highly‑optimized Triton‑augmented forward pass of the **outgoing**
    TriMul operator (AlphaFold‑3).

    The implementation uses:
        • native `torch.nn.functional.layer_norm` for the first normalization
        • a fused Triton kernel for the second LayerNorm + output‑gate
        • dense linear layers (`torch.nn.functional.linear`) and a batched
          matrix‑multiply for the triangular multiplicative update.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N               # rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use native PyTorch implementation)
    # --------------------------------------------------------------
    # Shape: [B, N, N, dim]  (float32)
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to half for the remaining FP16 pipeline
    x_norm_fp16 = x_norm.half()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections & gates (single GEMM)
    # --------------------------------------------------------------
    # Concatenate the five projection / gate matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*hidden, dim)

    # Perform the GEMM: (R, dim) @ (dim, 5*hidden).T  → (R, 5*hidden)
    proj_all = F.linear(x_norm_fp16.view(total_rows, dim), concat_w)  # (R, 5*hidden)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                           # each (R, hidden)

    # --------------------------------------------------------------
    # 2a) Optional binary mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R, 1)  → half
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)          # (R, hidden)
    right_gate = torch.sigmoid(right_gate_fp16)        # (R, hidden)
    out_gate = torch.sigmoid(out_gate_fp16)            # (R, hidden)

    left_fp16 = left_proj_fp16 * left_gate             # (R, hidden)
    right_fp16 = right_proj_fp16 * right_gate          # (R, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, hidden]
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # (B, hidden, N, N)  →  (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # Batched matrix multiplication (FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N)

    # Restore original layout: (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate (fused Triton kernel)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()      # (R, hidden) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)           # (R, hidden) half

    # Kernel launch configuration
    BLOCK_M = 64                     # rows per program
    BLOCK_D = 32                     # hidden columns per iteration

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                 # already sigmoid’ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection  (hidden → dim)
    # --------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)   # (R, dim)

    # Cast back to float32 and reshape to [B, N, N, dim]
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

        # multiply by the pre‑computed out‑gate
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the **outgoing** TriMul operation.

    Steps
    -----
    1️⃣  Input LayerNorm – native torch.nn.functional.layer_norm (FP32 → FP16)
    2️⃣  One large FP16 GEMM that produces the five projections/gates.
    3️⃣  Optional binary mask, sigmoid gates and element‑wise gating.
    4️⃣  Triangular multiplicative update via a batched FP16 matrix‑multiply.
    5️⃣  Second LayerNorm + out‑gate – a fused Triton kernel.
    6️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to float32.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim        = cfg["dim"]
    hidden     = cfg["hidden_dim"]
    eps        = cfg.get("eps", 1e-5)
    nomask     = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                       # rows = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (use the highly‑optimised PyTorch kernel)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (C = dim)
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                     # now FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fused projection + gates (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                        # shape (5*hidden, dim)

    # Linear map: (total_rows, dim) @ (dim, 5*hidden).T → (total_rows, 5*hidden)
    proj_all = F.linear(x_norm.view(total_rows, dim), concat_w)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, hidden)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask if required (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)    # (R, hidden)
    right_gate = torch.sigmoid(right_gate_fp16)   # (R, hidden)
    out_gate   = torch.sigmoid(out_gate_fp16)     # (R, hidden) – used later

    left_fp16  = left_proj_fp16  * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B,N,N,hidden] for easy permutation
    left  = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # Bring `hidden` into the batch dimension for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # FP16 batched matrix‑multiply (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N)

    # Restore original layout → (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣  Second LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, hidden) FP16
    out_ln_gate_fp16 = torch.empty_like(out_flat)          # (R, hidden) FP16

    # Launch configuration – tune to the hardware
    BLOCK_M = 64               # rows per program
    BLOCK_D = 32               # hidden columns per iteration (covers H ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_ln_gate_fp16,
        weights["to_out_norm.weight"],   # γ  (hidden,)
        weights["to_out_norm.bias"],     # β  (hidden,)
        out_gate,                        # already sigmoid‑ed (R, hidden)
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()            # (dim, hidden)
    final_fp16 = F.linear(out_ln_gate_fp16, to_out_w_fp16)     # (R, dim) FP16

    # Cast back to float32 and reshape to the original 4‑D shape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
