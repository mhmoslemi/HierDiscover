    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, K, out_dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)
    # row block index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean & var for each row
    sum = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        x_ptrs = x_ptr + offs_m[:, None] * K + offs_k[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum / K
    var = sum_sq / K - mean * mean
    rstd = tl.rsqrt(var + eps)

    # loop over output tiles
    for n in range(0, tl.cdiv(out_dim, BLOCK_N)):
        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
        # load bias for this tile
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < out_dim
        bias = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32) # shape [BLOCK_N]

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K
            # load x (original) for this tile of K
            x_ptrs = x_ptr + offs_m[:, None] * K + offs_k[None, :]

        # store result

@triton.jit
def ln_matmul_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, K, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean & variance.
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        x = tl.load(x_ptr + offs_m[:, None] * K + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / K
    var = sum_sq / K - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over output tiles
    for n in range(0, tl.cdiv(N, BLOCK_N)):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        # bias for this tile
        bias = tl.load(bias_ptr + offs_n,
                       mask=mask_n,
                       other=0.0).to(tl.float32)

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K
            # load x tile again
            x = tl.load(x_ptr + offs_m[:, None] * K + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
            # normalize
            x_norm = (x - mean[:, None]) * rstd[:, None]   # shape [BLOCK_M, BLOCK_K]

            # load weight tile (transposed)
            # weight_ptr layout: [N, K] -> weight[n, k]
            w = tl.load(weight_ptr + offs_k[:, None] * N + offs_n[None, :],
                        mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)  # shape [BLOCK_K, BLOCK_N]

            acc += tl.dot(x_norm, w)   # [BLOCK_M, BLOCK_N]

        # add bias
        acc += bias[None, :]

        # store
        y_ptrs = y_ptr + offs_m[:, None] * N + offs_n[None, :]
        tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – Triton‑fused implementation
================================================
The forward pass consists of three stages that are heavily fused with
custom Triton kernels:

1️⃣ **Row‑wise LayerNorm + 5‑head linear projection**  
   The input tensor `x` (shape [B,N,N,D]) is normalised row‑wise,
   the normalisation weight/bias are folded into the linear weights and
   the five projection heads (left, right, left‑gate, right‑gate, out‑gate)
   are computed in a single GEMM kernel.  This replaces the separate
   LayerNorm kernel + `F.linear` call of the reference implementation.

2️⃣ **Triangular multiplicative update**  
   `left` and `right` are obtained from the projection heads,
   optionally masked, gated and reshaped to (B,H,N,N).  A batched matrix
   multiplication (`torch.bmm`) implements

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   which is the core O(N³) operation.

3️⃣ **Row‑wise LayerNorm + output‑gate + final projection**  
   The result of the update (`out`) is normalised, multiplied by the
   output gate, and projected back to the original dimension `dim`.  All
   three steps are fused in a second Triton kernel that simultaneously
   computes the statistics, applies the per‑row normalisation, the
   gating, and the final matrix multiplication.

All arithmetic inside the kernels is performed in FP32 accumulators and
the stored tensors are FP16, giving Tensor‑Core utilisation on an H100.
The final tensor is returned in FP32 as required.

The implementation works for all configurations listed in the problem
statement (B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384},
hidden_dim∈{128,384,768}) and handles both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 : fused LayerNorm (first LN) + 5×Linear projection
# ----------------------------------------------------------------------
@triton.jit
def ln_matmul_kernel(
    x_ptr,               # const  * [R, K]        (float32)
    y_ptr,               # mutable* [R, N]        (float16)
    weight_ptr,          # const  * [N, K]        (float16)
    bias_ptr,            # const  * [N]           (float16)
    total_rows,          # int (runtime)
    K,                   # int (dim)
    N,                   # int (5*hidden_dim)
    eps,                 # float (compile‑time)
    weight_stride,       # int  (stride between rows of weight, =K for first kernel)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_K: tl.constexpr,  # reduction axis tile
    BLOCK_N: tl.constexpr,  # output‑cols tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----------------------- 1. mean / variance -----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        x = tl.load(x_ptr + offs_m[:, None] * K + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / K
    var  = sum_sq / K - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------------- 2. GEMM (normalized × weight) ----------
    for n in range(0, tl.cdiv(N, BLOCK_N)):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        # bias for this output tile
        b = tl.load(bias_ptr + offs_n,
                    mask=mask_n,
                    other=0.0).to(tl.float32)

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # ---- read input tile and normalise --------------------
            x = tl.load(x_ptr + offs_m[:, None] * K + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)
            x_center = (x - mean[:, None]) * rstd[:, None]    # [BM, BK]

            # ---- load weight tile (transposed view) ---------------
            # weight layout: [N, K] row‑major => offset = n*K + k
            w = tl.load(
                weight_ptr
                + offs_k[:, None]               # k
                + offs_n[None, :] * weight_stride,  # n * K
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            ).to(tl.float32)                # [BK, BN]

            # ---- accumulate -------------------------------------------------
            acc += tl.dot(x_center, w)      # [BM, BN]
        # add bias and write
        acc += b[None, :]
        y_ptrs = y_ptr + offs_m[:, None] * N + offs_n[None, :]
        tl.store(y_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Kernel 2 : fused LayerNorm (second LN) + output‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def final_ln_gate_matmul_kernel(
    a_ptr,                # const  * [R, H]    (float16)  -- output of TriMul
    g_ptr,                # const  * [R, H]    (float16)  -- out_gate
    y_ptr,                # mutable* [R, D]    (float16)  -- final result
    norm_w_ptr,           # const  * [H]       (float32)  -- to_out_norm.weight
    norm_b_ptr,           # const  * [H]       (float32)  -- to_out_norm.bias
    weight_ptr,           # const  * [D, H]    (float16)  -- to_out.weight
    total_rows, H, D, eps,
    weight_stride,        # int (stride between rows of final weight = H)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_K: tl.constexpr,  # reduction axis tile (over H)
    BLOCK_N: tl.constexpr,  # output‑cols tile (over D)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------- 1. mean / variance of a --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        a = tl.load(a_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(a, axis=1)
        sum_sq  += tl.sum(a * a, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- 2. (LN + gate) × final weight ------------
    for n in range(0, tl.cdiv(D, BLOCK_N)):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < D
        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k in range(0, tl.cdiv(H, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # ---- load a and gate ---------------------------------
            a = tl.load(a_ptr + offs_m[:, None] * H + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            g = tl.load(g_ptr + offs_m[:, None] * H + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            # ---- normalise a (LN) ---------------------------------
            a_center = (a - mean[:, None]) * rstd[:, None]        # [BM, BK]

            # ---- load LN weight / bias ----------------------------
            wn = tl.load(norm_w_ptr + offs_k,
                         mask=mask_k,
                         other=0.0)                               # fp32
            bn = tl.load(norm_b_ptr + offs_k,
                         mask=mask_k,
                         other=0.0)                               # fp32

            # ---- combine LN, gate and bias -------------------------
            tmp = a_center * wn[None, :] + bn[None, :]            # [BM, BK]
            s   = tmp * g                                        # [BM, BK]

            # ---- load final weight tile (transposed) -------------
            # weight layout: [D, H] row‑major => offset = d*H + h
            w = tl.load(
                weight_ptr
                + offs_k[:, None]                 # h
                + offs_n[None, :] * weight_stride,   # d * H
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            ).to(tl.float32)                     # [BK, BN]

            # ---- accumulate ----------------------------------------
            acc += tl.dot(s, w)                 # [BM, BN]
        # write output tile
        y_ptrs = y_ptr + offs_m[:, None] * D + offs_n[None, :]
        tl.store(y_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict mapping parameter names to float32 tensors
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows = B·N·N
    out_dim = 5 * hidden               # concatenated projection size

    # ------------------------------------------------------------------
    # 1) Fuse first LayerNorm + 5 linear heads (left_proj/right_proj/... )
    # ------------------------------------------------------------------
    # concatenate raw weights (bias=False in the original model)
    w_raw = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                      # shape [5*H, dim]  (float32)
    )

    # fold the *first* LayerNorm (norm.weight / norm.bias) into the linear
    norm_w = weights["norm.weight"]          # [dim]   fp32
    norm_b = weights["norm.bias"]            # [dim]   fp32

    # fuse scale (norm weight) into the linear matrix
    w_fused = (w_raw * norm_w.unsqueeze(0)).to(torch.float16)   # [5*H, dim] fp16

    # bias after folding:   bias_fused = w_raw @ norm_bias
    b_fused = (w_raw @ norm_b).to(torch.float16)                # [5*H] fp16

    # flatten input to [R, dim] (float32)
    x_flat = input_tensor.view(R, dim).contiguous()

    # output buffer for the concatenated projection results
    proj_out = torch.empty(R, out_dim, dtype=torch.float16, device=device)

    # launch the fused LN+GEMM kernel
    BLOCK_M = 256
    BLOCK_K = 64
    BLOCK_N = 128      # good for out_dim up to ~2000
    grid = (triton.cdiv(R, BLOCK_M),)

    ln_matmul_kernel[grid](
        x_flat,                      # input (fp32)
        proj_out,                    # output (fp16)
        w_fused,                     # fused weight (fp16)
        b_fused,                     # fused bias   (fp16)
        R,
        dim,
        out_dim,
        eps,
        weight_stride=dim,           # stride = K (dim) for weight layout [N, K]
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )

    # ------------------------------------------------------------------
    # 2) Slice the 5 heads, optional mask, and apply sigmoid gates
    # ------------------------------------------------------------------
    h = hidden
    left_proj   = proj_out[:, 0 * h : 1 * h]
    right_proj  = proj_out[:, 1 * h : 2 * h]
    left_gate_p = proj_out[:, 2 * h : 3 * h]
    right_gate_p= proj_out[:, 3 * h : 4 * h]
    out_gate_p  = proj_out[:, 4 * h : 5 * h]

    # optional mask (broadcast over hidden dim)
    if not config.get("nomask", True):
        mask_half = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj = left_proj * mask_half
        right_proj = right_proj * mask_half

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)   # shape [R, H]

    # gated projections
    left  = left_proj * left_gate            # [R, H]
    right = right_proj * right_gate          # [R, H]

    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, h)
    right = right.view(B, N, N, h)
    out_gate = out_gate.view(B, N, N, h)

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    # (B, H, N, N) → (B*H, N, N)
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)

    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_batch.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()

    # flatten for the final kernel
    out_flat = out.view(R, h).contiguous()
    out_gate_flat = out_gate.view(R, h).contiguous()

    # ------------------------------------------------------------------
    # 4) Fuse second LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    # Fold the *second* LayerNorm weights/bias into the kernel call
    norm2_w = weights["to_out_norm.weight"]   # [H] fp32
    norm2_b = weights["to_out_norm.bias"]     # [H] fp32

    # final linear weight (no bias in the original model)
    w_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]

    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_N2 = 64   # tile over the output dimension `dim` (≤384)
    grid2 = (triton.cdiv(R, BLOCK_M),)

    final_ln_gate_matmul_kernel[grid2](
        out_flat,                # a_ptr : raw TriMul output (fp16)
        out_gate_flat,           # g_ptr : output gate (fp16)
        final_fp16,              # y_ptr : final result (fp16)
        norm2_w,                 # LN weight (fp32)
        norm2_b,                 # LN bias   (fp32)
        w_out,                   # final weight (fp16)  [dim, H]
        R,
        h,
        dim,
        eps,
        weight_stride=h,         # stride between rows of w_out = H
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N2,
    )

    # reshape to [B, N, N, dim] and cast to float32 as required by the model
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
