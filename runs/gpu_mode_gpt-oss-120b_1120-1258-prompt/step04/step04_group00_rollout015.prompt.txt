import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel: TriMul “outgoing” (AlphaFold‑3) forward pass.

The implementation follows the reference PyTorch module but fuses the two
LayerNorms into tiny Triton kernels that work in FP32 internally and write
the normalized result directly in FP16.  All dense linear layers and the
batched N×N GEMM (the core Tri‑multiplicative update) are delegated to
PyTorch’s FP16 Tensor‑Core kernels, which on an H100 give the best
throughput.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor  [B,N,N,dim].
2️⃣  Fuse each projection + its gate into a single linear layer:
        (dim → hidden_dim)  +  (dim → hidden_dim)  →  (dim → 2*hidden_dim).
3️⃣  Apply the optional pair‑wise mask (broadcast over hidden_dim) and
    the sigmoid‑gates (still FP16).
4️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    is a batched GEMM over the hidden dimension (B·hidden_dim matrices of
    size N×N) performed with `torch.bmm`.
5️⃣  Second LayerNorm (FP16 output) → FP16.
6️⃣  Apply the output‑gate (sigmoid) and a final linear projection
    (hidden_dim → dim).  The result is cast back to FP32.

The kernel only implements LayerNorm; everything else uses highly‑optimised
PyTorch kernels, ensuring correctness and excellent performance on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Computes LayerNorm over the last dimension of a 2‑D tensor.
    The mean/variance are computed in FP32, the normalized values are
    written as FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1: compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Pass 2: write normalized values
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # weight
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # bias

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (float32) – may be unused
        weights_dict : {
            "norm.weight":        Tensor[dim],
            "norm.bias":          Tensor[dim],
            "left_proj.weight":   Tensor[hidden_dim, dim],
            "right_proj.weight":  Tensor[hidden_dim, dim],
            "left_gate.weight":   Tensor[hidden_dim, dim],
            "right_gate.weight":  Tensor[hidden_dim, dim],
            "out_gate.weight":    Tensor[hidden_dim, dim],
            "to_out_norm.weight": Tensor[hidden_dim],
            "to_out_norm.bias":   Tensor[hidden_dim],
            "to_out.weight":      Tensor[dim, hidden_dim],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [rows, dim]

    # allocate FP16 buffer for normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    # launch kernel
    BLOCK_M = 256                     # rows per program (tuned)
    BLOCK_D = 64                      # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync needed – subsequent ops use default stream.

    # reshape back to 4‑D
    x_norm_h = x_norm_h.view(B, N, N, dim)  # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total)
    # ------------------------------------------------------------------
    # concat (proj, gate) weights for left and right paths
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)   # [2*H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)   # [2*H, dim]

    # Linear: (rows, dim) @ (2*H, dim).t  → (rows, 2*H)
    rows = total_rows
    left_concat = F.linear(x_norm_h.view(rows, dim), left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_h.view(rows, dim), right_concat_w)

    # split into projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]            # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]            # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (computed once, shared for all positions)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h.view(rows, dim),
                 weights["out_gate.weight"].to(torch.float16, non_blocking=True))
    )   # [rows, H] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [rows, 1]
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates and multiply
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h   # [rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # permute to bring H into batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                         # effective batch size for GEMM
    left_mat  = left_perm.view(BH, N, N)        # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)       # [BH, N, N]

    # GEMM: (N×N) @ (N×N)^T  → sum over k dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (on hidden_dim)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    out_norm = out_norm * out_gate                     # gated output

    # final linear: hidden_dim → dim
    final_h = F.linear(
        out_norm.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, dim] (FP16)

    # cast to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
