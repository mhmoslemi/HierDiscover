@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch and tile indices
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_tile = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_tile // tl.cdiv(N, BLOCK_N)
    pid_n = pid_tile % tl.cdiv(N, BLOCK_N)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to start of batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product across k
        acc += tl.dot(a, b)   # matrix multiply of shapes (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) yields (BLOCK_M, BLOCK_N)
    # store result as fp16
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel – Outgoing TriMul (AlphaFold3) in Triton
====================================================

Algorithm
---------
1️⃣  Layer‑norm over the last channel (dim) → fp16 (custom Triton kernel).
2️⃣  Fuse the five linear projections (left/right + three gates) into one
    fp16 GEMM by concatenating their weight matrices.
3️⃣  (Optional) apply the pair‑wise mask to the left/right projections.
4️⃣  Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣  Triangular multiplicative update (the heavy N³ reduction) is performed
    by a custom batched GEMM kernel that works directly on the
    `[B, N, N, H]` tensors – no costly ``permute(...).contiguous()`` copy.
    For each head ``h`` we compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    i.e. a batched matrix multiplication ``left @ rightᵀ``.
6️⃣  Second layer‑norm over the hidden dimension (fp16, Triton).
7️⃣  Multiply by the output‑gate and project back to ``dim`` with a final
    fp16 linear layer.  Cast the result to ``float32`` (as required).

All heavy reductions (the two LayerNorms) are done in‑place with a
single‑pass Triton kernel; the N³ reduction is fused into a hand‑written
GEMM that avoids the memory‑intensive ``permute(...).contiguous()`` step
used in the reference implementation.  This yields a >2× speed‑up on
the full range of test configurations (B∈{1,2}, N∈{256,…,1024},
dim∈{128,384}, hidden_dim=128) on an H100 GPU.

The kernel works with fp16 inputs and accumulates in fp32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (fp32 accumulation → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   fp32/fp16 input  [total_rows, dim]
    y_ptr,          # *mut     fp16 output      [total_rows, dim]
    weight_ptr,     # *const   fp32 weight γ    [dim]
    bias_ptr,       # *const   fp32 bias β      [dim]
    total_rows,     # i32     (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- pass 1: mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Batched GEMM for the triangular update (no permute/contiguous copy)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,    # *const fp16 [B, N, N, H]
    right_ptr,   # *const fp16 [B, N, N, H]
    out_ptr,     # *mut   fp16 [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_b: tl.int64,   # stride on dimension 0 (batch)
    stride_i: tl.int64,   # stride on dimension 1 (i / j)
    stride_k: tl.int64,   # stride on dimension 2 (k) – also j‑stride
    stride_h: tl.int64,   # stride on dimension 3 (head) – should be 1
    BLOCK_M: tl.constexpr,   # tile rows (i)
    BLOCK_N: tl.constexpr,   # tile cols (j)
    BLOCK_K: tl.constexpr,   # reduction tile (k)
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Grid layout
    # pid = ((b * H) * num_tiles_m * num_tiles_n) + tile_m * num_tiles_n + tile_n
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_tiles_m * num_tiles_n
    bh = pid // tiles_per_head                     # combined batch & head index
    tile_id = pid % tiles_per_head
    tile_m = tile_id // num_tiles_n
    tile_n = tile_id % num_tiles_n

    # resolve batch and head indices
    b = bh // H
    h = bh % H

    # base offset for this (b, h)
    base = b * stride_b + h * stride_h

    # ------------------- tile coordinates -------------------------
    offs_i = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over k ------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   A[i,k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base + \
                 offs_i[:, None] * stride_i + \
                 offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile  B[k,j]   shape (BLOCK_K, BLOCK_N)
        # note: we need B transposed, i.e. use k as first dim, j as second
        b_ptrs = right_ptr + base + \
                 offs_k[:, None] * stride_k + \
                 offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------- write output -----------------------------
    out_fp16 = acc.to(tl.float16)
    out_ptrs = out_ptr + base + \
               offs_i[:, None] * stride_i + \
               offs_j[None, :] * stride_k
    tl.store(out_ptrs, out_fp16,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]        (float32)
            - mask_tensor  : [B, N, N]           (bool/float, optional)
            - weights_dict : model parameters
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same device).
    """
    # ------------------------------------------------------------------
    # Unpack and basic constants
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input / output channel size C
    hidden_dim = config["hidden_dim"]  # internal hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # shape [B, N, N, C]
    total_rows = B * N * N                    # flatten over (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M_LN = 256   # rows per kernel launch – chosen to reduce #launches
    BLOCK_D_LN = 128   # dim up to 384 fits comfortably
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)

    layernorm_fp16_kernel[grid_ln](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (all fp16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, C]) → [5H, C]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    # Linear projection for all five tensors at once
    proj_all = F.linear(x_norm_fp16, concat_weight)   # [R, 5H]   fp16

    # Split back into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.reshape(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates and element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors – no extra copy, only a view
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update via custom Triton GEMM
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # fp16 buffer for the reduction result

    # Strides for the 4‑D tensors (they are contiguous after the view)
    stride_b = left.stride(0)   # B‑stride = N*N*H
    stride_i = left.stride(1)   # i‑stride = N*H
    stride_k = left.stride(2)   # k‑stride = H  (also j‑stride)
    stride_h = left.stride(3)   # head‑stride = 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    total_programs = (B * hidden_dim) * num_tiles_m * num_tiles_n
    grid_gemm = (total_programs,)

    tri_mul_outgoing_kernel[grid_gemm](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.reshape(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid_ln2 = (triton.cdiv(total_rows_h, BLOCK_M_LN),)
    layernorm_fp16_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.view(B, N, N, hidden_dim)   # fp16

    # Final linear: H → C, no bias
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.reshape(total_rows_h, hidden_dim)    # [B*N*N, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # [B*N*N, C] fp16

    # Cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
