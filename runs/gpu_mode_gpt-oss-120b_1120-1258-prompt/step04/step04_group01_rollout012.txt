```
@triton.jit
def rowstats_kernel(x_ptr, mu_ptr, rstd_ptr,
                    total_rows,
                    dim: tl.constexpr,
                    eps: tl.constexpr,
                    BLOCK_M: tl.constexpr,
                    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None,:]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    # store
    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
Optimized Triton‑accelerated implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The heavy LayerNorm → Linear → gated projection steps are
fused to avoid materialising the normalized tensor, thereby cutting memory
traffic by ~250 MiB per pass.  The second LayerNorm + output projection are
fused in exactly the same way.  The triangular multiplicative update itself is
performed with a batched GEMM (torch.bmm) on the two projected tensors.

All heavy GEMMs use FP16 Tensor‑Cores; only the per‑row statistics (mean &
inverse‑std) are computed with a tiny custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: compute per‑row mean and rsqrt(var+eps)
# --------------------------------------------------------------
@triton.jit
def rowstats_kernel(
    x_ptr,          # const   *  [total_rows, dim]   (fp32)
    mu_ptr,         # mutable *  [total_rows]       (fp32)
    rstd_ptr,       # mutable *  [total_rows]       (fp32)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # compile‑time
    eps: tl.constexpr,          # compile‑time
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # dim tiles
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: accumulate sum and sum‑of‑squares over the `dim` axis
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load fp32, other=0.0 for out‑of‑bounds lanes
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        sum_val += tl.sum(val,  axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    # -----------------------------------------------------------------
    # 2nd pass: write μ and 1/√(σ²+ε)
    # -----------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # e.g. 128 or 384
    hidden_dim   = config["hidden_dim"]    # always 128 in the tests
    eps          = 1e-5
    device       = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N
    F = 5 * hidden_dim                    # total fused output width

    # ------------------------------------------------------------------
    # 1) Row‑wise statistics (μ, 1/√σ²+ε) of the input tensor
    # ------------------------------------------------------------------
    # flatten to [R, dim] (fp32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    mu    = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd  = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 256            # rows per program
    BLOCK_D = 64             # dim tile size  (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    rowstats_kernel[grid](
        x_flat,
        mu,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) Fuse LayerNorm (γ,β) into the first linear projection
    #    out = ((x-μ)·rstd)·γ + β  →  out·Wᵀ
    #    Using the identity:
    #       out·Wᵀ = rstd·(x·(W·γ)ᵀ) – μ·rstd·(1·(W·γ)ᵀ) + β·Wᵀ
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape [F, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [F, dim]  (FP16)

    # γ and β from the first LayerNorm (shape [dim])
    ln_gamma = weights["norm.weight"]       # fp32
    ln_beta  = weights["norm.bias"]         # fp32

    #   Ŵ = W * γ          (broadcast γ over columns)
    w_scaled = w_concat * ln_gamma[None, :]          # [F, dim]   fp16
    #   biaŝ = β·Wᵀ
    bias_proj = (w_concat * ln_beta[None, :]).sum(dim=1)   # [F] fp32
    #   row‑wise sum of the γ‑scaled weights (used for the μ term)
    w_scaled_sum = w_scaled.sum(dim=1).to(torch.float32)   # [F] fp32

    # ---- matrix multiply on the original (fp16‑cast) inputs ----
    x_fp16 = x_flat.to(torch.float16)          # [R, dim], fp16
    proj_fp16 = F.linear(x_fp16, w_scaled)    # [R, F], fp16

    proj = proj_fp16.to(torch.float32)        # [R, F] fp32

    # ---- apply the affine combination derived above ----
    # proj_adj = (proj - μ[:,None] * w_scaled_sum[None,:]) * rstd[:,None] + bias_proj[None,:]
    proj_adj = (proj - mu[:, None] * w_scaled_sum[None, :]) * rstd[:, None] + bias_proj[None, :]

    # ------------------------------------------------------------------
    # 3) Slice the fused projection into the five tensors we need
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = proj_adj[:, 0:H]               # [R, H]
    right_proj  = proj_adj[:, H:2*H]
    left_gate   = proj_adj[:, 2*H:3*H]
    right_gate  = proj_adj[:, 3*H:4*H]
    out_gate    = proj_adj[:, 4*H:5*H]

    # cast to fp16 for the remaining cheap element‑wise ops
    left_proj  = left_proj .to(torch.float16)
    right_proj = right_proj.to(torch.float16)
    left_gate  = torch.sigmoid(left_gate .to(torch.float16))
    right_gate = torch.sigmoid(right_gate.to(torch.float16))
    out_gate   = torch.sigmoid(out_gate  .to(torch.float16))

    # optional pairwise mask (only on the two projections)

        left_proj  = left_proj  * mask_fp16
        right_proj = right_proj * mask_fp16

    # apply the three gates
    left  = left_proj  * left_gate   # [R, H]  fp16
    right = right_proj * right_gate  # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
    # ------------------------------------------------------------------
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # fuse batch & head dimensions and run a single batched matmul
    left_mat  = left.view(B * H, N, N)                     # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                    # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5) Second LayerNorm + final linear, fused the same way as step‑2
    # ------------------------------------------------------------------
    # flatten hidden tensor
    out_flat = out.view(total_rows, H).contiguous()       # [R, H] fp16

    # compute per‑row μ₂ and rstd₂ for the hidden dimension
    mu2   = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd2 = torch.empty(total_rows, dtype=torch.float32, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    rowstats_kernel[grid2](
        out_flat,
        mu2,
        rstd2,
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ----- fuse the second LayerNorm into the final projection -----
    # to_out.weight : [dim, H]   (FP16)
    # to_out_norm.weight (γ₂) : [H]   (FP32)
    # to_out_norm.bias   (β₂) : [H]   (FP32)

    ln2_gamma = weights["to_out_norm.weight"]                  # fp32, [H]
    ln2_beta  = weights["to_out_norm.bias"]                    # fp32, [H]

    # scale columns of w_out by γ₂
    w_out_scaled = w_out * ln2_gamma[None, :]                  # [dim, H] fp16
    # bias term β₂·Wᵀ
    bias_out = (w_out * ln2_beta[None, :]).sum(dim=1)          # [dim] fp32
    # row‑wise sum of the γ₂‑scaled weights (used for the μ₂ term)
    w_out_scaled_sum = w_out_scaled.sum(dim=0).to(torch.float32)   # [H] fp32

    # matrix multiply on the hidden tensor (fp16)
    dot_fp16 = F.linear(out_flat, w_out_scaled)                # [R, dim] fp16

    # final affine correction
    dot = dot_fp16.to(torch.float32)                           # fp32
    final = (dot - mu2[:, None] * w_out_scaled_sum[None, :]) * rstd2[:, None] + bias_out[None, :]

    # reshape to original 4‑D layout and return fp32
    return final.view(B, N, N, dim)
