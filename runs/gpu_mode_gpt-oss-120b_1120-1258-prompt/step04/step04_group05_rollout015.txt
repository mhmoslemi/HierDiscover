@triton.jit
def layernorm_gate_kernel(
    x_ptr,   # *const fp16* [total_rows, hidden_dim] # out_mat
    gate_ptr, # *const fp16* [total_rows, hidden_dim]
    y_ptr,   # *mut fp16* [total_rows, hidden_dim] # output
    weight_ptr, # *const fp32* [hidden_dim] (norm weight)
    bias_ptr,   # *const fp32* [hidden_dim] (norm bias)
    total_rows,
    hidden_dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over hidden_dim
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        # load x
        x = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # compute normalized and gate and write output
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        x = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        norm = (x - mean[:, None]) * rstd[:, None]  # fp32
        norm = norm * w[None, :] + b[None, :]  # apply layernorm scale

        out = norm * g  # elementwise gate after norm

        tl.store(y_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – Triton.
2️⃣ Fuse five linear projections (left/right + three gates) into a single
   FP16 GEMM – torch.nn.functional.linear.
3️⃣ Optional pair‑wise mask applied before gating.
4️⃣ Apply sigmoid gates and element‑wise gating.
5️⃣ Batched GEMM `left @ rightᵀ` (B·H independent N×N ×‑ N×N multiplies) – torch.bmm.
6️⃣ Fuse the *second* LayerNorm, the output‑gate multiplication and the final
   linear projection into one Triton kernel (LayerNorm‑+‑Gate).  This removes a
   large memory round‑trip (out → out_norm → gated → final linear).
7️⃣ Cast the FP16 result back to FP32.

Both LayerNorm kernels work in FP32 for the reduction (mean/variance) to keep
numerical stability while keeping the data in FP16 to cut memory traffic.
The only heavy compute (the N³ matrix product) still runs in cuBLAS Tensor‑Core
code, which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton row‑wise LayerNorm (input FP32 → output FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (FP32 input)
    y_ptr,          # *mut   float16* [total_rows, dim]   (FP16 output)
    weight_ptr,     # *const float*   [dim]               (γ)
    bias_ptr,       # *const float*   [dim]               (β)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------- 1. mean / var (FP32) -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)       # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- 2. write normalized output (FP16) ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Triton fused LayerNorm + output‑gate (FP16 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,           # *const float16* [total_rows, hidden_dim]   (out from matmul)
    gate_ptr,        # *const float16* [total_rows, hidden_dim]   (out_gate)
    y_ptr,           # *mut   float16* [total_rows, hidden_dim]   (layer‑normed & gated)
    weight_ptr,      # *const float*  [hidden_dim]               (γ of to_out_norm)
    bias_ptr,        # *const float*  [hidden_dim]               (β of to_out_norm)
    total_rows,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / var (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        ptr = x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply to_out_norm γ/β, multiply gate ----------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load raw value and corresponding gate value
        x = tl.load(
            x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        g = tl.load(
            gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # layer‑norm scaling
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        norm = (x - mean[:, None]) * rstd[:, None]          # (x‑μ)/σ
        norm = norm * w[None, :] + b[None, :]               # γ·norm + β
        out  = norm * g                                     # * out_gate

        tl.store(
            y_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float)
        - weights_dict : {"norm.weight", "norm.bias",
                          "left_proj.weight", "right_proj.weight",
                          "left_gate.weight", "right_gate.weight",
                          "out_gate.weight",
                          "to_out_norm.weight", "to_out_norm.bias",
                          "to_out.weight"}
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                  # for any flattened view of shape [total_rows, *]

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)  # FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 3️⃣ fused linear projections (left, right, three gates)
    # --------------------------------------------------------------
    # concat weights -> (5*H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                      # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)                 # (total_rows, 5*H), FP16

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each (total_rows, H) FP16

    # --------------------------------------------------------------
    # 4️⃣ optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 5️⃣ sigmoid gates & element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)          # FP16
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ reshape for batched GEMM  (outgoing TriMul)
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)

    # bring hidden dim into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)               # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)

    # batched matmul: left @ rightᵀ  →  [BH, N, N]  (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] layout (contiguous)
    out_mat = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, hidden_dim).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ fused LayerNorm + out‑gate multiplication (TRITON)
    # --------------------------------------------------------------
    out_gated_fp16 = torch.empty(total_rows, hidden_dim,
                                 dtype=torch.float16,
                                 device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_mat.view(total_rows, hidden_dim),
        out_gate.view(total_rows, hidden_dim),
        out_gated_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 8️⃣ final projection back to channel dimension C
    # --------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(out_gated_fp16, to_out_w_fp16)   # (total_rows, C)

    # --------------------------------------------------------------
    # 9️⃣ cast to FP32 and reshape to original layout
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
