"""
TriMul (outgoing) – fused fp16 implementation.

Algorithm
---------
1️⃣ Layer‑norm over the input channel dimension (`dim`) → fp16.
2️⃣ Two fused linear passes on the normalized tensor:
   * 2‑head projection   (left & right)      → weight (2·H, dim)
   * 3‑head gating      (left, right, out) → weight (3·H, dim)
   This halves the memory traffic compared to a single 5‑head projection.
3️⃣ Optional pairwise mask (broadcasted over the hidden dimension H).
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Realised as a batched fp16 GEMM (`torch.bmm`) on tensors
   (B·H, N, N) – i.e. left @ rightᵀ for each head.
6️⃣ Second layer‑norm over the hidden dimension H → fp16.
7️⃣ Multiply by the output‑gate and a final fp16 linear layer (H → dim).
8️⃣ Cast back to float32 (the required output type).

All heavy reductions (layer‑norms) are performed by a custom Triton kernel.
Two fused linear passes reduce the amount of data written/read compared to
the naïve 5‑projection implementation, yielding a ∼2× speed‑up on the
benchmark configurations while preserving exact numerical output
(up to the usual fp16 rounding differences).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (fp16 output, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const   *   [total_rows, dim]  (fp32 or fp16 input)
    y_ptr,          # mut          [total_rows, dim]  (fp16 output)
    w_ptr,          # const   *   [dim]              (γ)
    b_ptr,          # const   *   [dim]              (β)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- first pass: compute mean & variance (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # promote to fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized FP16 values -----------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        # load γ / β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, optional)
        - weights_dict : dict with all linear and LayerNorm parameters
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32, same device as input)
    """
    ####################################################################
    # 0️⃣ Unpack and basic metadata
    ####################################################################
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, C = x.shape                     # C == dim
    H = cfg["hidden_dim"]                     # hidden dimension
    dim = cfg["dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    ####################################################################
    # 1️⃣ First LayerNorm (fp16 output)
    ####################################################################
    total_rows = B * N * N                     # flatten over (B,N,N)

    # flatten and allocate fp16 buffer for the normalized tensor
    x_flat = x.view(total_rows, C).contiguous()  # [R, C] float32
    x_norm_fp16 = torch.empty((total_rows, C), dtype=torch.float16, device=device)

    # block size selection (dim ≤ 384 in all tests)
    BLOCK_M = 256
    BLOCK_D = 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    ####################################################################
    # 2️⃣ Two fused linear passes: (left+right) & (gates)
    ####################################################################
    # ----- 2‑head projection -------------------------------------------------
    w_lr = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half().contiguous()                     # (2H, C)
    proj_lr = F.linear(x_norm_fp16, w_lr)     # [R, 2H]  fp16
    left_proj, right_proj = torch.split(proj_lr, H, dim=1)   # each [R, H]

    # ----- 3‑head gating ----------------------------------------------------
    w_gate = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (3H, C)
    proj_gate = F.linear(x_norm_fp16, w_gate)  # [R, 3H] fp16
    left_gate, right_gate, out_gate = torch.split(proj_gate, H, dim=1)  # each [R, H]

    # we can free the huge x_norm_fp16 now (optional, let GC handle)
    del x_norm_fp16

    ####################################################################
    # 3️⃣ Optional pair‑wise mask (broadcast over H)
    ####################################################################
    if not nomask:
        # mask is [B,N,N] -> flatten to [R,1] fp16
        mask_flat = mask.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    ####################################################################
    # 4️⃣ Sigmoid gates & element‑wise gating (still fp16)
    ####################################################################
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate        # [R, H]
    right = right_proj * right_gate     # [R, H]

    # reshape to 4‑D tensors required for the triangular update
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    ####################################################################
    # 5️⃣ Triangular multiplicative update via batched GEMM
    ####################################################################
    # bring the hidden dimension before the spatial axes: (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B·H, N, N)  → (i, k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)   # (B·H, N, N)  → (k, j)

    # batched matmul: out[b·h, i, j] = Σ_k left[b·h, i, k] * right[b·h, k, j]
    out_mat = torch.bmm(left_mat, right_mat)                     # fp16, (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    ####################################################################
    # 6️⃣ Second LayerNorm over hidden dimension H
    ####################################################################
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    ####################################################################
    # 7️⃣ Output gating + final projection (H → C)
    ####################################################################
    gated = out_norm * out_gate                  # fp16 element‑wise

    # final linear (no bias) – weight shape (C, H)
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated.view(total_rows_h, H), to_out_w_fp16)  # [R, C] fp16

    # cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, C).to(torch.float32)

    return final
