"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Row‑wise LayerNorm** on the input tensor (shape [B,N,N,D]).
   Implemented as a custom Triton kernel that directly writes the
   normalized result in fp16.

2. **Fused projection + gating**:
   The five dense layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) are merged into a single half‑precision GEMM
   (`torch.nn.functional.linear`).  The result is split into the five
   tensors of shape [B,N,N, H] and the three gate tensors are passed
   through a sigmoid.

3. **Mask & gating**:
   An optional mask is broadcast over the hidden dimension and fused
   in‑place with the gate multiplications, avoiding extra temporaries.

4. **Triangular multiplicative update**:
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
   This is a batched matrix multiplication performed by a single
   ``torch.einsum`` call on tensors of shape [B,N,N,H]; it removes the
   costly ``permute`` + ``contiguous`` round‑trip required by the
   naïve ``torch.bmm`` implementation.

5. **Second LayerNorm** (over the hidden dimension) – same Triton
   kernel as step 1.

6. **Output gating & final projection**:
   The normalized result is multiplied element‑wise with ``out_gate``
   and projected back to the original dimension with a half‑precision
   linear layer.

All heavy arithmetic runs in fp16, leveraging Tensor‑Core throughput
on an H100.  Only the two LayerNorms are custom‑implemented; the rest
uses highly‑optimized PyTorch kernels.  The implementation works for
all test configurations (with or without mask, various sequence
lengths, batch sizes and channel dimensions).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *  [total_rows, dim]   (float32 input)
    y_ptr,          # mutable *[total_rows, dim]   (float16 output)
    weight_ptr,     # const *  [dim]               (float32 γ)
    bias_ptr,       # const *  [dim]               (float32 β)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1st pass: compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2nd pass: write normalized result (γ·x̂ + β) in fp16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        # (x - μ) * rstd
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]    # fp32

        out_half = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool / float)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32, same device as input)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – fp16 output
    # --------------------------------------------------------------
    # flatten to [R, D]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Triton launch parameters (tuned for D ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                # input pointer (fp32)
        x_norm,                # output pointer (fp16)
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: [5*H, D]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, D]
            weights["right_proj.weight"],  # [H, D]
            weights["left_gate.weight"],   # [H, D]
            weights["right_gate.weight"],  # [H, D]
            weights["out_gate.weight"],    # [H, D]
        ],
        dim=0,
    ).to(torch.float16)

    # Linear:  [R, D] @ [5*H, D].T → [R, 5*H]   (fp16)
    proj_out = F.linear(x_norm, w_concat)   # shape [R, 5*H]

    # --------------------------------------------------------------
    # 3) Split and reshape the fused result
    # --------------------------------------------------------------
    H = hidden
    left_proj   = proj_out[:, 0*H:1*H]   # [R, H]
    right_proj  = proj_out[:, 1*H:2*H]
    left_gate_p = proj_out[:, 2*H:3*H]
    right_gate_p= proj_out[:, 3*H:4*H]
    out_gate_p  = proj_out[:, 4*H:5*H]

    # Sigmoids for the three gate tensors
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Reshape everything to [B, N, N, H]
    left_proj   = left_proj.view(B, N, N, H)
    right_proj  = right_proj.view(B, N, N, H)
    left_gate   = left_gate.view(B, N, N, H)
    right_gate  = right_gate.view(B, N, N, H)
    out_gate    = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcasted over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5) Apply gates in‑place  (fuse element‑wise mult.)
    # --------------------------------------------------------------
    left_proj.mul_(left_gate)          # now holds “left”
    right_proj.mul_(right_gate)        # now holds “right”

    # --------------------------------------------------------------
    # 6) Triangular multiplicative update (batched GEMM) via einsum
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k d, b j k d -> b i j d', left_proj, right_proj)
    # out shape: [B, N, N, H]  (fp16)

    # --------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, H)          # fp16

    # --------------------------------------------------------------
    # 8) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # fp16 element‑wise

    final_half = F.linear(
        out_norm.view(total_rows, H),                  # [R, H]
        weights["to_out.weight"].to(torch.float16)    # [D, H]
    )                                                   # [R, D] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
