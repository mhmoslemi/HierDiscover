"""
TriMul (outgoing) – fast forward‑pass implementation with Triton LayerNorm.

Algorithm
---------
1️⃣  Layer‑norm on the input channel dimension (fp16 output) – custom Triton
    kernel that computes per‑row mean/variance in fp32 and writes the
    normalized tensor in fp16.

2️⃣  Fuse the five linear projections (left/right projections + three gates)
    into a single fp16 GEMM by concatenating their weight matrices.
    The result is split back into the five tensors.

3️⃣  (Optional) apply the pair‑wise mask to the left/right projections.

4️⃣  Apply sigmoid gates and element‑wise gating (still fp16).

5️⃣  Triangular multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    realized as a batched fp16 GEMM (`torch.bmm`) on tensors of shape
    (B·H, N, N), where H = hidden_dim.

6️⃣  Second layer‑norm over the hidden dimension (fp16) – same Triton kernel.

7️⃣  Multiply by the output gate and project back to the original `dim`
    with a final fp16 linear layer.  Cast the result to float32 (as required).

All heavy reductions (the two LayerNorms) are performed by a custom Triton
kernel; dense linear algebra uses fp16 Tensor‑Core friendly ops.  The
implementation works for all test configurations (different batch sizes,
sequence lengths, channel sizes and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (fp16 output, fp32/fp16 input)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   fp32/fp16 input  [total_rows, dim]
    y_ptr,          # *mut     fp16 output     [total_rows, dim]
    weight_ptr,     # *const   fp32 weight     [dim]
    bias_ptr,       # *const   fp32 bias       [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # compile‑time constant
    eps: tl.constexpr,          # compile‑time constant
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute per‑row mean and variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # cast to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

            - config_dict : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input / output channel size C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → skip mask

    # --------------------------------------------------------------
    # Shape helpers
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape           # [B, N, N, C]
    total_rows = B * N * N                     # flatten over (B,N,N)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per kernel launch (tuned for H100)
    BLOCK_D = 64    # columns per tile (covers up to dim=384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse projections & gates (all fp16)
    # --------------------------------------------------------------
    # Concatenate the five projection‑gate weight matrices (each [H, C])
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                # shape (5*H, C)

    # Linear: [R, C] @ [C, 5H]^T → [R, 5H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16 result

    # Split back to the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast over H)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> flatten to [R, 1] and cast to fp16
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates & element‑wise gating (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate          # [R, H]
    right_fp16 = right_proj * right_gate       # [R, H]

    # --------------------------------------------------------------
    # 5️⃣ Reshape for the triangular multiplicative update
    # --------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # Bring hidden dim to the second axis for batched GEMM
    # left_perm: (i,k)   shape [B, H, N, N]
    # right_perm: (k,j)  shape [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N] (i,k)
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N] (k,j)

    # Batched matrix multiplication in fp16 -> [B*H, N, N] (i,j)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16 element‑wise

    # Final linear layer (H → C) – fp16 weight, no bias
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # [R_h, C] fp16

    # Cast back to float32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
