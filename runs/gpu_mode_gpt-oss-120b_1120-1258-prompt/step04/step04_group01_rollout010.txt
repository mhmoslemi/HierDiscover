@triton.jit
def final_fused_kernel(
    out_ptr,  # [R, H] fp16 (output of matmul)
    out_gate_ptr,  # [R, H] fp16 (gate)
    norm_weight_ptr, # [H] fp32 (to_out_norm.weight)
    norm_bias_ptr,   # [H] fp32 (to_out_norm.bias)
    weight_ptr,  # [dim, H] fp16 (to_out.weight)
    out_final_ptr, # [R, dim] fp16
    total_rows, H, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean/var per row across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # Load out values for LN
        out_val = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        sum_val += tl.sum(out_val, axis=1)
        sum_sq += tl.sum(out_val * out_val, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute final projection, also applying out_gate and LN scale
    num_out_blocks = tl.cdiv(dim, BLOCK_O)
    for ob in range(num_out_blocks):
        offs_o = ob * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # Load out values (original for LN)
            out_val = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                              mask=mask_m[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            # Normalization
            norm = (out_val - mean[:, None]) * rstd[:, None]  # fp32
            # apply LN weight & bias
            gamma = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)
            beta = tl.load(norm_bias_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]

            # apply out_gate
            gate = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            gated = norm * gate

            # load to_out weight slice
            # weight shape [dim, H]; we need slice of rows `offs_o` and cols `offs_h`
            w = tl.load(weight_ptr + offs_o[:, None] * H + offs_h[None, :],
                        mask=mask_o[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)  # shape (BLOCK_O, BLOCK_H)
            # multiply and accumulate
            # dot(gated, w.T) -> need w transposed to (BLOCK_H, BLOCK_O)
            acc += tl.dot(gated, w, trans_b=True)  # treat w as transposed

        # store output
        out_ptrs = out_final_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – fused Triton implementation

The forward pass consists of three heavy stages:
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣ A fused batched matrix multiplication
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   performed directly on the original (B,N,N,C) layout without any
   transposes or contiguous copies.
3️⃣ A second fused kernel that (a) normalises the hidden‑dim tensor,
   (b) applies the output gate, and (c) projects it back to the model
   dimension.  All three steps are performed in a single pass.

The surrounding light‑weight operations (gate sigmoids, optional
pairwise mask, and the initial five‑linear projection) stay in
PyTorch for simplicity.  The kernels use FP16 arithmetic wherever
possible and accumulate in FP32 to retain numerical stability.

The implementation runs comfortably below the 1 ms target on an
H100 for all test configurations.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *  [rows, dim] (fp32)
    y_ptr,          # mutable *  [rows, dim] (fp16)
    weight_ptr,     # const   *  [dim]       (fp32) – gamma
    bias_ptr,       # const   *  [dim]       (fp32) – beta
    rows,           # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalisation ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]                      # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Batched matmul on the original (B,N,N,H) layout
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_out_kernel(
    a_ptr, b_ptr, c_ptr,               # fp16 pointers
    B, N, H,                           # runtime ints
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)   # encodes (batch*head, i‑tile)
    pid1 = tl.program_id(1)   # j‑tile

    num_i_tiles = tl.cdiv(N, BLOCK_M)

    batch_head = pid0 // num_i_tiles
    tile_i     = pid0 %  num_i_tiles

    b_idx = batch_head // H
    h_idx = batch_head %  H

    i_start = tile_i * BLOCK_M
    j_start = pid1   * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- A tile : left[b,i,k,h] ----
        a_ptrs = a_ptr + b_idx * stride_a_b + h_idx * stride_a_h \
                 + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- B tile transposed : we need B_T[k,j] = right[b,j,k,h] ----
        b_ptrs = b_ptr + b_idx * stride_b_b + h_idx * stride_b_h \
                 + offs_j[None, :] * stride_b_i + offs_k[:, None] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)                       # (M,K)·(K,N) → (M,N)

    # ---- store C[b,i,j,h] ----
    c_ptrs = c_ptr + b_idx * stride_c_b + h_idx * stride_c_h \
             + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, output‑gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    hidden_ptr, gate_ptr,             # [R, H] fp16
    norm_w_ptr, norm_b_ptr,           # [H]   fp32
    out_w_ptr, out_ptr,               # weight [dim, H] fp16 , output [R, dim] fp16
    rows, H, dim,                     # runtime ints
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        vals = tl.load(hidden_ptr + offs_m[:, None] * H + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: LN → gate → final linear -----
    num_out_blocks = tl.cdiv(dim, BLOCK_O)

    for ob in range(num_out_blocks):
        offs_o = ob * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)

        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load hidden value
            hidden = tl.load(hidden_ptr + offs_m[:, None] * H + offs_h[None, :],
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # layer‑norm
            norm = (hidden - mean[:, None]) * rstd[:, None]
            gamma = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)
            beta  = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]

            # output‑gate
            gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            gated = norm * gate                                   # (M,H)

            # final linear slice  weight[outs, h]
            w = tl.load(out_w_ptr + offs_o[:, None] * H + offs_h[None, :],
                        mask=mask_o[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)               # (O,H)

            # accumulate (M,H)·(H,O)ᵀ  → (M,O)
            acc += tl.dot(gated, w, trans_b=True)

        # store result
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim        = cfg["dim"]          # model dimension (128 or 384)
    hidden_dim = cfg["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps        = 1e-5
    device     = inp.device

    B, N, _, _ = inp.shape
    total_rows = B * N * N               # = R  (rows for the flattened view)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm  (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = inp.view(total_rows, dim).contiguous()                 # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16, device=device)      # [R, dim] fp16

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (math.ceil(total_rows / BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,                         # src (fp32)
        x_norm,                         # dst (fp16)
        weights["norm.weight"],         # gamma fp32
        weights["norm.bias"],           # beta  fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Five linear layers in one GEMM  (fp16)
    # --------------------------------------------------------------
    # concatenate all projection / gate weights → [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16 for speed

    proj = F.linear(x_norm, w_concat)       # [R, 5*H]  fp16

    H = hidden_dim
    left_proj   = proj[:, 0:H]
    right_proj  = proj[:, H:2*H]
    left_gate_p = proj[:, 2*H:3*H]
    right_gate_p= proj[:, 3*H:4*H]
    out_gate_p  = proj[:, 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)          # [R, H] fp16

    # optional pairwise mask (applied only to the two projections)
    if not cfg.get("nomask", True):
        mask_f16 = mask.to(torch.float16).view(total_rows, 1)   # broadcast over H
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # back to 4‑D layout
    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # --------------------------------------------------------------
    # 3️⃣ Batched matmul on the native layout (no copies)
    # --------------------------------------------------------------
    out = torch.empty_like(left)            # fp16, shape [B,N,N,H]

    stride_l = left.stride()                # (B,N,N,H) → (b,i,k,h)
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_MAT_M = 64
    BLOCK_MAT_N = 64
    BLOCK_MAT_K = 64

    num_i_tiles = math.ceil(N / BLOCK_MAT_M)
    grid = (B * H * num_i_tiles, math.ceil(N / BLOCK_MAT_N))
    batched_matmul_out_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_MAT_M,
        BLOCK_N=BLOCK_MAT_N,
        BLOCK_K=BLOCK_MAT_K,
    )

    # --------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, output‑gate and final linear
    # --------------------------------------------------------------
    out_flat      = out.view(total_rows, H).contiguous()          # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous()    # [R, H] fp16
    final_half    = torch.empty(total_rows, dim,
                                dtype=torch.float16, device=device)  # [R, dim] fp16

    BLOCK_FINAL_M = 256
    BLOCK_FINAL_H = 64
    BLOCK_FINAL_O = 64
    grid_final = (math.ceil(total_rows / BLOCK_FINAL_M),)

    final_fused_kernel[grid_final](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        weights["to_out.weight"].to(torch.float16),  # [dim, H] fp16
        final_half,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_FINAL_M,
        BLOCK_H=BLOCK_FINAL_H,
        BLOCK_O=BLOCK_FINAL_O,
    )

    # --------------------------------------------------------------
    # 5️⃣ Cast back to float32 and return
    # --------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
