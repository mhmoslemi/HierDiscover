import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass: sum and sum of squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)
        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: write normalized value
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)
        norm = (val_f32 - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    # dimensions
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides (in elements, not bytes)
    stride_b: tl.constexpr,
    stride_i: tl.constexpr,
    stride_k: tl.constexpr,
    stride_h: tl.constexpr,
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # pid0 encodes batch * head (BH)
    pid_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head
    h = pid_head % H
    b = pid_head // H

    # offsets for rows and cols of the output tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base offset for this (b, h) slice
    base = b * stride_b + h * stride_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [M, K]   -> rows i, columns k
        left_ptrs = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed: [K, N]   -> rows k, columns j
        right_ptrs = right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot product over K
        acc += tl.dot(left, right)

    # write back result in fp16
    out_ptrs = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.
    1) Row‑wise LayerNorm (FP32 → FP16) via a custom Triton kernel.
    2) One fused linear projection that simultaneously produces the two
       projections and the three gating vectors.
    3) Optional mask + sigmoid gating (all FP16).
    4) Triangular multiplicative update implemented as a batched GEMM in
       Triton, avoiding the costly permute‑contiguous copies.
    5) Second LayerNorm (FP16) via the same Triton kernel.
    6) Output gate + final projection back to the model dimension.
    Returns a FP32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (row‑wise)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N

    # flatten to [R, dim] (the input is FP32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # output will be FP16
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 256
    BLOCK_D_LN = 128   # reduces number of passes over dim
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ------------------------------------------------------------------
    # 2) Fuse five linear layers (proj + three gates)
    # ------------------------------------------------------------------
    # concatenate weight matrices of shape [5*H, dim] (FP16)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused linear: [R, 5*H]
    proj_out = F.linear(x_norm, w_concat)   # bias is None as per model

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate_p = proj_out[:, 2*H:3*H]
    right_gate_p= proj_out[:, 3*H:4*H]
    out_gate_p  = proj_out[:, 4*H:5*H]

    # sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # optional mask – broadcast over the hidden dimension
    if not nomask:
        mask_expand = mask_tensor.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left_proj = left_proj.view(B, N, N, H) * mask_expand
        right_proj = right_proj.view(B, N, N, H) * mask_expand
    else:
        left_proj = left_proj.view(B, N, N, H)
        right_proj = right_proj.view(B, N, N, H)

    # gating
    left  = left_proj * left_gate.view(B, N, N, H)
    right = right_proj * right_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------
    # left/right/out_gate are now FP16 tensors of shape [B, N, N, H]
    out = torch.empty_like(left)   # allocate output tensor

    # strides for the (B,N,N,H) layout
    stride_b, stride_i, stride_k, stride_h = left.stride()
    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate   # element‑wise (FP16)

    out_norm_flat = out_norm.view(total_rows, H).contiguous()
    final_half = F.linear(out_norm_flat,
                          weights["to_out.weight"].to(torch.float16))

    # cast back to FP32
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

def layernorm_kernel(

    y_ptr,               # mutable *  [R, dim]   (fp16)
    weight_ptr,          # const   *  [dim]      (fp32)
    bias_ptr,            # const   *  [dim]      (fp32)
    total_rows,          # int (runtime)
    dim: tl.constexpr,   # compile‑time
    eps: tl.constexpr,   # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]

    # head‑wise program id (combined batch * head)
    pid_head = tl.program_id(0)
    pid_i    = tl.program_id(1)
    pid_j    = tl.program_id(2)

    # decode batch b and head h
    h = pid_head % H
    b = pid_head // H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (b,h) slice
    base = b * stride_b + h * stride_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  [M, K]   (rows i, cols k)
        left_ptrs = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed  [K, N]   (rows k, cols j)
        right_ptrs = right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (M,N) tile accumulation

    # write result back (FP16)
    out_ptrs = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry‑point: outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

    Steps
    -----
    1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16).
    2️⃣ One fused linear projection that simultaneously yields the two
       projections (`left`, `right`) and the three gating vectors.
    3️⃣ Optional pairwise mask (only on the two projections) followed by
       sigmoid gating (all FP16).
    4️⃣ Triangular multiplicative update implemented as a batched GEMM in
       Triton – avoids the costly ``permute``/``contiguous`` copies of the
       reference implementation.
    5️⃣ Second LayerNorm over the hidden dimension (FP16).
    6️⃣ Element‑wise output‑gate multiplication and final projection back to
       the model dimension.  The final tensor is cast to FP32.
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 256
    BLOCK_D_LN = 128                     # reduces passes over `dim`
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (proj + three gates)
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)   # [R, 5*H]  (fp16)

    H = hidden_dim
    left_proj   = proj[:, 0:H]
    right_proj  = proj[:, H:2*H]
    left_gate_p = proj[:, 2*H:3*H]
    right_gate_p= proj[:, 3*H:4*H]
    out_gate_p  = proj[:, 4*H:5*H]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # reshape to [B, N, N, H]
    left_proj = left_proj.view(B, N, N, H)
    right_proj = right_proj.view(B, N, N, H)

    # --------------------------------------------------------------
    # optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask_tensor.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # gating
    left = left_proj * left_gate.view(B, N, N, H)
    right = right_proj * right_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update – Triton GEMM
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # [B, N, N, H] fp16

    # strides for the (B,N,N,H) layout
    stride_b, stride_i, stride_k, stride_h = left.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Output‑gate & final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate           # element‑wise (fp16)

    out_norm_flat = out_norm.view(R, H).contiguous()
    final_fp16 = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)
    )                                        # [R, dim] fp16

    # cast to FP32 for downstream use
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
