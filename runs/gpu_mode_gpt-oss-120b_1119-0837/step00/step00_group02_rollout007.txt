@triton.jit
def trmul_outgoing_kernel(
   left_ptr, right_ptr, out_ptr,
   B, N, hidden_dim, eps,
   stride_left_i, stride_left_k, stride_left_d,
   stride_right_i, stride_right_k, stride_right_d,
   stride_out_i, stride_out_j, stride_out_d,
   BLOCK_K: tl.constexpr,
   hidden_dim: tl.constexpr
):
    pid_b = tl.program_id(2) # batch dimension maybe id 0 for i, id1 for j, id2 for batch, but can also combine.
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    # compute global i/j indices
    i = pid_i
    j = pid_j
    # check bounds
    mask = (i < N) & (j < N)
    # create accumulator
    acc = tl.zeros((hidden_dim,), dtype=tl.float32)

    # K loop
    for k_base in range(0, N, BLOCK_K):
        cur_k = k_base + tl.arange(0, BLOCK_K)
        # ensure k within bounds
        k_mask = cur_k < N
        # Load left values for current i across k: shape (BLOCK_K, hidden_dim)
        # we need left[i, k, d]; left stride: i*stride_left_i + k*stride_left_k + d*stride_left_d
        left_offset = i * stride_left_i + cur_k[:, None] * stride_left_k + tl.arange(0, hidden_dim)[None, :] * stride_left_d
        left_vals = tl.load(left_ptr + left_offset, mask=k_mask[:, None], other=0.0)  # shape (BLOCK_K, hidden_dim)
        # Load right values for current j across k: shape (BLOCK_K, hidden_dim)
        right_offset = j * stride_right_i + cur_k[:, None] * stride_right_k + tl.arange(0, hidden_dim)[None, :] * stride_right_d
        right_vals = tl.load(right_ptr + right_offset, mask=k_mask[:, None], other=0.0)  # shape (BLOCK_K, hidden_dim)
        # Multiply elementwise and sum across k
        # left_vals * right_vals -> shape (BLOCK_K, hidden_dim); sum over K
        # We can do tl.sum across K dimension after elementwise product.
        prod = left_vals * right_vals
        acc += tl.sum(prod, axis=0)  # sum across K, resulting shape (hidden_dim,)
    # store out
    out_offset = i * stride_out_i + j * stride_out_j + tl.arange(0, hidden_dim) * stride_out_d
    tl.store(out_ptr + out_offset, acc, mask=mask)  # Actually mask is per (i,j) not per d.

import triton.language as tl

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    i = pid_i
    j = pid_j

    mask = (i < N) & (j < N)
    if not mask:
        return

    # Accumulator for hidden_dim
    acc = tl.zeros((hidden_dim,), dtype=tl.float32)

    # Load offsets for d dimension (vectorized)
    offs_d = tl.arange(0, hidden_dim)

    # Loop over k dimension in tiles
    for k_base in range(0, N, BLOCK_K):

        k_mask = cur_k < N

        # Load left slice: shape (BLOCK_K, hidden_dim)

    left_ptr, right_ptr, out_ptr,

    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,

):

    i = pid_i
    j = pid_j

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr
):
    # Compute program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    # Compute base indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Compute bounds for i/j
    mask_i = i_base + tl.arange(0, BLOCK_M) < N
    mask_j = j_base + tl.arange(0, BLOCK_N) < N
    mask_ij = mask_i[:, None] & mask_j[None, :]  # shape (BLOCK_M, BLOCK_N)
    # If no valid i,j in this block, return early
    if not tl.reduce(mask_ij):
        return

    # Offsets for hidden dim
    offs_d = tl.arange(0, hidden_dim)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, hidden_dim), dtype=tl.float32)

    # Loop over k dimension
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = cur_k < N

        # Load left tile (M,K,D)
        i_off = i_base + tl.arange(0, BLOCK_M)[:, None, None]  # (M,1,1)
        k_off = cur_k[None, :, None]  # (1,K,1)
        left_offsets = i_off * stride_left_i + k_off * stride_left_k + offs_d[None, None, :] * stride_left_d
        left_tile = tl.load(left_ptr + left_offsets,
                            mask=mask_k[None, :, None],
                            other=0.0,
                            dtype=tl.float16).to(tl.float32)  # (M,K,D)

        # Load right tile (K,N,D) - note ordering K,N for broadcasting
        j_off = j_base + tl.arange(0, BLOCK_N)[None, :, None]  # (1,N,1)
        right_offsets = j_off * stride_right_i + k_off * stride_right_k + offs_d[None, None, :] * stride_right_d
        right_tile = tl.load(right_ptr + right_offsets,
                             mask=mask_k[:, None, None],
                             other=0.0,
                             dtype=tl.float16).to(tl.float32)  # (K,N,D)

        # Compute outer product across current K slice
        for kk in range(BLOCK_K):
            # Conditional: if cur_k[kk] out of bounds, the loaded vectors are zero due to mask
            left_vec = left_tile[:, kk, :]          # (M,D)
            right_vec = right_tile[kk, :, :]        # (N,D)
            # broadcast:
            prod = left_vec[:, None, :] * right_vec[None, :, :]  # (M,N,D)
            acc += prod

        k += BLOCK_K

    # Store result (M,N,D)
    # Compute offsets for output
    i_off = i_base + tl.arange(0, BLOCK_M)[:, None, None]  # (M,1,1)
    j_off = j_base + tl.arange(0, BLOCK_N)[None, :, None]  # (1,N,1)
    out_offsets = i_off * stride_out_i + j_off * stride_out_j + offs_d[None, None, :] * stride_out_d
    # Apply mask to avoid OOB writes
    store_mask = mask_ij[..., None]  # (M,N,1)
    tl.store(out_ptr + out_offsets, acc.to(tl.float16), mask=store_mask)

"""
tri_mul_outgoing.py
-------------------
A Triton‑accelerated implementation of the *outgoing* Triangle Multiplicative
Update (TriMul) used in AlphaFold3 and related protein‑structure models.
The heavy O(N³) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed in a custom Triton kernel.  All other
operations (layer‑norms, linear projections, gating) are kept in PyTorch
for simplicity and to reuse existing weight tensors.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: compute a tile of the TriMul output.
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # memory pointers
    N,                                          # sequence length (runtime)
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,                     # tile size in i dimension
    BLOCK_N: tl.constexpr,                     # tile size in j dimension
    BLOCK_K: tl.constexpr,                     # tile size in k dimension
    hidden_dim: tl.constexpr,                  # channel dimension (compile time)
):
    """Compute a BLOCK_M × BLOCK_N × hidden_dim tile of the outgoing TriMul.
    Each program (i.e. CUDA thread‑block) works on a tile of output pairs
    (i, j).  Inside the tile we keep an accumulator of shape
    (BLOCK_M, BLOCK_N, hidden_dim) in FP32, broadcast loads of half‑precision
    (FP16) left/right slices, multiply them and accumulate.
    """
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile column index

    i_base = pid_i * BLOCK_M          # first i handled by this tile
    j_base = pid_j * BLOCK_N          # first j handled by this tile

    # ------------------------------------------------------------------
    # Masks for out‑of‑bounds rows/cols (necessary for the last partially
    # filled tile).  The masks are later used for a safe store.
    # ------------------------------------------------------------------
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)
    mask_i = i_range < N               # (BLOCK_M,)
    mask_j = j_range < N               # (BLOCK_N,)
    mask_ij = mask_i[:, None] & mask_j[None, :]   # (BLOCK_M, BLOCK_N)

    # Offsets along the hidden dimension (vectorised load)
    offs_d = tl.arange(0, hidden_dim)          # (hidden_dim,)

    # ------------------------------------------------------------------
    # Accumulator for the tile – FP32 for numerical stability.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, hidden_dim), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K.
    # ------------------------------------------------------------------
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        mask_k = cur_k < N                         # (BLOCK_K,)

        # ------------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K, hidden_dim)
        #   left[i_base + i, cur_k + k, d]
        # ------------------------------------------------------------------
        i_off = i_base + tl.arange(0, BLOCK_M)[:, None, None]   # (M,1,1)
        k_off = cur_k[None, :, None]                           # (1,K,1)
        left_offsets = (i_off * stride_left_i
                        + k_off * stride_left_k
                        + offs_d[None, None, :] * stride_left_d)
        left_tile = tl.load(left_ptr + left_offsets,
                            mask=mask_k[None, :, None],
                            other=0.0,
                            dtype=tl.float16).to(tl.float32)   # (M,K,D)

        # ------------------------------------------------------------------
        # Load right tile: shape (BLOCK_K, BLOCK_N, hidden_dim)
        #   right[j_base + j, cur_k + k, d]  (note the order K×N)
        # ------------------------------------------------------------------
        j_off = j_base + tl.arange(0, BLOCK_N)[None, :, None]   # (1,N,1)
        right_offsets = (j_off * stride_right_i
                         + k_off * stride_right_k
                         + offs_d[None, None, :] * stride_right_d)
        right_tile = tl.load(right_ptr + right_offsets,
                             mask=mask_k[:, None, None],
                             other=0.0,
                             dtype=tl.float16).to(tl.float32)  # (K,N,D)

        # ------------------------------------------------------------------
        # Accumulate outer‑product over the K‑tile:
        #   acc[i,j,:] += Σₖ left[i,k,:] * right[k,j,:]
        # ------------------------------------------------------------------
        for kk in range(BLOCK_K):
            # left_vec: (BLOCK_M, hidden_dim)
            # right_vec: (BLOCK_N, hidden_dim)
            left_vec  = left_tile[:, kk, :]           # (M,D)
            right_vec = right_tile[kk, :, :]          # (N,D)

            # Broadcast and multiply -> (M,N,D)
            prod = left_vec[:, None, :] * right_vec[None, :, :]
            acc += prod

        k += BLOCK_K

    # ------------------------------------------------------------------
    # Store the tile back to global memory.  Mask prevents OOB writes.
    # ------------------------------------------------------------------
    i_off = i_base + tl.arange(0, BLOCK_M)[:, None, None]   # (M,1,1)
    j_off = j_base + tl.arange(0, BLOCK_N)[None, :, None]   # (1,N,1)
    out_offsets = (i_off * stride_out_i
                   + j_off * stride_out_j
                   + offs_d[None, None, :] * stride_out_d)

    # Apply the i/j mask; broadcast over the hidden dimension
    store_mask = mask_ij[..., None]                         # (M,N,1)
    tl.store(out_ptr + out_offsets,
             acc.to(tl.float16),
             mask=store_mask)


# ----------------------------------------------------------------------
# Public entry point expected by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input  : tensor of shape [B, N, N, dim]
        - mask   : tensor of shape [B, N, N] (may be None when config["nomask"]=True)
        - weights: dict with the same keys as the PyTorch reference module
        - config : dict containing ``dim``, ``hidden_dim`` and ``nomask``.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------- unpack ---------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------- LayerNorm -------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # --------------------------- Linear projections -----------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------- optional mask ---------------------------
    if not config.get("nomask", False):
        # mask has shape [B,N,N]; broadcast to the channel dimension
        mask_f = mask_tensor.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------- gating ----------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------- prepare for Triton -----------------------
    # Cast to FP16 to reduce memory bandwidth; accumulation stays FP32.
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_f16.shape  # re‑extract after possible broadcasting

    # Output buffer (FP16) – will be cast back to FP32 after the kernel.
    out_f16 = torch.empty((B, N, N, hidden_dim),
                          dtype=torch.float16,
                          device=device)

    # Strides (in number of elements, not bytes)
    # left/right: [B, i, k, d]
    s_li = left_f16.stride(1)   # stride over i
    s_lk = left_f16.stride(2)   # stride over k
    s_ld = left_f16.stride(3)   # stride over d (==1)
    s_ri = right_f16.stride(1)
    s_rk = right_f16.stride(2)
    s_rd = right_f16.stride(3)
    # out: [B, i, j, d]
    s_oi = out_f16.stride(1)
    s_oj = out_f16.stride(2)
    s_od = out_f16.stride(3)

    # Tile size – chosen to keep the per‑program register usage
    # within limits while exposing enough parallelism.
    BLOCK_M = 8
    BLOCK_N = 8
    BLOCK_K = 8

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N )

    # Launch one kernel per batch element (B≤2 for the test suite)
    for b in range(B):
        trmul_outgoing_kernel[grid](
            left_f16[b].data_ptr(),
            right_f16[b].data_ptr(),
            out_f16[b].data_ptr(),
            N,
            s_li, s_lk, s_ld,
            s_ri, s_rk, s_rd,
            s_oi, s_oj, s_od,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            hidden_dim=hidden_dim,
            num_warps=4,          # 4 warps ≈ 128 threads per block – good occupancy
        )

    # Cast back to FP32 for the remaining operations
    out = out_f16.to(torch.float32)

    # --------------------------- post‑processing -------------------------
    # LayerNorm on the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to the original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
