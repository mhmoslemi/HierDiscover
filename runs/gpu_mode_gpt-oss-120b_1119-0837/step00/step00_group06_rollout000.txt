"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided weight/bias.
2. Linear projections (left/right) and gated linear layers (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear.
3. Optional mask is applied to the left/right projections.
4. The core “triangular multiplicative update”:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is rewritten as a batch of independent matrix‑multiplications
   `out_d = left_d @ right_dᵀ` for every hidden dimension `d`.
   A custom Triton kernel fuses the k‑reduction, loads the
   left/right tiles with stride‑aware addressing and uses `tl.dot`
   (half‑precision arithmetic, accumulation in FP32) to compute the
   result.  The kernel works on the flattened `(B*hidden_dim, N, N)`
   view and writes the result directly into an FP16 tensor.
5. The result is cast back to FP32, layer‑normed, gated by `out_gate`,
   and finally projected back to the original channel dimension `C`.
6. The function returns a tensor of shape (B,N,N,C) in FP32.

The kernel processes one `(i‑block, j‑block)` tile per program instance,
loops over the `k` dimension in tiles, and respects arbitrary masks.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the triangular multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # B=batch, N=seq_len, H=hidden_dim
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H).  The kernel treats (b,h)
    as a merged axis and tiles the i/j dimensions.
    """
    pid = tl.program_id(0)          # encodes merged (b,h) and block_i
    pid_j = tl.program_id(1)        # block_j

    # number of i‑blocks
    num_blocks_i = tl.cdiv(N, BLOCK_M)

    # decode batch, hidden, and i‑block
    bh = pid // num_blocks_i                # merged batch*hidden index
    block_i = pid % num_blocks_i

    b = bh // H                              # original batch index
    h = bh % H                               # hidden dimension index

    # i / j offsets within this tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j   * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for the current (b,h) slice
    base_l = b * stride_l_batch + h * stride_l_h
    base_r = b * stride_r_batch + h * stride_r_h
    base_o = b * stride_o_batch + h * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)
        ptr_l = left_ptr + base_l \
                + offs_i[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        l = tl.load(ptr_l, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: shape (BLOCK_N, BLOCK_K)
        ptr_r = right_ptr + base_r \
                + offs_j[:, None] * stride_r_i \
                + offs_k[None, :] * stride_r_k
        r = tl.load(ptr_r, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot product over k (accumulation in FP32)
        acc += tl.dot(l, r)

    # write out: cast back to FP16 (the output tensor allocated as half)
    ptr_o = out_ptr + base_o \
            + offs_i[:, None] * stride_o_i \
            + offs_j[None, :] * stride_o_j
    tl.store(ptr_o, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # Apply optional mask (mask shape (B,N,N) or None)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half-precision for the heavy kernel
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h   = torch.empty_like(left_h)            # (B,N,N,H) in FP16

    # ------------------------------------------------------------------
    # Prepare stride arguments for Triton (all strides are in element units)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_h.stride()

    # ------------------------------------------------------------------
    # Launch TriMul kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    # merged batch*hidden dimension
    BH = B * hidden_dim
    grid = (BH * num_blocks_i, num_blocks_j)

    tri_mul_kernel[grid](
        left_h, right_h, out_h,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # a modest number of warps works well on H100
    )

    # ------------------------------------------------------------------
    # Continue in FP32
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # to_out_norm (LayerNorm on hidden dimension)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply final gate and linear projection back to C
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,C)

    return out
```
