    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute combined batch index
    pid_bh = tl.program_id(0)  # 0..B*H-1
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute batch index b, hidden index h
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for each operand
    # start offsets for pointers
    # for left: base + b*stride_lb + h*stride_lh
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # We need to keep within N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # Compute base pointers for this slice
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh

"""
custom_kernel: High‑performance forward pass of the TriMul “outgoing” operator.

Algorithm
---------
1. Layer‑norm on the input tensor (last dimension = dim).
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Apply optional mask (broadcast over hidden_dim) and element‑wise sigmoid gates.
4. Fuse the heavy N³ reduction with a custom Triton kernel:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   The kernel treats (batch, hidden) as a combined batch dimension and
   performs a batched matrix multiplication (A @ Bᵀ) for each hidden slice.
   Inputs are FP16, accumulation is FP32, output is stored as FP16.
5. Convert the result to FP32, apply a second Layer‑norm on the hidden_dim,
   multiply by the out‑gate, and finish with a linear projection back to dim.
The implementation mixes PyTorch (layer‑norms, linear layers, gating) with a
Triton kernel for the N³ inner product, achieving large speed‑ups on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # runtime sizes
    stride_lb, stride_li, stride_lk, stride_lh,       # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,       # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,       # out   strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (b,h) are fused into a single program‑id axis.
    """
    pid_bh = tl.program_id(0)      # combined (batch, hidden) index
    pid_m = tl.program_id(1)       # tile over i (rows)
    pid_n = tl.program_id(2)       # tile over j (cols)

    # De‑compose (batch, hidden) index
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left block (A) ----------
        # A[i,k]  with stride (row) = stride_li, (col) = stride_lk
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # -------- right block (Bᵀ) ----------
        # Bᵀ[k,j] where original right is indexed as (b, j, k, h).
        # Row stride = stride_rk (k dimension), column stride = stride_ri (j dimension)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate dot product (FP32)
        acc += tl.dot(a, b)

    # Write the result (cast back to FP16)
    c_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)    # (B,N,N,hidden_dim)

    # ---------- Optional mask ----------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask
    mask_tensor = mask_tensor.unsqueeze(-1)  # (B,N,N,1)

    left  = left * mask_tensor
    right = right * mask_tensor

    # ---------- Gating ----------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates and cast to FP16 for the kernel
    left = (left * left_gate).to(torch.float16).contiguous()
    right = (right * right_gate).to(torch.float16).contiguous()

    # ---------- Triton N³ reduction ----------
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # (B,N,N,hidden_dim)

    # Compute strides for the tensors (they are in row‑major order)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()   # (B,N,N,H)
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Grid: (batch*hidden, ceil(N/M), ceil(N/N))
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_matmul_kernel[grid](
        left,                     # left_ptr
        right,                    # right_ptr
        out,                      # out_ptr
        B, N, hidden_dim,         # runtime sizes
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ---------- Post‑processing ----------
    out = out.to(torch.float32)   # convert back to FP32 for the remaining ops

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape (B,N,N,dim)

    return out
