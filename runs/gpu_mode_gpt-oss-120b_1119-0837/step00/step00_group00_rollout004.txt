import torch
import triton
import triton.language as tl

def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    dtype = x.dtype
    device = x.device

    # LayerNorm: compute mean and var over last dim dim
    eps = 1e-5
    # x_norm = (x - mean) / sqrt(var + eps) * norm_weight + norm_bias
    # compute:
    x_flat = x.view(-1, dim)
    mean = x_flat.mean(-1, keepdim=True)
    var = x_flat.var(-1, unbiased=False, keepdim=True)
    x_norm = (x_flat - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    x_norm = x_norm.view(bs, N, N, dim)
    # small dims: okay

    # Convert to fp16 for speed
    x_norm_h = x_norm.to(torch.float16)
    # Linear projections
    left_proj_weight_h = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x_norm_h, left_proj_weight_h, bias=None)
    right = torch.nn.functional.linear(x_norm_h, right_proj_weight_h, bias=None)
    # left,right shape [bs,N,N,hidden_dim] (float16)
    
    # mask apply
    if not config.get("nomask", True):
        mask_h = mask.to(torch.float16).unsqueeze(-1) # shape [bs,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Gating
    left_gate_weight_h = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight_h = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_norm_h, left_gate_weight_h, bias=None)
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_weight_h, bias=None)
    out_gate = torch.nn.functional.linear(x_norm_h, out_gate_weight_h, bias=None)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # Core tri-mul: batched matmul across hidden_dim
    # reshape to (bs*hidden_dim, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N) # left: [bs, N, N, H] -> [bs*H, N, N]
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # compute torch.bmm left @ right^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2)) # shape [bs*H, N, N]
    # reshape back to [bs, N, N, H]
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute layernorm manually
    out_flat = out.view(-1, hidden_dim)
    mean_out = out_flat.mean(-1, keepdim=True)
    var_out = out_flat.var(-1, unbiased=False, keepdim=True)
    out_norm = (out_flat - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias
    out_norm = out_norm.view(bs, N, N, hidden_dim)

    # apply out_gate (broadcast)
    out_gate = out_gate  # shape [bs,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # final linear
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_final = torch.nn.functional.linear(out_norm.to(torch.float16), to_out_weight_h, bias=None)
    # return float32
    return out_final.to(torch.float32)

"""
TriMul custom kernel (outgoing version)

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Project the normalized tensor to a hidden dimension `hidden_dim` with two linear
   layers (`left_proj` and `right_proj`).  Biases are omitted as in the reference.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) with linear
   layers and apply them element‑wise to the projected tensors.  An optional
   `mask` (shape [B,N,N]) is also applied after gating.
4. Core operation – “TriMul” – is a batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   The hidden dimension `h` (including the batch dimension) is flattened and the
   multiplication is performed by a custom Triton kernel that computes many
   independent N×N matmuls (one per hidden slice) in FP16 with FP32 accumulation.
5. Layer‑norm the result over the hidden dimension, multiply by `out_gate`,
   and finally project back to the original dimension with a linear layer.
6. The final output is returned as float32.

The Triton kernel fuses the N³ work of the TriMul step, while the surrounding
operations (layer‑norm, linear projections and gating) are performed with
high‑performance PyTorch primitives.  The implementation works for the
configured ranges of batch size, sequence length and feature dimensions.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B^T) for many independent slices.
# The input tensors are assumed to be contiguous with shape
#   A: [batch, M, K]   (here M = N, K = N)
#   B: [batch, N, K]   (same layout as A; we use the transposed view by
#                       swapping the stride for the K‑dimension in the kernel)
# Output:
#   C: [batch, M, N]
# All tensors are FP16; accumulation happens in FP32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel computing C = A @ Bᵀ (A,B FP16, C FP16)."""
    pid_batch = tl.program_id(0)       # which slice (batch * hidden)
    pid_m = tl.program_id(1)           # block row index
    pid_n = tl.program_id(2)           # block column index

    # ------------------------------------------------------------------
    # Compute the tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows of A / C
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols of B / C
    offs_k = tl.arange(0, BLOCK_K)                     # reduction axis

    # Mask for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for A (shape [batch, M, K]) and B (shape [batch, N, K])
    # B will be accessed as Bᵀ by swapping the logical axes:
    #   Bᵀ[tile] = B[batch, ?, k] where ? = col index (offs_n)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                    + offs_m[:, None] * stride_a_m \
                    + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                    + offs_k[:, None] * stride_b_k \
                    + offs_n[None, :] * stride_b_n

    # ------------------------------------------------------------------
    # Accumulator (FP32) for the tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        # Load a tile of A and B (both FP16)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # Convert to FP32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Compute the dot product of the tile
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write the result back (cast to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                       + offs_m[:, None] * stride_c_m \
                       + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point callable from the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a Triton‑accelerated
    core multiplication.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # should be torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension `dim`
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear / matmul work
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Projections (left/right) and gates (left_gate/right_gate/out_gate)
    # ------------------------------------------------------------------
    # Linear weights (FP16)
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_w = weights["to_out.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # Linear projections (bias=False as in reference)
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (mask shape: [B,N,N]; broadcast over hidden dim)
    if not no_mask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3. Core TriMul operation: compute Σ_k left[...,k,:] * right[...,k,:]ᵀ
    #    We fuse the batch and hidden dimensions (B*hidden) and launch the
    #    Triton kernel described above.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_slices = B * hidden_dim                         # flattened batch‑hidden dimension

    # Rearrange tensors to [B*hidden, N, N] (contiguous)
    # Permute to bring the hidden dim next to batch, then flatten.
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(total_slices, N, N)       # [B*H, N, N]
    right_flat = right_perm.view(total_slices, N, N)

    # Allocate output tensor (FP16)
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (contiguous layout)
    stride_batch = left_flat.stride(0)    # = N * N
    stride_m = left_flat.stride(1)       # = N
    stride_k = left_flat.stride(2)       # = 1

    # Strides for the right matrix (we will access it as Bᵀ)
    # For transposed access we need:
    #   stride_b_n : stride of the column index j (original row) = N
    #   stride_b_k : stride of the reduction index k (original col) = 1
    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)    # = N   (row stride)
    stride_b_k = right_flat.stride(2)    # = 1   (col stride)

    # Output strides (same layout as left)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Block sizes – tuned for H100 (multiple of 32 for good occupancy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_slices,                      # axis 0 -> folded batch/hidden
        (N + BLOCK_M - 1) // BLOCK_M,      # axis 1 -> rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,      # axis 2 -> cols (j)
    )

    # Launch the kernel
    tri_mul_batch_matmul_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        total_slices,
        N,
        N,
        N,
        stride_batch,
        stride_m,
        stride_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=4,               # reasonable default for FP16 matmul
    )

    # Reshape back to [B,N,N,hidden]
    out_h = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4. Post‑processing: layer‑norm over hidden dim, gating, final projection
    # ------------------------------------------------------------------
    # Convert to float32 for the remaining ops (better numerical stability)
    out = out_h.to(torch.float32)

    # Layer‑norm over the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply the output gate (broadcasted over the hidden dim)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim` (bias=False)
    out = torch.nn.functional.linear(out.half(), to_out_w).to(torch.float32)

    return out
