@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None,:] < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)
        # Note: Actually we need to offset a_ptrs and b_ptrs each iteration by BLOCK_K * respective stride
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
        # increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy pairwise contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batched matrix multiplication:
    out = (left_reshape) @ (right_reshape.T)
where each batch corresponds to a (batch, hidden) pair of N×N matrices.
We implement this contraction with a Triton kernel that performs
a batched GEMM in FP16 (accumulated in FP32). All other
operations (LayerNorms, linear projections, gating, final
projection) are expressed with PyTorch – this satisfies the
requirement of having “part of the operations” in a kernel while
keeping the implementation simple and performant.

The kernel works for any batch size (1‑2), sequence length up to 1024,
input dimension `dim` (128 or 384) and hidden dimension `hidden_dim`
(typically 128). The final output is returned in float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute batched matrix multiplication:
        C[b] = A[b] @ B[b]   for b in [0, batch)
    where:
        A : (batch, M, K)
        B : (batch, K, N)
        C : (batch, M, N)
    Input matrices are expected in FP16 and accumulation is performed in FP32.
    """
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block column

    # Coordinates within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds accesses
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = offs_k < K

        # Pointers to the current blocks of A and B
        a_ptrs = a_ptr \
                 + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr \
                 + batch_id * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load with masks (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

    # Write result back in FP16
    c_ptrs = c_ptr \
             + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape (B, N, N, dim) – keep float32

    # ------------------------------------------------------------------
    # 2. Linear projections + gating
    # ------------------------------------------------------------------
    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Optional mask (broadcast to last dim)
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)       # (B,N,N,1)
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 before the heavy contraction to save memory / bandwidth
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Heavy contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM: out = left_reshape @ (right_reshape)^T
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) for easier batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)

    # Transpose the inner N×N of right so that we can do L @ R^T
    right_t = right_perm.transpose(-1, -2).contiguous()    # (B, H, N, N)

    # Fuse batch and hidden dimensions
    B_H = B * hidden_dim
    a = left_perm.reshape(B_H, N, N)          # (B*H, N, N)
    b = right_t.reshape(B_H, N, N)            # (B*H, N, N)

    # Allocate output tensor (FP16)
    c = torch.empty((B_H, N, N), dtype=torch.float16, device=device)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B_H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        batch=B_H,
        M=N,
        N=N,
        K=N,
        stride_a_batch=a.stride(0),
        stride_a_m=a.stride(1),
        stride_a_k=a.stride(2),
        stride_b_batch=b.stride(0),
        stride_b_k=b.stride(1),
        stride_b_n=b.stride(2),
        stride_c_batch=c.stride(0),
        stride_c_m=c.stride(1),
        stride_c_n=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape result back to (B, N, N, hidden)
    out = c.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out‑gate (broadcasted)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # Return in float32 (matching the reference implementation)
    return out.to(torch.float32)
