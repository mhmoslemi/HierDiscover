@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute row and col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_batch * stride_Ab + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak)
        b_ptrs = B_ptr + pid_batch * stride_Bb + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_Cb + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']
    device = input_tensor.device
    # LayerNorm on input
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    to_out_w = weights['to_out.weight']

    # Linear projections (no bias)
    # shape [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)   # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask if present (mask may be None? In dataset config, we have mask input always present but maybe all ones)
    if mask_tensor is not None:
        # mask shape [B,N,N], broadcast to hidden_dim
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Transpose right for multiplication (swap axes 1 and 2)
    # right shape [B,N,N,H] -> [B,N,N,H] after permute(0,2,1,3)
    right_T = right.permute(0,2,1,3).contiguous()

    # Prepare for kernel: bring batch and head dims together
    B, N, _, H = left.shape
    # Permute to [B, H, N, N] for both left and right_T
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right_T.permute(0,3,1,2).contiguous()
    # Cast to float16 for kernel (saving memory)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Reshape to (B*H, N, N)
    batch_head = B * H
    left_mat = left_fp16.view(batch_head, N, N)
    right_mat = right_fp16.view(batch_head, N, N)

    # Allocate output in float32 (accumulate in float32)
    out_mat = torch.empty((batch_head, N, N), dtype=torch.float32, device=device)

    # Define kernel block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Get strides
    stride_Ab, stride_Am, stride_Ak = left_mat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_mat.stride()

"""
custom_kernel
==============

Implements the *outgoing* Triangle Multiplicative Update (TriMul) operation from
AlphaFold‑3.  The heavy “pair‑wise” multiplication

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is fused into a single Triton kernel that performs a batched matrix multiply
over the sequence dimension *N* for every hidden channel *d* and batch *b*.
All other operations (LayerNorms, linear projections, sigmoids, masking) are
performed with regular PyTorch kernels.  Mixed‑precision (FP16 for the large
matrices, FP32 accumulation) is used to keep memory footprint low while still
delivering the required <1 ms runtime on an H100.

The function signature is:
    custom_kernel(data) → torch.Tensor
where ``data`` is a tuple ``(input, mask, weights, config)`` described in the
problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (FP16 inputs, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,        # pointers to A, B, C
    M, N, K,                    # matrix dimensions (all = N for our use case)
    stride_Ab, stride_Am, stride_Ak,   # strides of A: batch, row (M), col (K)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B: batch, row (K), col (N)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,      # tile size in M dimension
    BLOCK_N: tl.constexpr,      # tile size in N dimension
    BLOCK_K: tl.constexpr,      # tile size in K dimension
):
    # 3‑D grid: (batch_head, block_m, block_n)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B tiles
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + (offs_m[:, None] * stride_Am)
            + (offs_k[None, :] * stride_Ak)
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + (offs_k[:, None] * stride_Bk)
            + (offs_n[None, :] * stride_Bn)
        )

        # Load with masking (outside‑bounds → 0)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # a, b are FP16; the dot product uses Tensor‑cores and accumulates in FP32
        acc += tl.dot(a, b)

    # Write the result back to C (FP32)
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + (offs_m[:, None] * stride_Cm)
        + (offs_n[None, :] * stride_Cn)
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Main entry point
# ----------------------------------------------------------------------

    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------

        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights_dict : mapping from weight names to torch.Tensors (already on GPU)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (dtype float32)
    """

    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    # ------------------------------------------------------------------
    # Pull out weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape -> [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (hidden_dim, dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on last dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating logits + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Prepare operands for the heavy einsum:
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    = left[b,:, :, d] @ right[b,:, :, d].T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B = batch, N = seq_len, H = hidden_dim

    # Transpose the second N‑dimension of `right`
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, H]

    # Bring (batch, channel) together for a true batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_t.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # Cast to FP16 to save memory and accelerate Tensor‑core compute
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Merge batch & channel dimensions: (B*H, N, N)
    batch_head = B * H
    left_mat = left_fp16.view(batch_head, N, N)
    right_mat = right_fp16.view(batch_head, N, N)

    # Output buffer (FP32 accumulation)
    out_mat = torch.empty((batch_head, N, N), dtype=torch.float32, device=device)

    # ---------------------------------------------------------------
    # 7) Triton batched matrix multiplication
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = left_mat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_mat.stride()

    grid = (
        batch_head,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default for H100
    )

    # ---------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ---------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ---------------------------------------------------------------
    # 9) Final hidden‑dim layer‑norm, out‑gate, and linear projection
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) -> [B, N, N, dim]

    return out
```
