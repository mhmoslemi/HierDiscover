def trimul_kernel(left, right):
    B,N,N,H = left.shape
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left) # shape B,N,N,H
    # Launch kernel
    grid = (B * H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_j=right.stride(1), # j dimension corresponds to second dimension
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    stride_right_b, stride_right_j, stride_right_k, stride_right_h,

    mask_j = j_idx < N

"""
TriMul (outgoing) forward implementation.

The core of TriMul is a batched matrix multiplication over the pairwise
dimension `k`:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

For each head `h` (the hidden dimension) this is exactly a GEMM
`left_h @ right_h.T`.  A custom Triton kernel fuses the batch and head
dimensions so that each program instance computes a tile of the output
matrix for a single `(b, h)` pair.  The kernel works in fp16 for the
loads (half‑precision) while accumulating in fp32 to retain accuracy.
All other operations (LayerNorm, projection, gating, final linear) are
performed with PyTorch, but the heavy O(N³·H) multiplication is off‑loaded
to the Triton kernel.

The kernel is written for Triton 3.3.1 and targets an H100 GPU.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# --------------------------------------------------------------
# Triton kernel: tiled batched GEMM (left @ rightᵀ) over (i, j) tiles.
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where the loops over i/j are tiled (BLOCK_M × BLOCK_N) and the
    reduction over k is tiled with BLOCK_K.
    """
    # 3‑D launch: pid0 = batch*head, pid1 = tile_i, pid2 = tile_j
    pid_batch = tl.program_id(0)   # combined (b, h) index
    pid_m = tl.program_id(1)       # i‑tile index
    pid_n = tl.program_id(2)       # j‑tile index

    # decode batch and head from the combined index
    b = pid_batch // H
    h = pid_batch % H

    # starting indices of the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # 1‑D range objects
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # absolute coordinates
    i_idx = i_start + i          # (BLOCK_M,)
    j_idx = j_start + j          # (BLOCK_N,)

    # predicates for out‑of‑bounds rows / columns
    mask_i = i_idx < N
    mask_j = j_idx < N

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in BLOCK_K sized chunks.
    # ------------------------------------------------------------------
    num_k_iters = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_iters):
        k_start = kk * BLOCK_K
        k_idx = k_start + k                         # (BLOCK_K,)

        mask_k = k_idx < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_offset = i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_base + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)

        # right[b, j, k, h] needs to be transposed (k, j) for dot
        right_offset = k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j
        right = tl.load(right_base + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0, dtype=tl.float16)

        # Accumulate the matrix product of the two tiles
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the tile back to memory (FP32 output)
    # ------------------------------------------------------------------
    out_offset = i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_base + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python wrapper that prepares arguments and launches the kernel.
# ----------------------------------------------------------------------
def _trimul_kernel_fp16(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    Execute the Triton kernel on half‑precision tensors.
    Input/Output shape: (B, N, N, H)
    Returns:
        out (torch.float32) of shape (B, N, N, H)
    """
    assert left_fp16.is_contiguous() and right_fp16.is_contiguous()
    B, N, N2, H = left_fp16.shape
    assert N == N2
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # tile sizes – empirically good for H100 and the tested problem sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid: (batch*head, n_tiles_i, n_tiles_j)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left_fp16,
        right_ptr=right_fp16,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=left_fp16.stride(0),
        stride_left_i=left_fp16.stride(1),
        stride_left_k=left_fp16.stride(2),
        stride_left_h=left_fp16.stride(3),
        stride_right_b=right_fp16.stride(0),
        stride_right_j=right_fp16.stride(1),
        stride_right_k=right_fp16.stride(2),
        stride_right_h=right_fp16.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict mapping weight names to torch.Tensor
        - config       : dict with keys
              * dim        – input feature dimension
              * hidden_dim – hidden dimension after projection
              * nomask     – bool, if True ignore the mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (shape [B, N, N, 1])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # broadcast over hidden_dim
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (Linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy O(N³·H) kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Core multiplication via Triton kernel
    # ------------------------------------------------------------------
    out_fp32 = _trimul_kernel_fp16(left_fp16, right_fp16)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, out‑gate, final Linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    out = F.layer_norm(out_fp32,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                       # broadcast over hidden_dim
    out = F.linear(out, to_out_weight)        # back to dim

    return out
