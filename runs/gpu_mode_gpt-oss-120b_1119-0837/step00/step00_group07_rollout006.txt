@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_i, stride_j, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute global indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    h = pid_h

    # compute offsets ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this hidden channel
    left_ptr_h = left_ptr + h * stride_h
    right_ptr_h = right_ptr + h * stride_h
    out_ptr_h = out_ptr + h * stride_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M,K)
        a_ptrs = left_ptr_h + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (K,N) after transposition
        b_ptrs = right_ptr_h + offs_k[:, None] * stride_j + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product
        acc += tl.dot(a, b)

    # Store
    c_ptrs = out_ptr_h + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the 'outgoing' TriMul operator.
    This kernel performs a batch-wise tensor contraction:
        out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where left and right are projected and gated versions of the
    input tensor.
    The heavy contraction is offloaded to a custom Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections -----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [bs, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Mask handling -----
    # mask shape: [bs, N, N]; may be boolean or float
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # shape [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gates -----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------ Contraction using Triton ------
    out = trimul_contract(left, right)  # shape [bs, N, N, hidden]

    # ----- Output LayerNorm, gating and final linear -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_weight)   # shape [bs, N, N, dim]

    return out

"""
TriMul “outgoing” operator – Triton accelerated contraction.

The heavy operation of the TriMul module is a three‑tensor contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
where:
* `left` and `right` are the linearly projected and gated versions of the
  input tensor (shape [B, N, N, H]).
* The sum is taken over the middle sequence dimension `k` (size N).

The implementation below follows the reference PyTorch module, but the
N³ contraction is off‑loaded to a custom Triton kernel that performs a
batched GEMM for every hidden channel `h`.  The kernel tiles the `i`
and `j` dimensions (BLOCK_M×BLOCK_N) and iterates over `k` in blocks
(BLOCK_K).  Each program instance (identified by `program_id(2)`) works
on a single hidden channel, yielding a natural mapping to the hidden
dimension without extra synchronization.

All other operations (LayerNorm, Linear projections, gates) are kept
in PyTorch for simplicity; only the contraction is Triton‑fused.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    N,                 # int32   sequence length
    H,                 # int32   hidden dimension
    stride_i,          # int32   stride of the first N dimension
    stride_j,          # int32   stride of the second N dimension
    stride_h,          # int32   stride of the hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM over the middle dimension of the pairwise tensor.

    Each kernel instance computes the result for a single hidden channel `h`.
    The algorithm is:
        for i in block_i:
            for j in block_j:
                acc = 0
                for k in 0..N:
                    acc += left[i,k,h] * right[j,k,h]
                out[i,j,h] = acc
    """
    pid_i = tl.program_id(0)   # block row index (i)
    pid_j = tl.program_id(1)   # block col index (j)
    pid_h = tl.program_id(2)   # hidden channel index (h)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    h = pid_h

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers for the selected hidden channel (h)
    # ------------------------------------------------------------------
    left_h_ptr  = left_ptr  + h * stride_h
    right_h_ptr = right_ptr + h * stride_h
    out_h_ptr   = out_ptr   + h * stride_h

    # ------------------------------------------------------------------
    # Accumulator for the (BLOCK_M, BLOCK_N) output tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: left[i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_h_ptr + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block (transposed): right[j, k, h] as (k, j) -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_h_ptr + offs_k[:, None] * stride_j + offs_j[None, :] * stride_i
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # GEMM update
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result back to global memory
    # ------------------------------------------------------------------
    c_ptrs = out_h_ptr + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def trimul_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the contraction out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  `left` and `right` must have shape
    [B, N, N, H] and be contiguous on the CUDA device.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    device = left.device

    # Output tensor (same layout, float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides (in number of elements) for the contiguous layout
    stride_b, stride_i, stride_j, stride_h = left.stride()

    # Tiling parameters – tuned for H100 fp32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch one kernel per batch element (B ≤ 2 in the spec)
    for b in range(B):
        # Base pointers for this batch
        left_ptr  = left.data_ptr()  + b * stride_b
        right_ptr = right.data_ptr() + b * stride_b
        out_ptr   = out.data_ptr()   + b * stride_b

        # Grid: (i‑tiles, j‑tiles, hidden‑channels)
        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            H,
        )

        trimul_outgoing_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            H,
            stride_i,
            stride_j,
            stride_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=4,        # works well on H100
        )

    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – Tensor [B, N, N, dim] (float32, CUDA)
        mask    – Tensor [B, N, N] (float32 or bool) or None
        weights – dict of weight tensors matching the reference module
        config  – dict with keys ``dim`` and ``hidden_dim``.
    Returns
    -------
    Tensor with shape [B, N, N, dim].
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after a linear)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate   # element‑wise gating
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Contraction via Triton (the O(N³) part)
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    contracted = trimul_contract(left, right)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear (hidden → dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    contracted = F.layer_norm(contracted, (hidden_dim,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias)

    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_weight)   # back to `dim`

    return out
