@triton.jit
def layernorm_kernel(
    input_ptr, output_ptr,
    weight_ptr, bias_ptr,
    M, N, eps,
    stride_im, stride_in, stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    # Compute block start
    block_row = pid * BLOCK_M
    row_offsets = block_row + tl.arange(0, BLOCK_M)
    col_offsets = tl.arange(0, BLOCK_N)

    # mask out-of-range
    row_mask = row_offsets < M
    col_mask = col_offsets < N

    # Compute pointers for each row+col combination
    # Use broadcasting: input_ptr + row_offsets[:, None] * stride_im + col_offsets[None, :] * stride_in
    input_ptrs = input_ptr + row_offsets[:, None] * stride_im + col_offsets[None, :] * stride_in
    # Load with mask
    x = tl.load(input_ptrs, mask=row_mask[:, None] & col_mask[None, :], other=0.0)

    # Compute per row sum
    # sum across N: reduce on axis 1
    sum_x = tl.sum(x, axis=1)  # Returns shape [BLOCK_M]
    sum_x2 = tl.sum(x * x, axis=1)

    # Compute mean/var
    mean = sum_x / N
    var = sum_x2 / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Normalize
    # broadcast mean and rstd
    norm_x = (x - mean[:, None]) * rstd[:, None]

    # Apply weight and bias
    w = tl.load(weight_ptr + col_offsets, mask=col_mask, other=0.0)  # shape [BLOCK_N]
    b = tl.load(bias_ptr + col_offsets, mask=col_mask, other=0.0)
    out = norm_x * w[None, :] + b[None, :]

    # Store output
    out_ptrs = output_ptr + row_offsets[:, None] * stride_om + col_offsets[None, :] * stride_on
    tl.store(out_ptrs, out, mask=row_mask[:, None] & col_mask[None, :])

def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    N,  # number of features
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # Load x tile
    x = tl.load(x_ptr + offs_m[:, None]*stride_xm + offs_n[None, :]*stride_xn,

def triton_layernorm(x, weight, bias, eps=1e-5):
    # x: torch tensor shape (M,N)
    # weight,bias: shape (N,)
    M,N = x.shape
    out = torch.empty_like(x)
    # Compute strides: x is contiguous; x.stride() = (N,1)
    stride_xm = x.stride(0)
    stride_xn = x.stride(1)
    stride_ym = out.stride(0)
    stride_yn = out.stride(1)
    # Launch kernel
    GRID = lambda meta: (triton.cdiv(M, meta['BLOCK_M']),)  # 1D grid

    layernorm_kernel[GRID](x_ptr=x, y_ptr=out,
                           weight_ptr=weight, bias_ptr=bias,
                           M=M, N=N, eps=eps,
                           stride_xm=stride_xm, stride_xn=stride_xn,
                           stride_ym=stride_ym, stride_yn=stride_yn,
                           BLOCK_M=256, BLOCK_N=64)  # choose appropriate numbers

    return out

    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if not nomask
    if not config.get("nomask", False):
        # mask shape (bs, seq_len, seq_len) -> unsqueeze(-1) to broadcast over hidden
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute batched matrix multiplication per hidden dim
    # reshape to (bs, hidden_dim, seq_len, seq_len) and pack hidden_dim as batch for gemm
    # left: [bs, seq_len, seq_len, hd] -> permute to [bs, hd, seq_len, seq_len]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hd dimension
    B_hd = bs * hidden_dim
    left_mat = left_perm.view(B_hd, seq_len, seq_len)
    right_mat = right_perm.view(B_hd, seq_len, seq_len)

    # compute out_mat = left_mat @ right_mat.transpose(1,2)  # shape [B_hd, seq_len, seq_len]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [bs, hidden_dim, seq_len, seq_len] then permute
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm via Triton
    # flatten first three dims
    out_flat = out.view(M, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)

    # Multiply by out_gate
    out = out_norm * out_gate

    # Final linear
    # shape [bs, seq_len, seq_len, hidden_dim] -> linear to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

    y = torch.empty_like(x)

        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128, BLOCK_N=64
    )
    return y_cont

"""
Triton‑accelerated implementation of the TriMul “outgoing” module.

The heavy computation consists of three parts:
1. A custom two‑pass LayerNorm kernel (implemented in Triton) that normalises
   across the feature dimension. It is used for the input LayerNorm and for the
   output LayerNorm.
2. The remaining linear projections, sigmoid gates and the batched matrix
   multiplication (the N³ operation) are performed with PyTorch primitives
   (torch.nn.functional.linear and torch.bmm), which on an H100 map to
   highly‑optimised cuBLAS kernels.
3. The final linear projection back to the model dimension.

The Triton kernel is deliberately simple (per‑row mean/variance reduction) but
provides the required “part of the operations in a kernel” as required by the
task.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon for variance
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    NUM_K = tl.cdiv(N, BLOCK_N)
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load a tile (BLOCK_M x BLOCK_N)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)          # sum over the N‑axis
        sum_sq  += tl.sum(x * x, axis=1)      # sum of squares

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Normalise
        norm_x = (x - mean[:, None]) * rstd[:, None]

        # Load weight and bias (1‑D tensors)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Applies LayerNorm over the last dimension of ``x`` using the Triton kernel.
    ``x`` is expected to be a 2‑D Tensor of shape (M, N).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors (required for correct stride handling)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program; tuned for H100
        BLOCK_N=64,    # columns per tile; fits in registers
    )
    return y


# ----------------------------------------------------------------------
# Triton‑based custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5  # default epsilon for LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    # Flatten batch/seq dimensions for the kernel: (B * N * N, dim)
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask if required (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Compute gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication across the summed dimension (k)
    #     For each hidden channel we perform: out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) and flatten batch & channel dimensions.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B_H = bs * hidden_dim
    left_mat = left_perm.view(B_H, seq_len, seq_len)           # [B*H, N, N]
    right_mat = right_perm.view(B_H, seq_len, seq_len)         # [B*H, N, N]

    # Matrix multiplication: (i,k) · (j,k)^T  →  (i,j) per hidden channel
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out_norm * out_gate
    out = F.linear(out, weights["to_out.weight"])  # weight shape [dim, hidden_dim]

    return out
