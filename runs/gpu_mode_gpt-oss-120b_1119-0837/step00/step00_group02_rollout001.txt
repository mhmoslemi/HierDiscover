    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    # Let's compute "pid" from axis=0 (one-dimensional grid) spanning B*H*ceil(N/BLOCK_M)*ceil(N/BLOCK_N).
    pid = tl.program_id(0)
    num_bh = B * H
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_bh = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ensure within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute b, h
    b = pid_bh // H
    h = pid_bh % H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        mask_k = offs_k < N

        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None]*stride_left_i + offs_k[None, :]*stride_left_j
        # left dims: i (rows) then k (cols)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # if half, cast as needed

        acc += tl.dot(left, right_T)  # shape (BLOCK_M, BLOCK_N)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_j,
    stride_r_b, stride_r_h, stride_r_i, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute block coordinates
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    bh = pid // (num_pid_m * num_pid_n)   # index over batch*head
    pid_rem = pid - bh * (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # offsets for dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # decode batch and head from combined index
    b = bh // H
    h = bh % H

    # offset pointers for the blocks
    left_ptrs = left_ptr + \
        b * stride_l_b + h * stride_l_h + \
        offs_m[:, None] * stride_l_i + \
        tl.arange(0, BLOCK_K)[None, :] * stride_l_j

    right_ptrs = right_ptr + \
        b * stride_r_b + h * stride_r_h + \
        offs_n[:, None] * stride_r_i + \
        tl.arange(0, BLOCK_K)[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # actual offsets for this k-block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right block: (N, K) we need (K, N) for dot
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # transpose right to (K, N)
        right_T = tl.trans(right, (1, 0))

        # promote to fp32 for accumulation
        acc += tl.dot(left.to(tl.float32), right_T.to(tl.float32))

        # advance pointers for next k block
        left_ptrs += BLOCK_K * stride_l_j
        right_ptrs += BLOCK_K * stride_r_j

    # write result
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_m[:, None] * stride_o_i + \
        offs_n[None, :] * stride_o_j

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    operations are carried out with PyTorch.

    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]

    to_out_weight = weights["to_out.weight"]

"""
TriMul (outgoing) – Triton implementation
------------------------------------------------
Given an input tensor x of shape [B, N, N, C] this kernel computes the
“outgoing” triangle multiplicative update described in the AlphaFold3 paper:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where ‘left’ and ‘right’ are linear projections of the normalized input,
optionally masked and gated.  The heavy O(B·D·N³) part (the sum over k) is
implemented with a custom Triton kernel that performs a batched matrix‑multiply
for every hidden dimension d (D = hidden_dim).  All other operations (LayerNorm,
Linear, sigmoid gating) are performed with regular PyTorch for simplicity and
numerical stability.

The kernel works for the full range of shapes used in the test matrix
( B∈{1,2}, N∈{128,256,512,1024}, C∈{128,384}, D=hidden_dim∈{128} ).
It operates on half‑precision inputs (to reduce memory traffic) while
accumulating in single‑precision for accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # const __half*   [B, D, N, N]
    right_ptr,               # const __half*   [B, D, N, N]
    out_ptr,                 #       float*    [B, D, N, N]
    B, D, N,                 # int32 scalars
    stride_l_b, stride_l_d, stride_l_i, stride_l_j,
    stride_r_b, stride_r_d, stride_r_i, stride_r_j,
    stride_o_b, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,   # rows of the output block
    BLOCK_N: tl.constexpr,   # cols of the output block
    BLOCK_K: tl.constexpr,   # reduction size per iteration
):
    """Batched GEMM: out[b,d,i,j] = left[b,d,i,:] @ right[b,d,j,:]^T"""
    pid = tl.program_id(0)

    # -------------------------------------------------------------
    # 1) Decode the 4‑D grid (b, d, i_block, j_block)
    # -------------------------------------------------------------
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)

    # (b·d) index
    bd = pid // (num_i_blocks * num_j_blocks)
    pid_rem = pid - bd * (num_i_blocks * num_j_blocks)
    i_block = pid_rem // num_j_blocks
    j_block = pid_rem % num_j_blocks

    b = bd // D
    d = bd % D

    # -------------------------------------------------------------
    # 2) Compute the absolute offsets for the current block
    # -------------------------------------------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the leading dimensions (b, d)
    base_left = left_ptr + b * stride_l_b + d * stride_l_d
    base_right = right_ptr + b * stride_r_b + d * stride_r_d

    # -------------------------------------------------------------
    # 3) Accumulate over the reduction dimension k
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]   shape (M, K)
        left_ptrs = base_left \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_j
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right[j, k]   shape (N, K) -> we need (K, N) for the dot
        right_ptrs = base_right \
            + offs_j[:, None] * stride_r_i \
            + offs_k[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # transpose the right block to (K, N)
        right_T = tl.trans(right, (1, 0))

        # accumulate in fp32 (tensor‑core friendly)
        acc += tl.dot(left.to(tl.float32), right_T.to(tl.float32))

    # -------------------------------------------------------------
    # 4) Write the result
    # -------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_o_b + d * stride_o_d \
        + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            - mask: torch.Tensor, shape [B, N, N]   (float32 or bool)
            - weights: dict of torch.Tensors containing all linear / LN weights
            - config: dict with keys:
                * dim          – input channel dimension C
                * hidden_dim   – hidden dimension D
                * nomask (bool) – whether mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the input (per‑channel)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    #   left  = x @ left_proj_weight.T
    #   right = x @ right_proj_weight.T
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, D]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask already on device, shape [B,N,N])
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # Broadcast mask over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy bilinear update – Triton kernel
    # ------------------------------------------------------------------
    #   Permute to [B, D, N, N] for a clean batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to half‑precision to reduce memory traffic
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    B, D, N, _ = left_fp16.shape
    out_perm = torch.empty_like(left_fp16, dtype=torch.float32)  # [B, D, N, N]

    # stride information (in number of elements, not bytes)
    stride_l_b, stride_l_d, stride_l_i, stride_l_j = left_fp16.stride()
    stride_r_b, stride_r_d, stride_r_i, stride_r_j = right_fp16.stride()
    stride_o_b, stride_o_d, stride_o_i, stride_o_j = out_perm.stride()

    # Tune block sizes for H100 – they work well for all N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * D * num_i_blocks * num_j_blocks

    grid = (total_blocks,)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_perm,
        B, D, N,
        stride_l_b, stride_l_d, stride_l_i, stride_l_j,
        stride_r_b, stride_r_d, stride_r_i, stride_r_j,
        stride_o_b, stride_o_d, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # enough warps to occupy an SM
    )

    # Back to the original layout [B, N, N, D]
    out = out_perm.permute(0, 2, 3, 1)  # (B,N,N,D)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 7) Gated output
    # ------------------------------------------------------------------
    out = out * out_gate  # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 8) Final linear projection back to input dim (no bias)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, C]

    return out
