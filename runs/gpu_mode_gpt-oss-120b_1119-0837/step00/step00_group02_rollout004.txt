```python
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_size, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_m = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(M, BLOCK_M)
    block_n = pid % tl.cdiv(N, BLOCK_N)

    # Compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers for each batch
    a_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = B_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    for k in range(0, K, BLOCK_K):
        # load blocks
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # multiply-accumulate
        acc += tl.dot(a, b)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write output
    c_ptrs = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

"""
Optimized Triton implementation of the TriMul “outgoing” update.
- LayerNorm, linear projections, gating and final Linear are kept in
  PyTorch (float32) for numerical stability.
- The heavy O(N^3) term `einsum('... i k d, ... j k d -> ... i j d')`
  is rewritten as a batched matrix‑multiply:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  which is `left @ right.T` for each channel `d`.
- The batched matmul is executed by a custom Triton kernel that
  fuses the K‑loop and accumulates in float32 while loading the
  operands in float16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr,                     # *float16
    B_ptr,                     # *float16
    C_ptr,                     # *float16
    batch, M, N, K,            # int (constexpr)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B^T for a batch of matrices.
    A : (batch, M, K)
    B : (batch, N, K)   (stored as (batch, N, K); we index it as (K, N)
    C : (batch, M, N)
    The kernel iterates over K in blocks, loads tiles of A and B,
    performs a dot product and accumulates in float32.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # grid decomposition: each program handles one (batch, block_m, block_n)
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch
    pid = pid % blocks_per_batch
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # offsets inside the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = A_ptr + batch_id * stride_a_batch
    b_batch_ptr = B_ptr + batch_id * stride_b_batch
    c_batch_ptr = C_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the K dimension (shared dimension)
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # B tile (need B^T -> we load as (BLOCK_K, BLOCK_N))
        # B is stored as (batch, N, K); we want B_T[k, n]
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # dot product accumulates in float32
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D], float32

    # --------------------------------------------------------------
    # 2. Linear projections (float32)
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])        # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Optional mask (float32)
    # --------------------------------------------------------------
    if not nomask:
        # ensure mask is float32 and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (float32)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy O(N³) contracted product via Triton
    # --------------------------------------------------------------

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_flat = right_perm.view(bh, N, N).contiguous()

    # convert to float16 for the kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Kernel launch configuration
    # ------------------------------------------------------------------
    # Block sizes – tuned for typical N up to 1024 on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

        right_fp16,
        out_flat,
        bh, N, N, N,                      # batch, M, N, K

        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # cast result back to float32
    out = out_flat.to(torch.float32)

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, D]

    return out

"""
Optimized Triton implementation of the TriMul “outgoing” update.

Algorithm
---------
1. **LayerNorm** and the four linear projections (left, right, three gates) are
   performed in PyTorch (float32) for numerical stability.
2. The heavy O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is rewritten as a batched matrix multiplication:
       out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
   for each channel `d`.  The batched matmul is executed by a custom Triton
   kernel that:
   * loads tiles of A (left) and B (right transposed) in float16,
   * accumulates the dot product in float32,
   * writes the result back in float16.
3. The result is cast back to float32, reshaped, and passed through the
   output LayerNorm, multiplied by the output gate, and finally projected
   back to the original feature dimension with a linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr,                     # *float16  (batch, M, K)
    B_ptr,                     # *float16  (batch, N, K)   (we read as Bᵀ)
    C_ptr,                     # *float16  (batch, M, N)
    batch, M, N, K,            # int scalar arguments
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : (batch, M, K)            float16
    B : (batch, N, K)            float16   (treated as transposed)
    C : (batch, M, N)            float16   (result)
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decompose the 1‑D program space: (batch, block_m, block_n)
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    block_m = pid_in_batch // num_blocks_n
    block_n = pid_in_batch % num_blocks_n

    # Offsets within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = A_ptr + batch_id * stride_a_batch
    b_batch_ptr = B_ptr + batch_id * stride_b_batch
    c_batch_ptr = C_ptr + batch_id * stride_c_batch

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the shared dimension K
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # B is stored as (batch, N, K); we access it as (K, N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # Multiply‑accumulate (accumulator stays in float32)
        acc += tl.dot(a, b)

    # Write the tile back to C (still in float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, D] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least:
                "dim" (D), "hidden_dim" (H), optional "nomask" (bool), "eps" (float)
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (float32)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (float32)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction via Triton (batched matmul)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) and make contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & channel dimensions for the kernel
    bh = B * hidden_dim                               # total "batch" for kernel
    left_flat  = left_perm.view(bh, N, N).contiguous()
    right_flat = right_perm.view(bh, N, N).contiguous()

    # Convert to float16 for the kernel (the kernel accumulates in float32)
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer (float16)
    out_flat = torch.empty_like(left_fp16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = bh * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        bh, N, N, N,                               # batch, M, N, K
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to float32 and reshape to original layout
    out = out_flat.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, D]

    return out
