"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections → left/right tensors (shape [B,N,N,hd]).
3. Optional pair mask (broadcasted on the feature axis) applied to left/right.
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and apply left/right gates element‑wise.
5. The core O(N³·hd) operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication of shape
      (B·hd, N, N)  ×  (B·hd, N, N)ᵀ .
   This is implemented with a Triton kernel that processes a block of rows
   (i) and columns (j) while accumulating over K‑tiles.
6. Layer‑norm the result over the hidden dimension, multiply by out_gate,
   and apply a final linear projection back to the original feature size.

The Triton kernel handles arbitrary batch size and hidden dimension by
flattening them into a single “batch” axis (B·hd).  All arithmetic is
performed in float32 for numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul A (M×K) @ B (N×K)ᵀ  →  C (M×N)
#   A shape : (batch, M, K)
#   B shape : (batch, N, K)   (K‑dimension is the last axis)
#   C shape : (batch, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_az, stride_am, stride_ak,      # A: (batch, M, K)
    stride_bz, stride_bk, stride_bn,      # B: (batch, N, K)  (K is last dim)
    stride_cz, stride_cm, stride_cn,      # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)   # batch index (B·hd)
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # column tile

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator for the C‑tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over K‑dimension in tiles
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A‑tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp32 load

        # B‑tile: (BLOCK_K, BLOCK_N)
        # Note: B layout is (batch, N, K); we load it as (K, N) to get the
        # transposed view required for the dot‑product.
        b_ptrs = (
            b_ptr
            + pid_z * stride_bz
            + cur_k[:, None] * stride_bk   # K‑stride (inner)
            + offs_n[None, :] * stride_bn   # N‑stride (outer)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Dot‑product over the K‑axis
        acc += tl.dot(a, b)

    # --------------------------------------------------------------- #
    # Write the result tile to C
    # --------------------------------------------------------------- #
    c_ptrs = (
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Public entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Compute the forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of model parameters (see code)
        - config       : dict containing at least "dim", "hidden_dim" and
                         optional "nomask" flag

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # Shortcut to weight tensors
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1) Input layer‑norm
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        (input_tensor.shape[-1],),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hd]
    right = F.linear(x_norm, right_proj_weight)  # [B,N,N,hd]

    # ------------------------------------------------------------------- #
    # 3) Gates (sigmoid) – they operate on the *normalized* input
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------- #
    # 4) Optional pair mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask shape: [B,N,N] → [B,N,N,1] for broadcasting
        mask_bc = mask.unsqueeze(-1).to(dtype)
        left = left * mask_bc
        right = right * mask_bc

    # ------------------------------------------------------------------- #
    # 5) Apply per‑pair gates
    # ------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 6) Core O(N³·hd) operation via Triton
    #    Reshape (B,N,N,hd) → (B·hd, N, N) and perform batched matmul.
    # ------------------------------------------------------------------- #
    B, N, _, _ = input_tensor.shape

    # Permute to bring the hidden dim into the batch axis and make it
    # contiguous for the view.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,hd,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,hd,N,N]

    batch_total = B * hidden_dim
    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)

    # Allocate output tensor
    out_flat = torch.empty((batch_total, N, N), dtype=dtype, device=device)

    # Block sizes – chosen to give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          batch_total ),
    )

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_total,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),   # stride for K dimension in B (last dim)
        right_flat.stride(1),   # stride for N dimension in B
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B,N,N,hd]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 7) Output layer‑norm + out‑gate
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8) Final linear projection to the original feature dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
