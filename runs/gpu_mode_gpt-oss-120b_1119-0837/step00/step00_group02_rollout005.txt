                   N, K,
                   stride_aa, stride_ai, stride_ak,
                   stride_bb, stride_bi, stride_bk,
                   stride_cc, stride_ci, stride_cj,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # pointer base for this batch
    a_batch_ptr = a_ptr + pid_b * stride_aa
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cc

"""
TriMul (outgoing) – Triton implementation of the heavy N³ reduction.

The forward pass consists of:
1. Layer‑norm on the input.
2. Two linear projections (left/right) and three element‑wise gates.
3. A masked rank‑1 update over the third axis:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix multiply where each (batch, d) pair
   computes A @ Bᵀ with A,B ∈ ℝ^{N×N}.  The kernel fuses the K‑loop,
   loads a tile of A (i‑rows) and a transposed tile of B (k‑rows, j‑cols),
   performs a dot‑product with float‑32 accumulation and writes the result.
4. Layer‑norm on the hidden dimension, gating and a final linear projection.

All heavy work (the N³ reduction) lives inside a Triton kernel; the
remaining linear / norm / sigmoid steps use PyTorch which already offers
high‑performance kernels on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        - input     : [B, N, N, dim]   (float32)
        - mask      : [B, N, N]        (bool/float, may be all‑ones)
        - weights   : dict of model parameters
        - config    : {"dim": int, "hidden_dim": int, ...}
    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1. unpack & basic shapes
    # ------------------------------------------------------------------
    (x, mask, weights, config) = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 for the N³ kernel (fp32 accumulation inside)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Prepare tensors for the batched GEMM kernel
    #    we treat every pair (batch, hidden_dim) as an independent matrix.
    # ------------------------------------------------------------------
    # bring hidden_dim in front: [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch & hidden_dim → [B*hidden_dim, N, N]
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)  # fp16

    # strides for the flattened tensors (contiguous layout)
    stride_a_batch = left_flat.stride(0)   # N * N
    stride_a_row = left_flat.stride(1)    # N
    stride_a_col = left_flat.stride(2)    # 1

    stride_b_batch = right_flat.stride(0)
    stride_b_row = right_flat.stride(1)
    stride_b_col = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_row = out_flat.stride(1)
    stride_c_col = out_flat.stride(2)

    # ------------------------------------------------------------------
    # 8. Triton kernel – batched matmul A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 64   # i‑tile
    BLOCK_N = 64   # j‑tile
    BLOCK_K = 64   # k‑tile

    @triton.jit
    def tri_mul_kernel(
        a_ptr, b_ptr, c_ptr,
        N, K,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid_m = tl.program_id(0)          # tile over i
        pid_n = tl.program_id(1)          # tile over j
        pid_b = tl.program_id(2)          # batch * hidden_dim index

        # tile start indices
        i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_i = i < N
        mask_j = j < N

        # base pointers for this (b,d) slice
        a_base = a_ptr + pid_b * stride_a_batch
        b_base = b_ptr + pid_b * stride_b_batch
        c_base = c_ptr + pid_b * stride_c_batch

        # accumulator in float32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # number of K‑tiles
        num_k = tl.cdiv(K, BLOCK_K)

        for k in range(0, num_k):
            k_start = k * BLOCK_K
            cur_k = tl.arange(0, BLOCK_K) + k_start

            # ---- Load A tile (i‑rows, k‑cols) ----
            a_ptrs = a_base + i[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
            a = tl.load(
                a_ptrs,
                mask=mask_i[:, None] & (cur_k[None, :] < K),
                other=0.0,
                dtype=tl.float16,
            )
            a = a.to(tl.float32)   # promote for accumulation

            # ---- Load Bᵀ tile (k‑rows, j‑cols) ----
            # note: we treat B directly but index it as (k, j) → transpose on‑the‑fly
            b_ptrs = b_base + cur_k[:, None] * stride_b_row + j[None, :] * stride_b_col
            b = tl.load(
                b_ptrs,
                mask=(cur_k[:, None] < K) & mask_j[None, :],
                other=0.0,
                dtype=tl.float16,
            )
            b = b.to(tl.float32)

            # ---- Rank‑1 update (outer product) for this k‑tile ----
            acc += tl.dot(a, b)          # (M×K)·(K×N) → (M×N)

        # ---- Store result ----
        c_ptrs = c_base + i[:, None] * stride_c_row + j[None, :] * stride_c_col
        tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # launch grid: (tiles_i, tiles_j, batch*hidden_dim)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [B, N, N, hidden_dim] (float32 for downstream ops)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 10. Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # -> [B, N, N, dim]

    return out
```
