```python
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_k_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)  # block row index across i dimension
    pid_j = tl.program_id(1)  # block col index across j dimension
    pid_hb = tl.program_id(2)  # combined batch-head index (0 .. B*H-1)
    
    # compute which batch and hidden index
    b = pid_hb // H
    h = pid_hb % H

    # Compute offsets for M and N
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # bound check
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offsets for given batch and hidden
    left_base = left_ptr + b * stride_b_left + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base = out_ptr + b * stride_b_out + h * stride_h_out

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K dimension
    for kk in range(0, N, BLOCK_K):
        # compute K offsets
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_i_left) + (offs_k[None, :] * stride_k_left)
        # right transpose: load B_T (BLOCK_K, BLOCK_N)
        # right indices: element = right[b, j, k, h], we want B_T[k, j]
        right_ptrs = right_base + (offs_j[None, :] * stride_i_right) + (offs_k[:, None] * stride_k_right)

        # load with mask
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot
        acc += tl.dot(left_tile, right_tile)

    # apply mask for i and j
    out_ptrs = out_base + (offs_i[:, None] * stride_i_out) + (offs_j[None, :] * stride_k_out)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

```python
def outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,

    pid_hb = tl.program_id(2) # combined batch*head

    b = pid_hb // H
    h = pid_hb % H

    mask_j = offs_j < N

    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base = out_ptr + b * stride_b_out + h * stride_h_out

        offs_k = kk + tl.arange(0, BLOCK_K)
```

```python
def custom_kernel(data):
    """
    Triton accelerated outgoing TriMul kernel (AlphaFold3).
    Implements the forward pass only.
    Steps:
      1) LayerNorm on input.
      2) Linear projections (left/right) and gates.
      3) Apply mask (if any) and gating.
      4) Compute the pairwise dot-sum using a custom Triton kernel
         (out[i,j,d] = sum_k left[i,k,d] * right[j,k,d]).
      5) LayerNorm on hidden dimension, gate, final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections to hidden dim
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided and not disabled
    if mask is not None and not config.get("nomask", False):
        # ensure same dtype
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Triton matmul for the core $i,j$ summation
    out_hidden = triton_outgoing_matmul(left, right)   # shape [B,N,N,hidden_dim]

    # Post-processing
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out_hidden = out_hidden * out_gate   # broadcast over hidden dim
    out = F.linear(out_hidden, to_out_weight)  # shape [B,N,N,dim]
    return out
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Project the normalized tensor to a hidden dimension with two weight matrices
   (left_proj, right_proj).  Apply an optional mask.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input and multiply the projected tensors with their respective
   gates.
4. Core TriMul operation:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of matrix multiplications between left and the transpose of
   right for each hidden head.  It is implemented in a custom Triton kernel.
5. Apply a second LayerNorm on the hidden tensor, multiply by out_gate and
   linearly project back to the original dimension.

The Triton kernel fuses the reduction over `k` (the sequence dimension) for
each (batch, head) pair and therefore saves a large amount of memory traffic
compared with a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul of shape (B,N,N,H) where
#   out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h]
# --------------------------------------------------------------
@triton.jit
def _outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_k_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)          # block row   (i axis)
    pid_j = tl.program_id(1)          # block col   (j axis)
    pid_hb = tl.program_id(2)         # combined batch * head index

    # decode batch and hidden‑head
    b = pid_hb // H
    h = pid_hb % H

    # i, j coordinates for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the selected batch and hidden head
    left_base  = left_ptr  + b * stride_b_l + h * stride_h_l
    right_base = right_ptr + b * stride_b_r + h * stride_h_r
    out_base   = out_ptr   + b * stride_b_o + h * stride_h_o

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K) -> indices (i, k)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_i_l + \
                    offs_k[None, :] * stride_k_l
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile (transposed): (BLOCK_K, BLOCK_N) -> indices (k, j)
        right_ptrs = right_base + \
                     offs_j[None, :] * stride_i_r + \
                     offs_k[:, None] * stride_k_r
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # accumulate dot product over k
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # write back
    out_ptrs = out_base + \
               offs_i[:, None] * stride_i_o + \
               offs_j[None, :] * stride_k_o
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _triton_outgoing_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.  `left` and `right` must have shape
    [B, N, N, H] and be contiguous on the CUDA device.
    Returns a tensor of the same shape containing the pairwise summed product.
    """
    assert left.is_cuda and right.is_cuda
    B, N, N2, H = left.shape
    assert N == N2, "input must be square in the two sequence dimensions"

    # allocate output
    out = torch.empty_like(left)

    # ensure contiguity (required for correct stride layout)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements, not bytes)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_k_o, stride_h_o = out.stride()

    # tiling parameters – chosen to fit well on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch grid: (i‑tiles, j‑tiles, batch*head)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _outgoing_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_b_l, stride_i_l, stride_k_l, stride_h_l,
        stride_b_r, stride_i_r, stride_k_r, stride_h_r,
        stride_b_o, stride_i_o, stride_k_o, stride_h_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict mapping weight names to torch.Tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- weights ----------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ---------- 1) layer norm ----------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- 2) linear projections ----------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---------- 3) optional mask ----------
    if mask is not None and not nomask:
        # mask is broadcasted over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4) gates ----------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ---------- 5) core TriMul (batched matmul) ----------
    hidden = _triton_outgoing_matmul(left, right)   # [B,N,N,hidden_dim]

    # ---------- 6) post‑processing ----------
    hidden = F.layer_norm(hidden,
                         (hidden_dim,),
                         weight=to_out_norm_weight,
                         bias=to_out_norm_bias)
    hidden = hidden * out_gate            # broadcast over hidden dim
    out = F.linear(hidden, to_out_weight) # final projection back to `dim`

    return out
