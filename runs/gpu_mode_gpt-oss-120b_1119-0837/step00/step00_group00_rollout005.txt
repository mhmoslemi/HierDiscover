def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm (x)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Project and gating
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # Apply mask if not nomask
    if not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # shape B,N,N,1
        left = left * mask_f
        right = right * mask_f
    # Apply gating
    left = left * left_gate
    right = right * right_gate
    # Compute out via Triton
    out = tri_mul_kernel_launcher(left, right, B, N, hidden_dim)
    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Multiply out_gate
    out = out * out_gate
    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

@triton.jit
def tri_mul_kernel_launcher(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # total grid = B * num_i * num_j
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j

    # Compute start offsets
    i_offset = i_block * BLOCK_I
    j_offset = j_block * BLOCK_J

    # offsets vectors
    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # loop over hidden dim blocks
    for h_start in range(0, H, BLOCK_H):
        # accumulator for current hidden block ; shape (BLOCK_I, BLOCK_J, BLOCK_H)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # loop over k dimension tiles
        for k_start in range(0, N, BLOCK_K):
            # compute offsets for K
            cur_k = k_start
            offs_k = cur_k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_H)
            left_ptrs = left_ptr + b * stride_left_b \
                + offs_i[:, None, None] * stride_left_i \
                + offs_k[None, :, None] * stride_left_k \
                + (tl.arange(0, BLOCK_H)[None, None, :] + h_start) * stride_left_h
            left_tile = tl.load(left_ptrs,
                                mask=mask_i[:, None, None] & mask_k[None, :, None] & ((tl.arange(0, BLOCK_H) + h_start) < H)[None, None, :],
                                other=0.0)

            # Load right tile but transposed: shape (BLOCK_K, BLOCK_J, BLOCK_H)
            # We need to compute pointer: stride_right_k for rows (K axis) and stride_right_j for columns (J axis)
            right_ptrs = right_ptr + b * stride_right_b \
                + offs_j[None, :, None] * stride_right_j \
                + offs_k[:, None, None] * stride_right_k \
                + (tl.arange(0, BLOCK_H)[None, None, :] + h_start) * stride_right_h
            # This yields shape (BLOCK_K, BLOCK_J, BLOCK_H) Type: (K, J, H)
            right_tile = tl.load(right_ptrs,
                                 mask=((tl.arange(0, BLOCK_K) + cur_k) < N)[:, None, None] & mask_j[None, :, None] & ((tl.arange(0, BLOCK_H) + h_start) < H)[None, None, :],
                                 other=0.0)

            # Now compute accumulate: for each h in BLOCK_H, do dot product left[..., h] (I,K) x right[...,h] (K,J)
            # Use tl.dot for each h
            for h in range(0, BLOCK_H):
                # early exit if out-of-bound hidden dimension
                if h_start + h >= H:
                    break
                left_mat = left_tile[:, :, h]   # (I, K)
                right_mat = right_tile[:, :, h]  # (K, J)
                # dot: left_mat (I,K) @ right_mat (K,J) -> (I,J)
                # tl.dot expects (M,K) and (K,N). ensure shapes.
                acc[:, :, h] += tl.dot(left_mat, right_mat, out_dtype=tl.float32)
        # Store to out
        out_ptrs = out_ptr + b * stride_out_b \
                 + offs_i[:, None, None] * stride_out_i \
                 + offs_j[None, :, None] * stride_out_j \
                 + (tl.arange(0, BLOCK_H)[None, None, :] + h_start) * stride_out_h
        tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & ((tl.arange(0, BLOCK_H) + h_start) < H)[None, None, :])

    assert left.shape == right.shape

"""
TriMul "outgoing" forward pass (AlphaFold3) with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Project to the hidden dimension `H` with four linear layers
   (left/right projection and three gating layers).  All linear ops use
   `torch.nn.functional.linear` (bias‑free) and the gates are passed
   through a sigmoid.
3. Apply the pairwise mask (if present) to the projected tensors,
   then multiply by their respective gates.
4. Compute the core contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent matrix‑multiplications:
        left_h @ right_hᵀ  (size N×N).
   This O(N³·H) work is performed by a custom Triton kernel that
   tiles the i/j dimensions (64×64), tiles the reduction dimension k
   (32) and processes the hidden channels in blocks of 32.
5. Apply a second layer‑norm on the hidden dimension,
   multiply by the output gate, and finally project back to D
   with another linear layer.
All operations use float32; the Triton kernel accumulates in float32
to preserve precision.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes out = Σ_k left[...,k,h] * right[...,k,h]
# left  : [B, N, N, H]  (i,k,h)
# right : [B, N, N, H]  (j,k,h) – we load it transposed to (k,j,h)
# out   : [B, N, N, H]  (i,j,h)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *const float*
    right_ptr,              # *const float*
    out_ptr,                # *float*
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # number of tiles in i and j dimensions
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    # decode program id to (batch, i_tile, j_tile)
    b = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_tile = pid_rem // num_j
    j_tile = pid_rem % num_j

    # starting offsets for this tile
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over hidden‑channel blocks
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        # (BLOCK_I, BLOCK_J, BLOCK_H) accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the reduction dimension k
        # ------------------------------------------------------------------
        for k_start in range(0, N, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # hidden offsets for this block
            offs_h = h_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # --------------------------------------------------------------
            # Load left tile  : shape (I, K, H)
            # --------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + b * stride_l_b
                + offs_i[:, None, None] * stride_l_i
                + offs_k[None, :, None] * stride_l_k
                + offs_h[None, None, :] * stride_l_h
            )
            left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
            left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # --------------------------------------------------------------
            # Load right tile (transposed) : shape (K, J, H)
            # --------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + b * stride_r_b
                + offs_j[None, :, None] * stride_r_i   # j dimension uses stride_r_i (first N)
                + offs_k[:, None, None] * stride_r_k
                + offs_h[None, None, :] * stride_r_h
            )
            right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
            right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # --------------------------------------------------------------
            # For each hidden channel inside the block perform a small GEMM:
            #   (I, K) @ (K, J) -> (I, J)
            # --------------------------------------------------------------
            for hh in range(0, BLOCK_H):
                # guard against over‑run of the hidden dimension
                if h_start + hh >= H:
                    break
                left_mat  = left_tile[:, :, hh]   # (I, K)
                right_mat = right_tile[:, :, hh]  # (K, J)
                acc[:, :, hh] += tl.dot(left_mat, right_mat, out_dtype=tl.float32)

        # ------------------------------------------------------------------
        # Store the result block back to memory
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_o_b
            + offs_i[:, None, None] * stride_o_i
            + offs_j[None, :, None] * stride_o_j
            + offs_h[None, None, :] * stride_o_h
        )
        out_mask = mask_i[:, None, None] & mask_j[None, :, None] & (offs_h < H)[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.  `left` and `right` must have
    shape [B, N, N, H] and be contiguous float32 tensors.
    """
    assert left.shape == right.shape, "left/right must have identical shapes"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # compute strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tiling parameters (chosen for H100 Tensor Core friendly sizes)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_H = 32

    # one program per (batch, i_tile, j_tile)
    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    total_programs = B * grid_i * grid_j

    tri_mul_kernel[(total_programs,)](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,          # experimentally good for the above block sizes
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the forward pass of the TriMul "outgoing" module.
    `data` is a tuple (input_tensor, mask, weights, config) where:
        - input_tensor : torch.Tensor of shape [B, N, N, D]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of model parameters
        - config       : dict containing at least "dim" and "hidden_dim"
    Returns:
        output : torch.Tensor of shape [B, N, N, D]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Layer‑norm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (unsqueeze to broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction using Triton (produces [B,N,N,hidden_dim])
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    out = tri_mul_triton(left, right)   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Second Layer‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # Multiply by the output gate and final linear projection back to D
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
