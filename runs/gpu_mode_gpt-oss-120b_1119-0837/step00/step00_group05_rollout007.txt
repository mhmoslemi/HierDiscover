import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    using tiled matrix multiplication per hidden channel d.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decompose batch index into batch and hidden index
    b = pid_batch // H
    d = pid_batch % H

    # Offsets within N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_b_left + d * stride_h_left + \
            offs_m[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # Load right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_b_right + d * stride_h_right + \
            offs_k[:, None] * stride_k_right + offs_n[None, :] * stride_i_right
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float16)

        # Compute partial product and accumulate in FP32
        acc += tl.dot(left, right).to(tl.float32)

    # Write back
    out_ptrs = out_ptr + \
        b * stride_b_out + d * stride_h_out + \
        offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    \"\"\"
    Triton-accelerated implementation of the TriMul "outgoing" update.
    The heavy pairwise reduction (einsum ``left[..., i, k, d] * right[..., j, k, d]``)
    is performed by a custom Triton kernel; all surrounding LayerNorms,
    linear projections, and gating are handled in PyTorch (with half‑precision
    where possible).
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask:        torch.Tensor [B, N, N] (ignored when config['nomask'] is True)
            weights:     dict of model parameters
            config:      dict with keys ``dim`` and ``hidden_dim`` and ``nomask``
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    \"\"\"
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------ 1. LayerNorm ------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to half for the bulk of the work
    x_h = x_norm.to(torch.float16)

    # ------------------------ 2. Linear projections ------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # ------------------------ 3. Optional mask ------------------------
    if not nomask:
        # mask is 0/1; broadcast to hidden dimension.
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------ 4. Gating ------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------ 5. Triton kernel: core reduction ------------------------
    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty(
        (bs, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Block sizes – chosen to fit typical H100 warp configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides (in elements, not bytes)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_b_left,
        stride_i_left,
        stride_k_left,
        stride_h_left,
        stride_b_right,
        stride_i_right,
        stride_k_right,
        stride_h_right,
        stride_b_out,
        stride_i_out,
        stride_j_out,
        stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ---------------------------------------------------------

    # ------------------------ 6. Output LayerNorm and gate ------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(out.dtype)

    # ------------------------ 7. Final linear projection ------------------------
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)

    return out

"""
TriMul “outgoing” update – Triton‑accelerated core reduction.

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a per‑channel (hidden_dim) matrix multiplication of two
(N×N) matrices.  This kernel tiles the i, j, k dimensions (M, N, K)
and performs the reduction in FP16 while accumulating in FP32.
All surrounding LayerNorms, linear projections and gating are
implemented in PyTorch (half‑precision where possible) and the
resulting tensor is stitched back to FP32 for the final LayerNorm
and linear projection.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,                # *fp16
    right_ptr,               # *fp16
    out_ptr,                 # *fp32
    N,                       # seq_len   (constexpr int)
    H,                       # hidden dim (constexpr int)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr,   # tile size in i direction
    BLOCK_N: tl.constexpr,   # tile size in j direction
    BLOCK_K: tl.constexpr,   # tile size in k direction
):
    """
    Compute: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The hidden dimension d is fused into the batch axis (batch*H programs).

    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : base pointers of the tensors.
    N : int
        Sequence length.
    H : int
        Hidden dimension (size of the channel after projection).
    stride_* : int
        Strides (in elements) of the corresponding tensor.
    BLOCK_* : int (constexpr)
        Tile sizes.
    """
    # -------------------- program IDs --------------------
    pid_batch = tl.program_id(0)        # combined batch * hidden channel
    pid_m = tl.program_id(1)            # block row (i)
    pid_n = tl.program_id(2)            # block column (j)

    # de‑compose combined batch id
    b = pid_batch // H                 # original batch index
    d = pid_batch % H                  # hidden channel inside batch

    # -------------------- offsets in i/j --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- k‑loop (reduction) --------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_b_left + d * stride_h_left + \
            offs_m[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # right tile:  shape (BLOCK_K, BLOCK_N)
        # note: we need rows = k, columns = j  →  stride_k for rows, stride_i for columns
        right_ptrs = right_ptr + \
            b * stride_b_right + d * stride_h_right + \
            offs_k[:, None] * stride_k_right + offs_n[None, :] * stride_i_right
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # fused dot‑product, accumulate in FP32
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- write back --------------------
    out_ptrs = out_ptr + \
        b * stride_b_out + d * stride_h_out + \
        offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

        bs * hidden_dim,                                 # batch × hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,                    # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                    # j‑tiles
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_b_left,
        stride_i_left,
        stride_k_left,
        stride_h_left,
        stride_b_right,
        stride_i_right,
        stride_k_right,
        stride_h_right,
        stride_b_out,
        stride_i_out,
        stride_j_out,
        stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,         # 4 warps per block works well on H100
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------

    to_out_norm_b = weights["to_out_norm.bias"]
