def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    b = pid_bh // H
    d = pid_bh % H

    # mask out-of-bound i, j
    mask_i = i < N
    mask_j = j < N

    # offset pointers for left and right
    # left: B, N, N, H
    left_ptrs = left_ptr + b * stride_bl + d * stride_dl + i[:, None] * stride_il + tl.arange(0, BLOCK_K)[None, :] * stride_kl
    # right: B, N, N, H
    # we need to load right with k as rows? we will load right as (k, j)
    right_ptrs = right_ptr + b * stride_br + d * stride_dr + j[None, :] * stride_jr + tl.arange(0, BLOCK_K)[:, None] * stride_kr

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute mask for k
        mask_k = cur_k < N
        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load right tile, note shape (BLOCK_K, BLOCK_N)
        # for right we need stride over k and j.
        # Build pointer: base + cur_k * stride_kr + j * stride_jr + b*..., d*...
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)
        # Convert to fp32 if loaded as fp16
        left = left.to(tl.float32)
        right = right.to(tl.float32)
        # Compute dot product across K dimension: sum left[i,k] * right[k,j]
        # Use tl.dot for each row? Could use matmul: tl.dot(left[i, :], right[:, j])
        # We can use tl.dot for each block pair by using broadcasting.
        # Using tl.dot around matrix multiply:
        # Perform outer product: left @ right
        acc += tl.dot(left, right) # shape (BLOCK_M, BLOCK_N)
        # increment pointers
        left_ptrs += BLOCK_K * stride_kl
        right_ptrs += BLOCK_K * stride_kr

    # Write out
    out_ptrs = out_ptr + b * stride_bo + d * stride_do + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0) # i tile
    pid_j = tl.program_id(1) # j tile
    pid_bh = tl.program_id(2) # (batch, hidden)

    # get i and j indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch and hidden dim from linearized id
    b = pid_bh // H
    d = pid_bh % H

    # out-of-bound masks
    mask_i = i < N
    mask_j = j < N

    # offsets for left (L) and right (R)
    # base offset for batch and hidden
    offset_base = b * stride_bl + d * stride_dl

    # precompute static offset arrays for i and j
    offs_i = i[:, None] * stride_il
    offs_j = j[None, :] * stride_jr

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction dimension k
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)  # shape (BLOCK_K,)
        mask_k = cur_k < N

        # pointers
        a_ptrs = left_ptr + offset_base + offs_i + cur_k[None, :] * stride_kl
        b_ptrs = right_ptr + offset_base + offs_j + cur_k[:, None] * stride_kr

        # load tiles with masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # upcast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # write out
    out_ptrs = out_ptr + b * stride_bo + d * stride_do + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the outgoing version of the TriMul operator.
    The kernel computes for each batch b and hidden channel h:
        out[b,i,j,h] = sum_{k} left[b,i,k,h] * right[b,j,k,h]
    where left and right are linear projections of the input followed by
    element‑wise gated masking. The heavy cubic O(N^3) part is fused in a
    Triton matmul‑like kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias
    )  # shape [B,N,N,dim]

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim] (fp32)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if required
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_exp = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gating and cast to half for the kernel
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim

    # Allocate output buffer (float32 for accumulation)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Get strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # Triton block size parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    # launch kernel
    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # Apply output layernorm, gating, and final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate  # shape [B,N,N,hidden_dim]
    to_out_weight = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # final shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) – Triton implementation.

The heavy cubic part of the module is
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the input tensor,
masked and gated per element.  This kernel treats each pair (batch, hidden)
as an independent matmul:  left[...,h] @ right[...,h].T.
The reduction over the sequence dimension `k` is performed in a tiled
fashion using `tl.dot`, which maps to Tensor‑Core instructions on H100.
All other cheap operations (LayerNorm, linear projections, gating,
masking, output normalization and final projection) are done in PyTorch.
The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimension up to 384.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul for every (batch, hidden) pair:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
    """
    pid_i = tl.program_id(0)      # i‑tile
    pid_j = tl.program_id(1)      # j‑tile
    pid_bh = tl.program_id(2)     # flattened (batch, hidden)

    # i, j coordinates of the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # decode batch and hidden index from linear id
    b = pid_bh // H
    d = pid_bh % H

    # out‑of‑bounds masks for the edges
    mask_i = i < N
    mask_j = j < N

    # Base offset that contains the batch and hidden contributions
    offset_base = b * stride_bl + d * stride_dl

    # Pre‑compute static part of the address (i‑ and j‑offsets)
    offs_i = i[:, None] * stride_il          # (BLOCK_M,1)
    offs_j = j[None, :] * stride_jr          # (1,BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)   # (BLOCK_K,)
        mask_k = cur_k < N

        # Pointers to the current k‑slice of left and right tensors
        a_ptrs = left_ptr + offset_base + offs_i + cur_k[None, :] * stride_kl
        b_ptrs = right_ptr + offset_base + offs_j + cur_k[:, None] * stride_kr

        # Load tiles with edge handling
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K), dtype=fp16
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)          # (BLOCK_K, BLOCK_N), dtype=fp16

        # Up‑cast to FP32 for accumulation (TensorCore does this automatically)
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Matrix‑multiply tile and accumulate
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # Write result back to global memory
    out_ptrs = out_ptr + b * stride_bo + d * stride_do + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a fused Triton kernel.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # fp32
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ----------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_exp = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear projections of x)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates and cast to FP16 for the kernel
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, H = left.shape               # H == hidden_dim
    assert H == hidden_dim, "Hidden dimension mismatch."

    # ----------------------------------------------------------------------
    # 5. Triton kernel: cubic accumulation
    # ----------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides in element units
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # Tile sizes – tuned for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # j‑tiles
        B * H,                           # (batch, hidden) programs
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                    # enough warps for the tile size
    )

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                        # [B, N, N, hidden_dim]

    out = out * out_gate                     # element‑wise gating

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # → [B,N,N,dim]

    return out
