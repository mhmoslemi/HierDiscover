def trmul_forward(a, b):
    batch, M, K = a.shape
    assert a.shape == b.shape
    N = K
    c = torch.empty_like(a, dtype=torch.float32)
    # set block size constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)
    trmul_forward_kernel[grid](
        a_ptr=a, b_ptr=b, c_ptr=c,
        M=M, N=N, K=K,
        stride_Ab=a.stride(0), stride_Am=a.stride(1), stride_Ak=a.stride(2),
        stride_Bb=b.stride(0), stride_Bm=b.stride(1), stride_Bk=b.stride(2),
        stride_Cb=c.stride(0), stride_Cm=c.stride(1), stride_Cn=c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # get weights
    # compute layernorm on input
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)
    # left and right projection
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # apply mask if needed
    if mask is not None:
        # mask shape [bs,N,N]; expand
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # shape bs,N,N,1
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    # reshape for kernel
    # permute to (bs, hidden_dim, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # flatten batch*hidden_dim
    batch_dim = bs * hidden_dim
    left_reshape = left_perm.view(batch_dim, N, N)
    right_reshape = right_perm.view(batch_dim, N, N)
    # allocate output half
    out_half = torch.empty_like(left_reshape, dtype=torch.float16)
    # call kernel
    trmul_forward_kernel(
        left_reshape,
        right_reshape,
        out_half,
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=32,
        num_warps=4
    )
    # reshape back
    out = out_half.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    # multiply by out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” kernel (AlphaFold3 style) – highly optimized forward pass.

The operator consists of:
1) LayerNorm over the last feature dimension.
2) Linear projections (left/right) + masked gating.
3) A massive pair‑wise multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication of shape
   (batch*hidden, N, N) × (batch*hidden, N, N)^T.
   This step is implemented in Triton (FP16 loads, FP32 accumulation).
4) LayerNorm + an output gate, then a final Linear to the original dim.

Only the O(N³·hidden) multiplication is off‑loaded to Triton; the surrounding
operations stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Triton kernel: batched GEMM with transposed B (the “einsum” step)
# --------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    A_ptr,                         # [B*H, M, K] (FP16)
    B_ptr,                         # [B*H, M, K] (FP16) – we read B as (k, j)
    C_ptr,                         # [B*H, M, N] (FP16) – N == K
    M, N, K,                       # matrix dimensions (int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bm, stride_Bk,   # strides for B (original layout)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ Bᵀ for many independent matrices.
    A, B, C are FP16; accumulation is FP32.
    Grid: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch_dim)
    """
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block col index
    pid_b = tl.program_id(2)          # batch*hidden index

    # Offsets within the matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch slice
    a_batch = A_ptr + pid_b * stride_Ab
    b_batch = B_ptr + pid_b * stride_Bb
    c_batch = C_ptr + pid_b * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        # A[i, k]  – regular layout
        a_ptrs = a_batch + (offs_m[:, None] * stride_Am
                            + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_Ak)
        # Bᵀ uses B[k, j] → row index = k, column = j
        b_ptrs = b_batch + ((cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_Bm
                            + offs_n[None, :] * stride_Bk)

        # Load with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M)
                         & ((cur_k + tl.arange(0, BLOCK_K))[None, :] < K),
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=((cur_k + tl.arange(0, BLOCK_K))[:, None] < K)
                         & (offs_n[None, :] < N),
                    other=0.0, dtype=tl.float16)

        # Dot: FP16 inputs → FP32 accumulation
        acc += tl.dot(a, b)

    # Write result (cast back to FP16 for storage)
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_Cm
                        + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def _batched_gemm_fp16(A: torch.Tensor,
                       B: torch.Tensor,
                       BLOCK_M: int = 64,
                       BLOCK_N: int = 64,
                       BLOCK_K: int = 32,
                       num_warps: int = 4) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    A, B: [batch_dim, N, N] FP16 tensors (contiguous).
    Returns C: same shape, FP16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch_dim, M, K = A.shape
    N = K

    # Output allocation
    C = torch.empty((batch_dim, M, N), dtype=A.dtype, device=A.device)

    # Strides (as int64)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bm, stride_Bk = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch_dim)

    _trmul_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bm, stride_Bk,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    return C


# --------------------------------------------------------------------
# Main entry point exposed to the evaluator
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    torch.set_grad_enabled(False)  # no autograd needed

    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, H]

    # ----------------------------------------------------------------
    # 3) Optional mask (broadcast over the feature dim)
    # ----------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] (bool/int/float)
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------
    # 4) Gating (sigmoid on linear maps of the normalized input)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core triple‑multiplication via Triton (FP16 for bandwidth)
    # ----------------------------------------------------------------
    # Cast to half‑precision
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (batch, hidden, N, N) then collapse batch*hidden
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_h = left_h.view(bs * hidden_dim, N, N)   # [B*H, N, N]
    right_h = right_h.view(bs * hidden_dim, N, N) # [B*H, N, N]

    # Triton GEMM (A @ Bᵀ) → result shape [B*H, N, N] (FP16)
    out_h = _batched_gemm_fp16(left_h, right_h,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=32,
                               num_warps=4)

    # Restore original layout and convert back to FP32
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # ----------------------------------------------------------------
    # 6) Output LayerNorm + gate
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ----------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
