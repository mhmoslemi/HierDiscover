def trimul_einsum_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_j, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block col
    pid_h = tl.program_id(2) # head (batch*hidden)

    # Compute b and h from pid_h
    h_idx = pid_h % H
    b_idx = pid_h // H

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b,h)
    a_base = a_ptr + b_idx * stride_a_b + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_b + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_b + h_idx * stride_c_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < N

        # pointers
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j

        # load with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The heavy part of the module is the 4‑D einsum:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We treat each (batch, hidden) pair as an independent matrix‑multiply:
    out_h = left_h @ right_h.T
with left_h, right_h ∈ ℝ^{N×N}.  The kernel tiles the output (i, j) and
loops over the k‑dimension in blocks, using `tl.dot` (FP16 × FP16 → FP32)
for high throughput on H100.  All other tensor ops (LayerNorm, Linear,
gate × sigmoid, final projection) are performed with PyTorch.

The kernel works for any batch ∈ {1,2}, sequence length N ∈ {128,…,1024},
and hidden dimension H ∈ {128,384,768}.  The output dtype is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused einsum ("tri‑multiplicative update")
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    a_ptr,                # left  (B,N,N,H)  FP16
    b_ptr,                # right (B,N,N,H)  FP16
    c_ptr,                # out   (B,N,N,H)  FP16
    B, N, H,              # ints
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_j, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b,h) slice.  The kernel is launched with a 3‑D grid:
        grid = (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_h = tl.program_id(2)          # flattened (batch, hidden)

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    h_idx = pid_h % H                  # hidden dimension index
    b_idx = pid_h // H                 # batch index

    # ------------------------------------------------------------------
    # Offsets for the current block along i, j, k
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_base = a_ptr + b_idx * stride_a_b + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_b + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_b + h_idx * stride_c_h

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_off = k * BLOCK_K
        cur_k = k_off + offs_k
        mask_k = cur_k < N

        # -------------------------------
        # Load a block:  left[i, k, h]
        # -------------------------------
        a_ptrs = a_base + \
                 offs_i[:, None] * stride_a_i + \
                 cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # -----------------------------------------------
        # Load b block:  right[j, k, h]  (note the swap)
        # We need rows = k, cols = j for the matmul.
        # -----------------------------------------------
        b_ptrs = b_base + \
                 cur_k[:, None] * stride_b_k + \
                 offs_j[None, :] * stride_b_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # -------------------------------
        # Dot‑product: (M×K)·(K×N) → (M×N)
        # tl.dot returns FP32 when inputs are FP16.
        # -------------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write result back (store as FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + \
             offs_i[:, None] * stride_c_i + \
             offs_j[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    left, right: shape [B, N, N, H], dtype=torch.float16, contiguous.
    Returns: out of same shape and dtype.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape

    # Output tensor
    out = torch.empty_like(left)

    # Strides in number of elements (not bytes)
    stride_a_b, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – tuned for H100 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * H )
    )

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_a_b, stride_a_i, stride_a_k, stride_a_h,
        stride_b_b, stride_b_j, stride_b_k, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps per block give good occupancy on H100
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (bool/int/float) or None
        - weights      : dict of torch.Tensors containing all learned parameters
        - config       : dict with at least keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Ensure contiguous tensors on CUDA
    x = input_tensor.contiguous()
    device = x.device
    assert device.type == "cuda", "TriMul kernel requires CUDA tensors"

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if provided (mask is broadcasted on the feature dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_expand = mask_tensor.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask_expand
    right = right * mask_expand

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the costly Einstein sum
    # ------------------------------------------------------------------
    left_fp16 = left.half()
    right_fp16 = right.half()

    # ------------------------------------------------------------------
    # Fused einsum via Triton
    # ------------------------------------------------------------------
    out_fp16 = trimul_einsum(left_fp16, right_fp16)   # [B,N,N,hidden_dim] FP16

    # Back to FP32 for the rest of the pipeline
    out = out_fp16.to(x.dtype)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply the output gating
    # ------------------------------------------------------------------
    out = out * out_gate  # shape broadcasted: [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
