"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. **Layer‑norm on the input tensor** – same as `nn.LayerNorm(dim)`.
2. **Linear projections** (half precision)  
   * left  = x·W_left  (no bias)  
   * right = x·W_right
3. **Gating** (half precision)  
   * left_gate  = σ(x·W_left_gate)  
   * right_gate = σ(x·W_right_gate)  
   * out_gate   = σ(x·W_out_gate)   (kept in FP32 for the later output‑norm)
4. **Mask** (if provided) is applied after the projections.
5. **Element‑wise gating** – `left = left * left_gate`, `right = right * right_gate`.
6. **Core TriMul operation** – a batched “inner‑product” over the second
   sequence dimension:

   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`

   This is exactly the `einsum('... i k h, ... j k h -> ... i j h')`.
   A custom Triton kernel performs the sum‑over‑k as a tiled GEMM:
   each hidden channel is processed independently (program‑id axis‑2
   = batch × hidden_dim).  The kernel works on FP16 data and accumulates
   in FP32 for numerical stability.
7. **Layer‑norm on the hidden dimension** (`nn.LayerNorm(hidden_dim)`).
8. **Output gating** – multiply with `out_gate`.
9. **Final linear projection** back to `dim` (`W_out`).

Only step 6 is written in Triton; all other steps use PyTorch
operations (highly optimized on H100 and keep the code concise).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# -------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# -------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)            # i‑block
    pid_n = tl.program_id(1)            # j‑block
    pid_hb = tl.program_id(2)           # combined batch×hidden index

    # -----------------------------------------------------------------
    # Decode batch (b) and hidden channel (h)
    # -----------------------------------------------------------------
    total_hb = B * H
    b = pid_hb // H
    h = pid_hb % H

    # -----------------------------------------------------------------
    # Offsets for the output tile (M × N)
    # -----------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer to the output tile
    out_tile_ptr = out_ptr + \
        b * stride_out_b + \
        (offs_i[:, None] * stride_out_i) + \
        (offs_j[None, :] * stride_out_j) + \
        h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the K dimension (the “summation” dimension)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------------------
        # Load left tile: shape (M, K)
        # -------------------------------------------------------------
        left_tile_ptr = left_ptr + \
            b * stride_left_b + \
            (offs_i[:, None] * stride_left_i) + \
            (offs_k[None, :] * stride_left_k) + \
            h * stride_left_h
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------------------------------------------------------------
        # Load right tile transposed: shape (K, N)
        #   we want right_T[k, j] = right[b, j, k, h]
        # -------------------------------------------------------------
        right_tile_ptr = right_ptr + \
            b * stride_right_b + \
            (offs_k[:, None] * stride_right_i) + \
            (offs_j[None, :] * stride_right_k) + \
            h * stride_right_h
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # -------------------------------------------------------------
        # Accumulate the matrix product for this K‑slice
        # -------------------------------------------------------------
        acc += tl.dot(left, right)          # (M,K) @ (K,N) → (M,N)

    # -----------------------------------------------------------------
    # Store the result (FP32)
    # -----------------------------------------------------------------
    tl.store(out_tile_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    no_mask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm (same semantics as nn.LayerNorm(dim))
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2‑4. Linear projections + gating (use half‑precision for speed)
    # -----------------------------------------------------------------
    x_f16 = x.half()                                 # [B,N,N,dim]  FP16

    left_proj = F.linear(x_f16, weights["left_proj.weight"].half())      # [B,N,N,hidden]
    right_proj = F.linear(x_f16, weights["right_proj.weight"].half())    # [B,N,N,hidden]

    left_gate = F.linear(x_f16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].half()).sigmoid()
    out_gate  = F.linear(x, weights["out_gate.weight"]).sigmoid()        # keep FP32

    # -----------------------------------------------------------------
    # 5. Optional mask (applied after the projection, before gating)
    # -----------------------------------------------------------------
    if mask is not None and not no_mask:
        # mask: [B,N,N] -> [B,N,N,1] and to same dtype as left/right
        mask_f16 = mask.to(dtype=torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # -----------------------------------------------------------------
    # 6. Apply element‑wise gates (still FP16)
    # -----------------------------------------------------------------
    left = left_proj * left_gate            # [B,N,N,hidden]  FP16
    right = right_proj * right_gate         # [B,N,N,hidden]  FP16

    # -----------------------------------------------------------------
    # 7. Core TriMul:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # Allocate output (float32, because the accumulator is FP32)
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=device)

    # Ensure contiguous memory for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # Tile sizes – tuned for H100 (64×64 tiles, K‑slice of 32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim                         # batch × hidden
    )

    triton_tri_mul_kernel[grid](
        left,                                   # left_ptr  (fp16)
        right,                                  # right_ptr (fp16)
        out_hidden,                             # out_ptr   (fp32)
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Hidden‑dim layer‑norm
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden]  (float32)

    # -----------------------------------------------------------------
    # 9. Output gating
    # -----------------------------------------------------------------
    out_gated = out_norm * out_gate   # broadcast over hidden dimension

    # -----------------------------------------------------------------
    # 10. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out_gated, weights["to_out.weight"])   # [B,N,N,dim]

    return out
