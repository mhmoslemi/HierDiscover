def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # layer norm on dim dim
    # Use torch.mean/std along last dim.
    eps = 1e-5
    # norm
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    # compute normalized tensor
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # linear projections
    left_proj_w = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_w)  # out shape [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_w)

        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor
    out = torch.empty_like(left_fp16, dtype=torch.float16)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute strides
    # left stride
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    # right stride

    # out stride

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # get batch id
    batch_pid = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    b = batch_pid // H
    h = batch_pid % H

    # compute start offsets
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    # create range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for i,j boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers start for given batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (i,k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # load right tile (k,j) i.e., right transposed
        # right element: right[b, j, k, h]
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)
    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation.
---------------------------------

We implement the forward pass of AlphaFold3's triangular multiplicative update
(“outgoing”) using a handcrafted Triton kernel for the cubic‑complex part

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

where `left` and `right` are the projected pair representations.
The kernel treats each (batch, hidden) pair as an independent matrix‑multiplication
task:
    C = A @ Bᵀ   with A = left[b,:,:,h]  (N×N) and B = right[b,:,:,h] (N×N).

All LayerNorms, Linear projections and gating are performed in PyTorch; the
heavy, N³ reduction is accelerated with Triton.  The kernel works in FP16 for
speed while the surrounding PyTorch code stays in FP32 (except the final
output which is required in FP32).

The implementation supports:
  * batch size B ∈ {1,2}
  * sequence length N ∈ {128,256,512,1024}
  * feature dimension C = dim ∈ {128,384}
  * hidden dimension H = hidden_dim ∈ {128}
  * optional binary mask (broadcasted over the hidden dimension)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with the second operand transposed.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # FP16 pointer, shape [B,N,N,H]
    right_ptr,              # FP16 pointer, shape [B,N,N,H]
    out_ptr,                # FP16 pointer, shape [B,N,N,H]
    B, N, H,                # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile rows  (i dimension)
    BLOCK_N: tl.constexpr,  # tile cols  (j dimension)
    BLOCK_K: tl.constexpr   # reduction tile (k dimension)
):
    # --------------------------------------------------------------
    # 1) program IDs -> (batch*hidden), i‑tile, j‑tile
    # --------------------------------------------------------------
    pid_batch = tl.program_id(0)   # [0 .. B*H)
    pid_i = tl.program_id(1)       # i‑tile index
    pid_j = tl.program_id(2)       # j‑tile index

    # decode batch and hidden indices
    b = pid_batch // H
    h = pid_batch %  H

    # compute starting coordinates of the output tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # 2) create offset vectors and masks for the current tile
    # ------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) reduction over k (the third axis)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: A_{i,k} -----
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- right tile transposed: Bᵀ_{k,j} = right[b, j, k, h] -----
        b_ptrs = right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i   # note the swap (j ↔ k)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ----- dot‑product on the tile -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 4) write back the result (still in FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Python entry point – matches the required signature.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = C_z (often 128)

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias is optional – we are given weight & bias)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (FP16 for the heavy tensors)
    # ------------------------------------------------------------------
    # Project to hidden dimension (no bias)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # work in FP16 for the matrices that will be multiplied
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Apply mask if present (mask is broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left  * mask.to(left.dtype)
        right = right * mask.to(right.dtype)

    # ------------------------------------------------------------------
    # Gating (computed in FP32 for stability, then cast)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden] fp32
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,hidden] fp32
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden] fp32

    left  = left  * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # ------------------------------------------------------------------
    # Triton kernel: out = Σ_k left[...,k,:] * right[...,k,:]ᵀ
    # ------------------------------------------------------------------
    # Allocate output buffer (FP16)
    out_fp16 = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Tile sizes – chosen to respect the 1024‑thread limit per block
    BLOCK_M = 32   # rows  (i dimension)
    BLOCK_N = 32   # cols  (j dimension)
    BLOCK_K = 32   # reduction chunk

    # strides for the three tensors (all are contiguous in the last dim)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # grid: (batch*hidden, i_tiles, j_tiles)
    i_tiles = (seq_len + BLOCK_M - 1) // BLOCK_M
    j_tiles = (seq_len + BLOCK_N - 1) // BLOCK_N
    grid = (bs * hidden_dim, i_tiles, j_tiles)

    tri_mul_kernel[grid](
        left, right, out_fp16,
        bs, seq_len, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # fine‑tune for H100; 4 warps per block is a good default
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Convert back to FP32 for the remaining operations
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # shape [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, normalized_shape=(hidden_dim,),
                            weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # Apply the output gate (already computed in FP32)
    out_gated = out_norm * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # shape [dim, hidden_dim]
    final_out = F.linear(out_gated, to_out_w)   # [B, N, N, dim] (FP32)

    return final_out
