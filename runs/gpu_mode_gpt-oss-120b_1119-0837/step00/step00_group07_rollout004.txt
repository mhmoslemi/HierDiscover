```python
import triton
import triton.language as tl

@triton.jit
def trmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)
    batch = tl.program_id(axis=1)
    
    # Compute block positions
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n
    
    # Offsets for M and N within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Compute start pointers for this batch (including batch stride)
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Mask for out-of-range K
        mask_k = k_offset < K
        
        # Pointers for A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        # Load A tile
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & mask_k[None, :], other=0.0)
        # Convert to fp16 if needed (load returns same dtype as pointer; we assume half)
        # Pointers for B tile (K x N)
        b_ptrs = b_batch_ptr + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_n[None, :] < N), other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)
    
    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections (no bias). Use torch.nn.functional.linear.
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not config.get("nomask", True):
        mask_broadcast = mask[..., None].to(x.dtype)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, hidden_dim, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # (B, hidden_dim, N, N)
    right_T_fp16 = right_fp16.permute(0,1,3,2).contiguous()               # (B, hidden_dim, N, N) transposed

    _, _, _, N_ = right_T_fp16.shape

    left_mat = left_fp16.view(total_batches, M, K)
    right_mat = right_T_fp16.view(total_batches, K, N_)

    # Output allocation
    out_mat = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # Kernel config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_cm, stride_cn,

"""
TriMul (outgoing) – fast Triton implementation.

The operator works on a 4‑D tensor x of shape [B, N, N, D]:
1. Layer‑norm over the last dimension D.
2. Two linear projections (left/right) and three linear gates (left, right, out).
3. Optional element‑wise mask applied to the projected tensors.
4. Gated tensors are multiplied with a custom Triton kernel that computes,
   for every hidden channel h,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix‑multiply of shape (N×N)·(N×N)ᵀ per channel.
   The kernel works in FP16 (inputs) and accumulates in FP32.
5. Layer‑norm on the hidden dimension, out‑gate multiplication and a final
   linear projection back to the original dimension.

Only the heavy O(N³) contraction (step 4) is performed on the GPU via Triton;
all other operations use PyTorch which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel for the core contraction: out = left @ rightᵀ  (per hidden dim)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,              # A row‑stride, col‑stride
    stride_bk, stride_bn,              # B row‑stride, col‑stride
    stride_cm, stride_cn,              # C row‑stride, col‑stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b].T   (A: M×K, B: N×K)"""

    pid = tl.program_id(axis=0)          # 2‑D grid: (tiles over M,N)
    batch = tl.program_id(axis=1)        # each batch = (B * hidden_dim)

    # --------------------------------------------------------------------- #
    #   Tile indices
    # --------------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # offsets inside the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Base pointers for this batch
    # --------------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # --------------------------------------------------------------------- #
    #   Accumulator in FP32
    # --------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   Loop over the K dimension (the contracted axis)
    # --------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K                 # guard for partial tiles

        # A tile (M × K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # B tile (K × N) – note that B is already transposed before the call
        b_ptrs = b_batch + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # MAC
        acc += tl.dot(a, b)          # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Store result (convert back to FP16)
    # --------------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -----------------------------------------------------------------------------#
# Entry point called by the tester
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    #   1) Layer‑norm over the feature dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    #   2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    #   3) Optional mask
    # --------------------------------------------------------------------- #
    if not config.get("nomask", True):
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   4) Gating
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   5) Core contraction via Triton (FP16 for bandwidth, FP32 accumulation)
    # --------------------------------------------------------------------- #
    #   Permute to (B, hidden, N, N) and cast to half
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # (B, H, N, N)

    #   Transpose the last two axes of right so that we compute A @ Bᵀ
    right_T_fp16 = right_fp16.permute(0, 1, 3, 2).contiguous()            # (B, H, N, N)

    B_, H_, M, K = left_fp16.shape               # M == K == N
    _, _, _, N_ = right_T_fp16.shape
    assert M == K == N_ == N, "Shape mismatch"

    total_batches = B_ * H_                       # treat (B × hidden) as the batch dimension

    #   Flatten the first two dimensions – now we have classic batched GEMM tensors
    a = left_fp16.view(total_batches, M, K)       # (batch, M, K)  in FP16
    b = right_T_fp16.view(total_batches, K, N_)   # (batch, K, N)  in FP16
    c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # --------------------------------------------------------------------- #
    #   Launch Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (
        ( (M + BLOCK_M - 1) // BLOCK_M ) * ( (N_ + BLOCK_N - 1) // BLOCK_N ),
        total_batches,
    )

    # Strides for the flattened tensors (row‑major layout)
    stride_a_batch = M * K
    stride_am = K          # row‑stride inside A
    stride_ak = 1          # col‑stride inside A

    stride_b_batch = K * N_
    stride_bk = N_         # row‑stride inside B (since B is K×N)
    stride_bn = 1          # col‑stride inside B

    stride_c_batch = M * N_
    stride_cm = N_         # row‑stride inside C
    stride_cn = 1          # col‑stride inside C

    _trmul_kernel[grid](
        a, b, c,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------------- #
    #   Reshape back to [B, N, N, hidden_dim] (float32 for the rest)
    # --------------------------------------------------------------------- #
    out = c.view(B_, hidden_dim, M, N_).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------------- #
    #   6) Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to `dim`

    return out
