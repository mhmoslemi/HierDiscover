def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']
    eps = config.get('layernorm_eps', 1e-5)

    # layernorm
    x = (input_tensor - input_tensor.mean(dim=-1, keepdim=True)) / torch.sqrt(input_tensor.var(dim=-1, unbiased=False, keepdim=True) + eps)
    x = x * weights['norm.weight'] + weights['norm.bias']  # shape B,N,N,dim

    # linear proj
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # mask
    if mask is not None:
        left = left * mask.unsqueeze(-1).to(left.dtype)
        right = right * mask.unsqueeze(-1).to(right.dtype)

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # core multiplication
    # cast to fp16
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # shape [B,N,N,H]

    # stride info
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * H )

    trm_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # Out is shape [B,N,N,H] float32
    # Apply to_out_norm
    # First layernorm over hidden dimension
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_norm, weights['to_out.weight'])
    return out_final

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Input tensor `x` of shape [B, N, N, C] is layer‑normed.
2. Four linear projections (left, right, three gates) are applied using the
   supplied weight tensors.  The left/right projections are masked (if a
   mask is given) and multiplied by their respective sigmoid gates.
3. The core operation is an einsum:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is equivalent to a batched matrix multiplication
   `out = left @ right.transpose(-2, -1)` per hidden dimension `h`.
   This is implemented in a Triton kernel that tiles the (i, j) output
   matrix and reduces over the k‑dimension (the sequence length).
4. The result is layer‑normed, gated by `out_gate`, and passed through a
   final linear projection back to `dim`.
All operations except the large reduction are performed with PyTorch for
clarity; the reduction is fully fused in Triton to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr,            # *FP16  [B,N,N,H]
    right_ptr,           # *FP16  [B,N,N,H]
    out_ptr,             # *FP32  [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid_m = tl.program_id(0)   # i block
    pid_n = tl.program_id(1)   # j block
    pid_hb = tl.program_id(2)  # flattened (batch, hidden)

    # recover batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # offsets within the N dimension
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left[b, i, k, h] (shape MxK) -----
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h \
            + offs_m[:, None] * stride_l_i \
            + k_offset[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=(mask_m[:, None] & mask_k[None, :]),
                       other=0.0)

        # ----- load right[b, j, k, h] as transposed (KxN) -----
        # we need right_T[k, j] = right[b, j, k, h]
        right_t_ptrs = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h \
            + k_offset[:, None] * stride_r_k \
            + offs_n[None, :] * stride_r_j
        right_t = tl.load(right_t_ptrs,
                          mask=(mask_k[:, None] & mask_n[None, :]),
                          other=0.0)

        # ----- dot product over K -----
        acc += tl.dot(left, right_t)   # (M,N) += (M,K) * (K,N)

    # ----- store result -----
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + h * stride_o_h \
        + offs_m[:, None] * stride_o_i \
        + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # Shapes
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ----------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ----------------------------------------------------------------------
    # PyTorch does the mean/var reduction efficiently.
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])        # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # ----------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # Gating values
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Core tri‑multiplicative update (einsum) using Triton
    # ----------------------------------------------------------------------
    # Cast to FP16 for bandwidth‑limited part; accumulation is FP32.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output buffer (FP32) – same shape as left/right
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (must be int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 (64‑bit registers, shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (i blocks, j blocks, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dim, gating, final linear
    # ----------------------------------------------------------------------
    # to_out_norm operates over the hidden dimension
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden]

    # Apply output gate (broadcasted on hidden dim)
    out_norm = out_norm * out_gate

    # Final linear back to `dim`
    final_out = F.linear(out_norm, weights["to_out.weight"])

    return final_out
