def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # compute mean, var across dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias

    # Linear projections
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Triton kernel to compute out = sum_k left[..., i, k, d] * right[..., j, k, d]
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    out = torch.empty_like(left)  # shape same

    # Compile kernel with proper block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    # D is hidden_dim; pass as constexpr
    import triton
    import triton.language as tl

    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, D,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, D_: tl.constexpr
    ):
        pid = tl.program_id(0)
        num_i = tl.cdiv(N, BLOCK_I)
        num_j = tl.cdiv(N, BLOCK_J)
        num_blocks_per_batch = num_i * num_j
        b = pid // num_blocks_per_batch
        pid_in_batch = pid % num_blocks_per_batch
        i_block = pid_in_batch // num_j
        j_block = pid_in_batch % num_j

        # offsets
        i_offset = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
        j_offset = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
        i_mask = i_offset < N
        j_mask = j_offset < N

        # accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J, D_), dtype=tl.float32)

        # loop over k dimension
        for start_k in range(0, N, BLOCK_K):
            cur_k = tl.minimum(N - start_k, BLOCK_K)
            k_offset = start_k + tl.arange(0, BLOCK_K)
            k_mask = k_offset < N

            # pointers
            left_ptrs = left_ptr + b * stride_lb + i_offset[:, None, None] * stride_li + k_offset[None, :, None] * stride_lk + tl.arange(0, D_)[None, None, :] * stride_ld
            right_ptrs = right_ptr + b * stride_rb + j_offset[:, None, None] * stride_rj + k_offset[None, :, None] * stride_rk + tl.arange(0, D_)[None, None, :] * stride_rd
            # load
            left = tl.load(left_ptrs, mask = (i_mask[:, None, None] & k_mask[None, :, None]), other=0.0)
            right = tl.load(right_ptrs, mask = (j_mask[:, None, None] & k_mask[None, :, None]), other=0.0)

            # inner loop over K within block
            # we only need to iterate up to cur_k (actual block size)
            for kk in range(0, BLOCK_K):
                # condition to break if kk >= cur_k
                if kk >= cur_k:
                    break
                # left slice (BLOCK_I, D)
                left_ik = left[:, kk, :]   # (BLOCK_I, D)
                # right slice (BLOCK_J, D)
                right_jk = right[:, kk, :] # (BLOCK_J, D)

                # broadcast and accumulate
                # shape after broadcast (BLOCK_I, BLOCK_J, D)
                acc += left_ik[:, None, :] * right_jk[None, :, :]
        # store
        out_ptrs = out_ptr + b * stride_ob + i_offset[:, None, None] * stride_oi + j_offset[None, :, None] * stride_oj + tl.arange(0, D_)[None, None, :] * stride_od
        tl.store(out_ptrs, acc, mask = (i_mask[:, None, None] & j_mask[None, :, None]))
    # Compute grid size
    grid = (B * ((N + BLOCK_I - 1) // BLOCK_I) * ((N + BLOCK_J - 1) // BLOCK_J),)
    # Determine strides
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()
    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, D_=hidden_dim
    )
    # to_out_norm layernorm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    # Compute mean,var across last dim (hidden_dim)
    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = ((out - mean2) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean2) * torch.rsqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply gating
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    to_out_weight = weights['to_out.weight']
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)
    return out_final

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update (TriMul).
    The kernel fuses the heavy triple-wise multiplication
    out[i,j,:] = sum_k left[i,k,:] * right[j,k,:] for each batch and hidden dimension.
    Remaining linear projections, gating, and layer normalizations are performed in PyTorch.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------- Apply mask if present ----------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------- Gates ----------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    )

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Overview
    --------
    1. Layer‑norm the input pair representation.
    2. Compute left/right linear projections and their gates.
    3. Apply (optional) pair mask and element‑wise gating.
    4. **Heavy part** – perform the triple‑wise multiplication
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       with a custom Triton kernel that iterates over the sequence dimension.
    5. Layer‑norm the intermediate tensor, apply the output gate, and a final
       linear projection back to the original channel dimension.

    The kernel uses one program per (batch, i, j) pair; each program
    contains `hidden_dim` threads (one per hidden channel).  The sequence
    dimension is iterated in blocks of ``BLOCK_K`` (constexpr) and masked
    so the same kernel works for any sequence length up to 1024.
    """
    import torch
    import triton
    import triton.language as tl

    # ------------------- unpack inputs -------------------
    # data = (input_tensor, mask, weights, config)
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input / output channel dimension
    hidden_dim = config["hidden_dim"]  # hidden dimension (always 128 in tests)
    eps = 1e-5

    # ------------------- LayerNorm on input -------------------
    # Equivalent to nn.LayerNorm(dim, eps=eps) with given weight/bias
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------- Linear projections -------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------- Optional mask -------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------- Gating -------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------- Triton kernel: Σ_k left[i,k,d] * right[j,k,d] -------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # strides for the contiguous tensors (torch uses row‑major layout)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tuning parameters (these are compile‑time constants)
    BLOCK_K = 32                     # size of the k‑tile
    MAX_K_BLOCKS = (1024 + BLOCK_K - 1) // BLOCK_K   # enough for any N ≤ 1024

    @triton.jit
    def trimul_kernel(
        left_ptr,   # const float*
        right_ptr,  # const float*
        out_ptr,    # float*
        B, N, D,    # runtime integers
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_K: tl.constexpr,
        MAX_K_BLOCKS: tl.constexpr,
        D_: tl.constexpr,
    ):
        """
        Each program processes ONE (b,i,j) coordinate.
        Within the program, `tl.arange(0, D_)` gives a thread for each hidden channel.
        """
        pid = tl.program_id(0)

        # ---- decode (b,i,j) from pid ----
        total_ij = N * N
        b = pid // total_ij
        ij = pid % total_ij
        i = ij // N
        j = ij % N

        # hidden‑channel index for this lane
        d = tl.arange(0, D_)

        # base pointers (without the k‑dimension)
        left_base  = left_ptr  + b * stride_lb + i * stride_li
        right_base = right_ptr + b * stride_rb + j * stride_rj

        # accumulator for the dot‑product over k
        acc = tl.zeros((D_,), dtype=tl.float32)

        # ---- loop over the sequence dimension in BLOCK_K chunks ----
        for kb in range(0, MAX_K_BLOCKS):
            k_start = kb * BLOCK_K
            for kk in range(0, BLOCK_K):
                k = k_start + kk
                # mask out of range k values (handles any N ≤ 1024)
                k_valid = k < N

                # compute pointers for this k and hidden channel d
                left_ptr_k  = left_base  + k * stride_lk  + d * stride_ld
                right_ptr_k = right_base + k * stride_rk  + d * stride_rd

                # load values (zero if k is out of range)
                l_val = tl.load(left_ptr_k,  mask=k_valid, other=0.0)
                r_val = tl.load(right_ptr_k, mask=k_valid, other=0.0)

                # accumulate product
                acc += l_val * r_val

        # store the result for this (b,i,j) – one value per hidden channel
        out_ptr_ = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj + d * stride_od
        tl.store(out_ptr_, acc)

    # launch configuration: one program per (b,i,j) pair
    grid = (B * N * N,)

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_K=BLOCK_K,
        MAX_K_BLOCKS=MAX_K_BLOCKS,
        D_=hidden_dim,
    )
    # ---------------------------------------------------------

    # LayerNorm on the intermediate representation (hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to the original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
