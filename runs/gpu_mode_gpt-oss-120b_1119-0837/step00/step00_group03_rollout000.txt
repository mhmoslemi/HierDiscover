def batched_matmul_kernel(left_ptr, right_ptr, out_ptr, M, N, K,
                         stride_l_batch, stride_l_m, stride_l_k,
                         stride_r_batch, stride_r_n, stride_r_k,
                         stride_o_batch, stride_o_m, stride_o_n,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    num_blocks_M = tl.cdiv(M, BLOCK_M)
    num_blocks_N = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_M * num_blocks_N
    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    block_i = block_id // num_blocks_N
    block_j = block_id % num_blocks_N

    # Compute offsets in M and N dimensions
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # actual mask for valid ranges
    mask_m = offs_m < M
    mask_n = offs_n < N

    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # compute offsets in K
        offs_k = k + tl.arange(0, BLOCK_K)
        # masks for K
        mask_k = offs_k < K

        # Load left sub-matrix (M x K)
        left_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k)

        right_ptrs = right_batch_ptr + (offs_n[:, None] * stride_r_n + offs_k[None, :] * stride_r_k)

        acc += tl.dot(left, right_T)

    # Store accumulator result to output
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n)

"""
TriMul “outgoing” operator (AlphaFold3) – Triton accelerated version.

The forward pass follows the reference PyTorch implementation:

1. Layer‑Norm over the last channel (dim).
2. Linear projections `left_proj` and `right_proj` (dim → hidden_dim) without bias.
3. Optional mask (broadcasted on the channel axis) applied to the two projections.
4. Gating linear layers (`left_gate`, `right_gate`, `out_gate`) + sigmoid.
5. Element‑wise multiplication with the gates.
6. Core Tensor‑TriMul:
       out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   i.e. a batched matrix multiplication per hidden dimension:
       out_d = (left_d) @ (right_d).T
   This is implemented with a custom Triton kernel that processes the
   (batch * hidden_dim) independent N×N GEMMs in half‑precision while
   accumulating in float32.
7. Layer‑Norm over `hidden_dim`.
8. Element‑wise multiplication with `out_gate`.
9. Final linear projection back to `dim`.

The kernel works for arbitrary batch size (1‑2), sequence length up to 1024
and hidden dimensions that are multiples of 32.  All tensors are expected
to reside on the GPU.  The kernel uses FP16 for the two input matrices to
reduce bandwidth and accumulates in FP32 for numerical stability.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    left_ptr,          # FP16 tensor, shape [B*H, M, K]
    right_ptr,         # FP16 tensor, shape [B*H, N, K]
    out_ptr,           # FP32 tensor, shape [B*H, M, N]
    M, N, K,           # matrix dimensions (all equal to seq_len)
    stride_l_batch,    # stride to next batch in left
    stride_l_m,        # stride to next row in left (M dimension)
    stride_l_k,        # stride to next col in left (K dimension)
    stride_r_batch,    # stride to next batch in right
    stride_r_n,        # stride to next row in right (N dimension)
    stride_r_k,        # stride to next col in right (K dimension)
    stride_o_batch,    # stride to next batch in out
    stride_o_m,        # stride to next row in out (M dimension)
    stride_o_n,        # stride to next col in out (N dimension)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes a batch of independent matrix products:
        out[b] = left[b] @ right[b].T
    where each left/right matrix has shape (M, K) / (N, K) respectively.
    """

    pid = tl.program_id(0)

    # number of blocks covering the M and N dimensions
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # identify which batch (i.e. which (batch, hidden) pair) we are processing
    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # row / column offsets for the current tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch in each operand
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of left (M x BK) – FP16 -> FP32
        left_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m
                                      + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load a tile of right (N x BK) – FP16 -> FP32
        right_ptrs = right_batch_ptr + (offs_n[:, None] * stride_r_n
                                        + offs_k[None, :] * stride_r_k)
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate: left @ right.T  (M x BK) @ (BK x N) -> (M x N)
        acc += tl.dot(left, tl.trans(right))

    # Write back the result (FP32)
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_m
                                + offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm (no torch.nn.Module, operates on last dim)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight + bias
    return x_norm


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] on the same device as the inputs.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all tensors are assumed to be on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]    # [hidden_dim, dim]
    left_gate_weight = weights["left_gate.weight"]      # [hidden_dim, dim]
    right_gate_weight = weights["right_gate.weight"]    # [hidden_dim, dim]
    out_gate_weight = weights["out_gate.weight"]        # [hidden_dim, dim]
    to_out_norm_weight = weights["to_out_norm.weight"]  # [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"]      # [hidden_dim]
    to_out_weight = weights["to_out.weight"]            # [dim, hidden_dim]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_weight, norm_bias)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    if not nomask and mask_tensor is not None:
        # mask has shape [B, N, N]; broadcast to channel dimension
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3️⃣ Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core TriMul: batched matmul over the inner sequence axis
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    For each (b,d) pair this is a matrix product:
    #         (N×N) @ (N×N)^T
    #    We reshape to [B*hidden_dim, N, N] and launch the Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape   # N == seq_len

    # Rearrange to [B, hidden_dim, N, N] → flatten batch+hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Cast to FP16 for the kernel (saves bandwidth)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output in FP32 (accumulator type)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements, not bytes) for the kernel
    stride_l_batch, stride_l_m, stride_l_k = left_fp16.stride()
    stride_r_batch, stride_r_n, stride_r_k = right_fp16.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_fp32.stride()

    # Define block sizes – chosen to fit H100 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n

    # Launch kernel
    grid = (total_blocks,)
    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden_dim, gating, final linear projection
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_weight, to_out_norm_bias)
    out = out * out_gate
    # Final projection back to original dimension (dim)
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
