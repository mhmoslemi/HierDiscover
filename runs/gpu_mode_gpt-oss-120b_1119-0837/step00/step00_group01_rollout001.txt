```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch dimension id
    batch_id = tl.program_id(2)
    # block indices
    pid_m = tl.program_id(0)  # i dimension
    pid_n = tl.program_id(1)  # j dimension

    # compute b and h
    h = batch_id % H
    b = batch_id // H

    # compute starting offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # create range vectors
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute pointers base for given batch and hidden
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        # compute k offset
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # load left tile [M, K]
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed [K, N]
        # We need right[b, j, k, h] transposed => right index order: j (col) then k (row)
        # For tile we want rows = cur_k, cols = offs_j
        right_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product
        # left: [M, K], right: [K, N] => dot adds [M,N]
        acc += tl.dot(left, right, out_type=tl.float32)

        # advance k pointers: not needed as we compute directly each iteration

    # write output
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – Triton‑accelerated core einsum.

The heavy part of the TriMul forward pass is the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for a 4‑D tensor of shape [B, N, N, H].  For each pair (b, h) this is a
matrix multiplication L (i×k) @ Rᵀ (k×j).  The Triton kernel below treats
the (b, h) combination as an independent batch and tiles the i‑ and
j‑dimensions while accumulating over k.  The kernel works on fp16 inputs
(left/right) and accumulates in fp32, producing a fp32 output.  All other
operations (LayerNorm, linear projections, gating and the final linear)
are performed in PyTorch; the kernel only implements the O(N³) contraction,
which is the performance‑critical part of the module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_einsum_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Shapes / sizes
    B, N, H,
    # Strides for left  [b,i,k,h]
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # Strides for right [b,j,k,h]  (j is the second axis, k the third)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # Strides for out   [b,i,j,h]
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # Compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute the “outgoing” TriMul contraction via tiled matrix multiplication.
    Each program instance (pid) handles a tile (BLOCK_M×BLOCK_N) of the output
    for a specific (b, h) pair.
    """
    # -------------------------------------------------------------
    # 1️⃣ Identify the batch (b) and hidden‑channel (h) for this program
    # -------------------------------------------------------------
    pid_h = tl.program_id(2)               # flattened (b, h) index
    b = pid_h // H
    h = pid_h % H

    # -------------------------------------------------------------
    # 2️⃣ Tile indices in the i‑ and j‑ dimensions
    # -------------------------------------------------------------
    pid_i = tl.program_id(0)               # i‑tile id
    pid_j = tl.program_id(1)               # j‑tile id
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # -------------------------------------------------------------
    # 3️⃣ Base pointers for this (b, h) slice
    # -------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b   + h * stride_o_h

    # -------------------------------------------------------------
    # 4️⃣ Accumulate the dot product over k in blocks of size BLOCK_K
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = (N + BLOCK_K - 1) // BLOCK_K        # number of k‑tiles
    for k in range(num_k):
        k_start = k * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left tile   [M, K]  (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile  [K, N]  (k, j) – note the transposition
        right_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate matrix‑multiply (fp16×fp16 → fp32)
        acc += tl.dot(left, right, out_type=tl.float32)

    # -------------------------------------------------------------
    # 5️⃣ Write the result out
    # -------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The large N³ contraction is executed on the GPU via the Triton kernel
    `tri_mul_einsum_kernel`; all surrounding linear / LayerNorm / gating steps
    are performed with regular PyTorch operations.

    Parameters
    ----------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        (input_tensor, mask, weights, config) where
        - input_tensor : [B, N, N, dim] float32
        - mask         : [B, N, N] (bool/float) or None
        - weights      : model weight tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask may be bool or float; convert to same dtype as proj (float32) first
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Allocate output buffer for the contraction (fp32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch Triton kernel
    # ------------------------------------------------------------------
    # Gather strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes – empirically good for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
        B * hidden_dim,                      # (b, h) programs
    )

    tri_mul_einsum_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Post‑contraction LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
