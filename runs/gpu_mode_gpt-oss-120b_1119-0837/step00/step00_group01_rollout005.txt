    left_in_ptr, left_out_ptr,
    right_in_ptr, right_out_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,

    stride_ri_b, stride_ri_n, stride_ri_m, stride_ri_h,

    n_mask = n_off < N
    m_mask = m_off < N
    h_mask = h_off < H

    mask = tl.load(mask_ptr + pid_b * stride_mask_b + n_off[:, None] * stride_mask_n + m_off[None, :] * stride_mask_m,

def custom_kernel(data):
    """Custom TriMul kernel.
    Implements the outgoing triangle multiplicative update as described in AlphaFold3.
    Steps:
    1. LayerNorm over the last dimension.
    2. Linear projections left/right and gate linear layers.
    3. Optional mask gating performed in a fused Triton kernel.
    4. Batched matrix multiplication (left @ right^T) across the hidden dimension.
    5. Output LayerNorm + gating and final linear projection.
    """
    import torch
    import triton
    import triton.language as tl
    # unpack
    input_tensor, mask, weights, config = data
    # Ensure contiguous
    input_tensor = input_tensor.contiguous()
    mask = mask.contiguous() if mask is not None else None
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # Apply mask and gates.
    if config.get("nomask", False):
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure mask dtype matches tensors
        mask_f = mask.to(input_tensor.dtype)
        # allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        # strides
        stride_li_b, stride_li_n, stride_li_m, stride_li_h = left_proj.stride()
        stride_lo_b, stride_lo_n, stride_lo_m, stride_lo_h = left.stride()
        stride_ri_b, stride_ri_n, stride_ri_m, stride_ri_h = right_proj.stride()
        stride_ro_b, stride_ro_n, stride_ro_m, stride_ro_h = right.stride()
        stride_lg_b, stride_lg_n, stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rg_b, stride_rg_n, stride_rg_m, stride_rg_h = right_gate.stride()
        stride_mask_b, stride_mask_n, stride_mask_m = mask_f.stride()
        # Block sizes (tuned for H100)
        BLOCK_N = 64
        BLOCK_M = 64
        BLOCK_H = 32
        grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), B)
        mask_gate_kernel[grid](
            left_proj, left,
            right_proj, right,
            left_gate, right_gate,
            mask_f,
            B, N, hidden_dim,
            stride_li_b, stride_li_n, stride_li_m, stride_li_h,
            stride_lo_b, stride_lo_n, stride_lo_m, stride_lo_h,
            stride_ri_b, stride_ri_n, stride_ri_m, stride_ri_h,
            stride_ro_b, stride_ro_n, stride_ro_m, stride_ro_h,
            stride_lg_b, stride_lg_n, stride_lg_m, stride_lg_h,
            stride_rg_b, stride_rg_n, stride_rg_m, stride_rg_h,
            stride_mask_b, stride_mask_n, stride_mask_m,
            BLOCK_N=BLOCK_N, BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
        )
    # Batched matrix multiplication across hidden dimension
    # reshape to (B*hidden_dim, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # compute left @ right^T
    out = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))
    # reshape back to [B, N, N, hidden_dim]
    out = out.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    # Apply out gate
    out = out * out_gate
    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) kernel.

The forward pass follows the AlphaFold‑3 definition:

1. Layer‑norm over the channel dimension (dim).
2. Linear projections → left_proj, right_proj (dim → hidden_dim).
3. Linear gates → left_gate, right_gate, out_gate (dim → hidden_dim) and sigmoid.
4. Optional mask‑gating.  This element‑wise multiplication (mask * proj * gate)
   is fused into a single Triton kernel to hide memory latency.
5. For each hidden channel d we compute
        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   which is exactly a batched matrix multiplication:
        left_d @ right_dᵀ .
   The batch dimension is (batch * hidden_dim) and we use torch.bmm
   (highly tuned cuBLAS on H100).
6. Layer‑norm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to dim.

All heavy element‑wise work (mask + gating) is executed on the GPU
via a Triton kernel; the O(N³) matrix multiplication is delegated
to cuBLAS for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    # pointers
    left_in_ptr, left_out_ptr,
    right_in_ptr, right_out_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # dimensions
    B, N, H,
    # strides (B,N,N,H)
    stride_li_b, stride_li_n, stride_li_m, stride_li_h,
    stride_lo_b, stride_lo_n, stride_lo_m, stride_lo_h,
    stride_ri_b, stride_ri_n, stride_ri_m, stride_ri_h,
    stride_ro_b, stride_ro_n, stride_ro_m, stride_ro_h,
    stride_lg_b, stride_lg_n, stride_lg_m, stride_lg_h,
    stride_rg_b, stride_rg_n, stride_rg_m, stride_rg_h,
    stride_mask_b, stride_mask_n, stride_mask_m,
    # compile‑time constants
    BLOCK_N: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Fused mask and gate application.

    left_out  = left_in  * left_gate  * mask
    right_out = right_in * right_gate * mask
    """
    pid_b = tl.program_id(2)               # batch index
    pid_n = tl.program_id(0)               # block along first N‑axis
    pid_m = tl.program_id(1)               # block along second N‑axis

    # 1‑D offsets inside each tile
    n_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    m_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    h_off = tl.arange(0, BLOCK_H)

    # bounds
    n_mask = n_off < N
    m_mask = m_off < N
    h_mask = h_off < H

    # ------------ mask (broadcast over hidden dim) ------------
    # mask shape: [B, N, N]
    mask = tl.load(
        mask_ptr + pid_b * stride_mask_b
        + n_off[:, None] * stride_mask_n
        + m_off[None, :] * stride_mask_m,
        mask=n_mask[:, None] & m_mask[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # [BLOCK_N, BLOCK_M, 1]

    # ------------ left side ----------
    left = tl.load(
        left_in_ptr
        + pid_b * stride_li_b
        + n_off[:, None] * stride_li_n
        + m_off[None, :] * stride_li_m
        + h_off[None, None, :] * stride_li_h,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
        other=0.0,
    )
    left_gate = tl.load(
        left_gate_ptr
        + pid_b * stride_lg_b
        + n_off[:, None] * stride_lg_n
        + m_off[None, :] * stride_lg_m
        + h_off[None, None, :] * stride_lg_h,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
        other=0.0,
    )
    left_out = left * left_gate * mask
    tl.store(
        left_out_ptr
        + pid_b * stride_lo_b
        + n_off[:, None] * stride_lo_n
        + m_off[None, :] * stride_lo_m
        + h_off[None, None, :] * stride_lo_h,
        left_out,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
    )

    # ------------ right side ----------
    right = tl.load(
        right_in_ptr
        + pid_b * stride_ri_b
        + n_off[:, None] * stride_ri_n
        + m_off[None, :] * stride_ri_m
        + h_off[None, None, :] * stride_ri_h,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + pid_b * stride_rg_b
        + n_off[:, None] * stride_rg_n
        + m_off[None, :] * stride_rg_m
        + h_off[None, None, :] * stride_rg_h,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
        other=0.0,
    )
    right_out = right * right_gate * mask
    tl.store(
        right_out_ptr
        + pid_b * stride_ro_b
        + n_off[:, None] * stride_ro_n
        + m_off[None, :] * stride_ro_m
        + h_off[None, None, :] * stride_ro_h,
        right_out,
        mask=n_mask[:, None] & m_mask[None, :] & h_mask[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (ignored when config['nomask']==True)
        - weights      : dict of weight tensors (see docstring)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as inputs).
    """
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    input_tensor = input_tensor.contiguous()
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input layer‑norm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Gates (with sigmoid)
    # ------------------------------------------------------------------ #
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # later

    # ------------------------------------------------------------------ #
    # 4. Fuse mask + gates via Triton (if a mask is present)
    # ------------------------------------------------------------------ #
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # ensure mask is float and same device
        mask = mask.to(input_tensor.dtype).contiguous()

        # allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides for all tensors (C‑contiguous layout)
        s_li = left_proj.stride()
        s_lo = left.stride()
        s_ri = right_proj.stride()
        s_ro = right.stride()
        s_lg = left_gate.stride()
        s_rg = right_gate.stride()
        s_m = mask.stride()

        # block sizes (tuned for H100)
        BLOCK_N = 64
        BLOCK_M = 64
        BLOCK_H = 32

        grid = (
            (N + BLOCK_N - 1) // BLOCK_N,   # axis 0 → N
            (N + BLOCK_M - 1) // BLOCK_M,   # axis 1 → N
            B,                              # axis 2 → batch
        )

        mask_gate_kernel[grid](
            # pointers
            left_proj, left,
            right_proj, right,
            left_gate, right_gate,
            mask,
            # dimensions
            B, N, hidden_dim,
            # strides
            s_li[0], s_li[1], s_li[2], s_li[3],
            s_lo[0], s_lo[1], s_lo[2], s_lo[3],
            s_ri[0], s_ri[1], s_ri[2], s_ri[3],
            s_ro[0], s_ro[1], s_ro[2], s_ro[3],
            s_lg[0], s_lg[1], s_lg[2], s_lg[3],
            s_rg[0], s_rg[1], s_rg[2], s_rg[3],
            s_m[0], s_m[1], s_m[2],
            # compile‑time block sizes
            BLOCK_N=BLOCK_N,
            BLOCK_M=BLOCK_M,
            BLOCK_H=BLOCK_H,
        )
        # Free the temporary gate tensors (optional, let GC handle)

    # ------------------------------------------------------------------ #
    # 5. Batched matrix multiplication across the hidden channel
    #    (left @ right^T for each hidden slice)
    # ------------------------------------------------------------------ #
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # core O(N³) work – cuBLAS handles this extremely fast on H100
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
