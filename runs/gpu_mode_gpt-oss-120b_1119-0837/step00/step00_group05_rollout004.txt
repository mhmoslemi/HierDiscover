    left_proj_w = weights["left_proj.weight"]  # shape (hidden_dim, dim)

"""
custom_kernel
==============

Highly‑optimized Triton implementation of the “outgoing” TriMul operator
used in AlphaFold‑3 and related protein structure models.

The forward pass consists of:
1. Layer‑norm on the 4‑D input tensor.
2. Two linear projections (left/right) without bias.
3. Optional pairwise mask.
4. Three sigmoid gating linear layers (left, right, out).
5. Element‑wise gated left/right tensors.
6. **Batched matrix‑multiplication per hidden channel**:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (equivalently left @ rightᵀ for every hidden channel).  
   This O(N³·H) kernel is the only part written in Triton.
7. Layer‑norm on the hidden dimension.
8. Apply out‑gate and final linear projection back to the model dimension.

Only the heavy N³ reduction (step 6) is off‑loaded to Triton; everything
else uses PyTorch for simplicity. The kernel works with arbitrary batch
size (B∈{1,2}), sequence length N (up to 1024) and hidden dimension H
(typically 128). It operates in float16 internally for bandwidth
efficiency and accumulates in float32 to preserve numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul where the right operand is transposed.
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    B, N, H,           # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b, h) pair.  Grid dimensions are:
        axis 0 -> i‑tiles, axis 1 -> j‑tiles, axis 2 -> (b*H) instances.
    """
    pid_i = tl.program_id(0)               # i‑tile index
    pid_j = tl.program_id(1)               # j‑tile index
    pid_h = tl.program_id(2)               # combined batch‑hidden index

    # ------------------------------------------------------------------
    # Decode (batch, hidden) from the third grid dimension
    # ------------------------------------------------------------------
    bc = pid_h
    b = bc // H
    h = bc % H

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (float32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main reduction over the k‑dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        cur_k = tk * BLOCK_K
        cur_offs_k = cur_k + offs_k            # absolute k indices for this tile
        mask_k = cur_offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_li + cur_offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                 # fp16 -> promoted to fp32 later

        # Load B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + cur_offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Partial matrix multiplication
        acc += tl.dot(a, b)                    # acc stays fp32

    # ------------------------------------------------------------------
    # Store the result tile (cast back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N]  (bool/float) or None
        - weights      : dict of parameter tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)
    left  = F.linear(x, left_proj_w)               # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(x)    # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched matmul (left @ rightᵀ per hidden channel)
    #    Use float16 for bandwidth, accumulate in float32.
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, H = left_h.shape

    out_h = torch.empty((B, N, N, H), dtype=torch.float16, device=left_h.device)

    # Strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_rj, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out_h.stride()

    # Tunable block sizes (chosen for H100, fp16)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H                           # (batch, hidden) programs
    )

    trimul_matmul_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for the remaining operations
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Layer‑norm on the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to model dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # [B,N,N,dim]

    return out
