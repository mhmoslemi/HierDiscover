def batched_matmul_kernel(
    A_ptr,  # left pointer
    B_ptr,  # right_T pointer
    C_ptr,  # output pointer
    BATCH, M, N, K,
    stride_Ab, stride_Al, stride_Ak,
    stride_Bb, stride_Bl, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Pointers offset for batch
    A_batch_ptr = A_ptr + batch_id * stride_Ab
    B_batch_ptr = B_ptr + batch_id * stride_Bb
    C_batch_ptr = C_ptr + batch_id * stride_Cb
    
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask out of bounds
        a_ptrs = A_batch_ptr + (offs_m[:, None] * stride_Al + offs_k[None, :] * stride_Ak)
        b_ptrs = B_batch_ptr + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bl)
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
    # Write back
    c_ptrs = C_batch_ptr + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
TriMul “outgoing” forward implementation with a fused batched‑matrix‑multiply
kernel written in Triton.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ (B,N,N,D) (D = dim).
2. Linear projections:
   left  = x @ W_left   (shape B,N,N,H)
   right = x @ W_right  (shape B,N,N,H)   with H = hidden_dim.
3. Optional mask (B,N,N) is broadcast over the channel dimension.
4. Gating (sigmoid) on the *same* normalized input:
   left  *= sigmoid(x @ W_left_gate)
   right *= sigmoid(x @ W_right_gate)
   out_gate = sigmoid(x @ W_out_gate)               # (B,N,N,H)
5. Prepare the two tensors for a batched matrix multiplication:
   left  → (B,H,N,N)   permute(0,3,1,2)  → (B*H,N,N)   “A” (i,k)
   right → (B,H,N,N)   permute(0,3,2,1)  → (B*H,N,N)   “B” (k,j)
   The product is   out_{i,j} = Σ_k A_{i,k} * B_{k,j}.
   This is a standard batched matmul with batch = B*H.
6. The heavy N³ work is performed by `batched_matmul_kernel` (Triton).
   Inputs are cast to fp16, multiplication is accumulated in fp32,
   and the result is stored in fp32.
7. Reshape the result back to (B,N,N,H) and apply the final
   Layer‑Norm, gating and linear projection.
8. Return the tensor of shape (B,N,N,D).

The kernel works for any batch size ≤2, any sequence length N in
{128,256,512,1024,…} and hidden_dim up to 384.  No gradients are
computed – only the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    batch_dim, M, N, K,                 # sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched (A @ B) where A: (batch_dim, M, K), B: (batch_dim, K, N).
       The result C has shape (batch_dim, M, N).  All tensors are contiguous.
    """
    pid_batch = tl.program_id(0)   # which batch*head we are processing
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # pointers to the current sub‑tiles
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)

        # load fp16 tiles (other=0.0 for masked elements)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # dot product – accumulation in fp32
        acc += tl.dot(a, b)

    # write back with boundary check
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data

    # shape / config
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the channel dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: (B,N,N) → (B,N,N,1) and cast to input dtype
        mask_ = mask.to(dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) on the same normalized input
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate         # (B,N,N,hidden_dim)
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for batched matrix multiplication
    #    left  : (B, H, N, N)  with axes (i, k)
    #    right : (B, H, N, N)  with axes (k, j)  (i.e. permuted)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  k,j

    # collapse the (batch, head) dimensions → single batch for the kernel
    batch_dim = bs * hidden_dim
    left_view = left_perm.view(batch_dim, N, N)   # (B*H, M, K)
    right_view = right_perm.view(batch_dim, N, N) # (B*H, K, N)

    # cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32)
    out_view = torch.empty(
        (batch_dim, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # strides expressed in number of elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_fp16.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_fp16.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_view.stride()

    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_view,
        batch_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B,N,N,hidden_dim) and post‑processing
    # ------------------------------------------------------------------
    out_perm = out_view.view(bs, hidden_dim, N, N)          # (B,H,N,N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()        # (B,N,N,H)

    # layer‑norm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # gating
    out = out * out_gate

    # final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
