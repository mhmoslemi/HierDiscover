def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # IDs
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # Determine indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # Masks for bounds
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # Offsets for i,j,d
    # Compute base pointers
    b_offset_l = b * stride_l_b
    b_offset_r = b * stride_r_b
    b_offset_o = b * stride_o_b

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):

        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        left_ptrs = left_ptr + b_offset_l + i[:, None, None] * stride_l_i + k[None, :, None] * stride_l_k + d[None, None, :] * stride_l_h

        right_ptrs = right_ptr + b_offset_r + k[:, None, None] * stride_r_k + j[None, :, None] * stride_r_j + d[None, None, :] * stride_r_h

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0, dtype=tl.float16)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Convert to half for speed
    x = x.to(torch.float16)

    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # shape [..., hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # prepare mask as half
    if mask is None:
        mask_f = torch.ones_like(left, dtype=left.dtype)
    else:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # shape [B,N,N,1]

    left = left * mask_f
    right = right * mask_f

    # Gate computations
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor for hidden_dim stage (float16)
    out = torch.empty_like(left, dtype=torch.float16)

    # Kernel launch
    B, N, _, _ = left.shape
    # stride arrays
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 8

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            (hidden_dim + BLOCK_D - 1) // BLOCK_D)

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )

    # Convert to float32 for subsequent ops
    out = out.to(torch.float32)

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    # multiply by out_gate (out_gate is float16? we computed in float16)
    out_gate = out_gate.to(out.dtype)  # match dtype
    out = out * out_gate

    # final linear to dim
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)

    return out

    stride_r_b, stride_r_j, stride_r_k, stride_r_h,

    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul "outgoing" implementation (AlphaFold3 style)

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (dim = last axis).
2. Linear projections:
   * left = x @ W_left      (dim → hidden_dim)
   * right = x @ W_right    (dim → hidden_dim)
3. Apply the optional pairwise mask (broadcast on the hidden axis).
4. Compute three sigmoid gates from the normalized input:
   left_gate, right_gate, out_gate.
   Multiply left/right by their respective gates.
5. Core tensor contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply per hidden channel.
   It is performed by a custom Triton kernel that tiles the
   (i, j, k) dimensions and iterates over hidden channels in
   small blocks, accumulating in FP32.
6. Layer‑normalize the hidden‑dim result, multiply by out_gate,
   and apply the final linear projection back to the original dim.
All operations after step 5 remain in PyTorch for simplicity,
but the heavy N³ contraction is fully fused on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the einsum: out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float*  (B,N,N,H)
    right_ptr,         # const float*
    out_ptr,           # float*        (output)
    B, N, H,           # tensor sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,  # strides of left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,  # strides of right (j‑dim is second)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,  # strides of output
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    """Compute the TriMul contraction using block‑wise GEMM."""
    # ------------------------------------------------------------------
    # 3‑D grid: (batch+hidden‑block, block_i, block_j)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)          # encodes batch index + hidden‑block index
    pid_i = tl.program_id(1)        # block over i
    pid_j = tl.program_id(2)        # block over j

    # number of blocks that cover the hidden dimension
    num_d_blocks = tl.cdiv(H, BLOCK_D)

    # decode batch index and hidden‑block index from pid
    b = pid // num_d_blocks
    pid_d = pid % num_d_blocks
    d_base = pid_d * BLOCK_D

    # ------------------------------------------------------------------
    # Offsets for the (i, j, d) tile processed by this program instance
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = d_base + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left:   shape (M, K, D)  -> indices (i, k, d)
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + k[None, :, None] * stride_l_k
                     + d[None, None, :] * stride_l_h)

        # right:  shape (K, N, D)  -> indices (k, j, d)
        # note the transposition (k first, j second) to match GEMM layout
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + k[:, None, None] * stride_r_k
                      + j[None, :, None] * stride_r_j
                      + d[None, None, :] * stride_r_h)

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0,
                       dtype=tl.float32)      # (M, K, D)

        right = tl.load(right_ptrs,
                        mask=mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :],
                        other=0.0,
                        dtype=tl.float32)     # (K, N, D)

        # ------------------------------------------------------------------
        # Compute a small GEMM for each hidden sub‑block d
        # ------------------------------------------------------------------
        for d_off in range(0, BLOCK_D):
            # left[:, :, d_off] -> (M, K)
            # right[:, :, d_off] -> (K, N)
            acc[:, :, d_off] += tl.dot(left[:, :, d_off], right[:, :, d_off])

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + i[:, None, None] * stride_o_i
                + j[None, :, None] * stride_o_j
                + d[None, None, :] * stride_o_h)

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (bool / 0‑1)
        - weights      : dict of parameter tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the last axis
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]   (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if any).  Mask is broadcast on the hidden axis.
    # --------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones_like(left[..., :1])
    else:
        mask_f = mask.unsqueeze(-1).to(left.dtype)     # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction (tri‑mul) via Triton
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    # allocate output tensor for the hidden‑dim intermediate result
    out_hidden = torch.empty_like(left, dtype=torch.float32)

    # strides (as Python ints)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()

    # --------------------------------------------------------------
    # Tiling configuration (chosen to respect shared‑memory limits on H100)
    # --------------------------------------------------------------
    BLOCK_M = 16    # rows   (i‑axis)
    BLOCK_N = 16    # cols   (j‑axis)
    BLOCK_K = 16    # reduction axis (k‑axis)
    BLOCK_D = 8     # hidden‑channel sub‑block

    grid = (
        B * ((hidden_dim + BLOCK_D - 1) // BLOCK_D),   # batch × hidden‑block
        (N + BLOCK_M - 1) // BLOCK_M,                 # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,                 # j‑blocks
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_hidden.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # Ensure kernel completion before next ops (optional but safe)
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6. Output normalization, gating and final projection
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_hidden = out_hidden * out_gate  # broadcasted over hidden dim
    output = F.linear(out_hidden, weights["to_out.weight"])  # back to dim

    return output
