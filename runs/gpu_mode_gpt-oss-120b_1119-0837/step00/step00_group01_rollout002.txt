tri‑linear kernel.

The heavy part of the module is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite it as a batched matrix multiplication:
    left  → A(b,d) = left[b, i, k, d]   (shape N×N)
    right → B(b,d) = right[b, j, k, d]  (shape N×N)
    out   = A(b,d) @ B(b,d).T            (shape N×N)

All batch×hidden dimensions are collapsed and a single Triton kernel
computes many independent N×N matrix products in parallel.  The kernel
handles the k‑reduction in blocks, accumulating in FP32 while reading the
operands as FP32 (the modest extra bandwidth is fine on an H100 and
keeps numerical error low).  The remaining linear layers, LayerNorms and
gates are performed with torch operations.

The function `custom_kernel` follows the signature required by the test
harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel : batched matrix multiplication (A @ B.T)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr,                     # fp32*   [B*H, M, K]
    B_ptr,                     # fp32*   [B*H, K, N]   (B already transposed)
    C_ptr,                     # fp32*   [B*H, M, N]
    M, N, K,                   # int
    stride_ah, stride_am, stride_ak,   # int strides for A
    stride_bh, stride_bk, stride_bn,   # int strides for B
    stride_ch, stride_cm, stride_cn,   # int strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[h, i, j] = Σ_k A[h, i, k] * B[h, k, j]
    over a 3‑D grid (block_i, block_j, batch_h).
    """
    pid_h = tl.program_id(2)          # batch * hidden index
    pid_i = tl.program_id(0)          # block row
    pid_j = tl.program_id(1)          # block col

    # ------------------------------------------------------------------ #
    # Offsets for the current block
    # ------------------------------------------------------------------ #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in BLOCK_K sized chunks
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_iter * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----------------------------------------------------------------
        # Pointers with proper stride handling
        # A[h, i, k]
        a_ptrs = (
            A_ptr
            + pid_h * stride_ah
            + offs_i[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B[h, k, j]  (note that B is already transposed)
        b_ptrs = (
            B_ptr
            + pid_h * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn
        )

        # Load with out‑of‑bounds masking (zero‑pad)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Matrix‑multiply the micro‑tiles
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # -------------------------------------------------------------------- #
    # Store the result
    # -------------------------------------------------------------------- #
    c_ptrs = (
        C_ptr
        + pid_h * stride_ch
        + offs_i[:, None] * stride_cm
        + offs_j[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------------- #
# Public entry point used by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------- LayerNorm ------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------ Linear proj ---------------------------- #
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------ Masking -------------------------------- #
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(left)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------ Gates ---------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------- Prepare tensors for Triton -------------------- #
    B, N, _, _ = left.shape   # B, seq_len, seq_len, hidden_dim
    # collapse batch & hidden dimension (treated as independent matmul tasks)
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden, i, k)
    right_t = right.permute(0, 3, 2, 1).contiguous()       # (B, hidden, k, i)   <-- transposed for B.T
    batch_hidden = B * hidden_dim

    A = left_t.view(batch_hidden, N, N)    # (BH, M=N, K=N)
    B_mat = right_t.view(batch_hidden, N, N)   # (BH, K=N, N=N)

    # Output tensor (FP32)
    out_pre = torch.empty_like(A, dtype=torch.float32)   # (BH, N, N)

    # -------------------------- Triton launch ----------------------------- #
    # Block sizes – values tuned for typical N (256‑1024) on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    # 3‑D grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (grid_m, grid_n, batch_hidden)

    _tri_mul_matmul_kernel[grid](
        A,
        B_mat,
        out_pre,
        N,
        N,
        N,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B_mat.stride(0),
        B_mat.stride(1),
        B_mat.stride(2),
        out_pre.stride(0),
        out_pre.stride(1),
        out_pre.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # Reshape back to the original layout: (B, N, N, hidden_dim)
    out_pre = out_pre.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------- Final LN & Linear ---------------------- #
    out = F.layer_norm(
        out_pre,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
