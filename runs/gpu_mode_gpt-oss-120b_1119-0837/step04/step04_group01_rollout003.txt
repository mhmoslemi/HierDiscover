    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head
    b = batch_head // H
    h = batch_head % H

    # offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # compute base pointers for current batch and head
    left_batch_ptr = left_ptr + b * stride_left_b + h * stride_left_h
    right_batch_ptr = right_ptr + b * stride_right_b + h * stride_right_h
    out_batch_ptr = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # load left block [i,k]
        left_block = tl.load(
            left_batch_ptr + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k,
            mask=i_mask[:, None] & k_mask[None, :],

        # ensure correct dtype for dot: dot supports half

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold‑3).  The implementation fuses as many operations as possible
and eliminates expensive layout conversions that were present in the
reference implementation.

Key ideas
---------
1️⃣  FP32 LayerNorm (first norm) is performed on a flattened (M, D) tensor.
2️⃣  All five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) are fused into a single FP16 GEMM.
3️⃣  Mask is applied after gating (element‑wise) – this is cheap.
4️⃣  The core N³ contraction is implemented as a custom Triton batched
    GEMM that works directly on the original layout (B, N, N, H).  This
    avoids the two costly ``permute+contiguous`` copies required by the
    naïve ``torch.bmm`` approach.
5️⃣  The second LayerNorm runs in FP16 (with FP32 accumulation) and the
    final projection brings the hidden representation back to the model
    dimension D.

The whole forward pass runs without gradients and returns a float32 tensor
with shape [B, N, N, D].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 2‑pass LayerNorm kernel (float32 input → float32 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) on a 2‑D tensor [M, N] using Triton."""
    assert x.dim() == 2, "expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------
# 2‑pass LayerNorm kernel (FP16 input → FP16 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_x_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (FP16) on a 2‑D tensor [M, N] using Triton."""
    assert x.dtype == torch.float16, "expects FP16 input"
    assert x.dim() == 2, "expects a 2‑D tensor"

    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

        eps=eps,

        stride_ym=stride_ym,

        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------
# Custom Triton kernel for the N³ contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    without any layout permutations.  Block dimensions are chosen to
    fit H100 tensor‑core requirements.
    """
    pid_bh = tl.program_id(0)        # combined batch * head index
    pid_i  = tl.program_id(1)        # block row (i)
    pid_j  = tl.program_id(2)        # block col (j)

    # --------------------------------------------------------------
    # Decode batch and head
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # Offsets for the i / j tiles
    # --------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # base pointers for the current batch & head
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑dimension loop (reduce over sequence length)
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left block: shape (BLOCK_M, BLOCK_K) -> indices (i, k)
        left = tl.load(
            left_base + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0
        )
        left = tl.cast(left, tl.float16)

        # right block (transposed on‑the‑fly): shape (BLOCK_K, BLOCK_N)
        # we load right[j, k] but store it as (k, j) to match GEMM semantics
        right = tl.load(
            right_base + k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_i,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0
        )
        right = tl.cast(right, tl.float16)

        # Matrix multiplication for this K‑slice
        acc += tl.dot(left, right)    # (BLOCK_M, BLOCK_N) accumulated in FP32

    # --------------------------------------------------------------
    # Write back the result (cast back to FP16)
    # --------------------------------------------------------------
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_base + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j,
        out,
        mask=i_mask[:, None] & j_mask[None, :]
    )


# --------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys “dim”, “hidden_dim”, optional “nomask”
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # float32 expected

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                # (M, D)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)  # (M, D)
    x_norm = x_norm_flat.view(bs, N, N, dim)                           # (B,N,N,D)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single GEMM (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # concatenate weight matrices: [left_proj; left_gate; right_proj; right_gate; out_gate] → (5H, D)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                     # (5*H, D)

    # (M, D) @ (D, 5H).T  →  (M, 5H)
    fused_out = F.linear(x_fp16, fused_weight)      # bias = None
    fused_out = fused_out.view(bs, N, N, 5 * H)      # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply sigmoid gates and mask (if required)
    # ------------------------------------------------------------------
    h0, h1, h2, h3, h4, h5 = 0, H, 2 * H, 3 * H, 4 * H, 5 * H

    left_proj   = fused_out[..., h0:h1]   # (B,N,N,H)
    left_gate_p = fused_out[..., h1:h2]
    right_proj  = fused_out[..., h2:h3]
    right_gate_p = fused_out[..., h3:h4]
    out_gate_p   = fused_out[..., h4:h5]

    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # element‑wise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # apply mask *after* gating (the order does not affect the result)
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction – custom Triton kernel (no permute/copy)
    # ------------------------------------------------------------------
    # Ensure contiguity for the kernel (the tensors are already contiguous after the
    # element‑wise ops, but we call .contiguous() for safety)
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = torch.empty_like(left)   # (B,N,N,H)

    # strides for the input tensors
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out_hidden.stride()

    # launch configuration
    BLOCK_M = 64   # i‑tile
    BLOCK_N = 64   # j‑tile
    BLOCK_K = 32   # reduction tile (chosen to match tensor‑core requirements)

    grid = (
        bs * H,                     # batch × hidden dimension
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=bs,
        N=N,
        H=H,
        stride_left_b=s_left_b,
        stride_left_i=s_left_i,
        stride_left_k=s_left_k,
        stride_left_h=s_left_h,
        stride_right_b=s_right_b,
        stride_right_i=s_right_i,
        stride_right_k=s_right_k,
        stride_right_h=s_right_h,
        stride_out_b=s_out_b,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        stride_out_h=s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out_hidden_flat = out_hidden.view(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(
        out_hidden_flat,
        to_out_norm_w,
        to_out_norm_b,
        eps=eps,
    )  # (M, H) in FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to the model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(M, H)           # (M, H)
    gated = out_norm * out_gate_flat               # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_proj = F.linear(gated, to_out_w)          # (M, D)

    # reshape back to 4‑D and cast to float32
    output = out_proj.view(bs, N, N, dim).to(torch.float32)
    return output
