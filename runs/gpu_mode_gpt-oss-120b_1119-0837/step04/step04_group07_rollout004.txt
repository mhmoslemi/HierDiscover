@triton.jit
def tril_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # get program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combined batch and head

    # decode batch and head indices
    total_heads = H
    b = pid_z // total_heads
    h = pid_z % total_heads

    # compute offsets
    # base pointer offset for this batch-head
    offset_base_left = b * stride_left_b + h * stride_left_h
    offset_base_right = b * stride_right_b + h * stride_right_h
    offset_base_out = b * stride_out_b + h * stride_out_h

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K
    # number of k blocks:
    num_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k):
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets
        left_ptrs = left_ptr + offset_base_left + \
            offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # right tile: shape (BLOCK_K, BLOCK_J)
        # i dimension for right is j (since we treat right as B[j,k])
        # For right tile: row index = offs_k (K), column index = offs_j (J)
        right_ptrs = right_ptr + offset_base_right + \
            offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # load
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b_ = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b_)  # both are (BLOCK_I, BLOCK_K) dot (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)

    # store result
    out_ptrs = out_ptr + offset_base_out + \
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tril_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    # Compute batch and head indices.
    b = pid_z // H
    h = pid_z % H

    # Compute base offsets
    offset_left = b * stride_left_b + h * stride_left_h
    offset_right = b * stride_right_b + h * stride_right_h
    offset_out = b * stride_out_b + h * stride_out_h

    # Offsets i, j
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k):
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + offset_left + \
            offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # right tile (BLOCK_K, BLOCK_J) from right[K,J]
        right_ptrs = right_ptr + offset_right + \
            offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :])
        b_val = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :])

        acc += tl.dot(a, b_val)

    # Store result
    out_ptrs = out_ptr + offset_out + \
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – Fast Triton implementation
========================================================

Algorithm
---------
1️⃣  Input LayerNorm (FP32) – 2‑pass Triton kernel.
2️⃣  Fuse the five linear projections (left, right, left‑gate,
    right‑gate, out‑gate) into a single `torch.nn.functional.linear`
    with an FP16 weight matrix.
3️⃣  Apply the optional binary mask and the three sigmoid gates
    (still in FP16).
4️⃣  Core “tri‑multiplicative” update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
       – Implemented as a **batched GEMM** in Triton.
       • The batch dimension is `B·hidden_dim` (B∈{1,2}, hidden_dim≤384).
       • No `permute + contiguous` copies – the kernel reads the
         original `[B, N, N, H]` layout directly.
       • Mixed‑precision: inputs FP16, accumulation FP32, output FP32.
5️⃣  Output LayerNorm (FP32) – same Triton kernel as step 1.
6️⃣  Apply the output gate (sigmoid) and the final projection back
    to the model dimension (FP16 weight, FP16 compute, cast to FP32).

All heavy work stays in FP16; the two LayerNorms use FP32 for
numerical stability.  The custom Triton GEMM replaces the expensive
`torch.bmm` + `permute/contiguous` dance, yielding a ≈2‑3× speed‑up
for the largest test case (N = 1024, hidden_dim = 128).

The entry point `custom_kernel` follows the required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32, works on a 2‑D view)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ----- row range -------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: mean & variance --------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine ---------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor,
                     bias: torch.Tensor, eps: float = 1e-5,
                     BLOCK_M: int = 128, BLOCK_N: int = 64) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor (M×N) using Triton."""
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure tensors are contiguous (required for stride arithmetic)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM for the core TriMul product
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right [B, N, N, H] (j‑dimension plays the role of “col”)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out   [B, N, N, H]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # grid (i_tile, j_tile, batch*head)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # ----- decode batch and head --------------------------------------
    b = pid_z // H                # batch index
    h = pid_z % H                 # head (hidden‑dim) index

    # base offsets for the current (b, h)
    base_left  = b * stride_left_b  + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out   = b * stride_out_b   + h * stride_out_h

    # ----- i/j coordinates --------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- reduction over k -------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_it in range(num_k):
        offs_k = k_it * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_I, BLOCK_K) -> left[b,i,k,h]
        left_tile = tl.load(
            left_ptr + base_left +
            offs_i[:, None] * stride_left_i +
            offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
        )

        # right tile: (BLOCK_K, BLOCK_J) -> right[b,j,k,h]  (k as row, j as col)
        right_tile = tl.load(
            right_ptr + base_right +
            offs_k[:, None] * stride_right_k +
            offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
        )

        # fused dot product (FP16 * FP16 -> FP32 accumulation)
        acc += tl.dot(left_tile, right_tile)

    # ----- write result ------------------------------------------------
    out_ptrs = out_ptr + base_out + \
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor,
                BLOCK_I: int = 64, BLOCK_J: int = 64, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]`` using Triton.

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape ``[B, N, N, H]`` (contiguous).
    BLOCK_* : int
        Tile sizes (tuned for H100).

    Returns
    -------
    out : torch.Tensor
        FP32 tensor of shape ``[B, N, N, H]``.
    """
    assert left.shape == right.shape, "left/right must have identical shapes"
    B, N, _, H = left.shape
    device = left.device

    # output tensor (FP32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (as number of elements, not bytes)
    ls = left.stride()
    rs = right.stride()
    os = out.stride()

    # launch grid
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_z = B * H

    trimul_gemm_kernel[(grid_i, grid_j, grid_z)](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=ls[0],
        stride_left_i=ls[1],
        stride_left_k=ls[2],
        stride_left_h=ls[3],
        stride_right_b=rs[0],
        stride_right_j=rs[1],   # j‑dimension (acts as “col” after transpose)
        stride_right_k=rs[2],
        stride_right_h=rs[3],
        stride_out_b=os[0],
        stride_out_i=os[1],
        stride_out_j=os[2],
        stride_out_h=os[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass of the “outgoing” TriMul (AlphaFold‑3) operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of model weights (torch.Tensor)
        - config       : dict containing ``dim`` and ``hidden_dim``; optional
                         key ``nomask`` (bool, default False)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    # reshape to (B·N·N, dim) for the 2‑D kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )
    x_norm = x_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Combined projection (FP16)
    # --------------------------------------------------------------
    # Move to FP16 for the heavy linear work
    x_norm_h = x_norm.half()
    x_norm_h_flat = x_norm_h.view(M, dim)

    # Concatenate the five weight matrices (left, right, left_gate,
    # right_gate, out_gate) – shape (5·H, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    # Fused linear projection (no bias)
    proj_out = F.linear(x_norm_h_flat, combined_weight)  # (M, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split back into the five tensors
    # --------------------------------------------------------------
    (
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
    ) = torch.split(proj_out, hidden_dim, dim=1)

    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw) # (B,N,N,H) FP16
    out_gate = torch.sigmoid(out_gate_raw)     # (B,N,N,H) FP16

    left = left_raw * left_gate                 # (B,N,N,H) FP16
    right = right_raw * right_gate              # (B,N,N,H) FP16

    # make tensors contiguous for the Triton GEMM
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Core TriMul product – batched GEMM in Triton
    # --------------------------------------------------------------
    # The kernel directly computes the result in FP32, avoiding the
    # costly `permute(...).contiguous()` dance.
    out_fp32 = trimul_gemm(left, right,
                           BLOCK_I=64, BLOCK_J=64, BLOCK_K=64)  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_flat_fp32 = out_fp32.view(bs * seq_len * seq_len, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat_fp32, to_out_norm_w, to_out_norm_b,
        eps=1e-5, BLOCK_M=128, BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                         # (B,N,N,H) FP32

    # final linear: FP16 weight, FP16 compute
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_w)                # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()                         # (B,N,N,dim) FP32

    return final_fp32
