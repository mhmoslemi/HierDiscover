"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32) – two‑pass Triton kernel.
2️⃣ Fuse the five linear projections (left, right and three gates) into a
   single half‑precision GEMM.
3️⃣ Split the projection result, optionally apply the binary mask (broadcast
   over the hidden dimension) and compute the three sigmoid gates.
4️⃣ Multiply left/right by their respective gates (in‑place, FP16).
5️⃣ Compute the triple‑multiplicative update with a single `torch.einsum`,
   which under the hood reduces to a batched GEMM but avoids the
   intermediate `permute`/`contiguous` copies required by a naïve bmm.
6️⃣ Output LayerNorm (FP32) – same Triton kernel as step 1.
7️⃣ Apply the output gate and the final half‑precision linear projection back to
   the model dimension.  The result is cast to FP32 as required by the API.

All heavy arithmetic is performed in FP16 (with FP32 accumulation inside the
CUDA kernels) while the two LayerNorms run in FP32 for numerical stability.
Only the LayerNorms are implemented in Triton, satisfying the requirement to
use a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # [M, N] input
    y_ptr,                # [M, N] output
    weight_ptr, bias_ptr, # [N]
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row offset for this program instance
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass: compute mean & variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: normalise + affine
    # --------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)

        norm_x = (xf - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_N: int = 64,
) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor (M × N) using Triton."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make tensors contiguous for pointer arithmetic
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Outgoing TriMul kernel (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    # Collapse the first three axes so we can run a 2‑D LayerNorm kernel
    M = bs * seq_len * seq_len                     # number of rows
    x_flat = input_tensor.reshape(M, dim)          # (M, dim) FP32
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection → 5 × hidden_dim channels (FP16)
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()                      # (B,N,N,dim) FP16
    x_norm_h_flat = x_norm_h.view(M, dim)          # (M, dim)

    # Stack all five weight matrices (each [hidden_dim, dim]) → [(5*H), dim]
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5*H, dim)

    # Linear: (M, dim) × (5*H, dim)^T → (M, 5*H)
    proj = F.linear(x_norm_h_flat, combined_weight)   # (M, 5*H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split projection into tensors of shape (B,N,N,H)
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view   (bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view  (bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates and gating of left/right (in‑place where possible)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)          # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)         # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)           # (B,N,N,H) FP16

    left  = left_raw  * left_gate                      # (B,N,N,H) FP16
    right = right_raw * right_gate                     # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Triple‑multiplicative update via a single einsum
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    This avoids the extra `permute`+`contiguous` copies required by a
    #    manual batched‑gemm implementation.
    # ------------------------------------------------------------------
    # einsum internally reduces to a series of GEMMs and is highly optimized.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    out_fp32 = out.float()                                   # (B,N,N,H) FP32
    out_flat_fp32 = out_fp32.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat_fp32,
        to_out_norm_w,
        to_out_norm_b,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.float()                # (B,N,N,H) FP32
    gated = out_norm * out_gate_fp32                # (B,N,N,H) FP32

    # Final linear projection (half‑precision weight, FP16 arithmetic)
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    gated_h = gated.half()
    final_h = F.linear(gated_h, to_out_w)          # (B,N,N,dim) FP16
    final_fp32 = final_h.float()                    # (B,N,N,dim) FP32

    return final_fp32
