    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape

    nomask = config.get("nomask", False)

    x_flat = input_tensor.reshape(bs * seq_len * seq_len, dim)
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp = triton_layernorm(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm_fp.view(bs, seq_len, seq_len, dim)

    # combined projection
    x_norm_h = x_norm.half()
    proj = F.linear(x_norm_h.view(-1, dim), combined_weight) # will be half

    # split
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(proj, hidden_dim, dim=1)
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # mask
    if (not nomask) and (mask_tensor is not None):

        left_raw.mul_(mask_h)
        right_raw.mul_(mask_h)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    left = left_raw * left_gate
    right = right_raw * right_gate
    out_gate = torch.sigmoid(out_gate_raw)

    # core contraction using triton
    out = torch.empty_like(left)
    # get strides (need stride for left/right)

    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out.stride()
    # they all have H stride == 1

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1️⃣ Input LayerNorm            – custom Triton kernel (FP32 mean/var, FP32 output).
2️⃣ Fuse the five linear layers (left/right projection + 3 gates) into a single
   half‑precision GEMM by concatenating their weight matrices.
3️⃣ Split the fused result, apply optional binary mask (broadcast over hidden dim)
   and element‑wise sigmoid gates.  The gated left/right tensors stay in FP16.
4️⃣ Core contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton “batched GEMM” that works directly on the
   (B,N,N,H) layout, thus avoiding the costly permute‑and‑contiguous copies that
   a naïve torch.bmm would need.  The kernel tiles the (i,j) plane (BLOCK_M×BLOCK_N)
   and accumulates in FP32 while the inputs are FP16 → Tensor‑Core friendly.
5️⃣ Output LayerNorm (FP32) – same Triton kernel as step 1.
6️⃣ Apply the output gate (sigmoid) and the final linear projection back to the
   model dimension (half‑precision weight, final FP32 result).

Only the LayerNorm and the contraction are implemented in Triton, satisfying
the requirement that at least one operation runs in a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (FP32 mean/var, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output [M, C]
    w_ptr, b_ptr,               # *ptr to weight / bias  [C]
    M, C, eps,                  # dimensions
    stride_xm, stride_xc,       # strides of input
    stride_ym, stride_yc,       # strides of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # row range for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalisation + affine transform
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_C: int = 64) -> torch.Tensor:
    """Layer‑norm over the last dimension of a 2‑D tensor (M×C) using Triton."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) core contraction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,       # fp16 pointer, shape (B,N,N,H)
    right_ptr,      # fp16 pointer, shape (B,N,N,H)
    out_ptr,        # fp16 pointer, shape (B,N,N,H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k,   # strides for left  (b, i, k, h)
    stride_r_b, stride_r_i, stride_r_k,   # strides for right (b, j, k, h)
    stride_o_b, stride_o_i, stride_o_j,   # strides for out   (b, i, j, h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]."""
    pid_i = tl.program_id(0)               # tile along i
    pid_j = tl.program_id(1)               # tile along j
    pid_bh = tl.program_id(2)              # combined batch‑head index

    # ---- decode batch and head ---------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ---- base pointers for this (b,h) pair --------------------------
    left_base  = left_ptr  + b * stride_l_b + h
    right_base = right_ptr + b * stride_r_b + h
    out_base   = out_ptr   + b * stride_o_b + h

    # ---- offsets for the current output tile -------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over the reduction dimension k ----------------------
    for kk in range(tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M × K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # right tile transposed: (K × N)
        right = tl.load(
            right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # accumulate
        acc += tl.dot(left, right)   # acc is FP32

    # ---- write back -------------------------------------------------
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point: forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from parameter name to torch.Tensor (float32)
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * seq_len * seq_len, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp = triton_layernorm(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm_fp.view(bs, seq_len, seq_len, dim)   # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single GEMM (FP16)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()
    x_norm_h_flat = x_norm_h.view(-1, dim)               # (B*N*N, dim)

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                    # (5*H, dim)

    proj = F.linear(x_norm_h_flat, combined_weight)      # (B*N*N, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split and reshape back to (B,N,N,H)
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep everything in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw * left_gate          # (B,N,N,H) FP16
    right = right_raw * right_gate        # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Core contraction using a custom Triton kernel (no permute copy)
    # --------------------------------------------------------------
    out = torch.empty_like(left)          # (B,N,N,H) FP16

    # strides for the (B,N,N,H) layout (contiguous)
    s_b_l, s_i_l, s_k_l, s_h_l = left.stride()
    s_b_r, s_i_r, s_k_r, s_h_r = right.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),          # i‑tiles
        triton.cdiv(seq_len, BLOCK_N),          # j‑tiles
        bs * hidden_dim,                        # batch‑head dimension
    )

    triton_trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_l_b=s_b_l,
        stride_l_i=s_i_l,
        stride_l_k=s_k_l,
        stride_r_b=s_b_r,
        stride_r_i=s_i_r,
        stride_r_k=s_k_r,
        stride_o_b=s_b_o,
        stride_o_i=s_i_o,
        stride_o_j=s_j_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp = out.float()
    out_flat = out_fp.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_C=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to ``dim``
    # --------------------------------------------------------------
    gated = out_norm * out_gate.float()          # (B,N,N,H) FP32
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_fp16 = F.linear(gated.half(), to_out_weight)   # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()

    return final_fp32
