def custom_kernel(data):

    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # input LN
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # Cast to half for rest
    x_h = x_norm.half()

    # combined weights
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device, torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    # linear fused
    left_comb = F.linear(x_h, left_comb_w)      # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)    # (B,N,N,2H)

    # split
    left_proj = left_comb[..., :H]
    left_gate_pre = left_comb[..., H:]
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # apply mask to projections if needed
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # output gate
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # N^3 contraction using batched gemm
    left_perm = left.permute(0,3,1,2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0,3,1,2).contiguous()
    BH = bs * H

"""
Trion kernel-accelerated TriMul "outgoing" operator.

Algorithm:
1. LayerNorm over the model dimension (float32) with a two‑pass Triton kernel.
2. Cast to FP16 for the remaining heavy work.
3. Fuse projection and gate linear transforms by concatenating their weights and
   performing a single half‑precision matmul; split the result into projection
   and gate tensors.
4. Optionally apply the input binary mask (broadcast over the hidden dim).
5. Element‑wise gating of left/right projected tensors.
6. Compute the N³ contraction with a batched half‑precision GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
7. Hidden‑dimension LayerNorm (float16) using a second Triton kernel.
8. Apply the output gate and a final half‑precision linear projection back to the
   model dimension.
9. Cast the result to float32 and return.

Only the two LayerNorm steps are written in Triton; all matrix multiplies are
performed with torch.bmm / torch.nn.functional.linear which on an H100 map
to highly‑optimized Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # first pass: compute sum and sum‑squared over feature dimension N
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized result
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton kernel. Input shape (M, N)."""
    assert x.dim() == 2, "Input must be 2‑D"
    M, N = x.shape
    y = torch.empty_like(x)

    # make strides explicit and contiguous
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# 2‑pass LayerNorm – float16 → float16
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) via Triton kernel. Input shape (M, H)."""
    assert x.dtype == torch.float16, "expected half‑precision tensor"
    assert x.dim() == 2, "expects 2‑D"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

        weight_ptr=weight,

        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,

        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32/bool) or None
        - weights      : dict mapping module parameter names to tensors
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)   # (B·N·N, D)

    # LayerNorm parameters (float32)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)  # (B, N, N, D) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy ops
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear + gate for left/right sides
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)   # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)   # (2H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Combined matmuls – shape (..., 2H)
    left_comb  = F.linear(x_h, left_comb_w)   # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)  # (B,N,N,2H)

    # Split projection and gate streams
    left_proj = left_comb[..., :H]          # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]      # (B,N,N,H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional binary mask to the projections (before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool or floating point; cast to half and broadcast
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction: batched GEMM in FP16
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)     # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)    # (B·H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)   # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # half

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H) half
    gated = out_norm * out_gate_flat                     # (M, H) half

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)             # (M, D) half

    output = output_flat.view(bs, N, N, dim).float()    # return float32

    return output

    assert x.dim() == 2, "expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride extraction
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float16 → float16
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) via Triton kernel.  Input shape (M, H)."""
    assert x.dtype == torch.float16, "expected half‑precision tensor"
    assert x.dim() == 2, "expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 / bool) or None
        - weights      : dict mapping parameter names to tensors
        - config       : dict containing ``dim`` and ``hidden_dim`` (optional ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype  # expect torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                     # (B·N·N, D)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                  # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the remaining work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left/right sides
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                               # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                               # (2H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Single matmul for each side (produces proj & gate concatenated)
    left_comb  = F.linear(x_h, left_comb_w)    # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)   # (B,N,N,2H)

    # Split into projection and gate streams
    left_proj      = left_comb[..., :H]          # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]          # (B,N,N,H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)    # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask applied to the projections (before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool or floating; cast to half and broadcast
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched FP16 GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)          # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)         # (B·H, N, N)

    # Batched matmul: left @ rightᵀ  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)                     # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # half

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)           # (M, H) half
    gated = out_norm * out_gate_flat                 # (M, H) half

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)          # (M, D) half

    output = output_flat.view(bs, N, N, dim).float()  # final FP32 tensor

    return output
