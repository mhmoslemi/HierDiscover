import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(
    left_raw_ptr,               # fp16 * B*N*N*H
    right_raw_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,                   # fp16 * B*N*N
    H: tl.constexpr,            # hidden dimension
    total_elem: tl.constexpr,   # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):

    mask = offs < total_elem

    right_gate_h = right_gate.to(tl.float16)

    tl.store(left_raw_ptr + offs, out_left, mask=mask)
    tl.store(right_raw_ptr + offs, out_right, mask=mask)

    # Split into the five components
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(

        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,

        eps=1e-5,
    )   # fp16

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32) using the highly‑optimised torch.nn.functional.layer_norm.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single fp16 GEMM by concatenating their weight matrices.
3️⃣ Apply mask, sigmoid‑gates and the element‑wise multiplication in‑place with a
   custom Triton kernel (`gate_mask_kernel`).  This fuses three separate PyTorch
   kernels (mask, sigmoid, multiply) into one pass over memory.
4️⃣ Core contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched half‑precision GEMM via `torch.bmm`, which maps to
   cuBLAS/Tensor‑Core on H100.
5️⃣ Output LayerNorm (still fp16) – uses torch’s native kernel (still fast) and
   keeps everything in half precision.
6️⃣ Apply the output gate, final linear projection back to the model dimension,
   and cast the result to float32 for the caller.

Only the gating+mask step is custom‑written in Triton, satisfying the “must
contain a Triton kernel” requirement while keeping the overall runtime well
below the original ~3 ms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    left_raw_ptr,          # fp16 * (B*N*N*H)
    right_raw_ptr,         # fp16 * (B*N*N*H)
    left_gate_ptr,         # fp16 * (B*N*N*H)
    right_gate_ptr,        # fp16 * (B*N*N*H)
    mask_ptr,              # fp16 * (B*N*N)   broadcasted over H
    H: tl.constexpr,       # hidden dimension (size of mask broadcast)
    TOTAL_ELEM: tl.constexpr,   # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,   # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEM

    # load raw values
    left_raw  = tl.load(left_raw_ptr + offs,  mask=mask, other=0.0)
    right_raw = tl.load(right_raw_ptr + offs, mask=mask, other=0.0)

    left_gate_raw  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcast over the hidden dimension: each H consecutive elements share one mask entry
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # sigmoid in fp32 (more stable) then cast back to fp16
    lg_f32 = left_gate_raw.to(tl.float32)
    rg_f32 = right_gate_raw.to(tl.float32)

    left_gate = 1.0 / (1.0 + tl.exp(-lg_f32))
    right_gate = 1.0 / (1.0 + tl.exp(-rg_f32))

    left_gate_h  = left_gate.to(tl.float16)
    right_gate_h = right_gate.to(tl.float16)

    # apply mask and gates
    out_left  = left_raw  * left_gate_h  * mask_val
    out_right = right_raw * right_gate_h * mask_val

    # write back in‑place
    tl.store(left_raw_ptr + offs,  out_left,  mask=mask)
    tl.store(right_raw_ptr + offs, out_right, mask=mask)
