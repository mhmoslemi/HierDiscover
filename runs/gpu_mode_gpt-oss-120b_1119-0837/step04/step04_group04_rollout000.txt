"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the reference `TriMul` module but replaces the
two `LayerNorm`s and the cubic contraction with custom Triton kernels:

* `layernorm_kernel`  – Row‑wise LayerNorm (mean/var reduction in FP32,
  normalisation & affine transform in FP16).  Operates on a 2‑D view
  [M, C].

* `batched_matmul_kernel` – Computes the N³ contraction
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
  as a *batched* GEMM (left @ rightᵀ) for all heads in a single launch.
  The kernel works on tensors of shape (batch_h, M, K) and (batch_h, K, N),
  using Tensor‑core `tl.dot` (FP16 × FP16 → FP32) with FP32 accumulation.

The heavy compute is performed in FP16 for speed, while the two
LayerNorms use FP32 accumulators for numerical stability.  The final
output is cast back to FP32 as required by the model.

The code works for all configurations described in the problem statement
(batch ≤ 2, sequence length ∈ {256, 512, 768, 1024},
`dim` ∈ {128, 384}, `hidden_dim` = 128) and respects optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, C]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, C]
    y_ptr,                # *ptr to output [M, C]
    weight_ptr,           # *ptr to weight [C]
    bias_ptr,             # *ptr to bias   [C]
    M,                    # total rows
    C,                    # feature dim
    eps,                  # epsilon
    stride_xm, stride_xc,  # strides for input
    stride_ym, stride_yc,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        # (x - mean) * rstd  (broadcast mean/rstd over columns)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm via Triton.  ``x`` must be a 2‑D tensor."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity for correct strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=128,   # rows per program (tuned for H100)
        BLOCK_C=64,    # columns per tile
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel (the N³ contraction).
# Computes for each head h:
#   out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_h: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_ah: tl.constexpr, stride_am: tl.constexpr, stride_ak: tl.constexpr,
    stride_bh: tl.constexpr, stride_bk: tl.constexpr, stride_bn: tl.constexpr,
    stride_ch: tl.constexpr, stride_cm: tl.constexpr, stride_cn: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 3‑D launch: (batch_h, ceil(M/BM), ceil(N/BN))
    # --------------------------------------------------------------
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K in blocks
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a block of A (M×K) and B (K×N)
        a = tl.load(
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back FP16 result
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A: (batch_h, M, K)  FP16
    B: (batch_h, K, N)  FP16
    Returns C: (batch_h, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch_h, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch_h, M, N), dtype=A.dtype, device=A.device)

    # Tile sizes tuned for H100 FP16 Tensor‑cores.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch_h,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch_h=batch_h,
        M=M,
        N=N,
        K=K,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,

    return C

# Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]  or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (via Triton) over the last dimension ``dim``
    # --------------------------------------------------------------
    # Flatten to [M, dim] where M = bs * L * L
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim).contiguous()  # [B, L, L, D]

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy compute
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse four linear maps (proj + gates) into a single matmul
    # --------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    # Concatenate weights: shape (4*hidden_dim, dim)
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)  # (4H, D)

    # Apply the fused projection to the flattened tensor
    fused_out = F.linear(x_norm_flat, fused_w)                     # (M, 4H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)          # (B, L, L, 4H)

    left_proj   = fused_out[..., :hidden_dim]                                  # (B, L, L, H)
    right_proj  = fused_out[..., hidden_dim:2 * hidden_dim]                    # (B, L, L, H)
    left_gate   = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])# (B, L, L, H)
    right_gate  = torch.sigmoid(fused_out[..., 3 * hidden_dim:])              # (B, L, L, H)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if any) before the element‑wise gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B, L, L, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply gates
    left  = left_proj * left_gate     # (B, L, L, H)
    right = right_proj * right_gate   # (B, L, L, H)

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Bring the hidden dimension next to the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, L, L)

    # For the RHS we need (K, N) = (seq_len, seq_len) order
    right_t = right_perm.transpose(2, 3).contiguous()   # (B, H, L, L) -> (B, H, K, N)

    # Collapse batch & heads into a single dimension for the kernel
    batch_h = bs * hidden_dim
    left_mat  = left_perm.reshape(batch_h, L, L)   # (batch_h, M, K)
    right_mat = right_t.reshape(batch_h, L, L)    # (batch_h, K, N)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM (Trion kernel) – replaces torch.bmm
    # --------------------------------------------------------------
    out_mat = batched_matmul_triton(left_mat, right_mat)   # (batch_h, L, L), FP16

    # Restore the original layout: (B, L, L, H)
    out = (out_mat.view(bs, hidden_dim, L, L)
                   .permute(0, 2, 3, 1)
                   .contiguous())  # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) – linear on the *original* normed input
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h, out_gate_w))  # (M, H)
    out_gate = out_gate.view(bs, L, L, hidden_dim)           # (B, L, L, H)

    # --------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to ``dim``
    # --------------------------------------------------------------
    gated = out_norm * out_gate                 # (B, L, L, H) FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated, to_out_w)     # (B, L, L, D) FP16
    out_final = out_final_h.to(torch.float32)   # Cast back to float32

    return out_final
