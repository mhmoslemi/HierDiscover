"""
TriMul “outgoing’’ forward pass – Triton‑accelerated implementation.

The kernel performs the following steps:
1️⃣  Layer‑Norm the (B,N,N,D) input (Triton 2‑pass kernel, FP32 stats,
    result kept in FP16).
2️⃣  Fuse the five linear layers (left/right projections + three gates) into
    a single half‑precision GEMM.
3️⃣  Apply sigmoid to the gate tensors, optional pair‑wise mask and compute the
    gated left/right tensors.
4️⃣  Compute the heavy O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    by reshaping to (B·H,N,N) and using a batched FP16 GEMM (torch.bmm).
5️⃣  Layer‑Norm the intermediate result (Triton kernel, FP16).
6️⃣  Apply the output gate, a final FP16 linear projection back to the model
    dimension and cast the answer to FP32 (public API expectation).

All heavy linear algebra (GEMM, batched GEMM) leverages cuBLAS/Tensor‑Core
kernels, while the two LayerNorms are written in Triton for full‑GPU
fusion and reduced memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension). Supports any dtype (we use
# fp16 for the output but compute mean/var in fp32 for stability).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # number of rows   (B·N·N)
    N,                # feature dimension (D or H)
    eps,              # epsilon
    stride_xm, stride_xn,      # strides of x
    stride_ym, stride_yn,      # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- row offsets for this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute sum and sum‑sq (FP32) ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine -------------------------------
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (weight/bias may be fp16/ fp32 – cast to fp32)
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for the Triton LayerNorm kernel.
    ``x`` must be a 2‑D contiguous tensor (M, N) and the result has the same
    dtype as ``x``.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # layout / strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes chosen to work for both D=128 and D=384/768
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]  (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor  [B, N, N, D] (float32)
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton).  The kernel works on a 2‑D view.
    # -----------------------------------------------------------------------
    M = bs * seq_len * seq_len               # rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)    # (M, D)  FP32

    norm_w = weights["norm.weight"].to(device, dtype)   # (D)
    norm_b = weights["norm.bias"].to(device, dtype)     # (D)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) FP32

    # -----------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision (fp16) – the heavy work runs in fp16.
    # -----------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)          # (M, D) fp16

    # -----------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (left/right proj + 3 gates) into a
    #    single GEMM to reduce memory traffic.
    # -----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]           # (H, D)
    right_proj_w  = weights["right_proj.weight"]          # (H, D)
    left_gate_w   = weights["left_gate.weight"]           # (H, D)
    right_gate_w  = weights["right_gate.weight"]          # (H, D)
    out_gate_w    = weights["out_gate.weight"]            # (H, D)

    # Concatenate along the output‑feature axis → shape (5·H, D)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                 # (5*H, D)

    # Single fused linear projection
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)   # (M, 5·H) fp16

    # -----------------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor, apply sigmoid to gates and optional mask.
    # -----------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)      # (B,N,N,H) fp16
    right_gate  = torch.sigmoid(right_gate_r)     # (B,N,N,H) fp16
    out_gate    = torch.sigmoid(out_gate_r)       # (B,N,N,H) fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    – reshape to (B·H, N, N) and invoke torch.bmm (Tensor‑Core fp16).
    # -----------------------------------------------------------------------
    # Move the hidden dimension to the batch axis for bmm.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim                                 # B·H
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Restore original layout: (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H)

    # -----------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton).  We stay in fp16 throughout.
    # -----------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                 # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)   # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)     # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # -----------------------------------------------------------------------
    # 7️⃣ Apply the output gate, final projection and cast back to FP32.
    # -----------------------------------------------------------------------
    gated = out_norm * out_gate                         # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    # final linear (fp16) → (B,N,N,D)
    out_fp16 = F.linear(gated, to_out_w)                         # fp16
    out_fp32 = out_fp16.to(torch.float32)                        # required FP32 output

    return out_fp32
