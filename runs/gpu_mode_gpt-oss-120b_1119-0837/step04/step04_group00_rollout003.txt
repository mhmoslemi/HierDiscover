"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated.

Key ideas
---------
1️⃣  Input LayerNorm (float‑32) is performed with a custom Triton kernel.
2️⃣  All heavy linear work is kept in float‑16 (Tensor‑Core friendly).
3️⃣  The three separate linear projections (left‑proj, right‑proj, out‑gate)
    are fused into a single GEMM that produces a concatenated tensor of shape
    …×5·H.  This eliminates two extra reads of the large input tensor.
4️⃣  Optional binary mask is applied only to the projected tensors (before gating).
5️⃣  Gating uses a sigmoid on the split “gate” sub‑tensors.
6️⃣  The N³ contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is executed as a batched half‑precision GEMM via `torch.bmm`.
7️⃣  Hidden‑dim LayerNorm (float‑16) is done with a second custom Triton kernel.
8️⃣  An output gate (sigmoid of a linear) is applied, followed by the final
    linear projection back to the model dimension.
9️⃣  The final tensor is cast back to float‑32 for the return value.

Only the two small LayerNorm kernels are implemented in Triton; all other
operations are composed from fast PyTorch kernels while preserving the
correct semantics.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -------- second pass: write normalized ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton.

    Input/Output shape: (M, N) where N is the feature dimension.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def triton_layernorm_fp16(x: torch.Tensor,

                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) via Triton.

    Input shape (M, H) where H is the hidden dimension.
    """
    assert x.dtype == torch.float16, "FP16 layernorm expects half‑precision input"
    assert x.dim() == 2, "expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (float32 / bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys:
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input tensor must be square in the sequence dimensions"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – custom Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim).contiguous()        # (M, D)
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                 # (B, N, N, D)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the remaining heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse 3 linear projections (left, right, out_gate) into one GEMM
    # ------------------------------------------------------------------
    # Gather weights (all are (H, D) matrices – bias=False in the reference)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device, torch.float16).contiguous()                # (5*H, D)

    # fused linear: (M, D) @ (D, 5*H).T  → (M, 5*H)
    fused = F.linear(x_f16, fused_weight)                    # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 4️⃣ Split back to individual tensors
    # ------------------------------------------------------------------
    # shape helpers
    H = config["hidden_dim"]
    # slices along the last dimension
    left_proj      = fused[..., 0               : H]
    left_gate_pre  = fused[..., H               : 2*H]
    right_proj     = fused[..., 2*H             : 3*H]
    right_gate_pre = fused[..., 3*H             : 4*H]
    out_gate_pre   = fused[..., 4*H             : 5*H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 7️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)               # (B·H, N, N)

    # out_mat = left_mat @ right_matᵀ   → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)  (still FP16)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Hidden‑dim LayerNorm (FP16) – custom Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 🔟 Apply output gate & final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H).contiguous()
    gated = out_norm * out_gate_flat

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()  # (D, H)
    output_flat = F.linear(gated, to_out_w)  # (M, D), half‑precision

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32 tensor
    return output
