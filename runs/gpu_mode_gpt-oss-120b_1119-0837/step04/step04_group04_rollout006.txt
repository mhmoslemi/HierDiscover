def batched_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH: tl.constexpr, HEAD: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_A_batch, stride_A_head, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_head, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_head, stride_C_i, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch_idx and head_idx
    batch_idx = pid // HEAD
    head_idx = pid % HEAD

    # compute offsets for this block
    # offset for the head slice base
    a_base = batch_idx * stride_A_batch + head_idx * stride_A_head
    b_base = batch_idx * stride_B_batch + head_idx * stride_B_head
    c_base = batch_idx * stride_C_batch + head_idx * stride_C_head

    # row/col offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but replaces the N³
contraction with a custom Triton kernel that works on the strided view of the
projected tensors, thereby eliminating the expensive copies caused by
`.permute(...).contiguous()`.  All heavy arithmetic is performed in fp16 while
LayerNorm reductions use fp32 for numeric stability.

Key steps
---------
1️⃣ Input LayerNorm (row‑wise) on the last dimension.
2️⃣ Cast to fp16 for all subsequent ops.
3️⃣ Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
   into a single matmul.
4️⃣ Apply optional binary mask and the sigmoid gates.
5️⃣ Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
   Triton kernel that accesses the left/right tensors via strided views,
   avoiding any extra transposes / copies.
6️⃣ Output LayerNorm over the hidden dimension.
7️⃣ Output gating (sigmoid) from a linear projection of the normalized input.
8️⃣ Final linear projection back to the original feature dimension.
9️⃣ Cast the result to float32 (the model’s required dtype).

The kernel launches a single grid over all (batch × hidden) slices,
so the launch overhead is negligible even for many heads.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (operates on a 2‑D tensor [M, N])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (accumulate in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm via Triton. ``x`` must be 2‑D."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride extraction
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel (N³ contraction)
# ----------------------------------------------------------------------
@triton.jit
def batched_tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    HEAD: tl.constexpr,               # hidden dimension
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_A_batch, stride_A_head,
    stride_A_i, stride_A_k,
    stride_B_batch, stride_B_head,
    stride_B_k, stride_B_n,
    stride_C_batch, stride_C_head,
    stride_C_i, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # combined batch * head index
    pid_i = tl.program_id(1)    # row block
    pid_j = tl.program_id(2)    # col block

    # --------------------------------------------------------------
    # Decode the combined index into batch and head components
    # --------------------------------------------------------------
    batch_idx = pid // HEAD
    head_idx  = pid % HEAD

    # Base pointers for this (batch, head) slice
    a_base = batch_idx * stride_A_batch + head_idx * stride_A_head
    b_base = batch_idx * stride_B_batch + head_idx * stride_B_head
    c_base = batch_idx * stride_C_batch + head_idx * stride_C_head

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction axis K
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            A_ptr
            + a_base
            + offs_i[:, None] * stride_A_i
            + offs_k[None, :] * stride_A_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + b_base
            + offs_k[:, None] * stride_B_k
            + offs_j[None, :] * stride_B_n,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back (store as fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + c_base
        + offs_i[:, None] * stride_C_i
        + offs_j[None, :] * stride_C_n,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def batched_tri_mul(A: torch.Tensor, B: torch.Tensor,
                    hidden_dim: int) -> torch.Tensor:
    """
    Computes, for each head `h`:
        out[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]

    A, B : tensors of shape (B, L, L, hidden_dim) in fp16.
    Returns C of the same shape, fp16.
    """
    assert A.shape == B.shape
    bs, L, _, H = A.shape
    assert H == hidden_dim

    # --------------------------------------------------------------
    # Create strided views:
    #   left  -> (B, H, L, L)   (i,k)
    #   right -> (B, H, L, L)   (k,j)  (note the swapped axes)
    # --------------------------------------------------------------
    left_view = A.permute(0, 3, 1, 2)          # (B, H, L, L)  i,k
    right_view = B.permute(0, 3, 2, 1)         # (B, H, L, L)  k,j

    # --------------------------------------------------------------
    # Output tensor (B, H, L, L) – stored contiguously
    # --------------------------------------------------------------
    out = torch.empty((bs, hidden_dim, L, L),
                      dtype=torch.float16,
                      device=A.device)

    # --------------------------------------------------------------
    # Extract strides for the three tensors
    # --------------------------------------------------------------
    sA_batch, sA_head, sA_i, sA_k = left_view.stride()
    sB_batch, sB_head, sB_k, sB_n = right_view.stride()
    sC_batch, sC_head, sC_i, sC_n = out.stride()

    # --------------------------------------------------------------
    # Launch the Triton kernel
    # --------------------------------------------------------------
    BLOCK = 64
    grid = (
        bs * hidden_dim,               # combined (batch * head)
        triton.cdiv(L, BLOCK),        # row blocks
        triton.cdiv(L, BLOCK),        # col blocks
    )
    batched_tri_mul_kernel[grid](
        A_ptr=left_view,
        B_ptr=right_view,
        C_ptr=out,
        HEAD=hidden_dim,
        M=L,
        N=L,
        K=L,
        stride_A_batch=sA_batch,
        stride_A_head=sA_head,
        stride_A_i=sA_i,
        stride_A_k=sA_k,
        stride_B_batch=sB_batch,
        stride_B_head=sB_head,
        stride_B_k=sB_k,
        stride_B_n=sB_n,
        stride_C_batch=sC_batch,
        stride_C_head=sC_head,
        stride_C_i=sC_i,
        stride_C_n=sC_n,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
    )
    # --------------------------------------------------------------
    # Convert back to shape (B, L, L, H) for the rest of the pipeline
    # --------------------------------------------------------------
    return out.permute(0, 2, 3, 1).contiguous()   # (B, L, L, H)


# ----------------------------------------------------------------------
# Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (via Triton) over the last dimension ``dim``
    # --------------------------------------------------------------
    # Cast to half‑precision before LayerNorm to save memory.
    inp_fp16 = input_tensor.to(torch.float16)
    inp_flat = inp_fp16.reshape(bs * L * L, dim)          # [M, dim]
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    norm_flat = triton_layernorm(inp_flat, norm_w, norm_b, eps=eps)
    x_norm = norm_flat.view(bs, L, L, dim)               # [B, L, L, D]

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear maps (proj / gate) into a single matmul
    # --------------------------------------------------------------
    # Gather the four projection matrices.
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    fused_w = torch.cat([left_proj_w, right_proj_w,
                         left_gate_w, right_gate_w], dim=0)   # (4*H, D)

    # Linear projection on the flattened tensor
    x_norm_flat = x_norm.reshape(bs * L * L, dim)       # (M, D)
    fused_out = F.linear(x_norm_flat, fused_w)          # (M, 4*H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # Slice the four components
    left_proj = fused_out[..., :hidden_dim]                         # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # --------------------------------------------------------------
    # 3️⃣ Apply optional binary mask (before the sigmoid gates)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)      # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Gated values (still fp16)
    # --------------------------------------------------------------
    left = left_proj * left_gate      # (B, L, L, H)
    right = right_proj * right_gate   # (B, L, L, H)

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM (the N³ contraction)
    # --------------------------------------------------------------
    # out shape: (B, L, L, H)  – fp16
    out = batched_tri_mul(left, right, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.reshape(bs * L * L, hidden_dim)   # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)    # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) – linear on the *input* (not masked)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)       # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Apply gating and final projection back to original dim D
    # --------------------------------------------------------------
    gated = out_norm * out_gate                              # fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_fp16 = F.linear(gated, to_out_w)               # (B, L, L, D)
    out_final = out_final_fp16.to(torch.float32)             # required dtype

    return out_final
