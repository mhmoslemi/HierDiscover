Optimisations
-------------
1. Input LayerNorm – torch.nn.functional.layer_norm (highly‑optimised, no grads).
2. All heavy arithmetic is performed in float16.
3. Linear projections and their gating weights are concatenated so that the
   two linear layers (proj + gate) for the left and right branches are fused
   into a single matrix‑multiply, halving the number of kernel launches.
4. The O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is implemented as a custom Triton batched GEMM (kernel `tri_mul_gemm_kernel`).
   The kernel works directly on the native (B,N,N,H) layout, avoiding the
   expensive permute‑and‑reshape that the reference implementation needs.
5. Output LayerNorm (+ optional gating) is performed by a row‑wise Triton
   kernel (`layernorm_kernel`).  The kernel reduces each row in two passes,
   computing mean/variance then normalising + affine transform.
6. The final linear projection back to the model dimension D uses a single
   torch linear call.  The result is cast back to float32 for the API contract.

Overall the implementation reduces memory traffic (no extra copies),
fuses several matrix‑multiplies, and keeps the only remaining Python‑side
operations to lightweight element‑wise kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (used for the output norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # input  (M, N)
    y_ptr,   # output (M, N)
    w_ptr,   # weight (N,)
    b_ptr,   # bias   (N,)
    M,
    N,
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)               # row block id

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise, apply affine transform and store
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        norm_x = (xf - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w + b
        y = y.to(x.dtype)      # back to the input dtype (fp16/fp32)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    Input must be 2‑D (M, N); output has the same shape & dtype.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for H ≤ 128
    BLOCK_M = 256
    BLOCK_N = 128
    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,            # fp16, shape (B, N, N, H)
    right_ptr,           # fp16, shape (B, N, N, H)
    out_ptr,             # fp16, shape (B, N, N, H)
    B, N, H,
    # strides for left (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # rows  of A / out[i,:]
    BLOCK_N: tl.constexpr,   # cols  of B / out[:,j]
    BLOCK_K: tl.constexpr,   # reduction axis (k)
):
    # program ids: batch (b*h), block row, block col
    pid_batch = tl.program_id(0)          # 0 .. B*H-1
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # Decode batch and hidden index from flattened batch dimension
    b = pid_batch // H
    h = pid_batch % H

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A‑tile: left[b, i, k, h]  (M × K)
        a = tl.load(
            left_base
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Load B‑tile: right[b, j, k, h] transposed to (K × N)
        # (row = k, col = j)
        b_tile = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # Accumulate the block product
        acc += tl.dot(a, b_tile)

    # Write the result tile to out[b,i,j,h]
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    tl.store(
        out_base
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]     (float32) or None
        - weights      : dict of torch tensors containing the model parameters
        - config       : dict containing ``dim`` (model dim D) and
                         ``hidden_dim`` (H).  ``nomask`` (bool) is optional.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch implementation – fast)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse proj+gate for left and right branches
    # ------------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, D)

    # right side
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, D)

    # Combined linear pass (B, L, L, 2*H)
    left_comb = F.linear(x_h, left_comb_w)   # (B,L,L,2*H)
    right_comb = F.linear(x_h, right_comb_w)

    # Split projections and raw gates
    H = hidden_dim
    left_proj = left_comb[..., :H]        # (B,L,L,H)
    left_gate_raw = left_comb[..., H:]    # (B,L,L,H)
    right_proj = right_comb[..., :H]
    right_gate_raw = right_comb[..., H:]

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) prior to gating
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_fp16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate     # (B, L, L, H)
    right = right_proj * right_gate  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ N³ contraction – custom Triton batched GEMM
    # ------------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left)    # (B, L, L, H), fp16

    # Strides for left (B,N,N,H) layout -> (b,i,k,h)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    # Strides for right (b,j,k,h)
    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    # Strides for output (b,i,j,h)
    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Block sizes (tuned for H ≤ 128, N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * H,                     # batch × hidden dimension
        triton.cdiv(L, BLOCK_M),   # row blocks
        triton.cdiv(L, BLOCK_N),   # col blocks
    )

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=L,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (row‑wise) + output gate
    # ------------------------------------------------------------------
    # Flatten to (B*L*L, H) for the row‑wise LayerNorm kernel
    out_flat = out.reshape(bs * L * L, H).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, H)

    # Apply output gate (element‑wise, still fp16)
    out = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to model dimension D (float16 → float32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)          # (B, L, L, D) in FP16
    out = out.to(torch.float32)            # required output dtype

    return out
