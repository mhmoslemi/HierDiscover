@triton.jit
def trigemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute starting offset for this batch
    batch_offset_l = batch_id * stride_l_batch
    batch_offset_r = batch_id * stride_r_batch
    batch_offset_o = batch_id * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A (left)
        a_ptr = left_ptr + batch_offset_l + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B transposed (right)
        # B's shape is (N, N), we need rows = offs_k, cols = offs_n
        b_ptr = right_ptr + batch_offset_r + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store results
    out_ptrs = out_ptr + batch_offset_o + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. Input LayerNorm – use highly‑optimised torch.nn.functional.layer_norm.
2. All heavy compute is performed in float16.
3. Linear projections and gating are kept in PyTorch (fast BLAS + fused kernels).
4. The N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realized as a *batched* matrix‑multiply (A @ Bᵀ) where the batch
   dimension B·H is fused into a single Triton kernel.  The kernel
   processes tiles of size (BLOCK_M, BLOCK_N) and iterates over K in
   blocks (BLOCK_K).  Accumulation is done in fp32 and the result is stored
   back in fp16, halving memory traffic.
5. Output LayerNorm + bias (hidden_dim) is implemented in Triton
   (row‑wise two‑pass reduction).  The gate `out_gate` is applied afterwards.
6. Final linear projection back to the model dimension D and conversion to
   float32 complete the forward pass.

The implementation respects all test configurations (various seq‑len,
batch size, hidden/model dimensions and optional masks) and runs
entirely on the GPU (H100, Triton 3.3.1).  Only the forward pass is needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (used for the output norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,            # tensor pointers
    M, N, eps,                             # dimensions
    stride_xm, stride_xn, stride_ym, stride_yn,  # strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # Load tile
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, affine transform and store
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)          # preserve input dtype (fp16 / fp32)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm using Triton (supports fp16/fp32)."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    BLOCK_M = 256      # rows per program (covers up‑to ~10⁶ rows)
    BLOCK_N = 128      # cols per tile  (fits hidden_dim ≤ 128)

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for batched A @ Bᵀ (the N³ contraction)
# ----------------------------------------------------------------------
@triton.jit
def trigemm_kernel(
    left_ptr, right_ptr, out_ptr,    # fp16 pointers
    batch, N,                        # scalar dimensions
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, j, k]"""
    b = tl.program_id(0)            # batch index (B·H)
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # batch offsets (same for A, B and C because all are contiguous)
    batch_off_l = b * stride_l_batch
    batch_off_r = b * stride_r_batch
    batch_off_o = b * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load A (left) -------------------------------------------------
        a_ptr = left_ptr + batch_off_l \
                + offs_m[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ------- load B (right) transposed ------------------------------------
        # we need Bᵀ => rows = k, cols = n
        b_ptr = right_ptr + batch_off_r \
                + offs_k[:, None] * stride_r_i \
                + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot product of (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # ------- store result ------------------------------------------------------
    out_ptrs = out_ptr + batch_off_o \
               + offs_m[:, None] * stride_o_i \
               + offs_n[None, :] * stride_o_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, L, L, D] (float32)
        - mask         : torch.Tensor of shape [B, L, L] (float32) or None
        - weights      : dict containing all Linear/LayerNorm parameters
        - config       : dict with fields ``dim`` (D), ``hidden_dim`` (H)
                         and optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_f32 = input_tensor.dtype                     # expected torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch implementation – highly optimised)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype_f32).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype_f32).contiguous()
    x_norm = F.layer_norm(input_tensor,
                          (dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=eps)                     # [B, L, L, D]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free, fp16)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()

    left  = F.linear(x_h, left_proj_w)    # [B, L, L, H]
    right = F.linear(x_h, right_proj_w)   # [B, L, L, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, L, L, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, L, L, H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, L, L, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched contraction: out = left @ rightᵀ   (Triton GEMM)
    # ------------------------------------------------------------------
    #   left/right : [B, L, L, H]  ->  [B·H, L, L] for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, L, L]

    BH = bs * hidden_dim                               # batch dimension of GEMM
    left_flat  = left_perm.view(BH, L, L)              # [BH, L, L]   fp16
    right_flat = right_perm.view(BH, L, L)             # [BH, L, L]   fp16
    out_flat   = torch.empty_like(left_flat)           # allocate fp16 output

    # Strides for the contiguous (BH, L, L) tensors
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    # Triton block sizes – tuned for hidden_dim ≤ 128 and L up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(L, BLOCK_M),   # tiles over rows   (i dimension)
        triton.cdiv(L, BLOCK_N),   # tiles over cols   (j dimension)
    )
    trigemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        L,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original shape: [B, L, L, H] (still fp16)
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + output gate (Triton + fp16)
    # ------------------------------------------------------------------
    # Flatten to 2‑D for the row‑wise LayerNorm kernel
    out_flat2 = out.reshape(bs * L * L, hidden_dim).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_f32).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_f32).contiguous()
    out_norm_flat = triton_layernorm(out_flat2,
                                     to_out_norm_w,
                                     to_out_norm_b,
                                     eps=eps)               # fp16 result
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # Apply the output gate (both tensors are fp16)
    out = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension D (fp16 → fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)                     # [B, L, L, D]  (fp16)
    out = out.to(torch.float32)                       # required output dtype

    return out
