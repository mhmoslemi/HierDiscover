Highly‑optimized outgoing TriMul implementation (AlphaFold‑3).

Algorithm
---------
1. **Input LayerNorm** – fused Triton row‑wise LayerNorm on the
   flattened tensor (B*L*L, D).  The kernel computes mean/variance in FP32
   and writes the normalized values in the original dtype (FP32 or FP16).

2. **Single fused linear projection** – the four projection matrices
   (left_proj, left_gate, right_proj, right_gate) and the output‑gate
   matrix are concatenated into a single weight of shape (5*H, D) and
   multiplied once with the normalized input (FP16).  The result is split
   into the five components and the three gate tensors are passed through
   a sigmoid.  Left/right projections are masked (if a mask is present)
   before the gate multiplication.

3. **Batched GEMM contraction** – after gating we have
   `left` and `right` of shape (B, L, L, H).  They are reshaped to
   (B*H, L, L) and a single batched matrix‑multiply (`torch.bmm`) computes
   `out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]` as
   `left_flat @ right_flat.T`.

4. **Output LayerNorm + output‑gate** – the result is flattened to
   (B*L*L, H) and a second Triton row‑wise LayerNorm is applied.
   The normalized tensor is multiplied element‑wise by the previously
   computed output‑gate.

5. **Final projection** – a bias‑free linear layer maps the hidden
   dimension back to the model dimension `D`.  The final tensor is cast
   to float32 as required.

All heavy arithmetic is done in FP16 (except the reductions inside the
LayerNorm kernels which use FP32) and only a *single* matrix‑multiply is
performed for the five linear layers, dramatically reducing memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (re‑usable for input and output norms)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,  # INPUT  pointer
    y_ptr,  # OUTPUT pointer
    w_ptr,  # weight (scale) pointer
    b_ptr,  # bias pointer
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # Row indices handled by this program
    # ---------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------------------------------------------------------------
    # First pass: accumulate sum and sum‑sq for mean / var
    # ---------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalize, apply affine transform, store
    # ---------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be 2‑D (M, N).  Returns a tensor of the same shape/dtype.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make everything contiguous for stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # block sizes tuned for hidden dimensions ≤ 384
    BLOCK_M = 256
    BLOCK_N = 128

    grid = (triton.cdiv(M, BLOCK_M),)

    _layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul entry point (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]    (float32) or None
        - weights      : dict of torch tensors with the required keys
        - config       : dict with fields
                         * dim          – model dimension D
                         * hidden_dim   – hidden dimension H
                         * nomask (opt) – True ⇒ ignore ``mask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # float32
    eps = 1e-5

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    # flatten to 2‑D for the row‑wise kernel
    x_flat = input_tensor.reshape(bs * L * L, dim)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # reshape back to 4‑D
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projection (single matmul)
    #    Columns: 0..H-1      – left_proj
    #             H..2H-1    – left_gate
    #             2H..3H-1  – right_proj
    #             3H..4H-1  – right_gate
    #             4H..5H-1  – out_gate
    # ------------------------------------------------------------------
    # gather all weight matrices and cast to fp16
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    weight_cat = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape (5*H, D)

    # flatten input for the matmul
    x_h_flat = x_h.reshape(bs * L * L, dim)               # (M, D)

    # single fused GEMM (M, 5*H) = (M, D) @ (5*H, D).T
    fused_out = F.linear(x_h_flat, weight_cat)            # (M, 5*H), fp16

    # split back to the five branches
    split_tensors = torch.split(fused_out, hidden_dim, dim=1)
    left_proj_flat, left_gate_flat, right_proj_flat, right_gate_flat, out_gate_flat = split_tensors

    # reshape each branch to [B, L, L, H]
    left_proj = left_proj_flat.view(bs, L, L, hidden_dim)
    left_gate = left_gate_flat.view(bs, L, L, hidden_dim)
    right_proj = right_proj_flat.view(bs, L, L, hidden_dim)
    right_gate = right_gate_flat.view(bs, L, L, hidden_dim)
    out_gate = out_gate_flat.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Mask (if present)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction via batched GEMM
    #    left/right : (B, L, L, H) → (B*H, L, L)
    # ------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)

    # out_flat[b*h] = left_flat @ right_flat.T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, L, L, H] (still fp16)
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + output‑gate (Triton)
    # ------------------------------------------------------------------
    out_2d = out.reshape(bs * L * L, hidden_dim).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_2d = triton_layernorm(out_2d, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_2d.view(bs, L, L, hidden_dim)

    # apply output gate (both fp16)
    out = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)           # (B, L, L, D) in fp16
    out = out.to(torch.float32)             # required output dtype

    return out
