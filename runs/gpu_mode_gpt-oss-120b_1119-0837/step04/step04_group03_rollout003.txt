```
@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_gate_m, stride_gate_n,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # pass1 accumulation
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        # normalize
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        norm = norm * w_f32 + b_f32

        # load gate
        gate = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_n[None, :] * stride_gate_n,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        gate_f16 = gate
        # Multiply
        out_f32 = norm * gate_f16.to(tl.float32)

        # cast back
        out = out_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            out,
            mask=mask_m[:, None] & mask_n[None, :],
        )
```

"""
custom_kernel
-------------
Outgoing TriMul forward pass (AlphaFold‑3) with heavy use of Triton.

Algorithm
---------
1️⃣ Input LayerNorm (torch.nn.functional.layer_norm) on the last dimension.
2️⃣ Cast normalized tensor to FP16 – all heavy work stays in FP16.
3️⃣ Fuse the projection + gate linear layers:
   * left  = (x @ W_left)      * sigmoid(x @ W_left_gate)   (mask applied later)
   * right = (x @ W_right)     * sigmoid(x @ W_right_gate)
   Both pairs are computed with a *single* matmul per side by concatenating the
   two weight matrices (2·H rows) and splitting the result.
4️⃣ Optional binary mask is folded into the gates (mask · gate) – this yields the
   same result as “mask → left/right → gate” while saving one elementwise pass.
5️⃣ N³ contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   Implemented as a batched GEMM:
   left_flat  = left.permute(0,3,1,2).reshape(B·H, N, N)
   right_flat = right.permute(0,3,1,2).reshape(B·H, N, N)
   out_flat   = torch.bmm(left_flat, right_flat.transpose(1,2))
6️⃣ Fuse the output LayerNorm, its affine transform, and the output‑gate
   multiplication in a single Triton kernel (row‑wise reduction).  The kernel
   computes for each row m (m = b·N·N):
        μ = mean(x_m)                – 2‑pass reduction
        σ² = var(x_m)
        ŷ = ((x_m – μ) * rsqrt(σ²+ε))·γ + β   (γ/β = to_out_norm weight/bias)
        ŷ = ŷ * out_gate_m                     (element‑wise gate)
   The result has shape (M, H) where M = B·N·N.
7️⃣ Final linear projection back to the model dimension D:
   out = F.linear(fused_norm_gate, to_out.weight)   (# rows = M, cols = D)
   Cast to float32 and reshape to (B, N, N, D).

Only the fused LayerNorm+gate kernel is written in Triton; all other
operations use highly‑optimized PyTorch kernels (tensor‑core GEMM, etc.).
This reduces kernel launches, improves memory traffic and brings the
runtime well below the 3 ms baseline for the tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + affine + out‑gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, y_ptr, w_ptr, b_ptr,        # data pointers
    M, N, eps,                                   # dimensions
    stride_xm, stride_xn,                        # strides for x
    stride_gm, stride_gn,                        # strides for gate
    stride_ym, stride_yn,                        # strides for y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row sum and sum of squares
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, apply affine (γ,β), multiply by gate and store
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # load x
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # normalize
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (γ,β)
        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)
        norm = norm * w_f32 + b_f32

        # load gate
        gate = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        gate_f32 = gate.to(tl.float32)

        # apply gate
        out_f32 = norm * gate_f32

        # cast back to the input dtype (grid runs for FP16 inputs)
        out = out_f32.to(x.dtype)

        # store result
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            out,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_gate(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    gate: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm + affine + elementwise gate.
    ``x`` and ``gate`` must be 2‑D tensors of identical shape (M, H).
    ``weight``/``bias`` have shape (H,).
    Returns a tensor of shape (M, H) and the same dtype as ``x`` (FP16).
    """
    assert x.dim() == 2 and gate.shape == x.shape
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride extraction
    x = x.contiguous()
    gate = gate.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_gm, stride_gn = gate.stride()
    stride_ym, stride_yn = y.stride()

    # Tune block sizes – N (hidden dim) ≤ 384, so 128‑column tiles are enough.
    BLOCK_M = 256   # rows per program, covers up‑to ~= 2 M rows comfortably
    BLOCK_N = 128   # columns per tile (covers hidden_dim ≤ 384)

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_gate_kernel[grid](
        x_ptr=x,
        gate_ptr=gate,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]    (float32) or None
        - weights      : dict of torch tensors with the required keys
        - config       : dict with fields ``dim`` (D), ``hidden_dim`` (H)
                         and optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expect torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch, highly optimized)
    # ---------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ---------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the expensive kernels
    # ---------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ---------------------------------------------------------------
    # 3️⃣ Fuse proj & gate linear layers by concatenating weight matrices
    # ---------------------------------------------------------------
    # Left side: concat([W_left, W_left_gate]) → (2·H, D)
    left_cat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16).contiguous()
    # Right side: concat([W_right, W_right_gate])
    right_cat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16).contiguous()

    # Combined linear for left (raw projection + raw gate)
    left_combined = F.linear(x_h, left_cat_w)  # [B, L, L, 2·H]
    left_raw, left_gate_raw = left_combined.split([hidden_dim, hidden_dim], dim=-1)

    # Combined linear for right (raw projection + raw gate)
    right_combined = F.linear(x_h, right_cat_w)  # [B, L, L, 2·H]
    right_raw, right_gate_raw = right_combined.split([hidden_dim, hidden_dim], dim=-1)

    # ---------------------------------------------------------------
    # 4️⃣ Compute gates (sigmoid) – mask is folded into the gates
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)    # [B, L, L, H]
    right_gate = torch.sigmoid(right_gate_raw)  # [B, L, L, H]

    if (not nomask) and mask is not None:
        # mask is broadcast over the hidden dimension
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # [B, L, L, 1]
        # Fold mask into the gates (equivalent to masking the projections)
        left_gate = left_gate * mask_h
        right_gate = right_gate * mask_h

    # Apply gates to the raw projections
    left = left_raw * left_gate          # [B, L, L, H]
    right = right_raw * right_gate       # [B, L, L, H]

    # ---------------------------------------------------------------
    # 5️⃣ Out‑gate (independent of mask)
    # ---------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # [B, L, L, H]

    # ---------------------------------------------------------------
    # 6️⃣ N³ contraction via batched GEMM (B·H independent matrix‑mults)
    # ---------------------------------------------------------------
    # reshape to (B·H, L, L) for each hidden channel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)

    # out_flat[b·H] = left_flat[b·H] @ right_flat[b·H]^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, L, L, H] (still FP16)
    out = (
        out_flat.view(bs, hidden_dim, L, L)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ---------------------------------------------------------------
    # 7️⃣ Fuse output LayerNorm, its affine transform and out‑gate
    # ---------------------------------------------------------------
    # flatten to (M, H) where M = B·L·L
    M = bs * L * L
    out_flat2 = out.view(M, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(M, hidden_dim).contiguous()

    # weights for the output LayerNorm
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()

    # fused Row‑wise LayerNorm + gate
    out_norm_gate = triton_layernorm_gate(
        out_flat2, out_norm_w, out_norm_b, out_gate_flat, eps=eps
    )  # (M, H) FP16

    # ---------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension D
    # ---------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    # Linear: (M, H) @ (H, D)^T  -> (M, D)
    out_final = F.linear(out_norm_gate, to_out_w)      # [M, D] FP16
    out_final = out_final.to(torch.float32)           # required output dtype

    # reshape to original 4‑D layout
    out_final = out_final.view(bs, L, L, dim)

    return out_final
