def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # get dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Input LayerNorm using torch.nn.functional.layer_norm (FP32)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps)

    # Convert to FP16 for heavy compute
    x_f16 = x_norm.to(torch.float16)

    # Combine all linear weights for left_proj, left_gate, right_proj, right_gate, out_gate
    # All weights are stored without bias
    # weights dict: left_proj.weight, left_gate.weight, right_proj.weight, right_gate.weight, out_gate.weight
    # Shapes: (hidden_dim, dim) each.
    # Concatenate along dim0: order as left_proj, left_gate, right_proj, right_gate, out_gate
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)  # shape (5*hidden_dim, dim)

    # Compute combined linear
    # shape (B,N,N,5*hidden_dim)
    combined = torch.nn.functional.linear(x_f16, weight_all)  # no bias

    # Split
    H = hidden_dim
    left_proj = combined[..., :H]
    left_gate_z = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate_z = combined[..., 3*H:4*H]
    out_gate_z = combined[..., 4*H:5*H]

    # Optional mask multiplication on projections
    if not config.get("nomask", False) and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Gate activation (sigmoid)
    left_gate = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out_gate
    out_gate = torch.sigmoid(out_gate_z)

    # Core contraction: out = left @ right^T across k dimension
    # reshape to (B*H,N,N) for batched GEMM
    # permute to bring H to batch dimension, then view.
    # Use contiguous to ensure memory layout.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Compute batched matmul: (B*H,N,N) @ (B*H,N,N)^T
    # Use torch.bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Now apply second LayerNorm + out_gate gating using Triton kernel
    # Flatten for kernel: rows = B*N*N, cols = H
    M = bs * N * N
    out_flat = out.view(M, H)
    out_gate_flat = out_gate.view(M, H)

    # Load second LN weights
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, torch.float16)      # (H,)

    # Apply custom kernel: LN + gate
    out_norm_and_gate = triton_layernorm_fp16_gate(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        out_gate_flat,
        eps=eps,
    )  # shape (M, H)

    # Final linear projection back to model dim
    to_out_weight = weights["to_out.weight"].to(device, torch.float16)  # (dim, H)
    output_flat = torch.nn.functional.linear(out_norm_and_gate, to_out_weight)  # (M, dim)
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output

):

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # first pass: compute mean and var per row (fp32)
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

"""
TriMul (outgoing) forward pass – optimized Triton implementation.

Algorithm:
1. Input LayerNorm (float32, cuDNN).
2. Cast to fp16.
3. Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) and the out_gate projection into a single GEMM.
   The result is sliced into the five tensors we need.
4. Optional pair‑wise mask is applied in‑place.
5. Apply sigmoid gates and obtain the gated left/right tensors.
6. Batched GEMM implements the N³ contraction:
   reshape (B,N,N,H) → (B·H,N,N) and compute
   C = A @ Aᵀ using torch.bmm.
7. Fuse the second LayerNorm and the element‑wise output‑gate in a
   custom Triton kernel (`layernorm_fp16_gate_kernel`).
8. Final linear projection back to the model dimension (fp16 GEMM) and
   cast back to float32.

All heavy linear work stays in fp16; the only fp32 work is the
numerically‑stable two‑pass LayerNorm inside the Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm (fp16) with fused output‑gate scaling
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,                 # input  (M, H)  fp16
    y_ptr,                 # output (M, H)  fp16
    gate_ptr,              # gate   (M, H)  fp16
    weight_ptr,            # LN weight (H)  fp16
    bias_ptr,              # LN bias   (H)  fp16
    M,                     # number of rows
    H,                     # hidden dimension (per‑row length)
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,            # rows per program
    BLOCK_N: tl.constexpr             # cols per program (sub‑tile)
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣ First pass – compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass – write normalized + affine transformed values
    #                 and apply the element‑wise output gate.
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)

        # (x - μ) / σ
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # affine
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)
        y = norm * w[None, :] + b[None, :]

        # gate
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        y = y * g

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16_gate(
    x: torch.Tensor,          # (M, H) fp16
    weight: torch.Tensor,     # (H,) fp16
    bias: torch.Tensor,       # (H,) fp16
    gate: torch.Tensor,       # (M, H) fp16
    eps: float = 1e-5
) -> torch.Tensor:
    """LayerNorm (fp16) with fused gate scaling."""
    assert x.ndim == 2 and x.dtype == torch.float16
    M, H = x.shape
    y = torch.empty_like(x)

    # ensure contiguity for simple stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    gate = gate.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    stride_gm, stride_gn = gate.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_gate_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        gate_ptr=gate,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        BLOCK_M=256,
        BLOCK_N=64,     # tile size keeps register pressure low & works for H≤384
    )
    return y


# ----------------------------------------------------------------------
# Entry point – custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 native cuDNN)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, D) – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections (left, right, out_gate)
    # ------------------------------------------------------------------
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    all_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)  # shape (5*H, D)

    # One big GEMM: (B·N·N, D) @ (5·H, D).T → (B·N·N, 5·H)
    combined = torch.nn.functional.linear(x_f16, all_weight)  # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the combined tensor
    # ------------------------------------------------------------------
    left_proj   = combined[..., :H]                # (B, N, N, H)
    left_gate_z = combined[..., H:2*H]             # (B, N, N, H)
    right_proj  = combined[..., 2*H:3*H]           # (B, N, N, H)
    right_gate_z= combined[..., 3*H:4*H]           # (B, N, N, H)
    out_gate_z  = combined[..., 4*H:5*H]           # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 6️⃣ Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)

    left  = left_proj * left_gate    # (B, N, N, H)
    right = right_proj * right_gate  # (B, N, N, H)

    out_gate = torch.sigmoid(out_gate_z)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B·H, N, N) for a single torch.bmm call.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # (B·H, N, N) = (B·H, N, N) @ (B·H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to (B, N, N, H) – keep fp16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dim + fused output‑gate scaling
    # ------------------------------------------------------------------
    M = bs * N * N                      # total rows to normalise
    out_flat = out.view(M, H).contiguous()                # (M, H) fp16
    out_gate_flat = out_gate.view(M, H).contiguous()      # (M, H) fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)

        out_flat, to_out_norm_w, to_out_norm_b, out_gate_flat, eps=eps
    )  # (M, H) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to model dim (fp16) → fp32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = torch.nn.functional.linear(out_norm, to_out_w)    # (M, D) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)      # (B, N, N, D)

    return output

"""
TriMul (outgoing) forward pass – optimized Triton implementation.

Algorithm
---------
1️⃣ Input LayerNorm (float32) – uses cuDNN, no custom kernel.
2️⃣ Cast to fp16 for all heavy linear work.
3️⃣ Fuse the four side‑projections (left/right) and the output‑gate
   projection into a *single* GEMM:
   (B·N·N, dim) @ (5·hidden_dim, dim).T → (B·N·N, 5·hidden_dim).
   The result is sliced into left_proj, left_gate, right_proj,
   right_gate and out_gate.
4️⃣ Optional pair‑wise mask is applied element‑wise (broadcast over the
   hidden dimension).
5️⃣ Apply sigmoid gates and obtain the gated left/right tensors.
6️⃣ Core N³ contraction is performed with one batched fp16 GEMM:
   reshape the gated tensors to (B·hidden_dim, N, N) and compute
   C = A @ Aᵀ via `torch.bmm`.
7️⃣ Fuse the second LayerNorm and the element‑wise output‑gate in a
   custom Triton kernel (`layernorm_fp16_gate_kernel`).  The kernel
   uses a two‑pass (FP32) mean/variance computation for numerical
   stability.
8️⃣ Final linear projection back to the model dimension (fp16 GEMM) and
   cast to float32 for the returned tensor.

All expensive matrix work stays in fp16; the only FP32 work is the
stable LayerNorm inside the Triton kernel.  The implementation respects
all configuration options (bs, seqlen, dim, hidden_dim, nomask) and
works with or without a mask.  It returns a tensor of shape
[bs, seqlen, seqlen, dim] (float32)."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (fp16) with fused output‑gate scaling
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,                 # input  (M, H)  fp16
    y_ptr,                 # output (M, H)  fp16
    gate_ptr,              # gate   (M, H)  fp16
    weight_ptr,            # LN weight (H)  fp16
    bias_ptr,              # LN bias   (H)  fp16
    M,                     # number of rows
    H,                     # hidden dimension (per‑row length)
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,            # rows per program
    BLOCK_N: tl.constexpr             # cols per program (sub‑tile)
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣ First pass – compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass – write normalized + affine transformed values
    #                 and apply the element‑wise output gate.
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)

        # (x - μ) / σ
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # affine
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)
        y = norm * w[None, :] + b[None, :]

        # gate
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        y = y * g

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16_gate(
    x: torch.Tensor,          # (M, H) fp16
    weight: torch.Tensor,     # (H,) fp16
    bias: torch.Tensor,       # (H,) fp16
    gate: torch.Tensor,       # (M, H) fp16
    eps: float = 1e-5
) -> torch.Tensor:
    """LayerNorm (fp16) with fused gate scaling."""
    assert x.ndim == 2 and x.dtype == torch.float16
    M, H = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    gate = gate.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    stride_gm, stride_gn = gate.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_gate_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        gate_ptr=gate,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        BLOCK_M=256,               # tuned for H100
        BLOCK_N=64,                # works for H ≤ 384
    )
    return y


# ----------------------------------------------------------------------
# Entry point – custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data = (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor[B, N, N, D] (float32)
        mask         : torch.Tensor[B, N, N] or None
        weights      : dict of torch.Tensor (model parameters)
        config       : dict with keys
                       - dim          (model dimension D)
                       - hidden_dim   (H)
                       - nomask (bool, optional)

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – cuDNN
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # (B, N, N, D) – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into one GEMM
    #    Order: left_proj, left_gate, right_proj, right_gate, out_gate
    # ------------------------------------------------------------------
    all_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                 # (5·H, D)

    # (B·N·N, D) @ (5·H, D).T → (B·N·N, 5·H)
    combined = torch.nn.functional.linear(x_f16, all_weight)  # (B, N, N, 5·H)

    # ------------------------------------------------------------------
    # 4️⃣ Slice combined tensor
    # ------------------------------------------------------------------
    left_proj   = combined[..., :H]                 # (B, N, N, H)
    left_gate_z = combined[..., H:2*H]              # (B, N, N, H)
    right_proj  = combined[..., 2*H:3*H]            # (B, N, N, H)
    right_gate_z= combined[..., 3*H:4*H]            # (B, N, N, H)
    out_gate_z  = combined[..., 4*H:5*H]            # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 6️⃣ Sigmoid gates and gated tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)

    left  = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate        # (B, N, N, H)

    out_gate = torch.sigmoid(out_gate_z)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Bring the hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # (B·H, N, N) = (B·H, N, N) @ (B·H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to (B, N, N, H) – keep fp16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm + fused output‑gate scaling (Triton)
    # ------------------------------------------------------------------
    M = bs * N * N                                 # number of rows to normalise
    out_flat = out.view(M, H).contiguous()        # (M, H) fp16
    out_gate_flat = out_gate.view(M, H).contiguous()  # (M, H) fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)

    out_norm = triton_layernorm_fp16_gate(
        out_flat,
        to_out_norm_w,
        to_out_norm_b,
        out_gate_flat,
        eps=eps,
    )                                            # (M, H) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to model dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = torch.nn.functional.linear(out_norm, to_out_w)   # (M, D) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)    # (B, N, N, D)

    return output
