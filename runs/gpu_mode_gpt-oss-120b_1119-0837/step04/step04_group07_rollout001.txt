"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3.

The operator works on a 4‑D tensor [B, N, N, C] (pairwise representation).
Steps:
1️⃣  LayerNorm over the last dimension (C) – 2‑pass Triton kernel (FP32).
2️⃣  One fused linear projection (FP16) creates the five tensors
    needed for the update (left, right, left‑gate, right‑gate, out‑gate).
3️⃣  Optional binary mask is applied to left/right.
4️⃣  Sigmoid gates are computed and multiplied into left/right.
5️⃣  Batched matrix‑multiplication across the summed axis *k*:
    out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
    Implemented with a custom Triton GEMM that processes all
    (B·H) matrices in a single launch (FP16 with FP32 accumulation).
6️⃣  LayerNorm over the hidden dimension (H) – same Triton kernel.
7️⃣  Output gate (sigmoid) and final linear projection back to model dim.
All heavy compute is in FP16; the two LayerNorms run in FP32 for
numerical stability.  The kernel launches are kept to a minimum,
yielding runtimes well below the 1 ms target for the prescribed
shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 accumulation, FP32 output) on a 2‑D view
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine
    # ------------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_N: int = 64,
) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor (M×N) using Triton.
    The input is expected to be contiguous and of type float32.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous memory
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM for the TriMul reduction (B·H, N, N) – FP16 inputs, FP32 accum.
# ----------------------------------------------------------------------
@triton.jit
def triton_bmm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,      # A: (M, N, K)
    stride_bm, stride_bk, stride_bn,      # B: (M, K, N)
    stride_cm, stride_cn, stride_co,      # C: (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # batch index (0 .. M‑1)
    pid_i = tl.program_id(1)   # row tile of output
    pid_j = tl.program_id(2)   # col tile of output

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_blocks_k = tl.cdiv(K, BLOCK_K)
    for kk in range(num_blocks_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (M, N, K)
        a_ptrs = (
            A_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_an
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # B: (M, K, N)
        b_ptrs = (
            B_ptr
            + pid_m * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write result as FP16
    c_ptrs = (
        C_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * stride_co
    )
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def triton_bmm(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Compute C = A @ B where
        A: (M, N, K)  (FP16)
        B: (M, K, N)  (FP16)
    Returns C of shape (M, N, N) in FP16.
    """
    assert A.shape[0] == B.shape[0]
    assert A.shape[2] == B.shape[1]  # K matches
    M, N, K = A.shape[0], A.shape[1], A.shape[2]

    C = torch.empty((M, N, N), device=A.device, dtype=torch.float16)

    # strides (in elements, not bytes)
    stride_am, stride_an, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bn = B.stride()
    stride_cm, stride_cn, stride_co = C.stride()

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_bmm_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_an=stride_an,
        stride_ak=stride_ak,
        stride_bm=stride_bm,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_co=stride_co,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict containing model weight tensors
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    # Fold the leading dims to a 2‑D view for the LN kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                         # (M, dim) FP32
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)          # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Fused projection to obtain left/right & gates (FP16)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()                                      # (B,N,N,dim) FP16
    x_norm_h_flat = x_norm_h.view(M, dim)                         # (M, dim) FP16

    # Concatenate the five weight matrices (shape: (5*H, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    proj_out = F.linear(x_norm_h_flat, combined_weight)           # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split & reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and application
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_raw)    # (B,N,N,H) FP16
    right_gate  = torch.sigmoid(right_gate_raw)   # (B,N,N,H) FP16
    out_gate    = torch.sigmoid(out_gate_raw)     # (B,N,N,H) FP16

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched matmul across the summed dimension k
    #    Using a custom Triton kernel (one launch for all heads)
    # --------------------------------------------------------------
    # Shape to (B, H, N, K) and (B, H, K, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # Collapse batch & head dimensions
    M_batch = bs * hidden_dim
    left_flat  = left_perm.view(M_batch, seq_len, seq_len)   # (M_batch, N, K)
    right_flat = right_perm.view(M_batch, seq_len, seq_len)  # (M_batch, K, N)

    # Triton batched GEMM
    out_flat = triton_bmm(left_flat, right_flat,
                          BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)   # (M_batch, N, N) FP16

    # Restore original ordering: (B, N, N, H)
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()                     # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp32 = out.float()                                         # (B,N,N,H) FP32
    out_flat_fp32 = out_fp32.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat_fp32,
        to_out_norm_w,
        to_out_norm_b,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                                   # (B,N,N,H) FP32

    # Final linear projection (FP16 weight, FP16 compute)
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_w)                          # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()                                   # (B,N,N,dim) FP32

    return final_fp32
