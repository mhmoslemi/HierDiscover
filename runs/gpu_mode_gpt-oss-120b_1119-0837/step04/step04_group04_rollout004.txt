    the hidden‑dim output.

2️⃣  **Linear projections** – the four linear maps (left/right proj &
    left/right gate) are fused into a single half‑precision matmul
    (shape *M × 4·H*).

3️⃣  **Mask & gating** – applied with plain PyTorch elementwise ops
    (very cheap compared with the contraction).

4️⃣  **Batched GEMM (the N³ contraction)** – computes
    ``out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]`` as a batched
    matrix multiplication ``left @ rightᵀ``.  The kernel processes all
    ``batch × head`` matrices in one launch, using:
        * BLOCK_M = BLOCK_N = 128    (large tiles → fewer kernel launches)
        * BLOCK_K = 32               (fits Tensor‑core granularity)
        * fp16 inputs, fp32 accumulation (Tensor‑core friendly)

5️⃣  **Output LayerNorm**, **output gating** and **final projection**
    – performed with PyTorch primitives; the result is cast back to
    float32 as required by the model.

The heavy work is done in the two Triton kernels; everything else
leverages highly optimized PyTorch ops.  The code runs on an H100
GPU (triton 3.3.1) and satisfies the required API:

    custom_kernel((input, mask, weights, config)) → output

where *output* has shape ``[B, L, L, dim]`` and dtype ``float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,              # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,      # *ptr to affine parameters [N]
    M, N, eps,                 # dimensions + epsilon
    stride_xm, stride_xn,      # input strides
    stride_ym, stride_yn,      # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------- row offset ----------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm via Triton.
    ``x`` must be 2‑D; ``weight`` / ``bias`` are 1‑D.
    Returns a tensor with the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous tensors
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel (the N³ contraction).
# Computes C[b,i,j] = Σₖ A[b,i,k] * B[b,k,j] for each batch/head.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_h: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_ah: tl.constexpr, stride_am: tl.constexpr, stride_ak: tl.constexpr,
    stride_bh: tl.constexpr, stride_bk: tl.constexpr, stride_bn: tl.constexpr,
    stride_ch: tl.constexpr, stride_cm: tl.constexpr, stride_cn: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    pid_batch = tl.program_id(0)  # (batch × head)
    pid_m = tl.program_id(1)      # row block
    pid_n = tl.program_id(2)      # col block

    # --------------------------------------------------------------
    # block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: BLOCK_M × BLOCK_K
        a = tl.load(
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B: BLOCK_K × BLOCK_N
        b = tl.load(
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core lane (fp16×fp16→fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back (fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A: (batch_h, M, K)   dtype=fp16, contiguous
    B: (batch_h, K, N)   dtype=fp16, contiguous
    Returns C: (batch_h, M, N) dtype=fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch_h, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch_h, M, N), dtype=A.dtype, device=A.device)

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides (contiguous tensors)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch_h,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch_h=batch_h,
        M=M,
        N=N,
        K=K,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # The following lines let Triton choose an appropriate number of warps.
        # 8 warps works well on H100 for 128×128 tiles.
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Forward entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the last dimension ``dim``
    # --------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)                # [M, dim]
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)             # [B, L, L, D]

    # --------------------------------------------------------------
    # Cast to fp16 for the heavy computation
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear maps into a single matmul
    #    (left_proj, right_proj, left_gate, right_gate)
    # --------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    # (4*H, D) where H = hidden_dim
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)     # (4·H, D)

    x_flat_h = x_norm_h.reshape(M, dim)                     # (M, D)
    fused_out = F.linear(x_flat_h, fused_w)                 # (M, 4·H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)    # (B, L, L, 4·H)

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and the sigmoid gates (pure torch ops)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B, L, L, 1)
        left_proj = fused_out[..., :hidden_dim] * mask_h
        right_proj = fused_out[..., hidden_dim:2 * hidden_dim] * mask_h
    else:
        left_proj = fused_out[..., :hidden_dim]
        right_proj = fused_out[..., hidden_dim:2 * hidden_dim]

    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    left = left_proj * left_gate               # (B, L, L, H)
    right = right_proj * right_gate            # (B, L, L, H)

    # --------------------------------------------------------------
    # 4️⃣ Prepare tensors for the batched GEMM:
    #     left  → shape (batch_h, L, L) where batch_h = B·H
    #     right → transposed shape (batch_h, L, L) ready for A @ Bᵀ
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, L, L)

    # Transpose the inner axes of ``right`` so we can use a simple GEMM.
    right_t = right_perm.transpose(2, 3).contiguous()      # (B, H, L, L) → (B, H, K, N)

    batch_h = bs * hidden_dim
    left_mat = left_perm.reshape(batch_h, L, L)           # (batch_h, M, K)
    right_mat = right_t.reshape(batch_h, L, L)            # (batch_h, K, N)

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM (Triton) – replaces the O(N³) contraction
    # --------------------------------------------------------------
    out_mat = batched_matmul_triton(left_mat, right_mat)  # (batch_h, L, L), fp16

    # Restore original layout: (B, L, L, H)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)               # (M, H)
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)    # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) – linear over the *original* normalized x
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_flat_h, out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)        # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to ``dim``
    # --------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated, to_out_w)            # (B, L, L, D), fp16
    out_final = out_final_h.to(torch.float32)          # cast to required dtype

    return out_final
