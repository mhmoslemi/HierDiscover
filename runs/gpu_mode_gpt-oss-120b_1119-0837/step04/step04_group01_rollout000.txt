"""
Highly‑optimized Triton‑based implementation of the “outgoing” TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch module:

1️⃣  Layer‑norm the input (float32) with a custom 2‑pass Triton kernel.
2️⃣  Fuse the five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single FP16 GEMM.
3️⃣  Apply sigmoid gates and an optional pair‑mask to the projected tensors.
4️⃣  Perform the N³ contraction as a batched half‑precision GEMM:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
5️⃣  Layer‑norm the hidden result (FP16) with a second Triton kernel.
6️⃣  Apply the output gate and a final FP16‑to‑FP32 projection.

All heavy work is done in FP16; the final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# 2‑pass LayerNorm (float32 input → float32 output)
# ------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Determine the rows this program works on
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write (x - mean) * rstd  * weight + bias
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm on a 2‑D tensor [M, N] in FP32 using Triton."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# 2‑pass LayerNorm (FP16 input → FP16 output)
# ------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

        torch.Tensor[B, N, N, D] (float32)

    device = input_tensor.device
    dtype = input_tensor.dtype   # should be torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ----------------------------------------------------------------
    # Flatten batch & pair dims → (M, D) where M = B * N * N
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                     # (M, D)
    # Load LayerNorm params
    norm_w = weights["norm.weight"].to(device, dtype)         # (D,)
    norm_b = weights["norm.bias"].to(device, dtype)           # (D,)
    # Run Triton layernorm (float32)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)  # (M, D)
    # Reshape back to [B, N, N, D] and cast to FP16 for heavy ops
    x_norm = x_norm_flat.view(bs, N, N, dim)                  # (B,N,N,D)

    # 2️⃣ Fuse the five linear projections into a single GEMM
    # ----------------------------------------------------------------
    # Concatenate weights: order = left_proj, left_gate, right_proj, right_gate, out_gate
    weight_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)          # (5*H, D)

    # Perform (M, D) @ (D, 5*H)^T → (M, 5*H)  (F.linear does weight^T internally)
    fused = F.linear(x_f16, weight_fused)  # (M, 5*H)
    fused = fused.view(bs, N, N, 5 * H)    # (B,N,N,5H)

    # ----------------------------------------------------------------
    # 3️⃣ Slice, apply sigmoid gates & optional mask
    # ----------------------------------------------------------------
    # Slice the fused tensor into its components
    h0, h1 = 0, H
    h2, h3 = H, 2 * H
    h4, h5 = 2 * H, 3 * H
    h6, h7 = 3 * H, 4 * H
    h8, h9 = 4 * H, 5 * H

    left_proj   = fused[..., h0:h1]   # (B,N,N,H)
    left_gate_p = fused[..., h1:h2]
    right_proj  = fused[..., h2:h3]
    right_gate_p = fused[..., h3:h4]
    out_gate_p  = fused[..., h4:h5]

    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    if (not nomask) and (mask is not None):
        # mask shape [B,N,N] → broadcast over hidden dim
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Apply element‑wise gates
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # ----------------------------------------------------------------
    # 4️⃣ N³ contraction as batched half‑precision GEMM
    # ----------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B*H, N, N)

    # Compute out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) over hidden dimension H
    # ----------------------------------------------------------------
    out_flat = out.view(M, H)                         # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ----------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast to float32
    # ----------------------------------------------------------------
    out_gate_flat = out_gate.view(M, H)               # (M, H)
    gated = out_norm * out_gate_flat                  # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)           # (M, D)

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
