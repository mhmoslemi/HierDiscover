@triton.jit
def norm_gate_linear_kernel(
    out_ptr, out_gate_ptr,
    norm_w_ptr, norm_b_ptr,
    weight_ptr,
    out_ptr_final,
    M, H, D, eps,
    stride_out_m, stride_out_h,
    stride_gate_m, stride_gate_h,
    stride_norm_w_h, stride_norm_b_h,
    stride_weight_h, stride_weight_d,
    stride_out_final_m, stride_out_final_d,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # Compute sum and sumsq across H
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # iteration across H
    num_h = tl.cdiv(H, BLOCK_H)
    for k in range(num_h):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out
        out = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0)

        # cast to fp32
        out_fp32 = tl.cast(out, tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq += tl.sum(out_fp32 * out_fp32, axis=1)

    # compute mean and rstd
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass accumulate final output
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for k in range(num_h):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and out_gate
        out = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0)
        gate = tl.load(
            out_gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0)

        # load norm weights & bias for h block
        norm_w = tl.load(
            norm_w_ptr + offs_h,
            mask=mask_h,
            other=0.0)
        norm_b = tl.load(
            norm_b_ptr + offs_h,
            mask=mask_h,
            other=0.0)

        # cast to fp32
        out_f32 = tl.cast(out, tl.float32)
        gate_f32 = tl.cast(gate, tl.float32)
        norm_w_f32 = tl.cast(norm_w, tl.float32)
        norm_b_f32 = tl.cast(norm_b, tl.float32)

        # normalize and scale
        # (out - mean) * rstd  -> shape (BLOCK_M, BLOCK_H)
        normalized = (out_f32 - mean[:, None]) * rstd[:, None]  # (M_block, H_block)
        normalized = normalized * norm_w_f32[None, :] + norm_b_f32[None, :]
        gated = normalized * gate_f32

        # now compute GEMM part: gated (M_block, H_block) dot weight (H_block, D_block)
        # load weight block as (BLOCK_H, BLOCK_D)
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_weight_h + offs_d[None, :] * stride_weight_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0)

        # convert gated back to fp16? dot works with fp16 best. But we can use fp32 dot; not sure if tl.dot works for fp32? It should. If shape (BLOCK_M, BLOCK_H) and (BLOCK_H, BLOCK_D) and dtype fp32, dot yields fp32. That may be fine.

        # cast gated fp16? Actually gated is fp32; we can just use fp32 dot.

        acc += tl.dot(gated, w)  # (M_block, D_block)

             tl.cast(acc, tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # Input layernorm: combine weight and bias from weights "norm.weight", "norm.bias"
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)
    # col dims from weights as same dtype of input
    norm_w = weights["norm.weight"].to(device, input_tensor.dtype)
    norm_b = weights["norm.bias"].to(device, input_tensor.dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # Cast to fp16
    x_norm_fp16 = x_norm.to(torch.float16)
    x_norm_fp16_flat = x_norm_fp16.view(M, dim)

    # Combined linear for left_proj,right_proj and gates
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # shape (5*hidden_dim, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)

    # Linear projection
    proj_gate_flat = torch.nn.functional.linear(x_norm_fp16_flat, combined_w)  # shape (M, 5*H)
    proj_gate = proj_gate_flat.view(bs, N, N, 5*hidden_dim)

    H = hidden_dim
    left_proj = proj_gate[..., 0:H]
    left_gate_pre = proj_gate[..., H:2*H]
    right_proj = proj_gate[..., 2*H:3*H]
    right_gate_pre = proj_gate[..., 3*H:4*H]
    out_gate_pre = proj_gate[..., 4*H:5*H]

    # Apply sigmoid gated gating vectors
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply mask if any
    if not config.get("nomask", False) and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Perform contraction 'outgoing' style: sum over k dimension
    # left: (B,N,N,H) ; right: (B,N,N,H) ; out = einsum('i k h, j k h -> i j h')
    # Equivalent to batched GEMMs
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)
    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H, N, N)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # Second layernorm + output gate + final projection fused kernel
    # reshape to (M, H)
    out_flat = out.view(M, hidden_dim)   # fp16
    out_gate_flat = out_gate.view(M, hidden_dim)   # fp16

    # Fetch normalization parameters for second layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    # Final weight
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # shape (dim, hidden_dim)

    # Allocate output
    out_final_fp16 = torch.empty((M, dim), dtype=torch.float16, device=device)

    # Compute block sizes
    BLOCK_M = 128
    BLOCK_D = 64
    BLOCK_H = 32   # hidden dimension tiling

    # Compute grid sizes
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(dim, BLOCK_D))

    # call fused kernel
    fused_norm_gate_linear_kernel[grid](
        out_ptr=out_flat,
        out_gate_ptr=out_gate_flat,
        norm_w_ptr=to_out_norm_w,
        norm_b_ptr=to_out_norm_b,
        weight_ptr=to_out_w,
        out_ptr_final=out_final_fp16,
        M=M,
        H=hidden_dim,
        D=dim,
        eps=eps,
        stride_out_m=out_flat.stride(0), stride_out_h=out_flat.stride(1),
        stride_gate_m=out_gate_flat.stride(0), stride_gate_h=out_gate_flat.stride(1),
        stride_norm_w_h=to_out_norm_w.stride(0), stride_norm_b_h=to_out_norm_b.stride(0),
        stride_weight_h=to_out_w.stride(1), stride_weight_d=to_out_w.stride(0),
        stride_out_final_m=out_final_fp16.stride(0), stride_out_final_d=out_final_fp16.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    out_final = out_final_fp16.view(bs, N, N, dim).to(torch.float32)

    return out_final

```
@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr, out_gate_ptr, norm_w_ptr, norm_b_ptr,
    weight_ptr, out_ptr_final,
    M, H, D, eps,
    stride_out_m, stride_out_h,
    stride_gate_m, stride_gate_h,
    stride_norm_w, stride_norm_b,
    stride_weight_h, stride_weight_d,
    stride_out_final_m, stride_out_final_d,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_d = offs_d < D

    # First pass: compute sum and sum_sq over H for each row (position)
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iters_h = tl.cdiv(H, BLOCK_H)

    for kb in range(num_iters_h):
        offs_h = kb * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        out = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_f32 = tl.cast(out, tl.float32)
        sum_val += tl.sum(out_f32, axis=1)         # sum across H block
        sum_sq += tl.sum(out_f32 * out_f32, axis=1)

    # compute mean and rstd per row
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: compute final output via reduction over H
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for kb in range(num_iters_h):
        offs_h = kb * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and out_gate
        out = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        gate = tl.load(
            out_gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        # load norm weight and bias for this block of H
        norm_w = tl.load(
            norm_w_ptr + offs_h,
            mask=mask_h,
            other=0.0,
        )
        norm_b = tl.load(
            norm_b_ptr + offs_h,
            mask=mask_h,
            other=0.0,
        )
        # cast to fp32
        out_f32 = tl.cast(out, tl.float32)
        gate_f32 = tl.cast(gate, tl.float32)
        norm_w_f32 = tl.cast(norm_w, tl.float32)
        norm_b_f32 = tl.cast(norm_b, tl.float32)

        # Normalize, scale, bias
        # (out - mean) * rstd
        normed = (out_f32 - mean[:, None]) * rstd[:, None]
        # apply weight and bias
        normed = normed * norm_w_f32[None, :] + norm_b_f32[None, :]

        # apply gate
        gated = normed * gate_f32

        # load weight block for final linear: shape (BLOCK_H, BLOCK_D)
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_weight_h + offs_d[None, :] * stride_weight_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_fp32 = tl.cast(w, tl.float32)

        # dot product across H
        acc += tl.dot(gated, w_fp32)   # (BLOCK_M, BLOCK_D)

    # store
    tl.store(
        out_ptr_final + offs_m[:, None] * stride_out_final_m + offs_d[None, :] * stride_out_final_d,
        tl.cast(acc, tl.float16),
        mask=mask_m[:, None] & mask_d[None, :],
    )
```

"""
TriMul “outgoing’’ forward pass – Triton‑accelerated.

The implementation follows the reference PyTorch module but replaces the two
LayerNorms with custom Triton kernels and fuses the second LayerNorm,
output‑gate multiplication and the final linear projection into a single
kernel.  This removes one pass over the (B·N·N·H) tensor and eliminates the
temporary `out_norm * out_gate` buffer, cutting memory traffic and runtime.

Algorithm
---------
1️⃣  Triton 2‑pass LayerNorm on the input tensor (shape B·N·N·D).  Result is
    cast to fp16.
2️⃣  Fuse the five linear layers (left/right projections + three gates) into
    a single FP16 GEMM.
3️⃣  Apply sigmoid to the three gate vectors and (optionally) the pairwise mask.
4️⃣  Contract the gated left/right tensors with a batched GEMM
    (torch.bmm) – this is the O(N³·H) heavy part and runs on cuBLAS/Tensor‑Core.
5️⃣  Fuse the second LayerNorm, the output‑gate scaling and the final linear
    projection (dim←H) into a custom Triton kernel.
6️⃣  Cast the final result back to fp32 (the public API expects float32).

The only PyTorch‑level operations are the fused projection (step 2), the
batched GEMM (step 4) and a few simple element‑wise ops.  All LayerNorm work
and the final projection are done in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """Two‑pass layer‑norm over the last axis."""
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: mean / variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for i in range(iters_n):
        offs_n = i * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: normalise + affine ----------
    for i in range(iters_n):
        offs_n = i * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr  + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Convenient wrapper for the 2‑pass LayerNorm kernel."""
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias   = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    BLOCK_M = 128
    BLOCK_N = 64
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x, y_ptr=y,
        weight_ptr=weight, bias_ptr=bias,
        M=M, N=N, eps=eps,
        stride_xm=stride_xm, stride_xn=stride_xn,
        stride_ym=stride_ym, stride_yn=stride_yn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Fuse the second LayerNorm, the output‑gate and the final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr, out_gate_ptr,      # input tensors   [M, H]
    norm_w_ptr, norm_b_ptr,     # LayerNorm params    H
    weight_ptr,                 # final linear weight [D, H]
    out_ptr_final,              # result tensor    [M, D]
    M, H, D, eps,
    stride_out_m, stride_out_h,
    stride_gate_m, stride_gate_h,
    stride_norm_w, stride_norm_b,
    stride_weight_h, stride_weight_d,
    stride_out_final_m, stride_out_final_d,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute for each position ``m`` (where m indexes a (b,i,j) triplet):
        y_m = ( ( (x_m - μ_m) / σ_m ) * γ + β ) * g_m   @   W^T
    where:
        x_m            – contracted tensor   (shape H)
        μ_m , σ_m      – mean/variance over H
        γ , β          – to_out_norm.weight / bias   (size H)
        g_m            – out_gate (size H)
        W              – to_out.weight (shape D×H)
    Output y_m has shape D.
    The implementation does a 2‑pass reduction over H, then a batched GEMM
    (the dot product) using only one read of ``x_m`` and ``g_m``.
    """
    pid_m = tl.program_id(0)   # row block
    pid_d = tl.program_id(1)   # column (output‑feature) block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_d = offs_d < D

    # ------- first pass: mean / variance across H -------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_h = tl.cdiv(H, BLOCK_H)
    for k in range(iters_h):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load x (pre‑norm tensor)
        x = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------- second pass: normalise, apply gates, multiply by W -------
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for k in range(iters_h):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load x and gate
        x = tl.load(
            out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        g = tl.load(
            out_gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        # load LayerNorm parameters for this H‑slice
        w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)
        b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)

        # cast everything to fp32 for the arithmetic
        x_f32   = tl.cast(x, tl.float32)
        g_f32   = tl.cast(g, tl.float32)
        w_n_f32 = tl.cast(w_norm, tl.float32)
        b_n_f32 = tl.cast(b_norm, tl.float32)

        # normalise + affine
        y = (x_f32 - mean[:, None]) * rstd[:, None]         # (Mblk, Hblk)
        y = y * w_n_f32[None, :] + b_n_f32[None, :]         # affine
        y = y * g_f32                                        # output‑gate

        # load final linear weight block   shape (Hblk, Dblk)
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_weight_h + offs_d[None, :] * stride_weight_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )
        w_f32 = tl.cast(w, tl.float32)

        # GEMM over the H‑dimension
        acc += tl.dot(y, w_f32)   # (Mblk, Dblk)
    
    # store result (cast back to fp16)
    tl.store(
        out_ptr_final + offs_m[:, None] * stride_out_final_m + offs_d[None, :] * stride_out_final_d,
        tl.cast(acc, tl.float16),
        mask=mask_m[:, None] & mask_d[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Input  : (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]   (float32)
        - mask         : torch.Tensor  [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` (model dim) and
                         ``hidden_dim`` (projection dim). Optional key
                         ``nomask`` (bool) indicates that *mask* may be omitted.

    Output : torch.Tensor  [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype   # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) → fp16
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)            # (M, D) fp32
    norm_w = weights["norm.weight"].to(device, dtype)  # (D,)
    norm_b = weights["norm.bias"].to(device, dtype)    # (D,)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B, N, N, D) fp32
    x_norm_f16 = x_norm.to(torch.float16)                  # fp16 for the heavy ops
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)           # (M, D) fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (left/right proj + three gates)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                # (5·H, D) fp16

    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)   # (M, 5·H) fp16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)      # (B,N,N,H) fp16
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)   # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)  # (B·H, N, N)

    # (B·H, N, N) * (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse 2nd LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out_flat      = out.view(M, hidden_dim)          # (M, H) fp16
    out_gate_flat = out_gate.view(M, hidden_dim)    # (M, H) fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)   # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)     # (H,)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)

    # fused kernel
    out_final_fp16 = fused_norm_gate_linear(
        out=out_flat,
        out_gate=out_gate_flat,
        norm_w=to_out_norm_w,
        norm_b=to_out_norm_b,
        weight=to_out_w,
        eps=eps,
    )   # shape (M, dim) fp16

    out_fp32 = out_final_fp16.view(bs, seq_len, seq_len, dim).to(torch.float32)
    return out_fp32
