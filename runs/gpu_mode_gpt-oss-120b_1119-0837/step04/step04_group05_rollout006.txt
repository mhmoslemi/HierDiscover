@triton.jit
def tril_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # column block
    pid_b = tl.program_id(2) # batch block (B*H)

    # compute i0, j0 offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # compute b and h indices
    b = pid_b // H
    h = pid_b % H

    # compute base offsets for this batch
    left_batch_offset = b * stride_left_b + h * stride_left_h
    right_batch_offset = b * stride_right_b + h * stride_right_h
    out_batch_offset = b * stride_out_b + h * stride_out_h

    # define range offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute pointer to start of block in left and right
    # We'll accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k) + left_batch_offset
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right block shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j) + right_batch_offset
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j) + out_batch_offset
    # convert to fp16
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized outgoing TriMul forward pass for AlphaFold3‑style models.

Algorithm
---------
1️⃣ Input Layer‑Norm (FP32) – implemented in Triton.
2️⃣ Five linear projections (left, right, left_gate, right_gate, out_gate) are fused
   into a single FP16 GEMM.
3️⃣ Optional binary mask (broadcast over the hidden dimension) is applied.
4️⃣ Gating (sigmoid) is performed in FP16.
5️⃣ Core contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is executed by a custom Triton kernel that treats (b,h) as a batched
   dimension, avoiding the expensive permute+contiguous copies required
   by the naïve torch.bmm implementation.
6️⃣ Output Layer‑Norm (FP32) – Triton.
7️⃣ Output‑gate is applied and the final half‑precision linear projection
   maps back to the model dimension.
8️⃣ Result is cast to float32.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while the two
LayerNorms run in FP32 for numerical stability.  The kernel in step 5
is the main speed‑up: it eliminates two large memory copies and uses a
single batched GEMM fused with masking/gating logic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (FP32 accumulation, FP16/FMMA‑friendly output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output [M, C]
    weight_ptr, bias_ptr,       # *ptr to LN weight / bias [C]
    M, C, eps,                  # dimensions
    stride_xm, stride_xc,       # stride of input
    stride_ym, stride_yc,       # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_C: int = 64) -> torch.Tensor:
    """Layer‑norm over the last dimension of a 2‑D tensor using Triton."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x, dtype=torch.float32)

    # make contiguous
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y

# ----------------------------------------------------------------------
# Batched GEMM kernel used for the core TriMul contraction.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)   # tile over i dimension
    pid_n = tl.program_id(1)   # tile over j dimension
    pid_b = tl.program_id(2)   # (b,h) pair

    # compute i/j offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # recover batch and head index
    b = pid_b // H
    h = pid_b %  H

    # base offsets for the selected (b,h)
    left_base  = b * stride_left_b  + h * stride_left_h
    right_base = b * stride_right_b + h * stride_right_h
    out_base   = b * stride_out_b   + h * stride_out_h

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulation buffer (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + left_base + \
                    (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block: we need (BLOCK_K, BLOCK_N) (k as rows, j as cols)
        right_ptrs = right_ptr + right_base + \
                     (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j)
        b_mat = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # fused FP16×FP16→FP32 dot‑product
        acc += tl.dot(a, b_mat)   # a:(M,K) b_mat:(K,N) → (M,N)

    # write result back as fp16
    out_ptrs = out_ptr + out_base + \
               (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * N * N, dim).contiguous()
    x_norm_fp = triton_layernorm(
        x_flat,
        weights["norm.weight"].to(device, dtype=torch.float32),
        weights["norm.bias"].to(device, dtype=torch.float32),
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_C=64,
    )
    x_norm = x_norm_fp.view(bs, N, N, dim)          # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()
    x_norm_h_flat = x_norm_h.view(-1, dim)         # (B*N*N, dim)

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    proj = F.linear(x_norm_h_flat, combined_weight)  # (B*N*N, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split the concatenated projection back into its components
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, N, N, hidden_dim)
    right_raw     = right_raw.view(bs, N, N, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, N, N, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, N, N, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in FP16 for speed
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw * left_gate   # (B,N,N,H) FP16
    right = right_raw * right_gate # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Core contraction via custom Triton batched GEMM
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # (B,N,N,H) FP16, uninitialized

    # strides (in elements) of the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h   = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h       = out.stride()

    # BLOCK parameters – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),        # tiles over i
        triton.cdiv(N, BLOCK_N),        # tiles over j
        bs * hidden_dim,                # (b,h) pairs
    )

    batched_trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_i,   # j‑dimension uses the same stride as the 1st N‑axis
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # free intermediate tensors earlier to lower peak memory usage
    del left, right, left_raw, right_raw, left_gate_raw, right_gate_raw

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp = out.float()
    out_flat = out_fp.view(bs * N * N, hidden_dim)
    out_norm_flat = triton_layernorm(
        out_flat,
        weights["to_out_norm.weight"].to(device, dtype=torch.float32),
        weights["to_out_norm.bias"].to(device, dtype=torch.float32),
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_C=64,
    )
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate.float()               # (B,N,N,H) FP32
    final_fp16 = F.linear(gated.half(),
                          weights["to_out.weight"].to(device, dtype=torch.float16))  # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()
    return final_fp32
