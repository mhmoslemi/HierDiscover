@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh_idx = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Compute batch and hidden indexes
    b = bh_idx // H
    h = bh_idx % H

    # Starting offsets for i, j
    i_start = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_start < N
    mask_j = j_start < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # compute offsets for left tile
        # left offset = left_ptr + b*stride_left_batch + i_start*stride_left_i + k_start*stride_left_k + h*stride_left_h
        left_ptrs = left_ptr + \
            b * stride_left_batch + \
            i_start[:, None] * stride_left_i + \
            k_start[None, :] * stride_left_k + \
            h * stride_left_h

        # Similarly right ptrs for (j,k)
        right_ptrs = right_ptr + \
            b * stride_right_batch + \
            j_start[:, None] * stride_right_j + \
            k_start[None, :] * stride_right_k + \
            h * stride_right_h

        # Load tiles with masks
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Cast to fp16 for dot? left_tile and right_tile are half
        # But we want to compute dot with fp16 input and fp32 accumulate.
        # tl.dot expects two 2D blocks, returns fp32 by default? Actually dot returns same dtype? It can produce FP32 accumulate if we cast.
        acc += tl.dot(left_tile, right_tile)  # returns fp32

    # Write result to out
    # out ptr compute:
    out_ptrs = out_ptr + \
        b * stride_out_batch + \
        i_start[:, None] * stride_out_i + \
        j_start[None, :] * stride_out_j + \
        h * stride_out_h

    # Cast acc to fp16 for storage
    out_tile = acc.to(tl.float16)

"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Apply LayerNorm on the full pair‑wise tensor (float32, cuDNN).
2. Cast to fp16 and fuse the five linear projections (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single GEMM.
3. In‑place compute gated left/right tensors (and optionally apply the pair mask)
   – no extra allocations.
4. Perform the N³ contraction with a custom Triton kernel that treats the hidden
   dimension as a batch dimension, avoiding the costly permute‑contiguous copies
   required by torch.bmm.
5. Layer‑norm the hidden‑dim result (fp16, cuDNN), apply the out‑gate,
   and project back to the model dimension with a final fp16 GEMM.
6. Cast the final output back to float32.

Only the N³ contraction is written in Triton; all other heavy ops use the
high‑performance PyTorch/cuDNN kernels.  This reduces memory traffic and
kernel launch overhead, meeting the sub‑millisecond target on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel for the core contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,            # fp16*   left tensor  (B,N,N,H)
    right_ptr,           # fp16*   right tensor (B,N,N,H)
    out_ptr,             # fp16*   output tensor (B,N,N,H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # program identifiers
    # --------------------------------------------------------------
    bh = tl.program_id(0)          # combined batch * hidden index
    ib = tl.program_id(1)          # block row   (i)
    jb = tl.program_id(2)          # block col   (j)

    # --------------------------------------------------------------
    # decode batch and hidden indices
    # --------------------------------------------------------------
    b = bh // H
    h = bh % H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = ib * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = jb * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the current k‑tile of left and right
        left_off = (
            b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        right_off = (
            b * stride_r_b
            + offs_j[:, None] * stride_r_i    # note: use i‑stride for the “j” index
            + offs_k[None, :] * stride_r_k
            + h * stride_r_h
        )

        # load tiles (mask out‑of‑bounds elements)
        left_tile = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        right_tile = tl.load(
            right_ptr + right_off,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )

        # dot‑product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # write the tile back to the output tensor
    # --------------------------------------------------------------
    out_off = (
        b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    out_tile = acc.to(tl.float16)
    tl.store(
        out_ptr + out_off,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[bs, N, N, dim] (float32)
            mask         : torch.Tensor[bs, N, N] or None (float / bool)
            weights      : dict of torch.Tensor (model params)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 – cuDNN)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, D), float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and fuse all five linear projections
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # concatenate weight matrices: [left_proj, left_gate, right_proj,
    #                               right_gate, out_gate]   → (5*H, D)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)          # (5*H, D)

    # fused projection: (B*N*N, D) @ (D, 5*H) → (B*N*N, 5*H)
    x_flat = x_f16.view(-1, dim)         # (M, D), M = B*N*N
    proj_flat = F.linear(x_flat, w_fused)  # (M, 5*H)
    proj = proj_flat.view(bs, N, N, 5 * H)  # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply sigmoid gates and optional pair‑mask (in‑place)
    # ------------------------------------------------------------------
    h0 = 0
    h1 = H
    h2 = 2 * H
    h3 = 3 * H
    h4 = 4 * H
    h5 = 5 * H

    left       = proj[..., h0:h1]   # (B,N,N,H)
    left_gate  = torch.sigmoid(proj[..., h1:h2])
    right      = proj[..., h2:h3]   # (B,N,N,H)
    right_gate = torch.sigmoid(proj[..., h3:h4])
    out_gate   = torch.sigmoid(proj[..., h4:h5])  # (B,N,N,H)

    # in‑place gating
    left.mul_(left_gate)
    right.mul_(right_gate)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left.mul_(mask_f16)
        right.mul_(mask_f16)

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction – custom Triton batched GEMM
    # ------------------------------------------------------------------
    # out will have shape (B,N,N,H)
    out = torch.empty_like(left)  # fp16 tensor

    # strides (in elements) for left/right/out
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # grid dimensions: (BH, blocks_i, blocks_j)
    BH = bs * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=H,
        stride_l_b=sL_b,
        stride_l_i=sL_i,
        stride_l_k=sL_k,
        stride_l_h=sL_h,
        stride_r_b=sR_b,
        stride_r_i=sR_i,
        stride_r_k=sR_k,
        stride_r_h=sR_h,
        stride_o_b=sO_b,
        stride_o_i=sO_i,
        stride_o_j=sO_j,
        stride_o_h=sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16 – cuDNN)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # (B,N,N,H), fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to D
    # ------------------------------------------------------------------
    gated = out_norm * out_gate  # fp16, shape (B,N,N,H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    gated_flat = gated.view(-1, H)   # (M, H)
    out_fp16 = F.linear(gated_flat, to_out_w)  # (M, D)

    # reshape and cast back to float32
    output = out_fp16.view(bs, N, N, dim).to(torch.float32)
    return output
