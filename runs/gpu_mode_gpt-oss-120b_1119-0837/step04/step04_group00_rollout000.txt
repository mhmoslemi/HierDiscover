"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ LayerNorm on the model dimension (FP16) → custom Triton kernel.
2️⃣ All heavy linear work stays in FP16:
   - Fuse projection + gate for left and right sides in a single matmul
     (weight shape (2·H, D) → split after matmul).
   - Optional binary mask is applied to the projected tensors before gating.
   - Apply sigmoid gates and multiply.
3️⃣ Core N³ contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   implemented as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ using torch.bmm.
4️⃣ Hidden‑dim LayerNorm (FP16) → custom Triton kernel.
5️⃣ Apply output gate (sigmoid of a linear on the normalized *input*).
6️⃣ Final linear projection back to model dim (FP16) → cast to float32.
   Two Triton kernels (FP16 LayerNorms) guarantee “kernel” usage.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# Two‑pass LayerNorm – FP16 → FP16 (mean/var in FP32)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # -------------------------------------------------
    # Row range for this program
    # -------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -------------------------------------------------
    # First pass: compute mean & variance (in FP32)
    # -------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -------------------------------------------------
    # Second pass: write normalized output
    # -------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm on a 2‑D tensor (M, N) where N = feature dim.
    Input/Output dtype: torch.float16.
    """
    assert x.dim() == 2 and x.dtype == torch.float16
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors (important for stride extraction)
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” operator
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of parameter tensors (float32)
        - config       : dict with keys:
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ---------------------------- unpack -------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – operate directly in FP16 to cut bandwidth
    # ------------------------------------------------------------------
    # Cast input to half‑precision first
    x_fp16 = input_tensor.to(torch.float16)                # (B,N,N,D) FP16
    M = bs * N * N
    x_flat_fp16 = x_fp16.reshape(M, dim)                  # (B·N·N, D)

    # LayerNorm parameters (cast to FP16)
    norm_weight = weights["norm.weight"].to(device, torch.float16)
    norm_bias   = weights["norm.bias"].to(device, torch.float16)

    x_norm_flat_fp16 = triton_layernorm_fp16(
        x_flat_fp16, norm_weight, norm_bias, eps=eps
    )
    x_norm = x_norm_flat_fp16.view(bs, N, N, dim)          # (B,N,N,D) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left and right sides
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (2·H, D)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)  # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)  # (2H, D)

    # Linear → (B,N,N,2H) for both sides
    left_comb  = F.linear(x_norm, left_comb_w)   # (B,N,N,2H) FP16
    right_comb = F.linear(x_norm, right_comb_w)  # (B,N,N,2H) FP16

    # Split projection & gate heads
    left_proj = left_comb[..., :H]                # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]            # (B,N,N,H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    # Apply optional mask *before* gating (mask broadcast over hidden dim)
    if not nomask and mask is not None:
        # Convert mask to half‑precision and broadcast
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Sigmoid gates (in‑place to reduce allocations)
    left_gate  = torch.sigmoid(left_gate_pre)     # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # Apply gates (in‑place)
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))            # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction – batched FP16 GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)               # (B·H, N, N)

    # Batched matrix multiply: C = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm (FP16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)   # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate & final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H) FP16
    gated = out_norm * out_gate_flat                     # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output_flat = F.linear(gated, to_out_w)             # (M, D) FP16

    # Cast back to float32 as required by the API
    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
