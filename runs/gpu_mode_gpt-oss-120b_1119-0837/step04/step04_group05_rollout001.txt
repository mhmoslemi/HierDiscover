@triton.jit
def gate_and_apply_kernel(
    raw_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_raw_m, stride_raw_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m, stride_mask_h,  # mask has no hidden dimension but we treat stride_mask_h=0
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    
    mask_m = offs_m < M
    mask_h = offs_h < H
    
    raw = tl.load(raw_ptr + offs_m[:, None] * stride_raw_m + offs_h[None, :] * stride_raw_h,
                  mask=mask_m[:, None] & mask_h[None, :], other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
                  mask=mask_m[:, None] & mask_h[None, :], other=0.0)

@triton.jit
def gate_and_mask_sigmoid_kernel(
    raw_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_raw_m, stride_raw_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_m = offs_m < M
    mask_h = offs_h < H

    # Load raw and gate
    raw = tl.load(raw_ptr + offs_m[:, None] * stride_raw_m + offs_h[None, :] * stride_raw_h,
                  mask=mask_m[:, None] & mask_h[None, :],
                  other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
                   mask=mask_m[:, None] & mask_h[None, :],
                   other=0.0)
    # Compute sigmoid of gate (gate is half)
    gate_f = gate.to(tl.float32)
    sig_f = 1.0 / (1.0 + tl.exp(-gate_f))
    sig = sig_f.to(tl.float16)

    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m,
                       mask=mask_m,
                       other=0.0).to(tl.float16)  # shape (BLOCK_M,)
        mask = mask[:, None]  # broadcast across H
        out = raw * sig * mask
    else:
        out = raw * sig

    tl.store(out_ptr + offs_m[:, None] * stride_raw_m + offs_h[None, :] * stride_raw_h,
             out,
             mask=mask_m[:, None] & mask_h[None, :])
```

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 “outgoing” variant).

Algorithm
---------
1️⃣ Input LayerNorm (FP32) – performed with the fast native torch.nn.functional
   implementation (cuDNN).  This is numerically stable and substantially
   faster than our previous custom Triton implementation.

2️⃣ Five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
   fused into a single half‑precision GEMM by concatenating the weight
   matrices.  The result has shape (B·N·N, 5·H).

3️⃣ A custom Triton kernel fuses the *mask* (if present) and the *sigmoid*
   gating for the left and right streams:
      left  = left_raw  * sigmoid(left_gate_raw)  * mask
      right = right_raw * sigmoid(right_gate_raw) * mask
   The kernel works on the flattened (M, H) tensors (M = B·N·N) and
   therefore maximises memory‑bandwidth reuse.

4️⃣ The core contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realized as a batched matrix‑multiply (torch.bmm) after reshaping to
   (B·H, N, N) × (B·H, N, N)ᵀ.  This leverages Tensor‑Cores on the H100.

5️⃣ Output LayerNorm (FP32) – again the native torch implementation.

6️⃣ Output gating (sigmoid) and the final projected linear layer are
   performed entirely in FP16 (Tensor‑Core friendly) and the result is
   cast back to FP32.

At least one operation (the mask‑gating fusion) is executed in a Triton
kernel, satisfying the problem requirements while achieving a large speed‑up
compared with a purely‑Python implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : fused mask & sigmoid (used for left/right streams)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_sigmoid_kernel(
    raw_ptr,            # pointer to raw tensor   (M, H)  half
    gate_ptr,           # pointer to gate tensor  (M, H)  half
    mask_ptr,           # pointer to mask tensor  (M,)    half
    out_ptr,            # pointer to output       (M, H)  half (may be same as raw_ptr)
    M,                  # total number of rows  = B * N * N
    H,                  # hidden dimension
    stride_raw_m,       # stride for rows in raw  = H
    stride_raw_h,       # stride for cols in raw  = 1
    stride_gate_m,      # stride for rows in gate = H
    stride_gate_h,      # stride for cols in gate = 1
    stride_mask_m,      # stride for rows in mask = 1
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # Program IDs define a tile of size (BLOCK_M, BLOCK_H)
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # Load raw and gate values
    raw = tl.load(
        raw_ptr + offs_m[:, None] * stride_raw_m + offs_h[None, :] * stride_raw_h,
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    )

    # ---- sigmoid(gate) (computed in FP32 for accuracy) ----
    gate_f32 = gate.to(tl.float32)
    sig_f32 = 1.0 / (1.0 + tl.exp(-gate_f32))
    sig = sig_f32.to(tl.float16)

    # ---- optional mask multiplication (broadcast over H) ----
    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float16)
        mask = mask[:, None]               # (BLOCK_M, 1) -> broadcast over H
        out = raw * sig * mask
    else:
        out = raw * sig

    # Store result (may overwrite raw)
    tl.store(
        out_ptr + offs_m[:, None] * stride_raw_m + offs_h[None, :] * stride_raw_h,
        out,
        mask=mask_m[:, None] & mask_h[None, :],
    )

# ----------------------------------------------------------------------
# Helper: apply mask + sigmoid gating to a (M, H) tensor pair
# ----------------------------------------------------------------------
def apply_gate_and_mask(
    raw: torch.Tensor,                     # (M, H) half
    gate: torch.Tensor,                    # (M, H) half
    mask: torch.Tensor | None,             # (M,) half or None
    hidden_dim: int,
    block_m: int = 128,
    block_h: int = 64,
) -> torch.Tensor:
    """
    Returns gated (and optionally masked) tensor with same shape as raw.
    The operation is performed in‑place on `raw` to avoid extra allocations.
    """
    M = raw.shape[0]
    H = hidden_dim
    # Strides for a contiguous (M, H) layout
    stride_m = H
    stride_h = 1
    has_mask = 1 if mask is not None else 0
    # Dummy mask pointer when not used (never dereferenced because HAS_MASK=0)
    dummy_mask = raw if mask is None else mask

    grid = (triton.cdiv(M, block_m), triton.cdiv(H, block_h))
    gate_mask_sigmoid_kernel[grid](
        raw_ptr=raw,
        gate_ptr=gate,
        mask_ptr=mask if mask is not None else dummy_mask,
        out_ptr=raw,                # in‑place write
        M=M,
        H=H,
        stride_raw_m=stride_m,
        stride_raw_h=stride_h,
        stride_gate_m=stride_m,
        stride_gate_h=stride_h,
        stride_mask_m=1,
        BLOCK_M=block_m,
        BLOCK_H=block_h,
        HAS_MASK=has_mask,
        num_warps=4,
        num_stages=2,
    )
    return raw


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]       (float32) or None
        - weights_dict : mapping from string to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True -> no mask, False -> mask present

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – using the highly‑optimized cuDNN kernel
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    # collapse the last three dims for layer‑norm
    x_flat = input_tensor.reshape(bs * seq_len * seq_len, dim)
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).view(bs, seq_len, seq_len, dim)       # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single half‑precision GEMM
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()                # cast once and reuse
    x_norm_h_flat = x_norm_h.view(-1, dim)   # (B·N·N, dim)

    # concatenate weight matrices (all are bias‑free)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)   # (5·H, dim)

    proj = F.linear(x_norm_h_flat, combined_weight)   # (M, 5·H) half
    M = proj.shape[0]                                 # B·N·N

    # ------------------------------------------------------------------
    # 3️⃣ Split concatenated projection back into its components
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )   # each is (M, H) half

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension) + gating
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.to(device, dtype=torch.float16).reshape(-1)  # (M,)
    else:
        mask_h = None

    # Apply mask + sigmoid gating in‑place on left_raw / right_raw
    left = apply_gate_and_mask(
        raw=left_raw,
        gate=left_gate_raw,
        mask=mask_h,
        hidden_dim=hidden_dim,
    )
    right = apply_gate_and_mask(
        raw=right_raw,
        gate=right_gate_raw,
        mask=mask_h,
        hidden_dim=hidden_dim,
    )
    # out_gate: only sigmoid (no mask)
    out_gate = torch.sigmoid(out_gate_raw).half()          # (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    left_4d = left.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) half
    right_4d = right.view(bs, seq_len, seq_len, hidden_dim)

    left_b = left_4d.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    right_b = right_4d.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # core contraction: left @ rightᵀ   (FP16, Tensor‑Core friendly)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))      # (B·H, N, N) half
    out = out_b.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32) – cuDNN implementation again
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    out_flat = out_fp32.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    ).view(bs, seq_len, seq_len, hidden_dim)               # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection (all in FP16)
    # ------------------------------------------------------------------
    out_gate_4d = out_gate.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) half

    # gated hidden representation (FP16)
    gated_fp16 = (out_norm.half() * out_gate_4d)                    # (B,N,N,H) half

    # final projection back to model dimension
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)   # (dim, H)
    final_fp16 = F.linear(gated_fp16, to_out_weight)          # (B,N,N,dim) half
    final_fp32 = final_fp16.float()                           # cast back to FP32

    return final_fp32
