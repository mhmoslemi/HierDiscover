@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, g_ptr,
    M, N, eps,
    stride_xm, stride_xn, stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None]*stride_xm + offs_n[None,:]*stride_xn,
                    mask=mask_m[:, None] & mask_n[None,:], other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean*mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None]*stride_xm + offs_n[None,:]*stride_xn,
                    mask=mask_m[:, None] & mask_n[None,:], other=0.0)
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        g = tl.load(g_ptr + offs_m[:, None]*stride_gm + offs_n[None,:]*stride_gn,
                    mask=mask_m[:, None] & mask_n[None,:], other=0.0)  # gating per element

        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)
        g_f32 = g.to(tl.float32)

        y_f32 = (norm_x * w_f32 + b_f32) * g_f32
        y = y_f32.to(x.dtype)

        tl.store(y_ptr + offs_m[:, None]*stride_ym + offs_n[None,:]*stride_yn,
                 y,
                 mask=mask_m[:, None] & mask_n[None,:])

Potential issues: gating pointer stride across rows may be large. Provide stride_gm, stride_gn computed via out_gate_stride().

"""TriMul (Outgoing) – fused Triton + PyTorch implementation
==========================================================

The forward pass consists of
1️⃣ Input LayerNorm (row‑wise, implemented in Triton)
2️⃣ A single fused Linear that simultaneously computes the two projections,
   the three gates and the output gate (weights are concatenated).
3️⃣ Optional pair‑wise mask applied to the two projections.
4️⃣ Sigmoid gates and element‑wise gating.
5️⃣ The N³ contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   performed as a batched GEMM (cuBLAS `torch.bmm`).
6️⃣ Output LayerNorm + output‑gate fused in one Triton kernel.
7️⃣ Final Linear projection back to the model dimension and cast to float32.

The heavy linear projections (5 of them in the reference code) are reduced to
a single matrix‑multiply, and the two layer‑norms are written in Triton, which
removes the Python‐level kernel launch overhead.  All heavy work runs in
FP16 (Tensor‑cores), only the final result is returned in FP32."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,               # data pointers
    M, N, eps,                                # dimensions
    stride_xm, stride_xn, stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Row‑wise LayerNorm (no gating)."""
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: mean & variance ----------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalize, affine transform --------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y_f32 = norm * w.to(tl.float32) + b.to(tl.float32)
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, g_ptr,          # data pointers
    M, N, eps,                                   # dimensions
    stride_xm, stride_xn, stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Row‑wise LayerNorm with an extra per‑element gate (out_gate)."""
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute mean / var -------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalize, affine, gate -----------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # load data
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        g = tl.load(
            g_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        g_f32 = g.to(tl.float32)

        # normalize
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (weight & bias)
        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y_f32 = (norm * w + b) * g_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm (no gating) on a 2‑D tensor."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguity for stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    BLOCK_M = 256
    BLOCK_N = 128
    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]     (float32) or None
        - weights      : dict of torch tensors with the required keys
        - config       : dict with fields ``dim`` (D), ``hidden_dim`` (H)
                         and optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype               # expected torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise Triton)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()

    # cast to fp16 for heavy work
    x_fp16 = input_tensor.to(torch.float16)
    x_flat = x_fp16.view(bs * L * L, dim)                     # (M, D)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps)  # (M, D)
    x_norm = x_norm_flat.view(bs, L, L, dim)                  # (B, L, L, D)

    # ------------------------------------------------------------------
    # 2️⃣ Fused Linear: left_proj, right_proj, left_gate, right_gate, out_gate
    # ------------------------------------------------------------------
    # collect all projection & gate weights (shape: (H, D))
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    # concatenate on the out‑features dimension: (5*H, D)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )  # shape (5*H, D)

    # single fused matmul
    x_norm_flat = x_norm.view(-1, dim)                # (M, D)
    combined_out_flat = F.linear(x_norm_flat, combined_weight)  # (M, 5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice and reshape the fused output
    # ------------------------------------------------------------------
    M = bs * L * L
    H = hidden_dim
    combined_out = combined_out_flat.view(M, 5 * H)

    left      = combined_out[:, 0:H].view(bs, L, L, H)
    right     = combined_out[:, H:2*H].view(bs, L, L, H)
    left_gate = combined_out[:, 2*H:3*H].view(bs, L, L, H)
    right_gate= combined_out[:, 3*H:4*H].view(bs, L, L, H)
    out_gate  = combined_out[:, 4*H:5*H].view(bs, L, L, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied before the gates)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and apply to projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B*H, L, L) for bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_perm.view(bs * H, L, L)            # (B*H, L, L)
    right_flat = right_perm.view(bs * H, L, L)

    # out_flat: (B*H, L, L) = left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, L, L, H)
    out = out_flat.view(bs, H, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out_gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()

    out_flat2   = out.view(-1, H).contiguous()          # (M, H)
    out_gate_f  = out_gate.view(-1, H).contiguous()    # (M, H)

    # fused LN + gate
    out_ln_gate = triton_layernorm_gate(out_flat2, out_norm_w, out_norm_b, out_gate_f, eps)
    out = out_ln_gate.view(bs, L, L, H)                # (B, L, L, H)

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_flat3 = out.view(-1, H).contiguous()          # (M, H)
    out_proj  = F.linear(out_flat3, to_out_w)         # (M, D)  fp16
    out_proj  = out_proj.view(bs, L, L, dim).to(torch.float32)

    return out_proj
