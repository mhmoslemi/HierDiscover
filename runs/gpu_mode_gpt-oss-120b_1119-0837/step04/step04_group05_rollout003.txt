""" 
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 version).

Key ideas
---------
1️⃣ Input LayerNorm (FP32 accumulation, FP16 output) – Triton.
2️⃣ Five linear projections (left, right, left_gate, right_gate, out_gate) are
   fused into a single half‑precision GEMM.  A single sigmoid call processes
   all three gate vectors at once, reducing kernel launch overhead.
3️⃣ Optional binary mask is broadcast over the hidden dimension and applied
   directly to the masked projections.
4️⃣ Gating (sigmoid) and mask are fused with the projections, yielding the
   final left / right tensors in FP16.
5️⃣ Core contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is performed as a batched half‑precision GEMM via `torch.bmm`.
6️⃣ Output LayerNorm (FP32 accumulation, FP16 output) – Triton.
7️⃣ Output gate (sigmoid) is applied in FP16, followed by the final half‑
   precision linear projection back to the model dimension.
8️⃣ Result is cast to float32 for the user‑facing API.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while both LayerNorms
use FP32 accumulation for numerical stability.  Only the LayerNorms are
implemented in Triton, satisfying the requirement to contain at least one
custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 accumulation, support for FP16/FP32 I/O)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                # *ptr to input / output [M, C]
    weight_ptr, bias_ptr,        # *ptr to LN weight / bias [C]
    M, C, eps,                   # dimensions
    stride_xm, stride_xc,        # stride of input
    stride_ym, stride_yc,        # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]          # FP32 result

        # Store with automatic casting to the dtype of `y_ptr`
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_C: int = 64,
) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a 2‑D tensor ``x`` (shape M×C)
    using a fused Triton kernel.  The input / output dtype is preserved
    (either float32 or float16).  ``weight``/``bias`` are expected to be
    float32 – they are cast internally.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)  # same dtype as x

    # make sure strides are contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# TriMul forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 formulation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(-1, dim).contiguous()                 # (B·N·N, dim) FP32
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    # LayerNorm returns FP32 tensor; we cast it to FP16 once.
    x_norm_fp32 = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)                               # (M, dim) FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left, right, gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices (shape (5*hidden_dim, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                                   # (5*H, dim)

    proj = F.linear(x_norm, combined_weight)                              # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split and process projections
    # --------------------------------------------------------------
    H = hidden_dim
    left_raw   = proj[:, :H]                # (M, H)
    right_raw  = proj[:, H:2*H]             # (M, H)
    gate_raw   = proj[:, 2*H:]              # (M, 3*H)

    # One‑shot sigmoid for all gates
    gate_sig   = torch.sigmoid(gate_raw)    # (M, 3*H)
    left_gate  = gate_sig[:, :H]            # (M, H)
    right_gate = gate_sig[:, H:2*H]         # (M, H)
    out_gate   = gate_sig[:, 2*H:]          # (M, H)

    # Reshape to 4‑D tensors
    left_raw   = left_raw.view(bs, seq_len, seq_len, H)
    right_raw  = right_raw.view(bs, seq_len, seq_len, H)
    left_gate  = left_gate.view(bs, seq_len, seq_len, H)
    right_gate = right_gate.view(bs, seq_len, seq_len, H)
    out_gate   = out_gate.view(bs, seq_len, seq_len, H)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Apply gating (still half‑precision)
    # --------------------------------------------------------------
    left  = left_raw * left_gate            # (B,N,N,H) FP16
    right = right_raw * right_gate          # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Core contraction via batched GEMM (Tensor‑Core friendly)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for `torch.bmm`
    left_b  = left.permute(0, 3, 1, 2).reshape(bs * H, seq_len, seq_len)
    right_b = right.permute(0, 3, 1, 2).reshape(bs * H, seq_len, seq_len)

    # out_b = left_b @ right_bᵀ   (B·H, N, N)  FP16
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # back to 4‑D tensor (B,N,N,H)
    out = out_b.view(bs, H, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP16 I/O, FP32 accumulation) – Triton
    # --------------------------------------------------------------
    out_flat = out.view(-1, H)                                            # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    out_norm = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm.view(bs, seq_len, seq_len, H)                     # FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate, final projection, cast to FP32
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                            # (B,N,N,H) FP16
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)   # (dim, H)

    # Linear projection back to model dimension (half‑precision GEMM)
    final_fp16 = F.linear(gated.view(-1, H), to_out_w)                     # (B·N·N, dim) FP16

    # Reshape & convert to float32 for the API
    final = final_fp16.view(bs, seq_len, seq_len, dim).float()
    return final
