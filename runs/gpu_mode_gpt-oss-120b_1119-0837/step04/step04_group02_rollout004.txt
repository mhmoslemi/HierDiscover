@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BH, N,  # shapes
    stride_ah, stride_ai, stride_ak,  # a: (batch, i, k): batch stride, i stride, k stride
    stride_bh, stride_bi, stride_bk,  # b: (batch, j, k) but we treat transposed mapping, thus we use b stride for j and k
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr   # maybe not needed
):
    b = tl.program_id(0) # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute block start k
        k_offset = k * BLOCK_K
        # offsets for k dimension within this block
        cur_k = offs_k + k_offset
        mask_cur_k = cur_k < N

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + b * stride_ah + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_cur_k[None, :], other=0.0).to(tl.float32)

        # load B (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + b * stride_bh + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs, mask=mask_cur_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute block product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + b * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    # Possibly cast to fp16 if required
    # We assume output dtype same as a (fp16)
    # TL dot gave fp32, need to cast back
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Triton‑accelerated forward pass of the “outgoing’’ TriMul operator.

Algorithm
---------
1️⃣ Layer‑Norm the input (B,N,N,D) using the built‑in torch.nn.functional.layer_norm
   (fast cuDNN kernel).  The tensor is then cast to FP16.
2️⃣ Fuse the five dense projections (left/right projections + three gates) into a
   single GEMM (weight shape (5·H,D)).  Slice the result and apply sigmoid to the
   gating tensors.
3️⃣ Apply the pair‑wise mask (if present) and element‑wise gating to obtain the
   tensors *left* and *right* (both (B,N,N,H) FP16).
4️⃣ Compute the heavy O(N³·H) contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   with a custom Triton kernel that performs a batched matrix‑multiply over the
   hidden dimension H.  The kernel works on tiles (BLOCK_M×BLOCK_N) and uses
   FP32 accumulation for numerical stability.
5️⃣ Layer‑Norm the result (B,N,N,H) with torch.nn.functional.layer_norm.
6️⃣ Apply the output‑gate, the final linear projection back to the model
   dimension D and return a FP32 tensor.

Only the contraction is written in Triton – this satisfies the “at least one
kernel’’ requirement while keeping the rest of the pipeline in highly‑optimised
PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel (computes Σₖ left[...,k,:] * right[...,k,:]ᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # *ptr left  [B·H, N, N]  (FP16)
    b_ptr,               # *ptr right [B·H, N, N]  (FP16), will be read transposed
    c_ptr,               # *ptr out   [B·H, N, N]  (FP16)
    BH,                  # number of batch‑heads = B·H
    N,                   # sequence length
    stride_ah, stride_ai, stride_ak,   # strides for a: (batch, i, k)
    stride_bh, stride_bi, stride_bk,   # strides for b: (batch, i, k) – we will read it as (k, j)
    stride_ch, stride_ci, stride_cj,   # strides for c: (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # which head (B·H)
    pid_m   = tl.program_id(1)          # i‑tile
    pid_n   = tl.program_id(2)          # j‑tile

    # Offsets of the current tile within the N×N matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 to avoid overflow
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_ah
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load a tile of B transposed: (BLOCK_K, BLOCK_N)
        #   B is stored as (batch, i, k).  We want (k, j) → use stride_bk for k,
        #   stride_bi for j.
        b = tl.load(
            b_ptr
            + batch_id * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # GEMM on the current tile
        acc += tl.dot(a, b)

    # Write back results (FP16)
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch_id * stride_ch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch – fast cuDNN implementation)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the compute‑heavy parts
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                           # (B,N,N,D) fp16
    M = bs * seq_len * seq_len                              # rows for fused linear

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gating (one GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (left_proj, left_gate, right_proj,
    # right_gate, out_gate).  Shapes: each (H, D) → combined (5·H, D).
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)  # (5·H, D) fp16

    # Perform the fused linear transformation
    proj_gate_flat = F.linear(x_fp16.view(M, dim), combined_w)  # (M,5·H) fp16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5, hidden_dim)  # (B,N,N,5,H)
    proj_gate = proj_gate.permute(3, 0, 1, 2, 4)  # (5, B, N, N, H)

    left_proj   = proj_gate[0]  # (B,N,N,H)
    left_gate_r = proj_gate[1]
    right_proj  = proj_gate[2]
    right_gate_r= proj_gate[3]
    out_gate_r  = proj_gate[4]

    left_gate   = torch.sigmoid(left_gate_r)   # (B,N,N,H) fp16
    right_gate  = torch.sigmoid(right_gate_r)  # (B,N,N,H) fp16
    out_gate    = torch.sigmoid(out_gate_r)    # (B,N,N,H) fp16

    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate
    # ------------------------------------------------------------------
    # 4️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h] via Triton
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis: (B,H,N,N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim                     # B·H
    N = seq_len

    # View as (BH,N,N) – each head becomes an independent batch entry
    left_view  = left_perm.view(BH, N, N)
    right_view = right_perm.view(BH, N, N)

    # Allocate output buffer (FP16)
    out_view = torch.empty_like(left_view)

    # Strides for the three tensors (they are contiguous)
    stride_ah, stride_ai, stride_ak = left_view.stride()
    stride_bh, stride_bi, stride_bk = right_view.stride()
    stride_ch, stride_ci, stride_cj = out_view.stride()

    # Tunable tile sizes – work well up to N=1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        a_ptr=left_view,
        b_ptr=right_view,
        c_ptr=out_view,
        BH=BH,
        N=N,
        stride_ah=stride_ah,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_ch=stride_ch,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Restore the original 4‑D layout (B,N,N,H)
    out = out_view.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()        # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (torch implementation)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # (B,N,N,H) same dtype as out (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to model dim D
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H) fp16
    out_fp16 = F.linear(gated, to_out_w)                         # (B,N,N,D) fp16

    # Convert back to float32 for the public API
    out_fp32 = out_fp16.to(torch.float32)
    return out_fp32
```
