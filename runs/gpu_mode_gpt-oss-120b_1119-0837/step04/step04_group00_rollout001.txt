"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Cast input to FP16 and apply a fused FP16 LayerNorm (custom Triton kernel).
2️⃣  Fuse the three linear transforms (left_proj+left_gate, right_proj+right_gate,
    out_gate) into a single half‑precision GEMM → (M,5·H) tensor.
3️⃣  Split the fused tensor into the five heads, apply sigmoid to the three gate
    heads and (optionally) the binary mask to the projected tensors.
4️⃣  Compute the core N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched half‑precision matrix multiplication using torch.matmul:
        out = (B,H,N,N) @ (B,H,N,N)ᵀ  → (B,N,N,H)
5️⃣  Apply a second fused FP16 LayerNorm on the hidden dimension.
6️⃣  Multiply by the output gate and project back to the model dimension with a
    final half‑precision linear layer.
7️⃣  Return the result in float32.

All heavy linear work stays in FP16 (Tensor‑Core friendly) and the only
explicit Triton kernels are the two LayerNorm kernels (FP16 for both passes).
The three independent linear layers of the reference implementation have been
merged into one GEMM to cut memory traffic, and the N³ contraction uses a
single torch.matmul call instead of multiple bmm/view/transposes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# FP16 LayerNorm kernel (2‑pass, accumulates in FP32)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,   # input  pointer
    y_ptr,   # output pointer
    w_ptr,   # weight pointer (γ)
    b_ptr,   # bias pointer   (β)
    M,       # number of rows
    N,       # feature dimension
    eps,     # epsilon
    stride_xm, stride_xn,   # input strides
    stride_ym, stride_yn,   # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # Row range processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """2‑pass LayerNorm for FP16 tensors using Triton."""
    assert x.dtype == torch.float16
    assert x.dim() == 2, "Expected a 2‑D tensor (M, N)"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride extraction
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# Outgoing TriMul forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (float32 / bool) or None
        - weights      : dict of torch.nn.Parameter tensors
        - config       : dict with keys:
            * dim          – model dimension D
            * hidden_dim   – hidden dimension H
            * nomask (optional) – True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  FP16 input + FP16 LayerNorm (custom Triton kernel)
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)
    # flatten to (M, dim) for the LayerNorm kernel
    M = bs * N * N
    x_fp16_flat = x_fp16.reshape(M, dim)

    norm_w = weights["norm.weight"].to(device, torch.float16)
    norm_b = weights["norm.bias"].to(device, torch.float16)
    x_norm_flat = triton_layernorm_fp16(x_fp16_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)   # (B, N, N, D) in FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fuse three linear layers (left_proj+gate, right_proj+gate, out_gate)
    # ------------------------------------------------------------------
    # concatenate all weight matrices along the out‑feature dimension
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(device, torch.float16)          # (5H, D)

    # Single GEMM: (M, D) @ (D, 5H)ᵀ → (M, 5H)
    combined = F.linear(x_norm_flat, combined_w)   # (M, 5H)

    # ------------------------------------------------------------------
    # 3️⃣  Split heads
    # ------------------------------------------------------------------
    # view back to (B, N, N, H) groups
    left_proj   = combined[..., 0:H].view(bs, N, N, H)
    left_gate_p = combined[..., H:2*H].view(bs, N, N, H)
    right_proj  = combined[..., 2*H:3*H].view(bs, N, N, H)
    right_gate_p= combined[..., 3*H:4*H].view(bs, N, N, H)
    out_gate_p  = combined[..., 4*H:5*H].view(bs, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid) & optional mask
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    if not nomask and mask is not None:
        # mask shape (B,N,N) → (B,N,N,1) broadcast over hidden dim
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣  Apply projection gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Core N³ contraction via batched GEMM (torch.matmul)
    # ------------------------------------------------------------------
    # reshape for batched matmul: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # out_perm shape (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # bring back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣  Hidden‑dim LayerNorm (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(M, H)  # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, H)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 8️⃣  Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate               # (B,N,N,H) FP16
    gated_flat = gated.view(M, H)              # (M, H)
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output_flat = F.linear(gated_flat, to_out_w)  # (M, D) FP16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output
