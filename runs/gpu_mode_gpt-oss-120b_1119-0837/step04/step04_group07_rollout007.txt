    left_raw_ptr, right_raw_ptr,
    left_gate_raw_ptr, right_gate_raw_ptr, out_gate_raw_ptr,
    mask_ptr, # may be None
    left_out_ptr, right_out_ptr, out_gate_ptr,
    M, H,
    stride_lr_m, stride_lr_h,
    stride_rg_m, stride_rg_h,
    stride_og_m, stride_og_h,
    stride_mask_m,  # mask shape (M,)
    stride_lo_m, stride_lo_h,
    stride_ro_m, stride_ro_h,
    stride_oo_m, stride_oo_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):

    # rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

def apply_gates_kernel(
    left_raw_ptr, right_raw_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_ptr,

    left_out_ptr, right_out_ptr, out_gate_out_ptr,

                                  mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

* Output LayerNorm uses the same Triton kernel as the input LayerNorm.
* A final element‑wise gate and a half‑precision linear projection produce
  the `float32` output.

All heavy arithmetic is in FP16 (Tensor‑core friendly); only the two
LayerNorms run in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *pointer to input*   [M, N]
    y_ptr,                # *pointer to output*  [M, N]
    weight_ptr, bias_ptr, # [N]
    M, N, eps,            # dimensions
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row range for this program instance
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----------------------------------------------------
    # First pass – compute mean & variance per row
    # ----------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------
    # Second pass – normalise + affine
    # ----------------------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_N: int = 64,
) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor (M×N) using Triton."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure tensors are contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Fused mask + gate application
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_raw_ptr,          # [M, H]  fp16
    right_raw_ptr,         # [M, H]  fp16
    left_gate_raw_ptr,     # [M, H]  fp16
    right_gate_raw_ptr,    # [M, H]  fp16
    out_gate_raw_ptr,      # [M, H]  fp16
    mask_ptr,              # [M]    fp16  (optional)
    left_out_ptr,          # [M, H]  fp16
    right_out_ptr,         # [M, H]  fp16
    out_gate_out_ptr,      # [M, H]  fp16
    M, H,
    stride_m, stride_h,                # strides for all (M,H) tensors – they are identical
    stride_mask_m,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    MASK_ENABLED: tl.constexpr,         # 1 – load mask, 0 – ignore
):
    pid = tl.program_id(0)

    # Row indices for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Loop over hidden‑dimension tiles
    for start_h in range(0, H, BLOCK_H):
        offs_h = start_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load raw tensors (fp16 -> fp32 for arithmetic)
        left_raw = tl.load(
            left_raw_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_raw = tl.load(
            right_raw_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        left_gate_raw = tl.load(
            left_gate_raw_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_gate_raw = tl.load(
            right_gate_raw_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        out_gate_raw = tl.load(
            out_gate_raw_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # Sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # Optional mask (broadcast over hidden dim)
        if MASK_ENABLED:
            mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)  # (BLOCK_M,)
            mask_val = mask_val[:, None]                                            # (BLOCK_M,1)
        else:
            mask_val = tl.full((BLOCK_M, 1), 1.0, dtype=tl.float32)

        # Apply mask and gates (still fp32)
        left_out = left_raw * left_gate * mask_val
        right_out = right_raw * right_gate * mask_val
        out_gate_out = out_gate  # already fp32

        # Cast back to fp16 for storage
        left_out_fp16 = left_out.to(tl.float16)
        right_out_fp16 = right_out.to(tl.float16)
        out_gate_fp16 = out_gate_out.to(tl.float16)

        # Store
        tl.store(
            left_out_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            left_out_fp16,
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            right_out_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            right_out_fp16,
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_out_ptr + offs_m[:, None] * stride_m + offs_h[None, :] * stride_h,
            out_gate_fp16,
            mask=mask_m[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# TriMul forward pass (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype            # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len                     # rows for the flatten view
    x_flat = input_tensor.reshape(M, dim)          # (M, dim) – FP32
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )
    x_norm = x_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,dim) – FP32

    # --------------------------------------------------------------
    # 2️⃣ Combined projection (half‑precision)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()                         # (B,N,N,dim) – FP16
    x_norm_h_flat = x_norm_h.reshape(M, dim)          # (M, dim) – FP16

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)               # (5*H, dim)

    proj_out = F.linear(x_norm_h_flat, combined_weight)    # (M, 5*H) – FP16

    # --------------------------------------------------------------
    # 3️⃣ Split projected tensors (still (M, H) each)
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )  # each (M, H), FP16

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask, sigmoid gates and element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(device, dtype=torch.float16).reshape(M)  # (M,)
        mask_enabled = True
    else:
        # dummy mask (won’t be consulted because MASK_ENABLED = 0)
        mask_flat = torch.tensor([1.0], dtype=torch.float16, device=device)
        mask_enabled = False

    left_out_flat   = torch.empty_like(left_raw)
    right_out_flat  = torch.empty_like(right_raw)
    out_gate_flat   = torch.empty_like(out_gate_raw)

    # All the (M,H) tensors share the same strides, fetch once
    stride_m = left_raw.stride(0)
    stride_h = left_raw.stride(1)

    grid = (triton.cdiv(M, 128),)   # BLOCK_M = 128 (chosen for good occupancy)
    apply_gates_kernel[grid](
        left_raw_ptr=left_raw,
        right_raw_ptr=right_raw,
        left_gate_raw_ptr=left_gate_raw,
        right_gate_raw_ptr=right_gate_raw,
        out_gate_raw_ptr=out_gate_raw,
        mask_ptr=mask_flat,
        left_out_ptr=left_out_flat,
        right_out_ptr=right_out_flat,
        out_gate_out_ptr=out_gate_flat,
        M=M,
        H=hidden_dim,
        stride_m=stride_m,
        stride_h=stride_h,
        stride_mask_m=mask_flat.stride(0),
        BLOCK_M=128,
        BLOCK_H=64,
        MASK_ENABLED=int(mask_enabled),
        num_warps=4,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape back to (B,N,N,H) for the batched GEMM
    # --------------------------------------------------------------
    left = left_out_flat.view(bs, seq_len, seq_len, hidden_dim)
    right = right_out_flat.view(bs, seq_len, seq_len, hidden_dim)
    out_gate = out_gate_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) – fp16

    # --------------------------------------------------------------
    # 6️⃣ TriMul – batched GEMM over the summed axis k
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Move hidden dimension into the batch dimension (contiguous copy required)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    left_flat  = left_perm.reshape(bs * hidden_dim, seq_len, seq_len)   # (B·H, N, K)
    right_flat = right_perm.reshape(bs * hidden_dim, seq_len, seq_len)  # (B·H, K, N)

    out_flat = torch.bmm(left_flat, right_flat)   # (B·H, N, N) – FP16 math, FP32 acc.

    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()                     # (B, N, N, H) – FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp32 = out.float()                                          # (B,N,N,H) – FP32
    out_flat_fp32 = out_fp32.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat_fp32,
        to_out_norm_w,
        to_out_norm_b,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) – FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate.float()                               # (B,N,N,H) – FP32

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    final_fp16 = F.linear(gated.half(), to_out_w)                     # (B,N,N,dim) – FP16
    final_fp32 = final_fp16.float()                                   # (B,N,N,dim) – FP32

    return final_fp32
