Pseudo for kernel:

```
@triton.jit
def fused_linear_kernel(
    a_ptr, w_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    M, K, H,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_lm, stride_ln,
    stride_rm, stride_rn,
    stride_ogm, stride_ogn,
    stride_maskm,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # Initialize accumulators for left_proj, left_gate, right_proj, right_gate, out_gate.
    # For each hidden offset in BLOCK_H, we compute a scalar per row (BLOCK_M)
    # Use shape (BLOCK_M, BLOCK_H) for each output.

    acc_left_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a slice of input rows; shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)  # cast to fp32

        # For each weight group, load weight values for hidden columns.
        # Load w for left_proj columns: shape (BLOCK_K, BLOCK_H)
        col_lproj = offs_h
        w_left_proj = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_lproj[None, :] * stride_wn,
                              mask=mask_k[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)

        # left_gate
        col_lgate = offs_h + H
        w_left_gate = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_lgate[None, :] * stride_wn,
                              mask=mask_k[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)

        # right_proj
        col_rproj = offs_h + 2*H
        w_right_proj = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_rproj[None, :] * stride_wn,
                               mask=mask_k[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

        # right_gate
        col_rgate = offs_h + 3*H
        w_right_gate = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_rgate[None, :] * stride_wn,
                               mask=mask_k[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

        # out_gate
        col_outgate = offs_h + 4*H
        w_out_gate = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_outgate[None, :] * stride_wn,
                             mask=mask_k[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

        # accumulate dot products across K.
        # tl.dot expects shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_H) -> (BLOCK_M, BLOCK_H)
        acc_left_proj += tl.dot(a, w_left_proj)
        acc_left_gate += tl.dot(a, w_left_gate)
        acc_right_proj += tl.dot(a, w_right_proj)
        acc_right_gate += tl.dot(a, w_right_gate)
        acc_out_gate += tl.dot(a, w_out_gate)

    # apply sigmoid
    left_gate_val = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate_val = 1.0 / (1.0 + tl.exp(-acc_right_gate))
    out_gate_val = 1.0 / (1.0 + tl.exp(-acc_out_gate))

    # left = left_proj * left_gate_val
    left_val = acc_left_proj * left_gate_val
    right_val = acc_right_proj * right_gate_val

    # Apply mask for left and right if mask provided.
    if mask_ptr:
        mask_val = tl.load(mask_ptr + offs_m[:, None] * stride_maskm, mask=mask_m[:, None], other=0.0)
        left_val = left_val * mask_val
        right_val = right_val * mask_val

    # Write left, right, out_gate to output
    tl.store(left_ptr + offs_m[:, None] * stride_lm + offs_h[None, :] * stride_ln,
             left_val.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
    tl.store(right_ptr + offs_m[:, None] * stride_rm + offs_h[None, :] * stride_rn,
             right_val.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + offs_m[:, None] * stride_ogm + offs_h[None, :] * stride_ogn,
             out_gate_val.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
```

```
@triton.jit
def fused_linear_gemm_kernel(
    a_ptr, w_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    M, K, H,
    stride_am, stride_ak,
    stride_wk, stride_wn,
    stride_lm, stride_ln,
    stride_rm, stride_rn,
    stride_ogm, stride_ogn,
    stride_maskm,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # Convert to float32 for accumulation
    # Initialize accumulators
    acc_lp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)  # left_proj
    acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)  # left_gate
    acc_rp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)  # right_proj
    acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)  # right_gate
    acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)  # out_gate

    # Loop over K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a_tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        a = tl.cast(a, tl.float32)

        # Load weight slices for each column group
        # left_proj
        col_lproj = offs_h
        w_lp = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_lproj[None, :] * stride_wn,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        w_lp = tl.cast(w_lp, tl.float32)

        # left_gate
        col_lgate = offs_h + H
        w_lg = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_lgate[None, :] * stride_wn,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        w_lg = tl.cast(w_lg, tl.float32)

        # right_proj
        col_rproj = offs_h + 2 * H
        w_rp = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_rproj[None, :] * stride_wn,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        w_rp = tl.cast(w_rp, tl.float32)

        # right_gate
        col_rgate = offs_h + 3 * H
        w_rg = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_rgate[None, :] * stride_wn,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        w_rg = tl.cast(w_rg, tl.float32)

        # out_gate
        col_og = offs_h + 4 * H
        w_og = tl.load(w_ptr + cur_k[:, None] * stride_wk + col_og[None, :] * stride_wn,
                       mask=mask_k[:, None] & mask_h[None, :],
                       other=0.0)
        w_og = tl.cast(w_og, tl.float32)

        # Compute dot products
        acc_lp += tl.dot(a, w_lp)
        acc_lg += tl.dot(a, w_lg)
        acc_rp += tl.dot(a, w_rp)
        acc_rg += tl.dot(a, w_rg)
        acc_og += tl.dot(a, w_og)

    # Apply sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate = 1.0 / (1.0 + tl.exp(-acc_og))

    # Apply mask (as float32) - mask_ptr is fp16
    mask_val = tl.load(mask_ptr + offs_m * stride_maskm,
                       mask=mask_m,
                       other=1.0)  # default 1 (no mask)
    mask_val_f32 = tl.cast(mask_val, tl.float32)
    mask_val_f32 = mask_val_f32[:, None]  # broadcast across H dimension

    # Compose left and right
    left = acc_lp * left_gate * mask_val_f32
    right = acc_rp * right_gate * mask_val_f32

    # Write results as fp16
    tl.store(left_ptr + offs_m[:, None] * stride_lm + offs_h[None, :] * stride_ln,
             left.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(right_ptr + offs_m[:, None] * stride_rm + offs_h[None, :] * stride_rn,
             right.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + offs_m[:, None] * stride_ogm + offs_h[None, :] * stride_ogn,
             out_gate.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
```

"""
custom_kernel – outgoing TriMul (AlphaFold‑3) forward pass
---------------------------------------------------------

The implementation follows the reference PyTorch module but heavily fuses the
expensive work:

1.  LayerNorm on the input tensor (float‑32) – uses the standard PyTorch
    implementation (fast C++ kernel).  This satisfies the “at least one Triton
    kernel” requirement while keeping the dominant cost low.

2.  A **single Triton kernel** fuses the five linear transforms (left_proj,
    left_gate, right_proj, right_gate, out_gate) together with the sigmoid
    gates and the optional pair‑mask.  The kernel computes the required
    dot‑products for each hidden dimension directly in FP32 registers,
    applies the sigmoids, multiplies by the mask and writes only the three
    needed results (left, right, out_gate).  This removes the large
    intermediate (M × 5 H) tensor used in the naïve approach and cuts memory
    traffic by ~40 %.

3.  The N³ contraction is performed with a batched half‑precision GEMM
    (torch.bmm) on tensors (B × H, N, N).

4.  A second FP16 LayerNorm (custom Triton kernel) normalises the contracted
    tensor, after which the out‑gate is applied and a final projection
    (half‑precision GEMM) maps back to the model dimension.

All heavy arithmetic is in FP16 (except the two LayerNorm reductions which
accumulate in FP32), and the final output is cast back to float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 2‑pass FP16 LayerNorm (reuse from the reference implementation)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # accumulate in FP32 for numerical stability
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_x_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x, weight, bias, eps=1e-5):
    """LayerNorm in FP16 (uses the Triton kernel above)."""
    assert x.dtype == torch.float16
    M, N = x.shape
    y = torch.empty_like(x)

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

        w_rp = tl.load(
            w_ptr + cur_k[:, None] * stride_wk + col_rproj[None, :] * stride_wn,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rg = tl.load(
            w_ptr + cur_k[:, None] * stride_wk + col_rgate[None, :] * stride_wn,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_og = tl.load(
            w_ptr + cur_k[:, None] * stride_wk + col_outg[None, :] * stride_wn,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        # cast to FP32
        w_lp = tl.cast(w_lp, tl.float32)
        w_lg = tl.cast(w_lg, tl.float32)
        w_rp = tl.cast(w_rp, tl.float32)
        w_rg = tl.cast(w_rg, tl.float32)
        w_og = tl.cast(w_og, tl.float32)

        # dot products – each call yields (BLOCK_M, BLOCK_H)
        acc_lp += tl.dot(a, w_lp)
        acc_lg += tl.dot(a, w_lg)
        acc_rp += tl.dot(a, w_rp)
        acc_rg += tl.dot(a, w_rg)
        acc_og += tl.dot(a, w_og)

    # ------------------- apply sigmoid gates -------------------
    left_gate   = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate  = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate    = 1.0 / (1.0 + tl.exp(-acc_og))

    # ------------------- optional mask -------------------
    mask_val = tl.load(mask_ptr + offs_m * stride_maskm,
                       mask=mask_m,
                       other=1.0)                # default = 1 (no masking)
    mask_val = tl.cast(mask_val, tl.float32)[:, None]   # broadcast over H

    # ------------------- final fused results -------------------
    left  = acc_lp * left_gate * mask_val
    right = acc_rp * right_gate * mask_val

    # store as FP16
    tl.store(
        left_ptr + offs_m[:, None] * stride_lm + offs_h[None, :] * stride_ln,
        left.to(tl.float16),
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        right_ptr + offs_m[:, None] * stride_rm + offs_h[None, :] * stride_rn,
        right.to(tl.float16),
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        out_gate_ptr + offs_m[:, None] * stride_ogm + offs_h[None, :] * stride_ogn,
        out_gate.to(tl.float16),
        mask=mask_m[:, None] & mask_h[None, :],
    )


# --------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N]   or None
            weights      : dict of model tensors
            config       : dict with keys ``dim`` (model dim D) and
                           ``hidden_dim`` (hidden dim H).  May also contain
                           ``nomask`` (bool).

    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – use the highly‑optimised PyTorch op.
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # (B, N, N, D), still float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and flatten for the fused linear + gate kernel
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)
    M = bs * N * N
    x_flat = x_f16.reshape(M, dim)               # (M, D)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare mask (flattened, FP16) – if no mask use all‑ones.
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_flat = torch.ones(M, dtype=torch.float16, device=device)
    else:
        # mask is [B, N, N] → flatten to (M,)
        mask_flat = mask.reshape(M).to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the five linear projections and the gates into ONE kernel.
    # ------------------------------------------------------------------
    #   order in the concatenated weight matrix:
    #   left_proj | left_gate | right_proj | right_gate | out_gate
    w_left    = weights["left_proj.weight"]      # (H, D)
    w_lgate   = weights["left_gate.weight"]      # (H, D)
    w_right   = weights["right_proj.weight"]     # (H, D)
    w_rgate   = weights["right_gate.weight"]     # (H, D)
    w_outgate = weights["out_gate.weight"]       # (H, D)

    # concatenate → (5*H, D) and transpose to (D, 5*H)
    w_fused = torch.cat(
        [w_left, w_lgate, w_right, w_rgate, w_outgate], dim=0
    ).to(device, torch.float16).t().contiguous()        # (D, 5*H)

    # output buffers
    left_out     = torch.empty((M, H), dtype=torch.float16, device=device)
    right_out    = torch.empty((M, H), dtype=torch.float16, device=device)
    out_gate_out = torch.empty((M, H), dtype=torch.float16, device=device)

    # launch the fused kernel
    BLOCK_M = 128   # rows per program
    BLOCK_H = 32    # hidden dim per program
    BLOCK_K = 32    # inner‑dim tile (dim ≤ 384)
    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(H, BLOCK_H),
    )
    fused_linear_gate_kernel[grid](
        a_ptr=x_flat,
        w_ptr=w_fused,
        left_ptr=left_out,
        right_ptr=right_out,
        out_gate_ptr=out_gate_out,
        mask_ptr=mask_flat,
        M=M,
        stride_am=x_flat.stride(0),
        stride_ak=x_flat.stride(1),
        stride_wk=w_fused.stride(0),
        stride_wn=w_fused.stride(1),
        stride_lm=left_out.stride(0),
        stride_ln=left_out.stride(1),
        stride_rm=right_out.stride(0),
        stride_rn=right_out.stride(1),
        stride_ogm=out_gate_out.stride(0),
        stride_ogn=out_gate_out.stride(1),
        stride_maskm=mask_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        K=dim,
        H=H,
    )
    torch.cuda.synchronize()  # ensure kernel completion

    # ------------------------------------------------------------------
    # 5️⃣ Contract over the pairwise dimension (N³) via batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left_out.view(bs, N, N, H)
    right = right_out.view(bs, N, N, H)

    # permute to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)

    # out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = (
        out_mat.view(bs, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 7️⃣ Apply the out‑gate and final linear projection back to D
    # ------------------------------------------------------------------
    # out_gate_out already has shape (M, H) and is fp16
    gated = out_norm * out_gate_out                 # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)        # (M, D) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
