"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated.

Algorithm
---------
1️⃣  Input LayerNorm (float32) – use PyTorch’s cuDNN implementation.
2️⃣  Cast the normalized tensor to fp16 for all subsequent heavy work.
3️⃣  Fuse projection + gate for the *left* and *right* sides:
    a single linear call per side produces a tensor of shape (B,N,N,2·H);
    the first H channels are the projection, the remaining H are the
    pre‑sigmoid gate values.
4️⃣  Optional binary mask (broadcast over the hidden dimension) is applied
    to the projections before gating.
5️⃣  Apply sigmoid to the gate tensors and multiply with the projections.
6️⃣  Compute the core N³ contraction:
    out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d].
    This is a batched half‑precision GEMM performed with a single
    ``torch.bmm`` on tensors of shape (B·H,N,N).
7️⃣  Fuse the second LayerNorm (over the hidden dimension) together with the
    element‑wise multiplication by the output‑gate using a custom Triton kernel.
8️⃣  Final linear projection back to the model dimension (fp16 matmul) and
    cast back to float32 for the returned tensor.

At least one operation (the second LayerNorm + gating) is performed in a
custom Triton kernel, satisfying the problem requirements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: FP16 LayerNorm with fused element‑wise gate scaling
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,          # (M, H) input, fp16
    y_ptr,          # (M, H) output, fp16
    gate_ptr,       # (M, H) gate, fp16
    weight_ptr,     # (H,)   LN scale, fp16
    bias_ptr,       # (H,)   LN bias,  fp16
    M, H, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_N: tl.constexpr,  # cols per program (tile width)
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row indices for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣ First pass – compute per‑row mean and variance with FP32 acc.
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(H, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < H

        # Load a (BLOCK_M, BLOCK_N) tile
        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(xv, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass – write normalized + affine transformed values,
    #    then apply the element‑wise gate.
    # ------------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(xv, tl.float32)

        # (x - μ) / σ
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # affine transformation
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)
        y = norm * w[None, :] + b[None, :]

        # gate
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        y = y * g

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_fp16_gate(x: torch.Tensor,
                               weight: torch.Tensor,
                               bias: torch.Tensor,
                               gate: torch.Tensor,
                               eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm (fp16) with fused gate scaling using Triton.

    Arguments
    ---------
    x : (M, H) fp16 tensor – input to be normalised.
    weight : (H,) fp16 – LN scale.
    bias   : (H,) fp16 – LN bias.
    gate   : (M, H) fp16 – element‑wise gate (already after sigmoid).
    eps    : float – epsilon for numerical stability.

    Returns
    -------
    y : (M, H) fp16 – normalised tensor multiplied by the gate.
    """
    assert x.dim() == 2 and x.dtype == torch.float16
    M, H = x.shape
    y = torch.empty_like(x)

    # make everything contiguous to simplify stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    gate = gate.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    stride_gm, stride_gn = gate.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_gate_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        gate_ptr=gate,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        BLOCK_M=256,   # tuned for H100 – handles up to ~400 k rows comfortably
        BLOCK_N=128,   # full‑row tile for H ≤ 384
    )
    return y


# ----------------------------------------------------------------------
# Entry point – the custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] or None (float32/bool/uint8)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys:
            * dim         – model dimension (D)
            * hidden_dim  – hidden dimension (H)
            * nomask      – bool, optional (skip masking if True)

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32) – the result of the TriMul forward pass.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – cuDNN implementation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, D), float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left & right sides
    # ------------------------------------------------------------------
    # (2·H, D) weight that produces [proj | gate_pre_sigmoid] in one matmul
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, D)

    # Output‑gate weight (H, D)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Linear passes – each yields (B, N, N, 2·H)
    left_comb  = F.linear(x_f16, left_comb_w)      # (B, N, N, 2H)
    right_comb = F.linear(x_f16, right_comb_w)     # (B, N, N, 2H)

    # Split the combined tensors
    left_proj   = left_comb[..., :H]                # (B, N, N, H)
    left_gate_z = left_comb[..., H:]                # (B, N, N, H)
    right_proj  = right_comb[..., :H]
    right_gate_z= right_comb[..., H:]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask assumed float (1.0 for keep, 0.0 for mask)
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates to the gate tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)   # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_z)

    left  = left_proj  * left_gate    # (B, N, N, H)
    right = right_proj * right_gate   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction – batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # (B·H, N, N) = (B·H, N, N) @ (B·H, N, N)^T  (fp16 GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H), still fp16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dim + fused output‑gate scaling
    # ------------------------------------------------------------------
    M = bs * N * N                     # total rows to normalise
    out_flat = out.view(M, H).contiguous()            # (M, H) fp16
    out_gate_flat = out_gate.view(M, H).contiguous()  # (M, H) fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)

    # Fused LN + gate → (M, H) fp16
    out_norm_and_gate = triton_layernorm_fp16_gate(
        out_flat, to_out_norm_w, to_out_norm_b, out_gate_flat, eps=eps
    )  # (M, H), fp16

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to model dimension (fp16) → fp32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(out_norm_and_gate, to_out_w)            # (M, D) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)      # (B,N,N,D)

    return output
```

**Explanation of key choices**

* **Mixed‑precision pipeline:** All heavyweight matrix multiplications (the three projections, the N³ contraction, and the final projection) run in fp16, allowing Tensor Cores on an H100 to be fully utilized while keeping the final result in fp32 for downstream compatibility.

* **Projection‑gate fusion:** By concatenating the projection and gate weight matrices we halve the number of linear calls per side (one matmul → 2·H outputs), then split and apply `sigmoid` in‑place.

* **Mask handling:** The optional mask is broadcasted once over the hidden dimension and applied before gating, matching the reference implementation.

* **Batched GEMM for the N³ contraction:** The einsum reduces to many independent matrix multiplications across the hidden dimension. Reshaping to (B·H, N, N) and using a single `torch.bmm` is the most efficient approach on modern GPUs.

* **Triton‑fused second LayerNorm + output‑gate:** This kernel performs a two‑pass mean/variance computation in FP32 for numerical stability, applies the affine transformation, and finally multiplies by the pre‑computed output gate—all without extra memory passes.

The implementation respects all configuration options (batch size, sequence length, model dimensions, hidden size, optional masking, and input distributions) and runs entirely on the provided CUDA device.
