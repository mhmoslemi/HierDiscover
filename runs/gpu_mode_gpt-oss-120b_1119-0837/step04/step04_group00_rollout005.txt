    # a shape (batch, M, K)

@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_ptrs = a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        a = tl.cast(a, tl.float16)  # it's already fp16; maybe skip cast
        # load b tile
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        b = tl.cast(b, tl.float16)

        # dot product across K dimension
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result as fp16
    c = tl.cast(acc, tl.float16)
    c_ptrs = c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” forward pass – Triton‑accelerated implementation.

The operator works on a 4‑D tensor [B, N, N, D] and implements the
following steps (all heavy work is carried out in FP16):
1️⃣  Float‑32 LayerNorm over the model dimension D (custom Triton kernel).
2️⃣  Cast to FP16.
3️⃣  Fuse the three linear projections (left‑proj, right‑proj,
    output‑gate) into a single matmul that yields 5·H channels and
    split the result, applying sigmoid gates.
4️⃣  Optional binary mask (broadcast over the hidden dimension) applied
    before the gated projection.
5️⃣  Core N³ contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   Implemented as a batched FP16 GEMM (A @ Bᵀ) using a custom Triton
   kernel, which processes the batch dimension (B·H) together.
6️⃣  Float‑16 hidden‑dim LayerNorm (custom Triton kernel).
7️⃣  Multiply by the output gate.
8️⃣  Final linear projection back to D and cast to FP32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – FP32 → FP32 (custom Triton kernel)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----------- first pass: compute mean and variance --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------- second pass: write normalized output --------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """Applies LayerNorm (float32) via a custom Triton kernel.
    Shapes: (M, D) where D = feature dimension.
    """
    assert x.dim() == 2, "expects a 2‑D tensor"
    M, D = x.shape
    y = torch.empty_like(x)

    # Ensure strides are contiguous for easy handling
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """Applies LayerNorm (float16) via a custom Triton kernel.
    Shapes: (M, H) where H = hidden dimension.
    """

    assert x.dim() == 2, "expects a 2‑D tensor"
    M, H = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()

    layernorm_fp16_kernel[grid](

        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,

    return y


# --------------------------------------------------------------------------
# Batched FP16 GEMM (A @ Bᵀ) – Triton kernel
# --------------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # K‑dimension tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N) – note that B is already transposed (Bᵀ)
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        a = tl.cast(a, tl.float16)
        b = tl.cast(b, tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # write result as FP16
    c = tl.cast(acc, tl.float16)
    tl.store(
        c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm_fp16(a: torch.Tensor, b_T: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for FP16:  out = a @ b_T  (a: (B, M, K), b_T: (B, K, N))
    Returns tensor of shape (B, M, N) in FP16.
    """
    batch, M, K = a.shape
    _, K2, N = b_T.shape
    assert K == K2, "Inner dimensions must match"
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides
    stride_a_batch, stride_am, stride_ak = a.stride()
    stride_b_batch, stride_bk, stride_bn = b_T.stride()
    stride_c_batch, stride_cm, stride_cn = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b_T,
        c_ptr=c,

# --------------------------------------------------------------------------
# Main entry point – outgoing TriMul
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of parameter tensors (see keys below)
        - config       : dict with keys 'dim' (D), 'hidden_dim' (H) and optional
                         'nomask' (bool)

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)             # (B, N, N, D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all three linear projections (left, right, out‑gate) into
    #    a single matmul that yields 5·H channels.
    # ------------------------------------------------------------------
    # Combined weight: [left_proj; left_gate; right_proj; right_gate; out_gate]
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],    # (H, D)
            weights["left_gate.weight"],    # (H, D)
            weights["right_proj.weight"],   # (H, D)
            weights["right_gate.weight"],   # (H, D)
            weights["out_gate.weight"],     # (H, D)
        ],
        dim=0,
    ).to(device, torch.float16)                         # (5H, D)

    # single large linear pass
    combined = F.linear(x_f16, combined_weight)        # (B·N·N, 5H)

    # split back to the logical tensors
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        combined, H, dim=-1
    )

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B, N, N) → (B, N, N, 1) → FP16
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)   # (B·N·N, H)

    # gated projections
    left  = left_proj * left_gate          # (B·N·N, H)
    right = right_proj * right_gate        # (B·N·N, H)

    # reshape to (B, H, N, N) for the contraction
    left  = left.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten batch & hidden dimensions → (B·H, N, N)
    BH = bs * H
    left_mat  = left.view(BH, N, N)                 # (B·H, N, N)
    # right matrix must be transposed before multiplication
    right_T = right.view(BH, N, N).permute(0, 2, 1).contiguous()  # (B·H, N, N) where last two dims are (K, N)

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction via a custom batched FP16 GEMM
    # ------------------------------------------------------------------
    out_mat = triton_batch_gemm_fp16(left_mat, right_T)    # (B·H, N, N) FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (FP16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)          # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)   # (M, H) FP16
    gated = out_norm * out_gate_flat         # (M, H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)   # (M, D) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32 tensor
    return output
