@triton.jit
def apply_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    N: tl.constexpr, K: tl.constexpr, B: tl.constexpr, H: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    left,right: shape (B,N,N,H) => flatten index = (b*N*N + i*N + k)*H + h
    left_gate,right_gate: same shape.
    mask: shape (B,N,N) => broadcast over H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)  # flatten index
    mask = offs < (B * N * N * H)
    if not mask.any():  # maybe unnecessary
        return

    # compute indices for mask
    offset = offs // H      # integer division to get index for mask and non-H dims
    h = offs % H
    # now offset corresponds to (b * N * N + i * N + k)
    # compute indices b,i,k: we can derive:
    # b = offset // (N * N)
    # rem = offset % (N * N)
    # i = rem // N
    # k = rem % N
    # But we don't need i/k for mask retrieval, just need mask element at (b,i,k)
    b = offset // (N * N)
    rem = offset % (N * N)
    i = rem // N
    k = rem % N

    # compute mask offset in 3D tensor:
    mask_offset = b * N * N + i * N + k
    mask_val = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)  # default 1

    if has_mask:
        mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0.0))

    # load left/right and gate
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    tl.store(right_ptr + offs, right_out, mask=mask)

```
@triton.jit
def gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    H,
    total_elems,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    out_left = left * left_gate_sig
    out_right = right * right_gate_sig

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.
------------------------------------------------
1️⃣ Input LayerNorm  – use PyTorch's highly‑optimised native kernel.
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
   into a *single* FP16 GEMM.
3️⃣ A single Triton kernel reads the fused projection and
   ① applies the two sigmoid gates,
   ② writes the gated left/right tensors,
   ③ writes the (sigmoided) out‑gate.
   This eliminates several memory passes that the reference implementation
   performs with separate torch ops.
4️⃣ Optional binary mask is applied (broadcast over the hidden dimension).
5️⃣ Batched matrix‑multiply over the summed axis `k`:
   out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]   →   torch.bmm
   (B·H independent N×N GEMMs, each executed by cuBLAS tensor‑cores).
6️⃣ Output LayerNorm – again use the native kernel.
7️⃣ Apply the out‑gate, then the final FP16 linear projection back to `dim`.
All heavy arithmetic is in FP16; LayerNorms run in FP32 for numerical stability.
The only custom Triton kernel is the fused gate‑mask kernel which dramatically
reduces memory traffic and improves runtime (target < 1 ms for the benchmark
cases).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection → sigmoid gates → left/right/out_gate
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_kernel(
    proj_ptr,          # input: (M, 5*H)  – FP16
    left_out_ptr,      # output: (M, H)   – FP16
    right_out_ptr,     # output: (M, H)   – FP16
    out_gate_ptr,      # output: (M, H)   – FP16
    H: tl.constexpr,   # hidden dimension
    TOTAL_ELEMS: tl.constexpr,   # M*H
    BLOCK_SIZE: tl.constexpr,
):
    """Read the concatenated projection (5·H columns) and write:
       left = left_raw * sigmoid(left_gate_raw)
       right = right_raw * sigmoid(right_gate_raw)
       out_gate = sigmoid(out_gate_raw)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # split linear index into (row, col_in_hidden)
    row = offs // H          # 0 … M‑1
    col = offs % H           # 0 … H‑1

    # each row contains 5 blocks of size H, stride = 5*H
    stride_row = 5 * H

    base = row * stride_row   # integer offset to the beginning of the row

    # load the five slices (scalar per thread)
    left_raw      = tl.load(proj_ptr + base + col,                     mask=mask, other=tl.float16(0.0))
    right_raw     = tl.load(proj_ptr + base + H + col,                 mask=mask, other=tl.float16(0.0))
    left_gate_raw = tl.load(proj_ptr + base + 2 * H + col,             mask=mask, other=tl.float16(0.0))
    right_gate_raw= tl.load(proj_ptr + base + 3 * H + col,             mask=mask, other=tl.float16(0.0))
    out_gate_raw  = tl.load(proj_ptr + base + 4 * H + col,             mask=mask, other=tl.float16(0.0))

    # sigmoid = 1/(1+exp(-x))
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    # apply gates
    left_out  = left_raw  * left_gate_sig
    right_out = right_raw * right_gate_sig

    # store results
    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
    tl.store(out_gate_ptr + offs, out_gate_sig, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N]  (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – native PyTorch kernel
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )   # (B,N,N,dim)  FP32

    # --------------------------------------------------------------
    # 2️⃣ Project to five tensors in *one* FP16 linear layer
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()                     # (B,N,N,dim) – FP16
    M = bs * seq_len * seq_len                    # total number of (b,i,k) triples
    x_norm_h_flat = x_norm_h.view(M, dim)        # (M, dim)

    # concatenate the five weight matrices (shape (5*H, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    # (M, 5*H)   FP16
    proj_out = F.linear(x_norm_h_flat, combined_weight)

    # --------------------------------------------------------------
    # 3️⃣ Fuse sigmoid‑gates and split into left/right/out_gate
    # --------------------------------------------------------------
    # allocate output tensors for the three results
    left_gated  = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_gated = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    out_gate   = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    TOTAL_ELEMS = M * hidden_dim
    BLOCK_SIZE = 1024                # 1K threads per program instance (max on H100)
    grid = (triton.cdiv(TOTAL_ELEMS, BLOCK_SIZE),)

    fused_gate_kernel[grid](
        proj_ptr=proj_out,
        left_out_ptr=left_gated,
        right_out_ptr=right_gated,
        out_gate_ptr=out_gate,
        H=hidden_dim,
        TOTAL_ELEMS=TOTAL_ELEMS,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to (B,N,N,H)
    left  = left_gated.view(bs, seq_len, seq_len, hidden_dim)
    right = right_gated.view(bs, seq_len, seq_len, hidden_dim)
    out_gate = out_gate.view(bs, seq_len, seq_len, hidden_dim)   # FP16, already sigmoided

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication across the summed axis k
    # --------------------------------------------------------------
    # bring hidden dimension to the batch axis: (B, H, i, k)  / (B, H, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    left_flat  = left_perm.view(bs * hidden_dim, seq_len, seq_len)   # (B*H, i, k)
    right_flat = right_perm.view(bs * hidden_dim, seq_len, seq_len)  # (B*H, k, j)

    # batched GEMM (FP16 inputs, FP32 accumulation internally)
    out_flat = torch.bmm(left_flat, right_flat)                     # (B*H, i, j)

    # reshape to original 4‑D layout
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)           # (B, H, i, j)
    out = out.permute(0, 2, 3, 1).contiguous()                     # (B, i, j, H)  FP16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32) – native kernel
    # --------------------------------------------------------------
    out_fp32 = out.float()                                         # FP32 for numerical stability
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # (B, N, N, H)  FP32

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to model dim
    # --------------------------------------------------------------
    # out_gate is still FP16 → cast to FP32 for the element‑wise product
    gated = out_norm * out_gate.float()                # (B, N, N, H)  FP32

    # final linear (half‑precision weight, FP16 compute)
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)   # (dim, H)
    final_fp16 = F.linear(gated.half(), to_out_w)          # (B, N, N, dim)  FP16
    final_fp32 = final_fp16.float()                         # (B, N, N, dim)  FP32

    return final_fp32
