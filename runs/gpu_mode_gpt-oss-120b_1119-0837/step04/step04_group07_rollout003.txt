@triton.jit
def apply_gate_mask_kernel(
    val_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_val_m, stride_val_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m,
    stride_out_m, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    mask_m = offs_m < M
    mask_h = offs_h < H
    mask = tl.load(mask_ptr + offs_m * stride_mask_m, mask=mask_m, other=0.0)[:, None] # broadcast over h
    # Load val and gate
    val = tl.load(val_ptr + offs_m[:, None] * stride_val_m + offs_h[None, :] * stride_val_h, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
    # Compute sigmoid
    gate_f = gate.to(tl.float32)  # compute in FP32
    sigmoid_gate = 1.0 / (1.0 + tl.exp(-gate_f))
    # Multiply
    out = val * sigmoid_gate * mask
    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h, out, mask=mask_m[:, None] & mask_h[None, :])
```

    stride_mask_m,
    stride_lout_m, stride_lout_h,
    stride_rout_m, stride_rout_h,
    stride_ogout_m, stride_ogout_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  LayerNorm on the input tensor (FP32, torch.nn.functional.layer_norm).
2️⃣  Cast to FP16 and perform a *single* fused linear projection that produces the
    five tensors required for TriMul (left, right, left‑gate, right‑gate, out‑gate).
3️⃣  A small Triton kernel fuses the three sigmoid‑gates, the optional binary
    mask and the element‑wise multiplication of the left/right tensors.
4️⃣  Batched matrix multiplication across the summed axis *k*:
    for each head h compute  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
    Implemented with `torch.bmm` on (B·H, N, N) tensors – this maps to the
    highly‑optimised cuBLAS Tensor‑Core GEMM on H100.
5️⃣  LayerNorm on the hidden dimension (FP32, torch.nn.functional.layer_norm).
6️⃣  Apply the output gate (sigmoid) and a final FP16 linear projection back to
    the model dimension.
7️⃣  Return the result in FP32.

The only Triton code is the gated‑mask kernel (step 3), keeping the heavy
matrix‑multiply work in the cuBLAS library.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_and_mask_kernel(
    # input tensors
    left_raw_ptr, left_gate_ptr, right_raw_ptr, right_gate_ptr,
    mask_ptr,
    # output tensors
    left_out_ptr, right_out_ptr,
    # shapes
    M, H,
    # strides (row‑major contiguous)
    stride_lr_m, stride_lr_h,
    stride_lg_m, stride_lg_h,
    stride_rr_m, stride_rr_h,
    stride_rg_m, stride_rg_h,
    stride_mask_m,
    stride_lout_m, stride_lout_h,
    stride_rout_m, stride_rout_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Fuse sigmoid‑gates, binary mask and element‑wise product for left/right."""
    pid = tl.program_id(0)

    # row indices for this program instance
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H
    block_mask = mask_m[:, None] & mask_h[None, :]

    # load mask (broadcast over the hidden dimension)
    # if mask_ptr points to a tensor of ones the multiplication is a no‑op
    mask_val = tl.load(mask_ptr + offs_m * stride_mask_m, mask=mask_m, other=1.0)[:, None]

    # load raw tensors
    left_raw   = tl.load(
        left_raw_ptr + offs_m[:, None] * stride_lr_m + offs_h[None, :] * stride_lr_h,
        mask=block_mask,
        other=0.0,
    )
    left_gate  = tl.load(
        left_gate_ptr + offs_m[:, None] * stride_lg_m + offs_h[None, :] * stride_lg_h,
        mask=block_mask,
        other=0.0,
    )
    right_raw  = tl.load(
        right_raw_ptr + offs_m[:, None] * stride_rr_m + offs_h[None, :] * stride_rr_h,
        mask=block_mask,
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr + offs_m[:, None] * stride_rg_m + offs_h[None, :] * stride_rg_h,
        mask=block_mask,
        other=0.0,
    )

    # sigmoid in FP32 for numerical stability, then cast back to FP16
    left_gate_f  = left_gate.to(tl.float32)
    right_gate_f = right_gate.to(tl.float32)

    sigmoid_left  = 1.0 / (1.0 + tl.exp(-left_gate_f))
    sigmoid_right = 1.0 / (1.0 + tl.exp(-right_gate_f))

    sigmoid_left_h  = sigmoid_left.to(tl.float16)
    sigmoid_right_h = sigmoid_right.to(tl.float16)

    # fused compute:  raw * mask * sigmoid(gate)
    left_out  = left_raw  * mask_val * sigmoid_left_h
    right_out = right_raw * mask_val * sigmoid_right_h

    # store results
    tl.store(
        left_out_ptr + offs_m[:, None] * stride_lout_m + offs_h[None, :] * stride_lout_h,
        left_out,
        mask=block_mask,
    )
    tl.store(
        right_out_ptr + offs_m[:, None] * stride_rout_m + offs_h[None, :] * stride_rout_h,
        right_out,
        mask=block_mask,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32, torch native)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (FP16)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()
    M = bs * seq_len * seq_len                     # total number of pairwise entries
    x_flat = x_norm_h.view(M, dim)                 # (M, dim) FP16

    # concatenate the five weight matrices once (shape: 5*H x dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    proj_out = F.linear(x_flat, combined_weight)  # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split into the five tensors
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    # reshape back to (B,N,N,H)
    left_raw      = left_raw.view   (bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view  (bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Triton kernel: apply sigmoids, binary mask and elementwise product
    # --------------------------------------------------------------
    # flatten everything to (M, H) for kernel launch
    left_raw_f      = left_raw.contiguous().view(M, hidden_dim)
    right_raw_f     = right_raw.contiguous().view(M, hidden_dim)
    left_gate_f     = left_gate_raw.contiguous().view(M, hidden_dim)
    right_gate_f    = right_gate_raw.contiguous().view(M, hidden_dim)

    # prepare mask (binary) – if no mask, use a vector of ones
    if nomask or mask is None:
        mask_f = torch.ones(M, device=device, dtype=torch.float16)
    else:
        mask_f = mask.view(M).to(device, dtype=torch.float16)   # (M,)

    left_out_f  = torch.empty_like(left_raw_f)
    right_out_f = torch.empty_like(right_raw_f)

    # launch the kernel
    BLOCK_M, BLOCK_H = 256, 64                     # tuned for H100
    grid = (triton.cdiv(M, BLOCK_M),)
    gate_and_mask_kernel[grid](
        left_raw_ptr=left_raw_f,
        left_gate_ptr=left_gate_f,
        right_raw_ptr=right_raw_f,
        right_gate_ptr=right_gate_f,
        mask_ptr=mask_f,
        left_out_ptr=left_out_f,
        right_out_ptr=right_out_f,
        M=M,
        H=hidden_dim,
        stride_lr_m=left_raw_f.stride(0),
        stride_lr_h=left_raw_f.stride(1),
        stride_lg_m=left_gate_f.stride(0),
        stride_lg_h=left_gate_f.stride(1),
        stride_rr_m=right_raw_f.stride(0),
        stride_rr_h=right_raw_f.stride(1),
        stride_rg_m=right_gate_f.stride(0),
        stride_rg_h=right_gate_f.stride(1),
        stride_mask_m=mask_f.stride(0),
        stride_lout_m=left_out_f.stride(0),
        stride_lout_h=left_out_f.stride(1),
        stride_rout_m=right_out_f.stride(0),
        stride_rout_h=right_out_f.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )

    # reshape back to (B,N,N,H)
    left  = left_out_f .view(bs, seq_len, seq_len, hidden_dim)
    right = right_out_f.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Output gate (sigmoid) – stays in FP32 for the later multiplication
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw).float()     # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM across the summed axis (k)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # bring head dimension to the second axis
    left_perm  = left .permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, k, j)

    # collapse batch & head dimensions for torch.bmm
    left_flat  = left_perm .reshape(bs * hidden_dim, seq_len, seq_len)   # (B*H, N, K)
    right_flat = right_perm.reshape(bs * hidden_dim, seq_len, seq_len)   # (B*H, K, N)

    out_flat = torch.bmm(left_flat, right_flat)          # (B*H, N, N)  FP16→FP32 accumulation inside cuBLAS

    # reshape to original layout (B,N,N,H) and keep FP16
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out.float(),
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dimension
    # --------------------------------------------------------------
    gated = out_norm * out_gate                           # (B,N,N,H) FP32

    # final linear projection (half‑precision weight)
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)   # (dim, H)
    final_half = F.linear(gated.half(), to_out_w)                         # (B,N,N,dim) FP16
    final = final_half.float()                                            # (B,N,N,dim) FP32

    return final
