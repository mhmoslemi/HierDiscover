@triton.jit
def row_mean_var_kernel(
    x_ptr, mean_ptr, var_ptr,
    M, H,
    stride_xm, stride_xn,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # each block handles BLOCK_M rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # iterate over H dimension
    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = tl.arange(0, BLOCK_K) + k * BLOCK_K
        mask_k = offs_k < H
        xv = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xn,
                     mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        xf = tl.cast(xv, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(var_ptr + offs_m, var, mask=mask_m)
```

    gamma_ptr,  # (H,) fp16

    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,

    pid_col = tl.program_id(1)
    offs_m = pid_row * BLOCK_M + tl.arange(0, BLOCK_M)

8️⃣ Cast the final result back to float32 and reshape to [B, N, N, dim].

The implementation respects all configuration options (batch size, sequence
length, hidden dimension, optional mask) and uses mixed precision for maximum
throughput on an H100.  At least one operation (step 7) is performed with a
custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse LayerNorm, output‑gate and final linear projection
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_linear_kernel(
    x_ptr,           # (M, H)                fp16   – hidden tensor after N³ contraction
    gate_ptr,        # (M, H)                fp16   – output‑gate
    weight_ptr,      # (D, H)                fp16   – final Linear weight
    gamma_ptr,       # (H,)                  fp16   – LN scale
    beta_ptr,        # (H,)                  fp16   – LN bias
    mean_ptr,        # (M,)                  fp32   – per‑row mean
    inv_std_ptr,     # (M,)                  fp32   – 1 / sqrt(var + eps)
    out_ptr,         # (M, D)                fp16   – final output
    M, H, D,
    eps,
    stride_xm, stride_xn,
    stride_gm, stride_gn,
    stride_wm, stride_wn,
    stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_row = tl.program_id(0)           # row‑tile index
    pid_col = tl.program_id(1)           # col‑tile index

    offs_m = pid_row * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_col * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_d = offs_d < D

    # ------------------------------------------------------------------
    # Accumulator for the output tile (FP32 for numeric safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Load per‑row statistics (mean & 1/σ)
    # ------------------------------------------------------------------
    mean   = tl.load(mean_ptr   + offs_m, mask=mask_m, other=0.0)   # (BLOCK_M,)
    inv_std = tl.load(inv_std_ptr + offs_m, mask=mask_m, other=0.0)  # (BLOCK_M,)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension K = H (tiled by BLOCK_H)
    # ------------------------------------------------------------------
    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = tl.arange(0, BLOCK_H) + k * BLOCK_H
        mask_h = offs_h < H

        # ----- load hidden tensor -------------------------------------------------
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_h[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )                                 # fp16
        xf = tl.cast(x, tl.float32)      # (BLOCK_M, BLOCK_H)

        # ----- normalise (x - μ) / σ --------------------------------------------
        norm = (xf - mean[:, None]) * inv_std[:, None]   # fp32

        # ----- apply learned LN affline (γ,β) ------------------------------------
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # fp16
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # fp16
        gamma_f = tl.cast(gamma, tl.float32)
        beta_f  = tl.cast(beta,  tl.float32)
        tmp = norm * gamma_f[None, :] + beta_f[None, :]               # fp32

        # ----- multiply by output‑gate -------------------------------------------
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_h[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )                                 # fp16
        gf = tl.cast(g, tl.float32)
        gated = tmp * gf                 # fp32

        # ----- load final linear weight tile (H × D) -------------------------------
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_wm + offs_d[None, :] * stride_wn,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0,
        )                                 # fp16
        wf = tl.cast(w, tl.float32)      # (BLOCK_H, BLOCK_D)

        # ----- accumulate dot product ------------------------------------------------
        # (BLOCK_M, BLOCK_H) @ (BLOCK_H, BLOCK_D) -> (BLOCK_M, BLOCK_D)
        acc += tl.dot(gated, wf)
    # ------------------------------------------------------------------
    # Store the fused result (cast back to fp16)
    # ------------------------------------------------------------------
    tl.store(
        out_ptr + offs_m[:, None] * stride_om + offs_d[None, :] * stride_on,
        tl.cast(acc, tl.float16),
        mask=mask_m[:, None] & mask_d[None, :],
    )


# ----------------------------------------------------------------------
# Helper that runs the fused kernel
# ----------------------------------------------------------------------
def fused_ln_gate_linear(
    hidden: torch.Tensor,           # (M, H) fp16
    gate: torch.Tensor,             # (M, H) fp16
    weight: torch.Tensor,           # (D, H) fp16
    gamma: torch.Tensor,            # (H,)   fp16
    beta: torch.Tensor,             # (H,)   fp16
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Computes:
        out = Linear( LN(hidden) * gate, weight )
    where LN uses per‑row statistics.
    Returns tensor (M, D) fp16.
    """
    M, H = hidden.shape
    D = weight.shape[0]

    # per‑row mean / variance (torch uses highly‑optimised kernels)
    mean = hidden.mean(dim=1, dtype=torch.float32)               # (M,)
    var  = hidden.var(dim=1, unbiased=False, dtype=torch.float32)  # (M,)
    inv_std = torch.rsqrt(var + eps)                            # (M,)

    out = torch.empty((M, D), dtype=torch.float16, device=hidden.device)

    # make everything contiguous to get simple stride values
    hidden = hidden.contiguous()
    gate   = gate.contiguous()
    weight = weight.contiguous()
    gamma  = gamma.contiguous()
    beta   = beta.contiguous()
    out    = out.contiguous()
    mean   = mean.contiguous()
    inv_std= inv_std.contiguous()

    stride_xm, stride_xn = hidden.stride()
    stride_gm, stride_gn = gate.stride()
    stride_wm, stride_wn = weight.stride()   # (stride over H, stride over D)
    stride_om, stride_on = out.stride()

    grid = (triton.cdiv(M, 64), triton.cdiv(D, 64))
    ln_gate_linear_kernel[grid](
        x_ptr=hidden,
        gate_ptr=gate,
        weight_ptr=weight,
        gamma_ptr=gamma,
        beta_ptr=beta,
        mean_ptr=mean,
        inv_std_ptr=inv_std,
        out_ptr=out,
        M=M,
        H=H,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        stride_wm=stride_wm,
        stride_wn=stride_wn,
        stride_om=stride_om,
        stride_on=stride_on,
        BLOCK_M=64,
        BLOCK_D=64,
        BLOCK_H=64,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – highly‑optimised TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data = (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor[B, N, N, D] – float32
        mask         : torch.Tensor[B, N, N]   – optional / None
        weights      : dict of torch.Tensor   – model parameters
        config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor[B, N, N, D] – float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fast cuDNN FP32 implementation)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # (B, N, N, D) – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left & right sides
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2·H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2·H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # combined linear calls (single matmul per side)
    left_comb  = F.linear(x_f16, left_comb_w)      # (B, N, N, 2·H)
    right_comb = F.linear(x_f16, right_comb_w)     # (B, N, N, 2·H)

    # split projection / gate parts
    left_proj   = left_comb[..., :H]                # (B, N, N, H)
    left_gate_z = left_comb[..., H:]                # (B, N, N, H)
    right_proj  = right_comb[..., :H]
    right_gate_z= right_comb[..., H:]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)

    left  = left_proj  * left_gate    # (B, N, N, H)
    right = right_proj * right_gate   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output‑gate (same for all positions)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched half‑precision GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) and use torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # (B·H, N, N) = (B·H, N, N) @ (B·H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back‑to (B, N, N, H) – keep fp16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Fuse second LayerNorm, output‑gate multiplication and final
    #     projection into one Triton kernel.
    # ------------------------------------------------------------------
    M = bs * N * N
    out_flat = out.view(M, H).contiguous()
    out_gate_flat = out_gate.view(M, H).contiguous()

    # final projection weights (dim × hidden_dim)
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    # LayerNorm parameters for the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)

    # fused kernel produces (M, D) fp16
    fused_out_fp16 = fused_ln_gate_linear(
        hidden=out_flat,
        gate=out_gate_flat,
        weight=to_out_w,
        gamma=to_out_norm_w,
        beta=to_out_norm_b,
        eps=eps,
    )   # (M, D) fp16

    # reshape to (B, N, N, D) and cast back to float32
    output = fused_out_fp16.view(bs, N, N, dim).to(torch.float32)

    return output
