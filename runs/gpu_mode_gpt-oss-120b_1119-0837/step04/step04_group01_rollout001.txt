"""
Highly‑optimized Triton‑accelerated “outgoing” TriMul forward pass.

Algorithm
---------
1. Cast the [B,N,N,D] input to FP16.
2. Perform a fused FP16 LayerNorm on the flattened (B·N·N,D) tensor
   (mean/var are accumulated in FP32, one‑pass kernel).
3. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM: (M,D) @ (D,5·H) → (M,5·H).
4. Split the result, apply sigmoid gates, and (if present) apply the
   pair‑mask to the left/right projections only.
5. Compute the N³ contraction
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`
   as a batched half‑precision GEMM: (B·H,N,N) @ (B·H,N,N)ᵀ → (B·H,N,N).
6. Apply a second FP16 LayerNorm over the hidden dimension.
7. Multiply by the output gate, then a final half‑precision linear
   projection back to the model dimension D.
8. Cast the result back to FP32.

All heavy arithmetic is done in FP16 with FP32 accumulation for numerical
stability. Two custom Triton kernels implement the two LayerNorms; the
linear layers and the N³ batched GEMM use cuBLAS (via torch.nn.functional.linear
and torch.bmm), which on an H100 provides >200 TFLOP/s in FP16, easily
meeting the sub‑millisecond target for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# FP16 LayerNorm (single‑pass, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range handled by this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize and apply affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_x_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    FP16 LayerNorm using the Triton kernel above.
    Expects a 2‑D tensor of shape [M, N] where N is the feature dimension.
    """
    assert x.dtype == torch.float16, "input must be FP16"
    assert x.dim() == 2, "input must be 2‑D"
    M, N = x.shape
    y = torch.empty_like(x)

    # make all tensors contiguous (required for stride extraction)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # larger row block improves occupancy for large M
        BLOCK_N=64,
    )
    return y

# ----------------------------------------------------------------------
# Outgoing TriMul forward pass (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim          (model dimension D)
                           - hidden_dim   (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0️⃣ Cast input to FP16 (the rest of the pipeline runs in FP16)
    # --------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16, fused pass)
    # --------------------------------------------------------------
    M = bs * N * N
    x_flat = x.reshape(M, dim)                     # (M, D)
    norm_w = weights["norm.weight"].to(device, torch.float16)
    norm_b = weights["norm.bias"].to(device, torch.float16)
    x_norm_flat = triton_layernorm_fp16(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)       # (B, N, N, D)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # --------------------------------------------------------------
    # Gather weight tensors and convert to half once
    w_lproj   = weights["left_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rproj   = weights["right_proj.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    weight_fused = torch.cat(
        [w_lproj, w_lgate, w_rproj, w_rgate, w_outgate], dim=0
    ).to(device, torch.float16)                     # (5*H, D)

    # (M, D) @ (D, 5*H) -> (M, 5*H)
    fused = F.linear(x_norm, weight_fused)          # (M, 5*H)
    fused = fused.view(bs, N, N, 5 * H)             # (B, N, N, 5H)

    # --------------------------------------------------------------
    # 3️⃣ Split, apply sigmoid gates and (optional) mask
    # --------------------------------------------------------------
    h0 = 0
    h1 = H
    h2 = 2 * H
    h3 = 3 * H
    h4 = 4 * H
    h5 = 5 * H

    left_proj   = fused[..., h0:h1]   # (B,N,N,H)
    left_gate_p = fused[..., h1:h2]
    right_proj  = fused[..., h2:h3]
    right_gate_p = fused[..., h3:h4]
    out_gate_p  = fused[..., h4:h5]

    # Sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Apply pair‑mask before gating, exactly as in the reference code
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Element‑wise gating (in‑place to reduce temporaries)
    left = left_proj * left_gate     # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # --------------------------------------------------------------
    # 4️⃣ Core N³ contraction: batched half‑precision GEMM
    # --------------------------------------------------------------
    # Re‑order to (B, H, N, N) → view as (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).reshape(bs * H, N, N)   # (BH, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * H, N, N)  # (BH, N, N)

    # Batched matrix multiplication with the second operand transposed
    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))    # (BH, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16 over the hidden dimension)
    # --------------------------------------------------------------
    out_flat = out.reshape(M, H)                         # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to D
    # --------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H)
    gated = out_norm * out_gate_flat                     # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)              # (M, D)

    # Cast back to float32 and restore original shape
    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
```
