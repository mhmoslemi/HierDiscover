@triton.jit
def layernorm_forward_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn, stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass: sum and sumsq
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for i in range(tl.cdiv(N, BLOCK_N)):
        offs_n = i * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)
    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for i in range(tl.cdiv(N, BLOCK_N)):
        offs_n = i * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f = x.to(tl.float32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]

        # Cast to half for store
        tl.store(y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
                 y.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – optimized Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Input LayerNorm (FP32 statistics, FP16 output) – 2‑pass Triton kernel.
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    into one half‑precision GEMM.
3️⃣  Apply the binary mask (if present) and the three sigmoid gates.
4️⃣  Compute the tri‑multiplicative update with a single ``torch.einsum`` which
    internally folds into a batched GEMM (no explicit permute / view copies).
5️⃣  Output LayerNorm (same kernel as step 1, FP16 output).
6️⃣  Multiply by the output gate and perform the final linear projection back to
    the model dimension (all half‑precision).
7️⃣  Cast the result to ``float32`` – the only FP32 tensor that leaves the kernel.

All heavy arithmetic (projections, TriMul, final projection) is carried out in
FP16 with FP32 accumulation where needed (layer‑norm statistics).  The only
explicit Triton code is the two‑pass LayerNorm which eliminates the large
FP32↔FP16 conversions present in a naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 stats, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                      # [M, N] input (float32 or float16)
    y_ptr,                      # [M, N] output (float16)
    weight_ptr, bias_ptr,       # [N] (float32)
    M, N, eps,                  # dimensions
    stride_xm, stride_xn,       # input strides
    stride_ym, stride_yn,       # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row block for this program instance
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣ First pass – compute mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = it * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)               # cast to FP32 for statistics
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass – normalise, affine, cast to FP16
    # ------------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = it * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        # (x - mean) * rstd   ->  FP32
        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]   # still FP32
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(tl.float16),                  # cast to FP16 for the output
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_fp16(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_N: int = 64,
) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor (M×N) that returns
    ``float16`` output while using FP32 statistics.
    The input ``x`` may be float32 or float16; the output is always float16.
    """
    assert x.dim() == 2, "LayerNorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty(M, N, dtype=torch.float16, device=x.device)

    # ensure contiguity for stride calculations
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator (AlphaFold‑3 version).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]  (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or ``None``
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys ``dim`` (model dim), ``hidden_dim``,
                         and optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 stats → FP16 output)
    # ------------------------------------------------------------------
    # reshape to (B * N * N, dim) for 2‑D LayerNorm kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)              # (M, dim) float32
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"  ].to(device, dtype=torch.float32)
    x_norm_flat = triton_layernorm_fp16(
        x_flat, norm_weight, norm_bias, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )                                                   # (M, dim) float16
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 2️⃣ Combined projection (half‑precision)
    # ------------------------------------------------------------------
    # flatten for the fused GEMM
    x_norm_flat_h = x_norm.view(M, dim)                 # (M, dim) fp16
    # concatenate the five projection weight matrices
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                                          # (5*H, dim)
    ).to(device, dtype=torch.float16)

    proj_out = F.linear(x_norm_flat_h, combined_weight)  # (M, 5*H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split into left/right and gates
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    # reshape back to (B, N, N, H)
    left_raw      = left_raw.view      (bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view     (bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view (bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view  (bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and their application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)    # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_raw)   # (B,N,N,H) fp16
    out_gate   = torch.sigmoid(out_gate_raw)    # (B,N,N,H) fp16

    left  = left_raw  * left_gate               # (B,N,N,H) fp16
    right = right_raw * right_gate              # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative update via a fused einsum (batched GEMM)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 stats → FP16 output)
    # ------------------------------------------------------------------
    out_flat_fp16 = out.view(bs * seq_len * seq_len, hidden_dim)     # (M, H) fp16
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)
    out_norm_flat = triton_layernorm_fp16(
        out_flat_fp16, to_out_norm_w, to_out_norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )                                                               # (M, H) fp16
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H) fp16
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    final_fp16 = F.linear(gated, to_out_w)          # (B,N,N,dim) fp16
    final_fp32 = final_fp16.float()                 # (B,N,N,dim) float32

    return final_fp32
