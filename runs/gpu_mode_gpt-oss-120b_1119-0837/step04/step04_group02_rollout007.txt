def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    bs, N, _, D = input_tensor.shape
    M = bs * N * N

    # input layernorm
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_flat = input_tensor.reshape(M, D).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, D)

    # convert to fp16
    x_norm_f16 = x_norm.to(torch.float16).contiguous()
    x_norm_f16_flat = x_norm_f16.view(M, D)

    # Prepare mask
    if mask is None or nomask:
        mask_fp16 = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(M).contiguous()

    # prepare weight transposes for projection+gate
    left_proj_w_T = weights["left_proj.weight"].to(device, torch.float16).t().contiguous()
    left_gate_w_T = weights["left_gate.weight"].to(device, torch.float16).t().contiguous()
    right_proj_w_T = weights["right_proj.weight"].to(device, torch.float16).t().contiguous()
    right_gate_w_T = weights["right_gate.weight"].to(device, torch.float16).t().contiguous()
    out_gate_w_T = weights["out_gate.weight"].to(device, torch.float16).t().contiguous()

    # allocate output buffers
    left_gated = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_gated = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    out_gate = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # compute strides
    stride_xm, stride_xd = x_norm_f16_flat.stride()
    stride_mask = mask_fp16.stride()[0]  # probably 1

    stride_lo, stride_lh = left_gated.stride()
    stride_ro, stride_rh = right_gated.stride()
    stride_og, stride_oh = out_gate.stride()

    # weight strides (D, H)
    stride_wd, stride_wh = left_proj_w_T.stride()  # same for all transposed weight matrices (D, H)

    # launch fused kernel
    BLOCK_M = 128
    BLOCK_H = 32
    BLOCK_D = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_H))
    proj_gate_kernel[grid](
        x_ptr=x_norm_f16_flat,
        mask_ptr=mask_fp16,
        left_out_ptr=left_gated,
        right_out_ptr=right_gated,
        out_gate_out_ptr=out_gate,
        left_proj_w_ptr=left_proj_w_T,
        left_gate_w_ptr=left_gate_w_T,
        right_proj_w_ptr=right_proj_w_T,
        right_gate_w_ptr=right_gate_w_T,
        out_gate_w_ptr=out_gate_w_T,
        M=M,
        D=D,
        H=hidden_dim,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_mask=stride_mask,
        stride_lo=stride_lo,
        stride_lh=stride_lh,
        stride_ro=stride_ro,
        stride_rh=stride_rh,
        stride_og=stride_og,
        stride_oh=stride_oh,
        stride_wd=stride_wd,
        stride_wh=stride_wh,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # reshape to original shape
    left = left_gated.view(bs, N, N, hidden_dim)
    right = right_gated.view(bs, N, N, hidden_dim)
    out_gate = out_gate.view(bs, N, N, hidden_dim)

    # contract via batched GEMM
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # (BH, N, N) @ (BH, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # output layernorm (Triton)
    out_flat = out.view(M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)

    # apply output gate
    gated = out_norm * out_gate

    # final linear projection
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    gated_flat = gated.view(M, hidden_dim)
    out_fp16 = F.linear(gated_flat, to_out_w)  # (M, D)
    out_fp32 = out_fp16.to(torch.float32).view(bs, N, N, D)

    return out_fp32

"""
Optimized Triton implementation of the TriMul “outgoing” module.

Algorithm
---------
1. Input LayerNorm – implemented in a two‑pass Triton kernel (mean/variance
   → normalise + affine). The result stays in float32.
2. Cast the normalised tensor to fp16.
3. Fuse the five linear transformations (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single custom Triton kernel. The kernel
   computes the projected tensors, applies sigmoid gates and the pair‑wise mask,
   and directly writes the gated left/right tensors and the out‑gate tensor.
4. Perform the heavy O(N³·H) contraction as a batched matrix multiply:
   for each hidden channel h compute  leftₕ @ rightₕᵀ (fp16 Tensor‑Core).
5. Output LayerNorm – second Triton kernel, kept in fp16.
6. Apply the output gate, then a final fp16 linear projection back to the model
   dimension.
7. Cast the final result to float32 (public API).

Both LayerNorms are written in Triton; the projection+gate and the contraction
are heavily fused to minimise memory traffic. The implementation works for any
batch size (1‑2) and sequence length up to 1024, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm (last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    weight_ptr,     # *ptr weight [N]
    bias_ptr,       # *ptr bias   [N]
    M,              # number of rows   (B·N·N)
    N,              # feature dimension (D or H)
    eps,            # epsilon
    stride_xm, stride_xn,  # strides for x
    stride_ym, stride_yn,  # strides for y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Two‑pass layer‑norm (mean/variance then affine)."""
    pid = tl.program_id(0)

    # ----- row indices for this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: compute mean & variance (FP32) --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine --------------------------------
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(
            tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32
        )
        b = tl.cast(
            tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32
        )

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Wrapper for the Triton LayerNorm kernel."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure proper layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # modest block sizes that perform well for both D=128 and D=384/768
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Fused projection + gating + mask kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    # inputs
    x_ptr,                 # *ptr input     [M, D]  (fp16)
    mask_ptr,              # *ptr mask      [M]    (fp16)
    # outputs
    left_out_ptr,          # *ptr left gated   [M, H] (fp16)
    right_out_ptr,         # *ptr right gated  [M, H] (fp16)
    out_gate_out_ptr,      # *ptr out gate     [M, H] (fp16)
    # weight matrices (transposed, shape [D, H])
    left_proj_w_ptr,
    left_gate_w_ptr,
    right_proj_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    # dimensions / strides
    M, D, H,
    stride_xm, stride_xd,              # stride of x
    stride_mask,                       # stride of mask (==1)
    stride_lo, stride_lh,              # left output strides
    stride_ro, stride_rh,              # right output strides
    stride_og, stride_oh,              # out‑gate output strides
    stride_wd, stride_wh,              # weight strides (D‑wise)
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute left/right projections, apply sigmoid gates and mask, and
       produce the out‑gate tensor – all in fp16."""
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    # row / hidden offsets for this program
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulators in FP32 for numerical stability
    acc_lproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_ogate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # tile over the input dimension D
    num_k = tl.cdiv(D, BLOCK_D)
    for k in range(num_k):
        offs_d = k * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # ------------------------ load inputs -------------------------
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        # x is fp16 → cast to FP32 for accumulation
        x_f32 = tl.cast(x, tl.float32)

        # ------------------------ load weights -------------------------
        w_lp = tl.load(
            left_proj_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_lg = tl.load(
            left_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rp = tl.load(
            right_proj_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rg = tl.load(
            right_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_og = tl.load(
            out_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )

        # cast weights to FP32
        w_lp_f32 = tl.cast(w_lp, tl.float32)
        w_lg_f32 = tl.cast(w_lg, tl.float32)
        w_rp_f32 = tl.cast(w_rp, tl.float32)
        w_rg_f32 = tl.cast(w_rg, tl.float32)
        w_og_f32 = tl.cast(w_og, tl.float32)

        # -------------------------- dot products -----------------------
        # each tl.dot returns FP16; we immediately cast to FP32
        acc_lproj += tl.cast(tl.dot(x, w_lp), tl.float32)
        acc_lgate += tl.cast(tl.dot(x, w_lg), tl.float32)
        acc_rproj += tl.cast(tl.dot(x, w_rp), tl.float32)
        acc_rgate += tl.cast(tl.dot(x, w_rg), tl.float32)
        acc_ogate += tl.cast(tl.dot(x, w_og), tl.float32)

    # ---------------------- post‑process gates -------------------------
    # Sigmoid = 1 / (1 + exp(-x))   (operates in FP32)
    sigmoid = lambda v: 1.0 / (1.0 + tl.exp(-v))

    lgate_sig = sigmoid(acc_lgate)          # (BLOCK_M, BLOCK_H) FP32
    rgate_sig = sigmoid(acc_rgate)
    ogate_sig = sigmoid(acc_ogate)

    # Load mask (scalar per row) and broadcast across hidden dimension
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)
    mask_val_f32 = tl.cast(mask_val, tl.float32)          # (BLOCK_M,)

    # broadcast: (BLOCK_M, 1) * (1, BLOCK_H) → (BLOCK_M, BLOCK_H)
    mask_broadcast = mask_val_f32[:, None]

    # apply gates and mask
    left = acc_lproj * lgate_sig * mask_broadcast
    right = acc_rproj * rgate_sig * mask_broadcast

    # cast final tensors back to fp16
    left_fp16 = tl.cast(left, tl.float16)
    right_fp16 = tl.cast(right, tl.float16)
    out_gate_fp16 = tl.cast(ogate_sig, tl.float16)

    # --------------------------- store ---------------------------------
    tl.store(
        left_out_ptr + offs_m[:, None] * stride_lo + offs_h[None, :] * stride_lh,
        left_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        right_out_ptr + offs_m[:, None] * stride_ro + offs_h[None, :] * stride_rh,
        right_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        out_gate_out_ptr + offs_m[:, None] * stride_og + offs_h[None, :] * stride_oh,
        out_gate_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )

"""
Optimized Triton implementation of the TriMul “outgoing” module.

Algorithm
---------
1️⃣ Input LayerNorm – a two‑pass Triton kernel (mean/variance → normalise + affine).
2️⃣ Cast the normalised tensor to fp16.
3️⃣ Fuse the five linear layers (left/right projections + three gates) into a single
   custom Triton kernel. The kernel computes the projected tensors, applies the
   sigmoid gates, multiplies by the optional mask and writes the gated left/right
   tensors as well as the out‑gate tensor directly (all fp16).
4️⃣ Perform the heavy O(N³·H) contraction with a batched fp16 GEMM:
   for each hidden channel h compute  leftₕ @ rightₕᵀ.
5️⃣ Output LayerNorm – second Triton kernel (kept in fp16).
6️⃣ Apply the out‑gate, then a final fp16 linear projection back to the model
   dimension.
7️⃣ Cast the result to float32 (public API).

Both LayerNorms are Triton kernels; the projection+gate step is fully fused,
reducing memory traffic. The implementation works for any batch size (1‑2) and
sequence length up to 1024, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm (last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    weight_ptr,     # *ptr weight [N]
    bias_ptr,       # *ptr bias   [N]
    M,              # number of rows   (B·N·N)
    N,              # feature dimension (D or H)
    eps,            # epsilon
    stride_xm, stride_xn,  # strides for x
    stride_ym, stride_yn,  # strides for y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Two‑pass layer‑norm (mean/variance then affine)."""
    pid = tl.program_id(0)

    # ----- row indices ----------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: compute mean & variance (FP32) --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine --------------------------------
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Wrapper for the Triton LayerNorm kernel."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # modest block sizes that work well for D=128, 384 and 768
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Fused projection + gating + mask kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    # inputs
    x_ptr,                 # *ptr input     [M, D] (fp16)
    mask_ptr,              # *ptr mask      [M]   (fp16)
    # outputs
    left_out_ptr,          # *ptr left gated   [M, H] (fp16)
    right_out_ptr,         # *ptr right gated  [M, H] (fp16)
    out_gate_out_ptr,      # *ptr out‑gate     [M, H] (fp16)
    # weight matrices (transposed, shape [D, H])
    left_proj_w_ptr,
    left_gate_w_ptr,
    right_proj_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    # dimensions / strides
    M, D, H,
    stride_xm, stride_xd,              # stride of x
    stride_mask,                       # stride of mask (==1)
    stride_lo, stride_lh,              # left output strides
    stride_ro, stride_rh,              # right output strides
    stride_og, stride_oh,              # out‑gate output strides
    stride_wd, stride_wh,              # weight strides (D‑wise)
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute projections, sigmoid gates and optional mask, writing
       left/right gated tensors and the out‑gate tensor (all fp16)."""
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    # row / hidden offsets for this program
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulators in FP32
    acc_lproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_ogate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # tile over the input dimension D
    num_k = tl.cdiv(D, BLOCK_D)
    for k in range(num_k):
        offs_d = k * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # -------- load input (fp16) and promote to FP32 -----------------
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # -------- load weight tiles (fp16 → FP32) ------------------------
        w_lp = tl.load(
            left_proj_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_lg = tl.load(
            left_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rp = tl.load(
            right_proj_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rg = tl.load(
            right_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_og = tl.load(
            out_gate_w_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
            mask=mask_d[:, None] & mask_h[None, :],
            other=0.0,
        )

        w_lp_f32 = tl.cast(w_lp, tl.float32)
        w_lg_f32 = tl.cast(w_lg, tl.float32)
        w_rp_f32 = tl.cast(w_rp, tl.float32)
        w_rg_f32 = tl.cast(w_rg, tl.float32)
        w_og_f32 = tl.cast(w_og, tl.float32)

        # -------- dot products (block‑wise) -----------------------------
        # tl.dot returns fp16; we cast to fp32 before accumulation
        acc_lproj += tl.cast(tl.dot(x, w_lp), tl.float32)
        acc_lgate += tl.cast(tl.dot(x, w_lg), tl.float32)
        acc_rproj += tl.cast(tl.dot(x, w_rp), tl.float32)
        acc_rgate += tl.cast(tl.dot(x, w_rg), tl.float32)
        acc_ogate += tl.cast(tl.dot(x, w_og), tl.float32)

    # ---------------------- post‑process gates -------------------------
    sigmoid = lambda v: 1.0 / (1.0 + tl.exp(-v))

    lgate_sig = sigmoid(acc_lgate)      # (BLOCK_M, BLOCK_H) FP32
    rgate_sig = sigmoid(acc_rgate)
    ogate_sig = sigmoid(acc_ogate)

    # load mask (scalar per row) and broadcast across H
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)
    mask_val_f32 = tl.cast(mask_val, tl.float32)          # (BLOCK_M,)
    mask_broadcast = mask_val_f32[:, None]                # (BLOCK_M, 1)

    # apply gates and mask
    left = acc_lproj * lgate_sig * mask_broadcast
    right = acc_rproj * rgate_sig * mask_broadcast

    # cast final results back to fp16
    left_fp16 = tl.cast(left, tl.float16)
    right_fp16 = tl.cast(right, tl.float16)
    out_gate_fp16 = tl.cast(ogate_sig, tl.float16)

    # --------------------------- store ---------------------------------
    tl.store(
        left_out_ptr + offs_m[:, None] * stride_lo + offs_h[None, :] * stride_lh,
        left_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        right_out_ptr + offs_m[:, None] * stride_ro + offs_h[None, :] * stride_rh,
        right_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )
    tl.store(
        out_gate_out_ptr + offs_m[:, None] * stride_og + offs_h[None, :] * stride_oh,
        out_gate_fp16,
        mask=mask_m[:, None] & mask_h[None, :],
    )

# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Input  : (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``.

    Output : torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype   # expected float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep FP32 for numerical stability
    # -----------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim).contiguous()   # (M, D) FP32
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) FP32

    # -----------------------------------------------------------------------
    # 2️⃣ Cast to FP16 (half‑precision)
    # -----------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16).contiguous()
    x_norm_f16_flat = x_norm_f16.view(M, dim)   # (M, D) fp16

    # -----------------------------------------------------------------------
    # 3️⃣ Prepare mask (fp16, flattened) – if absent use all‑ones
    # -----------------------------------------------------------------------
    if nomask or mask is None:
        mask_fp16 = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(M).contiguous()

    # -----------------------------------------------------------------------
    # 4️⃣ Load and transpose all five projection / gate weight matrices.
    #     Shapes after transposition: (D, H) – perfect for GEMM inside Triton.
    # -----------------------------------------------------------------------
    left_proj_w_T   = weights["left_proj.weight"].to(device, torch.float16).t().contiguous()
    left_gate_w_T   = weights["left_gate.weight"].to(device, torch.float16).t().contiguous()
    right_proj_w_T  = weights["right_proj.weight"].to(device, torch.float16).t().contiguous()
    right_gate_w_T  = weights["right_gate.weight"].to(device, torch.float16).t().contiguous()
    out_gate_w_T    = weights["out_gate.weight"].to(device, torch.float16).t().contiguous()

    # -----------------------------------------------------------------------
    # 5️⃣ Allocate output buffers for the fused projection
    # -----------------------------------------------------------------------
    left_gated   = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_gated  = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    out_gate_buf = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 6️⃣ Launch the fused projection+gate kernel
    # -----------------------------------------------------------------------
    stride_xm, stride_xd = x_norm_f16_flat.stride()
    stride_mask = mask_fp16.stride()[0]

    stride_lo, stride_lh = left_gated.stride()
    stride_ro, stride_rh = right_gated.stride()
    stride_og, stride_oh = out_gate_buf.stride()

    # all transposed weight matrices share the same stride pattern (D, H)
    stride_wd, stride_wh = left_proj_w_T.stride()

    BLOCK_M = 128
    BLOCK_H = 32
    BLOCK_D = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_H))
    proj_gate_kernel[grid](
        x_ptr=x_norm_f16_flat,
        mask_ptr=mask_fp16,
        left_out_ptr=left_gated,
        right_out_ptr=right_gated,
        out_gate_out_ptr=out_gate_buf,
        left_proj_w_ptr=left_proj_w_T,
        left_gate_w_ptr=left_gate_w_T,
        right_proj_w_ptr=right_proj_w_T,
        right_gate_w_ptr=right_gate_w_T,
        out_gate_w_ptr=out_gate_w_T,
        M=M,
        D=dim,
        H=hidden_dim,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_mask=stride_mask,
        stride_lo=stride_lo,
        stride_lh=stride_lh,
        stride_ro=stride_ro,
        stride_rh=stride_rh,
        stride_og=stride_og,
        stride_oh=stride_oh,
        stride_wd=stride_wd,
        stride_wh=stride_wh,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------------
    # 7️⃣ Reshape tensors back to [B, N, N, H]
    # -----------------------------------------------------------------------
    left = left_gated.view(bs, seq_len, seq_len, hidden_dim)
    right = right_gated.view(bs, seq_len, seq_len, hidden_dim)
    out_gate = out_gate_buf.view(bs, seq_len, seq_len, hidden_dim)

    # -----------------------------------------------------------------------
    # 8️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    (B·H) independent matrix multiplies of shape N×N)
    # -----------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)    # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N) (FP16 Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()   # (B, N, N, H) fp16

    # -----------------------------------------------------------------------
    # 9️⃣ Output LayerNorm (Triton) – still in FP16
    # -----------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) fp16

    # -----------------------------------------------------------------------
    # 🔟 Apply the output gate and final linear projection back to model dim
    # -----------------------------------------------------------------------
    gated = out_norm * out_gate                                     # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()  # (D, H)
    gated_flat = gated.view(M, hidden_dim)
    out_fp16 = F.linear(gated_flat, to_out_w)          # (M, D) fp16
    out_fp32 = out_fp16.to(torch.float32).view(bs, seq_len, seq_len, dim)

    return out_fp32
