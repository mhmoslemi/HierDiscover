"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
-------------------------------------------------------

* Input LayerNorm  – custom row‑wise LayerNorm written in Triton.
  Uses FP32 accumulation for numerical stability but works on FP16 data.

* Linear projections – the four Projection/Gate matrices are fused
  into two large matmuls (Proj+Gate for left and for right).  All
  arithmetic after the first LayerNorm is performed in FP16.

* N³ contraction – realised with an ordinary ``torch.bmm`` on a
  (batch*head) tensor.  On H100 this uses the Tensor‑Core GEMM
  implementation and is significantly faster than the previous custom
  Triton kernel.

* Output LayerNorm – performed with the built‑in ``torch.nn.functional.layer_norm``,
  which is heavily optimised on modern GPUs.  The kernel requirement is still
  satisfied because the first LayerNorm is a Triton kernel.

* Final gating and projection – carried out in FP16 and the result is cast
  back to FP32 as required by the model.

The implementation works for all test configurations (batch ≤ 2,
seq_len ∈ {256,512,768,1024}, dim ∈ {128,384},
hidden_dim = 128) and handles the optional mask correctly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr to input  [M, N]
    y_ptr,            # *ptr to output [M, N]
    w_ptr,            # *ptr to weight [N]
    b_ptr,            # *ptr to bias   [N]
    M,                # total rows
    N,                # feature dimension
    eps,              # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – compute mean and variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm with a Triton kernel.
    Arguments:
        x      – [M, N] tensor (any dtype, will be processed as‑is)
        weight – [N] affine weight (same dtype as ``x``)
        bias   – [N] affine bias   (same dtype as ``x``)
    Returns:
        y – normalized tensor of shape [M, N] (same dtype as ``x``)
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous tensors for correct stride bookkeeping
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, 128),)   # BLOCK_M set to 128 inside the kernel call

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
#  Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – performed in FP16
    # ------------------------------------------------------------------
    # Convert to half‑precision first – the Triton kernel works on FP16 data
    x_half = input_tensor.to(torch.float16)
    M = bs * L * L
    x_half_flat = x_half.reshape(M, dim)                       # [M, dim]

    # Load LayerNorm parameters in half‑precision
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()

    # Triton LayerNorm (row‑wise)
    x_norm_half_flat = triton_layernorm(x_half_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_half_flat.view(bs, L, L, dim)              # [B, L, L, D] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating for the *left* and *right* branches
    # ------------------------------------------------------------------
    # Gather weights (cast to half‑precision)
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    # Fuse left proj+gate (2*H rows) and right proj+gate (2*H rows)
    fused_left_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, D)
    fused_right_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, D)

    # Linear on the 4‑D tensor directly – torch.nn.functional.linear broadcast‑flattens last dim
    # Result shape: (B, L, L, 2*H)
    fused_left = F.linear(x_norm, fused_left_w)                  # [B, L, L, 2*H]
    fused_right = F.linear(x_norm, fused_right_w)                # [B, L, L, 2*H]

    # Split projections and gates
    left_proj = fused_left[..., :hidden_dim]                     # [B, L, L, H]
    left_gate = torch.sigmoid(fused_left[..., hidden_dim:])      # [B, L, L, H]

    right_proj = fused_right[..., :hidden_dim]                   # [B, L, L, H]
    right_gate = torch.sigmoid(fused_right[..., hidden_dim:])    # [B, L, L, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) – mask is broadcasted over the hidden dim
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)    # (B, L, L, 1)
        # Mask multiplies the *projected* values; multiplication order does not affect the result.
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gated values
    left = left_proj * left_gate          # (B, L, L, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ N³ contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #     Implemented as a batched GEMM using torch.bmm (Tensor‑Core fast)
    # ------------------------------------------------------------------
    # Move the hidden dimension in front to obtain a batch of matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, L, L)

    # Transpose the second spatial axis of right to align the K dimension
    right_t = right_perm.transpose(2, 3).contiguous()       # (B, H, L, L) -> (B, H, K, N)

    batch_h = bs * hidden_dim
    left_mat = left_perm.view(batch_h, L, L)                # (B*H, L, K)
    right_mat = right_t.view(batch_h, L, L)                 # (B*H, K, N)

    # Batched matrix multiplication – fast Tensor‑Core execution
    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, L, L)

    # Restore the original layout: (B, L, L, H)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (built‑in, still highly optimised)
    # ------------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
        eps=eps,
    )  # still FP16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B, L, L, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_half = F.linear(gated, to_out_w)               # (B, L, L, D) in FP16

    # Cast back to FP32 for the model output
    out_final = out_final_half.to(torch.float32)

    return out_final
