"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with heavy parts
implemented in Triton.

Algorithm
---------
1️⃣ Input LayerNorm (FP32, cuDNN) → normalises last dimension *dim*.
2️⃣ Cast to FP16 for all matrix work.
3️⃣ Fuse projection + gate for the left and right sides:
   each side = Linear(x, [proj; gate])  → shape (B, N, N, 2·H).
4️⃣ (Optional) apply mask to the *projection* part only.
5️⃣ Sigmoid the gating parts and multiply → (B,N,N,H) tensors.
6️⃣ Compute the N³ contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is a batched GEMM of shape (B·H, N, N) performed with
   torch.bmm (the fastest path on an H100).
7️⃣ Second LayerNorm on the hidden dimension *H* **and**
   element‑wise multiplication by the output‑gate are fused into a
   single Triton kernel (`layernorm_fp16_gate_kernel`).  This removes
   an extra memory pass.
8️⃣ Final linear projection back to model dimension *dim* (FP16 matmul)
   and cast to FP32 for the returned tensor.

Only the second LayerNorm+gate step is written in Triton; all other
operations stay in PyTorch for simplicity while still achieving
high performance on the target hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: FP16 LayerNorm + element‑wise gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,       # input           (M, H)  fp16
    y_ptr,       # output          (M, H)  fp16
    gate_ptr,    # element‑wise gate (M, H) fp16
    w_ptr,       # LN weight       (H,)   fp16
    b_ptr,       # LN bias         (H,)   fp16
    M,           # number of rows
    H,           # hidden dimension per row
    eps,         # epsilon for variance
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_N: tl.constexpr,   # cols per program (tile width)
):
    pid = tl.program_id(0)

    ######################################################################
    # 1️⃣ Determine rows tackled by this program
    ######################################################################
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M                       # shape (BLOCK_M,)

    ######################################################################
    # 2️⃣ First pass – compute per‑row mean and variance (FP32)
    ######################################################################
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(H, BLOCK_N)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        # Load tile (BLOCK_M, BLOCK_N)
        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(xv, tl.float32)          # cast to FP32 for accumulation
        sum_val += tl.sum(xf, axis=1)         # sum over N dim
        sum_sq  += tl.sum(xf * xf, axis=1)

    # row‑wise statistics
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    ######################################################################
    # 3️⃣ Second pass – normalize, apply affine transform, then gate
    ######################################################################
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(xv, tl.float32)

        # (x - μ) / σ
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # affine transformation
        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        y = norm * w[None, :] + b[None, :]

        # gate (element‑wise multiplication)
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        y = y * g

        # store result
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_fp16_gate(x, weight, bias, gate, eps=1e-5):
    """
    Fused FP16 LayerNorm + gate scaling.

    Parameters
    ----------
    x : torch.Tensor
        Input of shape (M, H), dtype=torch.float16 (contiguous).
    weight, bias : torch.Tensor
        LayerNorm parameters of shape (H,), dtype=torch.float16.
    gate : torch.Tensor
        Gate tensor of shape (M, H), dtype=torch.float16 (contiguous).

    Returns
    -------
    torch.Tensor
        Normalised & gated output of shape (M, H), dtype=torch.float16.
    """
    assert x.dtype == torch.float16 and x.dim() == 2
    M, H = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous + simple strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    gate = gate.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    stride_gm, stride_gn = gate.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_gate_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        gate_ptr=gate,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        BLOCK_M=256,      # tuned for H100, works well for M up to several 10⁶
        BLOCK_N=128,      # full‑row tile (covers H ≤ 384 comfortably)
    )
    return y


# ----------------------------------------------------------------------
# Main entry point – custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] or None (bool/int/float)
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys:
              * dim        – model dimension D
              * hidden_dim – hidden size H (default 128)
              * nomask     – bool, True if mask should be ignored

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
        The processed tensor.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use cuDNN – fastest for FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # (B, N, N, D)   FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right sides
    # ------------------------------------------------------------------
    # left_comb_w : (2·H, D)   – first H rows = proj, next H rows = gate
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)          # (2*H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)          # (2*H, D)

    # out_gate weight : (H, D)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Perform two linear transformations (project+gate) in FP16
    left_comb  = F.linear(x_f16, left_comb_w)   # (B, N, N, 2·H)
    right_comb = F.linear(x_f16, right_comb_w)  # (B, N, N, 2·H)

    # Slice projection & gate components
    left_proj   = left_comb[..., :H]   # (B, N, N, H)
    left_gate_z = left_comb[..., H:]   # (B, N, N, H)
    right_proj  = right_comb[..., :H]
    right_gate_z = right_comb[..., H:]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool/int/float – cast to fp16 and broadcast
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)   # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_z)

    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction: batched GEMM over the third axis (k)
    #      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #      => for each (b,h) we compute left_mat @ right_matᵀ
    # ------------------------------------------------------------------
    # Rearrange tensors to (B·H, N, N) for a single torch.bmm call.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)    # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)   # (B·H, N, N)

    # Batched matrix product (B·H, N, N) * (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H) – still FP16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dimension + fused output‑gate scaling
    # ------------------------------------------------------------------
    M = bs * N * N                          # total rows to normalise
    out_flat = out.view(M, H).contiguous()                 # (M, H) FP16
    out_gate_flat = out_gate.view(M, H).contiguous()       # (M, H) FP16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)

    # Apply fused LayerNorm + gate
    out_norm_and_gate = triton_layernorm_fp16_gate(
        out_flat,
        to_out_norm_w,
        to_out_norm_b,
        out_gate_flat,
        eps=eps,
    )   # (M, H) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Final linear back to model dimension (FP16 matmul) → FP32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    # Torch linear expects weight shape (out_features, in_features) = (D, H)
    output_flat = F.linear(out_norm_and_gate, to_out_w)            # (M, D) FP16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)     # (B, N, N, D)

    return output
