"""
Efficient Triton‑accelerated forward pass for the outgoing TriMul operator
(AlphaFold3 “outgoing” variant).

Algorithm
---------
1️⃣ Input LayerNorm (float32) – performed with the highly‑optimized
    cuDNN implementation via `torch.nn.functional.layer_norm`.

2️⃣ Cast the normalized tensor to fp16 for the heavy linear work.

3️⃣ Fuse the three linear projections that share the same input:
   * left_proj + left_gate + out_gate  →  one matmul (3·H outputs)
   * right_proj + right_gate           →  one matmul (2·H outputs)

   The concatenated weight matrices are pre‑packed in fp16 and the
   matmuls are performed with a single `torch.nn.functional.linear`
   call each, drastically reducing kernel launch overhead.

4️⃣ Apply the (optional) pairwise mask to the left/right projections,
   compute the sigmoid gates and element‑wise gated tensors.

5️⃣ Core N³ contraction:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   → equivalent to a batched GEMM `left @ rightᵀ` over the hidden dimension.
   The tensor is reshaped to (B·H, N, N) and a single `torch.bmm` call
   computes all B·H matrix multiplications in one launch.

6️⃣ Fuse the second LayerNorm, its affine transformation and the output‑gate
   scaling into a single Triton kernel (`layernorm_fp16_gate_kernel`).
   The kernel performs a two‑pass mean/variance computation (FP32
   accumulation) and writes the normalized & gated result in FP16.

7️⃣ Final linear projection back to the model dimension (fp16 matmul) and
   cast the result to float32.

All heavy tensor arithmetic is carried out in FP16, while the two
LayerNorms use FP32 accumulators for numerical stability.  The routine
conforms to the required signature `custom_kernel(data)` and works for
all test configurations (varying batch size, sequence length, model
dimension, hidden dimension and optional mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (fp16) + element‑wise gate scaling
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,                 # input  (M, H)  fp16
    y_ptr,                 # output (M, H)  fp16
    gate_ptr,              # gate   (M, H)  fp16
    weight_ptr,            # LN weight (H)  fp16
    bias_ptr,              # LN bias   (H)  fp16
    M,                     # number of rows
    H,                     # hidden dimension
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_gm, stride_gn,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_N: tl.constexpr    # cols per program (sub‑tile)
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row range for this program
    # --------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: write normalized + affine + gate
    # --------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < H

        xv = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(xv, tl.float32)

        # (x - μ) / σ
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # affine: γ * norm + β
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)
        y = norm * w[None, :] + b[None, :]

        # gate
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        y = y * g

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16_gate(
    x: torch.Tensor,          # (M, H) fp16
    weight: torch.Tensor,     # (H,) fp16
    bias: torch.Tensor,       # (H,) fp16
    gate: torch.Tensor,       # (M, H) fp16
    eps: float = 1e-5
) -> torch.Tensor:
    """LayerNorm (fp16) with fused gate scaling via Triton."""
    assert x.dim() == 2 and x.dtype == torch.float16
    M, H = x.shape
    y = torch.empty_like(x)

    # ensure contiguous tensors for simple stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    gate = gate.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    stride_gm, stride_gn = gate.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_gate_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        gate_ptr=gate,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        BLOCK_M=256,                 # tuned for H100
        BLOCK_N=128,                 # full‑row tile for H≤384
    )
    return y


# ----------------------------------------------------------------------
# Entry point – custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional, defaults to False)

    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – cuDNN implementation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, D)  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left_proj + left_gate + out_gate, and right_proj + right_gate
    # ------------------------------------------------------------------
    # Concatenate weights for the fused linear layers (stored as (out_feat, in_feat))
    left_comb_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                     # (3·H, D)

    right_comb_w = torch.cat(
        [
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                     # (2·H, D)

    # Combined linear calls (single matmul each)
    left_comb  = F.linear(x_f16, left_comb_w)       # (B, N, N, 3·H)
    right_comb = F.linear(x_f16, right_comb_w)      # (B, N, N, 2·H)

    # Split the fused outputs
    left_proj, left_gate_z, out_gate_z = torch.split(left_comb, split_size_or_sections=H, dim=-1)
    right_proj, right_gate_z = torch.split(right_comb, split_size_or_sections=H, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)   # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_z)  # (B, N, N, H)
    out_gate   = torch.sigmoid(out_gate_z)    # (B, N, N, H)

    left  = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Transform tensors to (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # (B·H, N, N) = left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)  (still fp16)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm + output‑gate (fused via Triton)
    # ------------------------------------------------------------------
    M = bs * N * N
    out_flat = out.view(M, H).contiguous()            # (M, H) fp16
    out_gate_flat = out_gate.view(M, H).contiguous()  # (M, H) fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)

    # fused LN + gate → (M, H) fp16
    out_norm = triton_layernorm_fp16_gate(
        out_flat,
        to_out_norm_w,
        to_out_norm_b,
        out_gate_flat,
        eps=eps,
    )  # (M, H) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dim (fp16) → fp32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    output_flat = F.linear(out_norm, to_out_w)                     # (M, D) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)     # (B, N, N, D) fp32

    return output
