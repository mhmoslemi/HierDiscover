"""
custom_kernel – Triton‑accelerated “outgoing’’ TriMul forward pass.

Algorithm
---------
1️⃣  Fuse the three linear projections (left/right) and the three gating
    linear layers (left_gate, right_gate, out_gate) into a single GEMM.
2️⃣  Two Triton Layer‑Norm kernels (one on the input, one on the
    intermediate tensor).  Both kernels operate on a flat 2‑D view
    (M = B·N·N rows, D or H columns) and use FP32 accumulation for
    numerical stability while keeping the data in the original dtype
    (FP16 after the first norm).
3️⃣  The heavy O(N³·H) contraction is performed with a batched half‑precision
    matrix multiplication (torch.bmm) after reshaping the hidden dimension.
4️⃣  The second Layer‑Norm output is multiplied by the output‑gate and
    projected back to model dimension with a final half‑precision GEMM.
5️⃣  The result is cast back to FP32 (the public API expects FP32).

Only the Layer‑Norms are written in Triton – everything else leverages
high‑performance cuBLAS/Tensor‑Core kernels from PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm (mean/var → normalise + affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # rows   (B·N·N)
    N,              # cols   (D or H)
    eps,            # epsilon
    stride_xm, stride_xn,   # stride of input
    stride_ym, stride_yn,   # stride of output
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_N: tl.constexpr,  # cols per program
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = it * BLOCK_N + tl.arange(0, BLOCK_N)          # [BLOCK_N]
        mask_n = offs_n < N

        # Load a tile of shape (BLOCK_M, BLOCK_N)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / tl.float32(N)
    var  = sum_sq / tl.float32(N) - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine
    # ------------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = it * BLOCK_N + tl.arange(0, BLOCK_N)          # [BLOCK_N]
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # (x - mean) * rstd
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # Load affine parameters (they may be FP16/FP32, cast to FP32)
        w = tl.cast(
            tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32
        )  # [BLOCK_N]
        b = tl.cast(
            tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32
        )  # [BLOCK_N]

        y = norm_x * w[None, :] + b[None, :]

        # Store back in the original dtype (usually FP16)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Wrapper around :func:`layernorm_kernel` that works on a 2‑D tensor.
    Arguments ``weight`` and ``bias`` may be FP16 or FP32.
    The output dtype matches the input dtype.
    """
    assert x.dim() == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tuned block sizes – work well for D/H = 128, 384, 768
    BLOCK_M = 64   # rows per program (fits many programs in a grid)
    BLOCK_N = 128  # columns per program (good for 128/384/768)

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

    # 0️⃣ Unpack & basic meta‑information
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # public API expects torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep result in FP32 for stability,
    #    then cast to FP16 for the heavy compute.
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)  # (M, D) – still FP32

    norm_w = weights["norm.weight"].to(device, dtype)   # (D)
    norm_b = weights["norm.bias"].to(device, dtype)     # (D)

    # Triton LayerNorm returns same dtype as input (FP32 here)
    x_norm_fp32 = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # Cast to half‑precision before the bulk of the computation
    x_norm = x_norm_fp32.to(torch.float16).view(bs, seq_len, seq_len, dim)  # (B,N,N,D) fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (two projections + three gates)
    #    into a single GEMM.  The weight shape becomes (5·H, D).
    # --------------------------------------------------------------
    # Collect weight tensors (they are stored as [out_features, in_features])
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature axis
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)            # (5·H, D) fp16

    # Flatten input for the linear projection
    x_norm_flat = x_norm.reshape(M, dim)   # (M, D) fp16

    # Single fused linear projection → (M, 5·H)
    proj_gate_flat = F.linear(x_norm_flat, combined_w)   # fp16

    # --------------------------------------------------------------
    # 3️⃣ Unpack projections & gates, apply sigmoid to gates
    # --------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)
    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)    # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        # Broadcast mask over the hidden dimension
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Heavy contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Reshape to (B·H, N, N) and use torch.bmm (Tensor‑Core).
    # --------------------------------------------------------------
    # Move hidden dimension to axis=1 for easy view
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim                # batch * hidden
    left_mat  = left_perm.view(BH, seq_len, seq_len)   # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)  # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()    # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (Triton) – stays in FP16 to avoid extra cast
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                # (M, H) fp16
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)  # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)    # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, cast back to FP32
    # --------------------------------------------------------------
    gated = out_norm * out_gate                      # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) fp16
    # Final linear projection (no bias in original implementation)
    out_fp16 = F.linear(gated, to_out_w)           # (B,N,N,D) fp16
    out_fp32 = out_fp16.to(torch.float32)          # (B,N,N,D) fp32

    return out_fp32
