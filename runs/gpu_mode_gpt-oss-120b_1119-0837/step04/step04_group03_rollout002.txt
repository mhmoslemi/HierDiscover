def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, L, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input LayerNorm (rowwise)
    x_flat = input_tensor.reshape(-1, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device).contiguous()
    norm_bias = weights["norm.bias"].to(device).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # Cast to FP16 for heavy compute
    x_h = x_norm.to(torch.float16)

    # Merge left_proj and left_gate weights
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # shape (2*hidden, dim)

    # Merge right_proj and right_gate similarly
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Compute left and right+gate in one linear each
    left_and_gate = F.linear(x_h, left_concat_w)  # shape [B, L, L, 2*hidden]
    right_and_gate = F.linear(x_h, right_concat_w)

    # Split projections and gates
    left_proj = left_and_gate[..., :hidden]
    left_gate = left_and_gate[..., hidden:]
    right_proj = right_and_gate[..., :hidden]
    right_gate = right_and_gate[..., hidden:]

    # Optional mask (broadcast over hidden)
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # [B, L, L, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out_gate
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Batched GEMM: reshape and permute
    # shape to (bs*hidden, L, L)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden, L, L)

    # Compute out = left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, L, L, hidden]
    out = out_flat.view(bs, hidden, L, L).permute(0, 2, 3, 1).contiguous()

    # Output layernorm (rowwise)
    out_flat2 = out.view(bs * L * L, hidden).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps=1e-5)
    out_norm = out_norm_flat.view(bs, L, L, hidden)

    # Apply output gate
    out = out_norm * out_gate

    # Final projection
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)

    # Cast back to float32
    return out.to(torch.float32)

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr, M, N, eps,
                     stride_xm, stride_xn, stride_ym, stride_yn,
                     BLOCK_M: tl.constexpr,
                     BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)

    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (row‑wise) on the pairwise tensor via a custom Triton kernel.
2️⃣ Cast to FP16 for the heavy work.
3️⃣ Fuse left/right projections with their respective gates:
   * Concatenate the two weight matrices (proj & gate) → one linear call.
   * Split the output, apply mask (if any) to the projection and multiply by the
     sigmoid‑gated values.
4️⃣ Compute the N³ contraction through a batched GEMM:
   out_flat[b*h, i, j] = Σₖ left_flat[b*h, i, k] * right_flat[b*h, j, k].
   (Implemented with torch.bmm on FP16 data.)
5️⃣ Output LayerNorm (row‑wise) **+** output gate (sigmoid) via a Triton kernel.
6️⃣ Final linear projection back to the model dimension and cast to float32.

All heavy linear algebra runs in FP16 on Tensor Cores; LayerNorms use
FP32 accumulators for numerical stability.  The kernel works for any batch
size ≤ 2, sequence length up to 1024, model dim ∈ {128,384} and hidden dim =128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (2‑D input)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,           # data pointers
    M, N, eps,                            # dimensions
    stride_xm, stride_xn, stride_ym, stride_yn,  # strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Offsets for the rows processed by this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: row‑wise mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, affine transform and store
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    `x` must be 2‑D (M, N). Returns a tensor of the same shape & dtype as `x`.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tuned block sizes – we process many rows per block (1024) and
    # up to 128 columns (covers hidden ≤128 & dim ≤384).
    BLOCK_M = 1024
    BLOCK_N = 128

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul forward kernel (no grads)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, L, L, D] (float32)
        - mask         : torch.Tensor [B, L, L] (float32) or None
        - weights      : dict of torch tensors with the required keys
        - config       : dict with keys ``dim`` (D), ``hidden_dim`` (H) and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, L, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise, FP32)
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(-1, dim).contiguous()
    norm_w = weights["norm.weight"].to(device).contiguous()
    norm_b = weights["norm.bias"].to(device).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers
    # --------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, D)

    # right side
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, D)

    # single linear for each side (produces 2*H channels)
    left_all = F.linear(x_h, left_concat_w)    # [B, L, L, 2H]
    right_all = F.linear(x_h, right_concat_w)  # [B, L, L, 2H]

    # split back into projections and gates
    left_proj = left_all[..., :hidden]          # [B, L, L, H]
    left_gate = left_all[..., hidden:]          # [B, L, L, H]
    right_proj = right_all[..., :hidden]
    right_gate = right_all[..., hidden:]

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, L, L, H]

    # --------------------------------------------------------------
    # 7️⃣ Batched GEMM for the N³ contraction
    # --------------------------------------------------------------
    # reshape to (B*H, L, L) – each (b, h) pair is an independent matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden, L, L)

    # out_flat[b*h, i, j] = Σₖ left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, L, L, H]
    out = out_flat.view(bs, hidden, L, L).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Output LayerNorm + output‑gate (Triton + FP16)
    # --------------------------------------------------------------
    out_flat2 = out.view(bs * L * L, hidden).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden)

    # Apply the output gate (both in FP16)
    out = out_norm * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final projection back to model dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)  # [B, L, L, D] (FP16)

    # --------------------------------------------------------------
    # 10️⃣ Cast to float32 for the API contract
    # --------------------------------------------------------------
    return out.to(torch.float32)
