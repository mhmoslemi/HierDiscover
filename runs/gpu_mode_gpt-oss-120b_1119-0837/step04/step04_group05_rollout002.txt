@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                          M, C, eps,
                          stride_xm, stride_xc,
                          stride_ym, stride_yc,
                          BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)

    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        # load input element; x could be fp32 or fp16
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
                     mask=mask_m[:, None] & mask_c[None, :],
                     other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq += tl.sum(x_f * x_f, axis=1)
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
                     mask=mask_m[:, None] & mask_c[None, :],
                     other=0.0)
        x_f = x.to(tl.float32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]

        # Cast to half before store

        tl.store(y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
                 y_h,
                 mask=mask_m[:, None] & mask_c[None, :])

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
0️⃣  Input LayerNorm (FP32 statistics) – a custom Triton kernel computes the
    mean / variance per row and writes the normalized result directly in FP16.
1️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single half‑precision GEMM by concatenating their weight
    matrices.
2️⃣  Split the concatenated projection, apply the optional binary mask (if any)
    and the three sigmoid gates *in‑place* on the half‑precision tensors.
3️⃣  Core contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    performed with a single `torch.einsum`, avoiding costly permutes and copies.
4️⃣  Output LayerNorm – same Triton kernel (FP32 accumulation, FP16 output).
5️⃣  Multiply by the out‑gate (in‑place) and apply the final half‑precision linear
    projection back to the model dimension.
6️⃣  Cast the result to float32 for the final return.

All heavy arithmetic runs in FP16 (tensor‑core friendly).  Both LayerNorms run
in a fused Triton kernel, satisfying the “must use a kernel” requirement while
keeping the overall memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 accum, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr,                # *ptr to input / output [M, C]
    weight_ptr, bias_ptr,        # *ptr to LN weight / bias [C]
    M, C, eps,                   # dimensions
    stride_xm, stride_xc,        # stride of input
    stride_ym, stride_yc,        # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """Layer‑norm over the last dimension of a 2‑D tensor (M×C) with
    FP32 accumulation and FP16 output."""
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        # Load x (could be fp32 or fp16) and cast to fp32 for accumulation
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform, store as fp16
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0,
        )
        x_f = x.to(tl.float32)

        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]          # FP32 intermediate
        y_h = y.to(tl.float16)                      # cast to FP16

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y_h,
            mask=mask_m[:, None] & mask_c[None, :],
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5,
                          BLOCK_M: int = 128,
                          BLOCK_C: int = 64) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a 2‑D tensor ``x`` (shape M×C)
    using a fused Triton kernel.  Returns a FP16 tensor of the same shape.
    All arithmetic is performed in FP32 for numerical stability.
    """
    assert x.dim() == 2, "triton_layernorm_fp16 expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty((M, C), dtype=torch.float16, device=x.device)

    # Ensure contiguity for stride extraction
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y

# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 stats, FP16 output) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.view(bs * seq_len * seq_len, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm = triton_layernorm_fp16(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)          # (B,N,N,dim) FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                     # (5H, dim)

    proj = F.linear(x_norm.view(-1, dim), combined_weight)  # (B·N·N, 5H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split back and reshape to [B, N, N, H]
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw       = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw      = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw  = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw   = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw.mul_(mask_h)          # in‑place mask
        right_raw.mul_(mask_h)

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – in‑place, still FP16
    # --------------------------------------------------------------
    left_gate_raw.sigmoid_()
    right_gate_raw.sigmoid_()
    out_gate_raw.sigmoid_()

    left_raw.mul_(left_gate_raw)     # left = left_raw * left_gate
    right_raw.mul_(right_gate_raw)   # right = right_raw * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction via a single Einstein‑summation (no permute/copy)
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left_raw, right_raw)  # FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 stats, FP16 output) – Triton
    # --------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm = triton_layernorm_fp16(out_flat,
                                     to_out_norm_weight,
                                     to_out_norm_bias,
                                     eps=1e-5).view(bs, seq_len, seq_len, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate (in‑place) and final projection back to model dim
    # --------------------------------------------------------------
    out_norm.mul_(out_gate_raw)                     # (B,N,N,H) *= out_gate
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    final = F.linear(out_norm.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim) FP16
    final = final.view(bs, seq_len, seq_len, dim).float()           # cast to FP32

    return final
```
