@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_b = tl.program_id(2)

   # compute offset indices
   offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   num_k = tl.cdiv(N, BLOCK_K)
   for k in range(num_k):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # left: (BLOCK_M, BLOCK_K)
       left = tl.load(
           left_ptr + pid_b * stride_l_batch
                    + offs_i[:, None] * stride_l_row
                    + offs_k[None, :] * stride_l_col,
           mask=mask_i[:, None] & mask_k[None, :],
           other=0.0
       )
       left_f32 = tl.cast(left, tl.float32)

       # right: (BLOCK_N, BLOCK_K) where rows are j (offs_j)
       right = tl.load(
           right_ptr + pid_b * stride_r_batch
                     + offs_j[:, None] * stride_r_row
                     + offs_k[None, :] * stride_r_col,
           mask=mask_j[:, None] & mask_k[None, :],
           other=0.0
       )
       right_f32 = tl.cast(right, tl.float32)

       acc += tl.dot(left_f32, right_f32)

   # Convert back to fp16
   out = tl.cast(acc, tl.float16)

   # Store
   tl.store(
       out_ptr + pid_b * stride_o_batch
                + offs_i[:, None] * stride_o_row
                + offs_j[None, :] * stride_o_col,
       out,
       mask=mask_i[:, None] & mask_j[None, :]
   )

    # left, right shape (BH,N,N) fp16
    BH,N,_ = left.shape
    out = torch.empty_like(left)
    stride_l_batch, stride_l_row, stride_l_col = left.stride()

        stride_l_col=stride_l_col,

assistantfinal**Solution Overview**  
The *outgoing* `TriMul` operator can be broken down into three heavy‑weight stages:

1. **LayerNorm on the model dimension (`dim`)** – done by a custom two‑pass Triton kernel
   (mean / variance in the first pass, normalisation in the second).  
2. **Projection + gating** – all five linear maps (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`, `out_gate`) are stacked into a single matrix‑multiply,
   saving memory traffic.  The gates are turned into sigmoids and a binary mask
   (when present) is applied only to the *projection* tensors.  
3. **Core cubic contraction**  

   \[
   {\rm out}_{b,i,j,h}= \sum_{k=0}^{N-1}
        {\rm left}_{b,i,k,h}\;{\rm right}_{b,j,k,h}
   \]

   This is a batched GEMM with shape  
   `(B·H, N, N) × (B·H, N, N)ᵀ`.  
   We implement it with a dedicated Triton kernel (`tri_mul_gemm_kernel`).  
   The kernel loops over the summation axis `k` in blocks, loads a
   `(BLOCK_M×BLOCK_K)` tile of *left* and a `(BLOCK_N×BLOCK_K)` tile of *right*,
   performs a `tl.dot` (FP16 → FP32 accumulation) and writes the result back as
   FP16.  The kernel is launched with a three‑dimensional grid:
   `grid = (⌈N/BLOCK_M⌉, ⌈N/BLOCK_N⌉, B·H)`.

4. **Hidden‑dim LayerNorm → output gate → final projection** – the hidden
   LayerNorm is another custom Triton kernel (FP16), the output gate is multiplied
   element‑wise, and the final projection back to `dim` is a standard `F.linear`
   in FP32.

All heavy linear work stays in FP16 (Tensor‑Core friendly), while the only
operations kept in FP32 are the LayerNorm statistics, the gating sigmoids and the
final projection – guaranteeing numerical parity with the reference implementation.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# -------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------- row indices -------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------- first pass: mean & var -------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------- second pass: write normalized -------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton – input shape (M, N)."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# -------------------------------------------------------------------------
# 2‑pass LayerNorm – float16 → float16
# -------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) via Triton – input shape (M, N)."""
    assert x.dtype == torch.float16
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# -------------------------------------------------------------------------
# Batched GEMM for the cubic contraction – FP16->FP32 accumulation
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)   # row block
    pid_j = tl.program_id(1)   # col block
    pid_b = tl.program_id(2)   # batch‑head (BH)

    # ------------------- indices & masks -------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block : (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + pid_b * stride_l_batch
                     + offs_i[:, None] * stride_l_row
                     + offs_k[None, :] * stride_l_col,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        left_f32 = tl.cast(left, tl.float32)

        # right block (treated as rows = j, cols = k) : (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_ptr + pid_b * stride_r_batch
                      + offs_j[:, None] * stride_r_row
                      + offs_k[None, :] * stride_r_col,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        right_f32 = tl.cast(right, tl.float32)

        acc += tl.dot(left_f32, right_f32)   # (BLOCK_M, BLOCK_N)

    # write back as FP16
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_ptr + pid_b * stride_o_batch
                 + offs_i[:, None] * stride_o_row
                 + offs_j[None, :] * stride_o_col,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def tri_mul_gemm_fp16(left: torch.Tensor,
                      right: torch.Tensor,
                      BLOCK_M=64, BLOCK_N=64, BLOCK_K=32) -> torch.Tensor:
    """Batched GEMM implementing the TriMul N³ contraction.
    Arguments:
        left, right – tensors of shape (BH, N, N) and dtype torch.float16.
    Returns:
        out – tensor of the same shape and dtype."""
    assert left.shape == right.shape
    BH, N, _ = left.shape
    out = torch.empty_like(left)

    stride_l_batch, stride_l_row, stride_l_col = left.stride()
    stride_r_batch, stride_r_row, stride_r_col = right.stride()
    stride_o_batch, stride_o_row, stride_o_col = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH                         # batch‑head
    )
    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        stride_l_batch=stride_l_batch,
        stride_l_row=stride_l_row,
        stride_l_col=stride_l_col,
        stride_r_batch=stride_r_batch,
        stride_r_row=stride_r_row,
        stride_r_col=stride_r_col,
        stride_o_batch=stride_o_batch,
        stride_o_row=stride_o_row,
        stride_o_col=stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# -------------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[ B, N, N, D ] (float32)
        - mask         : torch.Tensor[ B, N, N ] (float32 / bool) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (optional ``nomask``)

    Returns
    -------
    torch.Tensor[ B, N, N, D ] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32
    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -----------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)               # (B·N·N, D)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)            # (B, N, N, D)

    # -----------------------------------------------------------------
    # 2️⃣ Single fused linear -> projections + gates (still in float32)
    # -----------------------------------------------------------------
    # concatenate the five weight matrices: [left_proj, left_gate,
    # right_proj, right_gate, out_gate] → shape (5·H, D)
    combined_w = torch.cat([
        weights["left_proj.weight"],   # (H, D)
        weights["left_gate.weight"],   # (H, D)
        weights["right_proj.weight"],  # (H, D)
        weights["right_gate.weight"],  # (H, D)
        weights["out_gate.weight"],    # (H, D)
    ], dim=0).to(device, torch.float32)                # (5·H, D)

    # one big matmul: (B·N·N, 5·H)
    combined = F.linear(x_norm, combined_w)            # (B,N,N,5·H)

    # split the fused result
    h0 = 0
    left_proj    = combined[..., h0: h0 + H];               h0 += H
    left_gate_pre = combined[..., h0: h0 + H];              h0 += H
    right_proj   = combined[..., h0: h0 + H];               h0 += H
    right_gate_pre = combined[..., h0: h0 + H];             h0 += H
    out_gate_pre   = combined[..., h0: h0 + H];             # h0 now = 4·H

    # sigmoids for the gates (still FP32)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)               # (B,N,N,H) FP32

    # -----------------------------------------------------------------
    # 3️⃣ Apply binary mask (if present) – broadcast over H
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f32 = mask.to(device, torch.float32).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_f32
        right_proj = right_proj * mask_f32

    # -----------------------------------------------------------------
    # 4️⃣ Gated projections
    # -----------------------------------------------------------------
    left  = left_proj * left_gate        # (B,N,N,H) FP32
    right = right_proj * right_gate      # (B,N,N,H) FP32

    # -----------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the cubic contraction
    # -----------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ---------------------------------------------------------------
    # 6️⃣ Core N³ contraction via the custom Triton GEMM
    # ---------------------------------------------------------------
    # reshape to (BH, N, N) where BH = B·H
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(-1, N, N)                     # (BH, N, N)
    right_mat = right_perm.view(-1, N, N)                   # (BH, N, N)

    out_mat = tri_mul_gemm_fp16(left_mat, right_mat,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)  # (BH,N,N)

    # reshape back to (B, N, N, H) – still FP16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (FP16)
    # -----------------------------------------------------------------
    out_flat = out.reshape(M, H)                     # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # FP16

    # -----------------------------------------------------------------
    # 8️⃣ Apply output gate (still FP32) and final projection back to D
    # -----------------------------------------------------------------
    # bring the normalized tensor back to FP32 for the last linear
    out_norm_fp32 = out_norm.to(torch.float32)                # (M, H)
    out_gate_flat = out_gate.reshape(M, H)                    # (M, H) FP32
    gated = out_norm_fp32 * out_gate_flat                      # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float32)  # (D, H)
    output_flat = F.linear(gated, to_out_w)                     # (M, D) FP32

    output = output_flat.view(bs, N, N, dim)                    # (B,N,N,D)
    return output
