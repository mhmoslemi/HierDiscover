"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold‑3).  The forward pass consists of:

1️⃣ Input LayerNorm (FP32) – a custom 2‑pass Triton kernel.
2️⃣ Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single FP16 GEMM.
3️⃣ Apply the pair‑wise mask (if present) and sigmoid gating.
4️⃣ Core N³ contraction:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   performed by a custom Triton batched‑gemm kernel that works directly on
   the original [B,N,N,H] layout, thus avoiding the expensive permutes and
   copies of the reference implementation.
5️⃣ Second LayerNorm (FP16) on the hidden dimension.
6️⃣ Apply the output gate and final projection back to the model dimension.

All heavy arithmetic is done in FP16 with FP32 accumulation for numerical
stability; the final result is cast back to float32 to match the PyTorch
reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (float32 → float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr,   # input [M, N]  (float32)
    y_ptr,   # output [M, N] (float32)
    w_ptr,   # weight [N]    (float32)
    b_ptr,   # bias   [N]    (float32)
    M, N,
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor (M,N)."""
    assert x.dim() == 2, "expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](

        b_ptr=bias,
        M=M, N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

    B, N, H,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs:
    bh_id   = tl.program_id(0)              # combined batch * head index (0 … B*H-1)
    block_i = tl.program_id(1)              # tile row index
    block_j = tl.program_id(2)              # tile col index

    # decode batch & head
    b = bh_id // H
    h = bh_id % H

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # tile offsets inside the N x N matrix
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (helps accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # right tile is taken transposed so we get shape (BLOCK_K, BLOCK_N)
        # we load right[k, j] directly
        right = tl.load(
            right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # accumulate in FP32
        acc += tl.dot(tl.cast(left,  tl.float32),
                      tl.cast(right, tl.float32))

    # write back in FP16
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def triton_pairwise_gemm(left: torch.Tensor,
                         right: torch.Tensor,
                         BLOCK_M: int = 64,
                         BLOCK_N: int = 64,
                         BLOCK_K: int = 64) -> torch.Tensor:
    """
    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    where left/right are [B, N, N, H] fp16 tensors.
    Returns a fp16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    assert left.shape == right.shape, "left/right must have identical shapes"

    out = torch.empty_like(left)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BH = B * H
    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    triton_pairwise_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_l_b=stride_l_b, stride_l_i=stride_l_i,
        stride_l_k=stride_l_k, stride_l_h=stride_l_h,
        stride_r_b=stride_r_b, stride_r_i=stride_r_i,
        stride_r_k=stride_r_k, stride_r_h=stride_r_h,
        stride_o_b=stride_o_b, stride_o_i=stride_o_i,
        stride_o_j=stride_o_j, stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expect torch.float32

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – flatten batch & pair dimensions
    # ------------------------------------------------------------------
    M = B * N * N
    x_flat = input_tensor.reshape(M, D).contiguous()     # (M, D)

    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)  # (M, D)
    x_norm = x_norm_flat.view(B, N, N, D)                                   # (B, N, N, D)

    # Cast to FP16 for the remaining heavy work
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    w_lproj = weights["left_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rproj = weights["right_proj.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    weight_fused = torch.cat(
        [w_lproj, w_lgate, w_rproj, w_rgate, w_outgate], dim=0
    ).to(device, torch.float16)                     # (5*H, D)

    fused = F.linear(x_fp16, weight_fused)          # (M, 5*H)
    fused = fused.view(B, N, N, 5 * H)              # (B, N, N, 5H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply sigmoid gates and (optional) mask
    # ------------------------------------------------------------------
    h0, h1, h2, h3, h4, h5 = 0, H, 2*H, 3*H, 4*H, 5*H
    left_proj   = fused[..., h0:h1]
    left_gate_p = fused[..., h1:h2]
    right_proj  = fused[..., h2:h3]
    right_gate_p = fused[..., h3:h4]
    out_gate_p   = fused[..., h4:h5]

    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    left  = left_proj * left_gate   # (B,N,N,H) fp16
    right = right_proj * right_gate # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction (batched GEMM) using Triton
    # ------------------------------------------------------------------
    out_hidden = triton_pairwise_gemm(left, right,
                                      BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_hidden_flat = out_hidden.reshape(M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_hidden_flat,
                                     to_out_norm_w,
                                     to_out_norm_b,
                                     eps=eps)               # (M, H) fp16
    # reshape back
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to D
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                    # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated.reshape(M, H), to_out_w)           # (M, D) fp16
    output = output_flat.view(B, N, N, D).to(torch.float32)         # final FP32 tensor

    return output
