@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bi, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # offsets for rows and columns within tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Determine batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this batch/head
    a_base = a_ptr + b * stride_am + h * stride_ah
    b_base = b_ptr + b * stride_bm + h * stride_bh
    c_base = c_ptr + b * stride_cm + h * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a = tl.load(
            a_base
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (K x N): need rows = k, cols = j
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # Cast to fp32 for accumulation
        a_f = tl.cast(a, tl.float32)
        b_f = tl.cast(b, tl.float32)

        # Multiply accumulate
        acc += tl.dot(a_f, b_f)  # (M,N)
    # End K loop

    # Write result as float16
    c = tl.cast(acc, tl.float16)

    tl.store(
        c_base
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj,
        c,
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated.

Key ideas
---------
1️⃣ Input LayerNorm (float32) → custom Triton kernel.
2️⃣ Cast to FP16 for the heavy linear work.
3️⃣ Fuse the four linear maps (left/right projections + gates) into a single
   matmul:   (left_proj , left_gate , right_proj , right_gate)  → (4·H).
4️⃣ Apply the optional mask *before* the gates.
5️⃣ Compute the N³ “triangle‑multiplicative” contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   with a bespoke batched GEMM kernel that reads the 4‑D tensors directly –
   no costly permute / view copies.
6️⃣ Hidden‑dim LayerNorm (FP16) → custom Triton kernel.
7️⃣ Apply the output gate (sigmoid of a linear on the normalized *input*).
8️⃣ Final linear projection back to the model dimension and cast to float32.

The only Python‑level operations are the cheap element‑wise steps;
the two LayerNorms and the O(N³) contraction are executed on‑GPU
via Triton for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------
# 2‑pass LayerNorm – FP32 → FP32 (Triton)
# --------------------------------------------------------------------------
@triton.jit
def _layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- pass 1 : mean & variance -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- pass 2 : write normalized -----
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """2‑pass LayerNorm (float32) on a 2‑D tensor (M, N)."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    _layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# 2‑pass LayerNorm – FP16 → FP16 (Triton)
# --------------------------------------------------------------------------
@triton.jit
def _layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """2‑pass LayerNorm (float16) on a 2‑D tensor (M, N)."""
    assert x.dtype == torch.float16
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    _layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Batched GEMM for the N³ contraction (Triton)
# --------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # left  tensor  (B,N,N,H)  half
    b_ptr,                # right tensor  (B,N,N,H)  half
    c_ptr,                # output tensor (B,N,N,H)  half
    B, N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bi, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids : (tile_i, tile_j, batch_head)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # ----- tile coordinates -----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- decode batch and hidden indices -----
    b = pid_bh // H          # batch index
    h = pid_bh %  H          # hidden‑dim (head) index

    # base pointers for the selected (b,h)
    a_base = a_ptr + b * stride_am + h * stride_ah
    b_base = b_ptr + b * stride_bm + h * stride_bh
    c_base = c_ptr + b * stride_cm + h * stride_ch

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A tile: shape (BLOCK_M, BLOCK_K) ----
        a = tl.load(
            a_base
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # ---- load B tile: we need rows = k, cols = j  → (BLOCK_K, BLOCK_N) ----
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )

        # cast to FP32, then perform a dot‑product for the tile
        acc += tl.dot(tl.cast(a, tl.float32), tl.cast(b, tl.float32))

    # write result back as FP16
    c = tl.cast(acc, tl.float16)
    tl.store(
        c_base
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj,
        c,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiply for the core TriMul contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : torch.Tensor[bs, N, N, H] (half) – must be contiguous.
    Returns
    -------
    out : torch.Tensor[bs, N, N, H] (half)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # ensure contiguous layout (required for stride calculations)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    stride_left = left.stride()   # (b,i,k,h)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * H,                        # batch‑head dimension
    )
    _batched_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_am=stride_left[0],
        stride_ai=stride_left[1],
        stride_ak=stride_left[2],
        stride_ah=stride_left[3],
        stride_bm=stride_right[0],
        stride_bi=stride_right[1],
        stride_bk=stride_right[2],
        stride_bh=stride_right[3],
        stride_cm=stride_out[0],
        stride_ci=stride_out[1],
        stride_cj=stride_out[2],
        stride_ch=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------
# TriMul forward entry point
# --------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 / bool) or None
        - weights      : dict of model parameters
        - config       : dict containing ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expect torch.float32

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                    # (B·N·N, D)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                 # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right proj + gate into a single matmul
    # ------------------------------------------------------------------
    #   [left_proj; left_gate; right_proj; right_gate]  →  (4·H, D)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)           # shape (4·H, D)

    # linear → (B,N,N,4·H)
    combined = F.linear(x_half, combined_weight)   # half

    # split
    H = hidden_dim
    left_proj      = combined[..., :H]               # (B,N,N,H)
    left_gate_pre  = combined[..., H:2 * H]
    right_proj     = combined[..., 2 * H:3 * H]
    right_gate_pre = combined[..., 3 * H:4 * H]

    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask before gating
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate      # (B,N,N,H)
    right = right_proj * right_gate    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction – custom Triton GEMM
    # ------------------------------------------------------------------
    out = batched_gemm_triton(left, right)   # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (float16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)               # (B·N·N, H)
    out_norm = triton_layernorm_fp16(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                          # half

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, weights["out_gate.weight"].to(device, torch.float16)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    # flatten gated tensors for the final linear
    out_norm_flat = out_norm.view(M, H)               # (M, H) half
    out_gate_flat = out_gate.view(M, H)               # (M, H) half
    gated = out_norm_flat * out_gate_flat              # (M, H) half

    out_proj = F.linear(
        gated,
        weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    )                                                  # (M, D) half

    output = out_proj.view(bs, N, N, dim).float()    # back to FP32

    return output
