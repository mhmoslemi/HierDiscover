    # 3. Combined projection+gate for left and right
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device, torch.float16)

"""
TriMul “outgoing” forward pass – Triton‑accelerated LayerNorm and
cuBLAS‑backed matrix multiplications.

Highlights
----------
* Input LayerNorm (float‑32) – custom 2‑pass Triton kernel.
* All heavy linear work (projections, gates, output gate, final projection)
  stays in FP16 and is executed by cuBLAS (torch.nn.functional.linear / torch.bmm).
* Core N³ contraction: for each hidden channel a separate (N×N) × (N×N)
  matrix multiplication is performed with `torch.bmm`, which on H100
  uses Tensor‑Cores and is far faster than a hand‑written kernel.
* Hidden‑dim LayerNorm (float‑16) – custom Triton kernel, followed by
  output‑gate element‑wise multiplication and final FP16→FP32 projection.
* Optional binary mask is applied before gating.

The implementation respects the required signature:
    custom_kernel((input, mask, weights, config)) → Tensor[B,N,N,D] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# LayerNorm for float‑32 (2‑pass) – Triton
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr,  # input  [M, D]
    y_ptr,  # output [M, D]
    w_ptr,  # weight [D]
    b_ptr,  # bias   [D]
    M, D, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- 1️⃣ mean / variance -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- 2️⃣ normalize + scale -----
    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over last dimension (float32) implemented via a Triton kernel.
    Expects x of shape [M, D] (2‑D) and weight/bias of length D.
    """
    assert x.dim() == 2
    M, D = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)
    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_D=64,
        num_warps=4,
    )
    return y


# --------------------------------------------------------------------------
# LayerNorm for float‑16 (2‑pass) – Triton
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,  # input  [M, D] (fp16)
    y_ptr,  # output [M, D] (fp16)
    w_ptr,  # weight [D] (fp16)
    b_ptr,  # bias   [D] (fp16)
    M, D, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over last dimension (float16) via Triton.
    Input shape must be [M, D] with dtype torch.float16.
    """
    assert x.dtype == torch.float16
    assert x.dim() == 2
    M, D = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()

    return y


# --------------------------------------------------------------------------
# TriMul “outgoing” forward pass entry point
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameter tensors
        - config       : dict with keys
            * dim          – model dimension D
            * hidden_dim   – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    torch._C._set_grad_enabled(False)  # no grad required

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype        # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(-1, dim)                 # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)            # (B, N, N, D) – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_fp16 = x_norm.half()      # (B, N, N, D) – FP16

    # ------------------------------------------------------------------
    # 3️⃣ Projection + gate (fused linear per side)
    # ------------------------------------------------------------------
    #   left_comb = [left_proj ; left_gate] , shape (2·H, D)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)      # (2H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)      # (2H, D)

    # combined matmuls (produce proj + gate in one go)
    left_comb = F.linear(x_fp16, left_comb_w)    # (B,N,N,2H)
    right_comb = F.linear(x_fp16, right_comb_w)  # (B,N,N,2H)

    # split back
    left_proj = left_comb[..., :H]                # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]            # (B,N,N,H)
    right_proj = right_comb[..., :H]              # (B,N,N,H)
    right_gate_pre = right_comb[..., H:]          # (B,N,N,H)

    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply gating (element‑wise)
    # ------------------------------------------------------------------
    left = left_proj * left_gate                   # (B,N,N,H)  fp16
    right = right_proj * right_gate                # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of the mask)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)   # (H, D)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))               # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction – use cuBLAS batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    #   reshape to (B·H, N, N)  – each hidden channel is an independent GEMM
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    batch = bs * H
    left_mat  = left_t.view(batch, N, N)              # (B·H, N, N)
    right_mat = right_t.view(batch, N, N)             # (B·H, N, N)

    # out_mat[b] = left_mat[b] @ right_mat[b].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, H)                         # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (FP16 → FP32)
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(-1, H)               # (B·N·N, H)
    gated = out_norm * out_gate_flat                    # (B·N·N, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                # (B·N·N, D)  fp16

    output = out_fp16.view(bs, N, N, dim).float()      # back to FP32
    return output
