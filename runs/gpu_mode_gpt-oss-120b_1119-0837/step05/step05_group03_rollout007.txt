@triton.jit
def batched_gemm_hidden_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,  # M = N (row dimension i), N = N (col dimension j), K = N (summation dimension)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    hidden_dim, # number of hidden dims (H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # tile row
    pj = tl.program_id(1) # tile col
    pb = tl.program_id(2) # batch*hidden idx
    # derive batch index and hidden index
    batch_id = pb // hidden_dim
    hidden_id = pb % hidden_dim

    # compute start offsets for this slice
    a_base = batch_id * stride_a_batch + hidden_id * stride_a_col  # because hidden col stride = 1 (depth)
    b_base = batch_id * stride_b_batch + hidden_id * stride_b_col
    c_base = batch_id * stride_c_batch + hidden_id * stride_c_col

    # tile offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pj * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None,:] * stride_a_col
    b_ptrs = b_ptr + b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_row + offs_n[None,:] * stride_b_col

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None,:], other=0.0)
        # load B tile
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None,:], other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row
    # store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_c_row + offs_n[None,:] * stride_c_col
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None,:])

"""
TriMul “outgoing” forward pass – Triton‐accelerated.

The implementation follows the reference module closely but
*   uses a Triton kernel for the two LayerNorms,
*   runs the heavy N³ contraction with a custom batched GEMM that
    reads directly from the original (B,N,N,H) layout, thus **avoiding the
    expensive permute‑and‑contiguous copies** required by a naïve approach,
*   keeps all large matrix work in FP16 (Tensor‑Core friendly) and only
    casts back to FP32 at the end.

The kernel `batched_gemm_hidden_fp16_kernel` treats each (batch, hidden)
slice as an independent GEMM, i.e. computes
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
for every hidden dimension `h`.  Strides are passed explicitly so the
kernel can work on the native layout (row‑major with the hidden dimension
as the innermost axis) without any data reshaping.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# 1) LayerNorm kernels (FP32 and FP16) ------------------------------------
# -------------------------------------------------------------------------

@triton.jit
def _layernorm_fp32_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- pass 1: mean / variance ---------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_N)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- pass 2: normalize, scale and bias -------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float32) via Triton."""
    M, N = x.shape
    y = torch.empty_like(x)
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    _layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


@triton.jit
def _layernorm_fp16_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float16) via Triton."""
    M, N = x.shape
    y = torch.empty_like(x)
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    _layernorm_fp16_kernel[grid](

        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# 2) Batched GEMM that works directly on (B,N,N,H) layout ---------------
# -------------------------------------------------------------------------

@triton.jit
def batched_gemm_hidden_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for every hidden slice without permuting.

    A : (B, N, N, H)   – interpreted as batch of H matrices of shape (N,N)
                         with row stride = stride_a_row,
                         col stride = stride_a_col.
    B : (B, N, N, H)   – treated as matrices (K,N) = (N,N) transposed
                         (row stride = stride_b_row, col stride = stride_b_col).
    Result C has the same layout as A.
    """
    pid_m = tl.program_id(0)      # tile over rows (i)
    pid_n = tl.program_id(1)      # tile over cols (j)
    pid_hb = tl.program_id(2)     # combined batch*hidden index

    # -----------------------------------------------------------------
    # decode batch id and hidden id from combined index
    # -----------------------------------------------------------------
    hb = hidden_dim
    batch_id = pid_hb // hb
    hidden_id = pid_hb - batch_id * hb  # same as pid_hb % hb but cheaper

    # base offsets for this (batch, hidden) slice
    a_base = batch_id * stride_a_batch + hidden_id
    b_base = batch_id * stride_b_batch + hidden_id
    c_base = batch_id * stride_c_batch + hidden_id

    # tile offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current tile
    a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    b_ptrs = b_ptr + b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)               # FP16·FP16 → FP32

        # advance to next k‑tile
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # write results
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None])


def triton_batched_gemm_hidden_fp16(left: torch.Tensor,
                                   right: torch.Tensor,
                                   hidden_dim: int,
                                   BLOCK_M: int = 128,
                                   BLOCK_N: int = 128,
                                   BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul contraction.
    left/right : (B, N, N, H) fp16
    returns out : (B, N, N, H) fp16
    """
    assert left.dim() == 4 and right.dim() == 4
    B, N, _, H = left.shape
    assert H == hidden_dim and right.shape == (B, N, N, H)

    out = torch.empty_like(left)

    # strides are in element units (not bytes)
    stride_a_batch, stride_a_row, stride_a_col, stride_a_h = left.stride()
    stride_b_batch, stride_b_row, stride_b_col, stride_b_h = right.stride()
    stride_c_batch, stride_c_row, stride_c_col, stride_c_h = out.stride()

    # sanity checks (they should hold for a contiguous last‑dim tensor)
    assert stride_a_h == 1 and stride_b_h == 1 and stride_c_h == 1

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * hidden_dim)  # one kernel instance per (batch, hidden)

    batched_gemm_hidden_fp16_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of model parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool) indicates that no mask
                         should be applied even if a mask tensor is supplied.

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * N * N, dim)                 # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                       # (B,N,N,D) FP32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # -----------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Projections + gating (two linear layers fused per side)
    # -----------------------------------------------------------------
    left_comb_w = torch.cat([weights["left_proj.weight"],
                             weights["left_gate.weight"]], dim=0).to(device, torch.float16)  # (2H,D)
    right_comb_w = torch.cat([weights["right_proj.weight"],
                              weights["right_gate.weight"]], dim=0).to(device, torch.float16)  # (2H,D)

    # fused projection+gate
    left_comb  = F.linear(x_f16, left_comb_w)       # (B,N,N,2H)
    right_comb = F.linear(x_f16, right_comb_w)      # (B,N,N,2H)

    # split
    left_proj = left_comb[..., :H]                   # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]               # (B,N,N,H)
    right_proj = right_comb[..., :H]                 # (B,N,N,H)
    right_gate_pre = right_comb[..., H:]             # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4️⃣ Mask (optional) – broadcasted over the hidden dim
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # -----------------------------------------------------------------
    # 5️⃣ Gating
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)        # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)       # (B,N,N,H)

    left  = left_proj * left_gate                    # (B,N,N,H)
    right = right_proj * right_gate                  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # -----------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device, torch.float16)    # (H,D)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_weight))               # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Core N³ contraction – custom batched GEMM (no permute!)
    # -----------------------------------------------------------------
    out_hidden = triton_batched_gemm_hidden_fp16(left, right, hidden_dim=H,
                                                 BLOCK_M=128, BLOCK_N=128, BLOCK_K=64)
    # out_hidden : (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16)
    # -----------------------------------------------------------------
    M = bs * N * N
    out_flat = out_hidden.reshape(M, H)                # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # -----------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # -----------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)             # (M, H) fp16
    gated = out_norm * out_gate_flat                    # (M, H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output_flat = F.linear(gated, to_out_w)            # (M, D) fp16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32
    return output
