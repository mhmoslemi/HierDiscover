"""
Highly‑optimized forward pass for the “outgoing’’ TriMul operator.

Algorithm
---------
1️⃣ Triton‑based LayerNorm over the last feature dimension of the input
   tensor (shape [B·N·N, D]) – this is the first LN in the paper.
2️⃣ Cast the normalised tensor to FP16 (half) for the compute‑heavy part.
3️⃣ Fuse the five linear layers (left/right projections and three gates) into a
   single half‑precision GEMM, then slice the result.
4️⃣ Apply sigmoid to the three gate tensors.  If a mask is provided, multiply
   it (broadcast) into the left/right tensors in‑place to avoid extra buffers.
5️⃣ Contract the two 4‑D tensors with an Einstein‑summation
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.  This replaces the
   expensive `permute+contiguous+view+torch.bmm` sequence and eliminates two
   large copies (≈256 MiB each for N=1024, H=128).
6️⃣ Second Triton‑based LayerNorm on the hidden dimension (shape [B·N·N, H]).
7️⃣ In‑place multiply by the output‑gate, then a final half‑precision linear
   projection back to the model dimension.  The result is cast to float32.
All heavy kernels (the two LayerNorms) run on the GPU via Triton; the bulk of
the compute (the fused projection, elementwise gates, the contraction and the
final projection) uses PyTorch’s highly‑optimised cuBLAS/Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # number of rows (M = B·N·N or B·N·N·H)
    N,              # feature dimension (D or H)
    eps,            # epsilon
    stride_xm, stride_xn,   # strides of x
    stride_ym, stride_yn,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----- row block ----------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: compute sum and sum‑sq (FP32) --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise, apply affine, store -------------------
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalisation (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine transform (w & b may be FP16, cast to FP32 for calc)
        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for `layernorm_kernel`.  Expects a 2‑D contiguous tensor (M, N)
    and returns a tensor of the same shape and dtype as ``x``.
    """
    assert x.ndim == 2, "LayerNorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # block sizes tuned for D∈{128,384,768} and H=128
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` plus optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep FP32 for numerical stability.
    # --------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)                     # (M, D) FP32
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)               # (B,N,N,D) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute.
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.half().contiguous()
    x_norm_f16_flat = x_norm_f16.view(M, dim)              # (M, D) FP16

    # --------------------------------------------------------------
    # 3️⃣ Fuse five linear layers (proj + gates) into a single GEMM.
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                               # (5·H, D) FP16

    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)    # (M, 5·H) FP16
    proj_gate = proj_gate_flat.view(bs, N, N, 5 * hidden_dim)  # (B,N,N,5·H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]               # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates and optional mask – in‑place to save memory.
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_r)          # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_r)         # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_r)           # (B,N,N,H) FP16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj
        left.mul_(left_gate)
        left.mul_(mask_f16)

        right = right_proj
        right.mul_(right_gate)
        right.mul_(mask_f16)
    else:
        left = left_proj
        left.mul_(left_gate)        # in‑place

        right = right_proj
        right.mul_(right_gate)      # in‑place

    # --------------------------------------------------------------
    # 5️⃣ Contract over the sequence dimension k.
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Using einsum removes the expensive permute+contiguous+view
    out = torch.einsum('bikd,bjkd->bijd', left, right)  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – still in FP16.
    # --------------------------------------------------------------
    out_flat = out.view(-1, H)                         # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, H)         # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate (in‑place) and final projection.
    # --------------------------------------------------------------
    out_norm.mul_(out_gate)                            # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    out_fp16 = F.linear(out_norm.view(-1, H), to_out_w)              # (M, D) FP16
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)       # (B,N,N,D) FP32

    return out_fp32
