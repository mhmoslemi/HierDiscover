"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – highly optimized.

Key design
-----------
1️⃣  Input LayerNorm (float32) – custom Triton 2‑pass kernel.
2️⃣  Cast to FP16 for all heavy work (Tensor‑Core friendly).
3️⃣  Fuse each projection + its gate into a single Linear (2·H channels).
4️⃣  Optional binary mask (broadcast over hidden dim) applied before gating.
5️⃣  Apply sigmoid gates to the projection streams.
6️⃣  Compute the N³ contraction (∑ₖ left[i,k,h]·right[j,k,h]) with
    a single batched `torch.bmm` on FP16 tensors (much faster than a
    hand‑written Triton GEMM for the typical hidden‑dim batch sizes).
7️⃣  Hidden‑dim LayerNorm (float16) – custom Triton kernel.
8️⃣  Apply the output gate (sigmoid of a linear on the normalized *input*).
9️⃣  Final linear projection back to the model dimension (FP16 matmul) and
    cast to float32.

All heavy linear work stays in FP16 and both LayerNorms are executed with
Triton kernels, satisfying the “must use a kernel” requirement while
leveraging cuBLAS/Tensor‑Core for the N³ contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# Float‑32 LayerNorm – 2‑pass (mean/var + normalize+scale) via Triton
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean and variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_N)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized result ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float32) using a Triton kernel."""
    assert x.dim() == 2, "Flattened input expected"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous tensors
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,

# Custom kernel entry point: TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of model parameter tensors
        - config       : dict with keys:
            * dim          – model dimension D
            * hidden_dim   – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32) – flatten to (M, dim)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)            # (B, N, N, D) – still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy work
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)                    # (B, N, N, D)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection+gate (2·H channels each) – fp16 matmul
    # ------------------------------------------------------------------
    # left: proj + gate
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                         # (2H, D)
    left_comb = F.linear(x_fp16, left_comb_w)           # (B, N, N, 2H)

    # right: proj + gate
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                         # (2H, D)
    right_comb = F.linear(x_fp16, right_comb_w)         # (B, N, N, 2H)

    # split
    left_proj = left_comb[..., :H]                       # (B, N, N, H)
    left_gate_pre = left_comb[..., H:]                   # (B, N, N, H)
    right_proj = right_comb[..., :H]                     # (B, N, N, H)
    right_gate_pre = right_comb[..., H:]                 # (B, N, N, H)

    # gates (sigmoid)
    left_gate = torch.sigmoid(left_gate_pre)             # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_pre)           # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (uses the *input* after LN)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a single batched torch.bmm (fp16)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (k,j)

    batch = bs * H
    left_mat = left_perm.view(batch, N, N)              # (B·H, N, K) where K=N
    right_mat = right_perm.view(batch, N, N)            # (B·H, K, N)

    # batched matrix multiplication – Tensor‑Core accelerated
    out_mat = torch.bmm(left_mat, right_mat)            # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (fp16) – flatten to (M, H)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)                         # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H) fp16
    gated = out_norm * out_gate_flat                      # (M, H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)                # (M, D) fp16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # (B,N,N,D) fp32
    return output
