"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
================================================================

Algorithm
---------
1. Row‑wise LayerNorm on the input tensor (shape [B, L, L, D]) using a
   custom Triton kernel. The kernel works on a 2‑D view [M, D] where
   M = B·L·L.
2. Cast the normalized tensor to FP16.
3. Fuse the four linear maps (left/right projections and their gates) into
   a single half‑precision GEMM (weight shape (4·H, D)).
4. Apply an optional binary mask (if `nomask=False`) before the elementwise
   gating.
5. Compute the N³ contraction as a batched half‑precision GEMM:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is implemented as a batch of matrix multiplications with
   `torch.matmul`.
6. Second LayerNorm over the hidden dimension (H) using the same Triton
   kernel.
7. Compute the output gating from the *first* normalized tensor, apply it,
   and finally project back to the original dimension D with a half‑precision
   linear layer.
8. Cast the result back to float32 (the expected output dtype).

All heavy arithmetic is performed in FP16 (Tensor‑core friendly); reduction
steps inside LayerNorm use FP32 for numerical stability.  At least one
operation (the LayerNorm) is implemented in a Triton kernel as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                # *ptr to input / output       [M, N]
    weight_ptr, bias_ptr,        # *ptr to affine parameters    [N]
    M, N,                        # rows, features
    eps,                         # epsilon
    stride_xm, stride_xn,        # strides for input tensor
    stride_ym, stride_yn,        # strides for output tensor
    BLOCK_M: tl.constexpr,       # rows per program
    BLOCK_N: tl.constexpr,       # columns per inner loop
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for the current program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (using FP32 accumulators)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine‑transformed values
    # ------------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # (x - mean) * rstd
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm via Triton.
    Expected shape of ``x``: (M, N) where the norm is performed over the
    last dimension (N). ``weight`` and ``bias`` are 1‑D tensors of length N.
    The output has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tunable)
        BLOCK_N=64,    # columns per inner loop (tunable)
    )
    return y

    x_norm_flat = triton_layernorm(x_flat, ln_weight, ln_bias, eps=eps)  # FP16 output
    x_norm = x_norm_flat.view(bs, L, L, dim)                            # [B, L, L, D]

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate linear maps (FP16)
    # --------------------------------------------------------------
    # Load projection / gate weights in half‑precision (Tensor‑core friendly)
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    # Concatenate into one big weight matrix: (4*H, D)
    fused_weight = torch.cat([left_proj_w,
                              right_proj_w,
                              left_gate_w,
                              right_gate_w], dim=0)  # shape (4*H, D)

    # Half‑precision linear projection (no bias)
    fused_out = F.linear(x_norm.view(-1, dim), fused_weight)          # (M, 4*H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # Split fused output
    left_proj = fused_out[..., :hidden_dim]                                   # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]                    # (B, L, L, H)
    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # --------------------------------------------------------------
    # 3️⃣ Apply mask before gating (if required)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Gated projections
    # --------------------------------------------------------------
    left  = left_proj * left_gate      # (B, L, L, H)
    right = right_proj * right_gate    # (B, L, L, H)

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM over the third dimension (the N³ contraction)
    # --------------------------------------------------------------
    # Rearrange for batched matmul: (B, H, L, L)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, L, L)

    # Transpose the inner dimensions of `right` to align the summation axis
    right_t = right_perm.transpose(2, 3).contiguous()    # (B, H, L, L) -> (B, H, K, N)

    # Collapse batch & heads -> (B*H, L, L)
    left_mat  = left_perm.view(-1, L, L)   # (B*H, L, K)
    right_mat = right_t.view(-1, L, L)     # (B*H, K, N)

    # Half‑precision batched matrix multiplication (Tensor‑core accelerated)
    out_mat = torch.matmul(left_mat, right_mat)   # (B*H, L, L)

    # Restore original layout: (B, L, L, H)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)   # (B*L*L, H)
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output gating (sigmoid) – linear over the *first* normalized input
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    # x_norm is already FP16
    out_gate = torch.sigmoid(F.linear(x_norm.view(-1, dim), out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)   # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                         # (B, L, L, H)
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_fp16 = F.linear(gated.view(-1, hidden_dim), to_out_w)   # (B*L*L, D)
    out_final = out_final_fp16.view(bs, L, L, dim).to(torch.float32)

    return out_final
```
