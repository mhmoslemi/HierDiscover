"""
custom_kernel – Triton‑accelerated “outgoing” TriMul
===================================================

The kernel implements the forward pass of the TriMul “outgoing’’ operator
used in AlphaFold‑3 and related models.  It consists of:

1️⃣  Two‑pass LayerNorm (mean/var → normalise → affine) implemented in
    Triton (the input tensor is reshaped to a 2‑D matrix).  The
    normalised tensor is cast to fp16 for the remaining computation.

2️⃣  A *single* fused linear projection that simultaneously computes the
    three linear transforms (left/right) and the three gating vectors
    (left‑gate, right‑gate, out‑gate).  The concatenated weight has shape
    (5·H, D) where H = hidden_dim.

3️⃣  Element‑wise sigmoid on the gates, optional masking and the gated
    tensors `left` and `right`.

4️⃣  Batched matrix multiplication
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    for every hidden channel `h`.  This is a GEMM of size N×N×N for each
    of the B·H matrices (B≤2, H≤128).  The multiplication is performed
    with a custom Triton kernel that treats the right operand as
    transposed on‑the‑fly (no extra copy) and accumulates in FP32 while
    storing the result in FP16.

5️⃣  Second LayerNorm over the hidden dimension (still in Triton).

6️⃣  Output gate, final linear projection back to the model dimension D
    and a cast back to FP32 (the public API expects float32).

All heavy arithmetic is carried out in FP16 (except the LayerNorm
statistics and the GEMM accumulation) which gives a large speed‑up on
H100 Tensor‑Cores.  The implementation works for any batch size 1‑2,
any sequence length up to 1024, optional masking and the supported
model dimensions (dim∈{128,384,768}, hidden_dim∈{128}).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (works on any dtype, computes mean/var in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr input  [M, N]
    y_ptr,                # *ptr output [M, N]
    weight_ptr,           # *ptr weight [N]
    bias_ptr,             # *ptr bias   [N]
    M,                    # rows (B·N·N)
    N,                    # feature dim (D or H)
    eps,                  # small epsilon
    stride_xm, stride_xn,   # input strides
    stride_ym, stride_yn,   # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row indices for the current program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine
    # ------------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """Apply the Triton 2‑pass LayerNorm on a 2‑D tensor ``x`` (M,N)."""
    assert x.dim() == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()
    BLOCK_M = 256
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM for the outgoing contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A, B, C : (batch, M, K) ; (batch, K, N) ; (batch, M, N)   (FP16 inputs, FP16 output)
    The right‑hand matrix is interpreted as transposed on‑the‑fly:
        Bᵀ[row=k, col=n] = B[batch, row=n, col=k].
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)          # block-row
    pid_n = tl.program_id(2)          # block-col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load a tile of Bᵀ (K×N) – we treat B as transposed via swapped strides
        b_ptrs = b_batch + cur_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out result (cast to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul_outgoing(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Batched GEMM C = left @ rightᵀ for tensors of shape (B·H, N, N) (fp16).
    Returns a tensor of the same shape and dtype (fp16).
    """
    assert left.shape == right.shape
    batch, N, N2 = left.shape
    assert N == N2

    out = torch.empty_like(left)

    # Strides for the two inputs (contiguous layout = (batch, row, col))
    stride_a_batch = left.stride(0)
    stride_a_row   = left.stride(1)  # N
    stride_a_col   = left.stride(2)  # 1

    stride_b_batch = right.stride(0)
    # For Bᵀ we swap the last two strides:
    stride_b_row   = right.stride(2)   # original col‑stride (=1)
    stride_b_col   = right.stride(1)   # original row‑stride (=N)

    # Output strides (same layout as inputs)
    stride_c_batch = out.stride(0)
    stride_c_row   = out.stride(1)
    stride_c_col   = out.stride(2)

    grid = (batch,
            triton.cdiv(N, block_m),
            triton.cdiv(N, block_n))

    batched_matmul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights      : dict of model parameters
        - config       : dict containing 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (Triton) – keep result in FP16
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len               # rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)    # (M, D)  FP32

    norm_w = weights["norm.weight"].to(device, dtype_in)   # (D,)
    norm_b = weights["norm.bias"].to(device, dtype_in)     # (D,)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # downstream ops are much faster in FP16
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣  Fuse five linear layers (left/right proj + three gates)
    # --------------------------------------------------------------
    # Gather the five weight matrices (all have shape [H, D]) and concatenate
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                     # (5·H, D)

    # Linear projection on the flattened tensor (no bias)
    proj_gate_flat = F.linear(
        x_norm.view(M, dim),   # (M, D)  FP16
        combined_w,            # (5·H, D) FP16
    )                           # -> (M, 5·H) FP16

    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5·H)

    H = hidden_dim
    left_proj    = proj_gate[..., 0:H]                # (B,N,N,H)
    left_gate_r  = proj_gate[..., H:2 * H]
    right_proj   = proj_gate[..., 2 * H:3 * H]
    right_gate_r = proj_gate[..., 3 * H:4 * H]
    out_gate_r   = proj_gate[..., 4 * H:5 * H]

    left_gate  = torch.sigmoid(left_gate_r)      # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 3️⃣  Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Change layout to (B, H, N, N) and flatten the first two dims
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim                                # B·H
    left_mat   = left_perm.view(BH, seq_len, seq_len)   # (B·H, N, N)  FP16
    right_mat  = right_perm.view(BH, seq_len, seq_len)  # (B·H, N, N)  FP16

    # Custom Triton GEMM (left @ rightᵀ)
    out_mat = triton_batched_matmul_outgoing(
        left_mat,
        right_mat,
        block_m=128,
        block_n=128,
        block_k=32,
    )   # (B·H, N, N)  FP16

    # Restore 4‑D shape (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 4️⃣  Output LayerNorm (Triton) – stays in FP16
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                      # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_in)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_in)    # (H,)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)    # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 5️⃣  Output gate, final projection and cast back to FP32
    # --------------------------------------------------------------
    gated = out_norm * out_gate                              # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    out_fp16 = F.linear(gated, to_out_w)                     # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)                    # API expects FP32

    return out_fp32
