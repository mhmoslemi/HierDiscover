import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
============================================================

The implementation follows the reference PyTorch module but replaces the two
LayerNorms with custom Triton row‑wise kernels and fuses the four linear
projections (left/right and their gates) into a single half‑precision matmul.
The expensive N³ contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed as a batched matrix multiplication using the highly‑optimised
CUDA GEMM implementation (torch.matmul), which on H100 runs on Tensor‑cores.
All heavy arithmetic is done in fp16; reductions inside LayerNorm use fp32 for
numerical stability. The final result is cast back to float32.

The function `custom_kernel` is the entry‑point required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,       # *ptr to affine parameters [N]
    M, N,                       # rows, features
    eps,                        # epsilon
    stride_xm, stride_xn,       # strides for input
    stride_ym, stride_yn,       # strides for output
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_N: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)

    # row offsets for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (accurate in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm via a Triton kernel.
    x must be 2‑D (M, N). The weight and bias are 1‑D tensors of length N.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Launch with a reasonable block size; BLOCK_M should cover many rows.
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,          # rows per program
        BLOCK_N=64,           # columns per inner loop
    )
    return y


# ----------------------------------------------------------------------
# Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension `dim`
    # --------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)                     # [M, dim]

    norm_weight = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_bias   = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)                  # [B, L, L, D]

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj + gate) into a single matmul
    # --------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)      # (4*H, D)

    x_norm_h_flat = x_norm_h.reshape(M, dim)                  # (M, D)
    fused_out = F.linear(x_norm_h_flat, fused_w)              # (M, 4*H)

    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    left_proj = fused_out[..., :hidden_dim]                       # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]      # (B, L, L, H)
    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) before gating
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate          # (B, L, L, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication: 
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (batch*head, L, L) layout expected by torch.matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, L, L)

    # Transpose the inner dimensions of `right` to get (K, N) ordering
    right_t = right_perm.transpose(2, 3).contiguous()          # (B, H, L, L) -> (B, H, K, N)

    batch_h = bs * hidden_dim                                   # B * H
    left_mat = left_perm.reshape(batch_h, L, L)                 # (batch_h, L, K)
    right_mat = right_t.reshape(batch_h, L, L)                  # (batch_h, K, N)

    # Use cuBLAS batched GEMM via torch.matmul (Tensor‑core friendly)
    out_mat = torch.matmul(left_mat, right_mat)                 # (batch_h, L, L) fp16

    # Restore original layout: (B, L, L, H)
    out = (out_mat.view(bs, hidden_dim, L, L)
                 .permute(0, 2, 3, 1)
                 .contiguous())

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension (Triton)
    # --------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)         # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output gating (sigmoid) – linear over the original normalized input
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)              # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Apply gating and final linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                 # fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated, to_out_w)                    # (B, L, L, D) fp16
    out_final = out_final_h.to(torch.float32)                   # required dtype

    return out_final
