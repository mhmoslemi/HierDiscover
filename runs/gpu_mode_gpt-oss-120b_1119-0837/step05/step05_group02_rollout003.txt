    input_tensor, mask, weights, config = data
    bs, N, N2, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps=1e-5
    nomask = config.get("nomask", False)

    # Input LayerNorm in fp32
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)  # (B,N,N,D)

    # Cast to fp16
    x_f16 = x_norm.to(torch.float16)

    # fuse linear projections
    # gather weights
    fused_weight = torch.cat(
        [weights["left_proj.weight"],
         weights["left_gate.weight"],
         weights["right_proj.weight"],
         weights["right_gate.weight"],
         weights["out_gate.weight"]],
        dim=0
    ).to(torch.float16)
    fused = torch.nn.functional.linear(x_f16, fused_weight)  # (B,N,N,5H)

    # split
    left_proj = fused[..., :H]
    left_gate_pre = fused[..., H:2*H]
    right_proj = fused[..., 2*H:3*H]
    right_gate_pre = fused[..., 3*H:4*H]
    out_gate_pre = fused[..., 4*H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply mask
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    left = left_proj * left_gate
    right = right_proj * right_gate

    # prepare for matmul
    # rearrange to (B*H,N,N)
    B = bs
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)

    LH = left_perm.reshape(B * H, N, N)
    RH = right_perm.reshape(B * H, N, N)

    out_mat = torch.bmm(LH, RH.transpose(1,2))

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # hidden layernorm with Triton FP16
    out_flat = out.reshape(B * N * N, H)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_weight, to_out_norm_bias, eps=eps)

"""
TriMul “outgoing” forward pass – Triton‑accelerated core.

Key points
----------
* Input LayerNorm (float32) uses the highly‑optimised torch.nn.functional.layer_norm.
* Five linear projections (left, right, their gates and the output gate) are fused
  into a single half‑precision GEMM to minimise memory traffic.
* Optional binary masking is applied only to the two projected tensors.
* The N³ contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
  is performed as a batched half‑precision GEMM (torch.bmm) after reshaping the
  hidden dimension into the batch dimension.
* A custom Triton kernel normalises the hidden‑dim tensor (float16 → float16)
  and runs in a single launch (two passes over each row).
* The output‑gate (sigmoid) is fused with the hidden‑dim LayerNorm by an
  elementwise multiplication after the Triton kernel.
* Final linear projection back to model dimension runs as a half‑precision GEMM.
* The result is cast back to float32 for downstream layers.

The implementation satisfies the requirement of having a custom Triton kernel
while keeping the heavy matrix multiplications in cuBLAS/Tensor‑Core‑friendly
PyTorch calls for maximal throughput on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------
# FP16 LayerNorm kernel (two‑pass algorithm)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # input pointer (float16)
    y_ptr,               # output pointer (float16)
    weight_ptr,          # gamma pointer (float16)
    bias_ptr,            # beta  pointer (float16)
    M, N, eps,           # shape and epsilon
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------ #
    # rows processed by this program
    # ------------------------------------------------------------------ #
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------ #
    # 1️⃣  compute row‑wise mean & variance
    # ------------------------------------------------------------------ #
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)          # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------ #
    # 2️⃣  write the normalised & scaled output
    # ------------------------------------------------------------------ #
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)

        # !! normalisation !!
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        # !! scale & bias !!
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    Half‑precision LayerNorm implemented in Triton.
    Input/Output shape: (M, N) where N is the feature dimension.
    """
    assert x.dtype == torch.float16, "fp16 LayerNorm expects half‑precision input"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous (important for stride computation)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tune block sizes – larger BLOCK_M reduces kernel launches.
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # good trade‑off for our dimensions
        BLOCK_N=128,
    )
    return y


# --------------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# --------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32/bool) or None
        - weights      : dict of parameter tensors (already on target device)
        - config       : dict with keys
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[bs, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise tensor must be square (seq_len × seq_len)"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – use torch's highly‑optimised kernel
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B, N, N, D), dtype=float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the 5 linear layers (left_proj, left_gate,
    #    right_proj, right_gate, out_gate) into one GEMM.
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, D)

    # Single GEMM – shape (B, N, N, 5*H)
    fused = F.linear(x_f16, fused_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Split the fused tensor back to its logical components
    # ------------------------------------------------------------------
    # fused shape (..., 5*H)
    left_proj      = fused[..., 0          : H]
    left_gate_pre  = fused[..., H          : 2 * H]
    right_proj     = fused[..., 2 * H      : 3 * H]
    right_gate_pre = fused[..., 3 * H      : 4 * H]
    out_gate_pre   = fused[..., 4 * H      : 5 * H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)    # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 7️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate     # (B,N,N,H)
    right = right_proj * right_gate    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Transform to (B*H, N, N) where each hidden channel is a separate batch.
    # The permute is a view; the subsequent reshape incurs one copy
    # (required for cuBLAS to see contiguous matrices per batch).
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    LH = left_perm.reshape(bs * H, N, N)   # (B*H, N, N)
    RH = right_perm.reshape(bs * H, N, N)  # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(LH, RH.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Hidden‑dim LayerNorm (fp16) – custom Triton kernel
    # ------------------------------------------------------------------
    # flatten to (B·N·N, H) for the kernel
    out_flat = out.view(bs * N * N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (sigmoid) and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(bs * N * N, H)
    gated = out_norm * out_gate_flat                         # (M, H)

    to_out_w = weights["to_out.weight"].to(torch.float16)    # (D, H)
    output_flat = F.linear(gated, to_out_w)                 # (M, D), fp16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
