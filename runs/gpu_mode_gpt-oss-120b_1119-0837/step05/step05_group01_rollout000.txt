"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. 2‑pass LayerNorm over the model dimension (float32) → Triton kernel.
2. Cast to fp16 for the heavy linear work.
3. Fuse each projection + its gate (proj & gate are linear on the same input)
   by concatenating their weight matrices → a single half‑precision matmul.
4. (Optional) element‑wise mask on the projected tensors (before gating).
5. Apply sigmoid gates to left/right streams.
6. Core N³ contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM on the view (B·H, N, N).
7. Hidden‑dim LayerNorm (fp16) → Triton kernel.
8. Multiply by the output gate (sigmoid of a linear on the normalized *input*).
9. Final linear projection back to the model dimension (fp16 matmul) and cast to fp32.
   
Only the two LayerNorm steps are written in Triton; all matmuls use
torch.nn.functional.linear / torch.bmm which map to Tensor‑Core kernels on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32 (Triton)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------ row indices ------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------ first pass: compute mean & variance ------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------- second pass: write normalized ---------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton.  Input shape (M, N)."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def layernorm_fp16_kernel(

    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)

    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 / bool) or None

        - config       : dict with keys ``dim`` (model dim) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """

    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                     # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, torch.float32)
    norm_bias   = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                  # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse (proj + gate) for left/right streams
    # ------------------------------------------------------------------
    #   [proj; gate] → (2·H, dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                               # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                               # (2H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Single linear per side (produces concatenated [proj, gate])
    left_comb  = F.linear(x_h, left_comb_w)    # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)   # (B,N,N,2H)

    # Split concatenated tensors
    left_proj      = left_comb[..., :H]          # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]          # (B,N,N,H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    # Gating sigmoids
    left_gate  = torch.sigmoid(left_gate_pre)    # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (applied before gating)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):

        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)          # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)         # (B·H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)                 # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and project back to model dimension
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)        # (M, H)
    gated = out_norm * out_gate_flat              # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    output_flat = F.linear(gated, to_out_w)      # (M, dim)  (still fp16)

    output = output_flat.view(bs, N, N, dim).float()  # final fp32 tensor
    return output
```
