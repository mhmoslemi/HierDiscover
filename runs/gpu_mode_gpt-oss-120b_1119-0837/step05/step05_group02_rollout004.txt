def tri_mul_fused_gate_kernel(
    fused_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    M, H,
    stride_fused_m, stride_fused_h,
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_out_gate_m, stride_out_gate_h,
    stride_mask_m,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
    apply_mask: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # load left_proj
    offs_h = tl.arange(0, BLOCK_H)
    # compute offsets for each slice
    left_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + offs_h[None,:] * stride_fused_h
    left_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (H + offs_h)[None,:] * stride_fused_h
    right_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (2*H + offs_h)[None,:] * stride_fused_h
    right_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (3*H + offs_h)[None,:] * stride_fused_h
    out_gate_ptr_f = fused_ptr + offs_m[:, None] * stride_fused_m + (4*H + offs_h)[None,:] * stride_fused_h

    # load slices
    left_proj = tl.load(left_proj_ptr, mask=mask_m[:, None] & (offs_h < H), other=0.0)
    left_gate_pre = tl.load(left_gate_ptr, mask=mask_m[:, None] & (offs_h < H), other=0.0)
    right_proj = tl.load(right_proj_ptr, mask=mask_m[:, None] & (offs_h < H), other=0.0)
    right_gate_pre = tl.load(right_gate_ptr, mask=mask_m[:, None] & (offs_h < H), other=0.0)
    out_gate_pre = tl.load(out_gate_ptr_f, mask=mask_m[:, None] & (offs_h < H), other=0.0)

    # compute sigmoid
    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_pre))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_pre))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate_pre))

    # apply mask if needed
    if apply_mask:
        mask_val = tl.load(mask_ptr + offs_m[:, None] * stride_mask_m, mask=mask_m[:, None], other=0.0)
        left_proj = left_proj * mask_val
        right_proj = right_proj * mask_val

    # gated projection
    left = left_proj * left_gate
    right = right_proj * right_gate

    # store results
    tl.store(left_ptr + offs_m[:, None] * stride_left_m + offs_h[None,:] * stride_left_h,
              left, mask=mask_m[:, None] & (offs_h < H))
    tl.store(right_ptr + offs_m[:, None] * stride_right_m + offs_h[None,:] * stride_right_h,
              right, mask=mask_m[:, None] & (offs_h < H))
    tl.store(out_gate_ptr + offs_m[:, None] * stride_out_gate_m + offs_h[None,:] * stride_out_gate_h,
              out_gate, mask=mask_m[:, None] & (offs_h < H))
```

                        stride_fused_m, stride_fused_h,
                        stride_left_m, stride_left_h,
                        stride_right_m, stride_right_h,
                        stride_out_gate_m, stride_out_gate_h,
                        stride_mask_m,
                        BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Load mask values
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)  # scalar per row

    for h_offset in range(0, H, BLOCK_H):
        cur_h = h_offset + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # compute pointers for each slice
        slice_lo_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + cur_h[None, :] * stride_fused_h
        slice_lo_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (H + cur_h)[None, :] * stride_fused_h
        slice_ri_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (2*H + cur_h)[None, :] * stride_fused_h
        slice_ri_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (3*H + cur_h)[None, :] * stride_fused_h
        slice_out_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (4*H + cur_h)[None, :] * stride_fused_h

        # load slices
        left_proj = tl.load(slice_lo_proj_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate_pre = tl.load(slice_lo_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm
    norm_weight = weights["norm.weight"].to(device, torch.float32)
    norm_bias = weights["norm.bias"].to(device, torch.float32)
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2) cast to fp16
    x_f16 = x_norm.to(torch.float16)

    # 3) fused linear
    # Gather weight matrices and concatenate
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(device, torch.float16)

    M = bs * N * N
    x_f16_flat = x_f16.reshape(M, dim)  # shape (M,D)
    fused = torch.nn.functional.linear(x_f16_flat, fused_weight)  # shape (M,5H)

    # 4) Allocate outputs
    left = torch.empty((M, H), dtype=torch.float16, device=device)
    right = torch.empty((M, H), dtype=torch.float16, device=device)
    out_gate = torch.empty((M, H), dtype=torch.float16, device=device)

    # 5) mask flatten
    if nomask or mask is None:
        mask_flat = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(device, torch.float16).reshape(M)

    # 6) call fused kernel
    gate_fused_kernel(
        fused_ptr=fused,
        left_ptr=left,
        right_ptr=right,
        out_gate_ptr=out_gate,
        mask_ptr=mask_flat,
        M=M,
        H=H,
        stride_fused_m=fused.stride(0),
        stride_fused_h=fused.stride(1),
        stride_left_m=left.stride(0),
        stride_left_h=left.stride(1),
        stride_right_m=right.stride(0),
        stride_right_h=right.stride(1),
        stride_out_gate_m=out_gate.stride(0),
        stride_out_gate_h=out_gate.stride(1),
        stride_mask_m=mask_flat.stride(0),
        BLOCK_M=256,
        BLOCK_H=64,
    )
    # free fused to free memory (optional)
    del fused

    # reshape left, right, out_gate
    left = left.view(bs, N, N, H)
    right = right.view(bs, N, N, H)
    out_gate = out_gate.view(bs, N, N, H)

    # 7) contraction: out = sum_k left[b,i,k,:] * right[b,j,k,:]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (BH,N,N)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 8) hidden LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = torch.nn.functional.layer_norm(out, normalized_shape=(H,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # 9) output gating
    gated = out_norm * out_gate

    # 10) final projection
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # shape (dim, H)
    gated_flat = gated.reshape(M, H)
    output_flat = torch.nn.functional.linear(gated_flat, to_out_w)  # (M, dim)

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output

@triton.jit
def gate_fused_kernel(
    fused_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    M, H,
    stride_fused_m, stride_fused_h,
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_out_gate_m, stride_out_gate_h,
    stride_mask_m,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # row offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # load mask for the rows (scalar per row)
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0) # shape (BLOCK_M,)

    # number of hidden blocks
    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        h_base = hi * BLOCK_H
        offs_h = h_base + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute offsets for each slice
        # each slice starts at respective offset in fused columns
        # left_proj: 0
        left_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (offs_h)[None, :] * stride_fused_h
        left_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (H + offs_h)[None, :] * stride_fused_h
        right_proj_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (2*H + offs_h)[None, :] * stride_fused_h
        right_gate_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (3*H + offs_h)[None, :] * stride_fused_h
        out_gate_pre_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (4*H + offs_h)[None, :] * stride_fused_h

        # load slices
        left_proj = tl.load(left_proj_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate_pre = tl.load(left_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_proj = tl.load(right_proj_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_gate_pre = tl.load(right_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        out_gate_pre = tl.load(out_gate_pre_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # sigmoid approximation
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_pre))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_pre))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_pre))

        # apply mask (broadcast mask_vals to block_h dimension)
        mask_vec = mask_vals[:, None]  # shape (BLOCK_M, 1)
        left_proj = left_proj * mask_vec
        right_proj = right_proj * mask_vec

        # gated projections
        left_val = left_proj * left_gate
        right_val = right_proj * right_gate

        # store results
        tl.store(left_ptr + offs_m[:, None] * stride_left_m + offs_h[None, :] * stride_left_h,
                 left_val,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_m[:, None] * stride_right_m + offs_h[None, :] * stride_right_h,
                 right_val,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * stride_out_gate_m + offs_h[None, :] * stride_out_gate_h,
                 out_gate,
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
TriMul “outgoing” forward (AlphaFold‑3 style) – Triton‑accelerated.

Key ideas
---------
1️⃣  Input LayerNorm (FP32) is done with torch.nn.functional.layer_norm – highly
    tuned C++ implementation.
2️⃣  All heavy linear work is performed in FP16 (Tensor‑Core friendly).
3️⃣  The four linear projections (left/right + their gates) and the output‑gate
    are fused into a single GEMM, eliminating extra reads of the large input.
4️⃣  Masking, gating (sigmoid) and the two projected tensors (left/right) are
    fused in one custom Triton kernel, saving memory traffic.
5️⃣  The core N³ contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is executed as a batched half‑precision GEMM via `torch.bmm`.
6️⃣  Hidden‑dim LayerNorm (FP16) is done with torch.nn.functional.layer_norm.
7️⃣  The output gate (sigmoid of a linear) is applied, followed by the final
    linear projection back to the model dimension.
8️⃣  The final tensor is cast back to float32.

Only the fused‑gate kernel is implemented in Triton; all other ops use fast
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
#  Triton kernel: fused mask, sigmoid gating and projection
# --------------------------------------------------------------------------
@triton.jit
def gate_fused_kernel(
    fused_ptr,        # (M, 5*H)  fused linear output   (fp16)
    left_ptr,         # (M, H)    left   tensor        (fp16)
    right_ptr,        # (M, H)    right  tensor        (fp16)
    out_gate_ptr,     # (M, H)    out‑gate tensor      (fp16)
    mask_ptr,         # (M,)      binary mask (0/1)    (fp16)
    M, H,
    stride_fused_m, stride_fused_h,
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_out_gate_m, stride_out_gate_h,
    stride_mask_m,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)

    # -----------------------------
    # 1) row (M) block
    # -----------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # load scalar mask per row (broadcast later)
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)  # fp16, shape (BLOCK_M,)

    # -----------------------------
    # 2) hidden (H) loop
    # -----------------------------
    num_h = tl.cdiv(H, BLOCK_H)
    for bh in range(num_h):
        h_start = bh * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # pointers for the 5 sub‑tensors inside `fused`
        # left_proj   : columns [0          .. H-1]
        # left_gate   : columns [H          .. 2*H-1]
        # right_proj  : columns [2*H        .. 3*H-1]
        # right_gate  : columns [3*H        .. 4*H-1]
        # out_gate    : columns [4*H        .. 5*H-1]
        # Note: stride_fused_h == 1 for a contiguous tensor.
        left_proj_ptr   = fused_ptr + offs_m[:, None] * stride_fused_m + (offs_h)[None, :] * stride_fused_h
        left_gate_ptr   = fused_ptr + offs_m[:, None] * stride_fused_m + (H + offs_h)[None, :] * stride_fused_h
        right_proj_ptr  = fused_ptr + offs_m[:, None] * stride_fused_m + (2*H + offs_h)[None, :] * stride_fused_h
        right_gate_ptr  = fused_ptr + offs_m[:, None] * stride_fused_m + (3*H + offs_h)[None, :] * stride_fused_h
        out_gate_pre_ptr = fused_ptr + offs_m[:, None] * stride_fused_m + (4*H + offs_h)[None, :] * stride_fused_h

        # load slices (fp16)
        left_proj   = tl.load(left_proj_ptr,   mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate_p = tl.load(left_gate_ptr,   mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_proj  = tl.load(right_proj_ptr,  mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_gate_p = tl.load(right_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        out_gate_p   = tl.load(out_gate_pre_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # ----- sigmoid (computed in fp32 for numerical stability) -----
        left_gate_f32  = 1.0 / (1.0 + tl.exp(-tl.cast(left_gate_p,  tl.float32)))
        right_gate_f32 = 1.0 / (1.0 + tl.exp(-tl.cast(right_gate_p, tl.float32)))
        out_gate_f32   = 1.0 / (1.0 + tl.exp(-tl.cast(out_gate_p,   tl.float32)))

        left_gate  = tl.cast(left_gate_f32,  tl.float16)
        right_gate = tl.cast(right_gate_f32, tl.float16)
        out_gate   = tl.cast(out_gate_f32,   tl.float16)

        # ----- mask application (broadcast mask_vals) -----
        mask_vec = mask_vals[:, None]                     # (BLOCK_M, 1)
        left_proj  = left_proj  * mask_vec
        right_proj = right_proj * mask_vec

        # ----- gated projection -----
        left_val  = left_proj  * left_gate
        right_val = right_proj * right_gate

        # ----- write results -----
        tl.store(left_ptr + offs_m[:, None] * stride_left_m + offs_h[None, :] * stride_left_h,
                 left_val,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_m[:, None] * stride_right_m + offs_h[None, :] * stride_right_h,
                 right_val,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * stride_out_gate_m + offs_h[None, :] * stride_out_gate_h,
                 out_gate,
                 mask=mask_m[:, None] & mask_h[None, :])


# --------------------------------------------------------------------------
#  Triton‑accelerated outgoing TriMul
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (bool/float) or None
        - weights      : dict of model parameters (torch.Tensor)
        - config       : dict with keys
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the sequence dimensions"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – fast torch implementation
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, torch.float32)
    norm_bias   = weights["norm.bias"].to(device, torch.float32)
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=eps)                      # (B, N, N, D)  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the 5 linear projections (left, left‑gate, right, right‑gate, out‑gate)
    # --------------------------------------------------------------
    # shape of each weight: (H, D)  (bias=False in the reference)
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device, torch.float16)                     # (5*H, D)

    M = bs * N * N                                           # total pair positions
    x_flat = x_f16.reshape(M, dim)                          # (M, D)
    fused = F.linear(x_flat, fused_weight)                 # (M, 5*H), FP16

    # --------------------------------------------------------------
    # 4️⃣ Allocate final tensors and launch fused gate kernel
    # --------------------------------------------------------------
    left      = torch.empty((M, H), dtype=torch.float16, device=device)
    right     = torch.empty((M, H), dtype=torch.float16, device=device)
    out_gate  = torch.empty((M, H), dtype=torch.float16, device=device)

    # mask handling – if no mask (or nomask) use all‑ones
    if nomask or mask is None:
        mask_flat = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(device, torch.float16).reshape(M)

    # launch kernel
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)
    gate_fused_kernel[grid](
        fused_ptr=fused,
        left_ptr=left,
        right_ptr=right,
        out_gate_ptr=out_gate,
        mask_ptr=mask_flat,
        M=M,
        H=H,
        stride_fused_m=fused.stride(0),
        stride_fused_h=fused.stride(1),
        stride_left_m=left.stride(0),
        stride_left_h=left.stride(1),
        stride_right_m=right.stride(0),
        stride_right_h=right.stride(1),
        stride_out_gate_m=out_gate.stride(0),
        stride_out_gate_h=out_gate.stride(1),
        stride_mask_m=mask_flat.stride(0),
        BLOCK_M=128,          # rows per program
        BLOCK_H=128,          # hidden features per iteration
    )
    # fused tensor no longer needed
    del fused

    # reshape tensors back to (B, N, N, H)
    left     = left.view(bs, N, N, H)
    right    = right.view(bs, N, N, H)
    out_gate = out_gate.view(bs, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³) contraction:  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) -> batch‑gemm of (B*H, N, N) @ (B*H, N, N)ᵀ
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                 # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm (FP16) – torch implementation
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = F.layer_norm(out,
                            normalized_shape=(H,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)                 # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate                              # (B, N, N, H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    gated_flat = gated.view(M, H)                            # (M, H)
    out_flat = F.linear(gated_flat, to_out_w)                # (M, D) fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the final answer
    # --------------------------------------------------------------
    output = out_flat.view(bs, N, N, dim).to(torch.float32)
    return output
