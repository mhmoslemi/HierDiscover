@triton.jit
def tri_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, L,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid batch head
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < L
    mask_j = offs_j < L

    # accumulator for result. Use fp32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    num_k = tl.cdiv(L, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < L

        # Load left tile (M, K)
        left = tl.load(left_ptr + pid_batch * stride_left_bh
                       + offs_i[:, None] * stride_left_i
                       + offs_k[None, :] * stride_left_k,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        # Load right tile (K, N)
        right = tl.load(right_ptr + pid_batch * stride_right_bh
                        + offs_k[:, None] * stride_right_k
                        + offs_j[None, :] * stride_right_j,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)
        acc += tl.dot(left, right)   # FP16 dot to FP32 accumulate
    
    # Write output
    out = acc.to(tl.float16)
    tl.store(out_ptr + pid_batch * stride_out_bh
             + offs_i[:, None] * stride_out_i
             + offs_j[None, :] * stride_out_j,
             out,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # flatten and layernorm
    x_flat = input_tensor.reshape(bs*L*L, dim) # contiguous? probably fine
    # weights for norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # convert to half and contiguous
    norm_weight_h = norm_weight.to(device, dtype=torch.float16).contiguous()
    norm_bias_h = norm_bias.to(device, dtype=torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_weight_h, norm_bias_h, eps) # output in half

    x_norm = x_norm_flat.view(bs, L, L, dim) # half

    # cast to fp16
    x_norm_h = x_norm # shape [bs, L, L, dim] is half

    # fused linear for left/right + gating weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    # all half
    lp_w_h = left_proj_weight.to(device, dtype=torch.float16).contiguous()
    rp_w_h = right_proj_weight.to(device, dtype=torch.float16).contiguous()
    lg_w_h = left_gate_weight.to(device, dtype=torch.float16).contiguous()
    rg_w_h = right_gate_weight.to(device, dtype=torch.float16).contiguous()
    fused_w = torch.cat([lp_w_h, rp_w_h, lg_w_h, rg_w_h], dim=0) # (4*hidden_dim, dim)

    # linear: (M,D)->(M,4*hidden_dim)
    x_norm_h_flat = x_norm_h.reshape(bs*L*L, dim)
    fused_out = F.linear(x_norm_h_flat, fused_w)  # shape (M,4*hidden_dim)
    fused_out = fused_out.view(bs, L, L, 4*hidden_dim)

    left_proj = fused_out[..., :hidden_dim]
    right_proj = fused_out[..., hidden_dim:2*hidden_dim]
    left_gate = torch.sigmoid(fused_out[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3*hidden_dim:])

    if not nomask and mask is not None:
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1) # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # permute for GEMM: shape (B, H, L, L)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right: (B, H, K, N)
    right_t = right_perm.transpose(2, 3).contiguous()

    batch_h = bs * hidden_dim
    # flatten batch_h dimension
    left_flat = left_perm.view(batch_h, L, L)
    right_flat = right_t.view(batch_h, L, L)

    # compute gemm via tri_gemm
    # allocate output
    out_flat = torch.empty_like(left_flat)  # same dtype half

    # compute strides
    left_strides = left_flat.stride()
    right_strides = right_flat.stride()
    out_strides = out_flat.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch kernel
    grid=(batch_h, triton.cdiv(L, BLOCK_M), triton.cdiv(L, BLOCK_N))
    tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        L,
        # strides
        left_strides[0], left_strides[1], left_strides[2],
        right_strides[0], right_strides[1], right_strides[2],
        out_strides[0], out_strides[1], out_strides[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # restructure out_flat to original shape
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()
    # alternatively, .permute(0,2,3,1) may create noncontiguous but we will need contiguous next step for layernorm input.
    # We'll create a contiguous view after permute.

    # output layernorm
    out_M = bs * L * L
    out_flat2 = out.reshape(out_M, hidden_dim)
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # out_gate linear
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_weight)) # (M, hidden_dim)
    out_gate = out_gate.view(bs, L, L, hidden_dim)

    # apply gating
    gated = out_norm * out_gate

    # final linear projection
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16).contiguous()
    out_final_h = F.linear(gated.view(-1, hidden_dim), to_out_weight)  # (M, dim) half
    out_final = out_final_h.view(bs, L, L, dim).to(torch.float32)

    return out_final

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
================================================================

This implementation follows the reference PyTorch `TriMul` module but replaces
the two costly LayerNorms with custom Triton row‑wise kernels and replaces the
N³ contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

with a fused batched GEMM written in Triton.  All heavy arithmetic is performed
in FP16 (with FP32 accumulation for numerical stability); the final result is
cast back to FP32 as required by the model.

The entry point required by the test harness is `custom_kernel`, which takes a
tuple `(input_tensor, mask, weights, config)` and returns the processed
tensor `[B, L, L, D]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,       # *ptr to affine parameters [N]
    M, N,                       # rows, features
    eps,                        # epsilon
    stride_xm, stride_xn,       # strides for input
    stride_ym, stride_yn,       # strides for output
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_N: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row offsets for this program
    # --------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass: compute mean & variance (accurate in FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # --------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm using the Triton kernel above.
    `x` must be 2‑D (M, N).  `weight` and `bias` are 1‑D tensors of length N.
    """
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure the tensors are contiguous (required for correct strides)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,          # rows per program
        BLOCK_N=64,           # columns per inner loop
    )
    return y


# ----------------------------------------------------------------------
# Triton batched GEMM for the TriMul contraction.
# out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, L,
    # strides (row‑major [batch, i, k] for left, [batch, k, j] for right)
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    bh_id = tl.program_id(0)          # which (batch * head) we are processing
    pid_i = tl.program_id(1)          # tile row index
    pid_j = tl.program_id(2)          # tile column index

    # offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < L
    mask_j = offs_j < L

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(L, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < L

        # left tile (M x K)
        left = tl.load(
            left_ptr
            + bh_id * stride_left_bh
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (K x N)
        right = tl.load(
            right_ptr
            + bh_id * stride_right_bh
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # accumulate: the dot product is promoted to FP32 automatically
        acc += tl.dot(left, right)

    # write back in FP16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + bh_id * stride_out_bh
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension `dim`
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)                     # [M, dim]

    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)                  # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm  # already FP16 after layernorm

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj + gate) into a single matmul
    # ------------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)      # (4*H, D)

    x_norm_h_flat = x_norm_h.reshape(M, dim)                  # (M, D)
    fused_out = F.linear(x_norm_h_flat, fused_w)              # (M, 4*H)

    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    left_proj = fused_out[..., :hidden_dim]                       # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]       # (B, L, L, H)
    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) before gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate          # (B, L, L, H)
    right = right_proj * right_gate       # (B, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication via a custom Triton GEMM
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (batch, hidden) dimension
    batch_h = bs * hidden_dim

    # reshape to [batch_h, L, L] layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, L, L)
    right_t = right_perm.transpose(2, 3).contiguous()          # (B, H, K, N) = (B, H, L, L)

    left_mat = left_perm.view(batch_h, L, L)   # (batch_h, L, L)
    right_mat = right_t.view(batch_h, L, L)    # (batch_h, L, L)
    out_mat = torch.empty_like(left_mat)       # allocate output in FP16

    # strides for the kernel (elements, not bytes)
    l_str = left_mat.stride()
    r_str = right_mat.stride()
    o_str = out_mat.stride()

    # Tunable block sizes – they work well for the tested N ∈ {256,…,1024}
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(L, BLOCK_M),
        triton.cdiv(L, BLOCK_N),
    )
    tri_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_h,
        L,
        l_str[0],
        l_str[1],
        l_str[2],
        r_str[0],
        r_str[1],
        r_str[2],
        o_str[0],
        o_str[1],
        o_str[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)

    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Output gating (sigmoid) – linear over the original normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_w))  # (M, H)
    out_gate = out_gate.view(bs, L, L, hidden_dim)                 # (B, L, L, H)

    # ------------------------------------------------------------------
    # 8️⃣ Apply gating and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, L, L, H), fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated.view(-1, hidden_dim), to_out_w)  # (M, D) fp16
    out_final = out_final_h.view(bs, L, L, dim).to(torch.float32)

    return out_final
