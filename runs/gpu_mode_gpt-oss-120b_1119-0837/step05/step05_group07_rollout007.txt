@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    M, N, K,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # M = N (= rows), K=N (common dimension)
    # program IDs
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32

    # loop over reduction dimension K
    for k_offset in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_offset * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile
        # left shape (BH, M, K)
        a = tl.load(left_ptr + bh*stride_l_batch + offs_m[:, None]*stride_l_m + offs_k[None,:]*stride_l_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0)
        # Load right tile transposed
        # right shape (BH, N, K) but we need right transposed (K, N)
        # So we load as (K,N)
        b = tl.load(right_ptr + bh*stride_r_batch + offs_k[:, None]*stride_r_k + offs_n[None,:]*stride_r_n,
                    mask=mask_k[:, None] & mask_n[None,:], other=0)

        # Compute dot product: each row of a and each col of b? Actually a shape (BLOCK_M,K) and b shape (BLOCK_K,BLOCK_N)
        # we want to accumulate a @ b
        # using tl.dot a row with b col is not directly done; we can do loop over k dimension for dot with inner product? Use tl.dot?
        # tl.dot expects shape (BLOCK_M,K) and (K,BLOCK_N) but returns (BLOCK_M, BLOCK_N) after broadcasting? Actually tl.dot expects two matrices: shape (M,K) * (K,N) -> (M,N). This matches.
        acc += tl.dot(a, b)  # automatically broadcast as needed and uses fp32 accumulation.

        # increment pointers (implicitly because we use indexing with offsets; we don't need to update pointers)
    # store normalized result
    # optionally apply layernorm? Not here.

    # Store output
    out = acc.to(tl.float16)  # convert back to fp16
    tl.store(out_ptr + bh*stride_o_batch + offs_m[:, None]*stride_o_m + offs_n[None,:]*stride_o_n,
             out,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # LayerNorm 1
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    # Cast weight,bias to same dtype as input (float32)
    norm_w = weights["norm.weight"].to(device).contiguous()
    norm_b = weights["norm.bias"].to(device).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # Cast result to fp16 for speed
    x_norm = x_norm_flat.view(bs, N, N, dim).to(torch.float16)

    # Combine linear weights
    # Using weight matrices placed on device and cast to fp16
    combined_w = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device, torch.float16)

    # Linear projection
    proj = F.linear(x_norm.view(M, dim), combined_w)  # shape (M, 5*H)
    proj = proj.view(bs, N, N, 5 * hidden_dim)  # reshape

    # Slice outputs
    H = hidden_dim
    left_proj = proj[..., 0:H]
    left_gate_raw = proj[..., H:2*H]
    right_proj = proj[..., 2*H:3*H]
    right_gate_raw = proj[..., 3*H:4*H]
    out_gate_raw = proj[..., 4*H:5*H]

    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Apply mask if needed
    if (not nomask) and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Compute contraction (out = left @ right^T across k) via Triton
    # Ensure left, right are contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape (B,N,N,H)
    # compute strides
    stride_left = left.stride()
    stride_batch = stride_left[0]
    stride_i = stride_left[1]
    stride_k = stride_left[2]

    BLOCK_N = 64

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    to_out_norm_b = weights["to_out_norm.bias"].to(device).contiguous()

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul
===================================================

The implementation follows the forward pass of the TriMul module from
AlphaFold3 (outgoing version).  It consists of:

1️⃣  Two‑pass LayerNorm (mean/variance → normalise + affine) implemented in
    Triton.  The first LayerNorm normalises the input tensor over the model
    dimension ``dim``; the second normalises the hidden representation over
    ``hidden_dim``.

2️⃣  A single fused Linear projection that computes the three projections
    (left, right) and the three gating vectors (left_gate, right_gate,
    out_gate) in one matmul.  The result is kept in FP16.

3️⃣  Element‑wise Sigmoid on the three gate tensors, optional mask
    multiplication, and the gated tensors ``left`` and ``right``.

4️⃣  Batched contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    implemented as a custom Triton kernel.  The hidden dimension is fused
    into the batch dimension (BH = B·H) to avoid expensive permute &
    contiguous copies.  The kernel performs a tiled GEMM with fp16 inputs
    and fp32 accumulation.

5️⃣  Second LayerNorm over the hidden dimension, gating by ``out_gate``,
    final Linear projection back to the model dimension and cast to FP32.

All heavy arithmetic is performed in FP16 (with fp32 accumulation for
numerical stability).  The only intermediate tensors allocated are the
projected activations and the final output, keeping the memory footprint
low.

The function signature required by the test harness is:

    def custom_kernel(data):
        # data = (input_tensor, mask, weights, config)
        # returns torch.Tensor of shape [B, N, N, dim] (float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel – works on any datatype (mean/var in fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # row indices for this program (block of rows)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣ First pass – compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass – normalise + affine transform
    # ------------------------------------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Apply the Triton two‑pass LayerNorm on a 2‑D tensor ``x`` ∈ ℝ^{M×N}.
    Returns a tensor with the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "LayerNorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes work well for D ∈ {128,384,768}
    BLOCK_M = 256
    BLOCK_N = 64

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Batched contraction kernel for the TriMul “outgoing” operation
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_batch, stride_i, stride_k,
    H, N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for all batches ``b`` and hidden channels ``h``.
    The tensors are stored in layout (B,N,N,H) (row‑major, hidden dimension
    innermost).  ``H`` and ``N`` are runtime scalars.
    """
    bh = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)       # tile row (i)
    pid_n = tl.program_id(2)       # tile col (j)

    # Decompose bh → (batch, hidden)
    b = bh // H
    h = bh - b * H                  # same as bh % H

    # Base offset for this (b, h) slice
    base = b * stride_batch + h     # stride_h == 1, so just +h

    # Tile coordinates within the (N,N) matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[b,i,k,h]
        a = tl.load(
            left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (transposed): right[b,k,j,h]  → we need right[b,j,k,h]ᵀ
        b_tile = tl.load(
            right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b_tile)   # fp16×fp16 → fp32 accumulation

    # Write the result (still layout (B,N,N,H))
    out_tile = acc.to(tl.float16)
    tl.store(
        out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# TriMul entry point (forward pass only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (tensors)
        - config       : dict with keys ``dim`` (model dim) and
                         ``hidden_dim`` (projection dim); optional
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – cast to fp16 thereafter
    # --------------------------------------------------------------
    rows = bs * seq_len * seq_len              # M
    x_flat = input_tensor.reshape(rows, dim)   # (M, D)

    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)  # (M, D) fp32
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused Linear projection for left/right + gates
    # --------------------------------------------------------------
    # Concatenate projection and gate weight matrices (shape: (5·H, D))
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)                      # (5·H, D)

    # Linear (no bias) – heavy operation performed by cuBLAS
    proj = F.linear(x_norm.view(rows, dim), combined_w)   # (M, 5·H)
    proj = proj.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B, N, N, 5·H)

    H = hidden_dim
    left_proj   = proj[..., 0:H]
    left_gate_r = proj[..., H:2 * H]
    right_proj  = proj[..., 2 * H:3 * H]
    right_gate_r= proj[..., 3 * H:4 * H]
    out_gate_r  = proj[..., 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates and optional mask
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if (not nomask) and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Batched contraction via custom Triton kernel
    # --------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty_like(left)  # shape (B, N, N, H), fp16

    # Strides for tensors of shape (B, N, N, H) – row‑major contiguous
    stride_batch, stride_i, stride_k, _ = left.stride()  # (B,N,N,H) → (N*N*H, N*H, H, 1)

    BH = bs * hidden_dim                     # B·H
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        BH,
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_k=stride_k,
        H=hidden_dim,
        N=seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # `out` now holds the contracted tensor (B, N, N, H) in fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(rows, hidden_dim)   # (M, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, and cast to FP32
    # --------------------------------------------------------------
    gated = out_norm * out_gate   # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()  # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                     # (B,N,N,D) fp16
    out_fp32 = out_fp16.to(torch.float32)                    # expected dtype

    return out_fp32
