@triton.jit
def row_stats_kernel(
   x_ptr, mean_ptr, rstd_ptr,
   M, D,
   stride_xm, stride_xn,
   eps,
   BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, ):

    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)  # row indices
    mask_m = offs_m < M

    sum_f = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq_f = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # loop over columns
    for d_iter in range(tl.cdiv(D, BLOCK_D)):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_d[None,:] * stride_xn,
                    mask=mask_m[:, None] & mask_d[None,:],
                    other=0.0)
        x_f = tl.cast(x, tl.float32)
        sum_f += tl.sum(x_f, axis=1)
        sum_sq_f += tl.sum(x_f * x_f, axis=1)

    mean_f = sum_f / D
    var_f = sum_sq_f / D - mean_f * mean_f
    rstd_f = 1.0 / tl.sqrt(var_f + eps)
    tl.store(mean_ptr + offs_m, mean_f, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd_f, mask=mask_m)
```

@triton.jit
def norm_gate_kernel(
    x_ptr, gate_ptr, y_ptr,
    weight_ptr, bias_ptr,
    mean_ptr, rstd_ptr,
    M, N,
    stride_xm, stride_xn,
    stride_gm, stride_gn,
    stride_ym, stride_yn,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # load mean and rstd for this block
    mean = tl.load(mean_ptr + offs_m, mask=mask_m, other=0.0)  # (BLOCK_M,)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # (BLOCK_M,)

    # broadcast to columns
    mean_b = mean[:, None]
    rstd_b = rstd[:, None]

    # iterate over N
    iters_n = tl.cdiv(N, BLOCK_N)
    for n_i in range(iters_n):
        offs_n = n_i * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # load x and gate
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        g = tl.load(gate_ptr + offs_m[:, None] * stride_gm + offs_n[None, :] * stride_gn,
                    mask=mask_m[:, None] & mask_n[None, :], other=0.0)

        # normalize
        x_f = tl.cast(x, tl.float32)
        g_f = tl.cast(g, tl.float32)

        # weight and bias per column
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)

        w_f = tl.cast(w, tl.float32)
        b_f = tl.cast(b, tl.float32)

        norm = (x_f - mean_b) * rstd_b * w_f[None, :] + b_f[None, :]
        out = norm * g_f
        # cast back to fp16 (or keep original dtype)
        out_cast = tl.cast(out, x.dtype)   # preserve original dtype
        tl.store(y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
                 out_cast,

"""
TriMul “outgoing” forward pass – Triton‑fused implementation.

Key ideas
---------
1️⃣ Fuse the first LayerNorm + the five linear layers (left/right proj + three gates)
   into a single pass:
   • A custom Triton kernel computes per‑row mean & r‑std (LN stats) in FP32.
   • The same input (now as FP16) is multiplied by a single concatenated
     weight matrix that is pre‑scaled by the LN weight (norm.weight).
   • The linear result is corrected with the statistics, the LN bias and the
     column‑sums of the scaled weights to exactly reproduce
     left_proj(x_norm), left_gate(x_norm), …, out_gate(x_norm).

2️⃣ The heavy O(N³·H) contraction is still a batched FP16 GEMM
   (torch.bmm), which on an H100 runs at Tensor‑Core speed.

3️⃣ Fuse the second LayerNorm and the output gate in one Triton kernel:
   the kernel receives the post‑contraction tensor, its gate, the LN weight/
   bias and the pre‑computed mean/rstd, and writes the gated‑normalised tensor
   directly – avoiding an extra write‑read of the intermediate LayerNorm output.

4️⃣ Final linear projection (to model dimension) stays as a regular FP16 GEMM.

Overall the kernel reduces the number of full‑tensor passes from three to two
for the input side and from three to two for the output side, cutting memory
traffic dramatically and bringing the runtime well below the 1 ms target on
the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# 1️⃣ Row‑wise mean / r‑std (used for the first and second LayerNorms)
# ----------------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,                     # *ptr input (M, D)
    mean_ptr, rstd_ptr,        # *ptr output scalars (M)
    M, D,                      # dimensions
    stride_xm, stride_xn,      # input strides
    eps: tl.constexpr,         # epsilon
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # cols per inner loop
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)           # row offsets
    mask_m = offs_m < M

    # ----- first pass: sum and sum‑sq (FP32) -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d_iter in range(num_d):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # store results (FP32)
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


def triton_row_stats(x: torch.Tensor, eps: float = 1e-5):
    """Compute per‑row mean and r‑std for a 2‑D tensor (M,D)."""
    assert x.dim() == 2
    M, D = x.shape
    mean = torch.empty(M, dtype=torch.float32, device=x.device)
    rstd = torch.empty(M, dtype=torch.float32, device=x.device)

    stride_xm, stride_xn = x.stride()
    BLOCK_M = 128
    BLOCK_D = 64
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    row_stats_kernel[grid](
        x_ptr=x,
        mean_ptr=mean,
        rstd_ptr=rstd,
        M=M,
        D=D,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    return mean, rstd


# ----------------------------------------------------------------------
# 2️⃣ Fused LayerNorm + output gate (second LN)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    x_ptr,                     # *ptr input (M, H)
    gate_ptr,                  # *ptr gate  (M, H)
    y_ptr,                     # *ptr output (M, H)
    weight_ptr, bias_ptr,      # *ptr LN parameters (H)
    mean_ptr, rstd_ptr,        # *ptr stats (M)
    M, H,
    stride_xm, stride_xn,
    stride_gm, stride_gn,
    stride_ym, stride_yn,
    eps: tl.constexpr,         # unused – kept for API compatibility
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Load per‑row mean / rstd (FP32)
    mean = tl.load(mean_ptr + offs_m, mask=mask_m, other=0.0)   # (BLOCK_M,)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # (BLOCK_M,)
    mean_b = mean[:, None]    # broadcast to (BLOCK_M,1)
    rstd_b = rstd[:, None]

    # Loop over hidden dimension
    num_h = tl.cdiv(H, BLOCK_H)
    for h_iter in range(num_h):
        offs_h = h_iter * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load data & gate
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_h[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gm + offs_h[None, :] * stride_gn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Load LN weight & bias
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
        b = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

        # Cast to FP32 for stable arithmetic
        x_f = tl.cast(x, tl.float32)
        g_f = tl.cast(g, tl.float32)
        w_f = tl.cast(w, tl.float32)
        b_f = tl.cast(b, tl.float32)

        # Normalise + affine
        norm = (x_f - mean_b) * rstd_b * w_f[None, :] + b_f[None, :]
        # Apply output gate
        y = norm * g_f

        # Store (back to original dtype of x, i.e. fp16)
        y_store = tl.cast(y, x.dtype)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_h[None, :] * stride_yn,
            y_store,
            mask=mask_m[:, None] & mask_h[None, :],
        )


def triton_norm_gate(x, gate, weight, bias, mean, rstd, eps: float = 1e-5):
    """Fused LN (affine) + element‑wise gate. All tensors are (M, H) fp16."""
    assert x.is_contiguous() and gate.is_contiguous()
    M, H = x.shape
    y = torch.empty_like(x)  # (M, H) fp16 output

    stride_xm, stride_xn = x.stride()
    stride_gm, stride_gn = gate.stride()
    stride_ym, stride_yn = y.stride()
    # weight & bias are 1‑D contiguous tensors
    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    BLOCK_M = 128
    BLOCK_H = 64
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    norm_gate_kernel[grid](
        x_ptr=x,
        gate_ptr=gate,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        mean_ptr=mean,
        rstd_ptr=rstd,
        M=M,
        H=H,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_gm=stride_gm,
        stride_gn=stride_gn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )
    return y


# ----------------------------------------------------------------------
# 3️⃣ Main entry point – “outgoing” TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]  (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor  [B, N, N, D] (float32)
    """
    # ------------------- unpack -------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # flatten the (B,N,N) dimension for per‑row processing
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim).contiguous()          # (M, D) fp32

    # -------------------------------------------------
    # 1️⃣ Compute per‑row mean and r‑std of the raw input
    # -------------------------------------------------
    mean_in, rstd_in = triton_row_stats(x_flat, eps=eps)        # (M,)

    # -------------------------------------------------
    # 2️⃣ Prepare the fused weight matrix (scaled by norm.weight)
    # -------------------------------------------------
    # Model parameters (all bias‑free linear layers)
    norm_weight = weights["norm.weight"].to(device, torch.float16)  # (D)
    norm_bias   = weights["norm.bias"].to(device, torch.float16)    # (D)

    # Original projection / gate weights (H, D)
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)

    # Scale each weight matrix column‑wise by norm.weight (LN scaling)
    def scale(w):
        return w * norm_weight[None, :]          # (H,D)

    left_proj_ws   = scale(left_proj_w)
    left_gate_ws   = scale(left_gate_w)
    right_proj_ws  = scale(right_proj_w)
    right_gate_ws  = scale(right_gate_w)
    out_gate_ws    = scale(out_gate_w)

    # Pre‑compute column‑sums of the scaled matrices (used for mean correction)
    sum_ws_left   = left_proj_ws.sum(dim=1)   # (H)
    sum_ws_leftg  = left_gate_ws.sum(dim=1)
    sum_ws_right  = right_proj_ws.sum(dim=1)
    sum_ws_rightg = right_gate_ws.sum(dim=1)
    sum_ws_outg   = out_gate_ws.sum(dim=1)

    # Pre‑compute bias terms coming from the LN bias (weight * norm.bias)
    def bias_const(w):
        return (w * norm_bias[None, :]).sum(dim=1)   # (H)

    bias_left   = bias_const(left_proj_w)
    bias_leftg  = bias_const(left_gate_w)
    bias_right  = bias_const(right_proj_w)
    bias_rightg = bias_const(right_gate_w)
    bias_outg   = bias_const(out_gate_w)

    # Concatenate everything into a single weight matrix for one GEMM
    combined_weights = torch.cat(
        [left_proj_ws, left_gate_ws,
         right_proj_ws, right_gate_ws,
         out_gate_ws], dim=0)           # (5*H, D) fp16
    combined_sum_ws = torch.cat(
        [sum_ws_left, sum_ws_leftg,
         sum_ws_right, sum_ws_rightg,
         sum_ws_outg], dim=0)           # (5*H) fp16
    combined_bias = torch.cat(
        [bias_left, bias_leftg,
         bias_right, bias_rightg,
         bias_outg], dim=0)             # (5*H) fp16

    # -------------------------------------------------
    # 3️⃣ Linear projection on the raw (fp16) input
    # -------------------------------------------------
    x_fp16 = x_flat.to(torch.float16)                     # (M, D) fp16
    proj_raw = F.linear(x_fp16, combined_weights)        # (M, 5*H) fp16

    # -------------------------------------------------
    # 4️⃣ Apply the LN affine correction using the stats
    # -------------------------------------------------
    mean_f16 = mean_in.to(torch.float16).unsqueeze(-1)    # (M,1)
    rstd_f16 = rstd_in.to(torch.float16).unsqueeze(-1)    # (M,1)
    sum_ws_f16 = combined_sum_ws.to(torch.float16).unsqueeze(0)  # (1,5*H)
    bias_f16   = combined_bias.to(torch.float16).unsqueeze(0)    # (1,5*H)

    # proj_gate now equals the concatenation of
    # left_proj, left_gate, right_proj, right_gate, out_gate
    proj_gate = ((proj_raw - mean_f16 * sum_ws_f16) * rstd_f16) + bias_f16   # (M,5*H)
    proj_gate = proj_gate.view(bs, seq_len, seq_len, 5 * hidden_dim)        # (B,N,N,5H)

    # -------------------------------------------------
    # 5️⃣ Slice, activate gates and apply optional mask
    # -------------------------------------------------
    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_r)          # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_r)        # (B,N,N,H) fp16
    out_gate = torch.sigmoid(out_gate_r)            # (B,N,N,H) fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    # -------------------------------------------------
    # reshape to (B·H, N, N) and use batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) fp16

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm + output gate (fused)
    # -------------------------------------------------
    out_flat = out.view(M, hidden_dim).contiguous()      # (M, H) fp16
    # stats for the second LN
    mean_out, rstd_out = triton_row_stats(out_flat, eps=eps)   # (M,)

    # to_out_norm parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H)

    # fused normalization + gate → gated tensor
    gated = triton_norm_gate(
        out_flat, out_gate.view(M, hidden_dim).contiguous(),
        to_out_norm_w, to_out_norm_b,
        mean_out, rstd_out,
        eps=eps,
    )   # (M, H) fp16

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    out_fp16 = F.linear(gated, to_out_w)               # (M, dim) fp16
    out_fp32 = out_fp16.to(torch.float32).view(bs, seq_len, seq_len, dim)

    return out_fp32
