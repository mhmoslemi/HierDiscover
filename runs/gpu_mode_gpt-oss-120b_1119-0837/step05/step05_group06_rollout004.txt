@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    # maybe we need stride_out_h?
    stride_out_h,
    eps? not needed
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
   # we map program ID axes:
   batch_id = tl.program_id(0)  # [0, B*H)
   i_tile = tl.program_id(1)    # blocks along i dimension
   j_tile = tl.program_id(2)    # blocks along j dimension

   # derive batch and hidden indices
   b = batch_id // H
   h = batch_id % H

   # Offsets for block start
   offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   # Compute base offset for left & right & out for this batch
   left_batch_ptr = left_ptr + b*stride_left_b + h  # base pointer for batch/h
   right_batch_ptr = right_ptr + b*stride_right_b + h
   out_batch_ptr = out_ptr + b*stride_out_b + h # we may add offset for each hidden

   # compute pointer offsets for left block (i,k)
   # need to loop over k dimension blocks

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # compute number of K blocks
   num_k_blocks = tl.cdiv(N, BLOCK_K)
   for k_iter in range(num_k_blocks):
       offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # left pointer offsets: i offset * stride_i + k offset * stride_k
       left_ptrs = left_batch_ptr + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

       # right pointer offsets: k offset * stride_right_k + j offset * stride_right_i
       right_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

       # load
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
       right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

       # cast to fp32 for accumulation
       left_f32 = tl.cast(left, tl.float32)
       right_f32 = tl.cast(right, tl.float32)

       # dot product across k dimension: accumulate sum_i_k_j
       # We can use tl.dot for each row of left and column of right.
       # However tl.dot expects 1D vectors; we can use matrix multiplication approach: accumulate
       # left_f32 (M,K) dot right_f32 (K,N) => shape (M,N)
       acc += tl.dot(left_f32, right_f32)  # This uses matrix multiplication for block?
   # after loop, store back with cast to fp16
   out_ptrs = out_batch_ptr + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
   # store FP16
   tl.store(out_ptrs, tl.cast(acc, tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing’’ forward pass – Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Layer‑Norm the input (B,N,N,D).  Implemented in Triton (2‑pass, FP32
   statistics, FP16 result).  The input is cast to fp16 before the first
   norm to keep the whole pipeline in half‑precision.

2️⃣ Fuse the five linear layers (left/right projections + three gates) into
   a single GEMM:  x_flat @ W_cat  →  (B·N·N, 5·H).  The result is sliced
   into left_proj, left_gate, right_proj, right_gate and out_gate.

3️⃣ Apply sigmoid to the three gate tensors, optionally multiply a pair‑wise
   mask, and obtain the gated left/right tensors (still (B,N,N,H) in fp16).

4️⃣ Heavy O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   is performed by a custom Triton kernel that works directly on the
   (B,N,N,H) layout – no costly .permute/.contiguous copies.  The kernel
   treats (b,h) as a combined batch dimension and executes a blocked
   FP16 matrix‑multiply (B·H independent GEMMs) with FP32 accumulation.

5️⃣ Layer‑Norm the intermediate result (again a Triton kernel).

6️⃣ Apply the output gate, a final linear projection back to the model
   dimension D and cast the answer to float32 (the public API type).

All the heavy arithmetic (GEMMs, batched GEMM, accumulations) runs in
FP16 with Tensor‑Core acceleration, while the two LayerNorms use a
lightweight Triton implementation to avoid extra memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (last dimension) – works with any dtype (fp16/fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # rows   (= B·N·N)
    N,                # feature dimension (D or H)
    eps,              # epsilon
    stride_xm, stride_xn,      # strides of x
    stride_ym, stride_yn,      # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- row offsets ---------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: sum and sum‑sq (FP32) -------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine -------------------------------
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for the Triton LayerNorm kernel.
    ``x`` must be a 2‑D contiguous tensor (M, N) and the result has the same
    dtype as ``x``.
    """
    assert x.ndim == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for the heavy O(N³·H) contraction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # *ptr left  [B, N, N, H]   (fp16)
    right_ptr,  # *ptr right [B, N, N, H]   (fp16)
    out_ptr,    # *ptr out   [B, N, N, H]   (fp16)
    B: tl.int64,
    N: tl.int64,
    H: tl.int64,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM denoted   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]"""

    batch_id = tl.program_id(0)      # 0 … B·H-1  (combined (b,h))
    tile_i   = tl.program_id(1)      # row‑tile index   (i dimension)
    tile_j   = tl.program_id(2)      # col‑tile index   (j dimension)

    # unpack (b, h) from the fused batch id
    b = batch_id // H
    h = batch_id % H

    # ---- base pointers for this (b,h) ----------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h   # +h because hidden stride == 1
    right_base = right_ptr + b * stride_right_b + h
    out_base   = out_ptr   + b * stride_out_b   + h

    # ---- block offsets -------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right: shape (BLOCK_K, BLOCK_N)  (note: we read (k,j) directly)
        right_ptrs = right_base \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_i
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate in FP32
        acc += tl.dot(tl.cast(left, tl.float32), tl.cast(right, tl.float32))

    # ---- write result ---------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(
        out_ptrs,
        tl.cast(acc, tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

    M = bs * seq_len * seq_len                     # = B·N·N
    x_fp16_flat = x_fp16.reshape(M, dim)          # (M, D) fp16

    # LayerNorm weights/bias must match the dtype of x_fp16
    norm_w = weights["norm.weight"].to(device, torch.float16)
    norm_b = weights["norm.bias"].to(device, torch.float16)

    x_norm_fp16_flat = triton_layernorm(x_fp16_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_fp16_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one GEMM
    # ------------------------------------------------------------------
    # Gather weight matrices (all stored as FP32 in the checkpoint)
    left_proj_w   = weights["left_proj.weight"]           # (H, D)
    right_proj_w  = weights["right_proj.weight"]          # (H, D)
    left_gate_w   = weights["left_gate.weight"]           # (H, D)
    right_gate_w  = weights["right_gate.weight"]          # (H, D)
    out_gate_w    = weights["out_gate.weight"]            # (H, D)

    # Concatenate along the output feature axis -> (5·H, D)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)

    # Single fused linear projection
    proj_gate_flat = F.linear(x_norm_fp16_flat, combined_w)   # (M, 5·H) fp16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply sigmoid, optional mask
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                       # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_r)                # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_r)              # (B,N,N,H) fp16
    out_gate = torch.sigmoid(out_gate_r)                  # (B,N,N,H) fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ O(N³·H) contraction via a custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16, device=device)   # (B,N,N,H)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for H100 (fp16 + TensorCore)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,                              # combined (b,h) batch
        triton.cdiv(seq_len, BLOCK_M),                # i‑tiles
        triton.cdiv(seq_len, BLOCK_N),                # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_right_b=stride_r_b,
        stride_right_i=stride_r_i,
        stride_right_k=stride_r_k,
        stride_out_b=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (still in fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)               # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)   # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)     # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply the output gate, final projection and cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                       # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    gated_flat = gated.view(M, hidden_dim)          # (M, H)
    out_fp16_flat = F.linear(gated_flat, to_out_w)  # (M, D) fp16

    out_fp32 = out_fp16_flat.view(bs, seq_len, seq_len, dim).to(torch.float32)

    return out_fp32
