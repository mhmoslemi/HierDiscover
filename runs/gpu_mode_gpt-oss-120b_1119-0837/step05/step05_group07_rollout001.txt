"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

The implementation follows the PyTorch reference in AlphaFold3 but
replaces the two LayerNorms with fast two‑pass Triton kernels and
fuses the five linear layers (left/right projections + three gates) into
a single GEMM.  The heavy inner contraction

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

is performed with a batched matrix multiplication (torch.bmm) which
exploits Tensor‑Core FP16 GEMMs on H100.  All intermediate tensors are
kept in FP16 to maximise throughput; the final result is cast back to
float32 as required by the public API.

The algorithm steps are:
1️⃣  LayerNorm over the last dimension (dim) → FP16.
2️⃣  Single linear projection producing
        left_proj, left_gate, right_proj, right_gate, out_gate
    (all shape …×hidden_dim) in FP16.
3️⃣  Sigmoid gates + optional binary mask.
4️⃣  Batched GEMM for the pairwise contraction (B·hidden_dim independent
    matrix multiplies of size N×N).
5️⃣  LayerNorm over the hidden dimension → FP16.
6️⃣  Apply output gate, final linear projection back to model dimension,
    cast to float32.

Only the two LayerNorms are implemented in Triton; the remaining ops
use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (TensorFloat‑32 precision for statistics)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # number of rows (B·N·N)
    N,              # feature dimension (D or H)
    eps,            # epsilon for stability
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # -------------------------- row indices ---------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------- first pass: mean & var ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_n = tl.cdiv(N, BLOCK_N)

    for it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -------------------- second pass: normalise + affine ------------
    for it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise in FP32
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # load affine parameters (may be FP16) and cast to FP32
        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm * w[None, :] + b[None, :]

        # store result with original dtype (FP16 or FP32)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Triton two‑pass LayerNorm on a 2‑D tensor (M, N).
    Returns a tensor with the same dtype as ``x``.
    """
    assert x.dim() == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for the H100.  BLOCK_M processes many rows at once,
    # BLOCK_N is chosen to get good occupancy for N∈{128,384,768}.
    BLOCK_M = 256
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (binary) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` (D) and ``hidden_dim`` (H);
                         optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32).
    """
    # ---------------------------- unpack -------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype  # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------- 1️⃣ LayerNorm ------------------------
    # Reshape to (M, D) where M = B·N·N
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim).contiguous()          # (M, D) FP32

    # LayerNorm parameters (float32)
    norm_w = weights["norm.weight"].to(device, dtype_in)
    norm_b = weights["norm.bias"].to(device, dtype_in)

    # Two‑pass LN → keep FP32 for numeric stability, then cast to FP16
    x_norm_fp32 = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # -------------------- 2️⃣ fused linear projection ---------------
    # Gather the five weight matrices (each H×D) and concat along the
    # output dimension → (5·H, D).  No bias terms.
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)               # (5·H, D)

    # Linear projection on the flattened input (M, D) → (M, 5·H)
    proj_flat = F.linear(
        x_norm.view(M, dim),   # (M, D) FP16
        combined_w,            # (5·H, D) FP16
    )                           # → (M, 5·H) FP16

    # -------------------------- 3️⃣ split & gate --------------------
    proj = proj_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)

    H = hidden_dim
    left_proj    = proj[..., 0:H]                 # (B,N,N,H)
    left_gate_r  = proj[..., H:2*H]
    right_proj   = proj[..., 2*H:3*H]
    right_gate_r = proj[..., 3*H:4*H]
    out_gate_r   = proj[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)      # FP16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        # `mask` is binary; broadcast on hidden dimension.
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------- 4️⃣ batched GEMM -------------------
    # Rearrange to (B, H, N, N) → each (b, h) forms a matrix.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)    # (B·H, N, N)

    # Perform (B·H) independent GEMMs: (N,N) @ (N,N)^T → (N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Restore (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) FP16

    # --------------------------- 5️⃣ second LN ----------------------
    out_flat = out.view(M, hidden_dim)                # (M, H) FP16
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_in)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_in)    # (H,)

    out_norm_fp32 = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_fp32.view(bs, seq_len, seq_len, hidden_dim)    # (B,N,N,H) FP16

    # --------------------------- 6️⃣ gating & final proj ----------
    gated = out_norm * out_gate                                   # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                          # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)                         # API expects FP32

    return out_fp32
