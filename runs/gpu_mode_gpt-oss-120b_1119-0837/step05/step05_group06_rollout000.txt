"""
TriMul “outgoing’’ forward pass – Triton‑accelerated implementation.

This implementation follows the reference PyTorch version but
* fuses the five linear layers (left/right projections + three gates) into a
  single GEMM,
* performs both LayerNorms with a custom 2‑pass Triton kernel (FP32 stats,
  FP16 data),
* keeps the heavy contraction in FP16 and uses batched cuBLAS GEMMs
  (torch.bmm) for the O(N³·H) operation,
* applies the optional input mask and all sigmoid gates,
* returns the final result in FP32 as required by the public API.

All tensors are assumed to reside on the same CUDA device; the code works
for the various test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension). Supports any dtype (fp16 /
# fp32 data, but computes mean/var in fp32 for stability).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # number of rows (B·N·N)
    N,                # feature dimension (D or H)
    eps,              # epsilon for stability
    stride_xm, stride_xn,      # strides of x
    stride_ym, stride_yn,      # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- row offsets for this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute sum and sum‑sq (FP32) ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N

        # Load a tile of x
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine -------------------------------
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # load affine parameters (weight, bias) – they may be fp16 or fp32
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Applies LayerNorm over the last dimension of a 2‑D tensor ``x`` (shape
    [M, N]) using the Triton kernel above. ``weight`` and ``bias`` are
    vectors of length N. The output has the same dtype and shape as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for D = 128, 384, 768. They fit comfortably in
    # shared memory and provide good occupancy on H100.
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys:
            * "dim"          – model dimension D
            * "hidden_dim"   – hidden dimension H
            * "nomask" (opt) – bool, true if mask should be ignored

    Returns
    -------
    torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # dim comes from config but we also verify consistency
    dim_cfg = config["dim"]
    assert dim == dim_cfg, f"Input dim {dim} does not match config dim {dim_cfg}"
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – over dim (last axis)
    # --------------------------------------------------------------
    # Flatten first three dimensions so we have a 2‑D tensor (M, D)
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)  # (M, D) – still FP32

    # Load LayerNorm parameters (FP32)
    norm_w = weights["norm.weight"].to(device, dtype)   # (D,)
    norm_b = weights["norm.bias"].to(device, dtype)     # (D,)

    # Apply the Triton layernorm; result stays FP32
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # Reshape back to the original 4‑D layout
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,D) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the heavy linear algebra runs in half precision
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)           # (B,N,N,D) fp16
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)    # (M, D) fp16

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (projections + three gates) into
    #    a single GEMM:   W_fused @ x_norm   → shape (M, 5*H)
    # --------------------------------------------------------------
    # Weight shapes from the reference code:
    #   proj: (H, D), gate: (H, D)  (bias=False) → we can concat them.
    left_proj_w   = weights["left_proj.weight"]   # (H, D)
    right_proj_w  = weights["right_proj.weight"]  # (H, D)
    left_gate_w   = weights["left_gate.weight"]   # (H, D)
    right_gate_w  = weights["right_gate.weight"]  # (H, D)
    out_gate_w    = weights["out_gate.weight"]    # (H, D)

    # Concatenate along the output feature dimension → (5*H, D)
    fused_W = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)  # (5*H, D)

    # Perform the single fused GEMM. No bias is needed (all Linear layers had bias=False).
    proj_gate_flat = F.linear(x_norm_f16_flat, fused_W)  # (M, 5*H) fp16

    # --------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor, apply sigmoid to gate tensors,
    #    and optionally mask left/right projections.
    # --------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    # Index ranges:
    #   [0:H]          → left_proj
    #   [H:2H]         → left_gate (pre‑sigmoid)
    #   [2H:3H]        → right_proj
    #   [3H:4H]        → right_gate (pre‑sigmoid)
    #   [4H:5H]        → out_gate (pre‑sigmoid)
    left_proj    = proj_gate[..., 0:H]
    left_gate_pre = proj_gate[..., H:2*H]
    right_proj   = proj_gate[..., 2*H:3*H]
    right_gate_pre = proj_gate[..., 3*H:4*H]
    out_gate_pre = proj_gate[..., 4*H:5*H]

    # Sigmoid for the gate tensors
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_pre)  # (B,N,N,H) fp16
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H) fp16

    # Apply optional mask. The mask has shape (B,N,N); broadcast to (B,N,N,1).
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16   # (B,N,N,H)
        right = right_proj * right_gate * mask_f16 # (B,N,N,H)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h]·right[b,j,k,h]
    #    → treat hidden dimension as batch (B·H, N, N) and use torch.bmm.
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) then flatten batch and hidden dimensions.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim  # total batch size for bmm
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    # Note: right_mat is transposed on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # Restore to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()    # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – over hidden_dim
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                # (B·N·N, H) fp16
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)    # (H,)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply the output gate, final projection back to model dim D,
    #    and cast to FP32 for the final API contract.
    # --------------------------------------------------------------
    gated = out_norm * out_gate                         # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                           # (B,N,N,D) fp16

    return out_fp16.to(torch.float32)   # final output in FP32
