"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated
====================================================================

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32) with a custom Triton kernel.
2️⃣ Cast the normalized tensor to FP16 (the heavy work stays in half‑precision).
3️⃣ Fuse each projection (left/right) with its gating linear layer:
   * one linear call per side producing 2·H channels → split into projection & gate.
4️⃣ (Optional) Apply the binary mask to the raw projections (broadcast over H).
5️⃣ Apply the sigmoid gates → obtain gated tensors *left* and *right* (fp16).
6️⃣ Compute the N³ contraction:
       out[i,j,h] = Σ_k left[i,k,h]·right[j,k,h]
   This is a batched matrix‑multiplication L @ Rᵀ for each head.
   We use the highly‑optimised cuBLAS batched GEMM (`torch.bmm`) which is
   substantially faster on an H100 than a hand‑written Triton GEMM.
7️⃣ Hidden‑dimension layer‑norm (float16) via a second Triton kernel.
8️⃣ Apply the output‑gate (sigmoid of a linear on the original input).
9️⃣ Final linear projection back to model dimension (fp16 → fp32).

Only the two layer‑norms are written with Triton, satisfying the “must use a
kernel” requirement while the dominant N³ contraction is delegated to cuBLAS.
All tensors returned are float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# Float‑32 LayerNorm (2‑pass) – Triton kernel
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -----------------------------------------------------------------
    # first pass – compute mean & variance per row
    # -----------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_N)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -----------------------------------------------------------------
    # second pass – write normalized result
    # -----------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float32) using a Triton kernel."""
    assert x.dim() == 2, "layernorm fp32 expects 2‑D input (M, D)"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Float‑16 LayerNorm (2‑pass) – Triton kernel
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float16) using a Triton kernel."""
    assert x.dtype == torch.float16, "fp16 layernorm expects half‑precision input"
    assert x.dim() == 2, "layernorm fp16 expects 2‑D input (M, D)"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Custom TriMul kernel entry point
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 version).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32, on CUDA)
        - mask         : torch.Tensor[bs, N, N] (bool/float) or None
        - weights      : dict containing the model parameters (see keys below)
        - config       : dict with at least the fields
            * dim        – model dimension (D)
            * hidden_dim – hidden dimension (H)
            * nomask     – optional bool, if True ignore the mask

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, Triton)
    # ------------------------------------------------------------------
    # Collapse batch & pair dimensions → (B·N·N, dim)
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)              # (M, D)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)           # (B, N, N, D)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gating
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for each side
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16)   # (H, D)
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16)   # (H, D)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)  # (H, D)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)  # (H, D)

    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, D)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2H, D)

    # One linear per side → shape (B, N, N, 2H)
    left_comb  = F.linear(x_f16, left_comb_w)    # (B, N, N, 2H)
    right_comb = F.linear(x_f16, right_comb_w)   # (B, N, N, 2H)

    # Split projection & gate streams
    left_proj = left_comb[..., :H]                # (B, N, N, H)
    left_gate_pre = left_comb[..., H:]            # (B, N, N, H)
    right_proj = right_comb[..., :H]              # (B, N, N, H)
    right_gate_pre = right_comb[..., H:]          # (B, N, N, H)

    # Gating activations
    left_gate  = torch.sigmoid(left_gate_pre)     # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_pre)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask may be bool or float – convert to half and broadcast
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (depends only on x)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)   # (H, D)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))               # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via cuBLAS batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → batch dimension becomes B*H
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = bs * H
    # shape (batch, N, N)
    left_mat  = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # Perform L @ Rᵀ for each head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (batch, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16, Triton)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)    # (H,)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)           # (M, H)
    gated = out_norm * out_gate_flat                 # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    output_flat = F.linear(gated, to_out_w)         # (M, dim) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # back to FP32

    return output
