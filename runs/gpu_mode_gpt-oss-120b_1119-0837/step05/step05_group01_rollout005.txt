@triton.jit
def fused_norm_gate_linear(
    out_ptr,    # M, D output (half)
    a_ptr,      # M, H input (half)
    gate_ptr,   # M, H gating values (half)
    w_ptr,      # H, D final weight (half)
    ln_weight_ptr,   # H
    ln_bias_ptr,     # H
    eps,
    M, H, D,
    stride_a_m, stride_a_h,
    stride_gate_m, stride_gate_h,
    stride_w_h, stride_w_d,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute per output block rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Compute sum, sumsq across H for each row
    # Use float32 accumulators
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        sum_val += tl.sum(a_f, axis=1)
        sum_sq += tl.sum(a_f * a_f, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Prepare output accumulator
    # Determine number of output columns
    offs_d = tl.arange(0, BLOCK_D)
    # We'll allocate accumulator across D dimension
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # Loop over K dimension again for matmul
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a and gating
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        # Cast to fp32
        a_f = tl.cast(a, tl.float32)
        g_f = tl.cast(g, tl.float32)

        # Normalization: (a - mean)*rstd
        # mean, rstd need broadcast across H, shape (BLOCK_M, 1)
        a_norm = (a_f - mean[:, None]) * rstd[:, None]

        # LayerNorm weight and bias per channel (H)
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
        ln_b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
        ln_w_f = tl.cast(ln_w, tl.float32)
        ln_b_f = tl.cast(ln_b, tl.float32)

        # Apply LN scaling/bias and gating
        # scaled = (a_norm * ln_w_f + ln_b_f) * g_f
        scaled = (a_norm * ln_w_f + ln_b_f) * g_f     # shape (BLOCK_M, BLOCK_H)

        # Load weight tile for final linear: w_ptr shape (H, D)
        offs_d = tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        w = tl.load(
            w_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :] * stride_w_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0
        )
        w_f = tl.cast(w, tl.float32)  # shape (BLOCK_H, BLOCK_D)

        # Multiply and accumulate
        acc += tl.dot(scaled, w_f)   # returns (BLOCK_M, BLOCK_D)

    # Write out result (cast to half)
    # Need to handle possible partial rows and columns
    out = tl.cast(acc, tl.float16)

    # Compute column offsets for output
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
        out,
        mask=mask_m[:, None] & mask_d[None, :]
    )

Pseudo-code:

```
@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr, a_ptr, gate_ptr, w_ptr,
    ln_weight_ptr, ln_bias_ptr,
    eps,
    M, H, D,
    stride_a_m, stride_a_h,
    stride_gate_m, stride_gate_h,
    stride_w_h, stride_w_d,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # Determine block coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(D, BLOCK_D)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # First pass: compute mean and variance for each row in this block.
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        sum_val += tl.sum(a_f, axis=1)
        sum_sq += tl.sum(a_f * a_f, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: compute normalized, gated, weight product.
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
    # For each H tile
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a and gate
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        g_f = tl.cast(g, tl.float32)

        # Normalize
        a_norm = (a_f - mean[:, None]) * rstd[:, None]

        # Apply LN weight and bias
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
        ln_b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
        ln_w_f = tl.cast(ln_w, tl.float32)
        ln_b_f = tl.cast(ln_b, tl.float32)

        scaled = (a_norm * ln_w_f + ln_b_f) * g_f   # shape (BLOCK_M, BLOCK_H)

        # Load weight tile (H x D)
        w = tl.load(
            w_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :] * stride_w_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0
        )
        w_f = tl.cast(w, tl.float32)   # shape (BLOCK_H, BLOCK_D)

        # Accumulate
        acc += tl.dot(scaled, w_f)   # (BLOCK_M, BLOCK_D)

    # Write out results (cast to half)
    out = tl.cast(acc, tl.float16)

    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
        out,
        mask=mask_m[:, None] & mask_d[None, :]
    )
```

@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr,                # *M*D output (half)
    a_ptr,                  # *M*H pre‑norm tensor (half)
    gate_ptr,               # *M*H gating tensor (half)
    weight_ptr,             # *H*D linear weight (half)
    ln_weight_ptr,          # *H* LayerNorm weight (half)
    ln_bias_ptr,            # *H* LayerNorm bias (half)
    eps,                    # eps for LN
    M, H, D,
    stride_a_m, stride_a_h,
    stride_gate_m, stride_gate_h,
    stride_weight_h, stride_weight_d,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    """Fused LN (FP16), gating, and final Linear (H×D)"""

    pid = tl.program_id(0)

    # 2‑D grid: rows (M) × cols (D)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(D, BLOCK_D)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # ----------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & rstd (float32)
    # ----------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        sum_val += tl.sum(a_f, axis=1)
        sum_sq  += tl.sum(a_f * a_f, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------------------------
    # 2️⃣ Main pass: normalized × LN‑scale/bias × gate, then dot‑weight
    # ----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a and its gate
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        g_f = tl.cast(g, tl.float32)

        # Normalise: (a - μ) / σ
        a_norm = (a_f - mean[:, None]) * rstd[:, None]

        # LayerNorm weight & bias (per hidden channel)
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
        ln_b = tl.load(ln_bias_ptr + offs_h,   mask=mask_h, other=0.0)
        ln_w_f = tl.cast(ln_w, tl.float32)
        ln_b_f = tl.cast(ln_b, tl.float32)

        # Apply LN scaling, bias and output‑gate
        # shape → (BLOCK_M, BLOCK_H)
        gated = (a_norm * ln_w_f + ln_b_f) * g_f

        # Load final linear weight tile (H×D)
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_weight_h + offs_d[None, :] * stride_weight_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0
        )
        w_f = tl.cast(w, tl.float32)   # (BLOCK_H, BLOCK_D)

        # Accumulate contribution of this H‑slice
        acc += tl.dot(gated, w_f)      # (BLOCK_M, BLOCK_D)

    # ----------------------------------------------------------------------
    # 3️⃣ Write output (cast back to FP16)
    # ----------------------------------------------------------------------
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
        out,
        mask=mask_m[:, None] & mask_d[None, :]
    )

# Helper that launches the fused kernel
# --------------------------------------------------------------------------
def fused_norm_gate_linear(
    a: torch.Tensor,
    gate: torch.Tensor,
    weight: torch.Tensor,
    ln_weight: torch.Tensor,
    ln_bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Compute:
        out = ((a - μ)/σ * ln_weight + ln_bias) * gate @ weight
    a, gate : (M, H) FP16
    weight   : (H, D) FP16   (already transposed)
    ln_weight, ln_bias : (H,) FP16
    Returns (M, D) FP16.
    """
    assert a.is_contiguous() and gate.is_contiguous()
    M, H = a.shape
    D = weight.shape[1]
    out = torch.empty((M, D), dtype=torch.float16, device=a.device)

    stride_a_m, stride_a_h = a.stride()
    stride_gate_m, stride_gate_h = gate.stride()
    stride_w_h, stride_w_d = weight.stride()
    stride_out_m, stride_out_d = out.stride()

    grid = lambda meta: (
        triton.cdiv(M, meta["BLOCK_M"]) *
        triton.cdiv(D, meta["BLOCK_D"]),
    )
    fused_norm_gate_linear_kernel[grid](
        out_ptr=out,
        a_ptr=a,
        gate_ptr=gate,
        weight_ptr=weight,
        ln_weight_ptr=ln_weight,
        ln_bias_ptr=ln_bias,
        eps=eps,
        M=M,
        H=H,
        D=D,
        stride_a_m=stride_a_m,
        stride_a_h=stride_a_h,
        stride_gate_m=stride_gate_m,
        stride_gate_h=stride_gate_h,
        stride_weight_h=stride_w_h,
        stride_weight_d=stride_w_d,
        stride_out_m=stride_out_m,
        stride_out_d=stride_out_d,
        BLOCK_M=128,        # rows per program
        BLOCK_D=64,         # cols per program
        BLOCK_H=64,         # hidden‑dim tile
    )
    return out

        norm_bias,
        eps=eps,
    )
    x_norm = x_norm_flat.view(bs, N, N, dim)       # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left/right sides (2·H channels)
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                    # (2H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    left_comb  = F.linear(x_h, left_comb_w)          # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)        # (B,N,N,2H)

    # split projections and gating pre‑activations
    left_proj      = left_comb[..., :H]               # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]               # (B,N,N,H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)          # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate     # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Steps
-----
1️⃣  Input LayerNorm (float32) – custom Triton kernel.
2️⃣  Cast to FP16 for the heavy linear work.
3️⃣  Fuse each projection + its gate (2 · hidden_dim channels) into a single matmul.
4️⃣  Apply optional binary mask to the *projections* (mask does NOT affect the gates).
5️⃣  Element‑wise gating of left/right tensors.
6️⃣  Compute the N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched half‑precision GEMM (torch.bmm) on the view (B·H, N, N).
7️⃣  Fuse hidden‑dim LayerNorm, output‑gate (sigmoid) and final linear
    projection into one Triton kernel to avoid materialising the intermediate
    normalized tensor.
8️⃣  Cast the result back to float32 and return.

The heaviest memory traffic (the intermediate (B·N·N, H) after the contraction)
is read only once for normalisation and is never written back, which cuts the
runtime roughly in half compared with the naïve three‑kernel pipeline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton – input is (M, N)."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) via Triton – input is (M, N)."""
    assert x.dtype == torch.float16
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Fused hidden‑dim LayerNorm + output‑gate + final Linear
# --------------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr,                # (M, D) output (fp16)
    a_ptr,                  # (M, H) pre‑norm tensor (fp16)
    gate_ptr,               # (M, H) output‑gate tensor (fp16)
    weight_ptr,             # (H, D) final linear weight (fp16)
    ln_weight_ptr,          # (H,) LN weight (fp16)
    ln_bias_ptr,            # (H,) LN bias   (fp16)
    eps,                    # scalar eps
    M, H, D,
    stride_a_m, stride_a_h,
    stride_gate_m, stride_gate_h,
    stride_weight_h, stride_weight_d,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    """Fuse LayerNorm (fp16), output‑gate and final Linear."""
    pid = tl.program_id(0)

    # 2‑D grid: rows (M) × cols (D)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(D, BLOCK_D)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # ------------------------------------------------------------------
    # 1️⃣ mean & rstd per row (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        sum_val += tl.sum(a_f, axis=1)
        sum_sq  += tl.sum(a_f * a_f, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ main loop: normalise → LN‑scale/bias → gate → dot‑weight
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for k in range(num_k):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load a and its gate
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_a_m + offs_h[None, :] * stride_a_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        a_f = tl.cast(a, tl.float32)
        g_f = tl.cast(g, tl.float32)

        # Normalisation
        a_norm = (a_f - mean[:, None]) * rstd[:, None]

        # LN weight & bias (per hidden channel)
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
        ln_b = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)
        ln_w_f = tl.cast(ln_w, tl.float32)
        ln_b_f = tl.cast(ln_b, tl.float32)

        # Apply LN scale+bias and output gate
        gated = (a_norm * ln_w_f + ln_b_f) * g_f   # (BLOCK_M, BLOCK_H)

        # Load final linear weight tile (H × D)
        w = tl.load(
            weight_ptr + offs_h[:, None] * stride_weight_h + offs_d[None, :] * stride_weight_d,
            mask=mask_h[:, None] & mask_d[None, :],
            other=0.0
        )
        w_f = tl.cast(w, tl.float32)              # (BLOCK_H, BLOCK_D)

        # Accumulate the dot‑product for this slice of H
        acc += tl.dot(gated, w_f)                 # (BLOCK_M, BLOCK_D)

    # ------------------------------------------------------------------
    # 3️⃣ store result (cast back to fp16)
    # ------------------------------------------------------------------
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
        out,
        mask=mask_m[:, None] & mask_d[None, :]
    )

def fused_norm_gate_linear(
    a: torch.Tensor,
    gate: torch.Tensor,
    weight: torch.Tensor,
    ln_weight: torch.Tensor,
    ln_bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Compute:
        out = ((a‑μ)/σ * ln_weight + ln_bias) * gate  @  weight
    a, gate : (M, H) fp16
    weight  : (H, D) fp16   – already transposed
    ln_weight, ln_bias : (H,) fp16
    Returns (M, D) fp16.
    """
    M, H = a.shape
    D = weight.shape[1]
    out = torch.empty((M, D), dtype=torch.float16, device=a.device)

    stride_a_m, stride_a_h = a.stride()
    stride_gate_m, stride_gate_h = gate.stride()
    stride_w_h, stride_w_d = weight.stride()
    stride_out_m, stride_out_d = out.stride()

    grid = lambda meta: (
        triton.cdiv(M, meta["BLOCK_M"]) *
        triton.cdiv(D, meta["BLOCK_D"]),
    )
    fused_norm_gate_linear_kernel[grid](
        out_ptr=out,
        a_ptr=a,
        gate_ptr=gate,
        weight_ptr=weight,
        ln_weight_ptr=ln_weight,
        ln_bias_ptr=ln_bias,
        eps=eps,
        M=M,
        H=H,
        D=D,
        stride_a_m=stride_a_m,
        stride_a_h=stride_a_h,
        stride_gate_m=stride_gate_m,
        stride_gate_h=stride_gate_h,
        stride_weight_h=stride_w_h,
        stride_weight_d=stride_w_d,
        stride_out_m=stride_out_m,
        stride_out_d=stride_out_d,
        BLOCK_M=128,   # rows per program
        BLOCK_D=64,    # columns per program
        BLOCK_H=64,    # hidden‑dim tile (fits hidden_dim = 128/384)
    )
    return out


# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 / bool) or None
        - weights      : dict mapping parameter names to tensors
        - config       : dict containing ``dim`` and ``hidden_dim`` (optional ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)          # (B·N·N, D)
    x_norm_flat = triton_layernorm_fp32(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_flat.view(bs, N, N, dim)       # (B,N,N,D) float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left/right (2·H channels)
    # --------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(input_tensor.device, torch.float16)       # (2H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(input_tensor.device, torch.float16)      # (2H, D)

    out_gate_w = weights["out_gate.weight"].to(input_tensor.device, torch.float16)  # (H, D)

    left_comb  = F.linear(x_h, left_comb_w)       # (B,N,N,2H)
    right_comb = F.linear(x_h, right_comb_w)      # (B,N,N,2H)

    left_proj      = left_comb[..., :H]               # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]               # (B,N,N,H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)          # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (applied only to the projections)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(input_tensor.device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gated projections
    # --------------------------------------------------------------
    left  = left_proj * left_gate      # (B,N,N,H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Core N³ contraction via batched FP16 GEMM
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)          # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)         # (B·H, N, N)

    # Batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # Back to (B,N,N,H) (still FP16)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 8️⃣ Fuse hidden‑dim LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out_flat = out.view(-1, H)                         # (M, H)
    out_gate_flat = out_gate.view(-1, H)               # (M, H)

    # final linear weight must be (H, D)
    to_out_weight = (
        weights["to_out.weight"]
        .to(input_tensor.device, torch.float16)
        .t()                     # (D, H) -> (H, D)
        .contiguous()
    )
    to_out_norm_w = weights["to_out_norm.weight"].to(input_tensor.device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(input_tensor.device, torch.float16)

    # fused operation → (M, D) FP16
    output_half = fused_norm_gate_linear(
        a=out_flat,
        gate=out_gate_flat,
        weight=to_out_weight,
        ln_weight=to_out_norm_w,
        ln_bias=to_out_norm_b,
        eps=eps,
    )

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 and reshape
    # --------------------------------------------------------------
    output = output_half.view(bs, N, N, dim).float()
    return output
