"""
High‑performance outgoing TriMul (Tensor‑wise multiplicative update) forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32 reduction → FP16 output) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM.
3️⃣ Apply the optional binary mask (broadcast over the hidden dimension) to the
   left/right raw projections.
4️⃣ Apply sigmoid gating (still in FP16) and combine with the projected tensors.
5️⃣ Core contraction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   performed with a single `torch.einsum`, which internally maps to a batched
   FP16 GEMM (Tensor‑Core friendly) without extra data copies.
6️⃣ Output LayerNorm (FP32 reduction → FP16 output) – reuse the Triton kernel.
7️⃣ Apply the output gate and the final half‑precision linear projection back to
   the model dimension.
8️⃣ Cast the result to float32 (the type expected by downstream modules).

All heavy arithmetic runs in FP16, while both LayerNorms keep full‑precision (FP32)
statistics for numerical stability.  The kernel written in Triton is the
LayerNorm implementation; the remaining operations are delegated to highly‑
optimized PyTorch kernels (cuBLAS, cuDNN, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # *ptr to input  [M, C] (float32 or float16)
    y_ptr,                     # *ptr to output [M, C] (float16)
    weight_ptr, bias_ptr,      # *ptr to LN weight/bias [C] (float16)
    M, C, eps,                 # dimensions
    stride_xm, stride_xc,      # strides of input
    stride_ym, stride_yc,      # strides of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offset for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        # Load a tile of the input (any dtype) and cast to FP32 for reduction
        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform (output in FP16)
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        # Normalisation (FP32)
        norm = (x - mean[:, None]) * rstd[:, None]

        # Load affine parameters (FP16) and promote to FP32 for computation
        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_c,   mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w + b                     # still FP32
        y = y.to(tl.float16)                 # cast to half for storage

        tl.store(
            y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc),
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm_fp16(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 256,
    BLOCK_C: int = 128,
) -> torch.Tensor:
    """Layer‑norm over the last dimension of a 2‑D tensor (M×C) using Triton.
    Input may be FP32 or FP16; output is FP16."""
    assert x.dim() == 2, "triton_layernorm_fp16 expects a 2‑D tensor"
    M, C = x.shape

    y = torch.empty((M, C), dtype=weight.dtype, device=x.device)

    # Ensure contiguous layout for correct stride computation
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)
    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=8,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Outgoing TriMul entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast LayerNorm parameters to FP16 once (weights are tiny)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 reduction → FP16 output)
    # ------------------------------------------------------------------
    # Collapse batch × seq_len × seq_len to a 2‑D matrix for the kernel
    x_flat = input_tensor.reshape(-1, dim).contiguous()           # (B·N·N, dim) FP32
    x_norm = triton_layernorm_fp16(
        x_flat,
        norm_weight,
        norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                                             # (B·N·N, dim) FP16
    # Restore the 4‑D view
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)                # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Stack the five weight matrices (no bias) -> (5·H, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                             # (5·H, dim)

    # Linear projection of all positions in one shot
    proj = F.linear(x_norm.view(-1, dim), combined_weight)        # (B·N·N, 5·H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split back into the individual tensors (views, no copy)
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to half and add a trailing singleton dim
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)      # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)     # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)       # (B,N,N,H) FP16

    left  = left_raw * left_gate   # (B,N,N,H) FP16
    right = right_raw * right_gate # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction – fused tensor‑wise multiplication
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented via a single torch.einsum (maps to batched GEMM)
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 reduction → FP16 output)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out_flat = out.view(-1, hidden_dim)                     # (B·N·N, H)
    out_norm = triton_layernorm_fp16(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                                        # (B·N·N, H) FP16
    out_norm = out_norm.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                                 # (B,N,N,H) FP16

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_half = F.linear(gated.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim) FP16
    final = final_half.view(bs, seq_len, seq_len, dim).float()        # cast to FP32

    return final
