@triton.jit
def hidden_norm_linear_kernel(
    out_ptr, out_gate_ptr,
    gamma_ptr, beta_ptr,
    weight_ptr,
    result_ptr,
    M, H, D,
    eps: tl.constexpr,
    stride_out_mh: tl.constexpr, stride_out_h: tl.constexpr,
    stride_gate_mh: tl.constexpr, stride_gate_h: tl.constexpr,
    stride_gamma_h: tl.constexpr, stride_beta_h: tl.constexpr,
    stride_weight_dh: tl.constexpr, stride_weight_h: tl.constexpr,
    stride_res_m: tl.constexpr, stride_res_d: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # compute mean and variance across H for each row
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # First pass across H
    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        val = tl.load(out_ptr + offs_m[:, None] * stride_out_mh + offs_h[None, :] * stride_out_h,
                      mask=mask_m[:, None] & mask_h[None, :],
                      other=0.0)
        val_f = tl.cast(val, tl.float32)
        sum_val += tl.sum(val_f, axis=1)
        sum_sq  += tl.sum(val_f * val_f, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # accumulator for result dims
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # second pass across H: compute gated normalized and accumulate
    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and out_gate
        out_val = tl.load(out_ptr + offs_m[:, None] * stride_out_mh + offs_h[None, :] * stride_out_h,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0)
        out_f = tl.cast(out_val, tl.float32)

        gate_val = tl.load(out_gate_ptr + offs_m[:, None] * stride_gate_mh + offs_h[None, :] * stride_gate_h,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        gate_f = tl.cast(gate_val, tl.float32)

        # normalize out_f
        norm = (out_f - mean[:, None]) * rstd[:, None]

        # apply gamma, beta
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # shape (BLOCK_H)
        beta  = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)
        norm = norm * gamma[None, :] + beta[None, :]

        gated = norm * gate_f

        # Contribution to output dims, loop over dim blocks
        for d0 in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # load weight matrix slice (dim, hidden) for this dim block and hidden block
            # weight shape (D, H); want slice of shape (BLOCK_D, BLOCK_H)
            w = tl.load(weight_ptr + offs_d[:, None] * stride_weight_dh + offs_h[None, :] * stride_weight_h,
                        mask=mask_d[:, None] & mask_h[None, :],
                        other=0.0)
            # weight slice shape (BLOCK_D, BLOCK_H)
            # compute dot: gated (BLOCK_M, BLOCK_H) @ w.T (BLOCK_H, BLOCK_D) -> (BLOCK_M, BLOCK_D)
            acc += tl.dot(gated, w.T)

    # store result
    # result pointer shape (M, D), stride_res_m = D, stride_res_d = 1
    tl.store(result_ptr + offs_m[:, None] * stride_res_m + offs_d[None, :] * stride_res_d,
             acc,
             mask=mask_m[:, None] & mask_d[None, :])
```

        stride_beta_h=stride_bias_h,
        stride_weight_dh=stride_weight_dh,
        stride_weight_h=stride_weight_h,
        stride_res_m=stride_res_m,

"""
TriMul “outgoing” forward – Triton‑accelerated.

Key ideas
---------
1️⃣  Input LayerNorm (FP32) – custom Triton kernel.
2️⃣  Three linear projections (left, right, out‑gate) are fused in a single
    `torch.nn.functional.linear` (FP16) to keep the heavy GEMM on Tensor‑Cores.
3️⃣  Masking and sigmoid gating are applied element‑wise on the projected tensors.
4️⃣  The N³ contraction `out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]` is performed
    as an efficient batched FP16 GEMM (`torch.bmm`).
5️⃣  **Hidden‑dim LayerNorm, gating, and the final linear projection are fused
    into one Triton kernel**.  This removes the intermediate normalized tensor
    and halves the memory traffic of the final stage.
6️⃣  The final output is cast back to float32.

The implementation satisfies the required API (`custom_kernel(data)`) and
uses Triton for the two LayerNorms and the fused hidden‑norm‑linear kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32 (same as the reference)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -------- second pass: write normalized ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton.
    Input/Output shape: (M, N) where N is the feature dimension.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Hidden‑dim LayerNorm + output‑gate + final linear (fused)
# --------------------------------------------------------------------------
@triton.jit
def hidden_norm_linear_kernel(
    out_ptr, out_gate_ptr,
    gamma_ptr, beta_ptr,
    weight_ptr,
    result_ptr,
    M, H, D,
    eps: tl.constexpr,
    stride_out_mh: tl.constexpr, stride_out_h: tl.constexpr,
    stride_gate_mh: tl.constexpr, stride_gate_h: tl.constexpr,
    stride_gamma_h: tl.constexpr, stride_beta_h: tl.constexpr,
    stride_weight_dh: tl.constexpr, stride_weight_h: tl.constexpr,
    stride_res_m: tl.constexpr, stride_res_d: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Fuse hidden‑dim LN, out‑gate multiplication and the final linear projection.

    Shapes (all in contiguous layout):
        out      : (M, H)   FP16   – result of the N³ contraction
        out_gate : (M, H)   FP16   – gate produced from the input
        gamma    : (H,)     FP16   – LayerNorm weight
        beta     : (H,)     FP16   – LayerNorm bias
        weight   : (D, H)   FP16   – final projection weight
        result   : (M, D)   FP32   – final Tensor (written)

    The kernel processes BLOCK_M rows at a time, iterating over the hidden
    dimension in blocks of size BLOCK_H and over the output dimension in
    blocks of size BLOCK_D.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row index for the batch (M = B * N * N)
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance over the hidden dimension H
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        v = tl.load(
            out_ptr + offs_m[:, None] * stride_out_mh + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        vf = tl.cast(v, tl.float32)
        sum_val += tl.sum(vf, axis=1)
        sum_sq  += tl.sum(vf * vf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Accumulate final results (M × D)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load values from the contraction and the gate
        v = tl.load(
            out_ptr + offs_m[:, None] * stride_out_mh + offs_h[None, :] * stride_out_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        vf = tl.cast(v, tl.float32)

        g = tl.load(
            out_gate_ptr + offs_m[:, None] * stride_gate_mh + offs_h[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        gf = tl.cast(g, tl.float32)

        # -------- hidden LayerNorm ----------
        norm = (vf - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)
        gamma_f = tl.cast(gamma, tl.float32)
        beta_f  = tl.cast(beta,  tl.float32)

        norm = norm * gamma_f[None, :] + beta_f[None, :]

        # -------- apply output gate ----------
        gated = norm * gf   # (BLOCK_M, BLOCK_H)

        # -------- final linear projection ----------
        for d0 in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # weight slice: (BLOCK_D, BLOCK_H)
            w = tl.load(
                weight_ptr + offs_d[:, None] * stride_weight_dh + offs_h[None, :] * stride_weight_h,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0
            )
            wf = tl.cast(w, tl.float32)          # now (BLOCK_D, BLOCK_H)

            # (BLOCK_M, BLOCK_H) @ (BLOCK_H, BLOCK_D)  -> (BLOCK_M, BLOCK_D)
            acc += tl.dot(gated, wf.T)

    # ------------------------------------------------------------------
    # 3️⃣  Store the result
    # ------------------------------------------------------------------
    tl.store(
        result_ptr + offs_m[:, None] * stride_res_m + offs_d[None, :] * stride_res_d,
        acc,
        mask=mask_m[:, None] & mask_d[None, :]
    )


def triton_hidden_norm_linear(out: torch.Tensor,
                              out_gate: torch.Tensor,
                              gamma: torch.Tensor,
                              beta: torch.Tensor,
                              weight: torch.Tensor,
                              eps: float = 1e-5) -> torch.Tensor:
    """
    Fuse hidden‑dim LayerNorm, multiplication with the output gate, and the
    final linear projection (out) → (B, N, N, D).

    Parameters
    ----------
    out, out_gate : (B·N·N, H) FP16 tensors (contiguous)
    gamma, beta   : (H,) FP16 tensors (contiguous)
    weight        : (D, H) FP16 tensor (contiguous)
    eps           : epsilon for LayerNorm

    Returns
    -------
    Tensor of shape (B·N·N, D) with dtype torch.float32
    """
    M, H = out.shape
    D = weight.shape[0]

    result = torch.empty((M, D), device=out.device, dtype=torch.float32)

    # strides of the contiguous tensors
    stride_out_mh, stride_out_h       = out.stride()
    stride_gate_mh, stride_gate_h     = out_gate.stride()
    stride_gamma_h = gamma.stride(0)   # =1
    stride_beta_h  = beta.stride(0)    # =1
    stride_weight_dh, stride_weight_h = weight.stride()
    stride_res_m, stride_res_d = result.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    hidden_norm_linear_kernel[grid](
        out_ptr=out,
        out_gate_ptr=out_gate,
        gamma_ptr=gamma,
        beta_ptr=beta,
        weight_ptr=weight,
        result_ptr=result,
        M=M,
        H=H,
        D=D,
        eps=eps,
        stride_out_mh=stride_out_mh,
        stride_out_h=stride_out_h,
        stride_gate_mh=stride_gate_mh,
        stride_gate_h=stride_gate_h,
        stride_gamma_h=stride_gamma_h,
        stride_beta_h=stride_beta_h,
        stride_weight_dh=stride_weight_dh,
        stride_weight_h=stride_weight_h,
        stride_res_m=stride_res_m,
        stride_res_d=stride_res_d,
        BLOCK_M=128,
        BLOCK_H=64,
        BLOCK_D=64,
    )
    return result


# --------------------------------------------------------------------------
# Full TriMul forward pass (outgoing version)
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (float32 / bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys:
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype            # expected torch.float32

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input tensor must be square in the sequence dimensions"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    M = bs * N * N
    # flatten to (M, D)
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                # (B, N, N, D)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the three linear projections (left, right, out‑gate)
    # ------------------------------------------------------------------
    # (each weight: (H, D) – bias=False in the reference)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device, torch.float16).contiguous()        # (5*H, D)

    # (B,N,N,D) @ (D,5*H)^T  →  (B,N,N,5*H)   (still FP16)
    fused = F.linear(x_f16, fused_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Split the five heads
    # ------------------------------------------------------------------
    left_proj      = fused[..., 0               : H]   # (B,N,N,H)
    left_gate_pre  = fused[..., H               : 2 * H]
    right_proj     = fused[..., 2 * H             : 3 * H]
    right_gate_pre = fused[..., 3 * H             : 4 * H]
    out_gate_pre   = fused[..., 4 * H             : 5 * H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 7️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate               # (B,N,N,H)
    right = right_proj * right_gate             # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ contraction (batched FP16 GEMM)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)               # (B·H, N, N)

    # out_mat = left_mat @ right_matᵀ   → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H) – this copy is cheap compared to the
    # huge GEMM and we need a contiguous layout for the next kernel.
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Fuse hidden LayerNorm, output‑gate multiplication and final projection
    # ------------------------------------------------------------------
    # flatten to (M, H) and (M, D) for the fused kernel
    out_flat = out.view(-1, H).contiguous()                     # (M, H), FP16
    out_gate_flat = out_gate.view(-1, H).contiguous()          # (M, H), FP16

    # hidden‑dim LayerNorm parameters (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    # final projection weight (FP16) – shape (D, H)
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()

    # fused kernel → result (M, D) float32
    output_flat = triton_hidden_norm_linear(
        out=out_flat,
        out_gate=out_gate_flat,
        gamma=to_out_norm_w,
        beta=to_out_norm_b,
        weight=to_out_w,
        eps=eps,
    )

    # reshape to the required output shape (B, N, N, D) – already FP32
    output = output_flat.view(bs, N, N, dim)
    return output
