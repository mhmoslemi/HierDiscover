```
@triton.jit
def trimul_outgoing_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0) # combined batch*hidden
    pid1 = tl.program_id(1) # i block
    pid2 = tl.program_id(2) # j block

    # decode b,h
    h = pid0 % H
    b = pid0 // H

    i_start = pid1 * BLOCK_I
    j_start = pid2 * BLOCK_J

    # offsets using arange
    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)
    # mask for i and j boundaries
    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float16)

    # number of k tiles
    num_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_tiles):
        k_start = tk * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # load left block: (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_off[:, None] * stride_left_i + \
            k_off[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0)

        # load right block: (k,j) transposed (k as leading dim)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_off[:, None] * stride_right_k + \
            j_off[None, :] * stride_right_j + \
            h * stride_right_h
        right = tl.load(right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0)

        # multiply-accumulate
        acc += tl.dot(left, right)
    
    # store result to out
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_off[:, None] * stride_out_i + \
        j_off[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :])
```

"""
TriMul outgoing operator – Triton-accelerated implementation.

The kernel follows the reference PyTorch version:
1️⃣  Layer‑Norm on the input tensor (B,N,N,D) – Triton 2‑pass kernel,
    FP32 statistics, FP16 result.
2️⃣  Fuse the five linear layers (proj‑left/right + three gates) into a
    single half‑precision GEMM (shape (B·N·N, 5·H)).
3️⃣  Split the fused result, apply sigmoid to the three gate tensors and
    optionally mask the gated left/right tensors.
4️⃣  Heavy O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is performed by a custom Triton kernel that computes a batched
    matrix‑multiply tile‑wise (i‑tile × j‑tile) using `tl.dot`.  The kernel
    iterates over the k‑dimension in blocks, accumulates in FP16 (Tensor‑Core
    friendly) and writes the (B,N,N,H) result directly to GPU memory – no
    extra copies or PyTorch `bmm` calls.
5️⃣  Layer‑Norm of the intermediate tensor (H‑dim) – Triton kernel.
6️⃣  Apply the output gate, final FP16 linear projection back to the model
    dimension D and cast the answer to FP32 (public API expectation).

All heavy linear algebra (the two LayerNorms and the O(N³) contraction) is
implemented in Triton while the cheap GEMM‑style operations stay in cuBLAS.
The implementation works for any batch size B∈{1,2}, sequence length N∈{128…1024},
model dimension D∈{128,384,768} and hidden dimension H=hidden_dim (typically 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension). Supports any dtype (we use
# fp16 for the output but compute mean/var in fp32 for stability).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # number of rows   (B·N·N)
    N,                # feature dimension (D or H)
    eps,              # epsilon
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- row offsets for this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute sum and sum‑sq (FP32) ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine -------------------------------
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (weight/bias may be fp16/fp32 – cast to fp32)
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Triton LayerNorm (2‑pass) for a 2‑D tensor (M,N)."""
    assert x.dim() == 2, "[triton_layernorm] expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for D/H = 128/384/768
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) contraction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,                           # *ptr left  [B, N, N, H] (fp16)
    right_ptr,                          # *ptr right [B, N, N, H] (fp16)
    out_ptr,                            # *ptr out   [B, N, N, H] (fp16)
    B, N, H,                            # scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,              # tile size for i dimension
    BLOCK_J: tl.constexpr,              # tile size for j dimension
    BLOCK_K: tl.constexpr,              # tile size for k dimension
):
    """Computes   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]"""
    pid0 = tl.program_id(0)   # encodes (b, h)
    pid_i = tl.program_id(1)  # i‑tile index
    pid_j = tl.program_id(2)  # j‑tile index

    # Decode batch and hidden index
    h = pid0 % H
    b = pid0 // H

    # Tile start positions
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Offsets within the N‑axis
    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # Accumulator – keep in fp16 to let Tensor Cores stay in the fast path
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float16)

    # Number of k‑tiles
    n_tiles_k = tl.cdiv(N, BLOCK_K)
    for tk in range(n_tiles_k):
        k_start = tk * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ---- left block: (i, k) ------------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_off[:, None] * stride_left_i + \
            k_off[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )                               # shape (BLOCK_I, BLOCK_K)

        # ---- right block: (k, j)  (note k is leading dimension) ----
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_off[:, None] * stride_right_k + \
            j_off[None, :] * stride_right_j + \
            h * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )                               # shape (BLOCK_K, BLOCK_J)

        # Multiply‑accumulate (Tensor‑Core friendly)
        acc += tl.dot(left, right)     # (BLOCK_I, BLOCK_J)

    # ---- store -------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_off[:, None] * stride_out_i + \
        j_off[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, D] (float32)
        mask         : torch.Tensor  [B, N, N] (float32) or None
        weights      : dict of torch.Tensors (model parameters)
        config       : dict containing "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype        # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – flatten to 2‑D for the kernel
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len               # rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)    # (M,D) FP32
    norm_w = weights["norm.weight"].to(device, dtype)   # (D)
    norm_b = weights["norm.bias"].to(device, dtype)     # (D)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)  # (M,D) FP32
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)            # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Convert to fp16 – the heavy work runs in half precision
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm.to(torch.float16)                # (B,N,N,D) fp16
    x_norm_fp16_flat = x_norm_fp16.reshape(M, dim)        # (M,D) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + three gates) into a
    #    single GEMM (M, 5·H) – fast cuBLAS/torch implementation.
    # ------------------------------------------------------------------
    # Extract weight tensors (already on device)
    left_proj_w   = weights["left_proj.weight"]          # (H, D)
    right_proj_w  = weights["right_proj.weight"]         # (H, D)
    left_gate_w   = weights["left_gate.weight"]          # (H, D)
    right_gate_w  = weights["right_gate.weight"]         # (H, D)
    out_gate_w    = weights["out_gate.weight"]           # (H, D)

    # Concatenate along the output‑feature dimension
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                        # (5·H, D)

    # Single fused linear projection
    proj_flat = F.linear(x_norm_fp16_flat, combined_w)   # (M, 5·H) fp16
    proj = proj_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)   # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 4️⃣ Split, apply sigmoid gates and optional mask
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = proj[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj[..., H:2*H]
    right_proj  = proj[..., 2*H:3*H]
    right_gate_r= proj[..., 3*H:4*H]
    out_gate_r  = proj[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)        # (B,N,N,H) fp16
    right_gate  = torch.sigmoid(right_gate_r)       # (B,N,N,H) fp16
    out_gate    = torch.sigmoid(out_gate_r)         # (B,N,N,H) fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ O(N³·H) contraction with custom Triton kernel
    # ------------------------------------------------------------------
    # Allocate output tensor (fp16) – will hold the intermediate result
    out_inter = torch.empty_like(left)   # (B,N,N,H), fp16

    # Strides for left/right tensors (contiguous)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out_inter.stride()

    # Grid: (B*H, ceil(N/BI), ceil(N/BJ))
    BLOCK_I = 32    # tile size for the i‑axis
    BLOCK_J = 32    # tile size for the j‑axis
    BLOCK_K = 32    # tile size for the reduction axis

    grid = (
        bs * hidden_dim,
        triton.cdiv(seq_len, BLOCK_I),
        triton.cdiv(seq_len, BLOCK_J),
    )

    trimul_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_inter,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_left_h=stride_l_h,
        stride_right_b=stride_r_b,
        stride_right_j=stride_r_i,   # j dimension has same stride as i in the tensor layout
        stride_right_k=stride_r_k,
        stride_right_h=stride_r_h,
        stride_out_b=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_k,    # same note as above for the j‑axis
        stride_out_h=stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # 6️⃣ Output LayerNorm (H‑dim) – Triton kernel, stays in fp16
    # ------------------------------------------------------------------
    out_flat = out_inter.view(M, hidden_dim)            # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)  # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)    # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection and cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                           # (B,N,N,D) fp16
    out_fp32 = out_fp16.to(torch.float32)                          # API expects FP32

    return out_fp32
```
