@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, L, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_id = pid // (tl.cdiv(L, BLOCK_I) * tl.cdiv(L, BLOCK_J))
    block_i = (pid // (tl.cdiv(L, BLOCK_J))) % tl.cdiv(L, BLOCK_I)
    block_j = pid % tl.cdiv(L, BLOCK_J)

    # compute offsets
    i_offsets = block_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = block_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # allocate accumulator for each i,j,h block:
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k blocks
    for k_start in range(0, L, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)

        # load left: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left = tl.load(left_ptr + batch_id*stride_left_b
                                   + i_offsets[:,None,None]*stride_left_i
                                   + k_offsets[None,:,None]*stride_left_k
                                   + tl.arange(0, BLOCK_H)[None,None,:]*stride_left_h,
                       mask=(i_offsets[:,None,None] < L) & (k_offsets[None,:,None] < L) & (tl.arange(0, BLOCK_H)[None,None,:] < H),
                       other=0.0)
        # load right: shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right = tl.load(right_ptr + batch_id*stride_right_b
                                    + j_offsets[:,None,None]*stride_right_j
                                    + k_offsets[None,:,None]*stride_right_k
                                    + tl.arange(0, BLOCK_H)[None,None,:]*stride_right_h,
                        mask=(j_offsets[:,None,None] < L) & (k_offsets[None,:,None] < L) & (tl.arange(0, BLOCK_H)[None,None,:] < H),
                        other=0.0)
        # compute product sum over k: need to broadcast: left shape (I,K,H), right shape (J,K,H)
        # compute sum over k of left[i,k,h] * right[j,k,h] = sum_k left[i,k,:] * right[j,k,:] elementwise across H.

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
============================================================

This implementation follows the reference PyTorch `TriMul` module but
optimises the heavy parts:

* A custom row‑wise LayerNorm kernel (Triton) replaces the PyTorch
  `nn.LayerNorm`.  The kernel works on a 2‑D view `[M, C]` where
  `M = B * L * L` and `C = dim` or `hidden_dim`.  Statistics are
  accumulated in fp32 for numerical stability.

* The four linear maps (`left_proj`, `right_proj`, `left_gate`,
  `right_gate`) are fused into a single `F.linear` call, saving two
  matrix multiplies.

* The N³ contraction  
  `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`  
  is performed by a single batched GEMM (`torch.matmul`).  The
  tensor is only permuted – no intermediate `.contiguous()` copies are
  made – therefore the operation runs on native Tensor‑core GEMM
  kernels without extra memory traffic.

* All heavy arithmetic is done in fp16; reductions (LayerNorm) use fp32.
  The final result is cast back to float32 as required.

The function `custom_kernel` is the entry‑point used by the test harness.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise, works on a 2‑D tensor `[M, C]`)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                    # *ptr to input / output   [M, C]
    weight_ptr, bias_ptr,            # *ptr to affine parameters [C]
    M, C,                            # rows, channels
    eps,                             # epsilon
    stride_xm, stride_xc,            # input strides (row, col)
    stride_ym, stride_yc,            # output strides (row, col)
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_C: tl.constexpr,           # channels per inner loop
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row offsets for this program
    # --------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass – compute mean & variance (in FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass – write normalised + affine transformed values
    # --------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented with a Triton kernel.
    Expected input shape is 2‑D `[M, C]`.  Returned tensor has the same
    shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    # Ensure tensors are contiguous (required for correct stride parsing)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    # Launch – choose block sizes that cover the typical channel sizes (≤384)
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=256,          # rows per program (tuneable)
        BLOCK_C=64,           # channels per inner loop
    )
    return y


# ----------------------------------------------------------------------
# Main entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension `dim`
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)                         # [M, dim]

    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)                      # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj + gate) into a single matmul
    # ------------------------------------------------------------------
    #   [left_proj, right_proj, left_gate, right_gate]  -> (4*H, D)
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)          # (4*H, D)

    # Linear projection (all four heads at once)
    fused_out = F.linear(x_norm_h.view(M, dim), fused_w)          # (M, 4*H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # Slice the four pieces
    left_proj  = fused_out[..., :hidden_dim]                                 # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate  = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) before gating – in‑place to save memory
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj.mul_(mask_h)
        right_proj.mul_(mask_h)

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (in‑place) – now we have `left` and `right`
    # ------------------------------------------------------------------
    left = left_proj
    left.mul_(left_gate)                # (B, L, L, H)  fp16
    right = right_proj
    right.mul_(right_gate)              # (B, L, L, H)

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a single `torch.matmul` over a view that avoids
    #    any `.contiguous()` copies.
    # ------------------------------------------------------------------
    #   left  : (B, L, L, H) -> (B, H, L, L)   (no copy, just a view)
    #   right : (B, L, L, H) -> (B, H, L, L)   (no copy)
    left_mat = left.permute(0, 3, 1, 2)          # (B, H, L, L)  non‑contiguous view
    right_mat = right.permute(0, 3, 1, 2)        # (B, H, L, L)

    #   Transpose the inner dimensions of `right` → (B, H, L, L) where
    #   the third dimension becomes the reduction dimension.
    right_mat_t = right_mat.transpose(2, 3)     # (B, H, L, L)

    # Batched matrix multiplication – one GEMM per head
    out_mat = torch.matmul(left_mat, right_mat_t)  # (B, H, L, L)

    # Bring the tensor back to the original layout (B, L, L, H)
    out = out_mat.permute(0, 2, 3, 1)           # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (Triton)
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)

    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)    # (B, L, L, H)

    # ------------------------------------------------------------------
    # 8️⃣ Output gating – linear over the *original* normalised input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h.view(M, dim), out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)        # (B, L, L, H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply gating and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm.mul_(out_gate)                               # (B, L, L, H)  in‑place

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(out_norm, to_out_w)            # (B, L, L, D)  fp16

    # Cast back to float32 as required by the surrounding model
    out_final = out_final_h.to(torch.float32)

    return out_final
