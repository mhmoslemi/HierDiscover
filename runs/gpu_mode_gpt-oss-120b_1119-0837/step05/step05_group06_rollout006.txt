@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh_idx = tl.program_id(2)
    b = bh_idx // H
    h = bh_idx % H

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # Offsets base pointer for this batch and head
    offset_bh = b * stride_left_b + h * stride_left_h

    # Initialize accumulator as float32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        # k offsets
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile
        left_ptrs = left_ptr + offset_bh \
            + i_offset[:, None] * stride_left_i \
            + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = tl.cast(left, tl.float32)

        # Load right tile transposed (k x j)
        right_ptrs = right_ptr + offset_bh \
            + k_offset[:, None] * stride_right_k \
            + j_offset[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = tl.cast(right, tl.float32)

        # Accumulate
        acc += tl.dot(left, right)

    # Store output
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
        + i_offset[:, None] * stride_out_i \
        + j_offset[None, :] * stride_out_j
    out_val = acc.to(tl.float16)
    tl.store(out_ptrs, out_val, mask=mask_i[:, None] & mask_j[None, :])
```

We need to pass stride_right_j (col stride) and stride_right_k (row stride). Right pointer offset uses k as leading dim, j as second dim. Already considered.

def trimul_gemm(left, right):
    # left: torch.Tensor of shape (B,N,N,H) dtype fp16
    # right: same.
    B,N,N2,H = left.shape
    assert N == N2.
    out = torch.empty_like(left) # shape (B,N,N,H)
    # compute strides
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # note: we exported stride_right_j and stride_right_k; they are from right's shape.
    # allocate grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,

"""
TriMul “outgoing’’ forward pass – Triton‑accelerated.

The implementation follows the reference PyTorch module but fuses the
most expensive pieces:

1️⃣  Input LayerNorm (2‑pass Triton kernel, FP32 mean/var, FP16 output).
2️⃣  Cast to FP16 and fuse the five linear layers (two projections + three
    gates) into a single GEMM (`torch.nn.functional.linear`).
3️⃣  Apply the sigmoid gates and the optional pair‑wise mask.
4️⃣  Compute the heavy tensor contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    with a custom Triton kernel that works directly on the
    (B,N,N,H) layout.  This eliminates the expensive `permute(...).contiguous()`
    copies required by a naïve `torch.bmm` approach.
5️⃣  Output LayerNorm (Triton, FP16).
6️⃣  Apply the output gate, final linear projection back to the model
    dimension and cast the result to FP32 (the public API type).

All kernels use half‑precision arithmetic where possible and accumulate in
FP32 for numerical stability.  The heavy O(N³·H) contraction now runs
entirely on‑device without extra memory copies, yielding a large speed‑up
over the previous implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension).  Works for any dtype.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # rows   (B·N·N)
    N,                # feature dim (D or H)
    eps,              # epsilon
    stride_xm, stride_xn,      # strides of x
    stride_ym, stride_yn,      # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row offsets for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute sum and sum‑sq (FP32) ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine -------------------------------
    for n_iter in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = tl.cast(x, tl.float32)
        norm_x = (xf - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Apply the 2‑pass LayerNorm kernel to a 2‑D tensor (M,N)."""
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make everything contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # block sizes chosen to work for D∈{128,384,768} and H=128
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) contraction
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of left (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides of right (B,N,N,H)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides of out (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid axes:
    pid_i = tl.program_id(0)               # block row   (i‑dim)
    pid_j = tl.program_id(1)               # block col   (j‑dim)
    pid_bh = tl.program_id(2)              # combined batch * head

    # decode batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets inside the N×N matrix
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # base offset for the selected (b,h)
    base_offset = b * stride_left_b + h * stride_left_h

    # accumulator (FP32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the shared dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_base = kk * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile: indices (i,k)
        left_ptrs = left_ptr + base_offset \
            + i_offset[:, None] * stride_left_i \
            + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        left = tl.cast(left, tl.float32)

        # right tile transposed: we need right[k,j] (k as row, j as col)
        right_ptrs = right_ptr + base_offset \
            + k_offset[:, None] * stride_right_k \
            + j_offset[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)
        right = tl.cast(right, tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) += (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write out the tile
    out_ptrs = out_ptr + base_offset \
        + i_offset[:, None] * stride_out_i \
        + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction for the TriMul operator.
    `left` and `right` must be fp16 tensors of shape (B,N,N,H) and be
    contiguous before calling this function.
    Returns a fp16 tensor of shape (B,N,N,H).
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # allocate output
    out = torch.empty_like(left)

    # make sure buffers are contiguous (required for correct strides)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – 64 works well for the tested N up to 1024 on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # (batch, head) tiles
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Triton‑accelerated TriMul “outgoing’’ entry point

    """
    Forward pass of the TriMul “outgoing’’ operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]  (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor  [B, N, N, D] (float32)
    """

    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                     # (M, D) float32
    norm_w = weights["norm.weight"].to(device, dtype)       # (D)
    norm_b = weights["norm.bias"].to(device, dtype)         # (D)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)     # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projections + three gates into a single matmul
    # ------------------------------------------------------------------
    # Linear weights (each of shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate on the output‑feature axis → shape (5*H, D)
    fused_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)  # (5*H, D)

    proj_gate_flat = F.linear(x_norm_f16.view(M, dim), fused_w)   # (M,5*H) fp16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)

    left_proj   = proj_gate[..., 0:H]
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and form the gated tensors
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity before feeding into the custom kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – custom Triton kernel (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = trimul_gemm(left, right)          # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)      # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)   # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)     # (H)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w,
                                     to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Gated output → final linear projection → FP32 cast
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                       # (B,N,N,H) fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    out_fp16 = F.linear(gated, to_out_w)             # (B,N,N,D) fp16
    out_fp32 = out_fp16.to(torch.float32)            # final FP32 result

    return out_fp32
