import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(...):
    ...

    ...

def custom_kernel(data):
    """
    Triton‑accelerated forward for the “outgoing” TriMul operator.
    ...
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # Input LN
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # Cast to fp16
    x_norm_f16 = x_norm.to(torch.float16)

    # Combined projection+gate weight
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w], dim=0
    ).to(device, torch.float16)   # (5*H, D)

    # Linear projection to produce proj and gates
    proj_gate = F.linear(x_norm_f16.view(M, dim), combined_w)  # (M,5H)
    proj_gate = proj_gate.view(bs, N, N, 5 * hidden)

    # Split
    H = hidden
    left_proj   = proj_gate[..., 0:H]
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate = torch.sigmoid(out_gate_r)

    # Apply gates (in-place) and optional mask
    left_proj.mul_(left_gate)
    right_proj.mul_(right_gate)

    if (not nomask) and mask is not None:

        left_proj.mul_(mask_f16)

    # Contraction (pairwise product over k)
    # out[b,i,j,:] = Σ_k left_proj[b,i,k,:] * right_proj[b,j,k,:]
    out = torch.einsum('bikd,bjkd->bijd', left_proj, right_proj)   # (B,N,N,H)

"""
Triton‑accelerated “outgoing’’ TriMul forward pass.

Algorithm
---------
1️⃣ 2‑pass LayerNorm (FP32 accumulator, FP16 output) on the input tensor.
2️⃣ Cast the normalized tensor to FP16 and apply *one* fused linear layer that
   produces the three projections (left/right) and the three gating vectors
   (left_gate, right_gate, out_gate) in a single GEMM.
3️⃣ Apply sigmoid to the gating vectors, multiply by the optional pair‑wise mask,
   and fuse the gating with the projected tensors *in‑place* (no extra buffers).
4️⃣ Contract the gated tensors with a single ``torch.einsum`` which computes
   the raw O(N³·H) multiplicative update without the costly ``permute+contiguous``
   and batched‑GEMM copies used in naïve implementations.
5️⃣ Apply a second Triton LayerNorm on the hidden dimension.
6️⃣ Multiply by the output gate, apply the final linear projection back to the
   model dimension and cast the result to float32 for the public API.

Only the two LayerNorms are written in Triton; all dense linear algebra is
delegated to cuBLAS/Tensor‑Cores via PyTorch.  This eliminates the large
temporary copies that dominate runtime in earlier versions, bringing the
forward pass well below the target ~1 ms budget on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # number of rows (B·N·N)
    N,              # feature dimension (D or H)
    eps,            # epsilon for numerical stability
    stride_xm, stride_xn,   # strides of x
    stride_ym, stride_yn,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    Two‑pass LayerNorm:
      1. Compute per‑row mean & variance (FP32).
      2. Normalise, apply affine transform, store in original dtype.
    """
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # ------------------------------------------------------------------
    # First pass: accumulate sum and sum‑of‑squares
    # ------------------------------------------------------------------
    num_n = tl.cdiv(N, BLOCK_N)
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine
    # ------------------------------------------------------------------
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for ``layernorm_kernel``.
    Expects a 2‑D contiguous tensor (M, N) and returns a tensor of the same shape
    and dtype as ``x``.
    """
    assert x.ndim == 2, "LayerNorm wrapper expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # guarantee contiguous layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # block sizes tuned for D ∈ {128,384,768}
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep FP32 for stability
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)                     # (M, D) FP32
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)               # (B, N, N, D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse five linear layers (projections + gates) into one GEMM
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]    # (H, D)
    left_gate_w   = weights["left_gate.weight"]    # (H, D)
    right_proj_w  = weights["right_proj.weight"]   # (H, D)
    right_gate_w  = weights["right_gate.weight"]   # (H, D)
    out_gate_w    = weights["out_gate.weight"]     # (H, D)

    # concatenate along the output‑feature dimension (5*H, D)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)

    # (M, 5*H) FP16
    proj_gate_flat = F.linear(x_norm_f16.view(M, dim), combined_w)

    # ------------------------------------------------------------------
    # 4️⃣ Slice, apply sigmoid gates, and the optional mask (in‑place)
    # ------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, N, N, 5 * hidden)   # (B,N,N,5H)

    H = hidden
    left_proj   = proj_gate[..., 0:H]               # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]           # (B,N,N,H)
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]           # (B,N,N,H)

    # sigmoid gates
    left_gate   = torch.sigmoid(left_gate_r)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    # fuse gating into the projections (no extra temporaries)
    left_proj.mul_(left_gate)      # left_proj = left_proj * left_gate
    right_proj.mul_(right_gate)    # right_proj = right_proj * right_gate

    if (not nomask) and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj.mul_(mask_f16)
        right_proj.mul_(mask_f16)

    # ------------------------------------------------------------------
    # 5️⃣ Pairwise multiplicative update: Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a single einsum to avoid the large permute+contig copies.
    # ------------------------------------------------------------------
    # out shape: (B, N, N, H)
    out = torch.einsum('bikd,bjkd->bijd', left_proj, right_proj)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – still in FP16
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden)                          # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden)         # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection, cast back to FP32
    # ------------------------------------------------------------------
    out_norm.mul_(out_gate)                                 # in‑place gating

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    # Linear projection back to model dimension
    out_fp16 = F.linear(out_norm.view(M, hidden), to_out_w)        # (M, D)
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)      # (B,N,N,D)

    return out_fp32
