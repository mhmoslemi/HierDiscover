import triton
import triton.language as tl

@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)
    
    # compute batch and head indices
    hb = pid_hb
    b = hb // H
    h = hb - b * H
    
    # offsets
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_offset = tl.arange(0, BLOCK_K)
    
    # mask by bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N
    
    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for k_block in range(num_blocks_k):
        k_base = k_block * BLOCK_K
        k_off = k_base + k_offset
        
        # compute valid mask per k
        mask_k_tile = k_off < N
        
        # Load A: left[b,i,k,h] shape (BLOCK_I, BLOCK_K)
        a_ptr = left_ptr + b * stride_left_b + i_offset[:, None] * stride_left_i + k_off[None, :] * stride_left_k + h * stride_left_h
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k_tile[None, :], other=0.0).to(tl.float32)
        
        # Load B_T: right[b,j,k,h] transposed into shape (BLOCK_K, BLOCK_J)
        b_ptr = right_ptr + b * stride_right_b + k_off[:, None] * stride_right_k + j_offset[None, :] * stride_right_i + h * stride_right_h
        b = tl.load(b_ptr, mask=mask_k_tile[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        
        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_I, BLOCK_J)
    
    # store result
    out_ptrs = out_ptr + b * stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

Highly‑optimized outgoing TriMul (Tensor‑wise multiplicative update) forward pass.

Algorithm
---------
1️⃣ Input LayerNorm with torch.nn.functional.layer_norm (FP32 reduction,
   FP16 arithmetic).  Output cast to FP16.
2️⃣ Fuse the five linear projections (left/right, three gates) into a single
   half‑precision GEMM.
3️⃣ Apply the binary mask (if present) and sigmoid gating – all in FP16.
4️⃣ Core contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton kernel that evaluates the N³ reduction
   directly on the original [B, N, N, H] layout – no costly permutes or
   copies.
5️⃣ Output LayerNorm (FP32 reduction, FP16 output) via torch.nn.functional.
6️⃣ Apply the output gate, final half‑precision linear projection back to the
   model dimension, and cast the result to float32.
All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while the two

    # ------------------------------------------------------------------

    # Loop over k‑tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left tile: shape (BLOCK_I, BLOCK_K) -------------
        a_ptr = (left_ptr
                 + b * stride_left_b
                 + offs_i[:, None] * stride_left_i
                 + offs_k[None, :] * stride_left_k
                 + h * stride_left_h)
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----------- load right tile transposed: shape (BLOCK_K, BLOCK_J) -
        b_ptr = (right_ptr
                 + b * stride_right_b
                 + offs_k[:, None] * stride_right_k
                 + offs_j[None, :] * stride_right_i
                 + h * stride_right_h)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ----------- accumulate block‑wise GEMM ---------------------------
        # tl.dot broadcasts over the outer dimensions:
        #   a: (BLOCK_I, BLOCK_K), b: (BLOCK_K, BLOCK_J) → (BLOCK_I, BLOCK_J)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_out_b
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j
                + h * stride_out_h)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def triton_outgoing_contraction(left: torch.Tensor, right: torch.Tensor,
                               BLOCK_I: int = 32, BLOCK_J: int = 32, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Contract two tensors of shape [B, N, N, H] using the Triton kernel.
    Returns a tensor of the same shape, in FP16.
    """
    assert left.shape == right.shape, "left and right must have identical shape"
    B, N, _, H = left.shape

    # Ensure contiguous layout (required for stride‑based addressing)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Extract strides (in element units, not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    grid = (triton.cdiv(N, BLOCK_I),        # i‑tiles
            triton.cdiv(N, BLOCK_J),        # j‑tiles
            B * H)                          # batch*head dimension

    outgoing_trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=H,
        stride_left_b=sL_b,
        stride_left_i=sL_i,
        stride_left_k=sL_k,
        stride_left_h=sL_h,
        stride_right_b=sR_b,
        stride_right_i=sR_i,
        stride_right_k=sR_k,
        stride_right_h=sR_h,
        stride_out_b=sO_b,
        stride_out_i=sO_i,
        stride_out_j=sO_j,
        stride_out_h=sO_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from parameter names to torch.Tensors
        - config_dict  : contains ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0️⃣ Input LayerNorm (FP32 reduction, FP16 arithmetic)
    # --------------------------------------------------------------
    # Convert weight/bias to FP32 for the native LayerNorm kernel (more stable)
    norm_weight = weights["norm.weight"].to(device)  # float32
    norm_bias   = weights["norm.bias"].to(device)    # float32
    # LayerNorm over last dimension `dim`
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=1e-5)                # still float32
    x_norm = x_norm.half()                         # cast to FP16 for the projections

    # --------------------------------------------------------------
    # 1️⃣ Fuse five linear projections into a single half‑precision GEMM
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)               # shape (5·H, dim)

    # Linear projection (batched) – result (B·N·N, 5·H) in FP16
    proj = F.linear(x_norm.view(-1, dim), combined_weight)

    # --------------------------------------------------------------
    # 2️⃣ Split back to the individual tensors and reshape
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    # reshape to [B, N, N, H]
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) – broadcast over the hidden dimension
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw * left_gate       # [B, N, N, H] FP16
    right = right_raw * right_gate     # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via custom Triton kernel
    # --------------------------------------------------------------
    out = triton_outgoing_contraction(left, right,
                                      BLOCK_I=32, BLOCK_J=32, BLOCK_K=32)  # FP16 output

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32 reduction, FP16 arithmetic)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)   # float32
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)     # float32
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)                                 # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate       # [B, N, N, H] FP16

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    final_half = F.linear(gated.view(-1, hidden_dim), to_out_weight)          # (B·N·N, dim) FP16
    final = final_half.view(bs, seq_len, seq_len, dim).float()                # cast to FP32

    return final
