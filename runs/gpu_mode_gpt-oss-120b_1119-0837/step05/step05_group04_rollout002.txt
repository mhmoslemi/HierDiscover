def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # Flatten x to (M,D)
    M = bs * N * N
    x_flat = x.view(M, dim)

    # Input LN (Triton)
    x_norm_flat = triton_layernorm(x_flat, weights["norm.weight"], weights["norm.bias"], eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim).to(torch.float16)   # fp16

    # fused projection for left + left_gate + right + right_gate + out_gate
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(x_norm.device, torch.float16)

    # linear projection
    proj = torch.nn.functional.linear(x_norm.view(M, dim), combined_weight) # (M,5H)
    # split
    H = hidden
    left_proj = proj[:, 0:H].view(bs, N, N, H)
    left_gate_raw = proj[:, H:2*H].view(bs, N, N, H)
    right_proj = proj[:, 2*H:3*H].view(bs, N, N, H)
    right_gate_raw = proj[:, 3*H:4*H].view(bs, N, N, H)
    out_gate_raw = proj[:, 4*H:5*H].view(bs, N, N, H)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.unsqueeze(-1).to(x_norm.dtype)  # (B,N,N,1)
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Contraction via einsum (no extra permute copies)
    # out shape (B,N,N,H) FP16
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # Output LN (Triton)
    out_flat = out.view(M, H)
    out_norm_flat = triton_layernorm(out_flat, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, H)

    # Apply out_gate
    out_gated = out_norm * out_gate

    # Final linear
    out_fp16 = torch.nn.functional.linear(out_gated.view(M, H), weights["to_out.weight"].to(x_norm.device, torch.float16))

    return out_fp16.view(bs, N, N, dim).to(torch.float32)
```

"""
TriMul – “outgoing’’ variant (AlphaFold‑3) accelerated with Triton.

Algorithm
---------
1️⃣ Input LayerNorm  – 2‑pass Triton kernel (FP32 accumulation, FP16 I/O).
2️⃣ Cast to FP16 and fuse the five linear layers (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single GEMM.
3️⃣ Apply sigmoid to the three gating tensors, multiply by the optional
   pair‑wise mask and obtain the gated left/right tensors.
4️⃣ Contract the two 4‑D tensors with a single Einstein‑summation
   (torch.einsum) – this avoids the costly permute+contiguous+view used
   in the reference implementation.
5️⃣ Output LayerNorm – same Triton kernel, now on the hidden‑dim tensor.
6️⃣ Apply the output gate, a final FP16 linear projection back to the model
   dimension, and cast the result to float32.

Only the two LayerNorms are written in Triton; all heavy dense linear algebra
uses the highly‑optimised cuBLAS (via PyTorch).  This yields a large reduction
in temporary memory traffic compared with the reference while keeping the
forward‑only requirement of the challenge.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm written in Triton.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # *ptr input  [M, D]
    y_ptr,  # *ptr output [M, D]
    w_ptr,  # *ptr weight [D]
    b_ptr,  # *ptr bias   [D]
    M,      # number of rows
    D,      # feature dimension
    eps,    # numerical stability
    stride_xm, stride_xd,   # strides of x
    stride_ym, stride_yd,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row block (M dimension)
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – compute row-wise mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d_it in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d_it * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalise, apply affine transform, write output
    # ------------------------------------------------------------------
    for d_it in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d_it * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f = tl.cast(x, tl.float32)

        # normalisation (FP32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        # affine parameters (may be FP16 or FP32)
        w = tl.cast(tl.load(w_ptr + offs_d, mask=mask_d, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_d, mask=mask_d, other=0.0), tl.float32)

        y = norm * w[None, :] + b[None, :]

        # store in original dtype of the input tensor
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yd,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    2‑pass LayerNorm wrapper.
    x must be a 2‑D contiguous tensor (M, D); the returned tensor has the
    same shape and dtype as ``x``.
    """
    assert x.ndim == 2, "layernorm expects a 2‑D tensor"
    M, D = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous strides for the kernel
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xd = x.stride()
    stride_ym, stride_yd = y.stride()

    # Block sizes – chosen for good occupancy across the whole D range
    BLOCK_M = 128                # rows per thread block
    BLOCK_D = 64                 # columns per thread block

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_ym=stride_ym,
        stride_yd=stride_yd,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass for the “outgoing’’ TriMul operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N]   or ``None``
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – stay in FP32 for stability
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)                     # (M, D) – FP32

    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    # Back to 4‑D tensor and cast to FP16 for the remaining ops
    x_norm = x_norm_flat.view(bs, N, N, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (projections + gates) into ONE GEMM
    # ------------------------------------------------------------------
    w_left     = weights["left_proj.weight"]          # (H, D)
    w_left_g   = weights["left_gate.weight"]          # (H, D)
    w_right    = weights["right_proj.weight"]         # (H, D)
    w_right_g  = weights["right_gate.weight"]         # (H, D)
    w_out_g    = weights["out_gate.weight"]           # (H, D)

    # (5*H, D) – all in FP16
    combined_w = torch.cat(
        [w_left, w_left_g, w_right, w_right_g, w_out_g], dim=0
    ).to(device, torch.float16)

    # Linear: (M, D) @ (5*H, D).T -> (M, 5*H)
    proj = F.linear(x_norm.view(M, dim), combined_w)   # (M, 5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice, activate gates, optionally apply mask
    # ------------------------------------------------------------------
    H = hidden_dim
    # The layout matches the concatenation order above
    left_proj      = proj[:, 0:H].view(bs, N, N, H)
    left_gate_raw  = proj[:, H:2*H].view(bs, N, N, H)
    right_proj     = proj[:, 2*H:3*H].view(bs, N, N, H)
    right_gate_raw = proj[:, 3*H:4*H].view(bs, N, N, H)
    out_gate_raw   = proj[:, 4*H:5*H].view(bs, N, N, H)

    left_gate  = torch.sigmoid(left_gate_raw)   # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    if not nomask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] (FP16)
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Contract via a single einsum (avoids the expensive permute+contiguous)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Result: (B,N,N,H) in FP16
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) – kept in FP16
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                       # (M, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)      # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final linear, cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B,N,N,H) FP16

    w_to_out = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    # Linear projection back to model dimension (FP16)
    out_fp16 = F.linear(gated.view(M, hidden_dim), w_to_out)      # (M, D)

    # Final reshape and FP32 cast – public API format
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)

    return out_fp32
