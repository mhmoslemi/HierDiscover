"""
custom_kernel – TriMul “outgoing” forward pass (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the last channel (float32) using a tiny Triton kernel.
2️⃣  Cast to FP16 for the heavy linear work (all linear layers).
3️⃣  Fuse each projection + its gate into a single matmul (2·H channels).
4️⃣  Apply the optional binary mask (broadcast over the hidden dimension).
5️⃣  Element‑wise gating of the left/right projections.
6️⃣  Compute the N³ contraction   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    – done as a batched matrix‑multiply: for every hidden channel h
      left_h (N×N) @ right_hᵀ (N×N).  This uses torch.bmm (cuBLAS) which
      is faster than a naïve Triton implementation on H100.
7️⃣  Hidden‑dim LayerNorm (float16) with another Triton kernel.
8️⃣  Apply the output‑gate (sigmoid of a linear on the original normed input).
9️⃣  Final projection back to model dimension (float16 matmul) and cast to float32.
The only Triton kernels are the two LayerNorms, satisfying the “must use a kernel”
requirement while the compute‑heavy N³ step leverages the highly‑optimized cuBLAS
backend.

All tensors are expected to already reside on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# Float‑32 LayerNorm – two‑pass (mean/var + affine) via Triton
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # row indices handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -----------------------------------------------------------------
    # first pass – compute mean and variance per row (float32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_N)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -----------------------------------------------------------------
    # second pass – write normalized + affine result
    # -----------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float32) with Triton."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
        num_warps=4,
    )
    return y

def layernorm_fp16_kernel(

    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )

        norm_f32 = (xf - mean[:, None]) * rstd[:, None]

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension (float16) with Triton."""
    assert x.dtype == torch.float16
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of model parameter tensors
        - config       : dict with keys:
            * dim          – model dimension D
            * hidden_dim   – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    # Flatten over the pairwise dimension so we have a 2‑D tensor (M,N')
    # where M = B·N·N and N' = dim.
    M_flat = bs * N * N
    x_flat = input_tensor.view(M_flat, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)          # (B,N,N,D) – still FP32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate (two linear layers → 2·H channels each)
    # ------------------------------------------------------------------
    left_weight   = torch.cat([weights["left_proj.weight"],
                               weights["left_gate.weight"]], dim=0).to(device, torch.float16)  # (2H, D)
    right_weight  = torch.cat([weights["right_proj.weight"],
                               weights["right_gate.weight"]], dim=0).to(device, torch.float16)  # (2H, D)
    out_gate_weight = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Combined matmuls (projection + its gate in a single call)
    left_comb  = F.linear(x_f16, left_weight)   # (B,N,N,2H)
    right_comb = F.linear(x_f16, right_weight)  # (B,N,N,2H)

    # Split back into projection and gate tensors
    left_proj  = left_comb[..., :H]   # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]  # (B,N,N,H)
    right_proj = right_comb[..., :H]  # (B,N,N,H)
    right_gate_pre = right_comb[..., H:]  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and optional binary mask
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)  # (B,N,N,H)

    if not nomask and mask is not None:
        # mask shape (B,N,N) → broadcast over hidden dim
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Apply element‑wise gating
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # --------------------------------------------------------------
    # 6️⃣ Output‑gate (computed from the *original* normed input)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_weight))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Core N³ contraction: out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    #    Implemented as a batched GEMM using cuBLAS (torch.bmm) which is
    #    faster than a pure‑Triton kernel on the H100 for the given sizes.
    # --------------------------------------------------------------
    # Move hidden dimension to the batch axis
    # left_perm : (B, H, N, N)  where inner dims are (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right_perm : (B, H, N, N) where inner dims are (j,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Flatten batch × hidden into a single batch dimension
    A = left_perm.view(bs * H, N, N)        # (B·H, N, N)
    B = right_perm.view(bs * H, N, N)       # (B·H, N, N)

    # Perform batched matmul: each slice computes A @ Bᵀ  (i.e. Σₖ left[i,k]·right[j,k])
    out_mat = torch.bmm(A, B.transpose(1, 2))   # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) – FP16

    # --------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton
    # --------------------------------------------------------------
    out_flat = out.view(M_flat, H)                     # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # --------------------------------------------------------------
    # 9️⃣ Apply output‑gate and final projection back to model dim
    # --------------------------------------------------------------
    out_gate_flat = out_gate.view(M_flat, H)           # (B·N·N, H)
    gated = out_norm * out_gate_flat                   # (M, H) – FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)            # (M, D) – FP16

    # Cast back to float32 and reshape to the original 4‑D layout
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output
