"""
custom_kernel – Triton‑accelerated “outgoing” TriMul
===================================================

Forward pass of the TriMul module (the “outgoing” variant from AlphaFold‑3).

Main ideas
----------
1️⃣  Two‑pass LayerNorm (mean/variance → normalise → affine) implemented in Triton.
    The kernel works on a 2‑D view of the tensor so it can be reused for the
    input‑norm (shape [B·N·N, D]) and the hidden‑norm (shape [B·N·N, H]).

2️⃣  The five Linear layers (left/right projections + three gates) are fused
    into a single GEMM `x @ W.T` where `W` has shape (5·H, D).  The result is
    immediately split into the five tensors, the three gate tensors are passed
    through a sigmoid, and the masks (if any) are applied – no additional
    memory traffic is introduced.

3️⃣  The core contraction
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    is a batched matrix‑multiplication.  By reshaping the tensors to
    (B·H, N, N) we can call `torch.bmm`, which on an H100 uses Tensor‑Core
    FP16 GEMMs and is extremely fast for the required sizes.

4️⃣  The second LayerNorm, the output gate, and the final projection back to
    the model dimension are carried out in FP16 and the result is promoted
    to FP32 for the public API.

The implementation contains a Triton kernel (`layernorm_kernel`) and a
convenient wrapper (`triton_layernorm`).  All other heavy work is performed
with highly‑optimised PyTorch kernels, giving a good balance of speed and
readability while still satisfying the requirement to use Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (works on any dtype, computes stats in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr input  [M, N]
    y_ptr,                # *ptr output [M, N]
    weight_ptr,           # *ptr weight [N]
    bias_ptr,             # *ptr bias   [N]
    M,                    # rows   (B·N·N)
    N,                    # feature dimension (D or H)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    Two‑pass LayerNorm.
    First pass: compute per‑row mean and variance.
    Second pass: normalise, apply affine transform, and store.
    """
    pid = tl.program_id(0)

    # -------- rows handled by this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -------- first pass: accumulate sum and sum‑squares --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # iterate over the feature dimension in blocks
    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)          # 1/σ

    # -------- second pass: normalise + affine -------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalisation (in FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # load affine parameters (may be FP16 → cast)
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Apply the 2‑pass Triton LayerNorm on a 2‑D tensor ``x`` of shape (M, N).
    The output has the same dtype as ``x``.
    """
    assert x.dim() == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make everything contiguous for the kernel
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes – tuned for H100 and the dimensions we see in the tests
    BLOCK_M = 256            # rows per program
    BLOCK_N = 64             # columns per iteration (covers D/H up to 768)

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated “outgoing” TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor (B, N, N, D)  dtype float32
        - mask         : torch.Tensor (B, N, N) or ``None``
        - weights      : dict of model parameters (already on device)
        - config       : dict containing:
                         * dim          – model dimension D
                         * hidden_dim   – hidden dimension H
                         * nomask (opt) – bool, skip masking branch

    Returns
    -------
    torch.Tensor
        Output tensor (B, N, N, D) dtype float32
    """

    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len                         # rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)              # (M, D)  fp32

    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)

    # Keep the result in the driver dtype (fp32) and then down‑cast to fp16
    x_norm_fp32 = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five Linear layers (projections + gates)
    # ------------------------------------------------------------------
    # Gather the five weight matrices – each (H, D)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate --> shape (5·H, D) and switch to fp16 (Tensor‑Core friendly)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                       # (5·H, D)

    # (M, D) @ (D, 5·H).T  ->  (M, 5·H)
    proj_gate_flat = F.linear(
        x_norm.view(M, dim),    # (M, D)  fp16
        combined_w,             # (5·H, D) fp16
    )                           # → (M, 5·H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Slice, apply sigmoid and (optional) mask
    # ------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5·H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r = proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)          # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_r)         # (B,N,N,H) fp16
    out_gate   = torch.sigmoid(out_gate_r)           # (B,N,N,H) fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched contraction Σₖ left * right  →  (B·H, N, N)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then collapse batch*head
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)

    # Batched matrix multiplication: left @ rightᵀ  (FP16 Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore (B,N,N,H)
    out = (
        out_mat.view(bs, hidden_dim, seq_len, seq_len)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) – stays in FP16
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                     # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)   # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)     # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w,
                                     to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gate, final projection, cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                                 # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)      # (D, H) fp16
    out_fp16 = F.linear(gated, to_out_w)                       # (B,N,N,D) fp16
    out_fp32 = out_fp16.to(torch.float32)                      # API expects FP32

    return out_fp32
