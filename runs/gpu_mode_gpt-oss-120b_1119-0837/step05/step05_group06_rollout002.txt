"""
Highly‑optimized Triton + PyTorch implementation of the TriMul “outgoing’’
operator (AlphaFold3).  

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input tensor (B,N,N,D).  Implemented with a custom
   2‑pass Triton kernel (FP32 mean/variance, FP16 output) to avoid
   memory traffic of the built‑in LayerNorm.

2️⃣ Cast the normalized tensor to FP16 – all heavy work runs in half precision.

3️⃣ Fuse the five linear layers (left/right projections + three gates) into a
   single GEMM:
   - Weight matrix has shape (5·H, D) (H = hidden_dim).
   - Output is reshaped to (B,N,N,5·H) and sliced into left_proj,
     left_gate, right_proj, right_gate, out_gate.

4️⃣ Apply element‑wise sigmoid to the three gate tensors, optionally apply the
   pair‑wise mask, and compute the gated left/right tensors.

5️⃣ Compute the O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   by reshaping to (B·H, N, N) and using a batched FP16 GEMM (torch.bmm).

6️⃣ Layer‑Norm the intermediate result (still FP16) with another Triton kernel.

7️⃣ Multiply by the output gate, apply the final linear projection back to the
   model dimension D and cast the result to float32 (the public API expects FP32).

The implementation keeps the heavy linear algebra in cuBLAS/Tensor‑Core (FP16)
while fusing memory‑bound operations in Triton, achieving a large speed‑up
over a pure‑PyTorch baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension).  Supports any dtype; we keep the
# output dtype identical to the input (typically fp16) but compute mean/var in
# fp32 for numerical stability.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # rows = B·N·N
    N,                # feature dimension (D or H)
    eps,              # epsilon for variance
    stride_xm, stride_xn,      # strides of x
    stride_ym, stride_yn,      # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- row range for this program ------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute sum and sum‑sq (FP32) -----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_iters = tl.cdiv(N, BLOCK_N)
    for n_iter in range(num_n_iters):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        # Load a BLOCK_M×BLOCK_N tile
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalize + affine -------------------------------
    for n_iter in range(num_n_iters):
        offs_n = tl.arange(0, BLOCK_N) + n_iter * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # Normalize (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # Load weight & bias (may be fp16/32) and cast to fp32
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        # Store in original dtype
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Simple wrapper around the Triton LayerNorm kernel.
    The input ``x`` must be a 2‑D contiguous tensor (M, N); the output has the
    same dtype and layout as ``x``.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for both D=128 and D=384/768 on H100.
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]   (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` (D) and ``hidden_dim`` (H);
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype   # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – work on a 2‑D view (M, D)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len                 # total rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)      # (M, D) – fp32

    norm_w = weights["norm.weight"].to(device, dtype_in)   # (D)
    norm_b = weights["norm.bias"].to(device, dtype_in)     # (D)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) – fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the remaining heavy work runs in half precision
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)   # (M, D) – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projections and three gates into one GEMM
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]          # (H, D)
    right_proj_w  = weights["right_proj.weight"]         # (H, D)
    left_gate_w   = weights["left_gate.weight"]          # (H, D)
    right_gate_w  = weights["right_gate.weight"]         # (H, D)
    out_gate_w    = weights["out_gate.weight"]           # (H, D)

    # Concatenate on the output‑feature axis → (5·H, D)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)               # (5*H, D)

    # Single fused linear projection (no bias)
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)   # (M, 5·H) – fp16

    # ------------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor, apply sigmoid gating and mask
    # ------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)   # (B,N,N,H) – fp16
    right_gate = torch.sigmoid(right_gate_r)  # (B,N,N,H) – fp16
    out_gate   = torch.sigmoid(out_gate_r)    # (B,N,N,H) – fp16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h]·right[b,j,k,h]
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis and make tensors contiguous.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim                               # B·H
    left_mat  = left_perm.view(BH, seq_len, seq_len)   # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)  # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (FP16 GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore (B,N,N,H) layout
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (still in FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                      # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_in)  # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_in)    # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Gating, final projection & cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B,N,N,H) – fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    # Final linear projection (FP16) → (B,N,N,D)
    out_fp16 = F.linear(gated, to_out_w)                           # (B,N,N,D)
    out_fp32 = out_fp16.to(torch.float32)                          # Return FP32 as the API expects

    return out_fp32
