"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

Algorithm
---------
1️⃣ Layer‑Norm on the input tensor (shape B×N×N×D).  Implemented with a
   two‑pass Triton kernel that computes per‑row mean/variance in FP32 and
   applies the affine transform.  The result is cast to FP16 for the heavy
   compute that follows.

2️⃣ Fuse the five linear projections (left/right projection + three gates)
   into one matrix‑multiply using a concatenated weight of shape
   (5·H, D) where H = hidden_dim.  A single `torch.nn.functional.linear`
   call produces a tensor of shape B·N·N×5·H in FP16.

3️⃣ Slice the fused result, apply a sigmoid to the three gating tensors,
   optionally mask the left/right tensors, and multiply the gates.

4️⃣ For each head h ∈ [0,H) we need
        out_{b,i,j,h} = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is exactly a batched matrix multiplication
        (B·H, N, N) @ (B·H, N, N)^T .
   We permute the tensors, reshape to (B·H,N,N) and use
   `torch.bmm`, which on H100 runs as a Tensor‑Core GEMM in FP16.

5️⃣ Second Layer‑Norm (on the hidden dimension H) – again via the Triton
   kernel, staying in FP16.

6️⃣ Multiply by the output gate, project back to the model dimension D
   with a final FP16 linear layer, and cast the result to FP32 (the public
   API expects float32).

All heavy compute runs in FP16; the two LayerNorms are the only parts
implemented in Triton, satisfying the requirement to include a custom
kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (two‑pass: mean/var → normalise + affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr input  [M, N]
    y_ptr,                # *ptr output [M, N]
    weight_ptr,           # *ptr weight [N]
    bias_ptr,             # *ptr bias   [N]
    M,                    # rows (M = B·N·N)
    N,                    # feature dimension (D or H)
    eps,                  # epsilon for numerical stability
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # row offsets for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iters_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iters_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)                # 1 / σ

    # ------------------------------------------------------------------
    # Second pass – normalise and apply affine transformation
    # ------------------------------------------------------------------
    for it in range(num_iters_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply a 2‑pass LayerNorm using Triton on a 2‑D tensor x of shape (M, N).
    Returns a tensor of the same shape and dtype as x.
    """
    assert x.dim() == 2, "LayerNorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous storage (required for correct stride calculations)
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for D ∈ {128,384,768}
    BLOCK_M = 256
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul “outgoing” forward pass (no gradient needed)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see keys in docstring)
        - config       : dict with at least ``dim`` and ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] and dtype torch.float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", False))
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep result in FP16
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len              # total rows after flattening
    x_flat = input_tensor.reshape(M, dim)   # (M, D) – still float32

    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)

    # Normalize in FP32, cast to FP16 for the rest of the pipeline
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (proj + three gates)
    # ------------------------------------------------------------------
    # Gather weight matrices (all shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate -> (5·H, D) and cast to FP16 for speed
    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)

    # Linear projection on the flattened tensor (no bias)
    # Result shape: (M, 5·H) in FP16
    proj_gate_flat = F.linear(
        x_norm.view(M, dim),   # (M, D) FP16
        combined_w,            # (5·H, D) FP16
    )

    # ------------------------------------------------------------------
    # 3️⃣ Slice tensors, apply sigmoid and optional mask
    # ------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)

    H = hidden_dim
    left_proj    = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r  = proj_gate[..., H:2*H]
    right_proj   = proj_gate[..., 2*H:3*H]
    right_gate_r = proj_gate[..., 3*H:4*H]
    out_gate_r   = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)      # FP16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        # mask is broadcastable over the hidden dimension
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched contraction Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) so each (b, h) becomes a separate matrix
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim                                 # B·H
    left_mat  = left_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)    # (B·H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) – stays in FP16
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                       # (B·N·N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w,
                                    to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection to dim, cast to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    out_fp16 = F.linear(gated, to_out_w)       # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)      # final dtype as required

    return out_fp32
```
