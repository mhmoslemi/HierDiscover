    import torch, torch.nn.functional as F, triton, triton.language as tl
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast layernorm weight/bias to FP16
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float16)

    # Input LayerNorm
    x_flat = input_tensor.reshape(-1, dim).contiguous()
    x_norm = triton_layernorm_fp16(x_flat, norm_weight, norm_bias, eps=1e-5)

    # fused projections weight matrix
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(device, dtype=torch.float16)

    proj = F.linear(x_norm, combined_weight)  # (B*N*N, 5*hidden_dim)
    # split
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(proj, hidden_dim, dim=1)
    # reshape to [B, N, N, hidden_dim]

"""
Highly‑optimized outgoing TriMul (Tensor‑wise multiplicative update) forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32 reduction, FP16 output) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM.
3️⃣ Apply optional binary mask (broadcast over hidden dimension) and sigmoid gates.
4️⃣ Core contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed by a batched half‑precision GEMM written in Triton, which works
   directly on the original (B,N,N,H) layout – avoiding the expensive
   contiguous + permute copies required by `torch.bmm`.
5️⃣ Output LayerNorm (FP32 reduction, FP16 output) – same Triton kernel.
6️⃣ Apply the output gate and final half‑precision linear projection back to the
   model dimension.  Cast to FP32 for the final return value.
All heavy arithmetic runs in FP16 (Tensor‑Core friendly).  Two LayerNorms run
in FP32 for numerical stability, but their outputs stay in FP16 to minimise
memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr, M, C, eps,
    stride_xm, stride_xc, stride_ym, stride_yc,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row offset
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: mean / variance (FP32) ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: normalise + affine ----------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w + b               # still FP32
        y = y.to(tl.float16)           # store as half

        tl.store(
            y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc),
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm_fp16(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 256,
    BLOCK_C: int = 128,
) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a 2‑D tensor ``x`` (shape M×C)
    using a fused Triton kernel.  Input may be FP32 or FP16; output is FP16.
    """
    assert x.dim() == 2, "triton_layernorm_fp16 expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty((M, C), dtype=weight.dtype, device=x.device)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)
    _layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=8,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Batched matrix multiplication kernel for the core TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch b and hidden head h:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    left/right/out have shape (B,N,N,H) with strides as provided.
    """
    batch_head = tl.program_id(0)      # combined index = b * H + h
    pid_i = tl.program_id(1)          # tile row
    pid_j = tl.program_id(2)          # tile col

    # Recover batch and head indices
    b = batch_head // H
    h = batch_head % H

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Offsets of the current tile in the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   : shape (BLOCK_M, BLOCK_K)  -> left[b,i,k,h]
        a = tl.load(
            left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile  : shape (BLOCK_K, BLOCK_N) -> right[b,j,k,h] (k as rows, j as cols)
        b_tile = tl.load(
            right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # Multiply‑accumulate (tensor‑core dot)
        acc += tl.dot(a, b_tile)

    # Write the result back in FP16
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple

        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0️⃣ Cast LayerNorm parameters to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 reduction, FP16 output)
    # ------------------------------------------------------------------
    # reshape to (B·N·N, dim) – treat as 2‑D matrix for the LN kernel
    x_flat = input_tensor.reshape(-1, dim).contiguous()
    x_norm = triton_layernorm_fp16(
        x_flat,
        norm_weight,
        norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                               # (B·N·N, dim) FP16
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                     # (5·H, dim)

    proj = F.linear(x_norm.view(-1, dim), combined_weight)   # (B·N·N, 5·H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split back into the individual tensors
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    # reshape to (B,N,N,H)
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)      # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)     # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)       # (B,N,N,H) FP16

    left  = left_raw * left_gate   # (B,N,N,H) FP16
    right = right_raw * right_gate # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction via custom Triton batch GEMM (FP16)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous so that their strides match the
    # expected layout (B,N,N,H) with stride order (B, i, k, h).
    left  = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # (B,N,N,H) FP16

    # Strides for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()   # o_j == stride on the second N axis

    # Triton block sizes – chosen to fit Tensor‑Core tiles on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,                     # combined (batch * head) dimension
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    _tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 reduction, FP16 output)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out_flat = out.view(-1, hidden_dim)
    out_norm = triton_layernorm_fp16(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                   # (B·N·N, H) FP16
    out_norm = out_norm.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H) FP16

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_half = F.linear(gated.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim) FP16
    final = final_half.view(bs, seq_len, seq_len, dim).float()       # cast to FP32

    return final
```
