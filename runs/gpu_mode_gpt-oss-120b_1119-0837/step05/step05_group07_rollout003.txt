def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # Step 1: LayerNorm x to half
    M = bs * seq_len * seq_len
    x_flat = input_tensor.view(M, dim).contiguous()
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # ensure same dtype as input (float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # Step 2: Combined linear projection for left/right and gates
    # Build combined weight: shape (5 * hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    combined_w = torch.cat([left_proj_w, left_gate_w,
                            right_proj_w, right_gate_w,
                            out_gate_w], dim=0).to(device, torch.float16)
    # Linear
    proj = F.linear(x_norm.view(M, dim), combined_w) # (M,5*H) in fp16
    # reshape
    proj = proj.view(bs, seq_len, seq_len, 5 * hidden_dim)

    # split
    H = hidden_dim
    left_proj = proj[..., 0:H]
    left_gate = torch.sigmoid(proj[..., H:2*H])
    right_proj = proj[..., 2*H:3*H]
    right_gate = torch.sigmoid(proj[..., 3*H:4*H])
    out_gate = torch.sigmoid(proj[..., 4*H:5*H])

    # mask
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Step 4: O(N^3) contraction using einsum
    # out shape (B,N,N,H)
    out = torch.einsum('bikc,bjkc->bijc', left, right)

    # Step 5: Output layernorm
    out_flat = out.view(M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)

    # Step 6: Final gating and projection back to dim
    gated = out_norm * out_gate
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (dim, hidden_dim)
    out_fp16 = F.linear(gated, to_out_w)  # (B,N,N,dim) fp16
    out_fp32 = out_fp16.to(torch.float32)

    return out_fp32

"""
TriMul “outgoing” implementation (AlphaFold3 style) with Triton‑accelerated LayerNorm
and a copy‑free einsum contraction.

Algorithm
---------
1️⃣  Layer‑norm the input tensor over the last dimension (dim) with a
    two‑pass Triton kernel (mean/var → normalise + affine).  The result is kept
    in FP16 to speed up the heavy compute.

2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single matrix multiplication:
    Concatenate the five weight matrices (shape (5·H, D)) and apply
    ``F.linear`` on the flattened input, obtaining a tensor of shape
    (B, N, N, 5·H).  Slice the result into the four projected tensors and the
    output‑gate tensor.  Apply a sigmoid to the three gate tensors.

3️⃣  If a mask is present (and ``nomask`` is ``False``) multiply the gated
    left/right tensors by the mask (broadcast on the hidden dimension).

4️⃣  Compute the O(N³·H) “triangle‑multiplicative” update with a
    copy‑free ``torch.einsum``:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This avoids the costly ``permute``/``contiguous`` dance required for a
    batched GEMM.

5️⃣  Apply a second LayerNorm (again Triton‑based) on the hidden dimension H.

6️⃣  Multiply by the output‑gate, linearly project back to the model dimension D,
    and cast the result to float32 (the public API expectation).

The implementation keeps all heavy compute in FP16 while performing mean/var
reduction in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (works on any dtype, computes mean/var in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr input  [M, N]
    y_ptr,                # *ptr output [M, N]
    weight_ptr,           # *ptr weight [N]
    bias_ptr,             # *ptr bias   [N]
    M,                    # rows (B·N·N)
    N,                    # feature dimension (D or H)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Two‑pass LayerNorm (mean/variance → normalise + affine)."""
    pid = tl.program_id(0)

    # Row indices for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass: compute per‑row mean & variance
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val = sum_val + tl.sum(x_f32, axis=1)
        sum_sq  = sum_sq  + tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)          # 1/σ

    # Second pass: normalise + affine transform
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (weight & bias may be FP16 → cast to FP32)
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        # store with original dtype
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply the Triton two‑pass LayerNorm on a 2‑D tensor ``x`` of shape (M,N).
    The output has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for stride extraction
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for H100 / D ∈ {128,384,768}
    BLOCK_M = 256
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated “outgoing” TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (all tensors on the target device)
        - config       : dict with keys
                         * dim        – model dimension D
                         * hidden_dim – projection dimension H
                         * nomask (opt) – bool, when True ignore the mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ----------------------------- unpack ---------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------- 1️⃣ Input LayerNorm (Triton) --------------------
    # Collapse the first three axes so that LayerNorm becomes a 2‑D op.
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim).contiguous()          # (M, D) FP32

    # LayerNorm weights & bias (float32 → keep for stability)
    norm_w = weights["norm.weight"].to(device, dtype=input_tensor.dtype)
    norm_b = weights["norm.bias"].to(device, dtype=input_tensor.dtype)

    # Normalisation (still FP32), then cast to FP16 for the rest of the pipeline
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).to(torch.float16)

    # ------------------- 2️⃣ Fuse five linear layers --------------------
    # Concatenate the five weight matrices (each of shape [H, D])
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)      # (5·H, D)

    # Linear projection (no bias) on the flattened tensor
    proj_flat = F.linear(
        x_norm.view(M, dim),    # (M, D) FP16
        combined_w,             # (5·H, D) FP16
    )                            # → (M, 5·H) FP16

    # Reshape back to the 4‑D layout
    proj = proj_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B, N, N, 5·H)

    H = hidden_dim
    left_proj   = proj[..., 0:H]                # (B, N, N, H)
    left_gate_r = proj[..., H:2*H]
    right_proj  = proj[..., 2*H:3*H]
    right_gate_r= proj[..., 3*H:4*H]
    out_gate_r  = proj[..., 4*H:5*H]

    # Apply sigmoid to the three gating tensors
    left_gate  = torch.sigmoid(left_gate_r)    # (B, N, N, H) FP16
    right_gate = torch.sigmoid(right_gate_r)  # (B, N, N, H) FP16
    out_gate   = torch.sigmoid(out_gate_r)    # (B, N, N, H) FP16

    # ------------------- 3️⃣ Mask & gating ------------------------------
    if (not nomask) and (mask is not None):
        # mask ∈ {0,1} (or bool), broadcast on hidden dimension
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- 4️⃣ Triangle‑Mul contraction --------------------
    # Using a copy‑free einsum avoids the expensive permute+contiguous step.
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikc,bjkc->bijc', left, right)   # (B, N, N, H) FP16

    # ------------------- 5️⃣ Output LayerNorm (Triton) -------------------
    out_flat = out.view(M, hidden_dim)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=input_tensor.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=input_tensor.dtype)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B, N, N, H) FP16

    # ------------------- 6️⃣ Gating, final projection & cast ------------
    gated = out_norm * out_gate                                   # (B,N,N,H) FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H) FP16
    out_fp16 = F.linear(gated, to_out_w)                          # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)                         # API expects FP32

    return out_fp32
