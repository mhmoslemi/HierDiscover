def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
   batch_head = tl.program_id(0)
   i_tile = tl.program_id(1)
   j_tile = tl.program_id(2)

   b = batch_head // H
   h = batch_head % H

   # base pointers offsets include batch and head offset
   base_left = left_ptr + b * stride_left_b + h * stride_left_h
   base_right = right_ptr + b * stride_right_b + h * stride_right_h
   base_out = out_ptr + b * stride_out_b + h * stride_out_h

   i_start = i_tile * BLOCK_I
   j_start = j_tile * BLOCK_J
   offs_i = i_start + tl.arange(0, BLOCK_I)
   offs_j = j_start + tl.arange(0, BLOCK_J)
   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

   # loop over k
   for k0 in range(0, N, BLOCK_K):
       k_start = k0
       offs_k = k_start + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # left tile [i,k]
       left_ptrs = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # right tile [j,k]
       right_ptrs = base_right + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
       right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # accumulate
       acc += tl.dot(left, right, trans_b=True)
   # end for

   # store output
   out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
================================================================

We follow the reference PyTorch module but replace the expensive
`einsum('... i k d, ... j k d -> ... i j d')` with a custom Triton kernel.
The kernel works on the already projected & gated tensors `left` and `right`
(which have shape ``[B, N, N, H]``) and computes for each head ``h``

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

using a blocked GEMM that reads the two tensors in tiles, accumulates in FP32
and writes the result back in FP16.  The rest of the pipeline (LayerNorm,
linear projections, sigmoid gates) is performed with standard PyTorch ops in
FP16; at least part of the computation (the N³ contraction) is therefore
fused in a Triton kernel as required.

The implementation works for both masked and un‑masked inputs and supports
all shapes listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Row‑wise LayerNorm (optional: used for the input and output norms)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,       # *ptr to affine parameters [N]
    M, N,                       # rows, features
    eps,                        # epsilon
    stride_xm, stride_xn,       # strides for input
    stride_ym, stride_yn,       # strides for output
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_N: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)

    # ------------------------------ row offsets ------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----------------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulator)
    # ----------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ----------------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm via a Triton kernel."""
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# N³ contraction kernel (tri‑multiplicative update)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # *ptrs  [B, N, N, H]
    B, N, H,                               # problem sizes
    # strides for the 4‑D tensors (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    All tensors are in FP16, accumulation in FP32.
    """
    # ----------------------- program IDs -----------------------
    batch_head = tl.program_id(0)                 # encodes (batch, head)
    ti = tl.program_id(1)                         # tile over i
    tj = tl.program_id(2)                         # tile over j

    # decode batch and head
    b = batch_head // H
    h = batch_head % H

    # base pointers that include batch & head offsets
    base_l = left_ptr + b * stride_left_b + h * stride_left_h
    base_r = right_ptr + b * stride_right_b + h * stride_right_h
    base_o = out_ptr + b * stride_out_b + h * stride_out_h

    # ----------------------- tile offsets -----------------------
    i_start = ti * BLOCK_I
    j_start = tj * BLOCK_J
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----------------------- loop over k -----------------------
    for k0 in range(0, N, BLOCK_K):
        k_start = k0
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = base_l \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile: shape (BLOCK_J, BLOCK_K)
        right_ptrs = base_r \
            + offs_j[:, None] * stride_right_i \
            + offs_k[None, :] * stride_right_k
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot product: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True)

    # ----------------------- write result -----------------------
    out_ptrs = base_o \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

    # 1️⃣ Input LayerNorm (fast built‑in version could be used,

    right_proj = fused_out[..., H:2*H]                    # (B, L, L, H)
    left_gate  = torch.sigmoid(fused_out[..., 2*H:3*H])   # (B, L, L, H)
    right_gate = torch.sigmoid(fused_out[..., 3*H:])     # (B, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (applied before gating, as in the paper)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = (left_proj * left_gate).contiguous()   # (B, L, L, H)  fp16
    right = (right_proj * right_gate).contiguous() # (B, L, L, H)  fp16

    # ------------------------------------------------------------------
    # 7️⃣ N³ contraction via custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                    # (B, L, L, H)  fp16

    # launch grid: (batch*head, ceil(N/BI), ceil(N/BJ))
    grid = (
        B * H,
        triton.cdiv(L, 64),   # BLOCK_I = 64
        triton.cdiv(L, 64),   # BLOCK_J = 64
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=L,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_I=64,
        BLOCK_J=64,
        BLOCK_K=32,
    )
    # ``out`` now holds the result of the einsum in FP16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (row‑wise, same kernel as the input norm)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, H)                         # [(B*L*L), H]
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(B, L, L, H)          # (B, L, L, H)  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Output gating (sigmoid over the original normalized input)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h.view(-1, D), out_gate_w))
    out_gate = out_gate.view(B, L, L, H)               # (B, L, L, H)  fp16

    # ------------------------------------------------------------------
    # 🔟 Apply gating and final linear projection back to ``dim``
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                        # fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated, to_out_w)            # (B, L, L, D)  fp16

    # Cast back to float32 for the official output dtype
    out_final = out_final_h.to(torch.float32)

    return out_final
