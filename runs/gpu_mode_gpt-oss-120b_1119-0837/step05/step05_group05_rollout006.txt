"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
============================================================

The operator processes a 4‑D tensor [B, N, N, C] and consists of
1️⃣ a LayerNorm over the channel dimension (implemented with a small Triton
    kernel),
2️⃣ four linear projections (left/right projections and their gates) fused into
    a single matrix multiplication,
3️⃣ optional mask‑application,
4️⃣ a batched GEMM that evaluates the N³ contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h],
5️⃣ a second LayerNorm (torch.nn.functional.layer_norm) and an output gate,
6️⃣ a final linear projection back to the original dimension.

All heavy arithmetic is performed in fp16 while the LayerNorm reductions are
done in fp32 for numerical stability.  The output is returned in fp32.
A Triton kernel is required for the input LayerNorm; the remaining
operations use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,       # *ptr to affine parameters [N]
    M, N,                       # rows, features
    eps,                        # epsilon
    stride_xm, stride_xn,       # strides for input
    stride_ym, stride_yn,       # strides for output
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_N: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – compute mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – write normalised + affine transformed values
    # ------------------------------------------------------------------
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm via the Triton kernel above.
    Expected input shape: (M, N) where N is the normalized feature dimension.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,          # rows per program
        BLOCK_N=64,           # columns per inner loop
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension `dim`
    # --------------------------------------------------------------
    # Cast input to fp16 before LayerNorm – arithmetic inside the kernel
    # promotes to fp32 for mean/var.
    x_fp16 = input_tensor.to(torch.float16)
    M = bs * L * L
    x_flat = x_fp16.reshape(M, dim)                 # [M, dim]

    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)  # fp16
    x_norm = x_norm_flat.view(bs, L, L, dim)        # [B, L, L, D] (fp16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear maps (proj + gate) and the out‑gate into one GEMM
    # --------------------------------------------------------------
    # Shape of each weight: (hidden_dim, dim)
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    og_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    # Cat along the output‑feature dimension: (5*H, D)
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w, og_w], dim=0)  # (5*H, D)

    # Linear projection in fp16
    x_norm_flat = x_norm_flat  # already fp16, shape (M, dim)
    fused_out = F.linear(x_norm_flat, fused_w)                # (M, 5*H)
    fused_out = fused_out.view(bs, L, L, 5 * hidden_dim)     # [B, L, L, 5*H]

    # --------------------------------------------------------------
    # 3️⃣ Slice the fused result
    # --------------------------------------------------------------
    h = hidden_dim
    left_proj = fused_out[..., :h]                              # (B,L,L,H)
    right_proj = fused_out[..., h:2 * h]                        # (B,L,L,H)
    left_gate = torch.sigmoid(fused_out[..., 2 * h:3 * h])      # (B,L,L,H)
    right_gate = torch.sigmoid(fused_out[..., 3 * h:4 * h])     # (B,L,L,H)
    out_gate = torch.sigmoid(fused_out[..., 4 * h:])           # (B,L,L,H)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask before gating (if present)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,L,L,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Apply gating (sigmoid) to the projections
    # --------------------------------------------------------------
    left = left_proj * left_gate        # (B, L, L, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Permute to (B, H, L, K) layout; torch.matmul will treat (B, H) as
    # batch dimensions and perform a GEMM for each head.
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, L, K)
    right_perm = right.permute(0, 3, 1, 2)        # (B, H, L, K)

    # GEMM (the transpose swaps K ↔ N for the second operand)
    out_mat = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, L, L)

    # Restore to original layout (B, L, L, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (torch built‑in, fast fused kernel)
    # --------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
        eps=eps,
    )  # fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate (computed earlier) and final linear projection
    # --------------------------------------------------------------
    gated = out * out_gate                               # (B, L, L, H) fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_fp16 = F.linear(gated, to_out_w)                # (B, L, L, D) fp16

    # Return in float32 as required by the reference implementation
    return out_fp16.to(torch.float32)
