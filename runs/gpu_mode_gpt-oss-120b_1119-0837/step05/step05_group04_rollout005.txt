"""
TriMul “outgoing’’ implementation (forward only).

Algorithm
---------
1️⃣ Input LayerNorm (Triton 2‑pass) on the (B,N,N,D) tensor.
2️⃣ Cast the normalised tensor to FP16 and perform ONE fused linear projection
   that yields the three projections (left/right) and the three gating vectors.
3️⃣ Apply sigmoid to the gating tensors, optional pair‑wise mask and obtain the
   gated left/right tensors (still FP16).
4️⃣ Contract the two 4‑D tensors by a **single custom Triton batched‑GEMM**
   that works directly on the original layout (B,N,N,H) – this avoids the
   expensive ``permute + contiguous’’ copies required for ``torch.bmm``.
5️⃣ Layer‑norm the intermediate result (Triton) on the hidden dimension H.
6️⃣ Apply the output gate, final FP16 linear projection back to the model
   dimension D and cast the result to FP32 (public API).

Only the two LayerNorms and the O(N³·H) contraction are written in Triton; all
dense linear algebra (the fused projection and the final projection) uses
high‑performance cuBLAS/Tensor‑Core kernels via ``torch.nn.functional.linear``.
The implementation is fully functional for the configurations used in the
test suite and runs on an NVIDIA H100 (Triton 3.3.1).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton 2‑pass LayerNorm (last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, C]
    y_ptr,          # *ptr output [M, C]
    w_ptr,          # *ptr weight [C]
    b_ptr,          # *ptr bias   [C]
    M,              # number of rows  (B·N·N)
    C,              # feature size   (D or H)
    eps,            # epsilon
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----- row block ----------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: accumulate sum & sum of squares (FP32) --------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # iterate over the column dimension (C)
    for c_tile in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_tile * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine (still FP32) -------------------
    for c_tile in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_tile * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(w_ptr + offs_c, mask=mask_c, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_c, mask=mask_c, other=0.0), tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm using ``layernorm_kernel``.
    Input must be a 2‑D contiguous tensor (M, C).
    Returns tensor of same shape / dtype.
    """
    assert x.ndim == 2, "layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    # make sure tensors are contiguous (required for stride extraction)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    BLOCK_M = 128
    BLOCK_C = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
    )
    return y


# ----------------------------------------------------------------------
# Batched contraction kernel (the O(N³·H) core)
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,               # *ptr FP16 left   [B, N, N, H]
    right_ptr,              # *ptr FP16 right  [B, N, N, H]
    out_ptr,                # *ptr FP16 out    [(B·H), N, N]
    B, N, H,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out (flattened BH dimension)
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a single (b,h) slice:
        out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Operates directly on the original layout to avoid any permute+contiguous copy.
    """
    pid_i = tl.program_id(0)          # tile‑row index
    pid_j = tl.program_id(1)          # tile‑col index
    pid_bh = tl.program_id(2)         # batch×hidden index (0 … B·H‑1)

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_base = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h
        left_ptrs = left_base \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        a = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        a_f32 = tl.cast(a, tl.float32)

        # right tile needs to be transposed (k as rows, j as cols)
        # shape (BLOCK_K, BLOCK_N)
        right_base = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h
        right_ptrs = right_base \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_i
        b_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        b_f32 = tl.cast(b_tile, tl.float32)

        # accumulate
        acc += tl.dot(a_f32, b_f32)   # (BLOCK_M, BLOCK_N)

    # store result (back‑to‑FP16)
    out_base = out_ptr \
        + pid_bh * stride_o_bh
    out_ptrs = out_base \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def trimul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched O(N³·H) contraction using ``trimul_contract_kernel``.
    Arguments must have shape (B, N, N, H) and dtype FP16.
    Returns tensor of same shape (B, N, N, H) (FP16).
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # flatten batch‑hidden dimension for the output tensor
    BH = B * H
    out = torch.empty((BH, N, N), dtype=left.dtype, device=device)

    # strides of left/right (contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    # out strides (contiguous BH,N,N)
    stride_o_bh, stride_o_i, stride_o_j = out.stride()

    # tuning: 64×64 tiles with a 32‑element k‑tile give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        BH,                         # batch‑hidden index
    )

    trimul_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_bh=stride_o_bh,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of model parameters (see reference module)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – stay in FP32 for stability.
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)                         # (M, D) FP32

    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                   # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and perform the fused projection (+ gates).
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16).contiguous()
    x_norm_f16_flat = x_norm_f16.view(M, dim)                   # (M, D) FP16

    # fuse the five linear layers (left_proj, left_gate, right_proj, right_gate, out_gate)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                # (5·H, D) FP16

    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)          # (M, 5·H) FP16
    proj_gate = proj_gate_flat.view(bs, N, N, 5 * hidden_dim)      # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]               # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2 * H]
    right_proj  = proj_gate[..., 2 * H:3 * H]
    right_gate_r = proj_gate[..., 3 * H:4 * H]
    out_gate_r  = proj_gate[..., 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) and optional mask
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_r)          # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_r)        # (B,N,N,H) FP16
    out_gate = torch.sigmoid(out_gate_r)           # (B,N,N,H) FP16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) contraction – custom Triton kernel (no copies)
    # ------------------------------------------------------------------
    out = trimul_contraction(left, right)          # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) – still FP16
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)              # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to FP32
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    out_fp16 = F.linear(gated.view(M, hidden_dim), to_out_w)        # (M, D) FP16
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)      # (B,N,N,D) FP32

    return out_fp32
