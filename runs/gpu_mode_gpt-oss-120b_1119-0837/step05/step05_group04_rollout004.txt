import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr,
                     M, N, eps,
                     stride_xm, stride_xn, stride_ym, stride_yn,
                     BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    # compute per-row mean & variance in FP32 two-pass
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M
    # first pass
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_n = tl.cdiv(N, BLOCK_N)
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)
    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass: normalise and affine
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = tl.cast(x, tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]
        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

                 tl.cast(y, x.dtype),
                 mask=mask_m[:, None] & mask_n[None, :])

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:

    assert x.ndim == 2
    M, N = x.shape
    y = torch.empty_like(x)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()

    BLOCK_N = 64

                           stride_ym=stride_ym,

    return y

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1️⃣ Input LayerNorm (Triton) -> shape (B, N, N, dim) (float32)
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)  # 2‑D view
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)  # (B,N,N,D)

    # 2️⃣ Switch to fp16 for the heavy path
    x_norm_h = x_norm.to(torch.float16).contiguous()
    x_norm_h_flat = x_norm_h.view(M, dim)

    # 3️⃣ Fuse 5 linear projections (proj + gates) into a single GEMM
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    # each weight is (hidden_dim, dim) (no bias)
    w_left = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # each weight already shape (hidden_dim, dim)
    combined_w = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate],
        dim=0,
    ).to(device, torch.float16)  # (5*H, D)

    # compute all five tensors in one shot (FP16)
    proj_gate = F.linear(x_norm_h_flat, combined_w)  # (M, 5*H)
    proj_gate = proj_gate.view(bs, N, N, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    left_proj = proj_gate[..., 0:H]
    left_gate_pre = proj_gate[..., H:2*H]
    right_proj = proj_gate[..., 2*H:3*H]
    right_gate_pre = proj_gate[..., 3*H:4*H]
    out_gate_pre = proj_gate[..., 4*H:5*H]

    # 4️⃣ Sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # 5️⃣ Apply mask (if any) and gating to get left / right tensors
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_h
        right = right_proj * right_gate * mask_h
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6️⃣ Batched contraction over the sequence axis k.
    # reshape to (B, H, N, N) -> (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # compute A @ B^T for each batch element (B*H of them)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (BH, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 7️⃣ Output LayerNorm (Triton) – remains in fp16 for speed
    out_flat = out.view(M, hidden_dim)  # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)  # (B,N,N,H)

    # 8️⃣ Apply output gate, final projection to original dim, back to fp32
    gated = out_norm * out_gate  # (B,N,N,H) fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    # final linear (FP16)
    gated_flat = gated.view(M, hidden_dim)
    out_fp16 = F.linear(gated_flat, to_out_w)  # (M, D)
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)

    return out_fp32

"""
Highly‑optimised forward pass for the “outgoing’’ TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Custom Triton 2‑pass LayerNorm on the last dimension of the input
   (shape [B·N·N, D]) – keeps FP32 accumulation for numerical stability.
2️⃣ Cast the normalised tensor to fp16 and fuse the five linear layers
   (left/right projections + three gating projections) into a single
   GEMM (B·N·N, D) × (D, 5·H) → (B·N·N, 5·H).
3️⃣ Slice the fused result, apply sigmoid gates and (optionally) a pair‑wise
   mask, producing the gated left/right tensors of shape (B,N,N,H).
4️⃣ Contract the two 4‑D tensors via a batched FP16 GEMM:
   for each hidden channel h we compute   Lₕ @ Rₕᵀ   (L,R∈ℝ^{N×N}).
5️⃣ Layer‑norm the intermediate result (still fp16) with a second Triton
   kernel.
6️⃣ Multiply by the output gate, apply the final fp16 linear projection
   back to the model dimension D and cast to float32 for the public API.
   (All heavy linear algebra is delegated to cuBLAS/Tensor‑Core.)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton 2‑pass LayerNorm (operates on a 2‑D tensor [M, N])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # number of rows (B·N·N)
    N,              # feature dimension (D or H)
    eps,            # epsilon for numerical stability
    stride_xm, stride_xn,   # strides of x
    stride_ym, stride_yn,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Two‑pass LayerNorm:
       1. per‑row mean & variance (FP32)
       2. normalise, apply affine, store back in original dtype
    """
    pid = tl.program_id(0)

    # ----- row block ----------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: accumulate sums ------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    n_blocks = tl.cdiv(N, BLOCK_N)
    for n_it in range(n_blocks):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine -------------------------------
    for n_it in range(n_blocks):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for ``layernorm_kernel``.
    Expects a contiguous 2‑D tensor (M, N) and returns a tensor of the same
    shape and dtype as ``x``.
    """
    assert x.ndim == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure strides are calculated on a contiguous layout
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # block sizes tuned for both D=128 and D∈{384,768}
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

# TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N]   or None
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # --------------------------------------------------------------- unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ----------------------------------------------------------- 1️⃣ LayerNorm (Triton)
    M = bs * N * N
    x_flat = input_tensor.view(M, dim)                      # (M, D) FP32
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                # (B,N,N,D) FP32

    # ------------------------------------------------------------- 2️⃣ fp16 cast
    x_norm_h = x_norm.to(torch.float16).contiguous()
    x_norm_h_flat = x_norm_h.view(M, dim)                    # (M, D) FP16

    # --------------------------------------------------- 3️⃣ fused proj+gate
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    combined_w = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate],
        dim=0,
    ).to(device, torch.float16)            # (5·H, D) in FP16

    proj_gate = F.linear(x_norm_h_flat, combined_w)           # (M, 5·H)
    proj_gate = proj_gate.view(bs, N, N, 5 * hidden_dim)     # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                         # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    # -------------------------------------------------------------- 4️⃣ gate sigmoid
    left_gate   = torch.sigmoid(left_gate_r)                 # (B,N,N,H) FP16
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)                  # (B,N,N,H) FP16

    # ----------------------------------------------------------- 5️⃣ mask + gating
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_h                  # (B,N,N,H)
        right = right_proj * right_gate * mask_h
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------- 6️⃣ contraction
    # bring hidden dim to second axis, then treat each (b,h) as a batch item
    left_perm  = left.permute(0, 3, 1, 2).contiguous()        # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()       # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                      # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                     # (BH, N, N)

    # BMM with transpose on the right side: L @ Rᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)) # (BH, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------- 7️⃣ output LayerNorm
    out_flat = out.view(M, hidden_dim)                        # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)       # (B,N,N,H) FP16

    # ---------------------------------------------------- 8️⃣ gate & final projection
    gated = out_norm * out_gate                               # (B,N,N,H) FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)

    # final linear (still FP16)
    gated_flat = gated.view(M, hidden_dim)
    out_fp16 = F.linear(gated_flat, to_out_w)                # (M, D)

    # cast back to float32 for the public API
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)
    return out_fp32
