Thus we can have:

```
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_A, batch_stride_B, batch_stride_C,
    M, N, K,
    stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    block_i = tl.program_id(0)
    block_j = tl.program_id(1)
    batch = tl.program_id(2)

    # Compute location for pointers with batch offset
    a_ptr = A_ptr + batch * batch_stride_A
    b_ptr = B_ptr + batch * batch_stride_B
    c_ptr = C_ptr + batch * batch_stride_C

    # pointers for subblocks
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ... same as matrix multiply
    a_ptrs = a_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
    b_ptrs = b_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=..., other=0.0)
        b = tl.load(b_ptrs, mask=..., other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=...)
```

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_A, batch_stride_B, batch_stride_C,
    M, N, K,
    stride_a_m, stride_a_k, stride_b_k, stride_b_n, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(2)
    block_i = tl.program_id(0)
    block_j = tl.program_id(1)

    # compute offsets
    a_batch_ptr = A_ptr + batch * batch_stride_A
    b_batch_ptr = B_ptr + batch * batch_stride_B
    c_batch_ptr = C_ptr + batch * batch_stride_C

    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    num_k = tl.cdiv(K, BLOCK_K)
    for kb in range(num_k):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A subblock [M, K]
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B subblock [K, N]
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Multiply-accumulate
        acc += tl.dot(a, b)  # a and b are at most BLOCK_M x BLOCK_K and BLOCK_K x BLOCK_N

    # Write back to C
    c = acc.to(tl.float16)

    tl.store(
        c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # dtype: we'll use fp16 for compute
    # We need to move weights to correct dtype and device
    device = input_tensor.device

    # LayerNorm (first)
    norm_weight = weights["norm.weight"].to(device)  # keep same dtype as input float32? We'll convert later.
    norm_bias = weights["norm.bias"].to(device)
    # Cast to appropriate dtype later
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=eps)
    # Cast to fp16 for heavy compute
    x_norm_h = x_norm.half()    # dtype torch.float16

    # Fused linear for left/right + gates
    lp_w = weights["left_proj.weight"].to(device).half()
    rp_w = weights["right_proj.weight"].to(device).half()
    lg_w = weights["left_gate.weight"].to(device).half()
    rg_w = weights["right_gate.weight"].to(device).half()

    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)  # shape (4*hidden_dim, dim)
    # x_norm_h_flat shape (B*L*L, dim)
    M = bs * L * L
    x_flat = x_norm_h.view(M, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_w)  # (M, 4*hidden_dim)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    left_proj = fused_out[..., 0:hidden_dim]   # B,L,L,H
    right_proj = fused_out[..., hidden_dim:2*hidden_dim]
    left_gate = torch.sigmoid(fused_out[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3*hidden_dim:4*hidden_dim])

    # Apply mask and gating
    if not nomask and mask is not None:
        mask_h = mask.to(device).half().unsqueeze(-1)  # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Permute to (B, H, L, L) for batched matmul
    left_perm = left.permute(0, 3, 1, 2)     # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).transpose(2, 3)  # (B, H, L, L) where inner dim transposed

    # reshape to batch_h, L, L
    batch_h = bs * hidden_dim
    left_mat = left_perm.reshape(batch_h, L, L)
    right_mat = right_perm.reshape(batch_h, L, L)

    # Compute batched matmul via Triton kernel
    out_mat = triton_batched_matmul(left_mat, right_mat)  # shape (batch_h, L, L), fp16

    # Reshape to (B, L, L, H)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1)  # (B, L, L, H)

    # Output layernorm using torch.nn.functional.layer_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                              weight=to_out_norm_weight,
                                              bias=to_out_norm_bias,
                                              eps=eps)

    # Output gating: compute from original normalized input
    out_gate_w = weights["out_gate.weight"].to(device).half()
    # Compute gating: out_gate = sigmoid(linear(x_norm_h_flat, out_gate_w))
    out_gate_res = torch.nn.functional.linear(x_flat, out_gate_w)  # (M, hidden_dim)
    out_gate_res = torch.sigmoid(out_gate_res).view(bs, L, L, hidden_dim)

    # Apply gating
    gated = out_norm.half() * out_gate_res.half()  # ensure fp16

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).half()
    gated_flat = gated.view(M, hidden_dim)
    out_final_h = torch.nn.functional.linear(gated_flat, to_out_w)
    out_final = out_final_h.view(bs, L, L, dim).float()

    return out_final

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_A_batch, stride_A_head, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_head, stride_B_k, stride_B_j,
    stride_C_batch, stride_C_head, stride_C_i, stride_C_j,
    M, N, K, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)  # combined batch*head index

    # Compute batch and head indices
    b = pid_h // H
    h = pid_h % H

    # Compute pointer offsets for each batch-head pair
    a_ptr = A_ptr + b * stride_A_batch + h * stride_A_head
    b_ptr = B_ptr + b * stride_B_batch + h * stride_B_head
    c_ptr = C_ptr + b * stride_C_batch + h * stride_C_head

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A sub-block
        a = tl.load(
            a_ptr + offs_i[:, None] * stride_A_i + offs_k[None, :] * stride_A_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B sub-block
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_B_k + offs_j[None, :] * stride_B_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Store result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + offs_i[:, None] * stride_C_i + offs_j[None, :] * stride_C_j,
        c,
        mask=mask_i[:, None] & mask_j[None, :],

"""
custom_kernel
==============

Forward pass of the “outgoing” TriMul operator (AlphaFold3) – fully
fused, mixed‑precision implementation.

Algorithm
---------

1. **Input LayerNorm** – use PyTorch’s highly‑optimised
   `torch.nn.functional.layer_norm` (float32 → float16 conversion after).

2. **Fused linear projections** – concatenate the four linear maps
   (left_proj, right_proj, left_gate, right_gate) and apply a single
   half‑precision GEMM:  
   `fused = x_norm @ W_fused.T`.  Slice the result and apply the sigmoids
   for the two gates.

3. **Mask & gating** – element‑wise multiplication with the optional
   mask (converted to fp16) and the gate tensors.

4. **Batched matrix multiplication** – compute the N³ contraction  
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.  
   The tensors are reshaped to `(B, H, N, N)` (no extra copies) and a
   custom Triton kernel performs a batched GEMM for all
   `batch_h = B·H` heads.  The kernel works on half‑precision data, uses
   a 32‑bit accumulator for numerical stability and writes the result
   back as fp16.

5. **Output LayerNorm** – again `torch.nn.functional.layer_norm`
   (half‑precision).

6. **Output gating** – compute `out_gate = sigmoid(x_norm @ W_out_gate.T)`.

7. **Final linear projection** – `to_out` projection (half‑precision) and
   cast the result back to float32.

The implementation keeps the heavy arithmetic in fp16, reduces memory
traffic by avoiding copies of the `(B, H, N, N)` tensors, and fuses the
two costly LayerNorms with a Triton kernel for the matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (computes C = A @ B_T for many heads)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_A_batch, stride_A_head, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_head, stride_B_k, stride_B_j,
    stride_C_batch, stride_C_head, stride_C_i, stride_C_j,
    M, N, K, H,                     # matrix dimensions (M=N=K=L) and hidden dim
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """A, B, C have shape (B, H, M, K) (A) and (B, H, K, N) (B),
    where the contraction is over K.  The kernel loops over K in blocks.
    """
    pid_i = tl.program_id(0)      # block row index
    pid_j = tl.program_id(1)      # block column index
    pid_h = tl.program_id(2)      # combined batch*head index

    # ---- decode combined index to (batch, head) ---------------------------------
    b = pid_h // H                 # batch index
    h = pid_h % H                  # head index

    # ---- compute base pointers for this (batch, head) -----------------------------
    a_base = A_ptr + b * stride_A_batch + h * stride_A_head
    b_base = B_ptr + b * stride_B_batch + h * stride_B_head
    c_base = C_ptr + b * stride_C_batch + h * stride_C_head

    # ---- row / column offsets for this block ------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ---------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_base + offs_i[:, None] * stride_A_i + offs_k[None, :] * stride_A_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_base + offs_k[:, None] * stride_B_k + offs_j[None, :] * stride_B_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write back ---------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base + offs_i[:, None] * stride_C_i + offs_j[None, :] * stride_C_j,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right_t: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape (B, H, N, N) – not required to be contiguous.
    right_t : torch.Tensor
        Tensor of shape (B, H, N, N) = right.permute(0,3,1,2).transpose(2,3).
    hidden_dim : int
        Value of H (used for index decoding inside the kernel).

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, H, N, N) – fp16.
    """
    B, H, N, _ = left.shape
    assert H == hidden_dim, "hidden_dim mismatch"
    assert right_t.shape == left.shape

    out = torch.empty_like(left)  # same dtype (fp16) and layout

    # Gather strides (in elements, not bytes)
    sA_b, sA_h, sA_i, sA_k = left.stride()
    sB_b, sB_h, sB_k, sB_j = right_t.stride()
    sC_b, sC_h, sC_i, sC_j = out.stride()

    # Block sizes – tuned for H100 fp16 Tensor Core GEMMs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    batched_gemm_kernel[grid](
        A_ptr=left,
        B_ptr=right_t,
        C_ptr=out,
        stride_A_batch=sA_b,
        stride_A_head=sA_h,
        stride_A_i=sA_i,
        stride_A_k=sA_k,
        stride_B_batch=sB_b,
        stride_B_head=sB_h,
        stride_B_k=sB_k,
        stride_B_j=sB_j,
        stride_C_batch=sC_b,
        stride_C_head=sC_h,
        stride_C_i=sC_i,
        stride_C_j=sC_j,
        M=N,
        N=N,
        K=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # good occupancy for 128×128 tiles on H100
    )
    return out

# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).half()
    norm_b = weights["norm.bias"].to(device).half()
    # torch.nn.functional.layer_norm works on half tensors directly
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )
    x_norm_h = x_norm.half()                    # [B, L, L, D] fp16

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the four linear maps (proj + gate) into a single GEMM
    # ------------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device).half()
    rp_w = weights["right_proj.weight"].to(device).half()
    lg_w = weights["left_gate.weight"].to(device).half()
    rg_w = weights["right_gate.weight"].to(device).half()
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)   # (4*H, D)

    M = bs * L * L
    x_flat = x_norm_h.view(M, dim)               # (M, D)
    fused_out = F.linear(x_flat, fused_w)        # (M, 4*H)
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣  Slice projections and gates
    # ------------------------------------------------------------------
    left_proj = fused_out[..., 0:hidden_dim]                       # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]        # (B, L, L, H)
    left_gate = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask (if present) before gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device).half().unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate          # (B, L, L, H) fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2)                 # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).transpose(2, 3)  # (B, H, N, N) = right_T

    # Perform the contraction with the Triton kernel
    out_perm = triton_batched_gemm(left_perm, right_perm, hidden_dim)  # (B, H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # still fp16

    # ------------------------------------------------------------------
    # 7️⃣  Output gating (sigmoid) – linear over the original normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).half()
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣  Apply gating and final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.half()                  # fp16
    to_out_w = weights["to_out.weight"].to(device).half()
    gated_flat = gated.view(M, hidden_dim)
    out_final_h = F.linear(gated_flat, to_out_w)        # (M, D) fp16
    out_final = out_final_h.view(bs, L, L, dim).float() # cast back to FP32

    return out_final
```
