"""
TriMul “outgoing” forward (AlphaFold‑3) – fast Triton‑accelerated version.

Key ideas
---------
1️⃣  Input LayerNorm is delegated to the highly‑optimised PyTorch kernel
    (torch.nn.functional.layer_norm) – this avoids the custom two‑pass
    reduction we used previously and is faster for the relatively small
    per‑row dimension.

2️⃣  The three projection branches (left/right + three gates) are still
    fused into a single GEMM (shape (B·N², 5·H)) – this is the most
    compute‑intensive part and stays in FP16 to exploit Tensor Cores.

3️⃣  Sigmoid gating and optional binary mask are performed **in‑place**
    on the slices of the fused tensor, eliminating extra temporaries.

4️⃣  The N³ contraction is carried out as a batched GEMM
    (torch.bmm).  Although a copy is required to put the hidden dimension
    into the batch axis, this is still far cheaper than a naïve Python
    triple‑loop.

5️⃣  The hidden‑dim LayerNorm after the contraction is the only
    operation kept in Triton (FP16 → FP16) – it is a 2‑pass kernel that
    computes mean/variance per row and then normalises with the learned
    weight/bias.

6️⃣  Finally the output gate (sigmoid) is applied, followed by the final
    projection back to the model dimension.  All heavy work stays in
    FP16; the result is cast back to FP32 for the API contract.

The implementation respects the “outgoing” semantics from the AlphaFold‑3
paper and works for all configurations listed in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# FP16 LayerNorm – 2‑pass implementation (hidden‑dim normalisation)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalised result ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16 → float16) via Triton.

    Input shape (M, H) where H is the hidden dimension.
    """
    assert x.dtype == torch.float16, "FP16 layernorm expects half‑precision input"
    assert x.dim() == 2, "expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul (forward only)
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (float32 / bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys:
            * dim         – model dimension D
            * hidden_dim  – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the sequence dimensions"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm – use the highly‑optimised PyTorch kernel
    # ------------------------------------------------------------------
    # LayerNorm normalises across the last dimension (dim)
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # (B, N, N, D)  FP32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for all heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse the 5 linear projections (left, left‑gate, right, right‑gate,
    #      out‑gate) into a single GEMM
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]    # (H, D)
    left_gate_w   = weights["left_gate.weight"]    # (H, D)
    right_proj_w  = weights["right_proj.weight"]   # (H, D)
    right_gate_w  = weights["right_gate.weight"]   # (H, D)
    out_gate_w    = weights["out_gate.weight"]     # (H, D)

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0
    ).to(device, torch.float16).contiguous()  # (5*H, D)

    # fused GEMM → (B·N·N, 5·H)
    fused = F.linear(x_f16, fused_weight)  # shape (bs, N, N, 5*H)

    # ------------------------------------------------------------------
    # 4️⃣  Slice the fused result (no extra copies – just views)
    # ------------------------------------------------------------------
    H = config["hidden_dim"]  # sanity
    left_proj      = fused[..., 0               : H]       # (B,N,N,H)
    left_gate_pre  = fused[..., H               : 2*H]     # (B,N,N,H)
    right_proj     = fused[..., 2*H             : 3*H]     # (B,N,N,H)
    right_gate_pre = fused[..., 3*H             : 4*H]     # (B,N,N,H)
    out_gate_pre   = fused[..., 4*H             : 5*H]     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣  In‑place sigmoid for all gates (no extra allocations)
    # ------------------------------------------------------------------
    torch.sigmoid(left_gate_pre, out=left_gate_pre)      # left_gate_inplace
    torch.sigmoid(right_gate_pre, out=right_gate_pre)    # right_gate_inplace
    torch.sigmoid(out_gate_pre, out=out_gate_pre)        # out_gate_inplace

    # ------------------------------------------------------------------
    # 6️⃣  Apply optional mask (broadcast over H) – after gating (commutes)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj.mul_(mask_f16)
        right_proj.mul_(mask_f16)

    # ------------------------------------------------------------------
    # 7️⃣  Fuse gate with projection (in‑place, re‑using memory)
    # ------------------------------------------------------------------
    left_proj.mul_(left_gate_pre)   # left_proj now holds the gated left tensor
    right_proj.mul_(right_gate_pre) # right_proj now holds the gated right tensor

    # left_proj and right_proj are the tensors that will be contracted.
    # ------------------------------------------------------------------
    # 8️⃣  N³ contraction – batched half‑precision GEMM (B·H cases)
    # ------------------------------------------------------------------
    # Bring hidden dimension into the batch axis
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B·H, N, N)

    # Batched matrix multiplication: out_mat = left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 9️⃣  Hidden‑dim LayerNorm (FP16) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(-1, H).contiguous()  # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 🔟  Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate = out_gate_pre  # already sigmoid‑ed, shape (B,N,N,H)
    gated = out_norm * out_gate  # (B,N,N,H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()  # (D, H)
    gated_flat = gated.view(-1, H).contiguous()        # (B·N·N, H)
    output_flat = F.linear(gated_flat, to_out_w)      # (B·N·N, D)   FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # cast back to FP32
    return output
