@triton.jit
def batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch index and tile indices
    total_blocks_m = tl.cdiv(N, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (total_blocks_m * total_blocks_n)
    block_idx = pid % (total_blocks_m * total_blocks_n)
    block_i = block_idx // total_blocks_n
    block_j = block_idx % total_blocks_n

    # offsets
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left and right tiles
        left = tl.load(left_ptr + batch*stride_left_batch
                                   + i_offset[:, None] * stride_left_i
                                   + k_offset[None, :] * stride_left_k,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_ptr + batch*stride_right_batch
                                    + k_offset[:, None] * stride_right_k  # note k as row
                                    + j_offset[None, :] * stride_right_j,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)
        # cast to fp32
        left_f32 = tl.cast(left, tl.float32)
        right_f32 = tl.cast(right, tl.float32)

        acc += tl.dot(left_f32, right_f32)

    # store output
    # result type same as left/right dtype (fp16)
    out = tl.cast(acc, left.dtype) # convert to fp16
    tl.store(out_ptr + batch*stride_out_batch
                     + i_offset[:, None] * stride_out_i
                     + j_offset[None, :] * stride_out_j,
             out,
             mask=mask_i[:, None] & mask_j[None, :])
```

@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,                     # seq_len
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_block in range(0, num_k_blocks):
        offs_k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left element: shape (BLOCK_M, BLOCK_K)
        left = tl.load(left_ptr + batch_idx * stride_left_batch
                                 + offs_i[:, None] * stride_left_i
                                 + offs_k[None, :] * stride_left_k,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        # Load right element transposed: shape (BLOCK_N, BLOCK_K)
        # original right layout: (batch, N, N)
        # We need right_T (k,j), so indices: row = k, col = j. So:
        right = tl.load(right_ptr + batch_idx * stride_right_batch
                                    + offs_k[:, None] * stride_right_k
                                    + offs_j[None, :] * stride_right_j,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Cast to fp32
        left_f32 = tl.cast(left, tl.float32)
        right_f32 = tl.cast(right, tl.float32)

        # dot: left (m,k) dot right (k,n) -> m,n
        acc += tl.dot(left_f32, right_f32)

    # Write result
    out = tl.cast(acc, left.dtype) # fp16
    tl.store(out_ptr + batch_idx * stride_out_batch
                     + offs_i[:, None] * stride_out_i
                     + offs_j[None, :] * stride_out_j,
                out,
                mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” kernel – fast forward pass.

The implementation follows the reference PyTorch module but:
*   LayerNorm (both input and post‑contraction) is written in Triton
    (2‑pass algorithm, FP32 mean/var, FP16 output) to keep the data on‑GPU.
*   The five linear layers (two projections + three gates) are fused into a
    single half‑precision GEMM (torch.nn.functional.linear).
*   The heavy O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is performed by a custom Triton batched GEMM kernel that works on the
    combined batch dimension B·H and avoids the expensive permute+contiguous
    copies required for torch.bmm.
*   The output gate, final Linear projection and FP32 cast are done in PyTorch
    after the second LayerNorm.

All heavy work runs in FP16 (inputs are cast once after the first LayerNorm);
accumulation is performed in FP32 for numerical stability.  The final
tensor matches the reference implementation up to the usual FP16 rounding error
and is returned in float32 as required by the public API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 2‑pass LayerNorm (last dimension).  Works on a 2‑D view (M, D).
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # *ptr input  [M, D]
    y_ptr,        # *ptr output [M, D]
    w_ptr,        # *ptr weight [D]
    b_ptr,        # *ptr bias   [D]
    M,            # number of rows (M = B·N·N)
    D,            # feature dimension (dim or hidden_dim)
    eps,          # epsilon
    stride_xm, stride_xd,   # strides of x
    stride_ym, stride_yd,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # row offsets for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean and variance (FP32) ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d_iters = tl.cdiv(D, BLOCK_D)
    for d_iter in range(num_d_iters):
        offs_d = tl.arange(0, BLOCK_D) + d_iter * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: normalize + affine ----------
    for d_iter in range(num_d_iters):
        offs_d = tl.arange(0, BLOCK_D) + d_iter * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(w_ptr + offs_d, mask=mask_d, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_d, mask=mask_d, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yd,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for the 2‑pass Triton LayerNorm.
    x must be a 2‑D contiguous tensor (M, D).  Returns y with the same dtype.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, D = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xd = x.stride()
    stride_ym, stride_yd = y.stride()

    BLOCK_M = 128          # enough rows for all test configs
    BLOCK_D = 64           # works for D = 128, 384, 768

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_ym=stride_ym,
        stride_yd=stride_yd,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    return y


# --------------------------------------------------------------
# Batched GEMM for the “outgoing” TriMul contraction.
# Computes, for each hidden channel h:
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def tri_gemm_outgoing_kernel(
    left_ptr,            # *ptr left  [B, N, N, H] (FP16)
    right_ptr,           # *ptr right [B, N, N, H] (FP16)
    out_ptr,             # *ptr out   [B, N, N, H] (FP16)
    N,                   # sequence length
    H,                   # hidden dimension
    stride_l_batch,      # stride over B dimension (N*N*H)
    stride_l_i,          # stride over first N
    stride_l_k,          # stride over second N
    stride_out_batch,    # stride over B dimension of output (same as left)
    stride_out_i,        # stride over first N of output
    stride_out_j,        # stride over second N of output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 0‑th axis = combined batch * hidden index (b*H + h)
    batch_h_idx = tl.program_id(0)
    pid_i = tl.program_id(1)  # row‑tile index (i)
    pid_j = tl.program_id(2)  # col‑tile index (j)

    # derive (b, h)
    b = batch_h_idx // H
    h = batch_h_idx % H

    # offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_batch  + h
    right_base = right_ptr + b * stride_l_batch  + h
    out_base   = out_ptr   + b * stride_out_batch + h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_block in range(num_k_blocks):
        offs_k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K) -> left[b, i, k, h]
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (transposed): shape (BLOCK_K, BLOCK_N) -> right[b, j, k, h]
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_l_k   # k dimension
            + offs_j[None, :] * stride_l_i,  # j maps to the first N
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate (FP32)
        acc += tl.dot(tl.cast(left, tl.float32), tl.cast(right, tl.float32))

    # store result (cast back to FP16)
    out = tl.cast(acc, tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]  (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor  [B, N, N, D] (float32)
    """
    # ----------------------------------------------------------
    # unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ----------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ----------------------------------------------------------
    M = bs * seq_len * seq_len                     # rows = B·N·N
    inp_flat = input_tensor.view(M, dim)           # (M, D) FP32

    norm_w = weights["norm.weight"].to(device, dtype)   # (D)
    norm_b = weights["norm.bias"].to(device, dtype)     # (D)

    inp_norm_flat = triton_layernorm(inp_flat, norm_w, norm_b, eps=eps)   # (M, D) FP32
    inp_norm = inp_norm_flat.view(bs, seq_len, seq_len, dim)             # (B,N,N,D) FP32

    # ----------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # ----------------------------------------------------------
    inp_norm_f16 = inp_norm.to(torch.float16)    # (B,N,N,D) FP16
    inp_norm_f16_flat = inp_norm_f16.view(M, dim)  # (M, D) FP16

    # ----------------------------------------------------------
    # 3️⃣ Fuse projections + gates into one GEMM
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]          # (H, D)
    right_proj_w  = weights["right_proj.weight"]         # (H, D)
    left_gate_w   = weights["left_gate.weight"]          # (H, D)
    right_gate_w  = weights["right_gate.weight"]         # (H, D)
    out_gate_w    = weights["out_gate.weight"]           # (H, D)

    # concatenate along the output feature dimension
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device, torch.float16)      # (5*H, D)

    # single fused linear (M, 5*H)
    proj_gate_flat = F.linear(inp_norm_f16_flat, combined_w)
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)          # (B,N,N,H) FP16
    right_gate  = torch.sigmoid(right_gate_r)         # (B,N,N,H) FP16
    out_gate    = torch.sigmoid(out_gate_r)           # (B,N,N,H) FP16

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------
    # 4️⃣ Heavy O(N³·H) contraction via a custom Triton GEMM
    # ----------------------------------------------------------
    # allocate output tensor
    out = torch.empty_like(left)                     # (B,N,N,H) FP16

    # strides of the tensors (contiguous layout)
    stride_l_batch = left.stride(0)   # N*N*H
    stride_l_i = left.stride(1)       # N*H
    stride_l_k = left.stride(2)       # H
    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    BH = bs * hidden_dim                               # combined batch dimension

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    tri_gemm_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=seq_len,
        H=hidden_dim,
        stride_l_batch=stride_l_batch,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ----------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton)
    # ----------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                # (B·N·N, H) FP16
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)   # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)     # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H) FP16
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP16

    # ----------------------------------------------------------
    # 6️⃣ Output gate + final projection back to model dimension
    # ----------------------------------------------------------
    gated = out_norm * out_gate                         # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    out_fp16 = F.linear(gated, to_out_w)               # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)              # cast to the public API dtype

    return out_fp32
