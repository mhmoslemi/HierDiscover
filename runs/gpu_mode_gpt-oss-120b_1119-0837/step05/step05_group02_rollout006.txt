def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype  # float32
    bs, N, _, dim = input_tensor.shape

    H = config["hidden_dim"]; eps=1e-5; nomask = config.get("nomask", False)

    # Input LN (FP32) with fused write to FP16:

    x_flat = input_tensor.reshape(bs*N*N, dim)

    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    # We'll implement layernorm_fp32_to_fp16 that writes half
    x_norm_fp16_flat = triton_layernorm_fp32_to_fp16(x_flat, norm_w, norm_b, eps)
    x_norm_fp16 = x_norm_fp16_flat.view(bs, N, N, dim)   # (B,N,N,D) half

    # fused linear projection with weight fused

    fused_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"], weights["right_proj.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0)
    fused_weight = fused_weight.to(device, torch.float16).contiguous()

    fused = F.linear(x_norm_fp16, fused_weight)   
    # shape (B,N,N,5*H) half

    # split
    left_proj = fused[..., :H]
    left_gate_pre = fused[..., H:2*H]
    right_proj = fused[..., 2*H:3*H]
    right_gate_pre = fused[..., 3*H:4*H]
    out_gate_pre = fused[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # (B,N,N,H), half

    # mask if needed
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1) half
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # gated projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # contraction: N³ trig

    left_perm = left.permute(0,3,1,2).contiguous() # (B, H, N, N) half
    right_perm = right.permute(0,3,1,2).contiguous()

    BH = bs * H

@triton.jit
def layernorm_fp32_to_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                                 M,N,eps,
                                 stride_xm, stride_xn,
                                 stride_ym, stride_yn,
                                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    # same as previous but y_ptr is half
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass store normalized weight
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)
        y_f32 = norm * w[None, :] + b[None, :]
        y_f16 = tl.cast(y_f32, tl.float16)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y_f16,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def layernorm_fp32_to_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                                 M, N, eps,
                                 stride_xm, stride_xn,
                                 stride_ym, stride_yn,
                                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
   pid = tl.program_id(0)
   offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
   mask_m = offs_m < M
   # sum for mean and variance
   sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
   sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)
   num_k = tl.cdiv(N, BLOCK_N)
   for k in range(num_k):
       offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
       mask_n = offs_n < N
       x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                   mask=mask_m[:, None] & mask_n[None, :],
                   other=0.0)
       sum_val += tl.sum(x, axis=1)
       sum_sq += tl.sum(x * x, axis=1)
   mean = sum_val / N
   var = sum_sq / N - mean * mean
   rstd = 1.0 / tl.sqrt(var + eps)

   for k in range(num_k):
       offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
       mask_n = offs_n < N
       x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                   mask=mask_m[:, None] & mask_n[None, :],
                   other=0.0)
       norm = (x - mean[:, None]) * rstd[:, None]   # x - mean
       w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
       b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)
       y_f32 = norm * w[None, :] + b[None, :]
       y_f16 = tl.cast(y_f32, tl.float16)
       tl.store(y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
                y_f16,
                mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def hidden_norm_gate_linear_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    weight_out_ptr,
    output_ptr,
    B, N, eps,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_weight,
    stride_norm_bias,
    stride_weight_out_d, stride_weight_out_h,
    stride_output_b, stride_output_i, stride_output_j, stride_output_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    offs = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask = offs < total

    # compute b,i,j
    NN = N * N
    b = offs // NN
    rem = offs - b * NN
    i = rem // N
    j = rem - i * N

    # h range
    h_idx = tl.arange(0, H)

    # load out values
    out_offs = (b * stride_out_b)[:, None] + (h_idx * stride_out_h)[None, :] + (i * stride_out_i)[:, None] + (j * stride_out_j)[:, None]
    out_vals = tl.load(out_ptr + out_offs, mask=mask[:, None], other=0.0)  # shape (BLOCK_M, H)

    # load out_gate values
    gate_offs = (b * stride_gate_b)[:, None] + (i * stride_gate_i)[:, None] + (j * stride_gate_j)[:, None] + (h_idx * stride_gate_h)[None, :]
    gate_vals = tl.load(out_gate_ptr + gate_offs, mask=mask[:, None], other=0.0)   # shape (BLOCK_M, H)

    # Load norm weight and bias (size H)
    norm_weight = tl.load(norm_weight_ptr + h_idx * stride_norm_weight)   # shape (H,)
    norm_bias = tl.load(norm_bias_ptr + h_idx * stride_norm_bias)         # shape (H,)

    # Cast to float32
    out_f32 = tl.cast(out_vals, tl.float32)
    gate_f32 = tl.cast(gate_vals, tl.float32)
    norm_weight_f32 = tl.cast(norm_weight, tl.float32)
    norm_bias_f32 = tl.cast(norm_bias, tl.float32)

    # Compute mean and variance per position
    sum_val = tl.sum(out_f32, axis=1)   # (BLOCK_M,)
    sum_sq = tl.sum(out_f32 * out_f32, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Normalize
    normed = (out_f32 - mean[:, None]) * rstd[:, None]   # (BLOCK_M, H)

    # Apply LN weight and bias
    normed = normed * norm_weight_f32[None, :] + norm_bias_f32[None, :]

    # Gate multiplication
    v = normed * gate_f32           # (BLOCK_M, H)

    # Loop over output channel blocks
    for d_offset in range(0, D, BLOCK_D):
        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

        # iterate over hidden channels
        for h in range(H):
            # weight column for this hidden h across block of D
            w = tl.load(weight_out_ptr + (d_offset + tl.arange(0, BLOCK_D)) * stride_weight_out_d + h * stride_weight_out_h, mask=tl.full((BLOCK_D,), True, dtype=tl.int1), other=0.0)
            w_f32 = tl.cast(w, tl.float32)  # shape (BLOCK_D,)
            v_h = v[:, h]    # shape (BLOCK_M,)
            acc += v_h[:, None] * w_f32[None, :]   # broadcast

        # write output
        d_idx = tl.arange(0, BLOCK_D)
        out_offs = (b * stride_output_b)[:, None] + (i * stride_output_i)[:, None] + (j * stride_output_j)[:, None] + ((d_offset + d_idx) * stride_output_d)[None, :]
        out_vals_f16 = tl.cast(acc, tl.float16)
        tl.store(output_ptr + out_offs, out_vals_f16, mask=mask[:, None])

"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated.

What we fuse
-------------
1️⃣ Input LayerNorm (float32 → float16) – one Triton kernel that directly
   writes half‑precision normalized activations, removing an extra FP32→FP16 copy.

2️⃣ The three projection / gate Linear layers are merged into a single GEMM
   (torch.nn.functional.linear) as in the reference implementation.

3️⃣ The N³ contraction is performed with a batched half‑precision bmm
   (torch.bmm) – the fastest path on H100 for dense matrix multiplication.

4️⃣ Hidden LayerNorm, output‑gate multiplication and the final Linear
   (hidden→model dimension) are fused into a single Triton kernel.
   This kernel reads the contraction result (shape B×H×N×N) and the
   output‑gate (shape B×N×N×H), computes per‑position LayerNorm,
   applies the gate and performs a small Gemm (H→D) with the output weight
   matrix, writing directly the final tensor (B×N×N×D) in half‑precision.
   By eliminating the intermediate tensors (out_norm, gated) we cut
   ~0.5 GB of memory traffic and reduce runtime dramatically.

The overall flow matches the reference PyTorch module but moves the heavy
memory‑bound parts into Triton kernels while keeping compute‑bound batched
matrix‑multiplication in the highly‑optimised torch.bmm implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# 1. Input LayerNorm – FP32 → FP16 (single kernel, no extra copy)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_to_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized FP16 ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y_f32 = norm * w[None, :] + b[None, :]            # FP32 result
        y_f16 = tl.cast(y_f32, tl.float16)                # down‑cast
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y_f16,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32_to_fp16(x: torch.Tensor,
                                 weight: torch.Tensor,
                                 bias: torch.Tensor,
                                 eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32 → float16) via Triton."""
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty((M, N), dtype=torch.float16, device=x.device)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_to_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# 2. Hidden LayerNorm + output‑gate + final Linear (single fused kernel)
# --------------------------------------------------------------------------
@triton.jit
def hidden_norm_gate_linear_kernel(
    # tensors
    out_ptr, out_gate_ptr,               # (B,H,N,N) and (B,N,N,H) – half
    norm_weight_ptr, norm_bias_ptr,      # (H,) – half
    weight_out_ptr,                      # (D,H) – half
    output_ptr,                          # (B,N,N,D) – half
    # scalars
    B, N, eps,
    # strides
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_weight,
    stride_norm_bias,
    stride_weight_out_d, stride_weight_out_h,
    stride_output_b, stride_output_i, stride_output_j, stride_output_d,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
    H: tl.constexpr, D: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    pos = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # (BLOCK_M,)
    mask = pos < total

    # decode (b,i,j) from linear index
    NN = N * N
    b = pos // NN
    rem = pos - b * NN
    i = rem // N
    j = rem - i * N

    # ------------------------------------------------------------------
    # Load tensors needed per position
    # ------------------------------------------------------------------
    h_idx = tl.arange(0, H)                               # (H,)

    # out_val : shape (BLOCK_M, H)
    out_offs = (b * stride_out_b)[:, None] + (h_idx * stride_out_h)[None, :] \
               + (i * stride_out_i)[:, None] + (j * stride_out_j)[:, None]
    out_val = tl.load(out_ptr + out_offs, mask=mask[:, None], other=0.0)

    # out_gate : shape (BLOCK_M, H)
    gate_offs = (b * stride_gate_b)[:, None] + (i * stride_gate_i)[:, None] \
                + (j * stride_gate_j)[:, None] + (h_idx * stride_gate_h)[None, :]
    gate_val = tl.load(out_gate_ptr + gate_offs, mask=mask[:, None], other=0.0)

    # LN weight / bias (H,)
    norm_w = tl.load(norm_weight_ptr + h_idx * stride_norm_weight)   # half
    norm_b = tl.load(norm_bias_ptr   + h_idx * stride_norm_bias)     # half

    # ------------------------------------------------------------------
    # 1️⃣ Hidden LayerNorm (float32 accumulation)
    # ------------------------------------------------------------------
    out_f32 = tl.cast(out_val, tl.float32)                     # (M, H)
    sum_val = tl.sum(out_f32, axis=1)                         # (M,)
    sum_sq  = tl.sum(out_f32 * out_f32, axis=1)               # (M,)
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    normed = (out_f32 - mean[:, None]) * rstd[:, None]        # (M, H)

    # apply LN weight & bias
    w_f32 = tl.cast(norm_w, tl.float32)                       # (H,)
    b_f32 = tl.cast(norm_b, tl.float32)                       # (H,)
    normed = normed * w_f32[None, :] + b_f32[None, :]          # (M, H)

    # ------------------------------------------------------------------
    # 2️⃣ Output gate (sigmoid already applied upstream) → element‑wise mul
    # ------------------------------------------------------------------
    gate_f32 = tl.cast(gate_val, tl.float32)                  # (M, H)
    v_f32 = normed * gate_f32                                 # (M, H)
    v_f16 = tl.cast(v_f32, tl.float16)                        # (M, H) – ready for WMMA

    # ------------------------------------------------------------------
    # 3️⃣ Final Linear (H → D) using WMMA (tl.dot)
    # ------------------------------------------------------------------
    for d_off in range(0, D, BLOCK_D):
        d_idx = tl.arange(0, BLOCK_D)                         # (BLOCK_D,)

        # weight chunk shape (H, BLOCK_D)
        weight_offs = (h_idx[:, None] * stride_weight_out_h) \
                      + ((d_off + d_idx)[None, :] * stride_weight_out_d)
        w_chunk = tl.load(weight_out_ptr + weight_offs)       # half, (H, BLOCK_D)

        # dot: (M, H) @ (H, BLOCK_D) → (M, BLOCK_D)   (FP32 accumulator)
        acc = tl.dot(v_f16, w_chunk)                         # (M, BLOCK_D), FP32

        # store result (FP16)
        out_store_offs = (b * stride_output_b)[:, None] \
                         + (i * stride_output_i)[:, None] \
                         + (j * stride_output_j)[:, None] \
                         + ((d_off + d_idx)[None, :] * stride_output_d)
        tl.store(output_ptr + out_store_offs,
                 tl.cast(acc, tl.float16),
                 mask=mask[:, None])

        weights["out_gate.weight"],
    ], dim=0).to(device, torch.float16).contiguous()    # (5*H, D)

    fused = F.linear(x_norm_fp16, fused_weight)         # (B,N,N,5*H) half

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor and apply gates / optional mask
    # ------------------------------------------------------------------
    left_proj   = fused[..., 0               : H]         # (B,N,N,H)
    left_gate_p = fused[..., H               : 2*H]
    right_proj  = fused[..., 2*H             : 3*H]
    right_gate_p= fused[..., 3*H             : 4*H]
    out_gate_p  = fused[..., 4*H             : 5*H]

    left_gate  = torch.sigmoid(left_gate_p)             # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)               # (B,N,N,H)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    left  = left_proj * left_gate                        # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction – batched half‑precision GEMM
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                  # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                 # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H,N,N) half
    # reshape to (B, H, N, N) – still half‑precision, no extra transpose copy
    out = out_mat.view(bs, H, N, N).contiguous()          # (B, H, N, N) half

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm + output‑gate + final Linear (fused)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, torch.float16).contiguous()  # (H,)
    to_out_w      = weights["to_out.weight"     ].to(device, torch.float16).contiguous()  # (D, H)

    # fused Triton kernel → final result in FP16
    final_fp16 = triton_hidden_norm_gate_linear(
        out, out_gate, to_out_norm_w, to_out_norm_b, to_out_w, eps=eps
    )   # (B,N,N,D) half

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to FP32 for the API contract
    # ------------------------------------------------------------------
    return final_fp16.to(torch.float32)
```
