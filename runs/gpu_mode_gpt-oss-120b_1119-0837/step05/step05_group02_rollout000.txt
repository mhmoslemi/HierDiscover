@triton.jit
def triton_outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, # maybe not needed H
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk, # Right shape (B*H,N,N)
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute block start indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    b = pid_bh
    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # load left tile
        offs_i = i_start + tl.arange(0, BLOCK_I)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N
        left_tile = tl.load(left_ptr + b*stride_lb + offs_i[:, None] * stride_li + offs_k[None,:] * stride_lk, mask=mask_i[:,None] & mask_k[None,:], other=0.0)
        # cast to fp32
        left_tile_f = tl.cast(left_tile, tl.float32)
        # load right tile (right matrix has shape (BH,N,N) but we need row j dimension across k)
        offs_j = j_start + tl.arange(0, BLOCK_J)
        right_tile = tl.load(right_ptr + b*stride_rb + offs_j[:, None] * stride_rj + offs_k[None,:] * stride_rk, mask=mask_j[:,None] & mask_k[None,:], other=0.0)
        right_tile_f = tl.cast(right_tile, tl.float32)
        # compute dot: left_tile_f (I,K) dotted with right_tile_f transposed (J,K) but we need sum over k of left[i,k] * right[j,k].
        # right_tile_f is shape (BLOCK_J, BLOCK_K). Need to transpose to (BLOCK_K, BLOCK_J) to dot.
        acc += tl.dot(left_tile_f, tl.trans(right_tile_f))  # shape (BLOCK_I,BLOCK_J)
    # after loop, store output
    # cast back to half
    out_tile = tl.cast(acc, tl.float16)
    # compute offsets for out
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N
    tl.store(out_ptr + b*stride_ob + offs_i[:,None] * stride_oi + offs_j[None,:] * stride_oj, out_tile, mask=mask_i[:,None] & mask_j[None,:])

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style)
----------------------------------------------------
The implementation follows the PyTorch reference while off‑loading the two
LayerNorms and the N³ contraction to custom Triton kernels.

1️⃣  Input LayerNorm (float‑32) – a two‑pass kernel that computes mean/var
    per row and writes the normalized + affine transformed output.

2️⃣  All heavy linear work runs in float‑16: a single fused Linear produces
    the concatenated tensors
        [left_proj, left_gate, right_proj, right_gate, out_gate]
    of shape (...,5·H).  This saves two extra passes over the large
    input tensor.

3️⃣  Optional binary mask (if present) is applied directly to the projected
    tensors before gating.

4️⃣  Gating uses a sigmoid per sub‑tensor.

5️⃣  Core contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is executed by a custom Triton kernel that treats each head (h) of each
    batch as a separate matrix‑multiply: (BH,N,N) × (BH,N,N)ᵀ → (BH,N,N).
    The kernel tiles the i, j and k dimensions (64×64×64) and uses
    half‑precision tensor‑core dot products with float‑32 accumulation.

6️⃣  Hidden‑dim LayerNorm (float‑16) – another two‑pass Triton kernel.

7️⃣  The result is gated by `out_gate`, linearly projected back to the model
    dimension and finally cast to float‑32.

All operations are memory‑coalesced and the kernels are written to make
full use of H100 Tensor‑Core capabilities.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------------------- first pass: mean / var ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------------------- second pass: write normalized -------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton.
    Input/Output shape: (M, N) where N is the feature dimension.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------------------
# N³ contraction – custom Triton kernel (outgoing TriMul)
# --------------------------------------------------------------------------
@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    # stride specifics for left (BH,N,N)
    stride_lb, stride_li, stride_lk,
    # stride specifics for right (BH,N,N)
    stride_rb, stride_rj, stride_rk,
    # stride specifics for out (BH,N,N)
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids: 0 → batch×head, 1 → i‑tile, 2 → j‑tile
    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    bh = pid_bh

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    # Iterate over K dimension in tiles
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K

        offs_i = i_start + tl.arange(0, BLOCK_I)
        offs_j = j_start + tl.arange(0, BLOCK_J)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left tile: (BLOCK_I, BLOCK_K)
        left_tile = tl.load(
            left_ptr + bh * stride_lb
                     + offs_i[:, None] * stride_li
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )

        # right tile: (BLOCK_J, BLOCK_K)
        right_tile = tl.load(
            right_ptr + bh * stride_rb
                      + offs_j[:, None] * stride_rj
                      + offs_k[None, :] * stride_rk,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )

        # half‑precision tensor‑core dot product; accumulation in FP32
        prod = tl.dot(left_tile, tl.trans(right_tile))          # (I,J) half
        acc += tl.cast(prod, tl.float32)

    # --------------------------------------------------------------
    # Store the result (cast back to half)
    # --------------------------------------------------------------
    out_tile = tl.cast(acc, tl.float16)

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    tl.store(
        out_ptr + bh * stride_ob
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# --------------------------------------------------------------------------
# Entry point – combines everything
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (float32 / bool) or None
        - weights      : dict of model parameters (all tensors on the same device)
        - config       : dict with keys:
            * dim        – model dimension D
            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise dimensions must be equal"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – custom Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N                                   # total rows after flattening
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)        # (B, N, N, D)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the three projections + three gates + out‑gate into one GEMM
    # ------------------------------------------------------------------
    # Weight tensors are all (H, D) in the reference implementation.
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device, torch.float16).contiguous()      # (5·H, D)

    # Shape after linear: (B, N, N, 5·H)
    fused = F.linear(x_f16, fused_weight)          # bias‑free

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused tensor into its logical components
    # ------------------------------------------------------------------
    left_proj      = fused[..., 0               : H]
    left_gate_pre  = fused[..., H               : 2*H]
    right_proj     = fused[..., 2*H             : 3*H]
    right_gate_pre = fused[..., 3*H             : 4*H]
    out_gate_pre   = fused[..., 4*H             : 5*H]

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 7️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8️⃣ N³ contraction via custom Triton kernel
    # ------------------------------------------------------------------
    # Rearrange to (B·H, N, N) to treat each head as an independent matrix
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N).contiguous()   # (BH, N, N) half
    right_mat = right_perm.view(BH, N, N).contiguous()  # (BH, N, N) half

    out_mat = torch.empty_like(left_mat)                # (BH, N, N) half

    # stride extraction (all tensors are contiguous, so strides are simple)
    stride_lb, stride_li, stride_lk = left_mat.stride()    # (N*N, N, 1)
    stride_rb, stride_rj, stride_rk = right_mat.stride()
    stride_ob, stride_oi, stride_oj = out_mat.stride()

    # Triton launch configuration
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64
    grid = (
        BH,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    outgoing_trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_rj, stride_rk,
        stride_ob, stride_oi, stride_oj,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,                      # tuning knob (H100 handles 4‑8 well)
    )

    # Reshape back to (B, N, N, H) – still FP16
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Hidden‑dim LayerNorm (FP16) – custom Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H).contiguous()
    gated = out_norm * out_gate_flat                     # element‑wise, still FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()  # (D, H)
    output_flat = F.linear(gated, to_out_w)            # (M, D) half

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32 tensor
    return output
