def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute total blocks per bh
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_blocks_m * num_blocks_n

    bh_idx = pid // blocks_per_head
    block_rem = pid % blocks_per_head
    pid_m = block_rem // num_blocks_n
    pid_n = block_rem % num_blocks_n

    # compute batch and head indices
    b_idx = bh_idx // H
    h_idx = bh_idx % H

    # offsets for i, j, k
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for this batch and head
    base_l = b_idx * stride_l_b + h_idx * stride_l_h
    base_r = b_idx * stride_r_b + h_idx * stride_r_h
    base_o = b_idx * stride_o_b + h_idx * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    num_k = tl.cdiv(N, BLOCK_K)

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

"""
Optimized Triton implementation of the outgoing TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (float32) – Triton kernel.
2️⃣ Cast the normalized tensor to fp16 for all heavy computation.
3️⃣ Fuse each projection + its gate into a single linear layer (2·H channels, split later).
4️⃣ Optional mask is applied to the projected tensors (before gating).
5️⃣ Gated left/right tensors are obtained.
6️⃣ Core N³ contraction (out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h])
   is performed by a custom strided‑batched GEMM Triton kernel that works on the
   original layout [B,N,N,H] – no expensive permute‑contiguous copies.
7️⃣ Hidden‑dim LayerNorm (fp16) – Triton kernel.
8️⃣ Element‑wise multiplication with the output‑gate.
9️⃣ Final linear projection back to the model dimension (fp16 matmul) and cast to float32.

Only the two LayerNorm passes and the N³ contraction are written in Triton;
all other operations use highly‑optimized torch kernels.
"""

@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # row offset for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: sum & sum_sq ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) – expects a 2‑D tensor (M, N)."""
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# LayerNorm – FP16 (custom kernel)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float16) – expects a 2‑D tensor (M, N)."""
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# Core N³ contraction – fused batched GEMM (Triton)
# --------------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """ Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] """
    pid = tl.program_id(0)

    # number of tiles along i/j dimensions
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_tiles_i * num_tiles_j

    # derive batch*head index and tile coordinates
    bh_idx = pid // tiles_per_head
    tile_idx = pid % tiles_per_head
    pid_i = tile_idx // num_tiles_j
    pid_j = tile_idx % num_tiles_j

    # split bh_idx into batch and head
    b_idx = bh_idx // H
    h_idx = bh_idx % H

    # offsets for i, j and the masks
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (b, h) pair
    base_l = b_idx * stride_l_b + h_idx * stride_l_h
    base_r = b_idx * stride_r_b + h_idx * stride_r_h
    base_o = b_idx * stride_o_b + h_idx * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (i , k)
        a_ptr = left_ptr + base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile (k , j) – note the transpose in memory accesses
        b_ptr = right_ptr + base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused multiply‑add (accumulator stays FP32)
        acc += tl.dot(a, b)

    # write out the result as fp16
    out_tile = acc.to(tl.float16)
    out_ptr_tile = out_ptr + base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptr_tile,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Input tensors must have shape (B, N, N, H) and dtype torch.float16.
    Returns a tensor of the same shape and dtype.
    """
    assert left.shape == right.shape, "left/right must share shape"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # ensure contiguous layout – required for correct stride calculation
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides for the 4‑D layout (B, N, N, H)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BH = B * H
    tiles_i = triton.cdiv(N, 64)   # BLOCK_M
    tiles_j = triton.cdiv(N, 64)   # BLOCK_N
    total_progs = BH * tiles_i * tiles_j

    grid = (total_progs,)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=s_lb,
        stride_l_i=s_li,
        stride_l_k=s_lk,
        stride_l_h=s_lh,
        stride_r_b=s_rb,
        stride_r_i=s_ri,
        stride_r_k=s_rk,
        stride_r_h=s_rh,
        stride_o_b=s_ob,
        stride_o_i=s_oi,
        stride_o_j=s_oj,
        stride_o_h=s_oh,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=64,
    )
    return out

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32/bool) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    # flatten to (B·N·N, dim) for the 2‑D kernel
    M = bs * N * N
    inp_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    inp_norm_flat = triton_layernorm_fp32(inp_flat, norm_weight, norm_bias, eps=eps)
    inp_norm = inp_norm_flat.view(bs, N, N, dim)   # (B,N,N,dim) – still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_h = inp_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left/right sides
    # ------------------------------------------------------------------
    # combine proj & gate weights so that a single linear gives [proj; gate]
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2·H, dim)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2·H, dim)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, dim)

    # one linear per side = (B,N,N,2·H)
    left_comb  = F.linear(x_h, left_comb_w)
    right_comb = F.linear(x_h, right_comb_w)

    # split projection and gate streams
    left_proj      = left_comb[..., :hidden_dim]   # (B,N,N,H)
    left_gate_pre  = left_comb[..., hidden_dim:]   # (B,N,N,H)
    right_proj     = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate       # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction – custom Triton GEMM (no permutes)
    # ------------------------------------------------------------------
    out = tri_mul_outgoing_fp16(left, right)      # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (fp16) – Triton
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, hidden_dim)          # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M,H) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, hidden_dim)      # (M,H) fp16
    gated = out_norm * out_gate_flat                     # (M,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    output_flat = F.linear(gated, to_out_w)                         # (M, dim) fp16
    output = output_flat.view(bs, N, N, dim).float()                # cast back to FP32

    return output
