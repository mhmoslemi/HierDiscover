"""
TriMul “outgoing’’ forward pass – Triton‑accelerated layer‑norms,
fused linear/gating, batched GEMM contraction and final projection.

Algorithm
---------
1️⃣ 2‑pass layer‑norm (FP32 accumulation, FP16 output) on the input
   tensor of shape (B·N·N, D).  Implemented in Triton.

2️⃣ Cast the normalised tensor to FP16 and perform a *single* fused
   linear projection that yields the three projections (left/right) and
   the three gating vectors (left_gate, right_gate, out_gate).  The
   weight matrices are concatenated and a single `F.linear` call is
   used – this reduces the number of reads of the huge (B·N·N, D)
   matrix.

3️⃣ Apply the sigmoid to the gating tensors, optionally multiply by the
   pair‑wise mask, and obtain the gated left/right tensors.

4️⃣ Contract the two 4‑D tensors using a batched GEMM:
   left_perm  = (B, H, N, N) → (B·H, N, N)
   right_perm = (B, H, N, N) → (B·H, N, N)
   out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))
   → (B·H, N, N) which is reshaped back to (B, N, N, H).

5️⃣ Second Triton layer‑norm on the contraction result (still FP16).

6️⃣ Multiply by `out_gate`, apply the final linear projection back to the
   model dimension `D`, and cast the result to float32 for the public API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *ptr input  [M, N]
    y_ptr,          # *ptr output [M, N]
    w_ptr,          # *ptr weight [N]
    b_ptr,          # *ptr bias   [N]
    M,              # number of rows   (= B·N·N)
    N,              # feature dimension (= D or H)
    eps,            # epsilon for numerical stability
    stride_xm, stride_xn,   # strides of x
    stride_ym, stride_yn,   # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    Two‑pass LayerNorm:
      1. Compute per‑row mean & variance (FP32).
      2. Normalise, apply affine transform, store in original dtype.
    """
    pid = tl.program_id(0)

    # ----- block of rows -----------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: accumulate sums ---------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine -----------------------------
    for n_it in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n_it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalisation (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        # store back using the original dtype of x
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for `layernorm_kernel`.
    Input ``x`` must be a contiguous 2‑D tensor (M, N).
    Returns a tensor of the same shape and dtype as ``x``.
    """
    assert x.ndim == 2, "layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Block sizes tuned for both small and large feature dimensions.
    BLOCK_M = 256   # more rows per program enables fewer launches
    BLOCK_N = 64    # works well for D ∈ {128,384,768}

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y

    # ------------------------------------------------------------------

    x_norm_f16_flat = x_norm_f16.view(M, dim)  # (M, D) – FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (projections + gates) into one GEMM.
    # ------------------------------------------------------------------
    # Grab the five weight tensors (FP16) – all have shape (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                # (5·H, D) – FP16

    # (M, 5·H) – FP16
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)

    # ------------------------------------------------------------------
    # 4️⃣ Slice, apply sigmoid to gates, and optionally apply mask.
    # ------------------------------------------------------------------
    # reshape to (B,N,N,5·H)
    proj_gate = proj_gate_flat.view(bs, N, N, 5 * hidden_dim)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_r)          # (B,N,N,H) – FP16
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)           # (B,N,N,H) – FP16

    if not nomask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] (FP16)
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left  = left_proj * left_gate * mask_f16     # (B,N,N,H)
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis for an easy view
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim                                 # B·H
    left_mat  = left_perm.view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                # (B·H, N, N)

    # batched GEMM using Tensor‑Core (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – still FP16.
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                       # (M, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)      # (B,N,N,H) – FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection, and cast back to FP32.
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B,N,N,H) – FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)

    # final linear (FP16) – (M, D)
    out_fp16 = F.linear(gated.view(M, hidden_dim), to_out_w)

    # Cast the public output back to float32
    out_fp32 = out_fp16.view(bs, N, N, dim).to(torch.float32)

    return out_fp32
```
