"""
Highly‑optimized outgoing TriMul (Tensor‑wise multiplicative update) forward pass.

Algorithm overview
------------------
1️⃣ Input LayerNorm – 2‑pass FP32 reduction in a fused Triton kernel, FP16
   output (shape B·N·N × dim).

2️⃣ Fused projection – a single half‑precision linear layer that simultaneously
   computes left, right, left‑gate, right‑gate and out‑gate:
   (B·N·N, dim) × (dim, 5·H) → (B·N·N, 5·H).

3️⃣ In‑place sigmoid gating & optional binary mask (broadcast over H).

4️⃣ Core contraction – batched half‑precision GEMM:
   left[b,h] : (N, N)   = left_proj[b,:, :, h]
   right_t[b,h] : (N, N) = right_proj[b,:, :, h] with the 2‑nd axis swapped.
   out[b,h] = left[b,h] @ right_t[b,h]   → (N, N, H) per batch.

5️⃣ Output LayerNorm – same Triton kernel as step 1 (now over the hidden dim).

6️⃣ Apply the out‑gate, final half‑precision linear projection back to `dim`,
   and cast to FP32.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly).  The two
LayerNorms use FP32 statistics for numerical stability, but the intermediate
tensors stay in half‑precision to keep memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,  # *ptr input  [M, C] (float16/float32)
    y_ptr,  # *ptr output [M, C] (float16)
    w_ptr,  # *ptr weight  [C]    (float16)
    b_ptr,  # *ptr bias    [C]    (float16)
    M, C, eps,
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = tl.arange(0, BLOCK_M) + pid * BLOCK_M
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean / variance in FP32
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine (output FP16)
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) * rstd   – FP32
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w + b                # FP32
        y = y.to(tl.float16)            # cast back to half

        tl.store(
            y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc),
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5,
                          BLOCK_M: int = 256,
                          BLOCK_C: int = 128) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor ``x`` (M×C).
    ``x`` may be FP32 or FP16; the output is FP16.
    """
    assert x.ndim == 2, "triton_layernorm_fp16 expects a 2‑D tensor"
    M, C = x.shape

    y = torch.empty((M, C), dtype=weight.dtype, device=x.device)

    # guarantee contiguity (required for correct stride handling)
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)
    _layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=8,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Entry point for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from PyTorch parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0️⃣ LayerNorm parameters (cast once)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 stats → FP16 tensor)
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(-1, dim).contiguous()          # (B·N·N, dim)
    x_norm = triton_layernorm_fp16(
        x_flat,
        norm_weight,
        norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                                            # (B·N·N, dim), FP16
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)               # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single GEMM
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5·H, dim)

    # (B·N·N, 5·H)  – half‑precision GEMM
    proj = F.linear(x_norm.view(-1, dim), combined_weight)

    # --------------------------------------------------------------
    # 3️⃣ Split back into the five tensors and reshape
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) – broadcast on hidden dimension
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating – Sigmoid + in‑place multiplication
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left_raw.mul_(left_gate)   # in‑place: left = left_raw * sigmoid(...)
    right_raw.mul_(right_gate)

    # tidy up RAM we no longer need
    del left_gate_raw, right_gate_raw, left_gate, right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction: batched GEMM over hidden dimension
    # --------------------------------------------------------------
    # shape -> (B·H, N, N) for torch.bmm
    left_b = left_raw.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim,
                                                            seq_len,
                                                            seq_len)

    # right needs a swap of the two sequence axes (i ↔ j) before batching
    right_t = right_raw.permute(0, 2, 1, 3).contiguous()       # (B, N, N, H) with i↔k
    right_b = right_t.permute(0, 3, 1, 2).contiguous().view(
        bs * hidden_dim, seq_len, seq_len
    )

    # Batched half‑precision GEMM – (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_b = torch.bmm(left_b, right_b)   # no transpose needed thanks to right_t

    # Restore the original 4‑D ordering
    out = out_b.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    # (B, N, N, H)

    # free large temporaries early
    del left_raw, right_raw, left_b, right_b, right_t

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 stats → FP16)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out_flat = out.view(-1, hidden_dim)                               # (B·N·N, H)
    out_norm = triton_layernorm_fp16(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                                                  # (B·N·N, H), FP16
    out_norm = out_norm.view(bs, seq_len, seq_len, hidden_dim)        # (B,N,N,H)

    # --------------------------------------------------------------
    # 8️⃣ Gating, final linear projection & cast to FP32
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)                # (B,N,N,H)
    out_norm.mul_(out_gate)                               # in‑place gating

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_half = F.linear(out_norm.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim)
    final = final_half.view(bs, seq_len, seq_len, dim).float()            # FP32 output

    return final
