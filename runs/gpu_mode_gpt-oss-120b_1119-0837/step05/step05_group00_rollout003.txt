    a_ptr, b_ptr, c_ptr,
    M, N, K,

    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # offsets
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb

    x_ptr,               # *ptr to input  [M, C]  (fp16 or fp32)
    y_ptr,               # *ptr to output [M, C]  (fp16)
    weight_ptr, bias_ptr,  # *ptr to LN weight & bias [C] (fp16)
    M, C, eps,           # tensor dimensions
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: mean & variance (FP32) ----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: normalise + affine (output FP16) ----
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w + b
        y = y.to(tl.float16)

        tl.store(
            y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc),
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5,
                          BLOCK_M: int = 256,
                          BLOCK_C: int = 128) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a 2‑D tensor (M×C) using a fused
    Triton kernel.  Input may be FP32/FP16, output is FP16.
    """
    assert x.dim() == 2, "layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty((M, C), dtype=weight.dtype, device=x.device)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)
    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=8,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ  (A,B : [B, N, N]  fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    N,                # N == K (square matrices)
    stride_ab, stride_ai, stride_ak,   # A: batch, rows (i), cols (k)
    stride_bb, stride_bk, stride_bn,   # B (transposed view): batch, rows (k), cols (j)
    stride_cb, stride_ci, stride_cj,   # C: batch, rows (i), cols (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)      # block along i (rows)
    pid_j = tl.program_id(1)      # block along j (cols)
    pid_b = tl.program_id(2)      # batch index (B * hidden_dim)

    # offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]
        a = tl.load(
            a_batch + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[k, j]  (i.e. B[j, k] in original layout)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # FP16 dot, accumulation in FP32

    # write back the result (FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj),
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def triton_batched_gemm_transpose(a: torch.Tensor,
                                  b: torch.Tensor,
                                  BLOCK_M: int = 128,
                                  BLOCK_N: int = 128,
                                  BLOCK_K: int = 32,
                                  num_warps: int = 8,
                                  num_stages: int = 3) -> torch.Tensor:
    """
    Batched matrix multiplication producing C = A @ Bᵀ.
    All tensors must be FP16, contiguous and have shape (B, N, N).
    Returns a tensor of shape (B, N, N) in FP16.
    """
    assert a.shape == b.shape, "A and B must have identical shapes"
    batch, N, K = a.shape
    assert K == N, "Square matrices expected"

    c = torch.empty_like(a)

    # strides (in elements)
    stride_ab, stride_ai, stride_ak = a.stride()
    stride_bb, stride_bk, stride_bn = b.stride()          # note: bn = stride over original rows (j)
    stride_cb, stride_ci, stride_cj = c.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑blocks
        triton.cdiv(N, BLOCK_N),   # j‑blocks
        batch,                     # one block per batch (B*hidden_dim)
    )

    batched_gemm_transpose_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        N=N,
        stride_ab=stride_ab,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cb=stride_cb,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=num_stages,
    )
    return c


# ----------------------------------------------------------------------
# Entry point: outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32) or ``None``
        - weights_dict : dict of model parameters (Tensor)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------- unpack ---------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- LayerNorm 1 ---------------------------
    # Convert LN parameters to FP16 once (they are tiny)
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float16)

    x_flat = input_tensor.reshape(-1, dim).contiguous()          # (B·N·N, dim)  FP32
    x_norm = triton_layernorm_fp16(
        x_flat,
        norm_weight,
        norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )                                                            # (B·N·N, dim)  FP16
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)               # (B, N, N, dim)

    # --------------------------- fused linear projections ----------
    # Stack all five linear weights → shape (5*H, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                           # (5·H, dim)

    proj = F.linear(x_norm.view(-1, dim), combined_weight)       # (B·N·N, 5·H)  FP16

    # split back
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------- mask (optional) -------------------
    if (not nomask) and (mask_tensor is not None):
        # mask broadcast over hidden dimension
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------- gating ----------------------------
    left_gate  = torch.sigmoid(left_gate_raw)      # (B,N,N,H)  FP16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw * left_gate
    right = right_raw * right_gate

    # --------------------------- core contraction -------------------
    # Rearrange to (B*H, N, N) for the batched GEMM kernel
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)   # (B·H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)  # (B·H, N, N)

    # Batched matmul: left @ rightᵀ  →  (B·H, N, N)
    out_flat = triton_batched_gemm_transpose(
        left_flat,
        right_flat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
        num_warps=8,
        num_stages=3,
    )   # FP16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # --------------------------- LayerNorm 2 -----------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out_flat2 = out.view(-1, hidden_dim).contiguous()
    out_norm = triton_layernorm_fp16(
        out_flat2,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )   # (B·N·N, H)  FP16
    out_norm = out_norm.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # --------------------------- output gate + final projection ---
    gated = out_norm * out_gate                               # (B,N,N,H)  FP16
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_half = F.linear(gated.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim) FP16
    final = final_half.view(bs, seq_len, seq_len, dim).float()        # cast to FP32

    return final
