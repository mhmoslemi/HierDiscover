"""
custom_kernel: Outgoing TriMul (Tensor‑wise multiplicative update) forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32 reduction, FP16 output) – Triton fused kernel.
2️⃣ Fuse the five linear projections (two value projs, three gates) into one
   half‑precision GEMM.
3️⃣ Apply the optional mask (broadcast over hidden dim) and sigmoid gates.
4️⃣ Core contraction:   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 GEMM using a custom Triton kernel
   (no extra permute/contiguous copies).
5️⃣ Output LayerNorm (FP32 reduction, FP16 output) – same Triton kernel.
6️⃣ Apply the output gate and final projection back to the model dimension.
7️⃣ Cast the result to FP32 (required by downstream code).

All heavy arithmetic stays in FP16 (Tensor‑Core friendly).  Two Triton kernels
are used: a fused LayerNorm and the contraction kernel.  The implementation
covers all test configurations (various seq_len, batch size, dim and hidden
dim, with or without mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # [M, C] input (fp16 or fp32)
    y_ptr,                     # [M, C] output (fp16)
    weight_ptr, bias_ptr,      # [C] (fp16)
    M, C, eps,                 # scalar meta‑parameters
    stride_xm, stride_xc,      # input strides
    stride_ym, stride_yc,      # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offset for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & var (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine (FP16 output)
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C

        x = tl.load(
            x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc),
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = (norm * w + b).to(tl.float16)

        tl.store(
            y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc),
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm_fp16(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 256,
    BLOCK_C: int = 128,
) -> torch.Tensor:
    """
    2‑D LayerNorm over the last dimension using the fused Triton kernel.
    Input may be FP16 or FP32; output is FP16.
    """
    assert x.dim() == 2, "triton_layernorm_fp16 expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty((M, C), dtype=weight.dtype, device=x.device)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)
    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=8,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Triton batched contraction kernel for the core TriMul operation
# ----------------------------------------------------------------------
@triton.jit
def trimul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N, hidden_dim,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # right strides (j uses the same stride as left's i)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # output row block (i)
    pid_j = tl.program_id(1)   # output col block (j)
    pid_batch = tl.program_id(2)  # combined (batch, hidden)

    # ---- decode combined batch index into (b, h) ----
    hidden_val = hidden_dim
    b = pid_batch // hidden_val
    h = pid_batch % hidden_val

    # ---- compute coordinate offsets ----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over the reduction dimension k ----
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + (
            b * stride_left_b + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + (
            b * stride_right_b + h * stride_right_h
            + offs_j[None, :] * stride_right_j
            + offs_k[:, None] * stride_right_k
        )
        b_tile = tl.load(
            b_ptrs,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        ).to(tl.float32)

        # accumulate block‑wise matmul
        acc += tl.dot(a, b_tile)   # (M×K)·(K×N) → (M×N)

    # ---- write out the result (cast back to fp16) ----
    out_ptrs = out_ptr + (
        b * stride_out_b + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping from parameter names to torch.Tensors
        - config_dict  : dictionary with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Cast LayerNorm parameters to FP16 (small tensors)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 reduction, FP16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(-1, dim).contiguous()  # (B·N·N, dim)  float32
    x_norm = triton_layernorm_fp16(
        x_flat,
        norm_weight,
        norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )  # (B·N·N, dim) fp16
    x_norm = x_norm.view(bs, seq_len, seq_len, dim)      # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left, right, three gates)
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden_dim, dim)
            weights["right_proj.weight"],  # (hidden_dim, dim)
            weights["left_gate.weight"],   # (hidden_dim, dim)
            weights["right_gate.weight"],  # (hidden_dim, dim)
            weights["out_gate.weight"],    # (hidden_dim, dim)
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5·hidden_dim, dim)

    proj = F.linear(x_norm.view(-1, dim), combined_weight)  # (B·N·N, 5·hidden_dim) fp16

    # --------------------------------------------------------------
    # 3️⃣ Split back into individual tensors
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) – broadcast over hidden dimension
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – stays in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # (B,N,N,hidden_dim) fp16
    right_gate = torch.sigmoid(right_gate_raw)  # (B,N,N,hidden_dim) fp16
    out_gate   = torch.sigmoid(out_gate_raw)    # (B,N,N,hidden_dim) fp16

    left  = left_raw * left_gate   # (B,N,N,hidden_dim) fp16
    right = right_raw * right_gate # (B,N,N,hidden_dim) fp16

    # --------------------------------------------------------------
    # 6️⃣ Core contraction via custom Triton kernel (no extra copies)
    # --------------------------------------------------------------
    out = torch.empty_like(left)          # allocate output tensor (fp16)
    # launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,
    )
    trimul_contraction_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        batch=bs * hidden_dim,
        N=seq_len,
        hidden_dim=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_j=right.stride(1),   # column stride (same as left i‑stride)
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 reduction, FP16 output)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out_flat = out.view(-1, hidden_dim)  # (B·N·N, hidden_dim)
    out_norm = triton_layernorm_fp16(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=256,
        BLOCK_C=128,
    )  # (B·N·N, hidden_dim) fp16
    out_norm = out_norm.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,hidden_dim) fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate  # (B,N,N,hidden_dim) fp16

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, hidden_dim)

    final_half = F.linear(gated.view(-1, hidden_dim), to_out_weight)  # (B·N·N, dim) fp16
    final = final_half.view(bs, seq_len, seq_len, dim).float()         # cast to fp32
    return final
