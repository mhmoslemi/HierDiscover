def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    H = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (fast built‑in)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # (B,N,N,D) float32

    # 2 Cast to fp16
    x_f16 = x_norm.half()

    # 3. fused linear: left_proj, left_gate, right_proj, right_gate, out_gate
    # Gather weights (all shape (H, dim))
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device, torch.float16).contiguous()
    # shape (5*H, dim)
    # linear: (B*N*N, dim) @ (dim, 5*H).T -> (B*N*N, 5*H)
    x_flat = x_f16.reshape(bs * N * N, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (M,5*H)
    # reshape back
    fused_out = fused_out.view(bs, N, N, 5 * H)

    # 4. split
    left_proj = fused_out[..., 0:H]
    left_gate_pre = fused_out[..., H:2 * H]
    right_proj = fused_out[..., 2 * H:3 * H]
    right_gate_pre = fused_out[..., 3 * H:4 * H]
    out_gate_pre = fused_out[..., 4 * H:5 * H]

    # 5. sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # 6. optional mask
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # 7. apply gates
    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # 8. N³ contraction: batch matmul per head
    # rearrange to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    # compute left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 9. hidden LayerNorm using Triton (fp16)
    out_flat = out.view(bs * N * N, H).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # 10. output gating and projection
    # multiply gate (fp16) - need to align shape
    out_gate_flat = out_gate.view(bs * N * N, H).contiguous().half()
    gated = out_norm * out_gate_flat

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    output_flat = torch.nn.functional.linear(gated, to_out_w)  # (M, dim) fp16
    output = output_flat.view(bs, N, N, dim).float()
    return output

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3 style) in Triton+PyTorch

Algorithm
---------
1️⃣ Input LayerNorm (float‑32) – use the highly‑optimised torch.nn.functional.layer_norm.
2️⃣ Cast the normalised tensor to fp16 (half‑precision) for the heavy work.
3️⃣ Fuse the three linear projections (left_proj, right_proj, out_gate) *and* the
   three gating linear layers into a single GEMM → tensor of shape …×5·H.
4️⃣ Split the fused tensor into (left_proj, left_gate, right_proj,
   right_gate, out_gate).  Apply sigmoid to the gate sub‑tensors.
5️⃣ (Optional) apply the binary mask to the two projected tensors.
6️⃣ Compute the gated tensors `left = left_proj * left_gate`,
   `right = right_proj * right_gate`.
7️⃣ N³ contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]  is
   performed as a batched half‑precision GEMM via `torch.bmm`.
8️⃣ Hidden‑dim LayerNorm (fp16→fp16) – custom Triton kernel (2‑pass, block‑wise).
9️⃣ Multiply by the output gate (sigmoid(out_gate)).
🔟 Final linear projection back to the model dimension (half‑precision GEMM) and
   cast the result to float‑32.

Only the hidden‑dim LayerNorm is implemented in Triton; all other expensive
operations use PyTorch’s native CUDA kernels, which on an H100 give > 1 TFLOP/s
for the N³ contraction and keep the total latency well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float16 → float16 (used on the hidden dimension)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---- first pass: compute mean and variance (in float32) ----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # load a (BLOCK_M × BLOCK_N) tile of the input (half)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: write the normalised values ----
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        # LayerNorm affine parameters are stored as half – convert to half for the final store
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)
        y = tl.cast(norm_f32, tl.float16) * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm (float16→float16) via Triton.
    Expected input shape (M, N) where N is the hidden feature dimension.
    """
    assert x.dtype == torch.float16, "fp16 LayerNorm kernel expects half‑precision input"
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Make sure everything is contiguous
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tune the block sizes – N (hidden_dim) ≤ 128, so a single loop suffices
    BLOCK_M = 128
    BLOCK_N = 128

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# --------------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - mask         : torch.Tensor[B, N, N] (bool / float) or None
        - weights      : dict of model tensors
        - config       : dict with at least the keys

            * hidden_dim – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "expected square pairwise tensor"
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float‑32) – use the highly‑optimised builtin
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # (B, N, N, D), float32

    # ------------------------------------------------------------------
    x_f16 = x_norm.half()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse 5 linear maps (proj + gate) into one GEMM
    # ------------------------------------------------------------------
    # All linear weights have shape (H, D) and have no bias.
    left_proj_w  = weights["left_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device, torch.float16).contiguous()          # (5*H, D)

    # Linear: (B·N·N, D) @ (D, 5·H).T → (B·N·N, 5·H)
    x_flat = x_f16.view(bs * N * N, dim)               # (M, D)
    fused_out = F.linear(x_flat, fused_weight)         # (M, 5*H)
    fused_out = fused_out.view(bs, N, N, 5 * H)        # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 4️⃣ Split fused tensor
    # ------------------------------------------------------------------
    left_proj      = fused_out[..., 0          : H]
    left_gate_pre  = fused_out[..., H          : 2 * H]
    right_proj     = fused_out[..., 2 * H      : 3 * H]
    right_gate_pre = fused_out[..., 3 * H      : 4 * H]
    out_gate_pre   = fused_out[..., 4 * H      : 5 * H]

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H) half
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 6️⃣ Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 7️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ contraction – batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → reshape to (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # out_mat[b*h,i,j] = Σ_k left_mat[b*h,i,k] * right_mat[b*h,j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H) – still half precision
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Hidden‑dim LayerNorm (fp16) – custom Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(bs * N * N, H).contiguous()      # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H)

    # ------------------------------------------------------------------
    # 🔟 Output gate & final projection back to model dimension
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(bs * N * N, H).contiguous()   # (M, H) half
    gated = out_norm * out_gate_flat                               # (M, H) half

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()   # (D, H)
    output_flat = F.linear(gated, to_out_w)                     # (M, D) half

    # Cast to float32 for the return value
    output = output_flat.view(bs, N, N, dim).float()
    return output
```
