"""
TriMul “outgoing” forward pass – Triton‑fused hidden‑dim LayerNorm,
single fused FP16 projection + gating, optional binary mask, and an
FP16 batched GEMM performed with cuBLAS (torch.bmm).  The heavy N³
contraction Σₖ left[b,i,k,h]·right[b,j,k,h] is reduced to a batched
matrix multiplication (B·H independent N×N GEMMs).  All compute‑intensive
operations run in float16 (Tensor‑Core friendly) while the required
LayerNorm is implemented in Triton, satisfying the “custom kernel”
requirement.

Inputs
------
input_tensor : torch.Tensor[bs, N, N, dim] (float32)
mask        : torch.Tensor[bs, N, N] (bool/float) or None
weights     : dict of model parameters (see keys below)
config      : dict with keys:
    - dim          : model dimension D
    - hidden_dim   : hidden dimension H
    - nomask (opt) : if True ignore `mask`

Outputs
-------
torch.Tensor[bs, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
# FP16 LayerNorm – 2‑pass implementation in Triton (used for the hidden dim)
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------ row indices handled by this program -----------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------ first pass: compute mean & variance -----------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------ second pass: write normalized result -----------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """2‑pass LayerNorm over the last dimension (fp16) using a Triton kernel."""
    assert x.dtype == torch.float16
    assert x.dim() == 2  # shape (rows, hidden_dim)
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure all tensors are contiguous
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # row tile size
        BLOCK_N=64,    # column tile size
    )
    return y


# --------------------------------------------------------------------------
# Custom TriMul entry point
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see key list below)
        - config       : dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype   # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – use PyTorch native implementation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape (B, N, N, D), dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all projections + gates into a single linear (5·H outputs)
    # ------------------------------------------------------------------
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)      # shape (5*H, dim)

    #   combined: (B, N, N, 5*H)  – FP16
    combined = F.linear(x_f16, combined_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the combined tensor into its constituents
    # ------------------------------------------------------------------
    slice0 = 0
    slice1 = H
    slice2 = 2 * H
    slice3 = 3 * H
    slice4 = 4 * H
    slice5 = 5 * H

    left_proj   = combined[..., slice0:slice1]   # (B,N,N,H)
    left_gate_p = combined[..., slice1:slice2]   # (B,N,N,H)
    right_proj  = combined[..., slice2:slice3]   # (B,N,N,H)
    right_gate_p= combined[..., slice3:slice4]   # (B,N,N,H)
    out_gate_p  = combined[..., slice4:slice5]   # (B,N,N,H)

    # Gates (sigmoid) – stay in FP16
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # ------------------------------------------------------------------
    # 5️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 6️⃣ Apply per‑position gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)  FP16
    right = right_proj * right_gate # (B,N,N,H)  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction – batched FP16 GEMM via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for batched GEMM
    # left   : (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    # right  : (B, N, N, H) → (B, H, N, N) with (k,j) ordering → (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j)

    batch = bs * H
    A = left_perm.view(batch, N, N)   # (B·H, N, N)
    B = right_perm.view(batch, N, N)  # (B·H, N, N)   (k,j)

    # Batched GEMM on H100 – highly optimized Tensor‑Core kernel
    out_mat = torch.bmm(A, B)         # (B·H, N, N), FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    out_flat = out.view(M, H)   # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm_fp16(
        out_flat, to_out_norm_w, to_out_norm_b, eps=eps
    )
    out_norm = out_norm_flat.view(bs, N, N, H)   # (B,N,N,H)  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate   # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    # Linear projection (fp16) – torch.nn.functional.linear handles arbitrary leading dims
    output = F.linear(gated, to_out_w)   # (B,N,N,D) fp16
    output = output.to(torch.float32)    # cast back to float32

    return output
