def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # Input layernorm
    x_flat = input_tensor.reshape(-1, dim)
    x_norm_flat = triton_layernorm_fp32(x_flat, weights["norm.weight"], weights["norm.bias"], eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # Cast to fp16
    x_h = x_norm.to(torch.float16)

    # Combine linear weights for left/right proj and gates
    # left_proj_weight (hidden_dim, dim), left_gate_weight etc.
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"]
    ], dim=0).to(x_h.device, torch.float16)   # shape (4*H, dim)

    # Linear: (B*N*N, dim) @ (dim, 4*H). Actually torch.nn.functional.linear(x_h, combined_weight) expects (M,D) * (out_features, D) -> (M, out_features)
    # Ensure shape: x_h shape (B,N,N,D). flatten -> (M,D)
    # Let's flatten: x_h_flat = x_h.reshape(-1, dim) (M, dim)
    x_h_flat = x_h.reshape(-1, dim)
    combined_out = F.linear(x_h_flat, combined_weight)   # (M, 4H)

    # Reshape back to (B,N,N,4H)
    combined_out = combined_out.view(bs, N, N, 4*H)

    # split
    left_proj = combined_out[..., :H]
    left_gate_pre = combined_out[..., H:2*H]
    right_proj = combined_out[..., 2*H:3*H]
    right_gate_pre = combined_out[..., 3*H:]

    # Gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # optional mask
    if not nomask and mask is not None:
        mask_h = mask.to(x_h.device, torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gated projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Output gate from x_h: out_gate = sigmoid(x_h @ out_gate_weight.T). We can compute as:
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(x_h.device, torch.float16))
    out_gate = torch.sigmoid(out_gate_pre)   # (B,N,N,H)

    # Core contraction:
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    # right permute so that last two axes are (k,j): originally right shape (B,N,N,H). we want (B,H,N,N) such that right[i,j] becomes (k,j) in matmul.
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B,H,N,N) where axes (b,h,k,j)
    BH = bs * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    # GEMM: left_mat @ right_mat -> shapes (BH,N,N) and (BH,N,N) => bmm left_mat (N,N) multiplied by right_mat (N,N) with sum over k: left_mat[i,k] * right_mat[k,j].
    out_mat = torch.bmm(left_mat, right_mat)   # (BH,N,N)

    # Reshape to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Hidden layernorm (fp16)
    out_flat = out.reshape(-1, H)
    out_norm_flat = triton_layernorm_fp16(out_flat, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)
    out_norm = out_norm_flat.view(bs, N, N, H)

    # Apply output gate: gating works on normalized hidden.
    gated = out_norm * out_gate   # (B,N,N,H)

    # Final linear
    to_out_weight = weights["to_out.weight"].to(x_h.device, torch.float16)   # shape (dim, H)
    # Flatten gated
    gated_flat = gated.reshape(-1, H)
    # Linear: (M,dim) output:
    out_final_flat = F.linear(gated_flat, to_out_weight)   # (M, dim)
    out_final = out_final_flat.view(bs, N, N, dim).float()  # back to fp32

    return out_final
```

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated.

Algorithmic outline
-------------------
1️⃣ Input LayerNorm (float32) – two‑pass Triton kernel.
2️⃣ Cast normalized tensor to FP16 (half) to exploit Tensor‑Cores.
3️⃣ Fuse left/right projections and their gates into a single half‑precision
   matmul:   X @ W_cat.T   (W_cat contains 4·hidden_dim rows).
   Split result → left_proj, left_gate, right_proj, right_gate.
4️⃣ Optional binary mask (broadcasted) applied to the two projection streams.
5️⃣ Element‑wise gating of left/right streams.
6️⃣ Compute the core N³ contraction as a batched half‑precision GEMM:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   We rearrange tensors so that a single ``torch.bmm`` computes all heads:
      left  → (B·H, N, N)   (i × k)
      right → (B·H, N, N)   (k × j)   (already transposed by permute)
7️⃣ Hidden‑dim LayerNorm (float16) – two‑pass Triton kernel.
8️⃣ Apply the output gate (sigmoid on a linear of the normalized input).
9️⃣ Final linear projection back to the model dimension (FP16 matmul) and cast
   to float32 for the returned tensor.

Only the two LayerNorm steps are written in Triton; the remaining dense linear
operations are performed with highly‑optimized PyTorch kernels (Tensor‑Core
accelerated on an H‑100).  This reduces kernel launch overhead while keeping
the heavy N³ contraction fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # row indices processed by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via Triton – input is (M, N)."""
    assert x.dim() == 2, "Input must be 2‑D"
    M, N = x.shape
    y = torch.empty_like(x)

    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def layernorm_fp16_kernel(

    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Accumulate in fp32 for numeric stability
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):

        mask_n = offs_n < N

            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0

        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

            mask=mask_m[:, None] & mask_n[None, :],

        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]

# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, D] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32/bool) or None
        - weights      : dict mapping parameter names to tensors
        - config       : dict containing ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool) indicates that
                         ``mask`` should be ignored.

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    assert dtype == torch.float32, "TriMul expects float32 input"

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – Triton kernel
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                     # (M, dim)
    x_norm_flat = triton_layernorm_fp32(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_flat.view(bs, N, N, dim)                  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the remaining work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate for LEFT and RIGHT sides
    #    (single matmul => (4·H) channels, split afterwards)
    # ------------------------------------------------------------------
    # Concatenate all four weight matrices: left_proj, left_gate,
    # right_proj, right_gate → shape (4·H, dim)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0
    ).to(device, torch.float16)          # (4·H, dim)

    # Flatten spatial dimensions for one large matmul
    x_h_flat = x_h.view(-1, dim)         # (M, dim)
    combined_out_flat = F.linear(x_h_flat, combined_weight)   # (M, 4·H)

    # Reshape back to (bs, N, N, 4·H) and split streams
    combined_out = combined_out_flat.view(bs, N, N, 4 * H)

    left_proj      = combined_out[..., :H]                     # (B,N,N,H)
    left_gate_pre  = combined_out[..., H:2*H]                  # (B,N,N,H)
    right_proj     = combined_out[..., 2*H:3*H]                # (B,N,N,H)
    right_gate_pre = combined_out[..., 3*H:]                  # (B,N,N,H)

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask applied to the projection streams
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B,N,N) → (B,N,N,1) in FP16
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Output gate computed from the *input‑norm* tensor (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(device, torch.float16))
    )                                   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a single batched half‑precision GEMM
    #    left : (B,N,N,H) → (B,H,N,N)   (i × k)
    #    right: (B,N,N,H) → (B,H,N,N)   (k × j)   (already permuted)
    #    No transpose needed – both tensors are laid out as (B·H, N, N)
    # ------------------------------------------------------------------
    # rearrange tensors; the permutes produce non‑contiguous views that can be
    # safely reshaped because the H dimension has stride 1.
    left_perm  = left.permute(0, 3, 1, 2).reshape(bs * H, N, N)   # (BH, N, N)  i‑k
    right_perm = right.permute(0, 3, 2, 1).reshape(bs * H, N, N)  # (BH, N, N)  k‑j

    # Batched matrix multiply on half‑precision tensors (leverages Tensor‑Cores)
    out_mat = torch.bmm(left_perm, right_perm)                 # (BH, N, N)

    # Reshape back to (B,N,N,H)
    out_hidden = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (float16) – Triton kernel
    # ------------------------------------------------------------------
    out_hidden_flat = out_hidden.view(-1, H)  # (M, H)
    out_norm_flat = triton_layernorm_fp16(
        out_hidden_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                          # (M, H)
    out_norm = out_norm_flat.view(bs, N, N, H)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                 # (B,N,N,H) element‑wise

    # Final linear projection (FP16) → FP32
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    gated_flat = gated.view(-1, H)               # (M, H)
    out_final_flat = F.linear(gated_flat, to_out_w)   # (M, dim)
    out_final = out_final_flat.view(bs, N, N, dim).float()  # back to FP32

    return out_final
