"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ Input LayerNorm (FP16) – custom Triton kernel.
2️⃣ Fuse the two projections + their gates for left / right sides into two
   half‑precision linear layers (project + gate are concatenated along the
   output channel dimension and then split).
3️⃣ Apply the optional binary mask to the projected tensors (still FP16).
4️⃣ Element‑wise gating of the left / right tensors.
5️⃣ Compute the N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision GEMM: treat each hidden slice h as an
   independent matrix multiply (B·H, N, N) using torch.bmm.
6️⃣ Hidden‑dim LayerNorm (FP16) – custom Triton kernel.
7️⃣ Apply the output gate (sigmoid on a linear of the normalized input).
8️⃣ Final linear projection back to the model dimension (FP16 matmul) and
   cast the result to float32.

The only kernels written in Triton are the two LayerNorm passes (FP16);
all heavy matrix‑multiplications are delegated to the highly‑optimized
cuBLAS/Tensor‑Core kernels via torch.nn.functional.linear / torch.bmm.
The design reduces memory traffic by staying in half‑precision everywhere
except for the final output and by fusing projection+gate steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – half‑precision (FP16) → half‑precision
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # rows handled by this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # load a block of X (half) -> cast to fp32 for reduction
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized ----------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """Half‑precision LayerNorm via Triton (2‑pass algorithm)."""
    assert x.dtype == torch.float16, "input must be fp16"
    assert x.dim() == 2, "expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous tensors for stride extraction
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool / float) or None
        - weights      : dict mapping parameter names to tensors (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (int),
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16) – Triton kernel
    # ------------------------------------------------------------------
    # Cast input to half‑precision *before* normalisation to halve memory traffic
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    M = bs * N * N                     # total number of “tokens”
    x_flat = x_fp16.view(M, dim)      # (M, dim)

    # LayerNorm parameters (cast to fp16)
    norm_w = weights["norm.weight"].to(torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(torch.float16).contiguous()
    x_norm_flat = triton_layernorm_fp16(x_flat, norm_w, norm_b, eps=eps)   # (M, dim)
    x_norm = x_norm_flat.view(bs, N, N, dim)   # (B, N, N, dim)  – still fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left / right sides (two matmuls)
    # ------------------------------------------------------------------
    # Concatenate the projection and gate weight matrices (2·H × dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, dim)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, dim)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, dim)

    # Linear maps (FP16) – each returns (B, N, N, 2H)
    left_comb  = F.linear(x_norm, left_comb_w)      # (B,N,N,2H)
    right_comb = F.linear(x_norm, right_comb_w)     # (B,N,N,2H)

    # Split into projection and gate streams (each (B,N,N,H))
    left_proj      = left_comb[..., :H]
    left_gate_pre  = left_comb[..., H:]
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    # Apply sigmoid gates
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask applied to the *projections* (before gating)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates → gated tensors (B,N,N,H) in FP16
    # ------------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Output‑gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then reshape to (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)          # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)         # (B·H, N, N)

    # Batched matrix multiplication: left @ rightᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (FP16) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H)

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(M, H)           # (M, H)
    gated = out_norm * out_gate_flat               # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (dim, H)
    output_flat = F.linear(gated, to_out_w)       # (M, dim)  – still FP16

    output = output_flat.view(bs, N, N, dim).float()   # Cast back to FP32
    return output
