```python
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # index of tile in M,N
    batch_id = tl.program_id(axis=1)  # batch (B*H)

    # Compute block coordinates
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Pointers for each batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Compute offsets for M and N dimensions
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension, load tiles of A and B, compute
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile, shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # Load B tile, shape (BLOCK_K, BLOCK_N). Note: B is already transposed (right_T)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Compute partial product
        acc += tl.dot(a, b)  # automatically fp16 x fp16 -> fp32

    # Write out
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
TriMul (outgoing) – faster Triton implementation.

The heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched GEMM (left @ rightᵀ) for every hidden channel `h`.
We accelerate this step with a dedicated Triton kernel that:
* works in float16 (half) for bandwidth,
* accumulates in float32 (tensor‑core friendly),
* uses large 2‑D tiles (128×128) and a modest K‑tile (32) that maps
  nicely onto the 16×16×16 FP16 tensor‑core granularity of an H100,
* processes **all** hidden channels as a flat batch dimension,
* launches one program per (M‑tile, N‑tile) per hidden‑batch, and
* runs with 8 warps and double‑buffering (num_stages=2).

All the remaining lightweight operations (layer‑norms, linear
projections, sigmoids, and optional masking) stay in PyTorch but are
performed in half‑precision to minimise memory traffic.  The final
output is cast back to float32 as required by the reference API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K → Bᵀ: K×N)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (FP16)
    M, N, K,                           # matrix shapes (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,              # A row/col stride inside a batch
    stride_bk, stride_bn,              # B (already transposed) row/col stride
    stride_cm, stride_cn,              # C row/col stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulation."""
    pid = tl.program_id(axis=0)          # tile index over (M,N)
    batch_id = tl.program_id(axis=1)     # flattened (batch * hidden) dimension

    # --------------------------------------------------------------- #
    #   Tile coordinates
    # --------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------- #
    #   Pointers for the current batch
    # --------------------------------------------------------------- #
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # --------------------------------------------------------------- #
    #   Accumulator (FP32)
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    #   Main K‑loop
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # Rank‑K update
        acc += tl.dot(a, b)   # FP16×FP16 → FP32

    # --------------------------------------------------------------- #
    #   Write back (store as FP16)
    # --------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -----------------------------------------------------------------------------#
# Public entry point
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float)
            - weights_dict: mapping of all module parameters (float32)
            - config_dict : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    #   1) Layer‑norm on the feature dimension (keep FP32 for stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    #   2) Cast to FP16 and fetch half‑precision weights once
    # --------------------------------------------------------------------- #
    x_fp16 = x.half()

    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # --------------------------------------------------------------------- #
    #   3) Linear projections (no bias) – result is FP16
    # --------------------------------------------------------------------- #
    left  = F.linear(x_fp16, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------------- #
    #   4) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if not nomask:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   5) Gating (sigmoid) – stay in FP16
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   6) Core contraction via Triton (batched GEMM)
    # --------------------------------------------------------------------- #
    #   (B, H, N, N) layout → (batch, M, K) and (batch, K, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    #   Transpose right so we can compute A @ Bᵀ
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    B_, H_, M, K = left_perm.shape      # M == K == N
    _, _, _, N_   = right_T.shape
    assert M == K == N_ == N, "Shape mismatch in contraction"

    total_batches = B_ * H_                # flattened (B * hidden)

    # Flatten the first two dimensions – classic batched GEMM layout
    a = left_perm.view(total_batches, M, K)          # (batch, M, K)  FP16
    b = right_T.view(total_batches, K, N_)           # (batch, K, N)  FP16
    c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # --------------------------------------------------------------------- #
    #   7) Launch the Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        ((M + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),
        total_batches,
    )

    # Strides for the flattened contiguous tensors
    stride_a_batch = M * K
    stride_am = K                 # row‑stride inside A
    stride_ak = 1                 # col‑stride inside A

    stride_b_batch = K * N_
    stride_bk = N_                # row‑stride inside B (K × N)
    stride_bn = 1

    stride_c_batch = M * N_
    stride_cm = N_                # row‑stride inside C (M × N)
    stride_cn = 1

    _trmul_gemm_kernel[grid](
        a, b, c,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,                 # H100 shows best occupancy with 8 warps
        num_stages=2,                # double‑buffering for latency hiding
    )

    # --------------------------------------------------------------------- #
    #   8) Reshape back to [B, N, N, H] (still FP16)
    # --------------------------------------------------------------------- #
    out = c.view(B_, hidden_dim, M, N_).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------- #
    #   9) Hidden‑dim layer‑norm, out‑gate and final linear (all in FP16)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate               # broadcast over hidden dim
    out = F.linear(out, to_out_w)      # back to original dimension D (FP16)

    # --------------------------------------------------------------------- #
    #   10) Cast to FP32 for downstream consumption
    # --------------------------------------------------------------------- #
    return out.float()
