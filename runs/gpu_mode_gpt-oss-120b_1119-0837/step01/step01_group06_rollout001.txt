"""
TriMul “outgoing” forward pass (alphaFold‑3 style) – Triton‑accelerated.

Algorithm
---------
1. **LayerNorm (input)** – fused mean/variance reduction in a single
   Triton kernel (float32 accumulation, half‑precision I/O).  Operates on the
   last feature dimension (dim).
2. **Linear projections + mask** –  half‑precision linear layers for
   `left_proj` and `right_proj`.  When a mask is present the scalar mask is
   applied after the projection (equivalent to pre‑masking).
3. **Gating** – sigmoid gates (`left_gate`, `right_gate`, `out_gate`) are
   computed with half‑precision linear layers (no bias) and multiplied
   element‑wise into `left` and `right`.
4. **Batched matrix multiplication** – for each hidden channel we need

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   which is exactly a batch of GEMMs of shape (N×N).  By permuting to
   (B, hidden_dim, N, N) we can call `torch.matmul` (or `torch.bmm`) once,
   letting cuBLAS/Tensor‑cores handle the N³ work efficiently.
5. **LayerNorm (output)** – another Triton kernel normalises over the hidden
   dimension.
6. **Output gate + final linear** – apply `out_gate`, then a half‑precision
   linear projection back to the model dimension and cast the final
   tensor to float32.

All heavy arithmetic is performed in FP16 (torch.float16) with float32
accumulators inside the Triton kernels, giving a large speed‑up on H100
Tensor‑cores while preserving the exact functional behaviour of the
reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm kernel (works for float16 I/O)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]   (half)
    y_ptr,                # *ptr to output [M, N]   (half)
    weight_ptr,           # *ptr to weight [N]      (half)
    bias_ptr,             # *ptr to bias   [N]      (half)
    M,                    # total rows
    N,                    # number of features
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: per‑row mean & var (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    NUM_K = tl.cdiv(N, BLOCK_N)
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load a tile (BLOCK_M x BLOCK_N) in half precision
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised values (still half I/O)
    # ------------------------------------------------------------------
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # Normalise in float32 then cast back to half
        norm_f32 = (x_f32 - mean[:, None]) * rstd[:, None]
        norm = norm_f32.to(tl.float16)

        # Load weight & bias (half)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Triton‑based LayerNorm on the last dimension.
    Supports float16 input/output; accumulation is performed in float32.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tune block sizes – BLOCK_N should be >= max feature dimension block size.
    # For dim up to 384 we use 128 (covers 3 tiles).  BLOCK_M can be 256.
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # rows per program
        BLOCK_N=128,   # columns per tile (covers up to 384 features)
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
            - mask         : torch.Tensor   [B, N, N]   (bool/int) or None
            - weights      : dict of torch.Tensors (model parameters)
            - config       : dict with ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # ------------------------------------------------------------------
    # Cast everything to FP16 for speed (the kernels work on half)
    # ------------------------------------------------------------------
    dtype = torch.float16
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – flatten to (M, dim)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_half = input_tensor.to(dtype)                      # [B,N,N,dim] fp16
    x_flat = x_half.view(M, dim)                         # [M, dim]

    norm_weight = weights["norm.weight"].to(dtype, device)
    norm_bias   = weights["norm.bias"].to(dtype, device)

    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Prepare mask (if any) – keep as half, broadcast over hidden_dim
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_f = None

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype, device)
    right_proj_w = weights["right_proj.weight"].to(dtype, device)

    left = F.linear(x_norm, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_w)  # [B,N,N,hidden_dim]

    # Apply mask after projection (equivalent to pre‑masking)
    if mask_f is not None:
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – compute on the **unmasked** norm tensor
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(dtype, device)
    right_gate_w = weights["right_gate.weight"].to(dtype, device)
    out_gate_w = weights["out_gate.weight"].to(dtype, device)

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication (N³ reduction) – fused GEMMs
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) so that we can use torch.matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Equivalent to: out = torch.matmul(left_perm, right_perm.transpose(-1, -2))
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # [B, H, N, N]

    # Back to original layout [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) over hidden_dim
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)  # [M, hidden_dim]

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype, device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype, device)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    out = out_norm * out_gate          # gating (still fp16)
    to_out_w = weights["to_out.weight"].to(dtype, device)  # [dim, hidden_dim]

    out = F.linear(out, to_out_w)      # [B,N,N,dim] in fp16
    # Cast back to float32 as required by the API
    return out.to(torch.float32)
