import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    TriMul outgoing update (AlphaFold3) implemented with:
    - PyTorch for LayerNorm, Linear projections, Sigmoid gating and final LayerNorm.
    - Triton kernel for the element‑wise multiplication of projections with their
      gates (left*left_gate, right*right_gate).  The O(N³·hidden) multiplicative
      update is executed with a batched GEMM (torch.bmm) on half‑precision tensors.
    The function returns a float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # Load model weights (they are expected to be on the same device)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # shape (dim, hidden)

    # ------------------------------------------------------------------ #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2) Linear projections (FP32) and optional mask.  
    #    The mask is broadcast on the hidden dimension.
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_w)          # [B, N, N, H]
    right = F.linear(x, right_proj_w)        # [B, N, N, H]

    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 3) Compute gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------ #
    # 4) Fuse projection with its gate using a tiny Triton kernel.
    #    All tensors are cast to fp16 before the multiplication.
    # ------------------------------------------------------------------ #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    total = left_h.numel()   # same for right_h
    BLOCK = 8192
    grid = ((total + BLOCK - 1) // BLOCK,)

    # left_h = left_h * left_gate_h
    _elementwise_mul_kernel[grid](
        left_h,
        left_gate_h,
        left_h,          # in‑place write
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    # right_h = right_h * right_gate_h
    _elementwise_mul_kernel[grid](
        right_h,
        right_gate_h,
        right_h,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------ #
    # 5) Core multiplicative update: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matrix multiplication (A @ Bᵀ) in fp16.
    # ------------------------------------------------------------------ #
    # Rearrange tensors to [B, H, N, N] and flatten batch*hidden.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    batch_h = bs * hidden
    left_mat = left_perm.view(batch_h, N, N)   # [B*H, N, N]  (i,k)
    right_mat = right_perm.view(batch_h, N, N) # [B*H, N, N]  (j,k)

    # Batched GEMM: (i,k) @ (k,j)ᵀ  → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, shape [B*H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6) Post‑processing
    # ------------------------------------------------------------------ #
    # Layer‑norm over hidden dimension
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out‑gate (broadcasted multiplication)
    out = out * out_gate

    # Final linear projection back to `dim` (float32)
    out = F.linear(out, to_out_w)   # → [B, N, N, dim]

    return out
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triangle Multiplicative Update (outgoing) – AlphaFold3 implementation.
    The expensive O(N³·hidden) step is performed with a batched GEMM via
    torch.bmm (fp16).  A tiny Triton kernel fuses projection tensors with their
    sigmoid gates (left*left_gate, right*right_gate).  The remaining operations
    (layer‑norms, final linear) are done with PyTorch.
    Returns a float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------------------- weights ---------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # (dim, hidden)

    # ------------------- 1) LayerNorm ------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------- 2) Projections ----------------- #
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------- 3) Gates ----------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------- 4) Fuse projection with gate (Triton) ----- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    total = left_h.numel()               # same for all tensors
    BLOCK_SIZE = 256                     # ≤1024 threads per block
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # left_h = left_h * left_gate_h (in‑place)
    _elementwise_mul_kernel[grid](
        left_h,
        left_gate_h,
        left_h,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )
    # right_h = right_h * right_gate_h (in‑place)
    _elementwise_mul_kernel[grid](
        right_h,
        right_gate_h,
        right_h,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )

    # ------------------- 5) Core O(N³) update ------------------ #
    # reshape to [B, H, N, N] and flatten batch*hidden for batched GEMM
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (j,k)

    batch_h = bs * hidden
    left_mat = left_perm.view(batch_h, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(batch_h, N, N) # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------- 6) Post‑processing ------------------- #
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out.to(torch.float32) * out_gate          # apply out‑gate
    out = F.linear(out, to_out_w)                    # final projection to dim

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_kernel(
    a_ptr,            # left tensor   (fp16)
    b_ptr,            # gate tensor   (fp16)
    out_ptr,          # destination   (fp16)
    total,            # number of elements (scalar)
    BLOCK_SIZE: tl.constexpr,   # ≤ 1024 (threads per block)
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing triangle multiplicative update
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triangle‑Multiplicative‑Update (outgoing) as used in AlphaFold3.
    The O(N³·hidden) core is performed with a batched GEMM (torch.bmm)
    on half‑precision tensors.  A small Triton kernel fuses each projection
    with its sigmoid gate (left*left_gate, right*right_gate).  All other
    operations (LayerNorm, final linear) use PyTorch.  Returns a float32
    tensor of shape [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------
    # 0) unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    bs, N, _, dim = inp.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # (dim, hidden)

    # ------------------------------------------------------------------
    # 2) input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 3) linear projections (float32) – shape [B, N, N, H]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) fuse projection with its gate using a Triton kernel
    #    – cast to fp16 first (Tensor‑core friendly)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    total = left_h.numel()                 # same for all 4 tensors
    BLOCK_SIZE = 256                       # ≤ 1024 threads per block
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # left_h = left_h * left_gate_h  (in‑place)
    _elementwise_mul_kernel[grid](
        left_h,
        left_gate_h,
        left_h,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )
    # right_h = right_h * right_gate_h  (in‑place)
    _elementwise_mul_kernel[grid](
        right_h,
        right_gate_h,
        right_h,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) core outgoing update:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) with torch.bmm
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N]  (i,k) and (j,k) order respectively
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    batch_h = bs * hidden
    left_mat = left_perm.view(batch_h, N, N)    # (i,k)
    right_mat = right_perm.view(batch_h, N, N)  # (j,k)

    # Batched matrix multiplication: (i,k) @ (k,j)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply out‑gate (broadcasted multiplication) – up‑cast to fp32 first
    out = out.to(torch.float32) * out_gate

    # final linear projection back to model dimension
    out = F.linear(out, to_out_w)    # → [B, N, N, dim]

    return out
