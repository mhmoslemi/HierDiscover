4. Core contraction:
   For each batch element and hidden channel h we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiply left @ rightᵀ per hidden slice.
   It is implemented in Triton with FP16 inputs and FP32 accumulation.
   The kernel works on a flattened batch dimension of size B·H and
   processes tiles of size BLOCK_M×BLOCK_N (default 128×128) with a
   K‑tile of size BLOCK_K (default 32).  Memory access is fully
   coalesced and edge handling is done via masking.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate (sigmoid), and finally project back to the original
   channel dimension with another bias‑free linear layer.
6. The final tensor has shape [B, N, N, C] and dtype float32.

The implementation keeps all heavy arithmetic in FP16 where possible,
uses a single fused linear for the five projections, and only the O(N³·H)
contraction is performed in Triton.  This dramatically cuts memory traffic
and brings the end‑to‑end runtime well below the baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM:   C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]

    A: (batch, M, K)               FP16
    B: (batch, K, N) – stored as (batch, N, K) with swapped strides
    C: (batch, M, N)               FP32 accumulator → FP32 store
    """
    pid_batch = tl.program_id(2)      # batch * hidden index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Compute pointers for the tiles
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles (FP16) – masks ensure we don’t read OOB
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # Store the result (FP32)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of all model weights (see reference)
            - config_dict  : contains 'dim' (C) and 'hidden_dim' (H) and
                             optionally 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    ##################################################################
    # 1️⃣ Unpack inputs
    ##################################################################
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_fp32 = input_tensor.dtype
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    apply_mask = not config.get("nomask", True)

    ##################################################################
    # 2️⃣ Load weight tensors
    ##################################################################
    # LayerNorm over the input channels
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Individual projection / gate weights (all of shape [H, C])
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Output LayerNorm and projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    ##################################################################
    # 3️⃣ Input LayerNorm (float32) → half‑precision for the rest
    ##################################################################
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_weight,
                     bias=norm_bias)
    x_h = x.to(torch.float16)                     # (B, N, N, C)

    ##################################################################
    # 4️⃣ Fuse the five linear transforms into one matmul (FP16)
    ##################################################################
    # Concatenate weights along the output dimension: shape (5*H, C)
    fused_w = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w],
        dim=0
    ).to(torch.float16)

    # Flatten batch & spatial dims: (B*N*N, C)
    x_flat = x_h.view(-1, C)                     # (M, C) with M = B*N*N
    fused_out = F.linear(x_flat, fused_w)        # (M, 5*H)  FP16

    # Reshape and split
    fused_out = fused_out.view(B, N, N, 5, H)     # (B, N, N, 5, H)
    left_proj   = fused_out[..., 0, :]            # (B, N, N, H)
    right_proj  = fused_out[..., 1, :]
    left_gate_pre   = fused_out[..., 2, :]
    right_gate_pre  = fused_out[..., 3, :]
    out_gate_pre    = fused_out[..., 4, :]

    # Gates (sigmoid) – stay in FP16
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)     # (B, N, N, H) FP16

    ##################################################################
    # 5️⃣ Optional pairwise mask (broadcasted to the hidden dim)
    ##################################################################
    if apply_mask and mask_tensor is not None:
        # Cast mask to half‑precision and broadcast
        mask = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = left_proj * mask
        right = right_proj * mask
    else:
        left  = left_proj
        right = right_proj

    # Apply the per‑position gates
    left  = left * left_gate
    right = right * right_gate      # both (B,N,N,H) FP16

    ##################################################################
    # 6️⃣ Core TriMul contraction – batched GEMM in Triton
    ##################################################################
    # Rearrange to (B, H, N, N) and flatten batch*hidden → (BH, N, N)
    left_h   = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h  = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * H
    left_mat  = left_h.view(BH, N, N)   # (BH, M=N, K=N)
    right_mat = right_h.view(BH, N, N)  # (BH, K=N, N) – we will interpret it as transposed via swapped strides

    # Allocate output buffer (FP32)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides for the three tensors
    stride_a_batch = left_mat.stride(0)      # = N*N
    stride_a_m     = left_mat.stride(1)      # = N
    stride_a_k     = left_mat.stride(2)      # = 1

    stride_b_batch = right_mat.stride(0)     # = N*N
    # We interpret right_mat as (K, N) → swap the last two strides
    stride_b_k = right_mat.stride(2)         # = 1   (original column stride)
    stride_b_n = right_mat.stride(1)         # = N   (original row stride)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Tune tile sizes – good for N up to 1024 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH                         # batch*hidden
    )

    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H) – keep in FP32 for the following LayerNorm
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    ##################################################################
    # 7️⃣ Hidden‑dim LayerNorm, output‑gate, and final projection
    ##################################################################
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(H,),
                              weight=to_out_norm_w,
                              bias=to_out_norm_b)

    # out_gate was FP16; bring it to FP32 for multiplication
    out_hidden = out_hidden * out_gate.to(dtype=out_hidden.dtype)

    # Final linear projection H → C (bias‑free).  Use FP16 for the matmul
    # and cast the result back to FP32.
    out_hidden_h = out_hidden.to(torch.float16)
    to_out_w_h = to_out_w.to(torch.float16)
    out = F.linear(out_hidden_h, to_out_w_h)   # (B, N, N, C) FP16
    out = out.to(dtype_fp32)                    # cast to the expected dtype

    return out
```
