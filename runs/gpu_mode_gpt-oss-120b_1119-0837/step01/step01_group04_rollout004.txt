    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))

"""
TriMul (outgoing) forward pass – Triton‑fused mask/gate and FP16‐permute,
batched FP16 GEMM via torch.matmul, and the remaining ops in PyTorch.

Algorithm
---------
1. Layer‑norm the input over the feature dimension.
2. Linear projections (left/right) to the hidden dimension.
3. Compute three sigmoid gates (left‑gate, right‑gate, out‑gate).
4. A tiny Triton kernel fuses:
      * optional pairwise mask,
      * element‑wise gating,
      * conversion to float‑16,
      * permutation from (B,N,N,HD) → (B,HD,N,N).
   The kernel writes two half‑precision tensors `left_fp16` and `right_fp16`
   ready for a batched matrix multiply.
5. Core cubic operation:
      out[b,d,i,j] = Σ_k left_fp16[b,d,i,k] * right_fp16[b,d,j,k]
   performed by a single `torch.matmul` call (FP16 × FP16 → FP16,
   accumulation in FP32 on H100 Tensor‑Cores).
6. Convert the result back to float‑32, undo the permutation,
   apply a second LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original feature dimension.

The expensive N³ term is handled by cuBLAS (via torch.matmul) which is
far faster than a hand‑written Triton GEMM for the sizes used in AlphaFold3.
The Triton kernel only touches each element once, drastically reducing
temporary allocations compared with a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fp16_kernel(
    # input tensors
    left_ptr,            # float32 *
    right_ptr,           # float32 *
    left_gate_ptr,       # float32 *
    right_gate_ptr,      # float32 *
    mask_ptr,            # float32 *   (optional)
    # output tensors (half‑precision, permuted)
    out_left_ptr,        # float16 *
    out_right_ptr,       # float16 *
    # static sizes
    B, N, H,                     # int
    # strides for the (B,N,N,H) layout (float32 tensors)
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    # strides for the (B,H,N,N) layout (float16 output tensors)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # strides for the mask (B,N,N) layout
    stride_mask_b, stride_mask_i, stride_mask_j,
    # total number of elements (B*N*N*H)
    total,
    # compile‑time constants
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)                      # linear block id
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total                         # mask for the tail

    # ----- decode linear index -> (b,i,j,h) ---------------------------------
    h = offs % H
    idx = offs // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- compute flat offsets for the various tensors ----------------------
    in_offset = (b * stride_in_b
                 + i * stride_in_i
                 + j * stride_in_j
                 + h * stride_in_h)

    out_offset = (b * stride_out_b
                  + h * stride_out_h
                  + i * stride_out_i
                  + j * stride_out_j)

    if HAS_MASK:
        mask_offset = (b * stride_mask_b
                       + i * stride_mask_i
                       + j * stride_mask_j)

    # ----- load values -------------------------------------------------------
    left_val      = tl.load(left_ptr      + in_offset, mask=valid, other=0.0)
    right_val     = tl.load(right_ptr     + in_offset, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + in_offset, mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + in_offset, mask=valid, other=0.0)

    # ----- gating ----------------------------------------------------------------
    left_val  = left_val  * left_gate_val
    right_val = right_val * right_gate_val

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + mask_offset, mask=valid, other=1.0)
        left_val  = left_val  * mask_val
        right_val = right_val * mask_val

    # ----- store as float16 (the pointer type determines the dtype) ----------
    tl.store(out_left_ptr  + out_offset, left_val,  mask=valid)
    tl.store(out_right_ptr + out_offset, right_val, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        * input      – Tensor of shape [B, N, N, dim] (float32)
        * mask       – Tensor of shape [B, N, N] (float32) or None
        * weights    – Mapping from parameter names to tensors
        * config     – Dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    B, N, _, _ = inp.shape   # inp.shape == [B, N, N, dim]

    # --------------------------------------------------------------
    # 1) input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3) gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4) Optional mask – fuse it together with gating, cast to fp16,
    #    and permute to (B, hidden, N, N) layout using a Triton kernel.
    # --------------------------------------------------------------
    use_mask = (not nomask) and (mask is not None)
    if use_mask:
        mask_fp32 = mask.to(torch.float32)
    else:
        # dummy tensor – never read when HAS_MASK==0
        mask_fp32 = torch.empty((1,), dtype=torch.float32, device=device)

    # output tensors (half‑precision, already in the required (B,HD,N,N) layout)
    left_fp16  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_fp16 = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

    # strides for the source tensors (shape: B,N,N,HD)
    stride_in_b = left.stride(0)
    stride_in_i = left.stride(1)
    stride_in_j = left.stride(2)
    stride_in_h = left.stride(3)   # =1 for a contiguous tensor

    # strides for the destination tensors (shape: B,HD,N,N)
    stride_out_b = left_fp16.stride(0)
    stride_out_h = left_fp16.stride(1)
    stride_out_i = left_fp16.stride(2)
    stride_out_j = left_fp16.stride(3)

    # mask strides (shape: B,N,N)
    stride_mask_b = mask_fp32.stride(0) if use_mask else 0
    stride_mask_i = mask_fp32.stride(1) if use_mask else 0
    stride_mask_j = mask_fp32.stride(2) if use_mask else 0

    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024

    grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # launch the fusion kernel
    mask_gate_fp16_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_fp32,
        left_fp16, right_fp16,
        B, N, hidden_dim,
        stride_in_b, stride_in_i, stride_in_j, stride_in_h,
        stride_out_b, stride_out_h, stride_out_i, stride_out_j,
        stride_mask_b, stride_mask_i, stride_mask_j,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=1 if use_mask else 0,
    )

    # --------------------------------------------------------------
    # 5) Core N³ operation – batched FP16 GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # left_fp16, right_fp16: (B, hidden_dim, N, N)
    # torch.matmul automatically batches over the first two dimensions
    out_fp16 = torch.matmul(left_fp16, right_fp16.transpose(-1, -2))   # (B, hidden, N, N)

    # back to (B, N, N, hidden) and to float32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6) second LayerNorm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7) apply out‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
