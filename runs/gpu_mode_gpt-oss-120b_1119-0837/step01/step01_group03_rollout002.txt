"""
TriMul (outgoing) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x`  (shape [B, N, N, C]).
2. Two bias‑free linear projections (`left`, `right`) to the hidden dimension H.
3. Optional mask (broadcast on H) is applied to the projected tensors.
4. Compute three sigmoid gates from the normalized input:
      left_gate, right_gate, out_gate  (shape [B, N, N, H]).
   Apply the left/right gates point‑wise.
5. Core O(N³·H) operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ for each
   (batch, hidden) pair.  We fuse all B·H independent GEMMs with a
   half‑precision Triton kernel that leverages Tensor‑cores (fp16·fp16→fp32
   accumulation).  The kernel tiles the matrices with
   BLOCK_M×BLOCK_N tiles and a BLOCK_K reduction dimension.
6. Second LayerNorm over the hidden dimension, point‑wise multiplication
   by `out_gate`, and a final bias‑free linear projection back to the
   original channel size C.

Only the heavy O(N³·H) matmul is performed in Triton; all other
operations use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Half‑precision batched GEMM kernel (C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to fp16 tensors
    M, N, K,                            # matrix dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,   # A strides
    stride_b_batch, stride_b_k, stride_b_n,   # B strides (treated as transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of (M×K)·(N×K)ᵀ products.

    A: [batch, M, K] (row‑major)
    B: [batch, N, K] (row‑major, interpreted as K×N after transposition)
    C: [batch, M, N]
    All tensors are fp16; accumulation is fp32.
    """
    pid_batch = tl.program_id(2)   # combined batch * hidden index
    pid_m = tl.program_id(0)       # block row
    pid_n = tl.program_id(1)       # block column

    # offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for the selected batch element
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # compute pointers for the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_a_m \
                       + (k_offset + offs_k)[None, :] * stride_a_k
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_b_k \
                       + offs_n[None, :] * stride_b_n

        # out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # dot accumulates in fp32 automatically (fp16·fp16→fp32)
        acc += tl.dot(a, b)

    # store the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/int/float)
        - weights_dict : dict of model parameters (all float32)
        - config_dict  : dict containing at least:
            * "dim"         – input channel dimension C
            * "hidden_dim"  – hidden dimension H
            * "nomask"      – bool, True ⇒ mask can be ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # retrieve model parameters
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, C]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) to hidden dimension H
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask is 0/1 – convert to same dtype as tensors
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) operation – batched GEMM (A @ Bᵀ) in half‑precision
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # B = batch size, H = hidden_dim

    # fuse batch and hidden dimensions to obtain a pure 3‑D tensor
    # shape after view: [B*H, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_t.view(B * H, N, N).to(torch.float16)   # fp16 for compute
    right_mat = right_t.view(B * H, N, N).to(torch.float16)

    # output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    # Tile sizes – chosen to fit in shared memory and expose Tensor‑cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over rows (M)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over cols (N)
        B * H                           # batch * hidden
    )

    batched_matmul_fp16_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_mat.stride(0),
        stride_a_m=left_mat.stride(1),
        stride_a_k=left_mat.stride(2),
        stride_b_batch=right_mat.stride(0),
        # for Bᵀ we swap the last two strides:
        stride_b_k=right_mat.stride(2),   # stride of original K dimension
        stride_b_n=right_mat.stride(1),   # stride of original N dimension
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # H100 prefers many warps per block
        num_stages=2,           # double‑buffering of shared memory
    )

    # reshape back to [B, N, N, H] (float32 for subsequent ops)
    out = out_mat.view(B, H, N, N)                # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous().float()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to [B, N, N, C]

    return out
