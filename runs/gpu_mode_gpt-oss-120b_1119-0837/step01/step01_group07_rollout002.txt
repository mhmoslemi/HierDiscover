"""
TriMul (outgoing) – optimized Triton implementation.

The module processes a 4‑D tensor `x` ∈ ℝ^{B×N×N×D}:

1. Layer‑norm over the feature dimension D.
2. Two linear projections (left/right) and three linear gates (left, right, out).
3. Optional element‑wise mask applied to the projected tensors.
4. Gated tensors are multiplied with a custom Triton batched‑GEMM:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched mat‑mul `left @ rightᵀ` for each hidden channel `h`.
   The kernel works in FP16 (bandwidth) and accumulates in FP32 (numerical
   stability) and exploits Tensor‑Core MMA on H100.
5. Hidden‑dim layer‑norm, out‑gate multiplication and a final linear
   projection back to `dim`.

Only the heavy O(N³) contraction is performed on the GPU via Triton.  All
other operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import triton.ops as triton_ops  # provides a tuned batched‑matmul


# -----------------------------------------------------------------------------#
# Triton kernel (fallback) – simple batched GEMM based on tl.dot.
#   This kernel is only used if triton.ops.matmul is unavailable.
# -----------------------------------------------------------------------------#
@triton.jit
def _fallback_trmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (M=N=N, K=N)
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col) – B is already transposed
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b] = A[b] @ B[b]ᵀ  with A: (M×K), B: (N×K)  (both FP16 → FP32 acc)
    The grid is (tiles over M,N) × (batch dimension).
    """
    pid = tl.program_id(axis=0)          # tiles over output M×N
    batch = tl.program_id(axis=1)        # (B * hidden) batch index

    # --------------------------------------------------------------------- #
    #   2‑D tiling indices
    # --------------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Compute batch‑specific base pointers
    # --------------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   K‑loop (contracted axis)
    # --------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A tile (M×K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # B tile (K×N) – note that B is already transposed
        b_ptrs = b_batch + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # MAC (Tensor‑Core friendly)
        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Store result (cast back to FP16)
    # --------------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

# Entry point – called by the test harness
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    #   Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    #   1) Layer‑norm over the feature dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    #   2) Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    #   3) Optional outer mask (broadcast over hidden_dim)
    # --------------------------------------------------------------------- #
    if not nomask:
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left  * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   4) Gating (sigmoid) – shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   5) Core contraction: batched GEMM left @ rightᵀ per hidden channel.
    #       * cast to FP16 for bandwidth
    #       * compute with a Triton kernel (high‑performance matmul)
    # --------------------------------------------------------------------- #
    #   Rearrange to (B, hidden, N, N) and convert to half‑precision
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # (B, H, N, N)

    #   Transpose the last two axes of `right` so that we can compute
    #   out[b, h] = left[b, h] @ right[b, h]ᵀ
    right_T_fp16 = right_fp16.transpose(-2, -1).contiguous()              # (B, H, N, N)

    # --------------------------------------------------------------------- #
    #   5a) Preferred path – use Triton's highly‑tuned batched matmul.
    #       This kernel internally launches a fused GEMM that exploits
    #       Tensor‑Core MMA on the H100.
    # --------------------------------------------------------------------- #
    try:
        # `triton.ops.matmul` automatically selects an optimal kernel.
        out_fp16 = triton_ops.matmul(left_fp16, right_T_fp16)   # (B, H, N, N), FP16
    except Exception:  # pragma: no cover – fallback for minimal environments
        # ----------------------------------------------------------------- #
        #   5b) Fallback custom kernel (still using TL dot) when ops.matmul
        #       is not available (unlikely on the evaluation platform).
        # ----------------------------------------------------------------- #
        B_, H_, M, K = left_fp16.shape
        _, _, _, N_ = right_T_fp16.shape
        assert M == K == N_, "unexpected shape mismatch for GEMM"

        total_batches = B_ * H_                     # (B × hidden) as batch dim
        a = left_fp16.view(total_batches, M, K)     # (batch, M, K)
        b = right_T_fp16.view(total_batches, K, N_) # (batch, K, N)

        # Output buffer
        c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

        # Triton launch configuration
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64   # tuned for H100
        grid = (
            ( (M + BLOCK_M - 1) // BLOCK_M ) * ( (N_ + BLOCK_N - 1) // BLOCK_N ),
            total_batches,
        )

        # Strides for flattened tensors (row‑major)
        stride_a_batch = M * K
        stride_am = K                      # row‑stride inside A
        stride_ak = 1                      # col‑stride inside A

        stride_b_batch = K * N_
        stride_bk = N_                     # row‑stride inside B (since B is K×N)
        stride_bn = 1                      # col‑stride inside B

        stride_c_batch = M * N_
        stride_cm = N_                     # row‑stride inside C
        stride_cn = 1                      # col‑stride inside C

            stride_cm, stride_cn,
            stride_a_batch, stride_b_batch, stride_c_batch,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,                 # more warps → higher occupancy on H100

        out_fp16 = c.view(B_, hidden_dim, M, N_)   # (B, H, N, N)
