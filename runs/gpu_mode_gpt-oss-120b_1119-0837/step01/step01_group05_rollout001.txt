"""
TriMul – outgoing multiplicative update – fused Triton kernel.

The forward pass implements the AlphaFold‑3 “outgoing” TriMul:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the input tensor
[x] ∈ ℝ^{B×N×N×C}.  After optional masking we apply per‑channel sigmoids
(gates) and finally a batched GEMM (A @ Bᵀ) for every hidden channel *h*.
The GEMM is performed by a custom Triton kernel that:
  * packs the (batch, hidden) dimensions into a single Triton batch index,
  * uses FP16 operands and FP32 accumulation (Tensor‑Core friendly),
  * tiles the output with BLOCK_M×BLOCK_N tiles and iterates over K in
    BLOCK_K chunks,
  * works for any B∈{1,2}, N≤1024 and hidden_dim ≤128.

All remaining operations (LayerNorm, Linear, sigmoid) are kept in
high‑performance PyTorch; the kernel only accelerates the O(N³·H) core.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr,           # left  matrix   : [batch_h, M=N, K=N]   (fp16)
    b_ptr,           # right matrix^T: [batch_h, K=N, N=N]   (fp16)
    c_ptr,           # output matrix : [batch_h, M=N, N=N]   (fp16)
    M, N, K,         # all equal to sequence length
    stride_a_batch,  # stride between batch_h entries in A
    stride_a_m,      # stride for i‑dimension (M) in A
    stride_a_k,      # stride for k‑dimension (K) in A
    stride_b_batch,  # stride between batch_h entries in B
    stride_b_k,      # stride for k‑dimension (K) in B
    stride_b_n,      # stride for n‑dimension (N) in B
    stride_c_batch,  # stride between batch_h entries in C
    stride_c_m,      # stride for i‑dimension (M) in C
    stride_c_n,      # stride for n‑dimension (N) in C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ   with A:[batch_h,M,K]  B:[batch_h,N,K] → Bᵀ:[batch_h,K,N]

    Each block computes a TILE_M×TILE_N output tile.
    """
    pid_m = tl.program_id(0)      # tile row   (i)
    pid_n = tl.program_id(1)      # tile col   (j)
    pid_b = tl.program_id(2)      # packed batch*hidden index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointers to the sub‑tiles of A and B
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Masks for out‑of‑bounds accesses (helps with non‑multiple‑of‑tile sizes)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Accumulate the matrix product
        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

    # Write back the result (store as fp16)
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    TriMul outgoing update (forward only).

    Args:
        data: tuple (inp, mask, weights, config)
            inp   – torch.Tensor [B, N, N, C]   (float32)
            mask  – torch.Tensor [B, N, N] or None
            weights – dict with all module parameters (float32)
            config  – dict containing at least:
                         "dim" (C), "hidden_dim" (H), optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, config = data
    device = inp.device
    bs, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # Load weights
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim)

    # ------------------------------------------------------------------ #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2) Linear projections (no bias) – keep in float32 for now
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 3) Optional mask (broadcast on feature dim)
    # ------------------------------------------------------------------ #
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5) Core TriMul – batched GEMM with Triton
    # ------------------------------------------------------------------ #
    # Cast to fp16 for the kernel (Tensor‑Core friendly)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare tensors in [B, H, N, N] layout where the last two dims are
    # (i,k) for left and (k,j) for right.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N] (i,k)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k,j)

    # Collapse batch and hidden into a single dimension for the kernel.
    batch_h = bs * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)                # [B*H, N, N]
    right_mat = right_perm.view(batch_h, N, N)              # [B*H, N, N]
    out_mat = torch.empty_like(left_mat)                    # fp16 accumulator

    # Strides in elements (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)          # M‑dimension (i)
    stride_a_k = left_mat.stride(2)          # K‑dimension (k)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)         # K‑dimension (k)
    stride_b_n = right_mat.stride(2)         # N‑dimension (j)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)           # M‑dimension (i)
    stride_c_n = out_mat.stride(2)           # N‑dimension (j)

    # Choose tile sizes. 128×128 tiles give good occupancy on H100 for
    # half‑precision tensor‑core kernels; K‑tile of 64 aligns with warp‑level WMMA.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        batch_h,                        # packed (batch, hidden)
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well for 128‑tile on H100
    )

    # Restore the original layout and convert back to float32
    out = out_mat.view(bs, hidden_dim, N, N)            # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------ #
    # 6) Post‑processing: layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                       # gate is float32, same shape
    out = F.linear(out, to_out_w)              # back to dim: [B, N, N, dim]

    return out
