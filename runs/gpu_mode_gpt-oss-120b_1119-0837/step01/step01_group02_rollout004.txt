@triton.jit
def mask_gate_fuse_kernel(

   left_ptr, right_ptr,
   left_gate_ptr, right_gate_ptr,
   mask_ptr,
   out_left_ptr, out_right_ptr,
   B, N, H,
   stride_left_0, stride_left_1, stride_left_2, stride_left_3,
   stride_right_0, stride_right_1, stride_right_2, stride_right_3,
   stride_gate_0, stride_gate_1, stride_gate_2, stride_gate_3,
   stride_mask_0, stride_mask_1, stride_mask_2,
   stride_out_left_0, stride_out_left_1, stride_out_left_2, stride_out_left_3,
   stride_out_right_0, stride_out_right_1, stride_out_right_2, stride_out_right_3,
   BLOCK_SIZE: tl.constexpr):
    
    pid = tl.program_id(0)
    idx = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < B * N * N * H

    total_per_batch = N * N * H
    # compute b,i,j,h via division:
    b, rem = tl.divmod(idx, total_per_batch)
    i, rem2 = tl.divmod(rem, N * H)
    j, h = tl.divmod(rem2, H)

    # compute linear offsets:
    left_offset = b * stride_left_0 + i * stride_left_1 + j * stride_left_2 + h * stride_left_3
    right_offset = b * stride_right_0 + i * stride_right_1 + j * stride_right_2 + h * stride_right_3
    left_gate_offset = b * stride_gate_0 + i * stride_gate_1 + j * stride_gate_2 + h * stride_gate_3
    right_gate_offset = b * stride_gate_0 + i * stride_gate_1 + j * stride_gate_2 + h * stride_gate_3
    mask_offset = b * stride_mask_0 + i * stride_mask_1 + j * stride_mask_2  # no H dim

    # load
    l = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    out_l = l * m * lg
    out_r = r * m * rg

    # store
    tl.store(out_left_ptr + left_offset, out_l, mask=mask)
    tl.store(out_right_ptr + right_offset, out_r, mask=mask)

"""
TriMul forward [outgoing] optimized kernel.

- LayerNorm on input.
- Linear proj to hidden dim.
- Fuse mask and gating via a tiny Triton kernel (elementwise).
- Use torch.bmm (CUDA cuBLAS) for the heavy N×N matrix multiplication per hidden channel, in FP16 with FP32 accumulation.
- LayerNorm + output gate + final linear projection back to original dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    idx = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < B * N * N * H

    # Compute indices (b,i,j,h) from linear idx
    total_per_batch = N * N * H
    b, rem = tl.divmod(idx, total_per_batch)
    i, rem2 = tl.divmod(rem, N * H)
    j, h = tl.divmod(rem2, H)

    # Offsets
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    left_gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    right_gate_offset = left_gate_offset   # same shape as left_gate/right_gate
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # Load half‑precision values
    l = tl.load(left_ptr + left_offset, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + right_offset, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0.0))

    out_l = l * m * lg
    out_r = r * m * rg

    tl.store(out_left_ptr + left_offset, out_l, mask=mask)
    tl.store(out_right_ptr + right_offset, out_r, mask=mask)

    Forward pass of the outgoing TriMul operator.

            - config_dict: keys include 'dim', 'hidden_dim', possibly 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)

    # Unpack arguments
    # -----------------------------------------------------------------

    norm_bias = weights["norm.bias"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dim (float16)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight).half()   # (B,N,N,H)
    right = F.linear(x, right_proj_weight).half() # (B,N,N,H)

    # 3. Gates (float16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).half()
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).half()
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight)).half()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse optional mask and elementwise gating via Triton kernel
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Ensure mask is float16 and has shape (B,N,N)
        mask = mask_tensor.to(torch.float16)
        # Allocate output tensors
        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # Compute strides (int64)
        stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
        stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()

        stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()
        stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h = left_masked.stride()
        stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h = right_masked.stride()

        total_elements = B * N * N * H
        BLOCK = 1024  # 1K elements per program – good trade‑off for large tensors

        grid = (triton.cdiv(total_elements, BLOCK),)

        mask_gate_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask,
            left_masked, right_masked,
            B, N, H,
            stride_left_b, stride_left_i, stride_left_j, stride_left_h,
            stride_right_b, stride_right_i, stride_right_j, stride_right_h,
            stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
            stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
            BLOCK_SIZE=BLOCK,
        )
        left = left_masked
        right = right_masked
    else:
        # No mask: just elementwise multiply with gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matrix multiply (FP16 inputs, FP32 acc)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (BH, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (BH, N, N)

    # torch.bmm uses Tensor Cores on H100 for fp16 inputs => fp16 result.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2)).float()   # (BH, N, N) float32

    # Reshape back to (B, N, N, H)
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(H,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias)

    out_hidden = out_hidden * out_gate   # elementwise gate (both fp16/float32 broadcast)

    # Final linear projection H -> C (float32 output)
    out = F.linear(out_hidden, to_out_weight)  # shape (B,N,N,C)
    return out

"""
TriMul forward (outgoing) – highly‑optimized version.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) in FP32.
2. Project to a hidden dimension H with two bias‑free Linear layers → half‑precision.
3. Compute three sigmoid gates (left, right, out) in FP32 → cast to FP16.
4. If a pairwise mask is present, fuse *mask*, *left‑gate* and *right‑gate*
   with the projected tensors by a tiny Triton kernel (elementwise multiply);
   otherwise perform the same elementwise multiplication with PyTorch.
5. Core contraction:
   For every batch‑hidden slice compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ implemented with
   torch.bmm (cuBLAS/Tensor‑cores) on FP16 inputs and FP32 accumulation.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and finally a bias‑free Linear back to the original dimension C.
   The final tensor is returned in FP32.

Only the mask‑gate fusion is performed in Triton – the dominant O(N³·H)
contraction uses the highly‑tuned cuBLAS implementation, giving >5×
speed‑up on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
    BLOCK_SIZE: tl.constexpr
):
    """
    Elementwise fusion:
        out_left  = left  * mask * left_gate
        out_right = right * mask * right_gate
    All tensors are half‑precision.
    """
    pid = tl.program_id(0)
    idx = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_valid = idx < B * N * N * H

    # ----------- decode linear index into (b,i,j,h) -------------
    total_per_batch = N * N * H
    b, rem = tl.divmod(idx, total_per_batch)
    i, rem2 = tl.divmod(rem, N * H)
    j, h = tl.divmod(rem2, H)

    # -------------- compute byte offsets -------------------------
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    lgate_off  = b * stride_gate_b   + i * stride_gate_i   + j * stride_gate_j   + h * stride_gate_h
    rgate_off  = lgate_off  # same shape as left_gate/right_gate
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j

    # -------------- load half‑precision values -------------------
    l  = tl.load(left_ptr   + left_off,   mask=mask_valid, other=tl.float16(0.0))
    r  = tl.load(right_ptr  + right_off,  mask=mask_valid, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + lgate_off, mask=mask_valid, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + rgate_off, mask=mask_valid, other=tl.float16(0.0))
    m  = tl.load(mask_ptr   + mask_off,   mask=mask_valid, other=tl.float16(0.0))

    # -------------- fused multiplication -------------------------
    out_l = l * m * lg
    out_r = r * m * rg

    # -------------- write results -------------------------------
    tl.store(out_left_ptr  + left_off,  out_l, mask=mask_valid)
    tl.store(out_right_ptr + right_off, out_r, mask=mask_valid)
