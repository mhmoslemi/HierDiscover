```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1) Layer‑norm the input tensor `x` over the last dimension (dim).
2) Project `x` to a hidden dimension (hidden_dim) with two linear layers:
       left = x @ W_left   (shape [B,N,N,HD])
       right= x @ W_right
3) (Optional) apply a pairwise mask of shape [B,N,N] to both tensors.
4) Compute three sigmoid gates from `x` (left‑gate, right‑gate, out‑gate) and
   multiply them element‑wise with `left` and `right`.
5) The core cubic operation:
       out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   is a batched matrix‑multiply for every hidden channel:
       out[:,:,d] = left[:,:,d] @ right[:,:,d].T
   This is implemented with a Triton kernel (`batched_gemm_kernel`) that
   processes a batch of independent GEMMs (`batch*hidden` matrices) in FP16
   while accumulating in FP32.
6) Apply a second LayerNorm on the hidden dimension, multiply with the
   out‑gate and a final linear projection back to `dim`.

The kernel works on tensors of shape (B,HD,N,N) that are contiguous in memory.
It supports any `B∈{1,2}`, `N∈{128,256,512,1024}`, `dim∈{128,384}` and
`hidden_dim∈{128,384,768}`.  All heavy computation (the N³ term) is done in
the Triton kernel; the surrounding linear / norm / sigmoid ops use PyTorch.
"""

import torch
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton GEMM kernel (batch of independent matrix multiplies)
# -------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)  (Note: we read B transposed)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # which matrix in the batch

    # ----- tile start indices -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base offsets for the current matrix
    a_batch_off = pid_b * stride_ah
    b_batch_off = pid_b * stride_bh
    c_batch_off = pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over the reduction dimension K ------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_off + \
                 offs_m[:, None] * stride_am + \
                 cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N).  We read B as if it were transposed,
        # i.e. rows correspond to the reduction index k and columns to j.
        b_ptrs = b_ptr + b_batch_off + \
                 cur_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # ----- write back ---------------------------------------------------------
    c_ptrs = c_ptr + c_batch_off + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for a batch of (B*HD) GEMMs
# ----------------------------------------------------------------------
def batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B: tensors of shape (batch, hidden, N, N) and dtype float16,
          contiguous in memory.
    Returns C of same shape and dtype.
    """
    assert A.shape == B.shape
    BATCH, HD, N, _ = A.shape
    # Collapse batch*hidden into a single dimension for the kernel
    A_flat = A.reshape(-1, N, N)
    B_flat = B.reshape(-1, N, N)
    C_flat = torch.empty_like(A_flat)   # (batch*hidden, N, N)

    # Block sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # rows
        (N + BLOCK_N - 1) // BLOCK_N,      # cols
        A_flat.shape[0],                   # number of independent GEMMs
    )

    # strides (in elements) for flattened tensors
    stride_ah = A_flat.stride(0)
    stride_am = A_flat.stride(1)
    stride_ak = A_flat.stride(2)

    stride_bh = B_flat.stride(0)
    # B is interpreted as transposed; column stride == 1, row stride == N
    stride_bk = B_flat.stride(2)   # stride for reduction index k (original col)
    stride_bn = B_flat.stride(1)   # stride for output column j (original row)

    stride_ch = C_flat.stride(0)
    stride_cm = C_flat.stride(1)
    stride_cn = C_flat.stride(2)

    batched_gemm_kernel[grid](
        A_flat, B_flat, C_flat,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (batch, hidden, N, N)
    return C_flat.reshape(BATCH, HD, N, N)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the forward pass of the TriMul (outgoing) module using a Triton
    kernel for the cubic pairwise reduction.

    data : Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]
        input      – Tensor of shape [B, N, N, dim]
        mask       – Tensor of shape [B, N, N]  (may be None)
        weights    – Mapping from layer‑name strings to parameter tensors
        config     – Dictionary with keys:
                     * "dim"         – input feature dimension
                     * "hidden_dim" – hidden projection dimension
                     * "nomask"      – bool, True if mask should be ignored
    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32).
    """
    # unpack
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # left / right shape: [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and gated projection
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the cubic part (reduces memory bandwidth)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) and make tensors contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 6) Core N³ operation via Triton batched GEMM
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # ------------------------------------------------------------------
    out_perm = batched_gemm(left_perm, right_perm)       # (B, hidden, N, N)

    # Back to the original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()      # (B, N, N, hidden)

    # Cast back to FP32 for the remaining FP32‑only ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8) Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
