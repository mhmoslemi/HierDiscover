"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the reference PyTorch version of the
‘outgoing’ TriMul module (AlphaFold‑3).  The heavy O(N³) “triangle”
multiplication

    out[b, i, j, h] = Σₖ left[b, i, k, h] · right[b, j, k, h]

is executed in a custom Triton kernel that treats each hidden
dimension (and batch) as an independent “channel”.  For a given
channel the operation is a standard matrix multiplication
L_h @ R_hᵀ, so the kernel is essentially a batched GEMM with fp16
operands and fp32 accumulation (Tensor‑Core friendly).

Key optimisation choices
-------------------------

* **Mixed‑precision** – all linear projections, gates and the
  Triton kernel use fp16.  Accumulation is fp32 to keep numerical
  stability.
* **Tile sizes** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 64` give
  64×64 output tiles while keeping each warp within the 1024‑thread
  limit.  The kernel is launched with 8 warps (`num_warps=8`) and
  4 pipeline stages (`num_stages=4`) to hide memory latency.
* **Memory layout** – after projection the tensors are permuted to
  `(B, H, N, N)` and reshaped to `(C_total, N, N)` where
  `C_total = B·H`.  This makes the channel dimension contiguous,
  allowing the Triton kernel to stride over it with a single program
  ID.
* **Mask handling** – when a mask is present it is applied before
  the triangular multiplication (as in the reference code) but is
  fused into the element‑wise gate multiplication to avoid extra
  memory passes.
* **Fusion outside Triton** – LayerNorms, sigmoid gates and the final
  projection are kept in PyTorch; they are cheap compared to the
  O(N³) core.

The function `custom_kernel` expects a tuple
`(input_tensor, mask, weights, config)` and returns the processed
tensor of shape `[B, N, N, C]` (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32 *
    a_ptr,              # fp16 *   (C, M, K)   – left
    b_ptr,              # fp16 *   (C, K, N)   – right (transposed)
    M, N, K,            # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,
    stride_a_m, stride_a_k, stride_a_c,
    stride_b_k, stride_b_n, stride_b_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the triangle multiplication.

    For each channel `c` (c = batch * hidden_dim) we compute:
        out[c, i, j] = Σ_k left[c, i, k] * right[c, j, k]
    with fp16 inputs and fp32 accumulation.
    """
    pid_m = tl.program_id(0)   # tile row  (i)
    pid_n = tl.program_id(1)   # tile col  (j)
    pid_c = tl.program_id(2)   # channel   (batch·hidden)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k      # (M, K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n      # (K, N)

        # Load with masking
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                           # fp16
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                           # fp16

        # fp16·fp16 → fp32 dot product (Tensor‑Core)
        acc += tl.dot(a, b)

    # Write the result tile
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) → left/right  (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – keep in FP16 for element‑wise multiplications
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # [B,N,N,H] – later used after the core multiplication

    # ------------------------------------------------------------------
    # 5) Apply gates and optional mask (broadcast over H)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if (not nomask) and mask is not None:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Prepare data for the Triton kernel
    #    Layout: (C_total, N, N) where C_total = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dim before the two sequence dimensions for contiguity
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)    # (C, M, K)
    right_flat = right.reshape(C_total, N, N)  # (C, K, N) – we will treat K dim as rows

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)   # K dimension (rows after transpose)
    stride_b_n = right_flat.stride(2)   # N dimension (cols)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (batch * hidden)
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9) Post‑processing: second LayerNorm, gating, final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply out_gate (converted to FP32 for stability)
    out = out * out_gate.to(torch.float32)

    # Final projection back to original dimensionality C
    out = F.linear(out, weights["to_out.weight"])

    return out
