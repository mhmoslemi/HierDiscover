"""
TriMul (outgoing) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm over the last feature dimension *dim*.
2. Two bias‑free linear projections (`left_proj`, `right_proj`) and three
   linear gates (`left_gate`, `right_gate`, `out_gate`). The gates are passed
   through a sigmoid and applied element‑wise.
3. An optional binary mask (broadcast over the hidden dimension) is applied
   to the projected tensors.
4. The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix‑multiply per hidden channel.  We implement it with a
   custom Triton kernel that works in FP16 for bandwidth and accumulates in
   FP32 (Tensor‑Core friendly).  The kernel is written so that the right
   operand is **read transposed on‑the‑fly** – no explicit `permute`/`contiguous`
   copy is needed, saving memory traffic.
5. The result (still FP16) is reshaped back to `[B,N,N,hidden]`, converted to
   FP32 and processed by a second layer‑norm, the output‑gate, and a final
   linear projection back to the original *dim*.

The implementation keeps all non‑linear / normalisation work in PyTorch (highly
optimised) and only the O(N³) GEMM‑like contraction is handled by Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : [batch, N, N])
#   A and B are FP16, accumulation in FP32.
#   Bᵀ is realized by swapping the logical row/col strides (no explicit transpose).
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,              # A row‑stride, col‑stride
    stride_bk, stride_bn,              # B row‑stride, col‑stride (transposed view)
    stride_cm, stride_cn,              # C row‑stride, col‑stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM:  C[b] = A[b] @ B[b].T   (A: M×K, B: N×K)"""
    pid = tl.program_id(axis=0)          # tile index over (M,N)
    batch = tl.program_id(axis=1)        # batch index (B * hidden)

    # ------------------------------------------------------------------- #
    #   Tile coordinates
    # ------------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------- #
    #   Base pointers for this batch
    # ------------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # ------------------------------------------------------------------- #
    #   Accumulator (FP32)
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    #   Main K‑loop
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # ----- Load a tile of A (M × K) ---------------------------------
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & mask_k[None, :],
            other=0.0,
        )

        # ----- Load a tile of Bᵀ (K × N) --------------------------------
        # B is stored as (row=i, col=k) – we want (row=k, col=j) → swap strides
        b_ptrs = b_batch + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )

        # ----- Multiply‑accumulate ---------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    # ------------------------------------------------------------------- #
    #   Store the result (convert back to FP16)
    # ------------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N)
    )


# --------------------------------------------------------------------------- #
# Entry‑point used by the test harness
# --------------------------------------------------------------------------- #

    """

    Parameters
    ----------

    """

    #   Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),

        bias=weights["norm.bias"],
        eps=eps,

    #   2) Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    #   3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not config.get("nomask", True):
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    #   4) Gating (sigmoid on linear maps)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    #   5) Core contraction – batched GEMM in FP16 (Tensor‑Cores)
    #       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #   We avoid an explicit transpose of the right operand by swapping strides.
    # ------------------------------------------------------------------- #
    #   (B, hidden, N, N) layout, then flatten the first two dims.
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # (B, H, N, N)

    B_, H_, N_, _ = left_fp16.shape          # B_ == B, H_ == hidden_dim, N_ == seq_len
    total_batches = B_ * H_

    # View as classic batched matrices (batch, M, K) with M=K=N_
    a = left_fp16.view(total_batches, N_, N_)
    b = right_fp16.view(total_batches, N_, N_)
    c = torch.empty((total_batches, N_, N_), dtype=torch.float16, device=device)

    # ----- Strides (in elements, not bytes) ---------------------------------
    stride_a_batch = N_ * N_
    stride_am = N_          # row‑stride of A
    stride_ak = 1           # col‑stride of A

    stride_b_batch = N_ * N_
    # B is accessed transposed (row ← k, col ← j)
    stride_bk = 1           # logical row‑stride after transposition
    stride_bn = N_          # logical column‑stride after transposition

    stride_c_batch = N_ * N_
    stride_cm = N_          # row‑stride of C
    stride_cn = 1           # col‑stride of C

    # ----- Tiling parameters -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        ((N_ + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),
        total_batches,
    )

    # Launch the kernel
    _tri_mul_gemm[grid](
        a, b, c,
        N_, N_, N_,                # M, N, K (all equal to seq_len)
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------- #
    #   Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # ------------------------------------------------------------------- #
    out_hidden = c.view(B_, hidden_dim, N_, N_).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------- #
    #   6) Post‑processing: hidden‑dim layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------- #
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])   # back to `dim`

    return out
```
