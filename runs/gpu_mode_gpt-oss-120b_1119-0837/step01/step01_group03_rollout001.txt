"""
TriMul (outgoing) – highly‑optimised forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input `x`  (shape [B,N,N,C]) in FP16.
2. Linear projections (bias‑free) to the hidden dimension H and the three
   gating matrices.  All linear layers are executed in FP16.
3. Optional mask (broadcasted on H) is applied in‑place.
4. Fuse the project‑and‑gate results:
        left  = (x @ W_left )  * sigmoid(x @ W_left_gate )
        right = (x @ W_right)  * sigmoid(x @ W_right_gate)
5. Core O(N³·H) operation: for every hidden slice h (and batch b) compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N}.  The kernel
   operates on FP16 operands and accumulates in FP32, thus using Tensor
   Cores on H100.
6. Second LayerNorm over H, multiply by the output‑gate and a final
   projection back to the original channel dimension C.
All heavy work (step 5) is performed by a single Triton kernel; the
remaining steps use PyTorch which already calls highly‑optimised cuBLAS /
cudnn kernels.  The whole pipeline stays in FP16 except for the final
output which is cast back to FP32 for downstream use.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : [B*H, M, K]   (row‑major, FP16)
# B : [B*H, N, K]   (row‑major, FP16, interpreted as K×N after transposition)
# C : [B*H, M, N]   (FP16 output, accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers to tensors
    M, N, K,                    # matrix sizes (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (treated as transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A and B are interpreted as FP16 matrices, accumulation happens in FP32.
    Each program instance processes one tile (BLOCK_M × BLOCK_N) of the
    output matrix for a single (batch, hidden) pair.
    """
    pid_b = tl.program_id(2)          # combined batch*hidden index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Pointers to the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_a_m \
                         + (k_offset + offs_k)[None, :] * stride_a_k
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_b_k \
                         + offs_n[None, :] * stride_b_n

        # Load with boundary checks
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # FP16

        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # Write the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weight_dict  : mapping of weight names to torch.Tensors
        - config_dict  : contains `dim`, `hidden_dim`, optional `nomask`

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the result of the TriMul
        module.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast everything to FP16 for the heavy part
    # ------------------------------------------------------------------
    dtype = torch.float16
    x = input_tensor.to(dtype)

    # ------------------------------------------------------------------
    # Load weights and cast to FP16 (no bias in the linear layers)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)

    left_proj_weight  = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left_gate_weight  = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight   = weights["out_gate.weight"].to(dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype)

    to_out_weight = weights["to_out.weight"].to(dtype)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, C]  (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Projections + 3️⃣ (optional) mask
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³·H operation – batched GEMM with Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim
    # Rearrange to [B, H, N, N] → view as (B*H, N, N) each GEMM operand
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_t.view(B * H, N, N)   # A : [B*H, M=N, K=N]
    right_mat = right_t.view(B * H, N, N)  # B : [B*H, N, K=N] (will be used transposed)

    # Output buffer (FP16)
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100)
    # ------------------------------------------------------------------
    # Tile sizes: 64 works well for all provided N, 128 is safe up to N≥512.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over rows (M)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over cols (N)
        B * H                           # batch × hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_mat.stride(0),
        stride_a_m=left_mat.stride(1),
        stride_a_k=left_mat.stride(2),
        stride_b_batch=right_mat.stride(0),
        # Bᵀ ⇒ swap the last two strides
        stride_b_k=right_mat.stride(2),
        stride_b_n=right_mat.stride(1),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 can efficiently run 8 warps per block
        num_stages=3,         # pipelined shared‑memory loads
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating and final projection (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # → [B, N, N, C] (FP16)

    # Return in FP32 as required by the harness
    return out.to(torch.float32)
