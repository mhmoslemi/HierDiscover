def custom_kernel(data):
    """...
    """
    input_tensor, mask, weights, config = data
    # Ensure they are on GPU
    device = input_tensor.device
    # Unpack config
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load all weights
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # 1. LayerNorm on last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # 2. Projections
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    # 3. optional mask
    if mask is not None and not nomask:
        # convert mask to same dtype, unsqueeze last dim
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # 5. Core tril multiplicative update via kernell
    # Convert to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

"""
TriMul (outgoing) – fast forward implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and computes

    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]

where
    left  = (x·W_left  )·σ(x·W_left_gate )  (optional pair‑wise mask)
    right = (x·W_right )·σ(x·W_right_gate)

All linear layers have no bias.  A layer‑norm is applied to the input,
another layer‑norm to the result, the result is gated by σ(x·W_out_gate)
and finally projected back to the original channel dimension C.

Only the cubic “core” (the batched GEMM C = A @ Bᵀ) is written in Triton.
The surrounding ops use PyTorch’s highly‑optimised kernels.
The kernel computes a batched fp16 GEMM with fp32 accumulation
and writes the result in fp16; the surrounding code converts back to fp32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    a_ptr,            # left  matrix A  [batch, M=N, K=N] (fp16)
    b_ptr,            # right matrix Bᵀ [batch, K=N, N] (fp16) – already transposed
    c_ptr,            # output C        [batch, M=N, N] (fp16)
    M, N, K,          # matrix sizes (all equal to seq_len)
    stride_a_batch,   # stride between consecutive batch entries in A
    stride_a_i,       # stride for the i‑dimension (M) in A
    stride_a_k,       # stride for the k‑dimension (K) in A
    stride_b_batch,   # stride between consecutive batch entries in B
    stride_b_k,       # stride for the k‑dimension (K) in Bᵀ
    stride_b_n,       # stride for the n‑dimension (N) in Bᵀ
    stride_c_batch,   # stride between consecutive batch entries in C
    stride_c_i,       # stride for the i‑dimension (M) in C
    stride_c_n,       # stride for the n‑dimension (N) in C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.

    A : (batch, M, K)   – fp16
    B : (batch, K, N)   – fp16   (already the transposed right matrix)
    C : (batch, M, N)   – fp16   (output)
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_b = tl.program_id(2)          # batch * hidden index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialise accumulator in higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to A[i, k] and B[k, n] tiles
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Masks for tiles that exceed matrix dimensions
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Accumulate the product (fp16 x fp16 → fp32)
        acc += tl.dot(a, b)

    # Store result back to C (convert to fp16)
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input   : torch.Tensor of shape [B, N, N, C] (float32)
        - mask    : torch.Tensor of shape [B, N, N] (optional, may be None)
        - weights : dict of model parameters (all tensors already on GPU)
        - config  : dict containing at least
                    * "dim"         – C
                    * "hidden_dim" – internal hidden size H
                    * "nomask"      – bool, True ⇒ ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – result of the TriMul update.
    """
    # ------------------------------------------------------------------ #
    # 0) Unpack arguments
    # ------------------------------------------------------------------ #
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1) Load model weights (all on the same device as the inputs)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # shape (C, H)

    # ------------------------------------------------------------------ #
    # 2) Input layer‑norm (over the C dimension)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(x, (C,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 3) Fuse the two projections on each side (proj + gate)
    #    This reduces three matmuls per side to a single matmul.
    # ------------------------------------------------------------------ #
    #   left_comb : [B,N,N,2*H]  = concat(left_proj, left_gate)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2H, C)

    left_comb = F.linear(x, left_comb_w)    # [B,N,N,2H]
    right_comb = F.linear(x, right_comb_w)  # [B,N,N,2H]

    # Split the concatenated dimensions
    left_proj, left_gate_raw = left_comb.chunk(2, dim=-1)   # each [B,N,N,H]
    right_proj, right_gate_raw = right_comb.chunk(2, dim=-1)

    # ------------------------------------------------------------------ #
    # 4) Optional pair‑wise mask (broadcast on the feature dim)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------ #
    # 5) Gating (sigmoid)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(left_gate_raw)    # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate                # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 6) Out‑gate (computed from the normalized input)
    # ------------------------------------------------------------------ #
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------ #
    # 7) Core multiplicative update – fused batched GEMM via Triton
    # ------------------------------------------------------------------ #
    # Convert to half‑precision for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, H, N, N] and make contiguous.
    # right must be transposed on its last two axes (k ↔ j) so that we
    # can perform a regular GEMM (A @ B) instead of A @ Bᵀ.
    left_h = left_h.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_h = right_h.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N]

    batch_h = B * H
    left_mat = left_h.view(batch_h, N, N)    # [B*H, N, N]
    right_mat = right_h.view(batch_h, N, N)  # [B*H, N, N]
    out_mat = torch.empty_like(left_mat)    # FP16 buffer for result

    # Strides in **elements** (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_i = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)   # K‑dimension
    stride_b_n = right_mat.stride(2)   # N‑dimension

    stride_c_batch = out_mat.stride(0)
    stride_c_i = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Tune block sizes – they are multiples of 16 to match tensor‑core tiles.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # Small sequence lengths need smaller tiles to avoid excessive padding.
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over cols (j)
        batch_h,                        # one block per (batch * head)
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_i=stride_c_i,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # more warps better utilisation on H100
        num_stages=3,         # pipelined shared‑memory pre‑fetching
    )

    # Convert back to float32 and restore the original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    # shape: [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 8) Post‑processing: layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    return out
