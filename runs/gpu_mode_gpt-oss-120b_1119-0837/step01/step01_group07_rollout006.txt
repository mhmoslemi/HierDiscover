"""
TriMul (outgoing) – fast Triton implementation.

The heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched GEMM (A @ Bᵀ) with:
    A : (N, N)   = left[b, :, :, h]   (rows=i, cols=k)
    B : (N, N)ᵀ = right[b, :, :, h]  (rows=k, cols=j)

Both A and B are stored in the native PyTorch layout
    [B, N, N, H]   (H = hidden_dim)
where the hidden dimension is the innermost (stride = 1).  
The kernel therefore works on *strided* matrices – no explicit
transpose or data copy is required.  The GEMM runs in FP16
(bandwidth‑friendly) while accumulation is done in FP32.
After the contraction the result is brought back to FP32,
layer‑normalized, gated and linearly projected to the original
feature dimension.

Key optimisations
-----------------
* No materialised transpose of the “right” tensor – we simply
  pass swapped strides to the kernel.
* The hidden dimension is mapped to a separate program‑ID axis,
  turning the (B × hidden) work‑group into a 3‑D grid
  (tiles over M/N, batch, hidden).  This keeps each GEMM batch
  stride constant and avoids irregular indexing.
* Large tile sizes (128 × 128) and 8 warps per block exploit the
  Tensor‑Core performance of H100.
* All non‑linear (layer‑norm, linear proj, gate, mask) work is
  performed with native PyTorch (highly optimised) and only the
  O(N³) contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel for the core contraction: out = left @ rightᵀ  (per hidden dim)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to the tensors (FP16)
    M, N, K,                           # dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,              # A row‑stride, col‑stride  (row=i, col=k)
    stride_bk, stride_bn,              # B row‑stride, col‑stride  (row=k, col=j)
    stride_cm, stride_cn,              # C row‑stride, col‑stride  (row=i, col=j)
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch (B) stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,h] = A[b,h] @ B[b,h].T

    A : (M, K)  – left[tile]
    B : (K, N)  – right[tile] (already transposed by strides)
    C : (M, N)  – output
    The hidden dimension `h` is encoded in program_id(axis=2).
    """
    pid = tl.program_id(axis=0)          # tile over M & N
    batch = tl.program_id(axis=1)        # batch index (0 .. B‑1)
    hidden = tl.program_id(axis=2)       # hidden index (0 .. hidden‑dim‑1)

    # ----------------------------------------------------------------------- #
    #   Tile coordinates
    # ----------------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows within A / C
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols within B / C

    # ----------------------------------------------------------------------- #
    #   Base pointers for this (batch, hidden) pair
    # ----------------------------------------------------------------------- #
    a_base = a_ptr + batch * stride_a_batch + hidden      # hidden stride = 1
    b_base = b_ptr + batch * stride_b_batch + hidden
    c_base = c_ptr + batch * stride_c_batch + hidden

    # ----------------------------------------------------------------------- #
    #   Accumulator (FP32)
    # ----------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------- #
    #   Loop over the reduction axis K
    # ----------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load a block of A (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs)

        # Load a block of B (K × N) – note that B is *already* transposed by using
        # swapped strides (row = k, col = j)
        b_ptrs = b_base + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs)

        # MAC – uses Tensor‑Core FP16×FP16→FP32 dot
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------------- #
    #   Write back the result (FP16)
    # ----------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16))


# -----------------------------------------------------------------------------#
# Entry point called by the tester
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, seq_len, seq_len, dim] (dtype=torch.float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------- #
    # 1) Layer‑norm over the feature dimension (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------- #
    # 2) Linear projections (bias=False) – produce left/right tensors
    # --------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3) Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------- #
    if not nomask:
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms)
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5) Core contraction via Triton (FP16 input, FP32 accumulation)
    # --------------------------------------------------------------- #
    # Cast to half‑precision (this creates a new contiguous tensor)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output tensor (same layout as inputs)
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------- #
    # Strides for the native layout [B, N, N, H] (contiguous)
    # ------------------------------------------------------------------- #
    # left
    a_stride_b, a_stride_i, a_stride_k, a_stride_h = left_fp16.stride()
    # right
    b_stride_b, b_stride_j, b_stride_k, b_stride_h = right_fp16.stride()
    # output
    c_stride_b, c_stride_i, c_stride_j, c_stride_h = out_fp16.stride()

    # ------------------------------------------------------------------- #
    # Kernel launch configuration
    # ------------------------------------------------------------------- #
    # Choose tile sizes that exactly divide the sequence length in all test cases
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Number of tiles covering the M×N output plane
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m * grid_n, B, hidden_dim)   # (tiles over M,N), batch, hidden

    # Launch the Triton kernel
    _trmul_kernel[grid](
        # pointers
        left_fp16, right_fp16, out_fp16,
        # problem sizes
        N, N, N,
        # A strides
        a_stride_i, a_stride_k,
        # B strides (note the swap: rows = k, cols = j)
        b_stride_k, b_stride_j,
        # C strides
        c_stride_i, c_stride_j,
        # batch strides (move to next element in B dimension)
        a_stride_b, b_stride_b, c_stride_b,
        # tile sizes
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Convert back to FP32 for the remaining post‑processing steps
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------- #
    # 6) Hidden‑dim layer‑norm, out‑gate multiplication and final linear
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
