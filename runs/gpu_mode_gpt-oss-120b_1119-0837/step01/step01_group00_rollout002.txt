"""
custom_kernel
==============

Forward pass for the *outgoing* TriMul module (AlphaFold‑3 style) using a
high‑performance Triton GEMM kernel.  

Algorithm
---------
1. **Layer‑norm** the (B,N,N,C) input.
2. **Half‑precision linear projections** to a hidden dimension H (no bias).
3. **Sigmoid gates** (`left_gate`, `right_gate`, `out_gate`), optional mask
   broadcasted over H, and gating of the projected tensors.
4. **Triangular multiplication**  
   For each batch‑head pair (b,h) compute  

   ``out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]``  

   which is exactly the matrix product ``A @ B.T`` with  
   ``A = left.reshape(-1,N,N)`` and ``B = right.reshape(-1,N,N)``.  
   This is performed by a custom Triton kernel that treats each
   (b,h) pair as an independent program‑ID and executes a tiled
   FP16‑Tensor‑Core GEMM (accumulating in FP32).
   Block sizes are chosen adaptively (64/128) to keep occupancy high on
   the H100.
5. **Post‑processing** – layer‑norm on the hidden output, gating with
   `out_gate`, and a final linear projection back to the original channel
   dimension C.

All heavy arithmetic (step 4) runs on the GPU in FP16 → FP32 with
Tensor‑Cores; the remaining ops use regular PyTorch (FP32 where needed).
The function returns a ``torch.float32`` tensor of shape
``[B, N, N, C]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,      # float32*
    a_ptr,        # float16*  (left)  shape (C, M, K)
    b_ptr,        # float16*  (right) shape (C, N, K) – we read it transposed
    M, N, K,      # matrix dims (all == seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left  strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides (after transpose)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul: out = A @ B.T  (all FP16, acc FP32)."""
    pid_m = tl.program_id(0)   # i‑tile index
    pid_n = tl.program_id(1)   # j‑tile index
    pid_c = tl.program_id(2)   # hidden‑channel index (batch * hidden_dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k      # (M,K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n      # (K,N)

        # Load tiles (FP16).  Use 0.0 for out‑of‑bounds elements.
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core compatible GEMM: FP16×FP16 → FP32
        acc += tl.dot(a, b)

    # Write the result tile (FP32) back to global memory
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] or None  (bool/float) – may be ignored.
            - weights: dict of torch.Tensor containing all linear / norm parameters.
            - config: dict with keys
                  "dim" (C), "hidden_dim" (H), optional "nomask" (bool)

    Returns:
        torch.Tensor[B, N, N, C] – float32 result.
    """
    # ------------------------------------------------------------------
    # 0) unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Input layer‑norm  (FP32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape (B, N, N, C)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear ops
    x_h = x.to(torch.float16)

    # 3) Linear projections (no bias)   → shape (B, N, N, H)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # (B,N,N,H)
    right = F.linear(x_h, right_proj_w)  # (B,N,N,H)

    # 4) Gate logits → sigmoid (still FP16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # (B,N,N,H)

    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6) Optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 7) Prepare for the GEMM kernel
    B, N, _, _ = left.shape
    # Move hidden dim before the sequence dims for a contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    C_total = B * hidden_dim                           # treat (b,h) as a single batch
    left_flat = left.reshape(C_total, N, N)            # (C, M, K)
    right_flat = right.reshape(C_total, N, N)          # (C, N, K) – we will read transposed

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) GEMM kernel launch
    # Choose block sizes based on sequence length (empirically good values)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Strides in element units (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # inner dim (K) after transpose
    stride_b_n = right_flat.stride(1)   # middle dim (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 can comfortably host 8 warps per block
        num_stages=2,         # double‑buffering of tiles
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 10) Post‑processing: norm → gate → linear back to C
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate currently FP16; promote to FP32 for the final multiplication
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to original channel dimension (C)
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,C)

    return out
