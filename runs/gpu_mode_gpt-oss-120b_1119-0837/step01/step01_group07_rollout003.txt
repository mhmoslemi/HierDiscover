"""
TriMul (outgoing) – high‑performance Triton implementation.

The heavy O(N³) contraction
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched GEMM per hidden channel (h) and batch (b):
   C[b,h] = A[b,h] @ B[b,h].T
with A and B of shape (N, N) and data type FP16 (bandwidth) while the
accumulator is FP32 (precision).  The kernel works directly on the
original [B, N, N, H] layout, avoiding any costly permutes or extra
copies.  All other operations (LayerNorm, Linear projections,
gating, final projection) are performed with plain PyTorch which is
already highly optimized.

Key ideas:
* Keep the memory layout (B,N,N,H) – the stride for the hidden dimension
  is 1, so each channel is contiguous and can be addressed via a
  separate program‑id dimension.
* Use three program‑id axes: tiles over the (M,N) output matrix, batch
  index (B) and hidden‑channel index (H).
* Load A and B tiles with the correct strides (A rows → dim‑1 stride,
  A cols → dim‑2 stride; B is accessed transposed using the same strides
  in swapped order).
* Accumulate in FP32, store as FP16 and cast back to FP32 for the
  remaining PyTorch post‑processing.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ  (per batch & per hidden channel)
# --------------------------------------------------------------------------- #
@triton.jit
def _trmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                      # base pointers
    M, N, K,                                   # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak,                      # A row‑stride, col‑stride
    stride_bk, stride_bn,                      # B (transposed) row‑stride, col‑stride
    stride_cm, stride_cn,                      # C row‑stride, col‑stride
    stride_batch, stride_hidden,               # batch‑stride (dim‑0) and channel‑stride (dim‑3)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for a single hidden channel:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    where A = left, B = right (transposed on‑the‑fly).
    """
    pid = tl.program_id(axis=0)         # tile over M×N
    batch_id   = tl.program_id(axis=1)  # batch (B)
    channel_id = tl.program_id(axis=2)  # hidden channel (H)

    # --------------------------------------------------------------- #
    #   Tile indices
    # --------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid %  num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------- #
    #   Base pointers for this (batch, channel)
    # --------------------------------------------------------------- #
    a_base = a_ptr + batch_id * stride_batch + channel_id * stride_hidden
    b_base = b_ptr + batch_id * stride_batch + channel_id * stride_hidden
    c_base = c_ptr + batch_id * stride_batch + channel_id * stride_hidden

    # --------------------------------------------------------------- #
    #   Accumulator (FP32)
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    #   Loop over the reduction axis K
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # ---- Load A[i, k] -------------------------------------------------
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # ---- Load Bᵀ[k, j] (i.e. right[j, k]) -------------------------------
        b_ptrs = b_base + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # ---- Multiply‑accumulate (tensor‑core friendly) --------------------
        acc += tl.dot(a, b)                     # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N)

    # --------------------------------------------------------------- #
    #   Store the result (convert back to FP16)
    # --------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

# Custom entry‑point used by the grader
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype=torch.float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections (bias=False) – produce tensors of shape [B,N,N,H]
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not config.get("nomask", True):
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid) – still in FP32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to FP16 (the heavy contraction works in half‑precision)
    # ------------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer (FP16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # 5a) Prepare strides for the Triton kernel (in elements)
    # --------------------------------------------------------------- #
    # left/right share the same layout: (B, N, N, H)
    stride_batch = left_fp16.stride(0)               # B‑stride = N*N*H
    stride_i     = left_fp16.stride(1)               # row‑stride (M) = N*H
    stride_k     = left_fp16.stride(2)               # col‑stride (K) = H
    stride_h     = left_fp16.stride(3)               # channel‑stride = 1

    # For the transposed right matrix we simply swap the i/k strides.
    #   Bᵀ has rows = k (stride = H) and cols = j (stride = N*H)
    stride_bk = stride_k        # row‑stride of Bᵀ
    stride_bn = stride_i        # col‑stride of Bᵀ

    # Output strides are identical to A's (row=i, col=j)
    stride_cm = stride_i
    stride_cn = stride_k

    # --------------------------------------------------------------- #
    # 5b) Launch the Triton kernel
    # --------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32   # good trade‑off on H100
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B, hidden_dim)

    _trmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,                     # M, N, K
        stride_i, stride_k,          # stride_am, stride_ak
        stride_bk, stride_bn,        # stride_bk, stride_bn
        stride_cm, stride_cn,        # stride_cm, stride_cn
        stride_batch, stride_h,      # batch‑stride, channel‑stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # H100 efficiently uses 8 warps per block
    )

    # --------------------------------------------------------------- #
    # 6) Convert back to FP32 for the remaining PyTorch ops
    # --------------------------------------------------------------- #
    out = out_fp16.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------- #
    # 7) Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to `dim`

    return out
```
