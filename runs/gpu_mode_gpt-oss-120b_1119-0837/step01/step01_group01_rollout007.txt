```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H, stride_b, stride_i, stride_j, stride_h,
    # right strides:
    stride_b_r, stride_i_r, stride_j_r, stride_h_r,
    # no need if same as left
    # For left (i,k) shape: stride_i = N*H, stride_k = H
    # For right (j,k) shape: stride_i = N*H (j), stride_k = H (k)
    # but we can use param K dimension, etc.
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined batch*hidden id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute b and h
    b = pid // H
    h = pid % H

    # offsets for rows i and columns j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for left slice A = left[b,:, :,h]
    base_a = left_ptr + b * stride_b + h * stride_h

    # pointers to A rows
    a_ptrs = base_a + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k

    # base pointer for right slice B = right[b,:, :,h]
    base_b = right_ptr + b * stride_b_R + h * stride_h_R

    # load B transposed: we want B[j,k,h] -> after transpose, row = k, col = j
    # So pointer: base_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_j
    b_ptrs = base_b + offs_k[:, None] * stride_k_R + offs_j[None, :] * stride_j_R

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_tiles):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_j[None, :], other=0.0)

        # convert to float32
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_k
        b_ptrs += BLOCK_K * stride_k_R  # but careful: we increment k dimension pointer

    # store result to out
    c = acc.to(tl.float16)
    # base pointer for output
    base_c = out_ptr + b * stride_b + h * stride_h
    c_ptrs = base_c + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j

    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – high‑performance implementation.
The heavy “TriMul” operation (einsum `'... i k d, ... j k d -> ... i j d'`)
is executed by a custom Triton kernel that performs many independent
N×N matmuls (one per hidden slice).  The kernel works directly on the
original [B, N, N, H] layout, so no costly permute‑+‑contiguous copy is
required.  All surrounding ops (LayerNorm, Linear projections, sigmoid
gates and optional mask) are carried out with PyTorch primitives in
float16, while the final result is returned as float32.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  for many (B×H) slices.
#   A : left  tensor  [B, N, N, H]   (row i, column k, hidden h)
#   B : right tensor  [B, N, N, H]   (row j, column k, hidden h)
#   C : output tensor [B, N, N, H]   (row i, column j, hidden h)
# All tensors are FP16, accumulation in FP32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid      = tl.program_id(0)          # index over (batch, hidden) = B*H
    pid_m    = tl.program_id(1)          # block row (i)
    pid_n    = tl.program_id(2)          # block col (j)

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from pid
    # ------------------------------------------------------------------
    b = pid // H
    h = pid %  H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows of A / C
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols of B / C
    offs_k = tl.arange(0, BLOCK_K)                    # reduction axis

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for the (b, h) slice.
    # The layout of left/right/out is [B, N, N, H] (contiguous in H).
    # ------------------------------------------------------------------
    base_l = left_ptr  + b * stride_l_batch + h * stride_l_h   # A slice
    base_r = right_ptr + b * stride_r_batch + h * stride_r_h   # B slice
    base_o = out_ptr   + b * stride_o_batch + h * stride_o_h   # C slice

    # Pointers to the current K‑tile of A and Bᵀ.
    # A[i, k]   : stride_i for i, stride_k for k
    # Bᵀ[k, j]  : stride_k for k, stride_i for j (since B is stored as B[j,k])
    a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        # Global k index for this tile (used for masking)
        cur_k = offs_k + tk * BLOCK_K
        mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    cache='ca')
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    cache='ca')

        # Cast to FP32 and perform the dot product
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # ------------------------------------------------------------------
    # Write the result back (cast to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point called from the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a Triton‑accelerated
    core multiplication.  The result is a float32 tensor of shape
    [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask    = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear / gating work
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections and sigmoid gates
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()
    to_out_w      = weights["to_out.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # Projections (bias=False)
    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast over hidden dim)
    if not no_mask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3. Core TriMul operation – batched matmul over (i,k) × (j,k)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    total_slices = B * H       # flattened batch‑hidden dimension

    # Allocate output tensor (FP16) with the same layout as left/right
    out = torch.empty_like(left)

    # Strides for left/right/out (in elements, not bytes)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 (64‑64 tiles, 32‑wide reduction)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_slices,                           # axis 0 -> (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,           # axis 1 -> rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,           # axis 2 -> cols (j)
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,      # good occupancy for FP16 matmul on H100
    )

    # ------------------------------------------------------------------
    # 4. Post‑processing: LayerNorm over hidden dim, gating & final proj.
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)   # [B,N,N,hidden]

    # Layer‑norm over the hidden dimension
    out_f32 = torch.nn.functional.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply the output gate (broadcasted)
    out_f32 = out_f32 * out_gate.to(torch.float32)

    # Final linear projection back to `dim` (bias=False)
    out_f32 = torch.nn.functional.linear(out_f32.half(), to_out_w).to(torch.float32)

    return out_f32
