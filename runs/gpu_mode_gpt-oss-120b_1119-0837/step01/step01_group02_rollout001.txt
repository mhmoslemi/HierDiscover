"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – high‑performance Triton version.

Algorithm
---------
1. Layer‑norm the (B, N, N, C) input (C = dim) in fp16.
2. Linear projections to a hidden size H (no bias) → `left` and `right`.
   The three gate‑linear layers are also evaluated in fp16 and passed through
   `sigmoid`.  All three tensors are multiplied point‑wise into `left`,
   `right` and the final output gate.
3. Optional pairwise mask (broadcasted on the hidden axis) is applied in fp16.
4. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix multiplication `left @ rightᵀ` per hidden channel.
   This heavy O(N³·H) work is performed by a fused Triton kernel that
   consumes fp16 inputs and accumulates in fp32 (Tensor‑Core friendly).
   The kernel is auto‑tuned for several block‑size configurations.
5. The result (shape (B, N, N, H), fp32) is layer‑normed, multiplied by the
   output gate (cast to fp32) and finally linearly projected back to `dim`
   via a fp32 weight matrix.

All non‑kernel operations are carried out with PyTorch’s highly optimised
kernels; the only custom Triton kernel is the GEMM‑style TriMul contraction.
Running entirely in mixed‑precision reduces memory traffic and improves
throughput on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Triton GEMM kernel used for the TriMul contraction.
#   A : (B*H, M=N, K=N)  – fp16, row‑major
#   B : (B*H, K=N, N=N)  – fp16, row‑major (used as is, i.e. rightᵀ)
#   C : (B*H, M, N)      – fp32 accumulator, row‑major
# --------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 64,  "BLOCK_K": 32}, num_stages=2, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_stages=3, num_warps=8),
        triton.Config({"BLOCK_M": 256, "BLOCK_N": 128, "BLOCK_K": 64}, num_stages=4, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out = left @ rightᵀ (fp16 → fp32)."""
    pid_batch = tl.program_id(2)          # combined batch * hidden index
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile col

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Global pointers for the current tile
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles (FP16) – out‑of‑bounds elements are zeroed
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot product, accumulate into FP32
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Store result
    # ---------------------------------------------------------------
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------
# Entry point – called by the evaluation harness.
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data (tuple): (input, mask, weights, config)
            - input  : torch.Tensor of shape [B, N, N, C] (float32)
            - mask   : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict mapping weight names to torch.Tensor (float32)
            - config : dict containing at least `dim` (C) and `hidden_dim` (H),
                       optionally `nomask` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    dim = config["dim"]                 # == C, kept for readability
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------
    # Convert everything that participates in the heavy GEMM to fp16.
    # ----------------------------------------------------------------
    dtype_fp16 = torch.float16
    x = input_tensor.to(dtype_fp16)                     # (B,N,N,C)

    # Layer‑norm (fp16)
    norm_w = weights["norm.weight"].to(dtype_fp16)
    norm_b = weights["norm.bias"].to(dtype_fp16)
    x = F.layer_norm(x, (C,), weight=norm_w, bias=norm_b)

    # ---------------------------------------------------------------
    # Linear projections and gates – all in fp16
    # ---------------------------------------------------------------
    # Projectors
    left_proj_w  = weights["left_proj.weight"].to(dtype_fp16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(dtype_fp16) # (H, C)

    # Gates
    left_gate_w  = weights["left_gate.weight"].to(dtype_fp16)   # (H, C)
    right_gate_w = weights["right_gate.weight"].to(dtype_fp16)  # (H, C)
    out_gate_w   = weights["out_gate.weight"].to(dtype_fp16)    # (H, C)

    # Linear + gate computations (no bias)
    left  = F.linear(x, left_proj_w)          # (B,N,N,H) fp16
    right = F.linear(x, right_proj_w)

    # Optional mask (broadcast on hidden dim)
    if (not nomask) and (mask_tensor is not None):
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates (sigmoid) – still fp16
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # Prepare tensors for the Triton GEMM.
    # We want a batched GEMM over the hidden dimension:
    #   for each h:  left[...,h] @ right[...,h].T
    # ----------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch*H → (BH, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = B * H
    left_mat  = left.view(BH, N, N)   # (BH, M=N, K=N)
    right_mat = right.view(BH, N, N)  # (BH, K=N, N=N) – used directly as RHS

    # Output accumulator (fp32)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ---------------------------------------------------------------
    # Strides for the Triton kernel (row‑major layout)
    # ---------------------------------------------------------------
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k     = right_mat.stride(1)   # K dimension stride
    stride_b_n     = right_mat.stride(2)   # N dimension stride

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # ----------------------------------------------------------------
    # Dispatch the Triton kernel.
    # Block size parameters are chosen by @triton.autotune.
    # ----------------------------------------------------------------
    grid = (
        triton.cdiv(N, 128),   # program_id(0) – rows (M)
        triton.cdiv(N, 128),   # program_id(1) – cols (N)
        BH                     # program_id(2) – batch*hidden
    )

    # The kernel receives the tile sizes as compile‑time constants; they are
    # inferred from the autotune configuration.
    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        # The following kwargs are filled in by the autotuner at JIT time.
        # They are kept here for explicitness.
        # BLOCK_M = BLOCK_M, BLOCK_N = BLOCK_N, BLOCK_K = BLOCK_K,
    )

    # ----------------------------------------------------------------
    # Reshape kernel output back to (B, N, N, H) (still fp32)
    # ----------------------------------------------------------------
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------------
    # Output layer‑norm, output‑gate multiplication and final projection.
    # ----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(H,),
                              weight=to_out_norm_w,
                              bias=to_out_norm_b)

    # out_gate was computed in fp16 – cast to fp32 before the element‑wise mul
    out_hidden = out_hidden * out_gate.to(torch.float32)

    # Final linear projection back to dim (float32 weight)
    to_out_w = weights["to_out.weight"]
    out = F.linear(out_hidden, to_out_w)   # (B,N,N,C)

    return out
