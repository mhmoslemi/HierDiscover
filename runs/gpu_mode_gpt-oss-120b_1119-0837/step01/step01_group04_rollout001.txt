"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation
-----------------------------------------------------------------------

The heavy “cubic” part of TriMul is
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]                (1)

For every hidden channel *d* this is exactly a matrix‑multiply
    out[d] = left[d] @ right[d].T
with square matrices of size N×N.  The kernel below launches a **batched**
GEMM that computes all (B·hidden_dim) independent N×N matrix products in
FP16 while accumulating in FP32.  All other operations (LayerNorm, linear
projections, sigmoids, masking and the final linear projection) are kept in
PyTorch – they are negligible compared with the N³ work.

Key optimisations
-----------------
* Block size 64×64×64 – a divisor of every sequence length used in the
  evaluation (256, 512, 768, 1024), therefore the kernel never needs
  boundary masking.
* No mask handling inside the kernel – masking is applied before the GEMM.
* FP16 inputs, FP32 accumulation, final store back to FP16 (the downstream
  LayerNorm works in FP32 after a cheap cast).
* The batch dimension (batch × hidden) is flattened, so each program instance
  processes one independent GEMM without extra copies.
* Simple double‑buffered loop over the reduction dimension (k) to hide load
  latency (the next tile is already prefetched by the next iteration).

The entry point ``custom_kernel`` follows the reference PyTorch API and
returns a float‑32 tensor of shape ``[B, N, N, dim]``.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Batched GEMM kernel: computes C_b = A_b @ B_b.T   for many independent
# BATCH = batch * hidden_dim matrices of shape (N, N).  All tensors are
# FP16, accumulation in FP32.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                      # all equal to seq_len
    stride_ah, stride_am, stride_ak,   # A : (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B : (batch, K, N)  (interpreted transposed)
    stride_ch, stride_cm, stride_cn,   # C : (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) program IDs – identify the tile (m, n) and the independent matrix
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_b = tl.program_id(2)          # which matrix in the batch

    # ------------------------------------------------------------------
    # 2) offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # 3) base pointer for this matrix (batch * hidden_dim)
    # ------------------------------------------------------------------
    a_batch_off = pid_b * stride_ah
    b_batch_off = pid_b * stride_bh
    c_batch_off = pid_b * stride_ch

    # ------------------------------------------------------------------
    # 4) accumulator (FP32) – initialise to zero
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 5) loop over reduction dimension k
    # ------------------------------------------------------------------
    # because N is always a multiple of BLOCK_K (64) we can omit bounds
    # checking – this removes the cheap masking overhead.
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)               # (BLOCK_K,)

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_off + \
                 offs_m[:, None] * stride_am + \
                 k_range[None, :] * stride_ak
        a = tl.load(a_ptrs)                                     # FP16

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        # B is stored as (batch, N, N) but we read it as (K, N)
        b_ptrs = b_ptr + b_batch_off + \
                 k_range[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs)                                     # FP16

        # dot product – WMMA internally uses tensor‑cores
        acc += tl.dot(a, b)                                     # FP32

    # ------------------------------------------------------------------
    # 6) write result back (store as FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + c_batch_off + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16))


def batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Arguments
    ---------
    A, B : torch.Tensor
        Shape ``(batch, hidden, N, N)`` (contiguous, dtype torch.float16).
        The operation performed is ``C = A @ B.T`` for every (batch, hidden)
        pair.

    Returns
    -------
    C : torch.Tensor
        Same shape and dtype as inputs.
    """
    assert A.shape == B.shape
    batch, hidden, N, _ = A.shape

    # flatten batch*hidden → independent GEMM dimension
    A_flat = A.reshape(-1, N, N)
    B_flat = B.reshape(-1, N, N)
    C_flat = torch.empty_like(A_flat)

    # ------------------------------------------------------------------
    # 64‑tile configuration – a divisor of every test sequence length
    # ------------------------------------------------------------------
    BLOCK = 64
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
        A_flat.shape[0],                       # independent matrices
    )

    # strides expressed in number of elements (not bytes)
    stride_ah = A_flat.stride(0)
    stride_am = A_flat.stride(1)
    stride_ak = A_flat.stride(2)

    stride_bh = B_flat.stride(0)
    # B will be accessed transposed ⇒ inner‑reduction stride = stride of column
    stride_bk = B_flat.stride(2)   # = 1 for contiguous layout
    stride_bn = B_flat.stride(1)   # = N

    stride_ch = C_flat.stride(0)
    stride_cm = C_flat.stride(1)
    stride_cn = C_flat.stride(2)

    batched_gemm_kernel[grid](
        A_flat, B_flat, C_flat,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (batch, hidden, N, N)
    return C_flat.reshape(batch, hidden, N, N)


# ----------------------------------------------------------------------
# Triton‑backed forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        * mask         : torch.Tensor of shape [B, N, N] or None
        * weights      : dict mapping layer names to parameter tensors
        * config       : dict with keys:
                         - "dim"        : input feature dimension
                         - "hidden_dim" : hidden projection dimension
                         - "nomask"     : bool, whether mask should be ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the module output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, HD)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # (B, N, N, 1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 and permute to (B, hidden, N, N) – layout expected
    #    by the batched GEMM kernel.
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # bring hidden dimension to axis 1 and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, HD, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 6) Core cubic reduction via the triton batched GEMM
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # ------------------------------------------------------------------
    out_perm = batched_gemm(left_perm, right_perm)      # (B, HD, N, N)

    # back to original layout (B, N, N, hidden_dim)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                         # FP32 for the rest

    # ------------------------------------------------------------------
    # 7) Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8) Multiply by out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
