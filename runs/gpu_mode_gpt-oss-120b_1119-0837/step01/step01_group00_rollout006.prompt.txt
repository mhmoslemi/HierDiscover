import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor (shape B×N×N×C).
2. Project to a hidden dimension H (no bias) → tensors `left` and `right`.
3. Compute three element‑wise sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Apply the two gates (and an optional mask) to `left`/`right`.
5. Compute the core 4‑D “triangle” multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden channel.
   It is implemented in Triton as a GEMM where each hidden‑channel (and batch)
   is treated as an independent program‑ID.
6. Layer‑norm the result (`to_out_norm`), apply the `out_gate`,
   and finally a linear projection back to the original dimensionality C.

All heavy arithmetic (step 5) is executed in a custom Triton kernel; the rest
uses regular PyTorch ops.  Half‑precision (float16) is used for the
intermediate projection/gate tensors, while accumulation in the kernel is
float32, yielding a float32 final output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,    # pointer to output (float32)
    a_ptr,      # pointer to left tensor (float16)  shape (C, M, K)
    b_ptr,      # pointer to right tensor (float16) shape (C, N, K) – we load it transposed
    M, N, K,    # matrix dimensions (all equal to seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides (C,M,K)
    stride_b_k, stride_b_n, stride_b_c,        # right strides (C,K,N) after transpose
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block row   -> i
    pid_n = tl.program_id(1)   # block col   -> j
    pid_c = tl.program_id(2)   # hidden channel (folded batch*H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (seq_len)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k   # (M,K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n   # (K,N)

        # load tiles with masking
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # FP16
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # FP16

        # GEMM accumulate (FP16 * FP16 → FP32)
        acc += tl.dot(a, b)

    # write the result tile
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (float32 or bool) – may be ignored if nomask=True
            - weights: dict with model parameters (see code)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor[B, N, N, C] – the processed tensor (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Layer‑norm on the input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # B×N×N×C  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to half‑precision for projections/gates
    x_h = x.to(torch.float16)

    # 3) Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, left_proj_w)    # B×N×N×H
    right = F.linear(x_h, right_proj_w)  # B×N×N×H

    # 4) Gates (sigmoid)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # B×N×N×H
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)) # B×N×N×H
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # B×N×N×H

    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6) Optional mask (broadcasted over the hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)          # B×N×N×1
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the Triton kernel:
    #    reshape to (C_total, N, N) where C_total = B * hidden_dim
    B, N, _, _ = left.shape
    # move hidden dim before the two sequence dims for contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # B×H×N×N
    right = right.permute(0, 3, 1, 2).contiguous() # B×H×N×N

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)   # (C, M, K)
    right_flat = right.reshape(C_total, N, N) # (C, N, K) – we will load it transposed

    # output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) Compute the triangle multiplication with Triton
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides in element units (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # last dim (K)
    stride_b_n = right_flat.stride(1)   # middle dim (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10) Post‑processing: to_out_norm, out_gate, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)   # gating in FP32
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, C)

    return out
