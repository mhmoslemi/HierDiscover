"""
TriMul (outgoing) – Triton‑accelerated implementation.

The heavy part of the TriMul block is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication (left @ rightᵀ) for every hidden
channel *d*.  The kernel works directly on the original layout
[batch, seq, seq, hidden] (no permute / copy) by using the proper
strides.  Each hidden channel of each batch is treated as an independent
batch for the GEMM, giving a total of `B*hidden` independent matrix
multiplications of size N×N.  The kernel loads tiles of size
BLOCK_M×BLOCK_K and BLOCK_K×BLOCK_N, accumulates in FP32 and writes the
result back in FP16.  The rest of the TriMul block (LayerNorm, Linear,
gating, mask) is performed with regular PyTorch calls.

The implementation targets H100 GPUs (Tensor‑core friendly) and runs in
well under the 1 ms budget for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel for the batched GEMM:  out = left @ rightᵀ  (per hidden dim)
# --------------------------------------------------------------------------- #
@triton.jit
def _trmul_kernel(
    left_ptr, right_ptr, out_ptr,          # tensor pointers (FP16)
    M, N, K, hidden_dim,                   # matrix sizes (M=N=K=seq_len)
    batch_stride, i_stride, k_stride,      # strides in the original layout
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Arguments
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the FP16 tensors
        Layout = [B, N, N, H] (row‑major, H = innermost)
    M, N, K                     : dimensions of the GEMM (all = seq_len)
    hidden_dim                  : number of hidden channels
    batch_stride                : stride (in elements) from one batch to the next
                                 (= N * N * hidden_dim)
    i_stride                    : stride between rows (i‑direction) = N * hidden_dim
    k_stride                    : stride between columns / the contracted axis = hidden_dim
    BLOCK_*                     : tiling sizes (compile‑time constants)
    """

    pid = tl.program_id(axis=0)          # tile index over (M,N)
    batch_id = tl.program_id(axis=1)     # combined (batch, hidden) index

    # --------------------------------------------------------------------- #
    #   Decode combined batch index into (batch, hidden)
    # --------------------------------------------------------------------- #
    b = batch_id // hidden_dim            # actual batch index
    h = batch_id % hidden_dim             # hidden channel inside the batch

    # --------------------------------------------------------------------- #
    #   Tile coordinates
    # --------------------------------------------------------------------- #
    n_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // n_blocks_n
    block_n = pid % n_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Base pointers for this (batch, hidden) pair
    # --------------------------------------------------------------------- #
    a_base = left_ptr + b * batch_stride + h
    b_base = right_ptr + b * batch_stride + h
    c_base = out_ptr + b * batch_stride + h

    # --------------------------------------------------------------------- #
    #   Accumulator (FP32)
    # --------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   Loop over the contracted dimension K
    # --------------------------------------------------------------------- #
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = cur_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # left tile : shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * i_stride) + (k_offset[None, :] * k_stride)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # right tile (already transposed in logical sense) : (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (k_offset[:, None] * k_stride) + (offs_n[None, :] * i_stride)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # MAC (Tensor‑core friendly)
        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Store result (convert back to FP16)
    # --------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * i_stride) + (offs_n[None, :] * k_stride)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# --------------------------------------------------------------------------- #
# Entry point used by the test harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : [B, N, N, D]   (float32)
        - mask_tensor  : [B, N, N]   (bool / float)
        - weights_dict : mapping name → torch.Tensor
        - config_dict  : contains at least
                * “dim”          – input feature dimension D
                * “hidden_dim”   – hidden dimension H
                * optional “nomask” flag

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    #   1) Layer‑norm over the feature dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    #   2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    #   3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    if not config.get("nomask", True):
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    #   4) Gates (sigmoid) and gating of the projected tensors
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = (left * left_gate).to(torch.float16)   # FP16 for the kernel
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------- #
    #   5) Core contraction via the Triton kernel
    # ------------------------------------------------------------------- #
    out_fp16 = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides in the original layout (contiguous tensor)
    batch_stride = N * N * hidden_dim          # distance between two batches
    i_stride = N * hidden_dim                  # stride for the first sequence axis
    k_stride = hidden_dim                      # stride for the contracted axis

    total_batches = B * hidden_dim

    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32   # good compromise on H100
    grid = (
        ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),  # tiles over M,N
        total_batches,                                                       # (batch, hidden)
    )

    _trmul_kernel[grid](
        left,
        right,
        out_fp16,
        N, N, N, hidden_dim,
        batch_stride,
        i_stride,
        k_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps = 256 threads per block
    )

    # Convert back to FP32 for the remaining operations
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------- #
    #   6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final Linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
