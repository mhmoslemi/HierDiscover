"""
TriMul (outgoing) – fast Triton implementation.

The heavy part of the TriMul forward pass is the “triangle” multiplication
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
For each hidden channel `h` (and each batch element) this is exactly a
matrix‑multiply `left @ rightᵀ` with shape (N,N)·(N,N) → (N,N).  The kernel
processes all channels in parallel by folding the batch dimension into the
channel axis, i.e. we work on tensors of shape (C_total,N,N) where
`C_total = B * hidden_dim`.  The computation is performed in FP16 for the
inputs and accumulated in FP32, yielding a final FP32 output.

All other operations (layer‑norms, linear projections, sigmoid gates and the
final projection) are kept in regular PyTorch – they are already highly
optimized (cuBLAS/cuDNN).  Only the O(B·N²·hidden_dim) core is off‑loaded to
Triton.

The kernel uses large tiles (128×128) and 8 warps, which matches the H100
SM dimensions and allows the whole K‑tile (64) to stay in registers.
Memory traffic is reduced by:
    * half‑precision storage for the projected tensors,
    * optional masking applied *before* the kernel,
    * a single contiguous layout (C_total, N, N) for both left/right.
The final output tensor is reshaped back to (B,N,N,dim) and passed through
the post‑processing (norm → gate → linear) in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ  for every hidden channel (C_total)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,    # *float32  (C_total, M, N)
    left_ptr,   # *float16  (C_total, M, K)  – original left
    right_ptr,  # *float16  (C_total, N, K)  – original right (will be read transposed)
    M, N, K,    # int32   matrix sizes (all equal to seq_len)
    # strides (in elements, not bytes)
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – each (pid_c, pid_m, pid_n) processes a tile
    # --------------------------------------------------------------
    pid_c = tl.program_id(2)   # hidden‑channel (B*H) index
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # iterate over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M, K)
        left_tile_ptr = left_ptr + pid_c * stride_left_c \
                         + offs_m[:, None] * stride_left_m \
                         + offs_k[None, :] * stride_left_k
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)                     # FP16

        # right tile (transposed on‑the‑fly): (K, N)
        right_tile_ptr = right_ptr + pid_c * stride_right_c \
                          + offs_k[:, None] * stride_right_k \
                          + offs_n[None, :] * stride_right_n
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)                    # FP16

        # GEMM accumulate
        acc += tl.dot(left, right)   # (M,N) ← (M,K)·(K,N)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                    + offs_m[:, None] * stride_out_m \
                    + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (bool or float) – may be None / ignored
            - weights: dict of model parameters (see reference)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to half, compute linear projections and gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # linear projections (no bias)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # sigmoid gates (half‑precision)
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))   # (B,N,N,H)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # apply optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Prepare tensors for the Triton kernel:
    #    we need shape (C_total, N, N) with C_total = B * hidden_dim.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dim before the two sequence dims -> contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)    # (C_total, M, K)
    right_flat = right.reshape(C_total, N, N)  # (C_total, N, K)  (we will load transposed)

    # Output buffer for the core multiplication (float32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 4) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # fits nicely into registers and reduces K‑loop overhead

    # Strides in element units (not bytes)
    stride_out_c, stride_out_m, stride_out_n = (
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2)
    )
    stride_left_c, stride_left_m, stride_left_k = (
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2)
    )
    stride_right_c, stride_right_k, stride_right_n = (
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1)
    )

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (folded batch*hidden)
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_left_c, stride_left_m, stride_left_k,
        stride_right_c, stride_right_k, stride_right_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim layer‑norm → out_gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B,N,N,H) float32

    # out_gate is half‑precision → convert to float32 before multiplication
    out = out * out_gate.to(torch.float32)

    # final projection back to original dimensionality (float32)
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
