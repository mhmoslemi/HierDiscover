"""
TriMul “outgoing” operator – fast half‑precision implementation.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor over the feature dimension.
   The reduction (mean/variance) is performed in a custom Triton kernel,
   while the normalisation itself runs in FP16 (statistics are accumulated
   in FP32 for numerical stability).

2. Compute the four linear projections (left/right and their gates) in
   FP16 with torch.nn.functional.linear.  The optional pairwise mask is
   applied immediately after the projection (as in the reference code).

3. Fuse the two projected tensors with their sigmoid gates:
      left  = (x @ W_left)       * sigmoid(x @ W_left_gate)   * mask
      right = (x @ W_right)      * sigmoid(x @ W_right_gate)  * mask

4. Perform the N³ contraction
      out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   via a single batched matrix‑multiply (torch.bmm) on FP16 tensors.
   This maps to a strided‑batched GEMM that fully exploits Tensor‑cores
   on an H100.

5. Layer‑norm the hidden‑dim result (again with the Triton kernel).

6. Apply the output sigmoid‑gate and the final linear projection back to
   the model dimension.  All remaining operations stay in FP16; the
   final tensor is cast back to FP32 as required by the API.

Performance notes
-----------------
* All heavy tensor‑wise work (layer‑norm, projections, gates, N³ GEMM)
  runs in FP16, cutting memory traffic by 2× and allowing Tensor‑core
  acceleration.
* The Triton LayerNorm kernel performs the two‑pass algorithm in a
  single launch and uses FP32 accumulators to preserve numerical quality.
* By keeping the intermediate tensors in half‑precision we minimise
  global‑memory reads/writes while still matching the reference output
  (within FP16 tolerance).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton two‑pass LayerNorm kernel (FP16 input, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]   (FP16)
    y_ptr,                # *ptr to output [M, N]   (FP16)
    weight_ptr,           # *ptr to weight [N]      (FP16)
    bias_ptr,             # *ptr to bias   [N]      (FP16)
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon for variance
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    NUM_K = tl.cdiv(N, BLOCK_N)
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load a tile (BLOCK_M × BLOCK_N) in FP16
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        x_f32 = tl.cast(x, tl.float32)                # promote for reduction
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values (FP16)
    # ------------------------------------------------------------------
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        x_f32 = tl.cast(x, tl.float32)
        norm_f32 = (x_f32 - mean[:, None]) * rstd[:, None]   # FP32 normalisation
        norm = tl.cast(norm_f32, tl.float16)                 # back to FP16

        # Load affine parameters (FP16)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=tl.float16(0.0))
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=tl.float16(0.0))

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor (M, N).
    Input/weight/bias are expected to be half‑precision (float16);
    the returned tensor has the same dtype.
    """
    assert x.dim() == 2, "triton_layernorm only supports 2‑D inputs"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure all tensors are contiguous (required for correct stride handling)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Grid: one program per BLOCK_M rows
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for H100
        BLOCK_N=64,    # tuned for register pressure
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (half‑precision core).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Cast all tensors to FP16 (the heavy computation runs in half‑precision)
    # ------------------------------------------------------------------
    # Input
    x = input_tensor.to(torch.float16)

    # Weights – everything is stored without bias, keep the original shape
    w = {}
    for name, tensor in weights.items():
        w[name] = tensor.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton – FP16)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = x.reshape(M, dim)                       # (B·N·N, dim)
    x_norm_flat = triton_layernorm(
        x_flat, w["norm.weight"], w["norm.bias"], eps=eps
    )
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16) – no bias
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_norm, w["left_proj.weight"])   # (B,N,N,hidden)
    right_proj = F.linear(x_norm, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise fusion
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))   # (B,N,N,hidden)

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication over the summed dimension (k)
    #     Here we treat each (batch, hidden) pair as an independent GEMM:
    #       out[b,h] = left[b,h] @ right[b,h].T
    #     The tensors are reshaped to (B*hidden, N, N) and a single
    #     strided‑batched GEMM (torch.bmm) runs on Tensor‑cores.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) and flatten the first two dims.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)    # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)   # (B*H, N, N)

    # GEMM: (B*H, N, N) × (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, hidden)
    out = (
        out_mat.view(bs, hidden_dim, seq_len, seq_len)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton – FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)
    out_norm_flat = triton_layernorm(
        out_flat, w["to_out_norm.weight"], w["to_out_norm.bias"], eps=eps
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to model dimension
    # ------------------------------------------------------------------
    out = out_norm * out_gate                       # (B,N,N,hidden) – FP16
    out = F.linear(out, w["to_out.weight"])         # (B,N,N,dim) – FP16

    # Cast back to float32 as required by the API
    return out.float()
