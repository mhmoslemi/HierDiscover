"""
TriMul (outgoing) – fast Triton implementation
================================================

The kernel computes the core contraction of the TriMul operator:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the linearly projected and gated pairwise
representations of the input tensor `x` (shape B×N×N×C).  The heavy
O(N³·H) work is performed by a fused batched GEMM kernel that
operates on half‑precision data (FP16) and accumulates in FP32.
All surrounding operations (layer‑norm, linear projections, sigmoid
gates, output normalisation and final projection) are done with
PyTorch primitives, which already provide highly‑optimised kernels.

Key ideas
---------
*   Fuse mask and gate multiplications into the projected tensors before
    the GEMM to reduce memory traffic.
*   Store `left` and `right` in FP16 – the kernel reads only 2 bytes per
    element, while accumulation stays in FP32 for numeric stability.
*   Collapse the batch and hidden dimensions (B·H) and run a single
    Triton kernel that performs a batched matrix‑multiply
    `M = N, K = N, N = N` for each hidden slice.
*   Tile the GEMM with configurable block sizes that are good for N
    up to 1024 on an H100.  The kernel uses `tl.dot` which maps to
    Tensor‑cores, giving near‑peak TFLOP performance.
*   No gradients are required – everything runs in inference mode.

The entry point is `custom_kernel(data)`, where `data` is a tuple
`(input, mask, weights, config)` as described in the problem statement.
The function returns a tensor of shape `[B, N, N, C]` (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (FP16 inputs → FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ Bᵀ for many independent (B*H) matrices.

    A : (BH, M, K)  – row‑major, FP16
    B : (BH, K, N)  – interpreted via swapped strides so that we read Bᵀ
    C : (BH, M, N)  – row‑major, FP32 accumulator
    """
    pid_batch = tl.program_id(2)          # flattened batch*hidden index
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile column

    # ------------------- tile offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- accumulator (FP32) ------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- K‑loop -----------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile of A and B
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles (FP16) – out‑of‑bounds elements are zero‑filled
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ------------------- store result ------------------
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: choose tile sizes based on N (empirically good for H100)
# ----------------------------------------------------------------------
def _choose_tile_sizes(N: int):
    """Return (BLOCK_M, BLOCK_N, BLOCK_K) tuned for the given N."""
    if N <= 256:
        return 64, 64, 16
    elif N <= 512:
        return 128, 128, 32
    else:       # up to 1024 (and a little beyond)
        return 128, 128, 32


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – optional
            - weights_dict : mapping weight name → torch.Tensor
            - config_dict  : contains 'dim' (C) and 'hidden_dim' (H)
                plus optional 'nomask' flag.

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2️⃣ load model weights (assumed already on the correct device)
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ Layer‑norm over the channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (C → H) – bias‑free
    # ------------------------------------------------------------------
    # These produce FP32 tensors; we will cast to FP16 after gating.
    left_proj  = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_weight)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional pairwise mask
    # ------------------------------------------------------------------
    apply_mask = not config.get("nomask", True)
    if apply_mask and mask_tensor is not None:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj  * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid of a second linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left  = (left_proj * left_gate).to(torch.float16)   # FP16 ready for GEMM
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ Core contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch*H → BH
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_mat = right_perm.view(BH, N, N)  # (BH, K=N, N) – we will treat it transposed

    # Prepare output buffer (FP32 accumulation)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    # We read rightᵀ, so K‑stride comes from the original column‑major stride
    stride_b_k     = right_mat.stride(2)   # original column stride (1)
    stride_b_n     = right_mat.stride(1)   # original row stride (N)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Choose tile sizes based on N
    BLOCK_M, BLOCK_N, BLOCK_K = _choose_tile_sizes(N)

    # Launch a 3‑D grid: rows × cols × batch*hidden
    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id(0) – M‑tiles
        triton.cdiv(N, BLOCK_N),   # program_id(1) – N‑tiles
        BH                         # program_id(2) – flattened batch*hidden
    )

    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H) – still FP32
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Output layer‑norm + final gate
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(H,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias)
    out_hidden = out_hidden * out_gate   # element‑wise gating (FP32)

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to original dimension C
    # ------------------------------------------------------------------
    out = F.linear(out_hidden, to_out_weight)   # (B, N, N, C), bias‑free

    return out
```
