import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul forward (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` of shape [B, N, N, C].
2. Linear projections (bias‑free) to a hidden dimension H:
      left  = x @ left_proj_weight.T
      right = x @ right_proj_weight.T
   (shape after projection: [B, N, N, H])
3. Optional mask (broadcasted on the hidden dimension) is applied to
   `left` and `right`.
4. Gating values are computed from the normalized input and multiplied
   point‑wise:
      left  = left  * sigmoid(left_gate(x))
      right = right * sigmoid(right_gate(x))
5. The core O(N³·H) operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (b, h) pair.
   We fuse the ~B·H independent GEMMs with a custom Triton kernel that
   computes C = A @ Bᵀ for each batch entry.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and finally a linear projection back to the original
   channel size C.

Only the heavy batched GEMM (step 5) is written in Triton; all other
operations use highly‑optimised PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (float32 accumulation)
# A : [BHD, M, K]   (row‑major)
# B : [BHD, N, K]   (row‑major, interpreted as K×N after transposition)
# C : [BHD, M, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (all integer)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (treated as transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index (combined batch*head)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # compute absolute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for the selected batch element
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # pointers for the current K‑tile (A is M×K, B is K×N after transposition)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True => mask can be ignored

    # ------------------------------------------------------------------
    # fetch model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden dimension (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³·H operation – batched GEMM using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # fold the hidden dimension into the batch dimension:
    #   (B, N, N, H) -> (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_t.view(B * H, N, N)      # A : [B*H, M=N, K=N]
    right_mat = right_t.view(B * H, N, N)    # B : [B*H, N, K=N]  (will be used transposed)

    # output placeholder
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # blocks over rows  (M)
        (N + BLOCK_N - 1) // BLOCK_N,      # blocks over cols  (N)
        B * H                               # batch × hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_mat.stride(0),
        stride_a_m=left_mat.stride(1),
        stride_a_k=left_mat.stride(2),
        stride_b_batch=right_mat.stride(0),
        # for Bᵀ we swap the last two strides:
        stride_b_k=right_mat.stride(2),   # stride of the original K‑dimension
        stride_b_n=right_mat.stride(1),   # stride of the original N‑dimension
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                      # reasonable default for H100
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection → [B, N, N, dim]

    return out
