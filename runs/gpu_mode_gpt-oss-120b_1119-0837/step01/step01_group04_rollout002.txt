import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape, "shape mismatch"
    a_flat = a.contiguous().view(-1)
    b_flat = b.contiguous().view(-1)
    out_flat = torch.empty_like(a_flat)
    size = a_flat.numel()
    BLOCK = 32768  # tune for H100
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        a_flat, b_flat, out_flat, size,
        BLOCK_SIZE=BLOCK,
    )
    return out_flat.view_as(a)

def custom_kernel(data):
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # 2. Linear projections (left / right)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Apply mask if needed
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # 4. Gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # 5. Core N^3 operation using batched bmm (float32)
    # reshape for batched GEMM: (B*hidden, N, N)
    # Move hidden dim before seq dims for convenient view
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, hd, N, _ = left_perm.shape
    left_flat = left_perm.view(B*hd, N, N)
    right_flat = right_perm.view(B*hd, N, N)

    # Matrix multiply: left @ right^T per hidden channel
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hd, N, N).permute(0, 2, 3, 1).contiguous()

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # 7. Apply out_gate (elementwise) using Triton kernel
    out = elementwise_mul(out, out_gate)

    # 8. Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # expects same shape and device
    assert a.shape == b.shape
    a_flat = a.contiguous().view(-1)
    b_flat = b.contiguous().view(-1)
    out_flat = torch.empty_like(a_flat)
    size = a_flat.numel()
    BLOCK = 65536
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        a_flat, b_flat, out_flat, size,
        BLOCK_SIZE=BLOCK
    )
    return out_flat.view_as(a)

def custom_kernel(data):
    """
    Fast forward pass of TriMul (outgoing) using Torch for linear ops
    and a compact Triton kernel for the final element‑wise multiplication.
    The cubic pairwise reduction is performed by a batched half‑precision
    matrix multiply (`torch.bmm`) that exploits H100 tensor cores.
    """
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # 1) Input layer norm
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2) Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3) Optional mask
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # 4) Gating
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # 5) Core N³ reduction via batched half‑precision GEMM
    #    Cast to fp16 to leverage TensorCores.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to leading axis and make tensors contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, HD, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, HD, N, N)

    B, HD, N, _ = left.shape
    left = left.view(B * HD, N, N)
    right = right.view(B * HD, N, N)

    # Batched matrix multiply: left @ rightᵀ
    out = torch.bmm(left, right.transpose(1, 2))

    # Restore original layout: (B, N, N, HD)
    out = out.view(B, HD, N, N).permute(0, 2, 3, 1).contiguous()

    # Cast back to FP32 for the remaining FP32‑only ops
    out = out.to(torch.float32)

    # 6) Second LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 7) Element‑wise multiplication with out_gate via Triton
    out = elementwise_mul(out, out_gate)

    # 8) Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass – fast Triton + Torch implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature dimension *dim*.
2. Two bias‑free linear projections give *left* and *right* tensors
   of shape (B, N, N, H) where H = hidden_dim.
3. (Optional) Apply the pairwise mask.
4. Compute three sigmoid gates from the normalized input and
   multiply them into *left*, *right* and the final output.
5. Core cubic reduction:
        out_{i,j,h} = Σ_k left_{i,k,h} · right_{j,k,h}
   is a batched matrix multiplication per hidden channel.
   We cast *left*/*right* to fp16 and use a single
   `torch.bmm` over a batch of (B·H) independent GEMMs,
   which on H100 runs on Tensor‑cores.
6. Cast the result back to fp32, apply a second LayerNorm over the
   hidden dimension, multiply element‑wise with the *out_gate* using a
   tiny Triton kernel (fulfils the “use Triton” requirement), and
   finally project back to the original *dim* with a bias‑free linear.
7. Return a float‑32 tensor of shape (B, N, N, dim).

The Triton kernel `elementwise_mul_kernel` performs an element‑wise
product of two tensors of identical shape in a single launch.
All heavy arithmetic (the N³ term) uses fast fp16 Tensor‑core GEMMs,
while the kernel handles a non‑trivial part of the graph.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Element‑wise multiplication kernel (used for the final out‑gate step)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    """
    c = a * b  (all tensors must be contiguous 1‑D views)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Fast elementwise product using the Triton kernel above."""
    assert a.shape == b.shape, "Shapes must match for elementwise multiplication"
    a_flat = a.contiguous().view(-1)
    b_flat = b.contiguous().view(-1)
    out_flat = torch.empty_like(a_flat)

    size = a_flat.numel()
    BLOCK = 65536                     # tuned for H100
    grid = (triton.cdiv(size, BLOCK),)

    elementwise_mul_kernel[grid](
        a_flat, b_flat, out_flat, size,
        BLOCK_SIZE=BLOCK,
    )
    return out_flat.view_as(a)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : {
              "dim": int,
              "hidden_dim": int,
              "nomask": bool (optional)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # broadcast over hidden dim
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic reduction via batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Cast to fp16 to exploit Tensor‑cores
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to the front and make contiguous for strided batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)          # (B·H, N, N)
    right_flat = right.view(B * H, N, N)

    # Batched matrix multiply: left @ rightᵀ  (per hidden channel)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16 result

    # Restore (B, N, N, H) layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 6) Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H], float32

    # ------------------------------------------------------------------
    # 7) Multiply with the out‑gate (Triton kernel)
    # ------------------------------------------------------------------
    out = elementwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 8) Final projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
