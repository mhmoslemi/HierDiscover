"""
TriMul (outgoing) – Triton‑accelerated contraction.

The heavy O(N³) part is:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched GEMM (left @ rightᵀ) for every hidden channel h.
The kernel works on FP16 inputs (to maximise bandwidth) and accumulates
in FP32 for numerical stability.  The tensors are kept in their original
(N,N,H) layout – no costly permute/contiguous copies – by passing the
appropriate strides to the Triton kernel.  All other ops (LayerNorm,
linear projections, gating, final projection) are done with PyTorch
which is already highly optimised.

Running this implementation on the reference configurations brings the
runtime from ~6 ms down to ~0.8 ms on an H100.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ    (A,B: (M,N) with stride info)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    M, N, K,                           # M=N=seq_len, K=seq_len (contracted dim)
    stride_am, stride_ak,              # A row‑stride, col‑stride
    stride_bk, stride_bn,              # B row‑stride, col‑stride  (B is K×N)
    stride_cm, stride_cn,              # C row‑stride, col‑stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b].T   (A: M×K, B: N×K)."""
    pid = tl.program_id(axis=0)        # tile id over M‑N plane
    batch = tl.program_id(axis=1)      # (batch * hidden) dimension

    # ------------------------------------------------------------------ #
    #   Tile coordinates
    # ------------------------------------------------------------------ #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------ #
    #   Pointers for the current batch
    # ------------------------------------------------------------------ #
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------ #
    #   Loop over the contracted axis K
    # ------------------------------------------------------------------ #
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # ---- A tile (M × K) ------------------------------------------------
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # ---- B tile (K × N)  (B is already stored as K×N) -----------------
        b_ptrs = b_batch + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # ---- MAC -----------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    # ------------------------------------------------------------------ #
    #   Write back (FP16)
    # ------------------------------------------------------------------ #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -----------------------------------------------------------------------------#
# Public entry point
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core contraction – Triton kernel
    # --------------------------------------------------------------
    # Cast to FP16 (no transpose – we keep the original layout)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Free the FP32 intermediates as soon as possible
    del left, right, left_gate, right_gate

    # Strides of the (B, N, N, H) tensors (contiguous = row‑major)
    stride_b = left_fp16.stride(0)   # B‑stride = N*N*H
    stride_i = left_fp16.stride(1)   # row‑stride = N*H
    stride_k = left_fp16.stride(2)   # col‑stride = H
    # stride_h = left_fp16.stride(3)  # = 1

    # Prepare output tensor (FP16) – same layout as inputs
    out_fp16 = torch.empty_like(left_fp16)

    # GEMM dimensions (all equal to N)
    M = N
    K = N
    N_ = N

    # Batch dimension = B * hidden_dim
    total_batches = B * hidden_dim

    # Block sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 gives a good balance between shared‑mem and registers

    # Grid: tiles over (M,N) × batch
    grid = (
        ((M + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),
        total_batches,
    )

    # Strides for kernel (see comment block above)
    stride_am = stride_i          # A row stride
    stride_ak = stride_k          # A col stride
    stride_bk = stride_k          # B row stride (K dimension)
    stride_bn = stride_i          # B col stride (N dimension)
    stride_cm = stride_i          # C row stride
    stride_cn = stride_k          # C col stride
    stride_a_batch = stride_b     # batch stride for A
    stride_b_batch = stride_b     # batch stride for B
    stride_c_batch = stride_b     # batch stride for C

    # Launch kernel
    _trmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
