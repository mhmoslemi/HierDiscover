import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel: Triangle Multiplicative Update (outgoing) with Triton fused GEMM.

The forward pass follows the AlphaFold3 TriMul implementation:

1. Layer‑Norm over the feature dimension.
2. Two linear projections (`left_proj`, `right_proj`) producing tensors of shape
   [B, N, N, H] where H = hidden_dim.
3. Optional pair‑wise mask multiplication.
4. Gating: element‑wise sigmoids from three extra linear layers
   (`left_gate`, `right_gate`, `out_gate`).
5. Core “outgoing” multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
6. Layer‑Norm + out‑gate on the result, followed by a final linear
   projection back to `dim`.

Only the O(N³·H) GEMM part is written in Triton; all other ops use
high‑performance PyTorch kernels.  The kernel works for any batch size
(≤2), sequence length (≤1024) and hidden dimension (<=128) on an H100.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_kernel(
    a_ptr,            # left matrix  : [B*H, M=N, K=N]  (fp16)
    b_ptr,            # right matrix : [B*H, K=N, N]    (fp16) – already transposed
    c_ptr,            # output       : [B*H, M=N, N]  (fp16)
    M, N, K,          # matrix sizes (all = seq_len)
    stride_a_batch,   # stride between consecutive batches in A
    stride_a_i,       # stride for the i‑dimension (M) in A
    stride_a_k,       # stride for the k‑dimension (K) in A
    stride_b_batch,   # stride between consecutive batches in B
    stride_b_k,       # stride for the k‑dimension (K) in B
    stride_b_n,       # stride for the n‑dimension (N) in B
    stride_c_batch,   # stride between consecutive batches in C
    stride_c_i,       # stride for the i‑dimension (M) in C
    stride_c_n,       # stride for the n‑dimension (N) in C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (A:[b,M,K], B:[b,N,K] → Bᵀ:[b,K,N])"""
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_b = tl.program_id(2)          # batch*hidden index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tiles of A and B
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Accumulate products
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write the result back to C (convert to FP16)
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Perform the TriMul outgoing update.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [bs, N, N, dim]
            - mask : torch.Tensor of shape [bs, N, N]  (may be None)
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, config = data
    bs, N, _, dim = inp.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # Load weight tensors
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # shape (dim, hidden)

    # ------------------------------------------------------------------ #
    # 1) Input layer‑norm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------ #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 3) Optional mask (broadcast on feature dim)
    # ------------------------------------------------------------------ #
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)    # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4) Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5) Core TriMul:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (A @ Bᵀ) with Triton.
    # ------------------------------------------------------------------ #
    # Cast to half‑precision for the kernel (speed on H100 tensor cores)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, H, N, N] and make them contiguous.
    # For the GEMM we need:
    #   A : (B*H, M=N, K=N)
    #   Bᵀ: (B*H, K=N, N)  (i.e. right with its last two axes swapped)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k‑j order)

    # Flatten the leading (B, H) dimensions into a single batch dimension.
    batch_h = bs * hidden
    left_mat = left_perm.view(batch_h, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(batch_h, N, N)  # [B*H, N, N]
    out_mat = torch.empty_like(left_mat)        # FP16 buffer for result

    # Strides (in elements) for the three tensors
    stride_a_batch = left_mat.stride(0)
    stride_a_i = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)   # dimension K
    stride_b_n = right_mat.stride(2)   # dimension N (after transpose)

    stride_c_batch = out_mat.stride(0)
    stride_c_i = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_h,
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_i=stride_c_i,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # reasonable default for H100
    )

    # Convert the result back to float32 and restore original layout
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    # shape now: [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 6) Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # -> [B, N, N, dim]

    return out
