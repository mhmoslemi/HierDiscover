"""
TriMul (outgoing) forward pass – highly‑optimized version.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x` of shape [B, N, N, C].
2. Linear projections (bias‑free) from `C` → `H`:
       left  = x @ left_proj_weight.T
       right = x @ right_proj_weight.T                # -> [B, N, N, H]
3. Optional mask (broadcast on the hidden dimension) is applied.
4. Gating values are computed from the normalized input:
       left_gate  = sigmoid(x @ left_gate_weight.T)   # -> [B,N,N,H]
       right_gate = sigmoid(x @ right_gate_weight.T)
       out_gate   = sigmoid(x @ out_gate_weight.T)   # -> [B,N,N,H]
   The projected tensors are multiplied point‑wise by their gates.
5. Cast the gated tensors to float16 and fuse the massive
   O(N³·H) operation using a custom Triton GEMM kernel that computes,
   for every (b,h) pair,   C = A @ Bᵀ   (A,B ∈ ℝ^{N×N}).
   The kernel works on half‑precision inputs with float32 accumulation,
   fully exploiting H100 tensor‑cores.
6. Reshape the result back to [B, N, N, H] and apply a second
   LayerNorm, the output gate and a final linear projection back to
   the original channel dimension `C`.
The heavy computation (step 5) is performed in Triton; all other
operations use highly‑optimised PyTorch kernels.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# accumulator float32)

    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(2)  # combined batch * head dimension
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block column

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch element
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K

        # pointers for the current K‑tile
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_a_m \
                 + (cur_k + offs_k)[None, :] * stride_a_k
        b_ptrs = b_base \
                 + (cur_k + offs_k)[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # out‑of‑bounds handling
        mask_a = (offs_m[:, None] < M) & ((cur_k + offs_k)[None, :] < K)
        mask_b = ((cur_k + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # half
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # half

        # tl.dot on half produces half; accumulation in float32 via +=
        acc += tl.dot(a, b)      # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # store the tile
    # ------------------------------------------------------------------
    c_ptrs = c_base \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

        - input_tensor : torch.Tensor, shape [bs, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [bs, N, N]   (or None)
        - weight_dict  : dict of torch.Tensors for all model weights
        - config_dict  : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in FP32 for numerical stability
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to half‑precision and run the batched GEMM (core O(N³·H))
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # B, seq_len, _, hidden_dim

    # reshape to (B*H, N, N) → each [row, col] matrix will be multiplied
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output placeholder (half)
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100, FP16)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 32   # reduction tile – aligns with tensor‑core granularity

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H                           # batch × hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_mat.stride(0),
        stride_a_m=left_mat.stride(1),
        stride_a_k=left_mat.stride(2),
        stride_b_batch=right_mat.stride(0),
        # treat B as transposed => swap last two strides
        stride_b_k=right_mat.stride(2),   # original K‑stride
        stride_b_n=right_mat.stride(1),   # original N‑stride
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # enough warps to keep the H100 busy
    )

    # back to FP32 and original layout
    out = out_mat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # apply the output gate (FP32)
    out = F.linear(out, to_out_weight)   # → [B, N, N, dim]

    return out
```

This implementation keeps the heavy O(N³·H) multiplication in a custom Triton kernel that works on half‑precision data (leveraging H100 tensor‑cores), while the surrounding linear layers, normalizations, and gating are performed with PyTorch’s highly‑optimised kernels. The kernel is launched with 64 × 64 tiles and a 32‑wide reduction, yielding a far‑lower runtime (target ≈ 1 ms for the largest test case).
