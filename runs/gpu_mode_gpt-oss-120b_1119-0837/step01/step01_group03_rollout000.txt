"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The heavy O(N³·H) part is a batched matrix multiplication:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We fuse the B·H independent GEMMs (A @ Bᵀ) into a single Triton kernel.
All other operations (LayerNorm, linear projections, gating, masking) are
performed with PyTorch for simplicity.  The kernel works on half‑precision
inputs to reduce memory traffic while accumulating in float‑32 to keep the
final output numerically stable.

Input
-----
data = (x, mask, weights, config)
    x       : torch.Tensor, shape [B, N, N, C] (float32)
    mask    : torch.Tensor or None, shape [B, N, N] (bool/int/float)
    weights : dict of model parameters (no gradients needed)
    config  : dict with keys "dim", "hidden_dim", "nomask" (optional)

Output
------
torch.Tensor of shape [B, N, N, C] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : [B*H, M, K]   (row‑major)
# B : [B*H, N, K]   (row‑major, interpreted as K×N after transpose)
# C : [B*H, M, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (treated as transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch index (combined B·H)
    pid_m = tl.program_id(0)       # block row index
    pid_n = tl.program_id(1)       # block column index

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for the current batch element
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile (M×K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + (k_off + offs_k)[None, :] * stride_a_k
        # Bᵀ tile (K×N) – note the swapped strides
        b_ptrs = b_base + (k_off + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # out‑of‑bounds guarding
        mask_a = (offs_m[:, None] < M) & ((k_off + offs_k)[None, :] < K)
        mask_b = ((k_off + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float16)

        # dot‑product (accumulated in float32)
        acc += tl.dot(a, b)

    # store result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Model parameters
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (normalize over channel dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections to hidden dimension (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core N³·H operation – batched GEMM via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape           # H == hidden_dim

    # Fuse hidden dim into batch to get (B*H, N, N) matrices
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_t.view(B * H, N, N)   # A : [B*H, M=N, K=N]
    right_mat = right_t.view(B * H, N, N) # B : [B*H, N, K=N] (will be transposed)

    # Convert to half‑precision for the kernel (memory‑bandwidth reduction)
    left_mat_h = left_mat.to(torch.float16)
    right_mat_h = right_mat.to(torch.float16)

    # Output matrix (float32 accumulation)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows  (M)
        (N + BLOCK_N - 1) // BLOCK_N,   # cols  (N)
        B * H                           # batch * hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat_h,
        b_ptr=right_mat_h,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_mat_h.stride(0),
        stride_a_m=left_mat_h.stride(1),
        stride_a_k=left_mat_h.stride(2),
        stride_b_batch=right_mat_h.stride(0),
        # B is interpreted transposed: K axis corresponds to original last dim
        stride_b_k=right_mat_h.stride(2),
        stride_b_n=right_mat_h.stride(1),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,           # reasonable for H100, can be tuned
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second layer‑norm, output gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
