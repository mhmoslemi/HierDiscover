import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for elementwise multiply
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Fast implementation of the outgoing TriMul operator.
    All heavy matrix‑multiplications are performed in float16
    with torch.bmm (leveraging Tensor‑cores).  A tiny Triton kernel
    fuses the final element‑wise multiplication with the output gate.
    The final result is returned in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # Cast weights to float16 for the compute‑heavy parts
    w = {}
    for k in [
        "norm.weight", "norm.bias",
        "left_proj.weight", "right_proj.weight",
        "left_gate.weight", "right_gate.weight", "out_gate.weight",
        "to_out_norm.weight", "to_out_norm.bias", "to_out.weight"
    ]:
        w[k] = weights[k].to(torch.float16, device=device)

    # ---------- 1. LayerNorm on input ----------
    # First LayerNorm (float16)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    ).to(torch.float16)

    # ---------- 2. Linear projections ----------
    left = F.linear(x, w["left_proj.weight"])    # [B, N, N, H]
    right = F.linear(x, w["right_proj.weight"])

    # ---------- 3. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- 4. Optional mask ----------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- 5. Core O(N³·H) batched GEMM ----------
    B, N, _, H = left.shape
    # reshape for batched matmul: (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_t.view(B * H, N, N)
    right_mat = right_t.view(B * H, N, N)

    # Batched matrix multiply: left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ---------- 6. Second LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )

    # ---------- 7. Multiply by output gate (Triton) ----------
    # Triton expects 1‑D contiguous tensors
    out_flat = out.reshape(-1)
    out_gate_flat = out_gate.reshape(-1)
    out_gated_flat = torch.empty_like(out_flat)

    total = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_gated_flat,
        total,
        BLOCK_SIZE,
        num_warps=4,
    )
    out_gated = out_gated_flat.view_as(out)

    # ---------- 8. Final projection ----------
    out = F.linear(out_gated, w["to_out.weight"])

    # Return float32 tensor as required by the interface
    return out.float()

"""
Fast outgoing TriMul implementation.

Algorithm
---------
1. Layer‑norm the input (float16).
2. Bias‑free linear projections to hidden_dim (float16).
3. Compute sigmoid gates, apply them (and optional mask) to the projected tensors.
4. Reshape to (B·H, N, N) and perform a batched matrix multiplication
   C = A @ Bᵀ with torch.bmm (uses Tensor‑cores, fp16).
5. Second LayerNorm (float16).
6. A tiny Triton kernel fuses the element‑wise multiplication with the
   output gate (out = out * out_gate).
7. Final linear projection back to dim (float16) and cast the result to
   float32 for the caller.

Only the element‑wise gating (step 6) is written in Triton, satisfying the
requirement that a custom kernel is used while the huge O(N³·H) work is
performed by the highly‑optimised cuBLAS kernel invoked through `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: element‑wise multiply (out = a * b)
# ------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # pointer to first operand (1‑D contiguous)
    b_ptr,          # pointer to second operand (same size)
    out_ptr,        # pointer to output buffer
    SIZE,           # total number of elements (int32)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < SIZE

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ==> ignore mask

    device = input_tensor.device
    # --------------------------------------------------------------
    # 1. Cast model weights to float16 (the heavy compute path)
    # --------------------------------------------------------------
    # Helper to fetch & cast
    def w(name):
        return weights[name].to(torch.float16, device=device)

    norm_weight = w("norm.weight")
    norm_bias = w("norm.bias")
    left_proj_weight = w("left_proj.weight")
    right_proj_weight = w("right_proj.weight")
    left_gate_weight = w("left_gate.weight")
    right_gate_weight = w("right_gate.weight")
    out_gate_weight = w("out_gate.weight")
    to_out_norm_weight = w("to_out_norm.weight")
    to_out_norm_bias = w("to_out_norm.bias")
    to_out_weight = w("to_out.weight")

    # --------------------------------------------------------------
    # 2. Layer‑norm on the input (float16)
    # --------------------------------------------------------------
    # layer_norm works with the same dtype as the input, so we
    # normalise in float16 after a cast.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)                     # shape [B, N, N, dim]

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and optional mask
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    if (not nomask) and (mask_tensor is not None):
        # mask: [B, N, N] -> [B, N, N, 1] broadcast over H
        mask = mask_tensor.to(torch.float16, device=device).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5. Core O(N³·H) operation – batched GEMM (fp16 Tensor‑cores)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                      # left: [B, N, N, H]

    # reshape to (B·H, N, N) for torch.bmm
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_t.view(B * H, N, N)               # [B·H, N, N]
    right_mat = right_t.view(B * H, N, N)             # [B·H, N, N]

    # Batched matmul: C = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )                                            # still fp16

    # --------------------------------------------------------------
    # 7. Element‑wise multiplication with the output gate (Triton)
    # --------------------------------------------------------------
    # Flatten tensors for the 1‑D kernel
    out_flat = out.reshape(-1)
    out_gate_flat = out_gate.reshape(-1)

    out_gated_flat = torch.empty_like(out_flat)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024               # ≤ 1024 threads per block on H100
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_gated_flat,
        total_elems,
        BLOCK_SIZE,
        num_warps=4,
    )
    out_gated = out_gated_flat.view_as(out)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out_gated, to_out_weight)   # [B, N, N, dim] (fp16)

    # Cast to float32 as required by the contract
    return out.float()
