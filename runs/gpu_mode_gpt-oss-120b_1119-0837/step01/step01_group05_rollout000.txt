"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1. Layer‑norm over the feature dimension (float16).
2. Four linear projections (left/right and three gates) are performed in
   half‑precision, so the tensors stay in FP16 throughout the bulk of the
   computation.
3. Optional pairwise mask is broadcast and applied in FP16.
4. Core triangular update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication A @ Bᵀ for each (b,h) pair.
   The multiplication is realised with a custom Triton kernel that
   processes tiles of size 128×128×64 (M,N,K) using FP16 inputs and
   FP32 accumulation – a shape that matches the H100 Tensor‑Core tile.
5. The result (still FP16) is layer‑normed, gated, and finally projected
   back to the original feature dimension with a half‑precision linear
   layer before being cast to float32 for the output.

The kernel covers the O(N³·H) work in a single launch (grid:
(M‑tiles, N‑tiles, batch*hidden)).  All other operators remain in PyTorch,
but the heaviest part (the triple product) is fully fused in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr,            # left  : [B*H, M=N, K=N]  (fp16)
    b_ptr,            # right : [B*H, K=N, N]    (fp16) – already transposed
    c_ptr,            # out   : [B*H, M=N, N]  (fp16)
    M, N, K,          # all equal to seq_len
    stride_a_batch,   # stride between consecutive (b,h) in A
    stride_a_i,       # stride for i‑dimension (M) in A
    stride_a_k,       # stride for k‑dimension (K) in A
    stride_b_batch,   # stride between consecutive (b,h) in B
    stride_b_k,       # stride for k‑dimension (K) in B (row)
    stride_b_n,       # stride for n‑dimension (N) in B (col)
    stride_c_batch,   # stride between consecutive (b,h) in C
    stride_c_i,       # stride for i‑dimension (M) in C
    stride_c_n,       # stride for n‑dimension (N) in C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A:[b,M,K] , B:[b,K,N])"""
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_b = tl.program_id(2)          # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialise accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers to tiles of A and B
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # accumulate (FP16 * FP16 -> FP32)
        acc += tl.dot(a, b)

    # write back (convert to FP16)
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Triton‑fused outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N]   (bool/float32) or None
            - weights: dict of model parameters (float32)
            - config : dict with keys "dim" and "hidden_dim" and optional
                       "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # 0) Unpack and basic shapes
    # ------------------------------------------------------------------ #
    inp, mask, weights, config = data
    bs, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1) Cast everything to FP16 early – GPU Tensor Cores love it
    # ------------------------------------------------------------------ #
    # dtype for compute = half
    dev = inp.device
    dtype = torch.float16

    # ---- LayerNorm weight/bias ------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)

    # ---- Linear layer weights --------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype)          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    # ---- Output layers ---------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w = weights["to_out.weight"].to(dtype)                # (dim, hidden)

    # ------------------------------------------------------------------ #
    # 2) Input LayerNorm (FP16)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)    # FP16 result
    x = x.to(dtype)  # ensure half

    # ------------------------------------------------------------------ #
    # 3) Linear projections + gating (all in FP16)
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_w)      # [B,N,N,hidden]  FP16
    right = F.linear(x, right_proj_w)    # [B,N,N,hidden]

    if (mask is not None) and (not nomask):
        # mask may be bool or float; bring it to half and broadcast
        mask_h = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4) Core TriMul (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------ #
    # reshape to (B*hidden, N, N) where the right matrix is already transposed
    left_t = left.permute(0, 3, 1, 2).contiguous()       # [B, hidden, i, k]
    right_t = right.permute(0, 3, 2, 1).contiguous()     # [B, hidden, k, j]

    batch_h = bs * hidden
    left_mat = left_t.view(batch_h, N, N)   # (B*H, M=N, K=N)   FP16
    right_mat = right_t.view(batch_h, N, N) # (B*H, K=N, N)    FP16
    out_mat = torch.empty_like(left_mat)   # FP16 buffer for result

    # compute strides in elements (Triton expects element strides)
    stride_a_batch = left_mat.stride(0)
    stride_a_i = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)      # K‑dimension
    stride_b_n = right_mat.stride(2)      # N‑dimension (after transpose)

    stride_c_batch = out_mat.stride(0)
    stride_c_i = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # launch parameters – tuned for H100 Tensor‑Core (128×128×64 tiles)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        batch_h,                        # one program per (b,head)
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_i=stride_c_i,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # enough warps for this tile size on H100
    )

    # reshape back to [B, N, N, hidden] (still FP16)
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5) Post‑processing (still in FP16)
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                     # gating after norm
    out = F.linear(out, to_out_w)            # back to original dim (FP16)

    # final cast to float32 as required by the API
    return out.to(torch.float32)
