"""
TriMul outgoing kernel (AlphaFold3 variant)

- Layer‑norm over the channel dimension.
- Linear projections to a hidden dimension (`left_proj`, `right_proj`).
- Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
- Apply optional pairwise mask.
- Fuse the N³ core `einsum('... i k d, ... j k d -> ... i j d')` into a
  Triton batched‑matmul kernel that computes, for each hidden slice,
  C = A @ Bᵀ  (A,B ∈ ℝ^{N×N}, FP16, accumulation in FP32).
- Post‑process with a second layer‑norm, output gate and a final linear
  projection back to the original feature dimension.
All heavy compute is done in FP16 with FP32 accumulation and the final
output is returned as FP32.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) for many independent slices.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr,                # *const half*, shape [S, M, K]
    b_ptr,                # *const half*, shape [S, N, K] (accessed transposed)
    c_ptr,                # *mut   half*, shape [S, M, N]
    S, M, N, K,           # int64 scalars
    stride_a_s, stride_a_m, stride_a_k,
    stride_b_s, stride_b_n, stride_b_k,
    stride_c_s, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    For each slice s ∈ [0, S):
        C[s] = A[s] @ B[s]^T
    where A[s] has shape (M, K) and B[s] has shape (N, K).  The kernel
    operates on FP16 inputs and stores the result in FP16 after FP32
    accumulation.
    """
    pid_s = tl.program_id(0)   # slice index
    pid_m = tl.program_id(1)   # block-row
    pid_n = tl.program_id(2)   # block-col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current tile
    a = a_ptr + pid_s * stride_a_s \
                + offs_m[:, None] * stride_a_m \
                + offs_k[None, :] * stride_a_k
    b = b_ptr + pid_s * stride_b_s \
                + offs_k[:, None] * stride_b_k \
                + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # Load tile of A (M×K) and Bᵀ (K×N)
        a_tile = tl.load(a,
                         mask=mask_m[:, None] & (offs_k[None, :] < K),
                         other=0.0)
        b_tile = tl.load(b,
                         mask=(offs_k[:, None] < K) & mask_n[None, :],
                         other=0.0)

        # Convert to FP32 for accumulation
        a_tile = a_tile.to(tl.float32)
        b_tile = b_tile.to(tl.float32)

        # Accumulate the matrix product for this K‑tile
        acc += tl.dot(a_tile, b_tile)

        # Advance pointers to the next K‑tile
        a += BLOCK_K * stride_a_k
        b += BLOCK_K * stride_b_k

    # Write back the result (cast to FP16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_s * stride_c_s \
                    + offs_m[:, None] * stride_c_m \
                    + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with Triton‑accelerated core.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the feature dimension
    # ------------------------------------------------------------------
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.half()   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 3. Linear projections (left/right) and gates (left/right/out)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].half())

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["out_gate.weight"].half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast over hidden dim)
    if not no_mask and mask_tensor is not None:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Core TriMul (batched matmul) – fused with Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    S = B * hidden_dim  # total number of independent matmul slices

    # Reshape to [S, N, N] where each slice corresponds to one hidden channel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(S, N, N)   # [S, N, N]
    right_flat = right_t.view(S, N, N) # [S, N, N]

    out_flat = torch.empty_like(left_flat)   # FP16

    # Strides for the flattened tensors (contiguous: stride0 = N*N, stride1 = N, stride2 = 1)
    stride_s = left_flat.stride(0)
    stride_m = left_flat.stride(1)
    stride_k = left_flat.stride(2)

    # Right matrix is accessed as transposed, so we need the row/col strides swapped
    stride_b_s = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)   # N
    stride_b_k = right_flat.stride(2)   # 1

    stride_c_s = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tuned block sizes for H100 (multiples of 32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        S,                                 # axis 0 -> slice
        (N + BLOCK_M - 1) // BLOCK_M,      # axis 1 -> rows
        (N + BLOCK_N - 1) // BLOCK_N,      # axis 2 -> cols
    )

    tri_mul_batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        S,
        N,
        N,
        N,
        stride_s,
        stride_m,
        stride_k,
        stride_b_s,
        stride_b_n,
        stride_b_k,
        stride_c_s,
        stride_c_m,
        stride_c_n,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=4,          # good default for FP16 on H100
    )

    # Reshape back to [B, N, N, hidden]
    out_h = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5. Post‑processing: layer‑norm on hidden, gating, final linear
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate (broadcast over hidden dimension)
    out = out * out_gate.to(torch.float32)

    # Final projection back to `dim` (bias=False)
    out = torch.nn.functional.linear(out.half(), weights["to_out.weight"].half()).to(torch.float32)

    return out

"""
TriMul (outgoing) – Triton‑accelerated core

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Two bias‑free linear projections (`left_proj`, `right_proj`) expand the
   normalized tensor to `hidden_dim`.  Three additional linear layers
   (`left_gate`, `right_gate`, `out_gate`) produce sigmoid gates.
3. Optional pairwise mask (shape [B,N,N]) is applied element‑wise to the
   projected tensors.
4. The expensive N³ operation
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is fused into a Triton kernel that performs a batched matrix multiplication
   `C = A @ Bᵀ` for each hidden slice `d`.  Inputs/outputs are FP16, accumulation
   is in FP32.
5. The result is layer‑normed over the hidden dimension, multiplied by the
   output gate and finally projected back to the original dimension.
6. The function returns a FP32 tensor of shape [B,N,N,dim].

All heavy compute (the N³ step) runs on the GPU via Triton; the surrounding
operations are performed with efficient PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B ∈ ℝ^{N×N}, FP16)
# It processes `S` independent slices (S = batch * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr,                # *const half*   [S, M, K]
    b_ptr,                # *const half*   [S, N, K] (accessed transposed)
    c_ptr,                # *mut   half*   [S, M, N]
    S, M, N, K,           # program‑time scalars
    stride_a_s, stride_a_m, stride_a_k,
    stride_b_s, stride_b_n, stride_b_k,
    stride_c_s, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    For each slice s ∈ [0, S):
        C[s] = A[s] @ B[s]^T
    A has shape (M, K) and B has shape (N, K).  The kernel works in FP16 with
    FP32 accumulation and writes the result back in FP16.
    """
    pid_s = tl.program_id(0)   # slice index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers for the current tile
    # --------------------------------------------------------------
    a = a_ptr + pid_s * stride_a_s \
                + offs_m[:, None] * stride_a_m \
                + offs_k[None, :] * stride_a_k   # (M_tile, K_tile)

    b = b_ptr + pid_s * stride_b_s \
                + offs_k[:, None] * stride_b_k   # (K_tile, N_tile)
                + offs_n[None, :] * stride_b_n

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a_tile = tl.load(a,
                         mask=mask_m[:, None] & (offs_k[None, :] < K),
                         other=0.0)
        b_tile = tl.load(b,
                         mask=(offs_k[:, None] < K) & mask_n[None, :],
                         other=0.0)

        a_tile = a_tile.to(tl.float32)
        b_tile = b_tile.to(tl.float32)

        acc += tl.dot(a_tile, b_tile)

        a += BLOCK_K * stride_a_k
        b += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # Write result (cast back to FP16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_s * stride_c_s \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a Triton‑accelerated core.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (float32) or None
            - weights      : dict of torch.Tensors containing all model parameters
            - config       : dict with keys ``dim`` and ``hidden_dim``; optional
                             ``nomask`` (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy part of the computation
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Linear projections + sigmoid gates
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].half()
    )   # [B, N, N, hidden]
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].half()
    )

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["left_gate.weight"].half())
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, weights["out_gate.weight"].half())
    )

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast over hidden dimension)
    if not no_mask and mask_tensor is not None:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Core TriMul: batched matmul C = A @ Bᵀ  (one matmul per hidden slice)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    S = B * hidden_dim                     # total independent matmul slices

    # Rearrange tensors to [S, N, N] where each slice corresponds to a hidden channel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(S, N, N)    # [S, N, N]
    right_flat = right_t.view(S, N, N)  # [S, N, N]

    out_flat = torch.empty_like(left_flat)   # FP16 output buffer

    # Strides for the flattened tensors (contiguous layout)
    stride_a_s = left_flat.stride(0)   # = N * N
    stride_a_m = left_flat.stride(1)   # = N
    stride_a_k = left_flat.stride(2)   # = 1

    stride_b_s = right_flat.stride(0)  # = N * N
    stride_b_n = right_flat.stride(1)  # = N   (row stride after transpose)
    stride_b_k = right_flat.stride(2)  # = 1   (col stride after transpose)

    stride_c_s = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Block sizes tuned for H100 (multiples of 32, keep register pressure modest)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        S,                                   # axis 0 – slice index
        (N + BLOCK_M - 1) // BLOCK_M,        # axis 1 – rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,        # axis 2 – cols (j)
    )

    tri_mul_batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        S,
        N,
        N,
        N,
        stride_a_s,
        stride_a_m,
        stride_a_k,
        stride_b_s,
        stride_b_n,
        stride_b_k,
        stride_c_s,
        stride_c_m,
        stride_c_n,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=4,          # good default for FP16 on H100
    )

    # Reshape back to the original layout: [B, N, N, hidden]
    out_h = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Post‑processing: layer‑norm over hidden dim, output gate, final linear
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate (broadcasted)
    out = out * out_gate.to(torch.float32)

    # Final projection back to `dim` (bias=False)
    out = torch.nn.functional.linear(
        out.half(),
        weights["to_out.weight"].half()
    ).to(torch.float32)

    return out
