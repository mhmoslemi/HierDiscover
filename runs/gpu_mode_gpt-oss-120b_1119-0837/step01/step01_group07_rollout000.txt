"""
TriMul (outgoing) – Triton‑accelerated implementation.

The heavy O(N³) part of the TriMul block is a batched matrix multiply:
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]      (h = hidden dim)
which is exactly a GEMM per hidden channel (A @ Bᵀ).  This kernel computes the
batched GEMM in FP16 (to exploit tensor‑cores) while accumulating in FP32.
All surrounding operations (LayerNorm, Linear, gating, final projection) are
performed with PyTorch which already provides highly‑optimized kernels.

The implementation follows these steps:
1. Layer‑norm over the input feature dimension.
2. Linear projections → left / right (shape B,N,N,H) and three linear gates.
3. Optional masking of left/right.
4. Apply left/right gates (element‑wise sigmoid) to the projected tensors.
5. Triton kernel computes the batched contraction left @ rightᵀ per hidden channel.
6. Hidden‑dim layer‑norm, out‑gate multiplication and final linear projection
   back to the original feature dimension.

The kernel is written to work with any batch size B∈{1,2}, sequence length N,
input dimension dim (e.g. 128 or 384) and hidden dimension hidden_dim (e.g. 128).
It runs efficiently on H100 (Tensor‑core friendly) and respects the requirement
that at least part of the computation is done in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K)  – one batch per
# hidden channel.  The kernel works on FP16 inputs and accumulates in FP32.
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix dims (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,                # A row‑stride, col‑stride
    stride_bk, stride_bn,                # B row‑stride, col‑stride (B is transposed)
    stride_cm, stride_cn,                # C row‑stride, col‑stride
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[batch] = A[batch] @ B[batch].T  (A:M×K, B:N×K)."""
    # --------------------------------------------------------------------- #
    #   Grid layout
    # --------------------------------------------------------------------- #
    pid = tl.program_id(axis=0)   # tile id over (M,N)
    batch_id = tl.program_id(axis=1)  # folds batch dimension (B×hidden)

    # tile coordinates
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # offsets within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Pointers for this batch (including batch stride)
    # --------------------------------------------------------------------- #
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   Loop over K dimension (contracted axis)
    # --------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # ---- A tile (M×K) -------------------------------------------------
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_offset[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & mask_k[None, :],
            other=0.0,
        )

        # ---- B tile (N×K) – already transposed, we need Bᵀ which is K×N -----
        b_ptrs = b_batch + (k_offset[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )

        # ---- MAC ---------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    # --------------------------------------------------------------------- #
    #   Store result (convert back to FP16)
    # --------------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# -----------------------------------------------------------------------------#
# Entry point used by the tester
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (ignored when config["nomask"] is True)
        - weights_dict : mapping of weight names -> torch.Tensor
        - config_dict  : must contain keys ``dim`` and ``hidden_dim`` and may contain
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32), result of the TriMul forward pass.
    """
    # --------------------------------------------------------------------- #
    #   Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    #   1) Input layer‑norm over feature dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    #   2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    #   3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   4) Gating (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   5) Core contraction via Triton (FP16 matmul, FP32 accumulation)
    # --------------------------------------------------------------------- #
    #   Re‑order to (B, hidden, N, N) and cast to half‑precision
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # (B, H, N, N)

    #   Transpose the last two dimensions of right so we can compute A @ Bᵀ
    right_T_fp16 = right_fp16.permute(0, 1, 3, 2).contiguous()            # (B, H, N, N)

    #   Flatten batch × hidden‑dim → classic batched GEMM
    B_, H_, M, K = left_fp16.shape                # M = K = N
    _, _, _, N_ = right_T_fp16.shape
    assert M == K == N_ == N, "Inconsistent shapes for contraction"

    total_batches = B_ * H_                        # (B × hidden)

    a = left_fp16.view(total_batches, M, K)        # (batch, M, K)   fp16
    b = right_T_fp16.view(total_batches, K, N_)    # (batch, K, N)   fp16
    c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # --------------------------------------------------------------------- #
    #   Triton kernel launch configuration
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64          # good trade‑off for H100
    grid = (
        ((M + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),  # tiles over M×N
        total_batches,                                                       # batch dimension
    )

    #   Strides for the flattened (batch, row, col) tensors (row‑major)
    stride_am = K                 # A row‑stride
    stride_ak = 1                 # A col‑stride
    stride_bk = N_                # B row‑stride (K dimension)
    stride_bn = 1                 # B col‑stride
    stride_cm = N_                # C row‑stride
    stride_cn = 1                 # C col‑stride

    stride_a_batch = M * K        # batch stride for A
    stride_b_batch = K * N_       # batch stride for B
    stride_c_batch = M * N_       # batch stride for C

    # Launch the kernel
    _trmul_kernel[grid](
        a, b, c,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 favors more warps
    )
    # Ensure kernel completion before moving on (required for correctness)
    torch.cuda.synchronize()

    # --------------------------------------------------------------------- #
    #   6) Reshape back to [B, N, N, hidden_dim]  (float32 for following ops)
    # --------------------------------------------------------------------- #
    out = c.view(B_, hidden_dim, M, N_).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------------- #
    #   7) Post‑processing: hidden‑dim layer‑norm, out‑gate and final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                # broadcasted element‑wise
    out = F.linear(out, weights["to_out.weight"])   # ----------> shape [B,N,N,dim]

    return out
