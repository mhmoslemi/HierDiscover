import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def batched_matmul_fp16_kernel(...):

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # unpack dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # "norm" layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear proj
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B,N,_,H = left.shape
    # reshape for batched GEMM
    left_perm = left.permute(0,3,1,2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0,3,1,2).contiguous()
    # view each head as batch
    left_mat = left_perm.view(B*H, N, N)    # shape [B*H, M=N, K=N]
    right_mat = right_perm.view(B*H, N, N)  # shape [B*H, N, K=N]

    # cast to half for kernel
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)
    # allocate output float32
    out_mat = torch.empty_like(left_fp16, dtype=torch.float32)  # [B*H, N, N]

    # kernel launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * H )
    batched_matmul_fp16_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_fp16.stride(0),
        stride_a_m=left_fp16.stride(1),
        stride_a_k=left_fp16.stride(2),
        stride_b_batch=right_fp16.stride(0),
        stride_b_k=right_fp16.stride(2),   # treat as transposed
        stride_b_n=right_fp16.stride(1),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # second layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass – highly‑fused Triton implementation.

The heavy O(N³·H) part
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched matrix‑multiply C = A @ Bᵀ, where each
(b,h) pair is an independent GEMM of size N×N.  The kernel loads the
matrices in float16, performs the dot‑product on tensor‑cores (half‑input,
float32 accumulation) and writes the result in float32.  All other
operations (layer‑norms, linear layers, gating, mask handling) are kept
in PyTorch, which is negligible compared with the N³ work.

The kernel is tuned for H100 (fp16 tensor‑core) with a 64×64×64 tile,
8 warps per block and uses masks to safely handle non‑multiple‑of‑tile
sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,  # strides of A (BHD, M, K)
    stride_b_batch, stride_b_k, stride_b_n,  # strides of B (treated as transposed)
    stride_c_batch, stride_c_m, stride_c_n,  # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]   (row‑major, fp16)
    B : [batch, N, K]   (row‑major, fp16) – accessed as K×N via swapped strides
    C : [batch, M, N]   (float32)
    """
    pid_batch = tl.program_id(2)          # combined batch*head index
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch element
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # Global pointers for the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + (k_off + offs_k)[None, :] * stride_a_k
        b_ptrs = b_base + (k_off + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Load with masking for out‑of‑bounds elements
        mask_a = (offs_m[:, None] < M) & ((k_off + offs_k)[None, :] < K)
        mask_b = ((k_off + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0, dtype=tl.float16)

        # WMMA‑accelerated dot product (half→float32 accumulation)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write out the tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : mapping of all linear / norm parameters
        - config_dict  : contains at least "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # bias‑free linear projections to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # core O(N³·H) operation – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B: batch size, N: seq_len, H: hidden_dim

    # reshape to (B*H, N, N) and ensure contiguous layout
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_t.view(B * H, N, N)    # [B*H, M=N, K=N]
    right_mat = right_t.view(B * H, N, N)  # [B*H, N, K=N]

    # cast inputs to fp16 for tensor‑core acceleration
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # output will be accumulated in FP32
    out_mat = torch.empty_like(left_fp16, dtype=torch.float32)  # [B*H, N, N]

    # launch parameters – tile size tuned for H100 fp16 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over rows   (M dimension)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over cols   (N dimension)
        B * H                           # batch × hidden
    )

    batched_matmul_fp16_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_fp16.stride(0),
        stride_a_m=left_fp16.stride(1),
        stride_a_k=left_fp16.stride(2),
        stride_b_batch=right_fp16.stride(0),
        stride_b_k=right_fp16.stride(2),   # treat right as transposed
        stride_b_n=right_fp16.stride(1),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                      # H100 works well with 8 warps per block
    )

    # reshape back to original 4‑D layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # second layer‑norm, output gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
