import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul forward kernel (outgoing version) for AlphaFold3.

Algorithm:
1. Layer‑norm the input tensor (B, N, N, C).
2. Linear projections to a hidden dimension H (C → H) for left/right.
3. Apply an optional pairwise mask.
4. Compute three gates (sigmoid(linear(input))) and multiply them
   into the projected tensors.
5. Core operation: for each hidden slice d, compute
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batched matrix multiply left @ rightᵀ per hidden channel.
   This is implemented in Triton (FP16 inputs, FP32 accumulation).
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and finally a linear projection back to the original dimension.

The heavy O(N³·H) contraction (step 5) is fully fused in Triton.
All other ops use PyTorch which already provides highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Triton kernel: batched GEMM on half‑precision inputs (A,B) -> FP32 C
# A : (B*H, M, K)   (row‑major)
# B : (B*H, K, N)   interpreted via swapped strides (no explicit transpose)
# C : (B*H, M, N)   (row‑major, FP32)
# --------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)           # batch*hidden index
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for edge handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with correct strides
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles (FP16) with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot -> FP32 accumulation
        acc += tl.dot(a, b)

    # Store the result (FP32) with mask
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------
# Entry point called from the evaluator
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping of weight names to torch.Tensor
            - config_dict: contains 'dim' (C) and 'hidden_dim' (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ----------------------------------------------------------------
    # Load weights
    # ----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------
    # 1. Input layer‑norm over the channel dimension C
    # ----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ----------------------------------------------------------------
    # 2. Linear projections to hidden dimension H (no bias)
    # ----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # (B, N, N, H)
    right = F.linear(x, right_proj_weight) # (B, N, N, H)

    # ----------------------------------------------------------------
    # 3. Optional pairwise mask (skip if config["nomask"] is True)
    # ----------------------------------------------------------------
    apply_mask = not config.get("nomask", True)
    if apply_mask and mask_tensor is not None:
        # mask Tensor shape [B, N, N]; broadcast to hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------
    # 4. Gating
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # (B, N, N, H)

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5. Core TriMul contraction via Triton
    #    Compute out_hidd = Σ_k left[..., k, :] * right[..., k, :].T
    # ----------------------------------------------------------------
    # Cast to half‑precision for the heavy GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, H, N, N) then flatten batch*H for the kernel
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * H
    left_mat = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_mat = right_perm.view(BH, N, N) # (BH, K=N, N) – we will treat it transposed via strides

    # Prepare output buffer (FP32)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides for A (left_mat)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    # Strides for B (right_mat). We need (K,N) layout -> use column‑major stride for K
    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(2)   # stride of the original column dimension (1)
    stride_b_n = right_mat.stride(1)   # stride of the original row dimension (N)

    # Strides for C (out_mat)
    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Tuning parameters (empirically good for N up to 1024 on H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # program_id(0) – rows
        triton.cdiv(N, BLOCK_N),          # program_id(1) – cols
        BH                                 # program_id(2) – batch*hidden
    )

    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------------
    # 6. Output layer‑norm, gating, and linear projection back to C
    # ----------------------------------------------------------------
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(H,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias)

    out_hidden = out_hidden * out_gate   # element‑wise gating

    # Final linear projection H -> C (no bias)
    out = F.linear(out_hidden, to_out_weight)  # (B, N, N, C)

    return out
