@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_i, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_b = tl.program_id(2)  # batch*hid

    # Offsets
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load a tile
        a_ptrs = a_ptr + pid_b * stride_a_batch + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n

        # mask loads
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_j[None, :] < N), other=0.0)

        acc += tl.dot(a, b)  # fp16->fp32 accumulation

    # store result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_i[:, None] < M) & (offs_j[None, :] < N))
```

def custom_kernel(data):
    # Unpack
    inp, mask, weights, config = data
    bs, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    # maybe also dim_z for gating? not needed.
    # Load weights
    # same as before.
    # layernorm
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Linear projections (FP16)
    # Use torch.nn.functional.linear which uses matmul
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None and not config.get("nomask", False):
        left = left * mask.unsqueeze(-1).to(left.dtype)
        right = right * mask.unsqueeze(-1).to(right.dtype)
    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # apply gating
    left = left * left_gate
    right = right * right_gate
    # cast to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute for GEMM: shape [B, hidden, N, N]
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 2, 1).contiguous()   # transpose second last dims
    # flatten B*hidden
    batch_h = bs * hidden
    left_mat = left_h.view(batch_h, N, N)
    right_mat = right_h.view(batch_h, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # compute strides
    stride_a_batch, stride_a_i, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride() # note right has shape (batch, K, N) after permute
    stride_c_batch, stride_c_i, stride_c_n = out_mat.stride()

    # Kernel launch
    # choose block sizes based on N:
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             batch_h)

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N, N=N, K=N,
        stride_a_batch=stride_a_batch, stride_a_i=stride_a_i, stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch, stride_b_k=stride_b_k, stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch, stride_c_i=stride_c_i, stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # reshape back
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # post layernorm and gating and projection
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) – Triton‑accelerated core update

The TriMul module from AlphaFold‑3 updates a pairwise tensor X ∈ℝ^{B×N×N×C}
as follows (outgoing version):

    1. Layer‑norm over C.
    2. Two linear projections → left, right ∈ℝ^{B×N×N×H}.
    3. (Optional) element‑wise mask multiplication.
    4. Gating: left_gate, right_gate, out_gate = sigmoid(linear(x)).
       left ← left * left_gate ,   right ← right * right_gate.
    5. Core update (O(N³·H)):
       
           out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

       which is a batched matrix‑multiply C = A @ Bᵀ with

           A : (B·H, M=N, K=N)  = left  permuted to (i,k)
           B : (B·H, K=N, N)    = right permuted to (k,j)

       The multiplication is performed in FP16 with FP32 accumulation
       using a custom Triton kernel that tiles the M‑ and N‑axis
       (BLOCK_M × BLOCK_N) and the reduction axis K (BLOCK_K).
       Tile sizes are chosen to saturate the H100 Tensor‑cores
       (BLOCK_M/N = 64‑128, BLOCK_K = 32‑64, num_warps = 4‑8).

    6. Layer‑norm + out_gate on the H‑dimensional result.
    7. Final linear projection back to C.

Only the O(N³·H) GEMM is written in Triton; all other
operations are executed with high‑performance PyTorch kernels.
The function ``custom_kernel`` follows the same signatures used in
the test harness.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A : [B·H, M, K],  B : [B·H, K, N])
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,                 # left matrix  (fp16)
    b_ptr,                 # right matrix (fp16, already transposed)
    c_ptr,                 # output matrix (fp16)
    M, N, K,               # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_i, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM using Tensor‑cores.

    Each program instance computes a tile (BLOCK_M×BLOCK_N) of C for a
    particular (batch*hidden) index.
    """
    pid_m = tl.program_id(0)          # tile row  (i)
    pid_n = tl.program_id(1)          # tile col  (j)
    pid_b = tl.program_id(2)          # batch * hidden

    # Offsets of the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for numerical stability.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current sub‑tiles of A and B.
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load with out‑of‑bounds guard.
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        # Tensor‑core dot (fp16·fp16 → fp32 accumulator).
        acc += tl.dot(a, b)             # (BLOCK_M, BLOCK_N)

    # Write the tile back to global memory (convert to fp16).
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Python entry‑point – performs the full TriMul forward pass.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weights_dict : mapping from module attribute name to torch.Tensor
        - config_dict  : must contain keys "dim" and "hidden_dim"
                        plus optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the updated pairwise
        representation.
    """
    # ------------------------------------------------------------------
    # 0) Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    no_mask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Load all required weight tensors (all are FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]                     # shape (dim, hidden)

    # ------------------------------------------------------------------
    # 2) Input Layer‑Norm
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – keep in FP32 for max accuracy
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # [B, N, N, H]
    right = F.linear(x, right_proj_w)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast on feature dim)
    # ------------------------------------------------------------------
    if mask is not None and not no_mask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Gating – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H]

    # Apply element‑wise gates.
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to FP16 (Tensor‑core friendly) and reshape for GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    #   left : (B, N, N, H) → (B, H, N, N)  (rows = i, cols = k)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    #   right: (B, N, N, H) → (B, H, N, N) then swap axes so we have (k, j)
    right_h = right_h.permute(0, 3, 2, 1).contiguous()

    # Collapse leading dimensions: each (batch, hidden) pair is an
    # independent GEMM instance.
    batch_h = B * hidden
    left_mat = left_h.view(batch_h, N, N)      # [B*H, M=N, K=N]
    right_mat = right_h.view(batch_h, N, N)    # [B*H, K=N, N]  (already “transposed”)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 7) Compute strides (in elements) – required by the kernel
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_i, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_i, stride_c_n = out_mat.stride()

    # ------------------------------------------------------------------
    # 8) Kernel launch configuration – tile size tuned for H100
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
        num_warps = 8
    else:  # up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along N (j)
        batch_h,                        # separate GEMM per (batch, hidden)
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_i=stride_c_i,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to the original layout (B, N, N, H) and FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden, N, N) \
                .permute(0, 2, 3, 1) \
                .contiguous() \
                .to(torch.float32)                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10) Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                      # element‑wise gating
    out = F.linear(out, to_out_w)             # → [B, N, N, dim]

    return out
