Implementation of the outgoing Triangle Multiplicative Update (TriMul) forward pass.
- Input is layer‑normalized.
- Linear projections are performed in FP16.
- A Triton kernel fuses the optional mask with the two per‑channel gates onto the projected
  left/right tensors, reducing two memory passes to one.
- The O(N³) contraction is carried out with a batched GEMM (torch.bmm) on half‑precision
  tensors, which uses Tensor‑core accelerated cuBLAS kernels.
- The hidden‑dimension output is layer‑normalized, gated, and projected back to the original
  dimension.
All operations except the gate‑mask fusion are performed with PyTorch; the final result is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # FP16 [B,N,N,H] – in‑place output
    right_ptr,              # FP16 [B,N,N,H] – in‑place output
    left_gate_ptr,          # FP16 [B,N,N,H]
    right_gate_ptr,         # FP16 [B,N,N,H]
    mask_ptr,               # FP16 [B,N,N] (flattened)
    total_elements: tl.int64,
    hidden_dim:    tl.int64,
    BLOCK_SIZE:    tl.constexpr,
):
    """Fuse mask (if present) and per‑channel gates onto the projected left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Determine which pairwise entry (b,i,j) this element belongs to.
    # Each pairwise entry repeats hidden_dim times, so we divide by hidden_dim.
    pair_idx = offs // hidden_dim

    # Load mask value for the pair (broadcast over hidden_dim)
    mask_val = tl.load(mask_ptr + pair_idx, mask=mask, other=tl.float16(0.0))

    left_val  = tl.load(left_ptr  + offs, mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    left_out  = left_val * left_g * mask_val
    right_out = right_val * right_g * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

            - weights_dict: mapping of all module parameters (float32)
            - config_dict: {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------ #
    # 2) Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------ #
    x_h = x.half()   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------ #
    # 3) Load FP16 weights
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------ #
    # 4) Linear projections (bias‑free) – half precision
    # ------------------------------------------------------------------ #
    left_proj  = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 5) Compute per‑channel gates (sigmoid)
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).half()   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).half()  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).half()    # [B,N,N,H]

    # ------------------------------------------------------------------ #
    # 6) Fuse mask and gates (Triton) – only needed when a mask is present
    # ------------------------------------------------------------------ #
    if nomask:

        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:

        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()

        total_elements = left_proj.numel()   # B * N * N * H
        BLOCK_SIZE = 1024
        grid = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elements,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
        left, right = left_proj, right_proj

    # ------------------------------------------------------------------ #
    # 7) Heavy O(N³) contraction via batched GEMM (cuBLAS)
    #    out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------ #
    # reshape to (B, H, N, N) → view as (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_flat  = left_perm.view(batch, N, N)               # (B*H, N, K)
    right_flat = right_perm.view(batch, N, N).transpose(1, 2)  # (B*H, K, N)

    # Batched matrix multiplication (FP16 → FP16, accumulation in FP32)
    out_flat = torch.bmm(left_flat, right_flat)            # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 8) Hidden‑dim layer‑norm, out‑gate and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original dim (FP16)

    # ------------------------------------------------------------------ #
    # 9) Cast back to FP32 for the downstream API
    # ------------------------------------------------------------------ #
    return out.float()

2. Cast to FP16 and perform the four linear projections (left/right and their gates).

5. Hidden‑dim layer‑norm, element‑wise out‑gate, and the final linear projection back

        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Convert mask to FP16 and flatten it (shape: B*N*N)
        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1)

        total_elements = left_proj.numel()          # B * N * N * H
        BLOCK_SIZE = 1024
        grid = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elements,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
        left, right = left_proj, right_proj

    # ------------------------------------------------------------------ #
    # 7) O(N³) contraction via batched GEMM (cuBLAS)
    #    out = left @ rightᵀ   (per hidden channel)
    # ------------------------------------------------------------------ #
    # Rearrange to (B, H, N, N) and view as a batch of matrices
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_flat  = left_perm.view(batch, N, N)                     # (B*H, N, K)
    right_flat = right_perm.view(batch, N, N).transpose(1, 2)    # (B*H, K, N)

    # Batched half‑precision matrix multiplication (accumulation in FP32 internally)
    out_flat = torch.bmm(left_flat, right_flat)                  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 8) Hidden‑dim LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                      # element‑wise gating
    out = F.linear(out, to_out_w)             # back to original dim (FP16)

    # ------------------------------------------------------------------ #
    # 9) Convert back to FP32 for the downstream API
    # ------------------------------------------------------------------ #
    return out.float()
```

This implementation follows the reference TriMul logic, uses a Triton kernel to fuse mask and gating, relies on a highly‑optimized batched GEMM for the cubic contraction, and returns the final result in `float32`.
