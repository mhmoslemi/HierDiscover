"""
TriMul (outgoing) – fast mixed‑precision implementation.

Algorithm
---------
1. Layer‑norm on the input tensor (float32 → float16).
2. Fuse the *projection* and *gate* linear layers for the left and right
   tensors by concatenating the weights (2 × hidden_dim rows).  One `F.linear`
   call yields both the projected values and the pre‑sigmoid gate values.
3. Apply `sigmoid` to the gate halves.
4. If a mask is present, a single Triton kernel fuses the mask, the
   left‑gate and the right‑gate onto the projected tensors:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   (otherwise perform the two element‑wise multiplications in PyTorch.)
5. Perform the O(N³) contraction as a batched GEMM:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   realized with `torch.bmm` on tensors of shape (B·H, N, N).
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to `dim`.
7. Return the result in float32.

The heavy compute (batched GEMM) stays on cuBLAS (Tensor‑cores) while the
mask‑gate fusion is performed by a lightweight Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------#
# Triton kernel: fuse element‑wise mask and the left/right gates.
# -----------------------------------------------------------------------------#
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # FP16 [B,N,N,H] – projected left (input & output)
    right_ptr,         # FP16 [B,N,N,H] – projected right (input & output)
    left_gate_ptr,     # FP16 [B,N,N,H] – sigmoid gate for left
    right_gate_ptr,    # FP16 [B,N,N,H] – sigmoid gate for right
    mask_ptr,          # FP16 [B,N,N]   – broadcast mask (flattened)
    total_elements: tl.int64,   # B * N * N * H
    hidden_dim:    tl.int64,    # H
    BLOCK_SIZE:    tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # mask is broadcast across the hidden dimension
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))

    left      = tl.load(left_ptr      + offs, mask=valid, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=valid, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr+ offs, mask=valid, other=tl.float16(0.0))

    left_out  = left  * left_g  * mask_val
    right_out = right * right_g * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


# -----------------------------------------------------------------------------#
# Public entry point
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping of all module parameters (float32)
            - config_dict: {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)
    x = x.half()  # cast to FP16 for the bulk of the computation

    # --------------------------------------------------------------------- #
    # 2) Fuse projection + gate Linear layers (two matmuls → two matmuls)
    # --------------------------------------------------------------------- #
    # concatenate projection and gate weights along the output dimension
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*H, dim)

    # one Linear call yields both projection and pre‑sigmoid gate values
    left_all = F.linear(x, left_combined_w)   # [B, N, N, 2*H]
    right_all = F.linear(x, right_combined_w)  # [B, N, N, 2*H]

    # split the concatenated dimension
    left_proj = left_all[..., :H]   # [B, N, N, H]
    left_gate = left_all[..., H:]   # [B, N, N, H]
    right_proj = right_all[..., :H]
    right_gate = right_all[..., H:]

    # sigmoid gates (stay in FP16)
    left_gate = torch.sigmoid(left_gate).half()
    right_gate = torch.sigmoid(right_gate).half()

    # out‑gate (used later)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].half())).half()

    # --------------------------------------------------------------------- #
    # 3) Apply mask and fuse gates (if a mask is present)
    # --------------------------------------------------------------------- #
    if not nomask:
        # Ensure contiguous layout for flat indexing
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Convert mask to FP16 and flatten (B * N * N)
        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1).contiguous()

        total_elems = left_proj.numel()  # B * N * N * H
        BLOCK_SIZE = 4096                # threads per program
        grid = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elems,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,   # 4 warps → 128 threads, good occupancy on H100
        )
        left = left_proj
        right = right_proj
    else:
        # simple element‑wise fusion when there is no mask
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 4) Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑cores)
    # --------------------------------------------------------------------- #
    # reshape to (B*H, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_mat = left_perm.view(batch, N, N)                 # (B*H, N, N)
    right_mat = right_perm.view(batch, N, N).transpose(1, 2)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)               # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 5) Hidden‑dim LayerNorm, output gate and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).half()

    out = out * out_gate                # element‑wise gating
    out = F.linear(out, weights["to_out.weight"].half())  # back to dim

    # --------------------------------------------------------------------- #
    # 6) Cast back to float32 for the downstream API
    # --------------------------------------------------------------------- #
    return out.float()
