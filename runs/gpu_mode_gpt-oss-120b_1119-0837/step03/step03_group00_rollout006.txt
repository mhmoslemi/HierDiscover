      stride_right_b, stride_right_i, stride_right_k, stride_right_h,
      stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    pid_j = tl.program_id(1)

    h = pid_bh % H

Pseudo-code:

```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr = 1e-5
):
    pid_i = tl.program_id(0) # row block
    pid_j = tl.program_id(1) # col block
    pid_bh = tl.program_id(2) # batch*head

    # decode batch and head
    b = pid_bh // H
    h = pid_bh % H

    # base offsets
    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    base_out = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets within block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = tl.cast(left, tl.float32)

        # load right tile (K x N) -> use j axis for columns
        right_ptrs = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = tl.cast(right, tl.float32)

        acc += tl.dot(left, right)

    # store result to out
    out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Optimized Triton implementation of the “outgoing” TriMul operator (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm over the model dimension (float32) – custom 2‑pass Triton kernel.
2️⃣ Cast to FP16 for the heavy work.
3️⃣ Fuse each projection + its gate into a single linear call
   (proj and gate weights are concatenated → one matmul each for left/right).
4️⃣ Optional mask applied to the projected tensors (broadcast over the hidden dim).
5️⃣ Element‑wise gating of the projected tensors.
6️⃣ Core N³ contraction (pairwise reduction) implemented as a
   batched half‑precision GEMM in a custom Triton kernel – no data copy, stride‑aware.
7️⃣ Layer‑norm over the hidden dimension (FP16) – another 2‑pass kernel.
8️⃣ Apply the output‑gate (sigmoid of a linear on the normalized input).
9️⃣ Final linear projection back to the model dimension (FP16 → FP32 output).

All heavy linear work stays in FP16 (Tensor‑Core friendly) and the only
full‑precision steps are the two LayerNorms (using FP32 accumulators).
The implementation works for any batch size ≤2, any sequence length in {128,…,1024},
and any model/hidden dimension from the test matrix.  The code respects
the optional mask flag (``nomask``) and returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ----- row range for this program -----
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: mean / variance (FP32) -----
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: write normalized + affine -----
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) on a 2‑D tensor [M, N] using the Triton kernel above."""
    assert x.dim() == 2, "expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )

# ----------------------------------------------------------------------
# N³ contraction kernel (batched half‑precision GEMM, stride‑aware)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments:
        left_ptr, right_ptr, out_ptr : base pointers (FP16)
        B, N, H                     : batch size, sequence length, hidden dim
        stride_*                    : strides of the corresponding tensors
        BLOCK_M, BLOCK_N, BLOCK_K   : tile sizes
    """
    pid_i = tl.program_id(0)   # tile row
    pid_j = tl.program_id(1)   # tile col
    pid_bh = tl.program_id(2)  # combined batch * head index

    # ----- decode batch (b) and head (h) from pid_bh -----
    b = pid_bh // H
    h = pid_bh % H

    # ----- compute base offsets for this (b,h) -----
    base_l = left_ptr + b * stride_l_b + h * stride_l_h
    base_r = right_ptr + b * stride_r_b + h * stride_r_h
    base_o = out_ptr + b * stride_o_b + h * stride_o_h

    # ----- tile offsets within N -----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k -----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K)   rows=i, cols=k
        l_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        l = tl.load(l_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        l = tl.cast(l, tl.float32)

        # right tile: (BLOCK_K, BLOCK_N)   rows=k, cols=j
        r_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        r = tl.load(r_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)
        r = tl.cast(r, tl.float32)

        # dot-product for this k‑tile
        acc += tl.dot(l, r)

    # ----- write back (convert to FP16) -----
    o_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    out = acc.to(tl.float16)
    tl.store(o_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_outgoing_fp16(left: torch.Tensor,
                          right: torch.Tensor,
                          out: torch.Tensor,
                          BLOCK_M=32,
                          BLOCK_N=32,
                          BLOCK_K=64):
    """Convenient wrapper that launches the Triton contraction kernel."""
    assert left.is_contiguous() and right.is_contiguous() and out.is_contiguous()
    B, N, _, H = left.shape
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Entry point – Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] (float32) or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict containing keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)            # (B, N, N, D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy lifting
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection & gate for left/right (single matmul each)
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                     # (2H, D)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Linear → (B, N, N, 2H)
    left_comb  = F.linear(x_f16, left_comb_w)       # (B, N, N, 2H)
    right_comb = F.linear(x_f16, right_comb_w)      # (B, N, N, 2H)

    # Split into (proj, gate)
    left_proj      = left_comb[..., :H]               # (B, N, N, H)
    left_gate_pre  = left_comb[..., H:]                # (B, N, N, H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)          # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via custom Triton GEMM kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                 # (B, N, N, H) ‑‑ FP16
    tri_mul_outgoing_fp16(left, right, out,
                          BLOCK_M=32, BLOCK_N=32, BLOCK_K=64)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)                # (B*N*N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)   # (M, H) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (both FP16)
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H) FP16
    gated = out_norm * out_gate_flat                     # (M, H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)               # (M, D) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)   # (B, N, N, D) FP32
    return output
```
