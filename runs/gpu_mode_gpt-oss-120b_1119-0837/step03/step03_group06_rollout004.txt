"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The operator takes a 4‑D tensor X ∈ ℝ^{B×N×N×C} and a (optional) binary mask.
It performs:

1. Layer‑norm on the last channel dimension (C) → fp32.
2. Cast to fp16 and apply four linear projections:
      left_proj , right_proj  → hidden_dim H
      left_gate , right_gate , out_gate → H (followed by sigmoid)
3. Multiply left/right by their gates (and by the mask if supplied).
4. Core “triangle‑multiplication’’:
      out_{i,j,h} = Σ_k left_{i,k,h} · right_{j,k,h}
   This is a batched matrix‑multiply (B·H GEMMs of size N×N×N).
   We implement it with a custom Triton kernel that
        * works on fp16 inputs,
        * accumulates in fp32,
        * stores the result in fp16,
        * packs the batch and head dimension into a single batch axis.
5. Layer‑norm over the hidden dimension, multiply by the out‑gate and a final
   linear projection back to C.
6. Return the result in fp32.

All heavy linear layers use torch.nn.functional for best cuBLAS performance.
The Triton kernel is the only custom kernel and is heavily tiled for the
H100 (Tensor‑Core) architecture.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                   # fp16*   output   [C, M, N]
    a_ptr,                     # fp16*   left     [C, M, K]
    b_ptr,                     # fp16*   right    [C, K, N] (logically transposed)
    M, N, K,                   # matrix dimensions (all == seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,        # left‑tensor strides
    stride_b_c, stride_b_k, stride_b_n,        # right‑tensor strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the outgoing TriMul:
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    fp16 inputs, fp32 accumulation, fp16 write‑back.
    """
    pid_m = tl.program_id(0)        # row block
    pid_n = tl.program_id(1)        # col block
    pid_c = tl.program_id(2)        # batch×head block

    # -------------------------------------------------------------
    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M,K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K,N)

        # Load with masking (out‑of‑bounds → zero)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16

        # GEMM accumulate (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # Write the result (convert to fp16)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data: tuple (x, mask, weights, config)
            x        : torch.Tensor[B, N, N, C]   (float32)
            mask     : torch.Tensor[B, N, N]     (bool/float) or None
            weights  : dict of model parameters (see keys below)
            config   : dict with keys
                - "dim"        : C    (input channel size)
                - "hidden_dim": H    (hidden dimension)
                - "nomask"    : bool (True → ignore mask, default True)

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = x.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (fp32) → fp16
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Linear projections (bias‑free, fp16)
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x, w_left_proj)   # [B,N,N,H] fp16
    right = F.linear(x, w_right_proj)  # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid, fp16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, w_left_gate))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, w_out_gate))    # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4) Apply gates and (optional) mask
    left  = left * left_gate
    right = right * right_gate

    if (not nomask) and (mask is not None):
        # mask is broadcast over the hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left  = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Pack batch and head axes → (C_total, N, N) where C_total = B*H
    B, N, _, _ = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    C_total = B * hidden_dim
    left_flat  = left.reshape(C_total, N, N)   # (C, M, K)
    right_flat = right.reshape(C_total, N, N)  # (C, K, N)

    # -----------------------------------------------------------------
    # 6) Triton GEMM kernel – batched matrix multiplication
    #    out_c[M,N] = left_c[M,K] @ right_c[K,N]
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Tile sizes – pick a good configuration for the target seq‑len
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:          # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides in element units (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)   # reduction (K) axis
    stride_b_n = right_flat.stride(2)   # N axis

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (batch * hidden)
    )

    # Choose a warp count that gives enough registers per thread
    num_warps = 16 if N > 512 else 8

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                       # M,N,K
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) – still fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Post‑processing: LayerNorm → out‑gate → final linear (all fp16)
    w_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_out_norm_w,
        bias=w_out_norm_b,
    )

    out = out * out_gate          # apply out‑gate (already fp16)

    w_to_out = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, w_to_out)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 9) Cast to float32 for downstream consumption
    return out.to(torch.float32)
