def custom_kernel(data):

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    norm_bias = weights["norm.bias"].to(device, dtype)

    # out_gate_pre: 4H:5H
    left_proj = combined[..., 0:H]
    left_gate_pre = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate_pre = combined[..., 3*H:4*H]
    out_gate_pre = combined[..., 4*H:5*H]

    # apply sigmoid
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # optional mask: broadcast to hidden dimension H
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # elementwise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 3. Core contraction: batched GEMM
    # reshape for batched multiplication
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = bs * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # GEMM: left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (BH,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 4. Second LayerNorm (FP16)
    out_flat = out.view(bs * N * N, H)  # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # (M, H)

    # 5. Gating and final projection
    out_gate_flat = out_gate.view(M, H)
    gated = out_norm * out_gate_flat
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # shape (D, H)
    # note: F.linear expects weight shape (out_features, in_features)
    output_flat = F.linear(gated, to_out_w)  # (M, D)

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output

"""
Highly‑optimized Triton‑accelerated implementation of the “outgoing” TriMul
operator (AlphaFold‑3).  Key ideas:
* First LayerNorm (float32) is performed by a custom 2‑pass Triton kernel.
* All heavy linear work is fused into **one** FP16 GEMM:
    - left_proj + left_gate + right_proj + right_gate + out_gate
    - weight shape (5*H, D) where H = hidden_dim, D = model dim.
* Mask is applied only to the projected tensors (left/right) after the
  combined GEMM.
* The N³ contraction is a batched half‑precision GEMM (torch.bmm) on
  tensors of shape (B*H, N, N).
* The second LayerNorm (FP16) is also done by a custom Triton kernel.
* Final projection back to D and cast to float32.

This reduces the number of separate GEMM launches from five to two and
keeps the memory traffic for the weight matrices minimal, yielding a
large speed‑up while preserving the mathematical semantics of the reference
PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 2‑pass LayerNorm kernel (float32 input → float32 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm in float32 using the Triton kernel above."""
    assert x.dim() == 2, "expects a 2‑D tensor [M, N]"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# 2‑pass LayerNorm kernel (FP16 input → FP16 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(

        xf = tl.cast(x, tl.float32)
        norm_x_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,

    """LayerNorm in FP16 using the Triton kernel above."""
    assert x.dtype == torch.float16, "expects FP16 input"
    assert x.dim() == 2, "expects a 2‑D tensor [M, N]"
    M, N = x.shape

    bias = bias.contiguous()

        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )

# --------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # should be torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – collapse batch & pair dimensions
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                # (M, D)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)  # (M, D)
    x_norm = x_norm_flat.view(bs, N, N, dim)                           # (B,N,N,D)

    # Cast to FP16 for all following heavy work
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse LEFT, RIGHT and OUT gate linear projections into ONE GEMM
    # ------------------------------------------------------------------
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_lproj = weights["left_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rproj = weights["right_proj.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    weight_fused = torch.cat(
        [w_lproj, w_lgate, w_rproj, w_rgate, w_outgate], dim=0
    ).to(device, torch.float16)                     # (5*H, D)

    # Linear: (M, D) @ (D, 5*H) -> (M, 5*H)
    fused = F.linear(x_f16, weight_fused)          # (M, 5*H)
    fused = fused.view(bs, N, N, 5 * H)            # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply sigmoid gates and optional pair‑mask
    # ------------------------------------------------------------------
    # slice boundaries
    h0 = 0
    h1 = H
    h2 = 2 * H
    h3 = 3 * H
    h4 = 4 * H
    h5 = 5 * H

    left_proj   = fused[..., h0:h1]    # (B,N,N,H)
    left_gate_p = fused[..., h1:h2]
    right_proj  = fused[..., h2:h3]
    right_gate_p = fused[..., h3:h4]
    out_gate_p  = fused[..., h4:h5]

    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Element‑wise gating
    left  = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate        # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)               # (B*H, N, N)

    # out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to D
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(M, H)           # (M, H)
    gated = out_norm * out_gate_flat               # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)       # (M, D)

    output = output_flat.view(bs, N, N, dim).to(torch.float32)
    return output
```
