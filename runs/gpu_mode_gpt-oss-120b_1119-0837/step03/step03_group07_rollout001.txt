"""
TriMul “outgoing” forward pass (AlphaFold‑3) – highly optimized.

Algorithm
---------
1️⃣  LayerNorm over the model dimension (C) – PyTorch FP32 functional layer_norm.
2️⃣  One fused FP16 linear projects the normalized tensor to five tensors:
    left, right, left‑gate, right‑gate, out‑gate (shape H = hidden_dim).
3️⃣  (Optional) binary mask is applied to left/right before gating.
4️⃣  Sigmoid gating is performed in FP16 and applied to left/right.
5️⃣  Batched matrix multiplication (contraction over the summed dimension k):
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h].
    Implemented with a custom Triton kernel that processes all heads
    (B·H) together, loads tiles in FP16, accumulates in FP32 and writes FP16.
6️⃣  LayerNorm over the hidden dimension (H) – PyTorch FP32 functional layer_norm.
7️⃣  Apply the output gate (sigmoid) and a final FP16 linear projection back
    to the model dimension C.
All heavy compute runs in FP16 (tensor‑core friendly) while the two LayerNorms
run in FP32 for numerical stability. The custom kernel fuses the N³ contraction
across heads and batch, which is the dominant cost.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched half‑precision matrix multiplication kernel for the
# TriMul “outgoing” contraction.
#   C[b] = A[b] @ B[b]    where A,B,C : (B·H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                         # logical dimensions (M = N = K = seq_len)
    stride_ah, stride_am, stride_ak, # strides for A: (batch, row, col)
    stride_bh, stride_bk, stride_bn, # strides for B: (batch, row, col)
    stride_ch, stride_cm, stride_cn, # strides for C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A, B, C have shape (BH, N, N) with dtype fp16.
    The kernel computes C = A @ B for each of the BH matrices.
    Accumulation is performed in fp32 for numerical safety.
    """
    pid_batch = tl.program_id(2)   # flattened batch*head index
    pid_m = tl.program_id(0)       # tile row index
    pid_n = tl.program_id(1)       # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A and B in fp16
        a = tl.load(
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product; a,b are fp16 -> result fp32
        acc += tl.dot(a, b)

    # Write back as fp16
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M: int = 128,
                         BLOCK_N: int = 128,
                         BLOCK_K: int = 128) -> torch.Tensor:
    """
    Batched matrix multiplication for tensors of shape (B, H, N, N) in fp16.
    Returns C of the same shape and dtype.

    Parameters
    ----------
    A, B : torch.Tensor
        Input tensors of shape (B, H, N, N) (dtype torch.float16).
    BLOCK_M, BLOCK_N, BLOCK_K : int
        Tile sizes. 128 works well on H100.

    Returns
    -------
    C : torch.Tensor
        Result of shape (B, H, N, N) (fp16).
    """
    B_dim, H_dim, N, _ = A.shape
    BH = B_dim * H_dim

    # Flatten batch‑head dimension to launch a single 3‑D grid.
    A_flat = A.view(BH, N, N).contiguous()
    B_flat = B.view(BH, N, N).contiguous()
    C_flat = torch.empty_like(A_flat)

    stride_ah, stride_am, stride_ak = A_flat.stride()
    stride_bh, stride_bk, stride_bn = B_flat.stride()
    stride_ch, stride_cm, stride_cn = C_flat.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        BH,                        # batch * head
    )

    triton_matmul_outgoing_kernel[grid](
        A_ptr=A_flat,
        B_ptr=B_flat,
        C_ptr=C_flat,
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps = 256 threads per block
        num_stages=5,       # pipeline depth
    )
    # Reshape back to (B, H, N, N)
    return C_flat.view(B_dim, H_dim, N, N)


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    # reshape to (B*N*N, C) for functional layer_norm
    x_flat = input_tensor.view(-1, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x_norm_flat = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # FP32
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim).half()  # FP16 for rest

    # --------------------------------------------------------------
    # 2️⃣ Combined projection (FP16 linear) → left, right, gates
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (out_features stacked)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)  # shape (5*H, C)

    # Linear: (M, C) @ (5*H, C).T  → (M, 5*H)
    proj = F.linear(x_norm.view(-1, dim), combined_weight)  # FP16, M = B*N*N

    # --------------------------------------------------------------
    # 3️⃣ Split projected tensors and reshape
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    # Reshape to (B, N, N, H)
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # FP16
    right_gate = torch.sigmoid(right_gate_raw)        # FP16
    out_gate = torch.sigmoid(out_gate_raw)            # FP16

    left = left_raw * left_gate        # (B,N,N,H) FP16
    right = right_raw * right_gate      # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication across summed dimension k
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) layout expected by the kernel:
    #   left : (i,k) -> (head, i, k)
    #   right: (k,j) -> (head, k, j) (already transposed)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Triton batched matmul
    out_perm = triton_batched_matmul(
        left_perm, right_perm,
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=128,
    )  # (B, H, N, N) FP16

    # Back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) over hidden dimension
    # --------------------------------------------------------------
    out_fp32 = out.float()
    out_flat = out_fp32.view(-1, hidden_dim).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    out_norm_flat = F.layer_norm(
        out_flat,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # FP32
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim C
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                     # (B,N,N,H) FP32

    # Final linear projection (FP16 weight, bias‑free)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()                               # (B,N,N,H) FP16
    final_fp16 = F.linear(gated_h, to_out_weight)        # (B,N,N,C) FP16
    final_fp32 = final_fp16.float()                       # back to FP32

    return final_fp32
