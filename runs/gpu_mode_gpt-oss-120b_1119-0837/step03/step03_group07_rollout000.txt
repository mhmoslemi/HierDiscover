"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1️⃣ 2‑pass LayerNorm (mean/var in FP32, output FP32) is implemented in Triton.
2️⃣  The five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
    fused into one FP16 GEMM by concatenating the weight matrices.
3️⃣  Mask (if present) is applied to the left/right projections *before* gating.
4️⃣  Gating uses element‑wise sigmoid in FP16.
5️⃣  The cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed with a single batched FP16 matrix multiplication
    (`torch.bmm`), which leverages cuBLAS/Tensor‑cores and is far faster than a
    hand‑written Triton kernel for this workload.
6️⃣  A second LayerNorm (again in Triton) normalises the intermediate result.
7️⃣  The output gate (sigmoid) and the final projection back to the model
    dimension are applied in FP16, with the final output cast to FP32.

All heavy arithmetic runs in FP16 (tensor‑core friendly) while the LayerNorms
run in FP32 for numerical stability, satisfying the requirement that at least
part of the computation is performed in a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (FP32 mean/var, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [M, C] input / output
    weight_ptr, bias_ptr,       # [C] affine parameters
    M, C, eps,                  # dimensions
    stride_xm, stride_xc,       # stride of input
    stride_ym, stride_yc,       # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range handled by this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: rowwise mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_C: int = 64,
) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor ``x`` (shape M×C) using Triton.
    Returns a tensor of the same shape and dtype (both FP32).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Entry point – forward pass of TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors with model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # --------------------------------------------------------------
    # 0️⃣  Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # Should be torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32 – Triton)
    # --------------------------------------------------------------
    # Collapse batch*seq*seq into the row dimension for the kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                 # (M, C) float32
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)

    x_norm_fp32 = triton_layernorm(
        x_flat, norm_weight, norm_bias, eps=1e-5,
        BLOCK_M=128, BLOCK_C=64,
    )
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim)   # (B, N, N, C) float32

    # --------------------------------------------------------------
    # 2️⃣  Fused projection for left/right and gates (single FP16 GEMM)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()                               # (B, N, N, C) fp16
    x_norm_h_flat = x_norm_h.view(M, dim)                  # (M, C) fp16

    # Concatenate the five weight matrices: (5*H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5*H, C)

    # Linear: no bias
    proj_out = F.linear(x_norm_h_flat, combined_weight)    # (M, 5*H) fp16

    # --------------------------------------------------------------
    # 3️⃣  Split projections
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    # reshape to (B, N, N, H)
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣  Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣  Gating (sigmoid) and apply to projected values
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B, N, N, H) fp16
    right_gate = torch.sigmoid(right_gate_raw) # (B, N, N, H) fp16
    out_gate  = torch.sigmoid(out_gate_raw)    # (B, N, N, H) fp16

    left = left_raw * left_gate
    right = right_raw * right_gate

    # --------------------------------------------------------------
    # 6️⃣  Cubic contraction via batched matrix multiplication (torch.bmm)
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------
    # Transform to (B, H, N, N) where left = (i,k) and right = (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, N)

    # Collapse batch and head dimensions for torch.bmm
    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, seq_len, seq_len)    # (BH, N, K)
    right_mat = right_perm.view(BH, seq_len, seq_len)  # (BH, K, N)

    # Batched matrix multiplication in FP16 (cuBLAS/Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat)          # (BH, N, N) fp16

    # Restore shape (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣  Output LayerNorm (FP32 – Triton)
    # --------------------------------------------------------------
    out_fp32 = out.float()                                  # cast to FP32 for LN
    out_flat = out_fp32.view(M, hidden_dim)                  # (M, H) FP32
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_C=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣  Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                     # (B,N,N,H) FP32

    # Final linear projection to original dimension (no bias)
    to_out_weight_fp16 = weights["to_out.weight"].to(device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_weight_fp16)  # (B,N,N,C) fp16
    final_fp32 = final_fp16.float()                     # (B,N,N,C) FP32

    return final_fp32
