```python
"""
TriMul (outgoing) Triton implementation

The core of the module computes, for each batch `b` and hidden channel `h`,
the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the linearly projected and gated pairwise
representations. The reduction over `k` is realized as a batched GEMM
`C = A @ Bᵀ` for a total batch size `C_total = B * hidden_dim`.  Each
stacked GEMM is performed by a custom Triton kernel that works in FP16
with FP32 accumulation and stores the result back in FP16.  All other
operations (LayerNorm, linear projections, sigmoid gates) are carried
out with torch.nn.functional for simplicity.

The kernel uses a 2‑D tiling (BLOCK_M × BLOCK_N) and a 1‑D tiling over
the reduction dimension (`BLOCK_K`).  Double‑buffering (prefetching the
next K‑tile while the current one is accumulated) reduces memory
latency.  Block sizes are chosen to fit the H100 tensor‑core
requirements while keeping register pressure modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,                     # fp16*
    a_ptr,                       # fp16* (C, M, K)  – left tensor
    b_ptr,                       # fp16* (C, K, N)  – right tensor (logically transposed)
    M, N, K,                     # matrix dims (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,
    stride_a_m, stride_a_k, stride_a_c,
    stride_b_k, stride_b_n, stride_b_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_c = tl.program_id(2)   # channel = batch * hidden

    # compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- pre‑load first K‑tile -----------------------------
    k0 = 0
    a_tile = a_ptr + pid_c * stride_a_c \
             + offs_m[:, None] * stride_a_m \
             + (k0 + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
    b_tile = b_ptr + pid_c * stride_b_c \
             + (k0 + tl.arange(0, BLOCK_K))[:, None] * stride_b_k \
             + offs_n[None, :] * stride_b_n

    a = tl.load(a_tile,
                mask=mask_m[:, None] & ((k0 + tl.arange(0, BLOCK_K)) < K),
                other=0.0)                 # fp16
    b = tl.load(b_tile,
                mask=((k0 + tl.arange(0, BLOCK_K))[:, None] < K) & mask_n[None, :],
                other=0.0)                 # fp16

    # number of K‑tiles
    num_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------- main K‑loop with 1‑stage prefetch -------------
    for _ in range(num_tiles):
        # prefetch next tile (k + BLOCK_K) unless we are at the last step
        k_next = tl.where(_ < num_tiles - 1, BLOCK_K * (_ + 1), K)   # dummy index after the end
        has_next = k_next < K

        a_next_tile = a_ptr + pid_c * stride_a_c \
                      + offs_m[:, None] * stride_a_m \
                      + (k_next + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        b_next_tile = b_ptr + pid_c * stride_b_c \
                      + (k_next + tl.arange(0, BLOCK_K))[:, None] * stride_b_k \
                      + offs_n[None, :] * stride_b_n

        # accumulate current tile
        acc += tl.dot(a, b)

        # load next tile (or zeros if we are past K)
        a = tl.load(a_next_tile,
                    mask=mask_m[:, None] & ((k_next + tl.arange(0, BLOCK_K)) < K) & has_next,
                    other=0.0)
        b = tl.load(b_next_tile,
                    mask=((k_next + tl.arange(0, BLOCK_K))[:, None] < K) & mask_n[None, :] & has_next,
                    other=0.0)

    # ------------------- write result ------------------------------------
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 variant).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] fp32

    # Convert to fp16 for the heavy lifting
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) – produce hidden_dim features
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 3) Gating networks (sigmoid) – fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4) Apply gates and optional mask
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Rearrange tensors so that the K‑dimension (the one summed over)
    #    becomes contiguous.  We permute to (B, H, N, N) and then collapse
    #    B and H into a single GEMM‑batch dimension C = B * H.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)        # (C, M, K) with K = N
    right_flat = right.reshape(C_total, N, N)      # (C, K, N) – will be interpreted as (K,N)

    # Output storage (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Triton batched GEMM launch
    # Choose block sizes that give good occupancy on H100.
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 16

    # Strides in elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    # For the right tensor we logically treat it as (C, K, N).
    # Its stored shape is (C, N, N); therefore the stride that
    # corresponds to the reduction dimension K is the innermost stride.
    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # innermost dimension → K
    stride_b_n = right_flat.stride(1)   # middle dimension → N

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_STAGES=2,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) – still fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Post‑processing: LayerNorm → out‑gate → final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate   # already fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # back to [B, N, N, C] (fp16)

    # -----------------------------------------------------------------
    # 9) Cast back to float32 as the public API expects float32
    return out.to(torch.float32)

"""
TriMul (outgoing) Triton implementation.

The core of the module computes, for each batch and hidden channel,
the contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the linearly projected and gated pairwise
representations.  The reduction over `k` is expressed as a batched GEMM
C = A @ Bᵀ with a batch size C = B * hidden_dim.  The heavy part is
implemented in a custom Triton kernel that works in FP16 (FP32
accumulation) and stores the result back in FP16.  All surrounding
operations (LayerNorm, linear projections, sigmoid gates) are performed
with `torch.nn.functional` for simplicity.

The kernel is tiled in two dimensions (BLOCK_M × BLOCK_N) and loops over
the reduction dimension with tiles of size BLOCK_K.  The tile sizes are
chosen to keep register pressure modest while exposing enough work per
thread‑block for the H100 tensor‑cores.  A modest increase in the number
of warps per block further improves occupancy on larger sequence lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,                     # fp16*
    a_ptr,                       # fp16* (C, M, K) – left tensor
    b_ptr,                       # fp16* (C, K, N) – right tensor (logically transposed)
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,
    stride_a_m, stride_a_k, stride_a_c,
    stride_b_k, stride_b_n, stride_b_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,      # kept for possible future staging
):
    """Batched GEMM for the TriMul contraction.
    Each program instance computes a tile of size (BLOCK_M, BLOCK_N) for a
    single channel `c = batch * hidden`.  The inputs are stored as
    (C, M, K) and (C, K, N).  The computation uses fp16 inputs, fp32
    accumulation, and stores the result in fp16.
    """
    pid_m = tl.program_id(0)   # row‑tile index
    pid_n = tl.program_id(1)   # col‑tile index
    pid_c = tl.program_id(2)   # channel index (batch * hidden)

    # --------------------------------------------------------------
    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current sub‑tiles of A and B
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k           # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n           # (K, N)

        # Load with masking (out‑of‑bounds positions are zero)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                                 # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                                 # fp16

        # GEMM accumulate: fp16 * fp16 → fp32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the tile back (convert to fp16 on store)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask"      – bool, True → ignore mask

            Expected weight keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numeric stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                       # [B, N, N, C] fp32
    x = x.to(torch.float16)                # down‑cast for the rest

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)        # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)      # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4) Apply gates and optional mask
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Rearrange tensors so that the reduction dimension `k` becomes
    #    contiguous.  We permute to (B, H, N, N) and then collapse the first
    #    two axes into a single GEMM batch dimension C = B * H.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)        # (C, M, K)   – K = N
    right_flat = right.reshape(C_total, N, N)      # (C, K, N)   – will be treated as (K,N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Triton GEMM launch – tile sizes tuned for H100
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 16

    # Strides in elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    # Right tensor is stored as (C, N, N); we interpret the innermost
    # dimension as K and the middle one as N.
    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # innermost → K
    stride_b_n = right_flat.stride(1)   # middle   → N

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_STAGES=2,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) – still fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Post‑processing: LayerNorm → out‑gate → final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate                       # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)               # back to original dim (fp16)

    # -----------------------------------------------------------------
    # 9) Cast to float32 for the API contract
    return out.to(torch.float32)
