"""
TriMul (outgoing) – fast mixed‑precision Triton implementation.

Algorithm
---------
1. **Layer‑norm** on the input (float32) → cast to fp16.
2. **Linear projections** (`left_proj`, `right_proj`) and **gates**
   (`left_gate`, `right_gate`, `out_gate`) are computed with fp16
   weights.  Gating is applied element‑wise; an optional binary mask
   is broadcast over the hidden dimension and applied after the gates
   (order is irrelevant because it is just a 0/1 multiplier).
3. The core “triangle” multiplication
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]` is a batched GEMM.
   We pack the batch and hidden dimensions (`C = B * hidden_dim`) and
   launch a Triton kernel that computes, for each channel `c`,
   `out_c[M,N] = left_c[M,K] @ right_c[K,N]` with
   `M = N = seq_len`.  Inputs are fp16, accumulation is fp32, and the
   result is stored as fp16.
4. The GEMM output is reshaped back to `(B, N, N, hidden_dim)`.
5. A second **Layer‑norm**, multiplication by the *out‑gate* and a
   final linear projection back to the original channel dimension are
   performed (all in fp16).  The result is cast to float32 for the
   caller.

Only the heavy triangular multiplication is performed in Triton; the
remaining ops use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                      # fp16*
    a_ptr,                        # fp16*  (C, M, K) – left tensor
    b_ptr,                        # fp16*  (C, K, N) – right tensor (logically transposed)
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,        # left tensor strides
    stride_b_c, stride_b_k, stride_b_n,        # right tensor strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each channel `c` (packed batch*hidden) compute
      out_c[M,N] = left_c[M,K] @ right_c[K,N].
    Inputs are fp16, accumulation in fp32, result stored as fp16.
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_c = tl.program_id(2)  # channel (batch*hidden)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k   # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n   # (K, N)

        # Load with zero‑padding for OOB elements
        a = tl.load(
            a_tile,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16
        b = tl.load(
            b_tile,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Accumulate: fp16 * fp16 → fp32
        acc += tl.dot(a, b)

    # Write back tile (cast to fp16 on store)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(
        out_tile,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) or None
            - weights: dict with the model parameters
            - config: dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability) → fp16 for the rest
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,C] float32
    x = x.to(torch.float16)  # now fp16

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B,N,N,H] fp16
    right = F.linear(x, right_proj_w)  # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 4️⃣ Apply gates (and optional mask)
    left.mul_(left_gate)      # in‑place: left = left * left_gate
    right.mul_(right_gate)    # in‑place: right = right * right_gate

    if not nomask and mask is not None:
        # mask: [B,N,N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left.mul_(mask_h)
        right.mul_(mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Prepare tensors for the Triton GEMM.
    #      Pack (batch, hidden) → a single GEMM batch dimension C = B*H.
    B, N, _, _ = left.shape
    # left: (B, H, i, k)   → (C, M, K)
    left = left.permute(0, 3, 1, 2).contiguous()
    left_flat = left.reshape(B * hidden_dim, N, N)

    # right: (B, H, k, j)  (note the transpose of the last two axes)
    right = right.permute(0, 3, 2, 1).contiguous()
    right_flat = right.reshape(B * hidden_dim, N, N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6️⃣ Triton batched GEMM launch
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides: number of elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # right is stored as (C, K, N) after the permutation above:
    stride_b_k = right_flat.stride(1)   # K‑step
    stride_b_n = right_flat.stride(2)   # N‑step

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        B * hidden_dim,                 # channel dimension C
    )

    # Use more warps when the problem size is larger (helps occupancy)
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N,
        N,
        N,
        stride_out_c,
        stride_out_m,
        stride_out_n,
        stride_a_c,
        stride_a_m,
        stride_a_k,
        stride_b_c,
        stride_b_k,
        stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Reshape kernel output back to (B, N, N, H) (still fp16)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8️⃣ Post‑processing: second LayerNorm → out‑gate → final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply the out‑gate (already fp16)
    out = out * out_gate

    # Final projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the expected output type
    return out.to(torch.float32)
