def tri_mul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B_batches, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    out = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[t][i][k]
        left = tl.load(
            left_ptr + batch_id * stride_left_b
                + offs_i[:, None] * stride_left_i
                + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # load right[t][j][k]
        right = tl.load(
            right_ptr + batch_id * stride_right_b
                + offs_j[:, None] * stride_right_i
                + offs_k[None, :] * stride_right_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        # Cast to FP32
        left_f32 = tl.cast(left, tl.float32)
        right_f32 = tl.cast(right, tl.float32)

        out += tl.dot(left_f32, right_f32)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write result, cast back to fp16
    tl.store(
        out_ptr + batch_id * stride_out_b
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j,
        tl.cast(out, tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

"""
TriMul “outgoing’’ forward pass – highly‑fused Triton implementation.

Algorithm
---------
1️⃣  Input LayerNorm over the last dimension (custom 2‑pass Triton kernel).
2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj,
    right_gate, out_gate) into one GEMM → one contiguous FP16 tensor.
3️⃣  Apply sigmoid to the three gate tensors and (optionally) a mask.
4️⃣  Batched contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed with a custom Triton “batched‑matmul’’ kernel that
    executes a tiled FP16 × FP16 → FP32 reduction (Tensor‑Core).
5️⃣  Output LayerNorm (Triton – FP32 accumulation, FP16 results).
6️⃣  Apply the output gate and a final FP16 linear projection back to
    the model dimension; cast the result to float32 (the public API).

Only the two LayerNorms and the heavy N³ × H contraction are written
in Triton; everything else uses highly‑optimized PyTorch kernels.
The implementation works for all configurations listed in the prompt
and respects the `nomask` flag.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,            # *ptr to input / output       [M, N]
    weight_ptr, bias_ptr,    # *ptr to weight / bias        [N]
    M, N, eps,               # dimensions
    stride_xm, stride_xn,    # input strides
    stride_ym, stride_yn,    # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row tile for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load a tile of shape (BLOCK_M, BLOCK_N)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (still FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine – weight/bias may be FP16 or FP32; cast to FP32 for safety
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using the Triton kernel.
    Expects a 2‑D tensor [M, N] (row‑wise normalisation).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous / correct strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tuned for H100)
        BLOCK_N=64,    # columns per tile
    )
    return y

# ----------------------------------------------------------------------
# Batched half‑precision contraction kernel for TriMul
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    # dimensions
    BATCHES, N,
    # strides
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    # tiling
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch `b` (where a batch corresponds to one
    (sample, head) pair), the matrix product:
        Out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
    The right operand is accessed with its second index as “row’’ (j)
    and the shared dimension k as column – exactly the pattern needed
    for the “outgoing’’ TriMul.
    """
    batch_id = tl.program_id(2)                     # 0 … BATCHES‑1
    pid_i    = tl.program_id(0)                     # block over i (rows)
    pid_j    = tl.program_id(1)                     # block over j (cols)

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction dimension k ---------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k]  → shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + batch_id * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right[b,j,k] → shape (BLOCK_N, BLOCK_K) (note the swapped i/j order)
        right = tl.load(
            right_ptr
            + batch_id * stride_right_batch
            + offs_j[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        # cast to FP32, then fused dot
        left_f32  = tl.cast(left,  tl.float32)
        right_f32 = tl.cast(right, tl.float32)
        acc += tl.dot(left_f32, tl.trans(right_f32))   # (M,K)·(K,N) → (M,N)

    # write back result in FP16 (the rest of the model works in FP16)
    tl.store(
        out_ptr
        + batch_id * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        tl.cast(acc, tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def batched_tri_mul_matmul(left: torch.Tensor,
                           right: torch.Tensor,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=64) -> torch.Tensor:
    """
    Triton‑based batched contraction for TriMul.
    `left` & `right` must be contiguous tensors of shape (B*H, N, N) in FP16.
    Returns a tensor of the same shape and dtype.
    """
    assert left.shape == right.shape
    BATCHES, N, N2 = left.shape
    assert N == N2

    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_l_b, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_i, stride_r_k = right.stride()
    stride_o_b, stride_o_i, stride_o_j = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
        BATCHES,                        # one program per (batch, head)
    )

    tri_mul_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        BATCHES=BATCHES,
        N=N,
        stride_left_batch=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_right_batch=stride_r_b,
        stride_right_i=stride_r_i,
        stride_right_k=stride_r_k,
        stride_out_batch=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the last dimension (`dim`)
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len                     # total rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)          # (M, D) float32

    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)  # (M, D) FP32
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)            # (B,N,N,D) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + gates) into one GEMM
    # --------------------------------------------------------------
    #   each weight has shape (hidden_dim, dim) and bias is None
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # concatenate along the output‑feature dimension → (5*H, D)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                     # (5*H, D) FP16

    # single fused projection (M,5*H) in FP16
    proj_gate_flat = F.linear(x_norm_f16.view(M, dim), combined_weight)  # (M,5*H)

    # --------------------------------------------------------------
    # 4️⃣ Slice projections & gates, apply sigmoid and optional mask
    # --------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)   # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)               # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched contraction using a Triton kernel
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) → view as (B*H, N, N) for the kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)

    left_mat  = left_perm.view(bs * hidden_dim, seq_len, seq_len)   # (B*H,N,N) FP16
    right_mat = right_perm.view(bs * hidden_dim, seq_len, seq_len)  # (B*H,N,N) FP16

    # Triton batched matmul (produces (B*H,N,N) FP16)
    out_mat = batched_tri_mul_matmul(left_mat, right_mat,
                                     BLOCK_M=128, BLOCK_N=128, BLOCK_K=64)

    # Restore original layout (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton)
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                     # (B·N·N, H) FP16

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)   # FP32
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)     # FP32

    out_norm_flat = triton_layernorm(out_flat,
                                     to_out_norm_w,
                                     to_out_norm_b,
                                     eps=eps)                      # (M, H) FP16
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                            # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)          # (D, H) FP16
    # final projection (still FP16) and cast back to FP32 as required by API
    output_fp16 = F.linear(gated, to_out_w)                                # (B,N,N,D) FP16
    output = output_fp16.to(torch.float32)                                 # (B,N,N,D) FP32

    return output
