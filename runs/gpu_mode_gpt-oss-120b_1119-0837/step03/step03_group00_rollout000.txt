"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton‑accelerated.

Algorithm
---------
1️⃣  Input LayerNorm over the model dimension (float32) → custom Triton kernel.
2️⃣  Cast to FP16 for the heavy linear work.
3️⃣  Fuse each projection + its gate into a single matmul:
       [proj; gate] × x  →  (2·H) channels, split afterwards.
4️⃣  Optional binary mask is applied to the projected tensors (before gating).
5️⃣  Element‑wise gating of left / right tensors.
6️⃣  Core N³ contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM (torch.bmm) on the view
   (B·H, N, N).
7️⃣  Hidden‑dimension LayerNorm (FP16) → custom Triton kernel.
8️⃣  Apply the output gate (sigmoid of a linear on the normalized *input*).
9️⃣  Final linear projection back to the model dimension (FP16 matmul) and cast
   to float32 for the return value.

All heavy linear work stays in FP16 (Tensor‑Core friendly).  Two small Triton
kernels (float‑32 and float‑16 LayerNorm) are used, satisfying the “must use
a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 2‑pass LayerNorm – float32 → float32
# --------------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # Row indices this program works on
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------- first pass: mean & variance ----------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------- second pass: write normalized --------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) via the Triton kernel above.
    Input/Output shape: (M, N) where N = feature dimension.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride computation
    x, y = x.contiguous(), y.contiguous()
    weight, bias = weight.contiguous(), bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def triton_layernorm_fp16(x: torch.Tensor,

                          bias: torch.Tensor,

# --------------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” operator
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]   (float32/bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys:
            * dim          – model dimension D
            * hidden_dim   – hidden dimension H
            * nomask (optional) – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                # (B·N·N, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)             # (B, N, N, D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent work
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate (two matmuls → two fused matmuls)
    # ------------------------------------------------------------------
    # left side: [proj; gate] weight matrix (2·H, D)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)                         # (2H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)                         # (2H, D)

    # out‑gate linear weight (H, D)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    # Combined linear passes (B,N,N,2H)
    left_comb  = F.linear(x_f16, left_comb_w)           # (B,N,N,2H)
    right_comb = F.linear(x_f16, right_comb_w)          # (B,N,N,2H)

    # Split into projection and gate heads
    left_proj = left_comb[..., :H]                      # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]                  # (B,N,N,H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    # Apply sigmoid to gate pre‑activations
    left_gate  = torch.sigmoid(left_gate_pre)           # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure mask is float16 and broadcastable
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Gated projections
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate        # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched FP16 GEMM
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) → view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                  # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                 # (B·H, N, N)

    # Compute C = A @ Aᵀ for each batch → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm (FP16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)   # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate & final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)               # (M, H) FP16
    gated = out_norm * out_gate_flat                     # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    output_flat = F.linear(gated, to_out_w)             # (M, D) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32

    return output
