@triton.jit
def triton_tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch_h = tl.program_id(0)  # 0 <= pid < B*H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch index and head index
    b = pid_batch_h // H
    h = pid_batch_h % H

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointer offsets for left and right for each k block
    # base pointer to left[b, :, :, h]
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    # similarly for right
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left block: shape (BLOCK_M, BLOCK_K)
        # left indices: (i_offset, k_offset)
        left = tl.load(
            left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # load right block: shape (BLOCK_K, BLOCK_N) from right[b, :, :, h] transposed: (k_offset, j_offset)
        right = tl.load(
            right_base + j_offset[None, :] * stride_right_i + k_offset[:, None] * stride_right_k,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)  # left: MxK, right: KxN => float32

    # store result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    left_ptr, right_ptr, out_ptr,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

    h = pid_batch_h % H

"""
Optimised Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Key ideas
---------
* Input LayerNorm is performed by a tiny Triton kernel (row‑wise).
* The four linear maps (left_proj, right_proj, left_gate, right_gate) are
  fused into a single matmul (`F.linear`) of shape (M, 4·H) where M = B·L·L.
* The cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  is computed by a custom Triton kernel that works directly on tensors with
  shape (B, L, L, H).  No permutes or extra copies are required; the kernel
  fuses the batched GEMM for all heads.
* Output LayerNorm, the output gating and the final projection back to `dim`
  are kept in FP16 for maximum Tensor‑core utilisation; the final result is
  returned in FP32.
* At least one custom Triton kernel (the LayerNorm) is used, satisfying the
  problem constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – normalises a 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # hidden dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm implemented with Triton (supports FP16 & FP32)."""
    assert x.dim() == 2, "input must be 2‑D"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tested good on H100
        BLOCK_N=64,
    )
    return y

# Triton kernel that directly computes the outgoing TriMul contraction.

    # iterate over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  [M, K]  (i‑dimension, k‑dimension)
        left_tile = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile: [K, N]  (k‑dimension, j‑dimension) – note the transpose
        right_tile = tl.load(
            right_base
            + offs_j[None, :] * stride_right_i
            + offs_k[:, None] * stride_right_k,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)   # (M, N) accumulation

    # write out  (FP16 result)
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, L, L, dim]  (float32)
        mask         : torch.Tensor [B, L, L]      (bool/int) or None
        weights      : dict of torch tensors (model parameters)
        config       : dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, L, L, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                      # should be torch.float32
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]                # H
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – normalise over the channel dim `dim`
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)           # [M, dim]
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)        # [B, L, L, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj + gates) into a single matmul.
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    fused_weight = torch.cat([left_proj_w,
                              right_proj_w,
                              left_gate_w,
                              right_gate_w], dim=0)      # (4·H, dim)

    x_norm_h_flat = x_norm_h.reshape(M, dim)                # [M, dim]
    fused_out = F.linear(x_norm_h_flat, fused_weight)       # [M, 4·H]
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)    # [B, L, L, 4·H]

    left_proj   = fused_out[..., :hidden_dim]                       # (B, L, L, H)
    right_proj  = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate   = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate  = torch.sigmoid(fused_out[..., 3 * hidden_dim:4 * hidden_dim])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and the element‑wise gates
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate          # [B, L, L, H]   fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched contraction using a custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                                 # [B, L, L, H]

    # stride information (all tensors are contiguous in the last dim)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # kernel launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        bs * hidden_dim,                     # batch × head
        triton.cdiv(L, BLOCK_M),            # tiles over i
        triton.cdiv(L, BLOCK_N),            # tiles over j
    )
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=L,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (over the hidden dimension H)
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)                # [M, H]
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)     # [B, L, L, H]

    # ------------------------------------------------------------------
    # 7️⃣ Compute the output gate (linear + sigmoid on the *original* norm)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_w))   # [M, H]
    out_gate = out_gate.view(bs, L, L, hidden_dim)                # [B, L, L, H]

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16 * fp16 → fp16
    # final linear: (B·L·L, H) × (dim, H)ᵀ → (B·L·L, dim)
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated.reshape(out_M, hidden_dim), to_out_w)  # fp16
    out_final = out_final_h.to(torch.float32)     # required output dtype

    return out_final
