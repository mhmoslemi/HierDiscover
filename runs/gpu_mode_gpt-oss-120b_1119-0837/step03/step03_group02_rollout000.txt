def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast input and parameters to fp16 for main compute
    x_h = input_tensor.to(torch.float16)

    # Input layernorm
    norm_weight = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_bias = weights["norm.bias"].to(device, torch.float16).contiguous()
    M = bs * L * L
    x_h_flat = x_h.reshape(M, dim)
    x_norm_flat = triton_layernorm(x_h_flat, norm_weight, norm_bias)
    x_norm = x_norm_flat.view(bs, L, L, dim)   # fp16

    # Fuse five linear maps: left_proj, right_proj, left_gate, right_gate, out_gate
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    fused_weight = torch.cat([left_proj_w,
                              right_proj_w,
                              left_gate_w,
                              right_gate_w,
                              out_gate_w], dim=0)   # shape (5*H, dim)

    # Linear projection for all five outputs
    fused_all = F.linear(x_norm_flat, fused_weight)   # (M,5*H) fp16
    fused_all = fused_all.view(bs, L, L, 5*hidden_dim)

    # Split
    left_proj = fused_all[..., :hidden_dim]                          # (B,L,L, H)
    right_proj = fused_all[..., hidden_dim:2*hidden_dim]
    left_gate = torch.sigmoid(fused_all[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(fused_all[..., 3*hidden_dim:4*hidden_dim])
    out_gate = torch.sigmoid(fused_all[..., 4*hidden_dim:5*hidden_dim])

    # apply mask if needed and compute left/right
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B,L,L,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Contract left/right via batched GEMM
    # Permute to bring hidden dimension to batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, L, L)   # (B*H,N,N), k/j swapped
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H,N,N)

    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # Output layernorm
    out_weight = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_bias = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_flat = out.reshape(bs * L * L, hidden_dim)   # (M, H)
    out_norm_flat = triton_layernorm(out_flat, out_weight, out_bias)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # fp16

    # Gating and final projection
    gated = out_norm * out_gate   # fp16

    # Final linear projection to dim
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    gated_flat = gated.reshape(M, hidden_dim)   # (M, H)
    out_final_flat = F.linear(gated_flat, to_out_w)   # (M, dim) fp16
    out_final = out_final_flat.view(bs, L, L, dim).to(torch.float32)

    return out_final

"""
Optimised Triton‑accelerated implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1.  Input LayerNorm (row‑wise) is performed by a tiny Triton kernel.
    The kernel works on fp16 data (mean/variance accumulated in fp32).

2.  Cast the normalised tensor to fp16 and fuse **five** linear maps
    (left_proj, right_proj, left_gate, right_gate, out_gate) into a
    single matrix multiplication:
        fused = x @ W_fused.T   where   W_fused = [W_lp; W_rp; W_lg; W_rg; W_og]
    This removes two separate matmuls and halves the amount of memory
    traffic for the large (B·L·L, dim) tensor.

3.  Split the fused result, apply the sigmoid gates and (optional) mask,
    and form the left/right tensors.

4.  The expensive contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is rewritten as a batched GEMM:
        left  → (B·H, L, L)   (permute H into batch dim)
        right → (B·H, L, L)   (permute and swap the last two axes)
        out   = torch.bmm(left, right)   # FP16 Tensor‑core GEMM
   This leverages cuBLAS‑backed batched GEMM which is already near‑optimal
   on H100.

5.  Output LayerNorm (again Triton) is applied on the hidden dimension.

6.  The gated output (out_norm * out_gate) is finally projected back to
    the model dimension with a single FP16 linear layer.

All heavy compute is done in fp16 (Tensor‑core friendly); only the two
LayerNorms use fp32 accumulation for numerical stability.  The implementation
covers all test configurations (B∈{1,2}, L up to 1024, dim∈{128,384},
hidden_dim=128) and respects an optional attention mask.

The required Triton kernel (`layernorm_kernel`) is kept deliberately simple
yet fast for the small feature dimensions (≤384) encountered here.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm: normalises a 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *ptr to input  [M, N]
    y_ptr,               # *ptr to output [M, N]
    weight_ptr,          # *ptr to weight [N]
    bias_ptr,            # *ptr to bias   [N]
    M,                   # total rows
    N,                   # feature dimension
    eps,                 # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (accumulate in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load as the original dtype (fp16 or fp32) and cast to fp32 for reduction
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented with Triton.

    Parameters
    ----------
    x : torch.Tensor
        Input tensor of shape (M, N). Can be fp16 or fp32.
    weight, bias : torch.Tensor
        LayerNorm parameters of shape (N,). Must have the same dtype as `x`.
    eps : float
        Numerical epsilon.

    Returns
    -------
    torch.Tensor
        Output tensor of shape (M, N) with the same dtype as `x`.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for H100
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D] (float32)
        - mask         : torch.Tensor   [B, L, L] (optional)
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``.
                        May contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Cast input to fp16 and apply input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)
    norm_weight = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_bias   = weights["norm.bias"].to(device, torch.float16).contiguous()
    M = bs * L * L
    x_fp16_flat = x_fp16.reshape(M, dim)                     # [M, D]
    x_norm_flat = triton_layernorm(x_fp16_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)                 # [B, L, L, D] (fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right proj & gates + out_gate)
    # ------------------------------------------------------------------
    # Gather and concatenate the five weight matrices (all bias‑free)
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    #   Shape   (5·H, D)
    fused_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)

    # One big linear projection
    fused_out = F.linear(x_norm_flat, fused_weight)          # [M, 5·H] (fp16)
    fused_out = fused_out.view(bs, L, L, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor and apply gates / mask
    # ------------------------------------------------------------------
    left_proj   = fused_out[..., :hidden_dim]                     # (B, L, L, H)
    right_proj  = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate   = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate  = torch.sigmoid(fused_out[..., 3 * hidden_dim:4 * hidden_dim])
    out_gate    = torch.sigmoid(fused_out[..., 4 * hidden_dim:5 * hidden_dim])

    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B, L, L, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM:   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Reshape so that (B·H) is the batch dimension for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    # right needs the last two axes swapped before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, L, L)

    out_mat = torch.bmm(left_mat, right_mat)                # (B·H, L, L)

    # Restore the original 4‑D layout
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_bias   = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_flat = out.reshape(M, hidden_dim)                    # [M, H]
    out_norm_flat = triton_layernorm(out_flat, out_norm_weight, out_norm_bias, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)      # (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ Gating & final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (fp16)

    to_out_weight = weights["to_out.weight"].to(device, torch.float16).contiguous()
    gated_flat = gated.reshape(M, hidden_dim)                 # [M, H]
    out_final_flat = F.linear(gated_flat, to_out_weight)     # [M, D] (fp16)
    out_final = out_final_flat.view(bs, L, L, dim).to(torch.float32)

    return out_final
