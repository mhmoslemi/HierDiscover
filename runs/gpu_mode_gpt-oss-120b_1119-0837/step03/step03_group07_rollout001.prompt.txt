```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward implementation.

Algorithm
---------
1️⃣ Input LayerNorm (per‑row FP32) – fused two‑pass Triton kernel.
2️⃣ A single half‑precision linear (F.linear) produces the five projections
   (left, right, left‑gate, right‑gate, out‑gate) by concatenating the
   weight matrices.
3️⃣ Mask (if present) is applied to left/right before gating.
4️⃣ Gating (sigmoid) is done element‑wise in FP16.
5️⃣ Batched matrix multiplication across the summed dimension *k* is performed
   with a custom Triton kernel: for each hidden channel we compute
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.  The kernel fuses all
   batch‑×‑head matrices in a single launch and accumulates in FP32 for
   numerical safety.
6️⃣ Output LayerNorm (FP32) – same Triton kernel as step 1.
7️⃣ Output gate (sigmoid) is applied, then a final half‑precision linear
   projects back to the model dimension.
All heavy compute runs in FP16 (tensor‑core friendly) while the two
LayerNorms run in FP32 for stability.  The implementation satisfies the
requirement of “at least part of the operations in a Triton kernel”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                      # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,              # *ptr to weight / bias   [N]
    M, N, eps,                         # dimensions
    stride_xm, stride_xn,              # stride of input
    stride_ym, stride_yn,              # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 64) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor ``x`` (shape M×N) using
    the Triton kernel above.  Returns a tensor of the same shape and dtype.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Batched half‑precision matrix multiplication kernel for the
# TriMul “outgoing” contraction.
#   C[b] = A[b] @ B[b]    where A,B,C : (B·H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                         # here M = N = K = seq_len
    stride_ah, stride_am, stride_ak, # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn, # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn, # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # flattened (batch * hidden) index
    pid_m = tl.program_id(0)       # tile row index
    pid_n = tl.program_id(1)       # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a = tl.load(
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Half‑precision dot, accumulate in FP32
        acc += tl.dot(a, b)

    # Store result (cast back to FP16 for downstream use)
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M: int = 64,
                         BLOCK_N: int = 64,
                         BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication for tensors of shape (B, H, N, N) (half‑precision).
    Returns C of the same shape and dtype.
    """
    # A,B are (B, H, N, N) – already contiguous and half‑precision
    B_dim, H_dim, N, _ = A.shape
    BH = B_dim * H_dim

    A_flat = A.view(BH, N, N)
    B_flat = B.view(BH, N, N)
    C_flat = torch.empty_like(A_flat)

    stride_ah, stride_am, stride_ak = A_flat.stride()
    stride_bh, stride_bk, stride_bn = B_flat.stride()
    stride_ch, stride_cm, stride_cn = C_flat.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over rows (i)
        triton.cdiv(N, BLOCK_N),   # tiles over cols (j)
        BH,                        # batch * hidden
    )

    triton_matmul_outgoing_kernel[grid](
        A_ptr=A_flat,
        B_ptr=B_flat,
        C_ptr=C_flat,
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    # reshape back to (B, H, N, N)
    return C_flat.view(B_dim, H_dim, N, N)


# ----------------------------------------------------------------------
# Entry point – forward pass of TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 – Triton)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                # (M, C) FP32
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp32 = triton_layernorm(x_flat, norm_weight, norm_bias)
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim)  # (B,N,N,C) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Combined projection for left/right + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()                                 # (B,N,N,C) FP16
    x_norm_h_flat = x_norm_h.view(M, dim)                     # (M, C) FP16

    # concatenate the five weight matrices (shape (5*H, C))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                         # (5*H, C)

    # fused linear – no bias
    proj_out = F.linear(x_norm_h_flat, combined_weight)       # (M, 5*H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split the concatenated projection
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)        # (B,N,N,H) FP16
    out_gate = torch.sigmoid(out_gate_raw)            # (B,N,N,H) FP16

    left = left_raw * left_gate
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication across the summed dimension (k)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) – left: (i,k) ; right: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # fused batched matmul via Triton
    out_perm = triton_batched_matmul(
        left_perm, right_perm, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64
    )  # (B, H, N, N) FP16

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()      # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 – Triton)
    # ------------------------------------------------------------------
    out_fp32 = out.float()                               # cast to FP32 for LN
    out_flat = out_fp32.view(M, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat, to_out_norm_weight, to_out_norm_bias, eps=1e-5
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dimension
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                     # (B,N,N,H) FP32

    # Final projection (half‑precision weight)
    to_out_weight_fp16 = weights["to_out.weight"].to(device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_weight_fp16)    # (B,N,N,C) FP16
    final_fp32 = final_fp16.float()                       # back to FP32

    return final_fp32
```
