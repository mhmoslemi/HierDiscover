"""
TriMul (outgoing) – fast Triton‑accelerated implementation.

Algorithm
---------
1.  Layer‐norm over the last channel (dim) in FP32 for stability.
2.  Cast to FP16 and fuse the five linear transforms
        left_proj, left_gate, right_proj, right_gate, out_gate
    into a single matmul (weight matrix concatenated along the output dim).
3.  Apply sigmoid to the three gate tensors (still in FP16).
4.  If a mask is present, a tiny Triton kernel fuses:
        * mask broadcast (B×N×N → B×N×N×H)
        * element‑wise multiplication by the left/right gates
        * mask‑application to the projected tensors.
   If no mask, a simple elementwise multiply is used.
5.  Heavy O(N³) contraction is performed with a strided‑batched GEMM:
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    (implemented as torch.bmm on tensors of shape (B·H, N, N)).
6.  Hidden‑dim layer‑norm, elementwise out‑gate, and final linear
    projection back to the original dimension D.
7.  Result is cast back to float32.

Only the mask‑gate fusion is written in Triton; all matmuls rely on
highly‑optimized cuBLAS/torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------#
# Triton kernel: fuse mask (if any) and the per‑channel gates onto the
# projected left/right tensors. Operates on flattened tensors of shape
# (B·N·N·H) and writes the result back in‑place.
# -----------------------------------------------------------------------------#
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # FP16 [B,N,N,H] – projected left (in‑place)
    right_ptr,              # FP16 [B,N,N,H] – projected right (in‑place)
    left_gate_ptr,          # FP16 [B,N,N,H] – sigmoid gate for left
    right_gate_ptr,         # FP16 [B,N,N,H] – sigmoid gate for right
    mask_ptr,               # FP16 [B,N,N]   – broadcast mask (flattened)
    total_elements: tl.int64,
    hidden_dim:    tl.int64,
    BLOCK_SIZE:    tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # ----- mask index (broadcast over hidden dimension) -----
    mask_index = offs // hidden_dim                # integer division
    mask_val = tl.load(mask_ptr + mask_index,
                       mask=mask,
                       other=tl.float16(0.0))

    # ----- load tensors -----
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    # ----- fused element‑wise operation -----
    left_out  = left  * left_g  * mask_val
    right_out = right * right_g * mask_val

    # ----- write back (in‑place) -----
    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights_dict: mapping of all module parameters (float32)
            - config_dict: {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1) Input layer‑norm (FP32 for numerical stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # 2) Cast to FP16 for the heavy work
    # --------------------------------------------------------------------- #
    x_h = x.half()                  # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------------- #
    # 3) Fuse the five linear projections into a single matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # --------------------------------------------------------------------- #
    # concatenate weights along the output dimension: shape (5*H, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                        # (5*H, dim)

    # flatten the (B,N,N) spatial dimensions
    flat_input = x_h.view(-1, dim)   # (B*N*N, dim)

    # single linear projection
    fused = F.linear(flat_input, w_concat)     # (B*N*N, 5*H)

    # split back into the five tensors (still views, no copy)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        fused, H, dim=1
    )                               # each (B*N*N, H)

    # reshape to (B, N, N, H)
    left_proj  = left_proj_raw.view(B, N, N, H)
    left_gate  = left_gate_raw.view(B, N, N, H)
    right_proj = right_proj_raw.view(B, N, N, H)
    right_gate = right_gate_raw.view(B, N, N, H)
    out_gate   = out_gate_raw.view(B, N, N, H)

    # --------------------------------------------------------------------- #
    # 4) Sigmoid gates (still in FP16)
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------------------- #
    # 5) Fuse mask (if present) and gate multiplication onto the projected tensors
    # --------------------------------------------------------------------- #
    if nomask:
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # make sure tensors are contiguous for flat indexing
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # mask → FP16 and flatten (B * N * N)
        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1).contiguous()

        total_elems = left_proj.numel()           # B * N * N * H
        BLOCK_SIZE = 1024                         # threads per program
        grid = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elems,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,               # good occupancy on H100
        )
        left, right = left_proj, right_proj

    # --------------------------------------------------------------------- #
    # 6) Heavy O(N³) contraction via strided batched GEMM:
    #    out = left @ rightᵀ  (per hidden channel)
    # --------------------------------------------------------------------- #
    # bring hidden dim to the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_flat  = left_perm.view(batch, N, N)                     # (B*H, N, N)
    right_flat = right_perm.view(batch, N, N).transpose(1, 2)    # (B*H, N, N) transposed

    # batched matrix multiplication (FP16 inputs, FP32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat)                  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                            # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 7) Hidden‑dim layer‑norm, out‑gate, final projection back to dim
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate                         # gating
    out = F.linear(out, weights["to_out.weight"].half())   # (B,N,N,dim)

    # --------------------------------------------------------------------- #
    # 8) Return in float32 (the public API expects FP32)
    # --------------------------------------------------------------------- #
    return out.float()
