"""
TriMul “outgoing’’ forward pass – Triton‑accelerated implementation.

Key ideas
---------
1️⃣  Layer‑Norm the input tensor (B,N,N,D) with a custom Triton kernel.
2️⃣  Cast the normalised tensor to FP16 and perform **one** fused linear
    projection that simultaneously produces the three projections (left,
    right) and the three gating vectors (left_gate, right_gate, out_gate).
3️⃣  Apply the sigmoid to the gating tensors, mask (if present) and compute
    the gated tensors ``left`` and ``right``.
4️⃣  Contract the two 4‑D tensors using a single batched half‑precision
    GEMM (torch.bmm) – the heavy O(N³·H) part.
5️⃣  Layer‑Norm the result with a second Triton kernel (kept in FP16).
6️⃣  Apply the output gate, a final FP16 linear projection back to the model
    dimension and cast the answer to FP32 (the public API expects FP32).

Only the two LayerNorms are written in Triton; all dense linear algebra
leverages the highly‑optimised cuBLAS/Tensor‑Core kernels present in PyTorch.
The implementation works for any batch size (1‑2) and any sequence length
up to 1024, with or without a pair‑wise mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *ptr input  [M, N]
    y_ptr,            # *ptr output [M, N]
    weight_ptr,       # *ptr weight [N]
    bias_ptr,         # *ptr bias   [N]
    M,                # number of rows   (batch * seq * seq)
    N,                # feature dimension (D or H)
    eps,              # small epsilon
    stride_xm, stride_xn,  # strides of x
    stride_ym, stride_yn,  # strides of y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """Two‑pass layer‑norm (mean/var then normalise+affine)."""
    pid = tl.program_id(0)

    # ----- row indices for this program ---------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass: compute mean & variance (FP32) --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    iters_n = tl.cdiv(N, BLOCK_N)
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise + affine --------------------------------
    for k in range(iters_n):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalisation (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine (weight/bias may be FP16 or FP32 → cast to FP32)
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        # store back in original dtype (usually FP16)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for the Triton layer‑norm kernel.
    `x` must be a contiguous 2‑D tensor (M, N). The result has the same dtype
    as ``x`` (typically fp16 after the first norm, fp16 after the second).
    """
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure proper layout for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # modest block sizes that work well for both D=128 and D=384/768
    BLOCK_M = 128
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.
    -------------------------------------------------
    Input  : (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` (model dim),
                         ``hidden_dim`` (projection dim) and optional
                         ``nomask`` (bool).
    Output : torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – keep FP32 for numerical stability, result
    #    in the original dtype (float32) which we immediately cast to fp16.
    # -----------------------------------------------------------------------
    M = bs * seq_len * seq_len                      # rows = B·N·N
    x_flat = input_tensor.reshape(M, dim)          # (M, D) FP32

    norm_w = weights["norm.weight"].to(device, dtype)  # (D)
    norm_b = weights["norm.bias"].to(device, dtype)    # (D)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)     # (B,N,N,D) FP32

    # -----------------------------------------------------------------------
    # 2️⃣ Cast to FP16 (half‑precision) – the dominant compute will run in fp16.
    # -----------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)    # (M, D) fp16

    # -----------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (left/right projections + 3 gates) into
    #    a single GEMM.
    #    Shapes:
    #       left_proj   : (H, D)
    #       left_gate   : (H, D)
    #       right_proj  : (H, D)
    #       right_gate  : (H, D)
    #       out_gate    : (H, D)
    #    After concatenation we obtain a weight of shape (5·H, D).
    # -----------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                 # (5·H, D) fp16

    # single fused linear projection
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_w)   # (M, 5·H) fp16

    # -----------------------------------------------------------------------
    # 4️⃣ Slice the concatenated result and apply sigmoid to the three gate
    #    tensors. Mask (if any) is applied now as a scalar multiplier.
    # -----------------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)  # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_r)    # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    We reshape to (B·H, N, N) and call torch.bmm (FP16 Tensor‑Core).
    # -----------------------------------------------------------------------
    # bring the hidden dimension to the second axis for easier view
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim                                 # B·H
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B·H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B·H, N, N)

    # batched GEMM: (B·H, N, N) * (B·H, N, N)^T  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) fp16

    # -----------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton).  We keep the tensor in fp16 throughout
    #    the second norm to avoid extra conversions.
    # -----------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)  # (H)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)    # (H)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) fp16

    # -----------------------------------------------------------------------
    # 7️⃣ Apply the output gate, final projection and cast back to FP32.
    # -----------------------------------------------------------------------
    gated = out_norm * out_gate                      # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) fp16
    # final linear (fp16) – result is (B,N,N,D) fp16
    out_fp16 = F.linear(gated, to_out_w)
    out_fp32 = out_fp16.to(torch.float32)           # public API expects FP32

    return out_fp32
